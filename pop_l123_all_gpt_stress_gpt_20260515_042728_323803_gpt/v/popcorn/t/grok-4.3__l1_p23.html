import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
__global__ void mean_dim0(const float* __restrict__ input, float* __restrict__ output, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < B * D) {
        int b = idx / D;
        int d = idx % D;
        float sum = 0.0f;
        for(int r = 0; r < R; r++) {
            sum += input[(r * B + b) * D + d];
        }
        output[idx] = sum / static_cast<float>(R);
    }
}

}
