      float2 g = __half22float2(gate);
      float2 u = __half22float2(up);
      g.x = fminf(g.x, limit);
      g.y = fminf(g.y, limit);
      u.x = fmaxf(fminf(u.x, limit), -limit);
      u.y = fmaxf(fminf(u.y, limit), -limit);
      gate = __float22half2_rn(g);
      up = __float22half2_rn(u);
    }
    return __hmul2(PACKED_ACT_FN(gate), up);
  } else {
    packed_t gate = x;
    packed_t up = y;
    if constexpr (HAS_CLAMP) {
      float2 g = __half22float2(gate);
      float2 u = __half22float2(up);
      g.x = fmaxf(fminf(g.x, limit), -limit);

      u.x = fminf(u.x, limit);
      u.y = fminf(u.y, limit);
      gate = __float22half2_rn(g);
      up = __float22half2_rn(u);
    }
    return __hmul2(gate, PACKED_ACT_FN(up));
  }
}
