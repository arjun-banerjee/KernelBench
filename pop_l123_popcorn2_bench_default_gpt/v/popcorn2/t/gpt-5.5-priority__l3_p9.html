import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Bayesian linear regression with a conjugate Gaussian prior.
    Given features X and targets y, computes the posterior distribution
    over weights: p(w | X, y) = N(mu_post, Sigma_post) in closed form
    via the matrix inversion lemma / Cholesky decomposition.
    """

    def __init__(self, input_dim, prior_precision=1.0, noise_precision=1.0):
        super().__init__()
        self.input_dim = input_dim
        self.prior_precision = prior_precision
        self.noise_precision = noise_precision

    def forward(self, X: torch.Tensor, y: torch.Tensor, X_test: torch.Tensor) -> tuple:
        """
        Args:
            X:      (B, N, D) – training features
            y:      (B, N, 1) – training targets
            X_test: (B, M, D) – test features
        Returns:
            (pred_mean, pred_var):
                pred_mean: (B, M, 1)
                pred_var:  (B, M)
        """
        D = X.shape[-1]
        eye = torch.eye(D, device=X.device, dtype=X.dtype).unsqueeze(0)
        Lambda_post = self.prior_precision * eye + self.noise_precision * (X.transpose(-1, -2) @ X)
        L = torch.linalg.cholesky(Lambda_post)
        rhs = self.noise_precision * (X.transpose(-1, -2) @ y)
        mu_post = torch.cholesky_solve(rhs, L)
        pred_mean = X_test @ mu_post
        Sigma_post = torch.cholesky_inverse(L)
        pred_var = (X_test @ Sigma_post * X_test).sum(-1) + 1.0 / self.noise_precision
        return (pred_mean, pred_var)

                           const float* __restrict__ y,
                           const float* __restrict__ Xt,
                           float* __restrict__ out_mean,
                           float* __restrict__ out_var,
                           int Bx, int By, int Bt, int N, int D, int M,
                           float prior, float noise) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    int bx = (Bx == 1 ? 0 : b);
    int by = (By == 1 ? 0 : b);
    int bt = (Bt == 1 ? 0 : b);

    extern __shared__ float sm[];
    float* sX = sm;
    float* sy = sX + (size_t)N * D;
    float* A = sy + N;
    float* L = A + D * D;
    float* Linv = L + D * D;
    float* Sig = Linv + D * D;
    float* rhs = Sig + D * D;
    float* mu = rhs + D;

    const float* Xb = X + (size_t)bx * N * D;
    const float* yb = y + (size_t)by * N;
    const float* Xtb = Xt + (size_t)bt * M * D;

    int XD = N * D;
    for (int k = tid; k < XD; k += blockDim.x) sX[k] = Xb[k];
    for (int k = tid; k < N; k += blockDim.x) sy[k] = yb[k];
    __syncthreads();

    int DD = D * D;
    for (int idx = tid; idx < DD; idx += blockDim.x) {
        int i = (D == 16) ? (idx >> 4) : (idx / D);
        int j = idx - i * D;
        float sum = 0.0f;
        for (int n = 0; n < N; ++n) sum += sX[n * D + i] * sX[n * D + j];
        A[idx] = noise * sum + ((i == j) ? prior : 0.0f);
    }
    for (int i = tid; i < D; i += blockDim.x) {
        float sum = 0.0f;
        for (int n = 0; n < N; ++n) sum += sX[n * D + i] * sy[n];
        rhs[i] = noise * sum;
    }
    __syncthreads();

    if (tid == 0) {
        for (int i = 0; i < DD; ++i) { L[i] = 0.0f; Linv[i] = 0.0f; Sig[i] = 0.0f; }
        for (int i = 0; i < D; ++i) {
            for (int j = 0; j <= i; ++j) {
                float s = A[i * D + j];
                for (int k = 0; k < j; ++k) s -= L[i * D + k] * L[j * D + k];
                L[i * D + j] = (i == j) ? sqrtf(fmaxf(s, 1.0e-20f)) : (s / L[j * D + j]);
            }
        }
        for (int i = 0; i < D; ++i) {
            float s = rhs[i];
            for (int k = 0; k < i; ++k) s -= L[i * D + k] * mu[k];
            mu[i] = s / L[i * D + i];
        }
        for (int i = D - 1; i >= 0; --i) {
            float s = mu[i];
            for (int k = i + 1; k < D; ++k) s -= L[k * D + i] * mu[k];
            mu[i] = s / L[i * D + i];
        }
        for (int i = 0; i < D; ++i) {
            Linv[i * D + i] = 1.0f / L[i * D + i];
            for (int j = 0; j < i; ++j) {
                float s = 0.0f;
                for (int k = j; k < i; ++k) s += L[i * D + k] * Linv[k * D + j];
                Linv[i * D + j] = -s / L[i * D + i];
            }
        }
        for (int i = 0; i < D; ++i) {
            for (int j = 0; j < D; ++j) {
                float s = 0.0f;
                int start = (i > j ? i : j);
                for (int k = start; k < D; ++k) s += Linv[k * D + i] * Linv[k * D + j];
                Sig[i * D + j] = s;
            }
        }
    }
    __syncthreads();

    float inv_noise = 1.0f / noise;
    for (int m = tid; m < M; m += blockDim.x) {
        const float* x = Xtb + (size_t)m * D;
        float mean = 0.0f;
        for (int i = 0; i < D; ++i) mean += x[i] * mu[i];
        float var = 0.0f;
        for (int i = 0; i < D; ++i) {
            float row = 0.0f;
            for (int j = 0; j < D; ++j) row += Sig[i * D + j] * x[j];
            var += x[i] * row;
        }
        out_mean[(size_t)b * M + m] = mean;
        out_var[(size_t)b * M + m] = var + inv_noise;
    }
}

std::vector<torch::Tensor> blr_forward(torch::Tensor X, torch::Tensor y, torch::Tensor Xtest, double prior_precision, double noise_precision) {
    TORCH_CHECK(X.is_cuda() && y.is_cuda() && Xtest.is_cuda(), "inputs must be CUDA tensors");
    TORCH_CHECK(X.scalar_type() == torch::kFloat32 && y.scalar_type() == torch::kFloat32 && Xtest.scalar_type() == torch::kFloat32, "only float32 supported");
    X = X.contiguous();
    y = y.contiguous();
    Xtest = Xtest.contiguous();
    int Bx = (int)X.size(0), N = (int)X.size(1), D = (int)X.size(2);
    int By = (int)y.size(0), Bt = (int)Xtest.size(0), M = (int)Xtest.size(1);
    TORCH_CHECK(y.size(1) == N && y.size(2) == 1 && Xtest.size(2) == D, "bad shapes");
    int Bout = Bx;
    if (By != 1 && Bout == 1) Bout = By;
    if (Bt != 1 && Bout == 1) Bout = Bt;
    TORCH_CHECK((Bx == 1 || Bx == Bout) && (By == 1 || By == Bout) && (Bt == 1 || Bt == Bout), "batch dimensions are not broadcastable");
    c10::cuda::CUDAGuard device_guard(X.device());
    auto mean = torch::empty({Bout, M, 1}, X.options());
    auto var = torch::empty({Bout, M}, X.options());
    int threads = 256;
    size_t shmem = ((size_t)N * D + N + (size_t)4 * D * D + (size_t)2 * D) * sizeof(float);
    cudaFuncSetAttribute(blr_kernel, cudaFuncAttributeMaxDynamicSharedMemorySize, (int)shmem);
    blr_kernel<<<Bout, threads, shmem, at::cuda::getCurrentCUDAStream()>>>(X.data_ptr<float>(), y.data_ptr<float>(), Xtest.data_ptr<float>(), mean.data_ptr<float>(), var.data_ptr<float>(), Bx, By, Bt, N, D, M, (float)prior_precision, (float)noise_precision);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return {mean, var};
}
