import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Pack tokens into expert-major order for MoE execution.

    Each token is assigned to one expert and one slot within that expert's
    buffer. The kernel writes token activations into the packed dispatch layout.
    """

    def __init__(self):
        super().__init__()

    def forward(self, token_hidden: torch.Tensor, expert_idx: torch.Tensor, slot_idx: torch.Tensor, expert_offsets: torch.Tensor) -> torch.Tensor:
        num_rows = int(expert_offsets[-1].item())
        out = torch.zeros(num_rows, token_hidden.shape[1], dtype=token_hidden.dtype, device=token_hidden.device)
        for token in range(token_hidden.shape[0]):
            expert = int(expert_idx[token].item())
            row = int(expert_offsets[expert].item() + slot_idx[token].item())
            out[row] = token_hidden[token]
        return out

    return p.moe_dispatch_inputs(num_tokens, hidden_dim, num_experts, mode=mode)

from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void moe_dispatch_kernel(
    const float* __restrict__ token_hidden,
    const int64_t* __restrict__ expert_idx,
    const int64_t* __restrict__ slot_idx,
    const int64_t* __restrict__ expert_offsets,
    float* __restrict__ out,
    int num_tokens,
    int hidden_dim
) {
    int token = blockIdx.x * blockDim.x + threadIdx.x;
    if (token >= num_tokens) return;
    
    int expert = expert_idx[token];
    int slot = slot_idx[token];
    int64_t base_row = expert_offsets[expert];
    int out_row = static_cast<int>(base_row + slot);
    
    const float* src = token_hidden + token * hidden_dim;
    float* dst = out + out_row * hidden_dim;
    
    for (int i = 0; i < hidden_dim; ++i) {
        dst[i] = src[i];
    }
}

torch::Tensor moe_dispatch_cuda(
    torch::Tensor token_hidden,
    torch::Tensor expert_idx,
    torch::Tensor slot_idx,
    torch::Tensor expert_offsets
) {
    int num_tokens = token_hidden.size(0);
    int hidden_dim = token_hidden.size(1);
    
    int64_t num_rows = expert_offsets[expert_offsets.size(0)-1].item<int64_t>();
    auto out = torch::zeros({num_rows, hidden_dim}, token_hidden.options());
    
    const int threads = 256;
    const int blocks = (num_tokens + threads - 1) / threads;
    
    moe_dispatch_kernel<<<blocks, threads>>>(
        token_hidden.data_ptr<float>(),
        expert_idx.data_ptr<int64_t>(),
        slot_idx.data_ptr<int64_t>(),
        expert_offsets.data_ptr<int64_t>(),
        out.data_ptr<float>(),
        num_tokens,
        hidden_dim
    );
    
    return out;
}
