import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Multi-head sparse aggregation with per-edge attention weights.

    Each edge carries one weight per head, and source node features are arranged
    as `(num_nodes, num_heads, head_dim)`. The output aggregates weighted
    source features into `(num_nodes, num_heads, head_dim)`.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, col_idx: torch.Tensor, edge_weight: torch.Tensor, node_feat: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        num_heads = node_feat.shape[1]
        head_dim = node_feat.shape[2]
        out = torch.zeros(num_nodes, num_heads, head_dim, dtype=node_feat.dtype, device=node_feat.device)
        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end > start:
                src = col_idx[start:end].long()
                weights = edge_weight[start:end].unsqueeze(-1)
                out[dst] = (node_feat[src] * weights).sum(dim=0)
        return out

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
__global__ void csr_multihead_spmm_kernel(
    const int* __restrict__ row_ptr,
    const int* __restrict__ col_idx,
    const float* __restrict__ edge_weight,
    const float* __restrict__ node_feat,
    float* __restrict__ out,
    int num_nodes,
    int num_heads,
    int head_dim
) {
    int dst = blockIdx.x;
    int h = threadIdx.y;  // head index
    int d = threadIdx.x;  // head_dim index
    
    if (dst >= num_nodes || h >= num_heads || d >= head_dim) return;
    
    int start = row_ptr[dst];
    int end = row_ptr[dst + 1];
    
    // Initialize output
    float sum = 0.0f;
    
    // Accumulate weighted source features
    for (int e = start; e < end; e++) {
        int src = col_idx[e];
        float weight = edge_weight[e * num_heads + h];
        sum += weight * node_feat[src * num_heads * head_dim + h * head_dim + d];
    }
    
    out[dst * num_heads * head_dim + h * head_dim + d] = sum;
}

torch::Tensor csr_multihead_spmm_cuda(
    torch::Tensor row_ptr,
    torch::Tensor col_idx,
    torch::Tensor edge_weight,
    torch::Tensor node_feat
) {
    int num_nodes = row_ptr.numel() - 1;
    int num_heads = node_feat.size(1);
    int head_dim = node_feat.size(2);
    
    // Ensure indices are int32
    auto row_ptr_int = row_ptr.to(torch::kInt32);
    auto col_idx_int = col_idx.to(torch::kInt32);
    
    auto out = torch::zeros({num_nodes, num_heads, head_dim}, 
        torch::dtype(node_feat.scalar_type()).device(node_feat.device()));
    
    dim3 grid(num_nodes);
    dim3 block(head_dim, num_heads);  // 32 x 8 = 256 threads
    
    csr_multihead_spmm_kernel<<<grid, block>>>(
        row_ptr_int.data_ptr<int>(),
        col_idx_int.data_ptr<int>(),
        edge_weight.data_ptr<float>(),
        node_feat.data_ptr<float>(),
        out.data_ptr<float>(),
        num_nodes,
        num_heads,
        head_dim
    );
    
    return out;
}
