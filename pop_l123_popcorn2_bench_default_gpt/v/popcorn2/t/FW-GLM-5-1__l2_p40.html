import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Differentiable Smith-Waterman-style local sequence alignment scoring.
    Uses a soft-max relaxation of the DP recurrence so gradients can flow
    through the alignment matrix.  
    """

    def __init__(self, gap_open=-1.0, temperature=1.0):
        super().__init__()
        self.gap_open = gap_open
        self.temperature = temperature

    def forward(self, score_matrix: torch.Tensor) -> torch.Tensor:
        """
        Args:
            score_matrix: (B, M, N) – per-position similarity scores
        Returns:
            alignment_score: (B,) – soft local-alignment score (log-sum-exp over DP cells)
        """
        (B, M, N) = score_matrix.shape
        neg_inf = torch.tensor(float('-inf'), device=score_matrix.device, dtype=score_matrix.dtype)
        H = score_matrix.new_full((B, M + 1, N + 1), 0.0)
        for i in range(1, M + 1):
            for j in range(1, N + 1):
                match = H[:, i - 1, j - 1] + score_matrix[:, i - 1, j - 1]
                delete = H[:, i - 1, j] + self.gap_open
                insert = H[:, i, j - 1] + self.gap_open
                zero = torch.zeros(B, device=score_matrix.device)
                stack = torch.stack([zero, match, delete, insert], dim=-1)
                H[:, i, j] = self.temperature * torch.logsumexp(stack / self.temperature, dim=-1)
        return H.view(B, -1).max(dim=-1).values

__inline__ __device__ float logsumexp4(float a, float b, float c, float d) {
    // Compute log(exp(a) + exp(b) + exp(c) + exp(d)) in a numerically stable way
    float max_val = fmaxf(fmaxf(fmaxf(a, b), c), d);
    if (max_val == -INFINITY) return -INFINITY;
    float sum = expf(a - max_val) + expf(b - max_val) + expf(c - max_val) + expf(d - max_val);
    return max_val + logf(sum);
}

__global__ void smith_waterman_kernel(
    const float* __restrict__ score_matrix,
    float* __restrict__ H,
    float gap_open,
    float temp,
    int B, int M, int N
) {
    // Each thread handles one batch element
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b >= B) return;

    // Pointers for this batch element
    const float* score_b = score_matrix + b * M * N;
    float* H_b = H + b * (M + 1) * (N + 1);

    // Initialize first row and column to 0
    for (int j = 0; j <= N; j++) {
        H_b[j] = 0.0f;
    }
    for (int i = 1; i <= M; i++) {
        H_b[i * (N + 1)] = 0.0f;
    }

    // Process anti-diagonals
    for (int d = 2; d <= M + N; d++) {
        int i_start = max(1, d - N);
        int i_end = min(M, d - 1);
        
        for (int i = i_start; i <= i_end; i++) {
            int j = d - i;
            if (j < 1 || j > N) continue;

            int idx = i * (N + 1) + j;
            int idx_diag = (i - 1) * (N + 1) + (j - 1);
            int idx_up = (i - 1) * (N + 1) + j;
            int idx_left = i * (N + 1) + (j - 1);
            int score_idx = (i - 1) * N + (j - 1);

            float match = H_b[idx_diag] + score_b[score_idx];
            float delete_val = H_b[idx_up] + gap_open;
            float insert_val = H_b[idx_left] + gap_open;

            // H[i,j] = temp * logsumexp(stack / temp)
