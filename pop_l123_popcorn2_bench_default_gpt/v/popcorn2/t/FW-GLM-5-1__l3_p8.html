import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class MSARowSelfAttention(nn.Module):

    def __init__(self, msa_dim, num_heads):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        self.norm = nn.LayerNorm(msa_dim)
        self.q = nn.Linear(msa_dim, msa_dim, bias=False)
        self.k = nn.Linear(msa_dim, msa_dim, bias=False)
        self.v = nn.Linear(msa_dim, msa_dim, bias=False)
        self.out = nn.Linear(msa_dim, msa_dim)

    def forward(self, msa):
        (B, S, N, D) = msa.shape
        (h, d) = (self.num_heads, self.head_dim)
        x = self.norm(msa)
        q = self.q(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        k = self.k(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = self.v(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        attn = q @ k.transpose(-1, -2) / math.sqrt(d)
        attn = F.softmax(attn, dim=-1)
        out = (attn @ v).permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        return self.out(out)

class OuterProductMeanSmall(nn.Module):

    def __init__(self, msa_dim, pair_dim, hidden=32):
        super().__init__()
        self.norm = nn.LayerNorm(msa_dim)
        self.a = nn.Linear(msa_dim, hidden)
        self.b = nn.Linear(msa_dim, hidden)
        self.out = nn.Linear(hidden * hidden, pair_dim)

    def forward(self, msa):
        x = self.norm(msa)
        a = self.a(x)
        b = self.b(x)
        outer = torch.einsum('bsih,bsjk->bijhk', a, b)
        S = msa.shape[1]
        outer = outer / max(S, 1)
        (B, N1, N2, H1, H2) = outer.shape
        return self.out(outer.reshape(B, N1, N2, H1 * H2))

class TriMulOut(nn.Module):

    def __init__(self, pair_dim, hidden):
        super().__init__()
        self.norm = nn.LayerNorm(pair_dim)
        self.pa = nn.Linear(pair_dim, hidden)
        self.pb = nn.Linear(pair_dim, hidden)
        self.ga = nn.Linear(pair_dim, hidden)
        self.gb = nn.Linear(pair_dim, hidden)
        self.norm_out = nn.LayerNorm(hidden)
        self.out = nn.Linear(hidden, pair_dim)
        self.gate_out = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair):
        x = self.norm(pair)
        a = self.pa(x) * torch.sigmoid(self.ga(x))
        b = self.pb(x) * torch.sigmoid(self.gb(x))
        o = torch.einsum('bikh,bjkh->bijh', a, b)
        o = self.out(self.norm_out(o))
        return torch.sigmoid(self.gate_out(pair)) * o

class Model(nn.Module):
    """
    Full Evoformer block from AlphaFold2 combining MSA row self-attention,
    outer-product mean update, and triangular multiplicative update of the
    pair representation.  A single block of the iterative refinement stack.
    """

    def __init__(self, msa_dim, pair_dim, num_heads, tri_hidden):
        super().__init__()
        self.msa_attn = MSARowSelfAttention(msa_dim, num_heads)
        self.opm = OuterProductMeanSmall(msa_dim, pair_dim)
        self.tri_mul = TriMulOut(pair_dim, tri_hidden)

    def forward(self, msa_repr: torch.Tensor, pair_repr: torch.Tensor) -> tuple:
        """
        Args:
            msa_repr:  (B, S, N, msa_dim)
            pair_repr: (B, N, N, pair_dim)
        Returns:
            (updated_msa, updated_pair)
        """
        msa_repr = msa_repr + self.msa_attn(msa_repr)
        pair_repr = pair_repr + self.opm(msa_repr)
        pair_repr = pair_repr + self.tri_mul(pair_repr)
        return (msa_repr, pair_repr)

constexpr int WARP_SIZE = 32;

template <typename scalar_t>
__device__ __forceinline__ scalar_t warp_reduce_sum(scalar_t val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset /= 2) {
        val += __shfl_down_sync(0xffffffff, val, offset);
    }
    return val;
}

template <typename scalar_t>
__device__ __forceinline__ scalar_t warp_reduce_max(scalar_t val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset /= 2) {
        val = max(val, __shfl_down_sync(0xffffffff, val, offset));
    }
    return val;
}

// Fused LayerNorm + QKV projection
template <typename scalar_t>
__global__ void layernorm_qkv_kernel(
    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ ln_weight,
    const scalar_t* __restrict__ ln_bias,
    const scalar_t* __restrict__ q_weight,
    const scalar_t* __restrict__ k_weight,
    const scalar_t* __restrict__ v_weight,
    scalar_t* __restrict__ q_out,
    scalar_t* __restrict__ k_out,
    scalar_t* __restrict__ v_out,
    int N_rows, int D, scalar_t eps) {
    
    int row_idx = blockIdx.x;
    int tid = threadIdx.x;
    
    if (row_idx >= N_rows) return;
    
    const scalar_t* in_row = input + row_idx * D;
    
    __shared__ scalar_t s_mean, s_var;
    
    scalar_t sum = 0.0f;
    scalar_t sum_sq = 0.0f;
    for (int i = tid; i < D; i += blockDim.x) {
        scalar_t val = in_row[i];
        sum += val;
        sum_sq += val * val;
    }
    
    sum = warp_reduce_sum(sum);
    sum_sq = warp_reduce_sum(sum_sq);
    
    __shared__ scalar_t s_partial_mean[32], s_partial_var[32];
    int warp_id = tid / WARP_SIZE;
    int lane_id = tid % WARP_SIZE;
    
    if (lane_id == 0) {
        s_partial_mean[warp_id] = sum;
        s_partial_var[warp_id] = sum_sq;
    }
    __syncthreads();
    
    if (warp_id == 0) {
        int num_warps = (blockDim.x + WARP_SIZE - 1) / WARP_SIZE;
        scalar_t warp_sum = (lane_id < num_warps) ? s_partial_mean[lane_id] : 0.0f;
        scalar_t warp_sum_sq = (lane_id < num_warps) ? s_partial_var[lane_id] : 0.0f;
        warp_sum = warp_reduce_sum(warp_sum);
        warp_sum_sq = warp_reduce_sum(warp_sum_sq);
        
        if (lane_id == 0) {
            s_mean = warp_sum / D;
            s_var = warp_sum_sq / D - s_mean * s_mean;
        }
    }
    __syncthreads();
    
    scalar_t mean = s_mean;
    scalar_t inv_std = rsqrt(s_var + eps);
    
    for (int o = tid; o < D; o += blockDim.x) {
        scalar_t q_acc = 0.0f;
        scalar_t k_acc = 0.0f;
        scalar_t v_acc = 0.0f;
        
        for (int i = 0; i < D; i++) {
            scalar_t norm_val = (in_row[i] - mean) * inv_std * ln_weight[i] + ln_bias[i];
            q_acc += norm_val * q_weight[o * D + i];
            k_acc += norm_val * k_weight[o * D + i];
            v_acc += norm_val * v_weight[o * D + i];
        }
        
        q_out[row_idx * D + o] = q_acc;
        k_out[row_idx * D + o] = k_acc;
        v_out[row_idx * D + o] = v_acc;
    }
}

// MSA Row Attention with fused softmax
template <typename scalar_t>
__global__ void msa_row_attention_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    scalar_t* __restrict__ attn_out,
    int bs_size, int H, int N, int D_head, scalar_t scale) {
    
    int block_idx = blockIdx.x;
    int bs_idx = block_idx / H;
    int h_idx = block_idx % H;
    int tid = threadIdx.x;
    int warp_id = tid / WARP_SIZE;
    int lane_id = tid % WARP_SIZE;
    
    const int D = H * D_head;
    
    const scalar_t* q_ptr = q + (bs_idx * H + h_idx) * N * D_head;
    const scalar_t* k_ptr = k + (bs_idx * H + h_idx) * N * D_head;
    const scalar_t* v_ptr = v + (bs_idx * H + h_idx) * N * D_head;
    
    extern __shared__ char smem[];
    scalar_t* s_attn = reinterpret_cast<scalar_t*>(smem);
    scalar_t* s_out = s_attn + N * N;
    
    for (int n_q = 0; n_q < N; n_q++) {
        for (int n_k = tid; n_k < N; n_k += blockDim.x) {
            scalar_t score = 0.0f;
            for (int d = 0; d < D_head; d++) {
                score += q_ptr[n_q * D_head + d] * k_ptr[n_k * D_head + d];
            }
            s_attn[n_q * N + n_k] = score * scale;
        }
    }
    __syncthreads();
    
    for (int n_q = warp_id; n_q < N; n_q += (blockDim.x / WARP_SIZE)) {
        scalar_t max_val = -1e30f;
        for (int n_k = lane_id; n_k < N; n_k += WARP_SIZE) {
            max_val = max(max_val, s_attn[n_q * N + n_k]);
        }
        max_val = warp_reduce_max(max_val);
        max_val = __shfl_sync(0xffffffff, max_val, 0);
        
        scalar_t sum_exp = 0.0f;
        for (int n_k = lane_id; n_k < N; n_k += WARP_SIZE) {
            sum_exp += expf(s_attn[n_q * N + n_k] - max_val);
        }
        sum_exp = warp_reduce_sum(sum_exp);
        sum_exp = __shfl_sync(0xffffffff, sum_exp, 0);
        
        for (int n_k = lane_id; n_k < N; n_k += WARP_SIZE) {
            s_attn[n_q * N + n_k] = expf(s_attn[n_q * N + n_k] - max_val) / sum_exp;
        }
    }
    __syncthreads();
    
    for (int idx = tid; idx < N * D_head; idx += blockDim.x) {
        int n_q = idx / D_head;
        int d = idx % D_head;
        scalar_t val = 0.0f;
        for (int n_k = 0; n_k < N; n_k++) {
            val += s_attn[n_q * N + n_k] * v_ptr[n_k * D_head + d];
        }
        s_out[idx] = val;
    }
    __syncthreads();
    
    for (int idx = tid; idx < N * D_head; idx += blockDim.x) {
        int n_q = idx / D_head;
        int d = idx % D_head;
        int out_pos = bs_idx * N * D + n_q * D + h_idx * D_head + d;
        attn_out[out_pos] = s_out[idx];
    }
}

// Fused output projection with residual
template <typename scalar_t>
__global__ void output_proj_residual_kernel(
    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ residual,
    const scalar_t* __restrict__ weight,
    const scalar_t* __restrict__ bias,
    scalar_t* __restrict__ output,
    int N_rows, int D) {
    
    int row_idx = blockIdx.x;
    int tid = threadIdx.x;
    
    if (row_idx >= N_rows) return;
    
    const scalar_t* in_row = input + row_idx * D;
    const scalar_t* res_row = residual + row_idx * D;
    scalar_t* out_row = output + row_idx * D;
    
    for (int o = tid; o < D; o += blockDim.x) {
        scalar_t acc = 0.0f;
        for (int i = 0; i < D; i++) {
            acc += in_row[i] * weight[o * D + i];
        }
        out_row[o] = res_row[o] + acc + bias[o];
    }
}

std::vector<torch::Tensor> layernorm_qkv_cuda(
    torch::Tensor input,
    torch::Tensor ln_weight,
    torch::Tensor ln_bias,
    torch::Tensor q_weight,
    torch::Tensor k_weight,
    torch::Tensor v_weight,
    double eps) {
    
    auto N_rows = input.size(0);
    auto D = input.size(1);
    
    auto q_out = torch::empty({N_rows, D}, input.options());
    auto k_out = torch::empty({N_rows, D}, input.options());
    auto v_out = torch::empty({N_rows, D}, input.options());
    
    int threads = 256;
    int blocks = N_rows;
    
    AT_DISPATCH_FLOATING_TYPES(input.scalar_type(), "layernorm_qkv", ([&] {
        layernorm_qkv_kernel<scalar_t><<<blocks, threads>>>(
            input.data_ptr<scalar_t>(),
            ln_weight.data_ptr<scalar_t>(),
            ln_bias.data_ptr<scalar_t>(),
            q_weight.data_ptr<scalar_t>(),
            k_weight.data_ptr<scalar_t>(),
            v_weight.data_ptr<scalar_t>(),
            q_out.data_ptr<scalar_t>(),
            k_out.data_ptr<scalar_t>(),
            v_out.data_ptr<scalar_t>(),
            N_rows, D, static_cast<scalar_t>(eps));
    }));
    
    return {q_out, k_out, v_out};
}

torch::Tensor msa_attention_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    int H,
    double scale) {
    
    auto bs_size = q.size(0);
    auto N = q.size(2);
    auto D_head = q.size(3);
    auto D = H * D_head;
    
    auto attn_out = torch::empty({bs_size, N, D}, q.options());
    
    int threads = 128;
    int blocks = bs_size * H;
    int shared_mem = N * N * sizeof(float) + N * D_head * sizeof(float);
    
    AT_DISPATCH_FLOATING_TYPES(q.scalar_type(), "msa_attention", ([&] {

            q.data_ptr<scalar_t>(),
            k.data_ptr<scalar_t>(),
            v.data_ptr<scalar_t>(),
            attn_out.data_ptr<scalar_t>(),
            bs_size, H, N, D_head, static_cast<scalar_t>(scale));
    }));
    
    return attn_out;
}

class ModelNew(nn.Module):
    def __init__(self, msa_dim, pair_dim, num_heads, tri_hidden):
        super().__init__()
        self.msa_dim = msa_dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        self.tri_hidden = tri_hidden
        
        # MSA Row Self-Attention components
        self.msa_norm = nn.LayerNorm(msa_dim)
        self.msa_q = nn.Linear(msa_dim, msa_dim, bias=False)
        self.msa_k = nn.Linear(msa_dim, msa_dim, bias=False)
        self.msa_v = nn.Linear(msa_dim, msa_dim, bias=False)
        self.msa_out = nn.Linear(msa_dim, msa_dim)
        
        # Outer Product Mean components
        self.opm_norm = nn.LayerNorm(msa_dim)
        self.opm_a = nn.Linear(msa_dim, 32)
        self.opm_b = nn.Linear(msa_dim, 32)
        self.opm_out = nn.Linear(32 * 32, pair_dim)
        
        # Triangular Multiplication components
        self.tri_norm = nn.LayerNorm(pair_dim)
        self.tri_pa = nn.Linear(pair_dim, tri_hidden)
        self.tri_pb = nn.Linear(pair_dim, tri_hidden)
        self.tri_ga = nn.Linear(pair_dim, tri_hidden)
        self.tri_gb = nn.Linear(pair_dim, tri_hidden)
        self.tri_norm_out = nn.LayerNorm(tri_hidden)
        self.tri_out = nn.Linear(tri_hidden, pair_dim)
        self.tri_gate_out = nn.Linear(pair_dim, pair_dim)

    def forward(self, msa_repr: torch.Tensor, pair_repr: torch.Tensor) -> tuple:
        B, S, N, D = msa_repr.shape
        h = self.num_heads
        d = self.head_dim
        
        # MSA Row Self-Attention
        # Reshape for batch processing: (B*S*N, D)
        msa_flat = msa_repr.reshape(B * S * N, D).contiguous()
        
        # Fused LayerNorm + QKV projection
        qkv = evoformer_ops.layernorm_qkv_cuda(
            msa_flat, self.msa_norm.weight, self.msa_norm.bias,
            self.msa_q.weight, self.msa_k.weight, self.msa_v.weight, 1e-5)
        q_flat, k_flat, v_flat = qkv
        
        # Reshape to (B*S, H, N, D_head)
        q = q_flat.view(B, S, N, h, d).permute(0, 1, 3, 2, 4).reshape(B * S, h, N, d).contiguous()
        k = k_flat.view(B, S, N, h, d).permute(0, 1, 3, 2, 4).reshape(B * S, h, N, d).contiguous()
        v = v_flat.view(B, S, N, h, d).permute(0, 1, 3, 2, 4).reshape(B * S, h, N, d).contiguous()
        
        # Custom attention kernel
        attn_out = evoformer_ops.msa_attention_cuda(q, k, v, h, 1.0 / math.sqrt(d))
        
        # Reshape back
        attn_out = attn_out.view(B * S * N, D)
        
        # Fused output projection + residual
        msa_flat = evoformer_ops.output_proj_residual_cuda(
            attn_out, msa_flat, self.msa_out.weight, self.msa_out.bias)
        msa_repr = msa_flat.view(B, S, N, D)
        
        # Outer Product Mean
        x = self.opm_norm(msa_repr)
        a = self.opm_a(x)
        b = self.opm_b(x)
        outer = torch.einsum('bsih,bsjk->bijhk', a, b)
        outer = outer / max(S, 1)
        pair_repr = pair_repr + self.opm_out(outer.reshape(B, N, N, 32 * 32))
        
        # Triangular Multiplication
        x = self.tri_norm(pair_repr)
        a = self.tri_pa(x) * torch.sigmoid(self.tri_ga(x))
        b = self.tri_pb(x) * torch.sigmoid(self.tri_gb(x))
        o = torch.einsum('bikh,bjkh->bijh', a, b)
        o = self.tri_out(self.tri_norm_out(o))
        pair_repr = pair_repr + torch.sigmoid(self.tri_gate_out(pair_repr)) * o
        
        return msa_repr, pair_repr
