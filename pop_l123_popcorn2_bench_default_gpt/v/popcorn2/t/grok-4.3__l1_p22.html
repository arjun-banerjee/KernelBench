import torch
import torch.nn as nn

class Model(nn.Module):
    """
    SE(3)-invariant linear layer.  Maps node features that are augmented
    with 3-D coordinate information through a linear transform that
    respects rotational and translational invariance.  The layer operates
    on (scalar_features, vector_features) pairs where scalar features
    transform trivially and vector features transform as 3-D vectors.
    Used in equivariant GNNs (EGNN, PaiNN, TFN).
    """

    def __init__(self, scalar_in, scalar_out, vector_in, vector_out):
        super().__init__()
        self.scalar_linear = nn.Linear(scalar_in + vector_in, scalar_out)
        self.vector_weight = nn.Parameter(torch.randn(vector_out, vector_in))
        nn.init.xavier_uniform_(self.vector_weight)
        self.vector_gate = nn.Linear(scalar_in + vector_in, vector_out)

    def forward(self, scalar_feat: torch.Tensor, vector_feat: torch.Tensor) -> tuple:
        """
        Args:
            scalar_feat: (B, N, scalar_in) – invariant node features
            vector_feat: (B, N, vector_in, 3) – equivariant vector features
        Returns:
            (scalar_out, vector_out):
                scalar_out: (B, N, scalar_out)
                vector_out: (B, N, vector_out, 3)
        """
        vec_norm = vector_feat.norm(dim=-1)
        combined = torch.cat([scalar_feat, vec_norm], dim=-1)
        s_out = self.scalar_linear(combined)
        v_out = torch.einsum('oi,bnid->bnod', self.vector_weight, vector_feat)
        gate = torch.sigmoid(self.vector_gate(combined)).unsqueeze(-1)
        v_out = gate * v_out
        return (s_out, v_out)

    int b = idx / N;
    int n = idx % N;
    long long offset_s = (long long)b * N * scalar_in + n * scalar_in;
    long long offset_v = (long long)b * N * vector_in * 3 + n * vector_in * 3;

    // Compute vec_norm using registers or shared
    float norms[32]; // assume vector_in <=32
    for(int i=0; i<vector_in; i++) {
        float vx = vector_feat[offset_v + i*3];
        float vy = vector_feat[offset_v + i*3 +1];
        float vz = vector_feat[offset_v + i*3 +2];
        norms[i] = sqrtf(vx*vx + vy*vy + vz*vz);
    }

    // scalar_linear
    long long out_offset_s = (long long)b * N * scalar_out + n * scalar_out;
    long long w_stride = scalar_in + vector_in;
    for(int o=0; o<scalar_out; o++) {
        float val = scalar_bias[o];
        for(int i=0; i<scalar_in; i++) {
            val += scalar_feat[offset_s + i] * scalar_weight[o * w_stride + i];
        }
        for(int i=0; i<vector_in; i++) {
            val += norms[i] * scalar_weight[o * w_stride + scalar_in + i];
        }
        s_out[out_offset_s + o] = val;
    }

    // vector part
    long long out_offset_v = (long long)b * N * vector_out * 3 + n * vector_out * 3;
    for(int o=0; o<vector_out; o++) {
        float gate = vector_gate_bias[o];
        for(int i=0; i<scalar_in; i++) {
            gate += scalar_feat[offset_s + i] * vector_gate_weight[o * w_stride + i];
        }
        for(int i=0; i<vector_in; i++) {
            gate += norms[i] * vector_gate_weight[o * w_stride + scalar_in + i];
        }
        gate = 1.0f / (1.0f + expf(-gate));

        for(int d=0; d<3; d++) {
            float vval = 0;
            for(int i=0; i<vector_in; i++) {
                vval += vector_weight[o * vector_in + i] * vector_feat[offset_v + i*3 + d];
            }
            v_out[out_offset_v + o*3 + d] = gate * vval;
        }
    }
}

std::vector<torch::Tensor> se3_forward(
    torch::Tensor scalar_feat,
    torch::Tensor vector_feat,
    torch::Tensor scalar_weight,
    torch::Tensor scalar_bias,
    torch::Tensor vector_weight,
    torch::Tensor vector_gate_weight,
    torch::Tensor vector_gate_bias,
    int scalar_in, int scalar_out, int vector_in, int vector_out
) {
    int B = scalar_feat.size(0);
    int N = scalar_feat.size(1);
    auto options = scalar_feat.options();
    auto s_out = torch::empty({B, N, scalar_out}, options);
    auto v_out = torch::empty({B, N, vector_out, 3}, options);

    int total = B * N;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;

    se3_kernel<<<blocks, threads>>>(
        scalar_feat.data_ptr<float>(),
        vector_feat.data_ptr<float>(),
        scalar_weight.data_ptr<float>(),
        scalar_bias.data_ptr<float>(),
        vector_weight.data_ptr<float>(),
        vector_gate_weight.data_ptr<float>(),
        vector_gate_bias.data_ptr<float>(),
        s_out.data_ptr<float>(),
        v_out.data_ptr<float>(),
        B, N, scalar_in, scalar_out, vector_in, vector_out
    );

    return {s_out, v_out};
}

class ModelNew(nn.Module):
    def __init__(self, scalar_in, scalar_out, vector_in, vector_out):
        super().__init__()
        self.scalar_linear = nn.Linear(scalar_in + vector_in, scalar_out)
        self.vector_weight = nn.Parameter(torch.randn(vector_out, vector_in))
        nn.init.xavier_uniform_(self.vector_weight)
        self.vector_gate = nn.Linear(scalar_in + vector_in, vector_out)
        self.module = module

    def forward(self, scalar_feat: torch.Tensor, vector_feat: torch.Tensor) -> tuple:
        s_out, v_out = self.module.se3_forward(
            scalar_feat.contiguous(), vector_feat.contiguous(),
            self.scalar_linear.weight, self.scalar_linear.bias,
            self.vector_weight,
            self.vector_gate.weight, self.vector_gate.bias,
            scalar_feat.size(-1), self.scalar_linear.out_features,
            vector_feat.size(-2), self.vector_weight.size(0)
        )
        return (s_out, v_out)
