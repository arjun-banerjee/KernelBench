import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Grounded top-2 MoE router.

    Router logits are biased by token-expert similarity against learned
    grounding embeddings, then normalized over the selected top-2 experts.
    """

    def __init__(self):
        super().__init__()

    def forward(self, token_hidden: torch.Tensor, router_logits: torch.Tensor, expert_ground: torch.Tensor, alpha: float) -> torch.Tensor:
        grounded = router_logits + alpha * (token_hidden @ expert_ground.t())
        (top_vals, top_idx) = torch.topk(grounded, k=2, dim=-1)
        top_weights = torch.softmax(top_vals, dim=-1)
        return torch.stack((top_idx.to(torch.float32), top_weights), dim=-1)
num_tokens = 8192
hidden_dim = 160
num_experts = 16
alpha = 0.35

def get_inputs():
    p = popcorn_pri
    mode = p.sample_input_mode()
    nt = p.trial_dim(num_tokens, 'num_tokens', mode=mode)
    hd = p.trial_dim(hidden_dim, 'hidden_dim', mode=mode, align=16)
    ne = p.trial_dim(num_experts, 'num_experts', mode=mode, minimum=2)
    token_hidden = torch.randn(nt, hd, dtype=torch.float32)
    router_logits = torch.randn(nt, ne, dtype=torch.float32)
    expert_ground = torch.randn(ne, hd, dtype=torch.float32)
    return [token_hidden, router_logits, expert_ground, alpha]

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cfloat>

__global__ void grounded_top2_kernel(
    const float* __restrict__ token_hidden,
    const float* __restrict__ router_logits,
    const float* __restrict__ expert_ground,
    float* __restrict__ output,
    int N, int H, int E, float alpha
) {
    int token = blockIdx.x * blockDim.x + threadIdx.x;
    if (token >= N) return;

    const float* token_vec = token_hidden + token * H;
    
    float scores[16];  // assume E <= 16
    for (int e = 0; e < E; ++e) {
        float dot = 0.0f;
        const float* expert = expert_ground + e * H;
        for (int h = 0; h < H; ++h) {
            dot += token_vec[h] * expert[h];
        }
        scores[e] = router_logits[token * E + e] + alpha * dot;
    }

    // find top-2
    float top_val[2] = {-FLT_MAX, -FLT_MAX};
    int top_idx[2] = {-1, -1};
    for (int e = 0; e < E; ++e) {
        if (scores[e] > top_val[0]) {
            top_val[1] = top_val[0];
            top_idx[1] = top_idx[0];
            top_val[0] = scores[e];
            top_idx[0] = e;
        } else if (scores[e] > top_val[1]) {
            top_val[1] = scores[e];
            top_idx[1] = e;
        }
    }

    // softmax
    float maxv = fmaxf(top_val[0], top_val[1]);
    float sum = expf(top_val[0] - maxv) + expf(top_val[1] - maxv);
    float w0 = expf(top_val[0] - maxv) / sum;
    float w1 = expf(top_val[1] - maxv) / sum;

    // output: [N, 2, 2] -> for each token, [[idx0, w0], [idx1, w1]]
    output[token * 4 + 0] = (float)top_idx[0];
    output[token * 4 + 1] = w0;
    output[token * 4 + 2] = (float)top_idx[1];
    output[token * 4 + 3] = w1;
}

torch::Tensor grounded_top2_cuda(
    torch::Tensor token_hidden,
    torch::Tensor router_logits,
    torch::Tensor expert_ground,
    float alpha
) {
    int N = token_hidden.size(0);
    int H = token_hidden.size(1);
    int E = router_logits.size(1);

    auto options = torch::TensorOptions().dtype(torch::kFloat32).device(token_hidden.device());
    auto output = torch::empty({N, 2, 2}, options);

    int threads = 256;
    int blocks = (N + threads - 1) / threads;

    grounded_top2_kernel<<<blocks, threads>>>(
        token_hidden.data_ptr<float>(),
        router_logits.data_ptr<float>(),
        expert_ground.data_ptr<float>(),
        output.data_ptr<float>(),
        N, H, E, alpha
    );

    return output;
}
