import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <vector>
#include <stdint.h>

static __forceinline__ __device__ int8_t f32_to_i8_rn_sat(float x) {
    uint32_t dst;
    asm volatile("cvt.rni.sat.s8.f32 %0, %1;" : "=r"(dst) : "f"(x));
    return reinterpret_cast<const int8_t&>(dst);
}

static __forceinline__ __device__ float warp_reduce_max(float v) {
    v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 16));
    v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 8));
    v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 4));
    v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 2));
    v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 1));
    return v;
}

__global__ __launch_bounds__(256, 2)
void dyn_int8_quant_half_kernel(const half* __restrict__ input,
                                int8_t* __restrict__ output,
                                float* __restrict__ scales,
                                int hidden) {
    const int row = (int)blockIdx.x;
    const int tid = (int)threadIdx.x;
    const int64_t base = (int64_t)row * hidden;
    const half* __restrict__ in = input + base;
    int8_t* __restrict__ out = output + base;

    float tmax = 0.0f;

    // Vectorized reduction.  For the benchmark shape hidden is 4096 and aligned.
    const int n2 = hidden >> 1;
    const half2* __restrict__ in2 = reinterpret_cast<const half2*>(in);
    for (int i = tid; i < n2; i += blockDim.x) {
        half2 h = in2[i];
        float2 f = __half22float2(h);
        tmax = fmaxf(tmax, fabsf(f.x));
        tmax = fmaxf(tmax, fabsf(f.y));
    }
    if ((hidden & 1) && tid == 0) {
        tmax = fmaxf(tmax, fabsf(__half2float(in[hidden - 1])));
    }

    tmax = warp_reduce_max(tmax);

    __shared__ float smem[8];
    const int lane = tid & 31;
    const int warp = tid >> 5;
    if (lane == 0) smem[warp] = tmax;
    __syncthreads();

    float absmax = 0.0f;
    if (warp == 0) {
        absmax = (lane < (blockDim.x >> 5)) ? smem[lane] : 0.0f;
        absmax = warp_reduce_max(absmax);
        if (lane == 0) {
            scales[row] = absmax * (1.0f / 127.0f);
            smem[0] = (absmax == 0.0f) ? 0.0f : (127.0f / absmax);
        }
    }
    __syncthreads();

    const float inv_s = smem[0];

    // Quantize.  Use CUDA's round-to-nearest-even saturating conversion to match PyTorch/vLLM.
    for (int i = tid; i < hidden; i += blockDim.x) {
        float v = __half2float(in[i]) * inv_s;
        out[i] = f32_to_i8_rn_sat(v);
    }
}

void dyn_int8_quant_float_kernel(const float* __restrict__ input,
                                 int8_t* __restrict__ output,
                                 float* __restrict__ scales,
                                 int hidden) {
    const int row = (int)blockIdx.x;
    const int tid = (int)threadIdx.x;
    const int64_t base = (int64_t)row * hidden;
    const float* __restrict__ in = input + base;
    int8_t* __restrict__ out = output + base;

    float tmax = 0.0f;
    for (int i = tid; i < hidden; i += blockDim.x) {

    }

    if (lane == 0) smem[warp] = tmax;

std::vector<torch::Tensor> dynamic_quant(torch::Tensor input) {
    TORCH_CHECK(input.is_cuda(), "input must be CUDA");
    TORCH_CHECK(input.is_contiguous(), "input must be contiguous");
    TORCH_CHECK(input.dim() >= 1, "input must have at least one dimension");

    const int64_t hidden64 = input.size(input.dim() - 1);
    TORCH_CHECK(hidden64 > 0 && hidden64 <= INT_MAX, "unsupported hidden size");
    const int hidden = (int)hidden64;
    const int64_t rows64 = input.numel() / hidden64;
    TORCH_CHECK(rows64 <= INT_MAX, "too many rows");
    const int rows = (int)rows64;

    auto out = torch::empty(input.sizes(), input.options().dtype(torch::kInt8));
    auto scales = torch::empty({rows64}, input.options().dtype(torch::kFloat32));

    const dim3 grid(rows);
    const dim3 block(256);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    if (input.scalar_type() == torch::kFloat16) {
        dyn_int8_quant_half_kernel<<<grid, block, 0, stream>>>(
            reinterpret_cast<const half*>(input.data_ptr<at::Half>()),
            reinterpret_cast<int8_t*>(out.data_ptr<int8_t>()),
            scales.data_ptr<float>(), hidden);
    } else if (input.scalar_type() == torch::kFloat32) {
        dyn_int8_quant_float_kernel<<<grid, block, 0, stream>>>(
            input.data_ptr<float>(),
            reinterpret_cast<int8_t*>(out.data_ptr<int8_t>()),
            scales.data_ptr<float>(), hidden);
    } else {
        TORCH_CHECK(false, "only float16 and float32 are supported");
    }
    return {out, scales};
}
