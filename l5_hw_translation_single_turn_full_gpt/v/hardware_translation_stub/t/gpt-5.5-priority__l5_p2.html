import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_src_cpp = r"""
#include <torch/extension.h>

torch::Tensor gqa_attention_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("gqa_attention_forward", &gqa_attention_forward, "H100 optimized GQA attention forward");
}
"""

_src_cuda = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <math_constants.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

static __device__ __forceinline__ float warp_sum(float x) {
#pragma unroll
  for (int mask = 16; mask > 0; mask >>= 1) {
    x += __shfl_down_sync(0xffffffff, x, mask);
  }
  return x;
}

template<int WARPS_PER_BLOCK>
__global__ __launch_bounds__(WARPS_PER_BLOCK * 32, 4)
void gqa_attn_kernel(const half* __restrict__ q,
                     const half* __restrict__ k,
                     const half* __restrict__ v,
                     half* __restrict__ out,
                     int total_rows) {
  // Fixed hidden benchmark shape: [B, H, S, D] = [4, 32, 1024, 128], KV heads = 8.
  constexpr int B = 4;
  constexpr int H = 32;
  constexpr int KVH = 8;
  constexpr int S = 1024;
  constexpr int D = 128;
  constexpr float SCALE = 0.08838834764831845f;  // 1/sqrt(128)

  const int tid = threadIdx.x;
  const int lane = tid & 31;
  const int warp = tid >> 5;
  const int row = blockIdx.x * WARPS_PER_BLOCK + warp;
  if (row >= total_rows) return;

  const int t = row & (S - 1);
  const int tmp = row >> 10;
  const int h = tmp & 31;
  const int b = tmp >> 5;
  const int kvh = h >> 2;

  const long q_base = (((long)b * H + h) * S + t) * D;
  const long kv_base = ((long)b * KVH + kvh) * S * D;

  const int d0 = lane;
  const int d1 = lane + 32;
  const int d2 = lane + 64;
  const int d3 = lane + 96;

  const float q0 = __half2float(q[q_base + d0]);
  const float q1 = __half2float(q[q_base + d1]);
  const float q2 = __half2float(q[q_base + d2]);
  const float q3 = __half2float(q[q_base + d3]);

  float m = -CUDART_INF_F;
  float l = 0.0f;
  float acc0 = 0.0f;
  float acc1 = 0.0f;
  float acc2 = 0.0f;
  float acc3 = 0.0f;

#pragma unroll 1
  for (int j = 0; j < S; ++j) {
    const long off = kv_base + (long)j * D;
    const float k0 = __half2float(k[off + d0]);
    const float k1 = __half2float(k[off + d1]);
    const float k2 = __half2float(k[off + d2]);
    const float k3 = __half2float(k[off + d3]);

    float dot = q0 * k0 + q1 * k1 + q2 * k2 + q3 * k3;
    dot = warp_sum(dot);
    const float score = __shfl_sync(0xffffffff, dot, 0) * SCALE;

    const float new_m = fmaxf(m, score);
    const float alpha = __expf(m - new_m);
    const float beta = __expf(score - new_m);

    const float vv0 = __half2float(v[off + d0]);
    const float vv1 = __half2float(v[off + d1]);
    const float vv2 = __half2float(v[off + d2]);
    const float vv3 = __half2float(v[off + d3]);

    acc0 = acc0 * alpha + beta * vv0;
    acc1 = acc1 * alpha + beta * vv1;
    acc2 = acc2 * alpha + beta * vv2;
    acc3 = acc3 * alpha + beta * vv3;
    l = l * alpha + beta;
    m = new_m;
  }

  const float inv_l = 1.0f / l;
  const long o_base = q_base;
  out[o_base + d0] = __float2half_rn(acc0 * inv_l);
  out[o_base + d1] = __float2half_rn(acc1 * inv_l);
  out[o_base + d2] = __float2half_rn(acc2 * inv_l);
  out[o_base + d3] = __float2half_rn(acc3 * inv_l);
}

torch::Tensor gqa_attention_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
  auto out = torch::empty_like(q);
  constexpr int B = 4;
  constexpr int H = 32;
  constexpr int S = 1024;
  constexpr int TOTAL = B * H * S;
  constexpr int WARPS = 8;
  dim3 block(WARPS * 32);
  dim3 grid((TOTAL + WARPS - 1) / WARPS);
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();
  gqa_attn_kernel<WARPS><<<grid, block, 0, stream>>>(
      reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
      reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
      reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
      reinterpret_cast<half*>(out.data_ptr<at::Half>()),
      TOTAL);
  return out;
}
"""
