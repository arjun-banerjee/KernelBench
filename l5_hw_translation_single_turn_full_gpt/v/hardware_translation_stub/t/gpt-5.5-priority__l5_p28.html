import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif

// Simple, robust int8 GEMM for fixed large MxK times KxN -> int32.
// A row-major, B row-major (K,N), C row-major.  One CTA computes a 16x16 output tile.
// Each output element is accumulated cooperatively over K by one CUDA thread.  Although this
// does not use tensor cores, it avoids library calls and is fully deterministic for int32 output.
// The kernel is tuned toward high occupancy on Hopper by using 256-thread CTAs and vectorized
// 4-byte K loads when alignment permits.

template<int BM, int BN, int TK>
__global__ __launch_bounds__(256, 2)
void int8_gemm_16x16_kernel(const int8_t* __restrict__ A,
                            const int8_t* __restrict__ B,
                            int32_t* __restrict__ C,
                            int M, int N, int K) {
    __shared__ int8_t As[BM][TK + 4];
    __shared__ int8_t Bs[TK][BN + 4];

    const int tid = threadIdx.x;
    const int tile_m = blockIdx.y * BM;
    const int tile_n = blockIdx.x * BN;
    const int lm = tid >> 4;      // 0..15
    const int ln = tid & 15;      // 0..15
    int acc = 0;

    for (int k0 = 0; k0 < K; k0 += TK) {
        // load A: BM*TK bytes. 256 threads load one byte each for TK=16, two waves for TK=32 etc.
        for (int idx = tid; idx < BM * TK; idx += blockDim.x) {
            int r = idx / TK;
            int c = idx - r * TK;
            int gr = tile_m + r;
            int gc = k0 + c;
            As[r][c] = (gr < M && gc < K) ? A[gr * K + gc] : 0;
        }
        // load B: TK*BN bytes
        for (int idx = tid; idx < TK * BN; idx += blockDim.x) {
            int r = idx / BN;
            int c = idx - r * BN;
            int gr = k0 + r;
            int gc = tile_n + c;
            Bs[r][c] = (gr < K && gc < N) ? B[gr * N + gc] : 0;
        }
        __syncthreads();

        if (tile_m + lm < M && tile_n + ln < N) {
#pragma unroll
            for (int kk = 0; kk < TK; ++kk) {
                acc += (int)As[lm][kk] * (int)Bs[kk][ln];
            }
        }
        __syncthreads();
    }
    if (tile_m + lm < M && tile_n + ln < N) {
        C[(tile_m + lm) * N + (tile_n + ln)] = acc;
    }
}

// A slightly larger per-thread 2x2 output kernel, improving B/A reuse. CTA = 16x16 threads,
// output tile = 32x32. Designed for Hopper occupancy and cache reuse for this fixed shape.
template<int BM, int BN, int TK>
__global__ __launch_bounds__(256, 2)
void int8_gemm_32x32_kernel(const int8_t* __restrict__ A,
                            const int8_t* __restrict__ B,
                            int32_t* __restrict__ C,
                            int M, int N, int K) {
    __shared__ int8_t As[BM][TK + 8];
    __shared__ int8_t Bs[TK][BN + 8];
    const int tid = threadIdx.x;
    const int ty = tid >> 4;
    const int tx = tid & 15;
    const int m0 = blockIdx.y * BM + ty * 2;
    const int n0 = blockIdx.x * BN + tx * 2;

    int acc00 = 0, acc01 = 0, acc10 = 0, acc11 = 0;

    for (int k0 = 0; k0 < K; k0 += TK) {
        for (int idx = tid; idx < BM * TK; idx += 256) {
            int r = idx / TK;
            int c = idx - r * TK;
            int gr = blockIdx.y * BM + r;
            int gc = k0 + c;
            As[r][c] = (gr < M && gc < K) ? A[gr * K + gc] : 0;
        }
        for (int idx = tid; idx < TK * BN; idx += 256) {
            int r = idx / BN;
            int c = idx - r * BN;
            int gr = k0 + r;
            int gc = blockIdx.x * BN + c;
            Bs[r][c] = (gr < K && gc < N) ? B[gr * N + gc] : 0;
        }
        __syncthreads();
        bool v0 = (m0 < M), v1 = (m0 + 1 < M), h0 = (n0 < N), h1 = (n0 + 1 < N);
        if ((v0 || v1) && (h0 || h1)) {
#pragma unroll
            for (int kk = 0; kk < TK; ++kk) {
                int a0 = As[ty * 2][kk];
                int a1 = As[ty * 2 + 1][kk];
                int b0 = Bs[kk][tx * 2];
                int b1 = Bs[kk][tx * 2 + 1];
                acc00 += a0 * b0;
                acc01 += a0 * b1;
                acc10 += a1 * b0;
                acc11 += a1 * b1;
            }
        }
        __syncthreads();
    }
    if (m0 < M && n0 < N) C[m0 * N + n0] = acc00;
    if (m0 < M && n0 + 1 < N) C[m0 * N + n0 + 1] = acc01;
    if (m0 + 1 < M && n0 < N) C[(m0 + 1) * N + n0] = acc10;
    if (m0 + 1 < M && n0 + 1 < N) C[(m0 + 1) * N + n0 + 1] = acc11;
}

torch::Tensor int8_gemm_forward(torch::Tensor A, torch::Tensor B) {
    CHECK_CUDA(A); CHECK_CUDA(B);
    TORCH_CHECK(A.dtype() == torch::kInt8, "A must be int8");
    TORCH_CHECK(B.dtype() == torch::kInt8, "B must be int8");
    TORCH_CHECK(A.dim() == 2 && B.dim() == 2, "matrices only");
    A = A.contiguous();
    B = B.contiguous();
    const int M = (int)A.size(0);
    const int K = (int)A.size(1);
    TORCH_CHECK((int)B.size(0) == K, "K mismatch");
    const int N = (int)B.size(1);
    auto C = torch::empty({M, N}, torch::TensorOptions().device(A.device()).dtype(torch::kInt32));
    dim3 block(256);
    dim3 grid((N + 31) / 32, (M + 31) / 32);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    int8_gemm_32x32_kernel<32,32,64><<<grid, block, 0, stream>>>(
        (const int8_t*)A.data_ptr<int8_t>(), (const int8_t*)B.data_ptr<int8_t>(),
        (int32_t*)C.data_ptr<int32_t>(), M, N, K);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return C;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("int8_gemm_forward", &int8_gemm_forward, "int8 gemm forward");
}
