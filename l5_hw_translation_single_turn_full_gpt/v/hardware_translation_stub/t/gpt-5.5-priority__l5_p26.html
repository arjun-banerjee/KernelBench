// Output is written directly in Tensor4DPermuteBMM0213 layout:
//   C[b,m,n] -> D[b/D1, m, b%D1, n]
__global__ void bmm_permute_wmma_kernel(
    const half* __restrict__ A,
    const half* __restrict__ W,
    half* __restrict__ D,
    int B, int M, int K, int N, int D1, float alpha) {

  constexpr int WM = 16;
  constexpr int WN = 16;
  constexpr int WK = 16;
  constexpr int WARPS_PER_BLOCK = 4;

  int warp_in_block = threadIdx.x >> 5;
  int lane = threadIdx.x & 31;
  int global_warp = blockIdx.x * WARPS_PER_BLOCK + warp_in_block;

  int mt = (M + WM - 1) / WM;
  int nt = (N + WN - 1) / WN;
  int tiles_per_batch = mt * nt;
  int total_tiles = B * tiles_per_batch;
  if (global_warp >= total_tiles) return;

  int b = global_warp / tiles_per_batch;
  int rem = global_warp - b * tiles_per_batch;
  int tile_m = rem / nt;
  int tile_n = rem - tile_m * nt;

  int row = tile_m * WM;
  int col = tile_n * WN;

  wmma::fragment<wmma::matrix_a, WM, WN, WK, half, wmma::row_major> a_frag;
  wmma::fragment<wmma::matrix_b, WM, WN, WK, half, wmma::row_major> b_frag;
  wmma::fragment<wmma::accumulator, WM, WN, WK, float> c_frag;
  wmma::fill_fragment(c_frag, 0.0f);

  const half* Abase = A + ((size_t)b) * M * K;
  const half* Wbase = W + ((size_t)b) * K * N;

  // Dimensions in the benchmark are multiples of 16. The guarded path below keeps the kernel valid
  // for any future multiple/edge case without affecting the hot path.
  bool full_tile = (row + WM <= M) && (col + WN <= N) && ((K & 15) == 0);

    #pragma unroll
    for (int kk = 0; kk < K; kk += WK) {

      wmma::load_matrix_sync(b_frag, Wbase + kk * N + col, N);
      wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);
