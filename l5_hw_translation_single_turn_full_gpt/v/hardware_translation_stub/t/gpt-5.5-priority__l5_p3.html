import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

__inline__ __device__ float warp_sum(float v) {
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        v += __shfl_down_sync(0xffffffff, v, off);
    }
    return v;
}

__inline__ __device__ float block_sum_256(float v) {
    __shared__ float smem[8];
    int lane = threadIdx.x & 31;
    int wid = threadIdx.x >> 5;
    v = warp_sum(v);
    if (lane == 0) smem[wid] = v;
    __syncthreads();
    v = (threadIdx.x < 8) ? smem[lane] : 0.0f;
    if (wid == 0) v = warp_sum(v);
    return v;
}

// Specialized fast path for the benchmark shape: [512, 4096], fp16, contiguous.
// One CTA per token. Each CTA reads the row once into shared memory, computes the
// fp32 RMS, then normalizes from shared memory to global memory.
__global__ __launch_bounds__(256, 2)
void rmsnorm_h4096_kernel(const half* __restrict__ x,
                          half* __restrict__ y,
                          float eps) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const half* __restrict__ row_x = x + ((size_t)row << 12); // row * 4096
    half* __restrict__ row_y = y + ((size_t)row << 12);

    extern __shared__ half s[]; // 4096 halfs = 8 KiB

    // 16 halfs/thread, vectorized as two 16-byte transactions.
    const int base = tid << 4;
    const uint4 v0 = *reinterpret_cast<const uint4*>(row_x + base);
    const uint4 v1 = *reinterpret_cast<const uint4*>(row_x + base + 8);
    *reinterpret_cast<uint4*>(s + base) = v0;
    *reinterpret_cast<uint4*>(s + base + 8) = v1;

    const half* hp0 = reinterpret_cast<const half*>(&v0);
    const half* hp1 = reinterpret_cast<const half*>(&v1);
    float ss = 0.0f;
    #pragma unroll
    for (int i = 0; i < 8; ++i) {
        float a = __half2float(hp0[i]);
        float b = __half2float(hp1[i]);
        ss = fmaf(a, a, ss);
        ss = fmaf(b, b, ss);
    }

    float sum = block_sum_256(ss);
    __shared__ float inv_rms_s;
    if (tid == 0) inv_rms_s = rsqrtf(sum * (1.0f / 4096.0f) + eps);
    __syncthreads();
    const float inv_rms = inv_rms_s;

    // Convert/multiply in fp32 and round once to half, matching torch RMSNorm tolerance.
    half outv[16];
    #pragma unroll
    for (int i = 0; i < 16; ++i) {
        float a = __half2float(s[base + i]);
        outv[i] = __float2half_rn(a * inv_rms);
    }
    *reinterpret_cast<uint4*>(row_y + base) = *reinterpret_cast<uint4*>(outv);
    *reinterpret_cast<uint4*>(row_y + base + 8) = *reinterpret_cast<uint4*>(outv + 8);
}

// Generic fp16 fallback for other hidden sizes divisible by 8.
__global__ __launch_bounds__(256, 2)
void rmsnorm_generic_kernel(const half* __restrict__ x,
                            half* __restrict__ y,
                            int hidden,
                            float eps) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const half* row_x = x + (size_t)row * hidden;
    half* row_y = y + (size_t)row * hidden;
    float ss = 0.0f;
    for (int i = tid; i < hidden; i += 256) {
        float a = __half2float(row_x[i]);
        ss = fmaf(a, a, ss);
    }
    float sum = block_sum_256(ss);
    __shared__ float inv_rms_s;
    if (tid == 0) inv_rms_s = rsqrtf(sum / (float)hidden + eps);
    __syncthreads();
    float inv_rms = inv_rms_s;
    for (int i = tid; i < hidden; i += 256) {
        row_y[i] = __float2half_rn(__half2float(row_x[i]) * inv_rms);
    }
}

torch::Tensor rmsnorm_forward(torch::Tensor x, double eps) {
    TORCH_CHECK(x.is_cuda(), "x must be CUDA");
    TORCH_CHECK(x.scalar_type() == at::kHalf, "x must be float16");
    TORCH_CHECK(x.is_contiguous(), "x must be contiguous");
    const int64_t hidden64 = x.size(-1);
    TORCH_CHECK(hidden64 > 0 && hidden64 <= INT_MAX, "bad hidden size");
    const int hidden = (int)hidden64;
    const int64_t rows64 = x.numel() / hidden64;
    TORCH_CHECK(rows64 <= INT_MAX, "too many rows");
    const int rows = (int)rows64;
    auto y = torch::empty_like(x);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const half* xp = reinterpret_cast<const half*>(x.data_ptr<at::Half>());
    half* yp = reinterpret_cast<half*>(y.data_ptr<at::Half>());
    if (hidden == 4096) {
        rmsnorm_h4096_kernel<<<rows, 256, 4096 * sizeof(half), stream>>>(xp, yp, (float)eps);
    } else {
        rmsnorm_generic_kernel<<<rows, 256, 0, stream>>>(xp, yp, hidden, (float)eps);
    }
    return y;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("rmsnorm_forward", &rmsnorm_forward, "Hopper optimized fp16 RMSNorm forward");
}
