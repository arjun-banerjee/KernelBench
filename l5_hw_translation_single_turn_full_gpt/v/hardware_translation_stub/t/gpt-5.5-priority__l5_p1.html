import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor gqa_attn_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, bool causal);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gqa_attn_forward", &gqa_attn_forward, "Hopper-optimized GQA attention forward (CUDA)");
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <math_constants.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

// One warp computes one (batch, head, query-position) output vector.
// Specialized for HEAD_SIZE=128 and FP16 inputs/outputs. Four warps per CTA.
__global__ __launch_bounds__(128, 4)
void gqa_attn_warp_kernel(const half* __restrict__ q,
                          const half* __restrict__ k,
                          const half* __restrict__ v,
                          half* __restrict__ out,
                          int B, int H, int KVH, int L,
                          float scale, int causal) {
    const int tid = threadIdx.x;
    const int lane = tid & 31;
    const int warp = tid >> 5;                 // 0..3
    const int qpos = blockIdx.x * 4 + warp;
    const int h = blockIdx.y;
    const int b = blockIdx.z;
    if (qpos >= L) return;

    constexpr int D = 128;
    const int group = H / KVH;
    const int kvh = h / group;

    const long q_base = (((long)b * H + h) * L + qpos) * D;
    const long kv_base0 = ((long)b * KVH + kvh) * L * D;
    const long o_base = q_base;

    // Each lane owns four output channels.
    const int d0 = lane;
    const int d1 = lane + 32;
    const int d2 = lane + 64;
    const int d3 = lane + 96;

    const float q0 = __half2float(q[q_base + d0]);
    const float q1 = __half2float(q[q_base + d1]);
    const float q2 = __half2float(q[q_base + d2]);
    const float q3 = __half2float(q[q_base + d3]);

    float acc0 = 0.0f, acc1 = 0.0f, acc2 = 0.0f, acc3 = 0.0f;
    float lsum = 0.0f;
    float m = -CUDART_INF_F;  // meaningful in lane 0; kept per-lane for simplicity

    const unsigned mask = 0xffffffffu;
    const int kend = causal ? (qpos + 1) : L;

    #pragma unroll 1
    for (int j = 0; j < kend; ++j) {
        const long kb = kv_base0 + (long)j * D;

        float dot = q0 * __half2float(k[kb + d0])
                  + q1 * __half2float(k[kb + d1])
                  + q2 * __half2float(k[kb + d2])
                  + q3 * __half2float(k[kb + d3]);

        // Warp reduction over the 32 lanes.
        dot += __shfl_down_sync(mask, dot, 16);
        dot += __shfl_down_sync(mask, dot, 8);
        dot += __shfl_down_sync(mask, dot, 4);
        dot += __shfl_down_sync(mask, dot, 2);
        dot += __shfl_down_sync(mask, dot, 1);

        float alpha = 0.0f;
        float beta = 0.0f;
        if (lane == 0) {
            const float score = dot * scale;
            const float new_m = fmaxf(m, score);
            alpha = __expf(m - new_m);
            beta = __expf(score - new_m);
            m = new_m;
        }
        alpha = __shfl_sync(mask, alpha, 0);
        beta = __shfl_sync(mask, beta, 0);

        const float vv0 = __half2float(v[kb + d0]);
        const float vv1 = __half2float(v[kb + d1]);
        const float vv2 = __half2float(v[kb + d2]);
        const float vv3 = __half2float(v[kb + d3]);

        acc0 = acc0 * alpha + beta * vv0;
        acc1 = acc1 * alpha + beta * vv1;
        acc2 = acc2 * alpha + beta * vv2;
        acc3 = acc3 * alpha + beta * vv3;
        lsum = lsum * alpha + beta;
    }

    const float inv_l = 1.0f / lsum;
    out[o_base + d0] = __float2half_rn(acc0 * inv_l);
    out[o_base + d1] = __float2half_rn(acc1 * inv_l);
    out[o_base + d2] = __float2half_rn(acc2 * inv_l);
    out[o_base + d3] = __float2half_rn(acc3 * inv_l);
}

torch::Tensor gqa_attn_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, bool causal) {
    TORCH_CHECK(q.is_cuda() && k.is_cuda() && v.is_cuda(), "inputs must be CUDA tensors");
    TORCH_CHECK(q.scalar_type() == at::kHalf && k.scalar_type() == at::kHalf && v.scalar_type() == at::kHalf,
                "only float16 tensors are supported");
    TORCH_CHECK(q.dim() == 4 && k.dim() == 4 && v.dim() == 4, "expected rank-4 tensors");
    TORCH_CHECK(q.is_contiguous() && k.is_contiguous() && v.is_contiguous(), "inputs must be contiguous");

    const int B = (int)q.size(0);
    const int H = (int)q.size(1);
    const int L = (int)q.size(2);
    const int D = (int)q.size(3);
    const int KVH = (int)k.size(1);
    TORCH_CHECK(D == 128, "this kernel is specialized for head_size=128");
    TORCH_CHECK(k.size(0) == B && v.size(0) == B && k.size(2) == L && v.size(2) == L &&
                k.size(3) == D && v.size(3) == D && v.size(1) == KVH, "incompatible tensor shapes");
    TORCH_CHECK(H % KVH == 0, "num_heads must be divisible by num_kv_heads");

    auto out = torch::empty_like(q);
    dim3 block(128);
    dim3 grid((L + 3) / 4, H, B);
    const float scale = 0.08838834764831845f; // 1/sqrt(128)
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    gqa_attn_warp_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()),
        B, H, KVH, L, scale, causal ? 1 : 0);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
"""

    def __init__(self, num_heads: int, num_kv_heads: int, head_size: int):
        super().__init__()
        self.num_heads = int(num_heads)
        self.num_kv_heads = int(num_kv_heads)
        self.head_size = int(head_size)
        # Decoder masked attention corresponding to the source paged-attention kernel.

    def forward(self, q, k, v):
