import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_src_cpp = r"""
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> rotary_forward(torch::Tensor q,
                                          torch::Tensor k,
                                          torch::Tensor cos,
                                          torch::Tensor sin,
                                          int64_t rot_dim);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("rotary_forward", &rotary_forward, "Hopper optimized fp16 rotary embedding");

"""

_src_cuda = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>
#include <vector>

// One CTA handles one token.  The per-token cos/sin rows are tiny (64 fp16 in
// the benchmark), so they are staged once in shared memory and reused by all
// query and key heads.  The arithmetic is performed in fp32 and rounded once to
// fp16, matching PyTorch fp16 tolerances while keeping register pressure low.
__global__ __launch_bounds__(256, 4)
void rotary_fwd_kernel(const __half* __restrict__ q,
                       const __half* __restrict__ k,
                       const __half* __restrict__ cos,
                       const __half* __restrict__ sin,
                       __half* __restrict__ q_out,
                       __half* __restrict__ k_out,
                       int num_tokens,
                       int num_heads,
                       int num_kv_heads,
                       int head_size,
                       int rot_dim) {
  const int token = blockIdx.x;
  const int tid = threadIdx.x;
  const int half_rot = rot_dim >> 1;

  __shared__ __half scos[128];
  __shared__ __half ssin[128];

  for (int d = tid; d < rot_dim; d += blockDim.x) {
    scos[d] = cos[token * rot_dim + d];
    ssin[d] = sin[token * rot_dim + d];
  }
  __syncthreads();

  const int q_pair_tasks = num_heads * half_rot;
  const int k_pair_tasks = num_kv_heads * half_rot;
  const int pair_tasks = q_pair_tasks + k_pair_tasks;

  // Rotary part: NeoX/rotate_half layout over the first rot_dim dimensions.
  for (int t = tid; t < pair_tasks; t += blockDim.x) {
    if (t < q_pair_tasks) {
      const int h = t / half_rot;
      const int off = t - h * half_rot;
      const int base = (token * num_heads + h) * head_size;

      const float x = __half2float(q[base + off]);
      const float y = __half2float(q[base + off + half_rot]);
      const float c0 = __half2float(scos[off]);
      const float s0 = __half2float(ssin[off]);
      const float c1 = __half2float(scos[off + half_rot]);
      const float s1 = __half2float(ssin[off + half_rot]);

      q_out[base + off] = __float2half_rn(fmaf(-y, s0, x * c0));
      q_out[base + off + half_rot] = __float2half_rn(fmaf(x, s1, y * c1));
    } else {
      const int u = t - q_pair_tasks;
      const int h = u / half_rot;
      const int off = u - h * half_rot;
      const int base = (token * num_kv_heads + h) * head_size;

      const float x = __half2float(k[base + off]);
      const float y = __half2float(k[base + off + half_rot]);
      const float c0 = __half2float(scos[off]);
      const float s0 = __half2float(ssin[off]);
      const float c1 = __half2float(scos[off + half_rot]);
      const float s1 = __half2float(ssin[off + half_rot]);

      k_out[base + off] = __float2half_rn(fmaf(-y, s0, x * c0));
      k_out[base + off + half_rot] = __float2half_rn(fmaf(x, s1, y * c1));
    }
  }

  // Pass-through tail [rot_dim, head_size).  This is kept in the same launch to
  // avoid extra kernels/copies.
  const int tail = head_size - rot_dim;
  if (tail > 0) {
    const int q_tail_tasks = num_heads * tail;
    const int k_tail_tasks = num_kv_heads * tail;
    const int total_tail = q_tail_tasks + k_tail_tasks;
    for (int t = tid; t < total_tail; t += blockDim.x) {
      if (t < q_tail_tasks) {
        const int h = t / tail;
        const int r = t - h * tail;
        const int idx = (token * num_heads + h) * head_size + rot_dim + r;
        q_out[idx] = q[idx];
      } else {
        const int u = t - q_tail_tasks;
        const int h = u / tail;
        const int r = u - h * tail;
        const int idx = (token * num_kv_heads + h) * head_size + rot_dim + r;
        k_out[idx] = k[idx];
      }
    }
  }
}

std::vector<torch::Tensor> rotary_forward(torch::Tensor q,
                                          torch::Tensor k,
                                          torch::Tensor cos,
                                          torch::Tensor sin,
                                          int64_t rot_dim64) {
  TORCH_CHECK(q.is_cuda() && k.is_cuda() && cos.is_cuda() && sin.is_cuda(), "all tensors must be CUDA");
  TORCH_CHECK(q.scalar_type() == torch::kFloat16 && k.scalar_type() == torch::kFloat16,
              "q and k must be fp16");
  TORCH_CHECK(cos.scalar_type() == torch::kFloat16 && sin.scalar_type() == torch::kFloat16,
              "cos and sin must be fp16");

  const at::cuda::OptionalCUDAGuard device_guard(device_of(q));
  auto q_out = torch::empty_like(q);
  auto k_out = torch::empty_like(k);

  const int num_tokens = (int)q.size(0);
  const int num_heads = (int)q.size(1);
  const int head_size = (int)q.size(2);
  const int num_kv_heads = (int)k.size(1);
  const int rot_dim = (int)rot_dim64;

  const dim3 grid(num_tokens);
  const dim3 block(256);
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();

  rotary_fwd_kernel<<<grid, block, 0, stream>>>(
      reinterpret_cast<const __half*>(q.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(k.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(cos.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(sin.data_ptr<at::Half>()),
      reinterpret_cast<__half*>(q_out.data_ptr<at::Half>()),
      reinterpret_cast<__half*>(k_out.data_ptr<at::Half>()),
      num_tokens, num_heads, num_kv_heads, head_size, rot_dim);

  C10_CUDA_KERNEL_LAUNCH_CHECK();
  return {q_out, k_out};
}
"""
