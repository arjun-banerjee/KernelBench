#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <mma.h>
#include <cuda_fp16.h>

using namespace nvcuda;

// One CTA computes a 16 x 64 output tile.  Four warps in the CTA compute
// independent 16 x 16 N-slices using tensor cores.  This is specialized for
// the benchmark shape M=16, K=4096, N=4096, but accepts runtime K/N as long as
// they are multiples of 16 and 64 respectively.
__global__ __launch_bounds__(128, 2)
void gemm_m16_n64_wmma_kernel(const half* __restrict__ A,
                              const half* __restrict__ B,
                              half* __restrict__ C,
                              int K, int N) {
    const int warp_id = threadIdx.x >> 5;       // 0..3
    const int lane_id = threadIdx.x & 31;
    const int n0 = blockIdx.x * 64 + warp_id * 16;

    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> acc;
    wmma::fill_fragment(acc, 0.0f);

    #pragma unroll 4
    for (int k0 = 0; k0 < K; k0 += 16) {
        // A is [16,K] row-major, B is [K,N] row-major.
        wmma::load_matrix_sync(a_frag, A + k0, K);
        wmma::load_matrix_sync(b_frag, B + k0 * N + n0, N);
        wmma::mma_sync(acc, a_frag, b_frag, acc);
    }

    wmma::store_matrix_sync(C + n0, acc, N, wmma::mem_row_major);
}

// Variant for a transposed/linear-style weight layout: B is logically [N,K]
// row-major and C = A * B^T.  Kept separate so ModelNew can expose the common
// nn.Linear-compatible parameter layout if a copied state_dict supplies it.
__global__ __launch_bounds__(128, 2)
void gemm_m16_n64_wmma_bt_kernel(const half* __restrict__ A,
                                 const half* __restrict__ B,
                                 half* __restrict__ C,
                                 int K, int N) {
    const int warp_id = threadIdx.x >> 5;
    const int n0 = blockIdx.x * 64 + warp_id * 16;

    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::col_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> acc;
    wmma::fill_fragment(acc, 0.0f);

    #pragma unroll 4
    for (int k0 = 0; k0 < K; k0 += 16) {
        wmma::load_matrix_sync(a_frag, A + k0, K);
        // Tile is B[n0:n0+16, k0:k0+16], interpreted as col_major matrix B^T.
        wmma::load_matrix_sync(b_frag, B + n0 * K + k0, K);
        wmma::mma_sync(acc, a_frag, b_frag, acc);
    }

    wmma::store_matrix_sync(C + n0, acc, N, wmma::mem_row_major);
}

// Fallback scalar half2 kernel for non-canonical sizes.  It is not intended to
// be used by the supplied benchmark, but makes the operator fully functional.
__global__ void gemm_small_fallback_kernel(const half* __restrict__ A,
                                           const half* __restrict__ B,
                                           half* __restrict__ C,
                                           int M, int K, int N,
                                           bool bt) {
    int n = blockIdx.x * blockDim.x + threadIdx.x;
    int m = blockIdx.y;
    if (m >= M || n >= N) return;
    float acc = 0.0f;
    for (int k = 0; k < K; ++k) {
        half bv = bt ? B[n * K + k] : B[k * N + n];
        acc += __half2float(A[m * K + k]) * __half2float(bv);
    }
    C[m * N + n] = __float2half_rn(acc);
}

torch::Tensor gemm_forward(torch::Tensor A, torch::Tensor B, bool bt) {
    TORCH_CHECK(A.is_cuda() && B.is_cuda(), "inputs must be CUDA tensors");
    TORCH_CHECK(A.scalar_type() == at::ScalarType::Half && B.scalar_type() == at::ScalarType::Half,
                "only float16 is supported");
    TORCH_CHECK(A.is_contiguous() && B.is_contiguous(), "inputs must be contiguous");
    const int M = (int)A.size(0);
    const int K = (int)A.size(1);
    const int N = bt ? (int)B.size(0) : (int)B.size(1);
    auto C = torch::empty({M, N}, A.options());

    const half* Ap = reinterpret_cast<const half*>(A.data_ptr<at::Half>());
    const half* Bp = reinterpret_cast<const half*>(B.data_ptr<at::Half>());
    half* Cp = reinterpret_cast<half*>(C.data_ptr<at::Half>());

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if (M == 16 && (K % 16) == 0 && (N % 64) == 0) {
        dim3 grid(N / 64);
        dim3 block(128);
        if (bt) {
            gemm_m16_n64_wmma_bt_kernel<<<grid, block, 0, stream>>>(Ap, Bp, Cp, K, N);
        } else {
            gemm_m16_n64_wmma_kernel<<<grid, block, 0, stream>>>(Ap, Bp, Cp, K, N);
        }
    } else {
        dim3 block(256);
        dim3 grid((N + 255) / 256, M);
        gemm_small_fallback_kernel<<<grid, block, 0, stream>>>(Ap, Bp, Cp, M, K, N, bt);
    }
    return C;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gemm_forward", &gemm_forward, "H100 specialized M=16 fp16 GEMM");
}

        # Marlin-style GEMM uses a logical [K, N] right-hand matrix.  Keeping it
