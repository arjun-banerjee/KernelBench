import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor silu_mul_h100(torch::Tensor input);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("silu_mul_h100", &silu_mul_h100, "H100 optimized SiLU-and-mul (FP16)");
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>
#include <vector>

namespace {

union Half2x4Vec {
    uint4 v;
    __half2 h2[4];
};

__device__ __forceinline__ __half2 silu_h2(__half2 x) {
    const __half2 one = __float2half2_rn(1.0f);
    // sigmoid(x) = 1 / (1 + exp(-x)); h2exp maps to Hopper half2 math and is
    // sufficiently accurate for fp16 while avoiding fp32 scalarization.
    __half2 e = h2exp(__hneg2(x));
    __half2 s = __h2div(one, __hadd2(one, e));
    return __hmul2(x, s);
}

__global__ __launch_bounds__(256, 2)
void silu_mul_h2x4_kernel(__half* __restrict__ out,
                          const __half* __restrict__ input,
                          int d) {
    const int token = blockIdx.x;
    const int tid = threadIdx.x;
    const __half* __restrict__ x_ptr = input + static_cast<long long>(token) * 2LL * d;
    const __half* __restrict__ y_ptr = x_ptr + d;
    __half* __restrict__ o_ptr = out + static_cast<long long>(token) * d;

    const int chunks = d >> 3;  // one uint4 = 8 half = 4 half2
    const uint4* __restrict__ x4 = reinterpret_cast<const uint4*>(x_ptr);
    const uint4* __restrict__ y4 = reinterpret_cast<const uint4*>(y_ptr);
    uint4* __restrict__ o4 = reinterpret_cast<uint4*>(o_ptr);

    #pragma unroll 1
    for (int c = tid; c < chunks; c += blockDim.x) {
        Half2x4Vec gx, uy, r;
        gx.v = x4[c];
        uy.v = y4[c];

        #pragma unroll
        for (int j = 0; j < 4; ++j) {
            r.h2[j] = __hmul2(silu_h2(gx.h2[j]), uy.h2[j]);
        }
        o4[c] = r.v;
    }
}

__global__ __launch_bounds__(256, 2)
void silu_mul_scalar_kernel(__half* __restrict__ out,
                            const __half* __restrict__ input,
                            int d) {
    const int token = blockIdx.x;
    const int tid = threadIdx.x;
    const __half* __restrict__ x_ptr = input + static_cast<long long>(token) * 2LL * d;
    const __half* __restrict__ y_ptr = x_ptr + d;
    __half* __restrict__ o_ptr = out + static_cast<long long>(token) * d;

    for (int i = tid; i < d; i += blockDim.x) {
        float x = __half2float(x_ptr[i]);
        float y = __half2float(y_ptr[i]);
        float s = x / (1.0f + expf(-x));
        o_ptr[i] = __float2half_rn(s * y);
    }
}

} // namespace

torch::Tensor silu_mul_h100(torch::Tensor input) {
    TORCH_CHECK(input.is_cuda(), "input must be a CUDA tensor");
    TORCH_CHECK(input.scalar_type() == at::kHalf, "input must be float16");
    TORCH_CHECK(input.dim() >= 1, "input must have at least one dimension");

    const at::cuda::OptionalCUDAGuard device_guard(device_of(input));

    const int64_t last = input.size(-1);
    const int d = static_cast<int>(last / 2);
    std::vector<int64_t> out_sizes(input.sizes().begin(), input.sizes().end());
    out_sizes.back() = d;
    torch::Tensor out = torch::empty(out_sizes, input.options());

    const int64_t num_tokens = input.numel() / last;
    if (num_tokens == 0) return out;

    const __half* in_ptr = reinterpret_cast<const __half*>(input.data_ptr<at::Half>());
    __half* out_ptr = reinterpret_cast<__half*>(out.data_ptr<at::Half>());

    const dim3 grid(static_cast<unsigned int>(num_tokens));
    const dim3 block(256);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    if ((d & 7) == 0) {
        silu_mul_h2x4_kernel<<<grid, block, 0, stream>>>(out_ptr, in_ptr, d);
    } else {
        silu_mul_scalar_kernel<<<grid, block, 0, stream>>>(out_ptr, in_ptr, d);
    }
    return out;
}
"""
