class Model(nn.Module):
    """
    Edge-only sampled dense-dense matmul.

    For each graph edge (src, dst), compute a dot product between source and
    destination node embeddings. This pattern appears in link prediction,
    sampled attention, and graph contrastive learning.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        src_idx: torch.Tensor,
        dst_idx: torch.Tensor,
        src_feat: torch.Tensor,
        dst_feat: torch.Tensor,
    ) -> torch.Tensor:
        src = src_feat[src_idx.long()]
        dst = dst_feat[dst_idx.long()]
        return (src * dst).sum(dim=-1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Compile the CUDA kernel
cuda_source = """
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <torch/extension.h>

// Edge dot product kernel
// Each thread processes one edge
// Uses float4 for vectorized loads (128 bits)
__global__ void edge_dot_kernel(
    const int* __restrict__ src_idx,
    const int* __restrict__ dst_idx,
    const float* __restrict__ src_feat,
    const float* __restrict__ dst_feat,
    float* __restrict__ output,
    int num_edges,
    int feat_dim
) {
    const int edge_id = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (edge_id >= num_edges) return;
    
    const int src_node = src_idx[edge_id];
    const int dst_node = dst_idx[edge_id];
    
    const float* src_row = src_feat + (long long)src_node * feat_dim;
    const float* dst_row = dst_feat + (long long)dst_node * feat_dim;
    
    float sum = 0.0f;
    
    // Process using float4 for vectorized loads
    const int vec_dim = feat_dim / 4;  // Number of float4 elements
    const float4* src_vec = reinterpret_cast<const float4*>(src_row);
    const float4* dst_vec = reinterpret_cast<const float4*>(dst_row);
    
    #pragma unroll 8
    for (int i = 0; i < vec_dim; i++) {
        const float4 s = src_vec[i];
        const float4 d = dst_vec[i];
        sum += s.x * d.x + s.y * d.y + s.z * d.z + s.w * d.w;
    }
    
    output[edge_id] = sum;
}

torch::Tensor edge_dot_cuda(
    torch::Tensor src_idx,
    torch::Tensor dst_idx,
    torch::Tensor src_feat,
    torch::Tensor dst_feat
) {
    // Ensure indices are contiguous int32
    auto src_idx_contig = src_idx.contiguous().to(torch::kInt32);
    auto dst_idx_contig = dst_idx.contiguous().to(torch::kInt32);
    
    const int num_edges = src_idx_contig.size(0);
    const int feat_dim = src_feat.size(1);
    
    // Ensure features are contiguous float32
    auto src_feat_contig = src_feat.contiguous().to(torch::kFloat32);
    auto dst_feat_contig = dst_feat.contiguous().to(torch::kFloat32);
    
    auto output = torch::empty({num_edges}, src_feat_contig.options());
    
    const int block_size = 256;
    const int num_blocks = (num_edges + block_size - 1) / block_size;
    
    edge_dot_kernel<<<num_blocks, block_size>>>(
        src_idx_contig.data_ptr<int>(),
        dst_idx_contig.data_ptr<int>(),
        src_feat_contig.data_ptr<float>(),
        dst_feat_contig.data_ptr<float>(),
        output.data_ptr<float>(),
        num_edges,
        feat_dim
    );
    
    return output;
}
"""
