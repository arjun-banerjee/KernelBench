Simulated broadcast of row 0 into masked columns for all rows: where mask[j] is True,

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>

__global__ void broadcast_mask_kernel(const float* __restrict__ x, const bool* __restrict__ mask, float* __restrict__ out, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx < total) {
        int j = idx % D;
        if (mask[j]) {
            out[idx] = x[j];
        } else {
            out[idx] = x[idx];
        }
    }
}

torch::Tensor broadcast_mask_cuda(torch::Tensor x, torch::Tensor mask) {
    auto out = torch::empty_like(x);
    int B = x.size(0);
    int D = x.size(1);
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    broadcast_mask_kernel<<<blocks, threads>>>(x.data_ptr<float>(), mask.data_ptr<bool>(), out.data_ptr<float>(), B, D);
    return out;
}
"""

cpp_source = """
torch::Tensor broadcast_mask_cuda(torch::Tensor x, torch::Tensor mask);
