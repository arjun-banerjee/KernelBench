import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Scalar-gated delta-rule linear attention recurrence.

    A recurrent state matrix is decayed by a scalar gate per head, updated by a
    rank-1 key/value outer product, then read out by a query vector.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
        beta: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-1).unsqueeze(-1)
            beta_t = beta[:, t].unsqueeze(-1).unsqueeze(-1)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + beta_t * kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void delta_rule_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    const float* __restrict__ beta,
    float* __restrict__ output,
    int B, int S, int H, int D
) {
    extern __shared__ float state[];
    int bh = blockIdx.x;
    int b = bh / H;
    int h = bh % H;
    if (b >= B) return;

    for (int i = threadIdx.x; i < D*D; i += blockDim.x) {
        state[i] = 0.0f;
    }
    __syncthreads();

    for (int t = 0; t < S; t++) {
        float g = gate[((b * S + t) * H + h)];
        float be = beta[((b * S + t) * H + h)];

        // outer = k[t] outer v[t]
        // state = g * state + be * outer
        for (int i = threadIdx.x; i < D*D; i += blockDim.x) {
            int row = i / D;
            int col = i % D;
            float kval = k[((b * S + t) * H + h) * D + row];
            float vval = v[((b * S + t) * H + h) * D + col];
            float outer = kval * vval;
            state[i] = g * state[i] + be * outer;
        }
        __syncthreads();

        // y = q[t] @ state   (q row vector * matrix)
        for (int col = threadIdx.x; col < D; col += blockDim.x) {
            float sum = 0.0f;
            for (int r = 0; r < D; r++) {
                float qval = q[((b * S + t) * H + h) * D + r];
                sum += qval * state[r * D + col];
            }
            output[((b * S + t) * H + h) * D + col] = sum;
        }
        __syncthreads();
    }
}
