import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_src = r'''
#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

__global__ void qgemm_kernel(const half* __restrict__ x,
                            const signed char* __restrict__ q,
                            const half* __restrict__ s,
                            half* __restrict__ out,
                            int M, int N, int K, int G) {
    int n = blockIdx.x * blockDim.x + threadIdx.x;
    int m = blockIdx.y;
    if (n >= N || m >= M) return;
    float acc = 0.0f;
    const half* xr = x + m * K;
    for (int k = 0; k < K; ++k) {
        float xv = __half2float(xr[k]);
        float wv = (float)q[k * N + n] * __half2float(s[(k / G) * N + n]);
        acc += xv * wv;
    }
    out[m * N + n] = __float2half_rn(acc);
}

torch::Tensor qgemm(torch::Tensor x, torch::Tensor q, torch::Tensor s, int G) {
    int M = (int)x.size(0), K = (int)x.size(1), N = (int)q.size(1);
    auto out = torch::empty({M, N}, x.options());
    dim3 block(256);
    dim3 grid((N + 255) / 256, M);
    qgemm_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>((const half*)x.data_ptr<at::Half>(),
        (const signed char*)q.data_ptr<int8_t>(), (const half*)s.data_ptr<at::Half>(),
        (half*)out.data_ptr<at::Half>(), M, N, K, G);
    return out;
}
'''

class ModelNew(nn.Module):
    def __init__(self, K, N, GROUP_SIZE):
