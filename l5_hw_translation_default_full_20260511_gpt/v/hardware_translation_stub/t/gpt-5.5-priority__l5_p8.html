#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_fp16.h>
#include <mma.h>
using namespace nvcuda;

__global__ void m16_gemm_kernel(const half* __restrict__ x,
                                const half* __restrict__ w,
                                float* __restrict__ acc,
                                int K, int N) {
    int n0 = blockIdx.x * 16;
    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> c;
    wmma::fill_fragment(c, 0.0f);
    #pragma unroll 4
    for (int k0 = 0; k0 < K; k0 += 16) {
        wmma::load_matrix_sync(a, x + k0, K);
        wmma::load_matrix_sync(b, w + k0 * N + n0, N);
        wmma::mma_sync(c, a, b, c);
    }
    wmma::store_matrix_sync(acc + n0, c, N, wmma::mem_row_major);
}

__global__ void cast_kernel(const float* __restrict__ acc, half* __restrict__ y, int total) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < total) y[i] = __float2half_rn(acc[i]);
}

torch::Tensor matmul_forward(torch::Tensor x, torch::Tensor w) {
    int K = (int)x.size(1);
    int N = (int)w.size(1);
