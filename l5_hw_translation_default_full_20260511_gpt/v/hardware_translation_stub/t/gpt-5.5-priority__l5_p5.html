import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp = r'''
#include <torch/extension.h>
void rope(torch::Tensor q, torch::Tensor k, torch::Tensor cos, torch::Tensor sin, int64_t rot_dim);
PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) { m.def("rope", &rope, "rope"); }
'''

_cuda = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ __launch_bounds__(256, 2) void rope_token_kernel(
        half* __restrict__ q,
        half* __restrict__ k,
        const half* __restrict__ cos,
        const half* __restrict__ sin) {
    constexpr int NUM_HEADS = 32;
    constexpr int NUM_KV_HEADS = 8;
    constexpr int HEAD_SIZE = 128;
    constexpr int ROT_DIM = 64;
    constexpr int HALF_DIM = 32;

    const int token = blockIdx.x;
    const int pairlane = threadIdx.x & 15;
    const int subwarp = threadIdx.x >> 4;
    const int r = pairlane << 1;

    const half2 c0 = *reinterpret_cast<const half2*>(cos + token * ROT_DIM + r);
    const half2 s0 = *reinterpret_cast<const half2*>(sin + token * ROT_DIM + r);
    const half2 c1 = *reinterpret_cast<const half2*>(cos + token * ROT_DIM + r + HALF_DIM);
    const half2 s1 = *reinterpret_cast<const half2*>(sin + token * ROT_DIM + r + HALF_DIM);

    #pragma unroll
    for (int h = subwarp; h < NUM_HEADS + NUM_KV_HEADS; h += 16) {
        half* base = (h < NUM_HEADS)
            ? (q + ((token * NUM_HEADS + h) * HEAD_SIZE))
            : (k + ((token * NUM_KV_HEADS + (h - NUM_HEADS)) * HEAD_SIZE));
        const half2 x = *reinterpret_cast<const half2*>(base + r);
        const half2 y = *reinterpret_cast<const half2*>(base + r + HALF_DIM);
        *reinterpret_cast<half2*>(base + r) = __hsub2(__hmul2(x, c0), __hmul2(y, s0));
        *reinterpret_cast<half2*>(base + r + HALF_DIM) = __hadd2(__hmul2(y, c1), __hmul2(x, s1));
    }
}

void rope(torch::Tensor q, torch::Tensor k, torch::Tensor cos, torch::Tensor sin, int64_t rot_dim) {
    rope_token_kernel<<<512, 256, 0, at::cuda::getCurrentCUDAStream()>>>(
        (half*)q.data_ptr<at::Half>(),
        (half*)k.data_ptr<at::Half>(),
        (const half*)cos.data_ptr<at::Half>(),
        (const half*)sin.data_ptr<at::Half>());
}
'''
