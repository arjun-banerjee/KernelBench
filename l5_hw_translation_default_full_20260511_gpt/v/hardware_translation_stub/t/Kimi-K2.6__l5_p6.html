template <int WS>
__launch_bounds__(256, 2)
__global__ void allreduce_kernel(const c10::Half* __restrict__ input,
                                 c10::Half* __restrict__ output,
                                 int64_t numel) {
    const float scale = static_cast<float>(WS);
    int64_t tid = blockIdx.x * blockDim.x + threadIdx.x;
    int64_t stride = gridDim.x * blockDim.x;
    int64_t n8 = numel >> 3;

    const Vec8h2* in8 = reinterpret_cast<const Vec8h2*>(input);
    Vec8h2* out8 = reinterpret_cast<Vec8h2*>(output);

    #pragma unroll 2
    for (int64_t i = tid; i < n8; i += stride) {
        Vec8h2 v = in8[i];

        float2 f0 = __half22float2(v.x);
        float2 f1 = __half22float2(v.y);
        float2 f2 = __half22float2(v.z);
        float2 f3 = __half22float2(v.w);
