import torch.nn as nn
from torch.utils.cpp_extension import load_inline

torch::Tensor rms_norm_forward(torch::Tensor x, torch::Tensor weight, double eps);

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__inline__ __device__ float warp_reduce_sum(float v) {
    v += __shfl_down_sync(0xffffffff, v, 16);
    v += __shfl_down_sync(0xffffffff, v, 8);
    v += __shfl_down_sync(0xffffffff, v, 4);
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

__global__ __launch_bounds__(128, 4)
void rms_norm_fp16_h4096_kernel(const half* __restrict__ x,
                                const half* __restrict__ w,
                                half* __restrict__ y,
                                const float eps) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int lane = tid & 31;
    const int warp = tid >> 5;
    __shared__ float smem[4];

    const half* __restrict__ row_x = x + ((size_t)row << 12);
    half* __restrict__ row_y = y + ((size_t)row << 12);
    const uint4* __restrict__ vx = reinterpret_cast<const uint4*>(row_x);

    float sum = 0.0f;
#pragma unroll
    for (int idx = tid; idx < 512; idx += 128) {
        uint4 a = vx[idx];
        const half2* h = reinterpret_cast<const half2*>(&a);
#pragma unroll
        for (int k = 0; k < 4; ++k) {
            float2 f = __half22float2(h[k]);
            sum = fmaf(f.x, f.x, sum);
            sum = fmaf(f.y, f.y, sum);
        }
    }

    sum = warp_reduce_sum(sum);
    if (lane == 0) smem[warp] = sum;
    __syncthreads();

    float block_sum = (tid < 4) ? smem[tid] : 0.0f;
    if (warp == 0) block_sum = warp_reduce_sum(block_sum);
    __shared__ float scale_s;
    if (tid == 0) scale_s = rsqrtf(block_sum * (1.0f / 4096.0f) + eps);
    __syncthreads();
    const float scale = scale_s;

    uint4* __restrict__ vy = reinterpret_cast<uint4*>(row_y);
#pragma unroll
    for (int idx = tid; idx < 512; idx += 128) {
        uint4 ax = vx[idx];
        half2* hx = reinterpret_cast<half2*>(&ax);
        uint4 outv;
        half2* ho = reinterpret_cast<half2*>(&outv);
#pragma unroll
        for (int k = 0; k < 4; ++k) {
            float2 xf = __half22float2(hx[k]);
            ho[k] = __floats2half2_rn(xf.x * scale, xf.y * scale);
        }
        vy[idx] = outv;
    }
}

torch::Tensor rms_norm_forward(torch::Tensor x, torch::Tensor weight, double eps) {
    auto y = torch::empty_like(x);
    const int64_t tokens64 = x.numel() >> 12;
    const dim3 grid((unsigned int)tokens64);
    const dim3 block(128);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    rms_norm_fp16_h4096_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<const half*>(x.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(weight.data_ptr<at::Half>()),
        reinterpret_cast<half*>(y.data_ptr<at::Half>()),
        (float)eps);
    return y;
}
