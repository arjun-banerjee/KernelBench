#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")

__global__ __launch_bounds__(256, 2)
void silu_mul_vec128_kernel(const half* __restrict__ input,
                            half* __restrict__ output) {
    constexpr int D = 2048;
    const int tok = blockIdx.x;

    const int4* __restrict__ x4 =
        reinterpret_cast<const int4*>(input + tok * (2 * D));
    const int4* __restrict__ y4 =
        reinterpret_cast<const int4*>(input + tok * (2 * D) + D);
    int4* __restrict__ o4 = reinterpret_cast<int4*>(output + tok * D);

    const int idx = threadIdx.x;  // 2048 halves / 8 halves per 128-bit vector

    int4 xv = __ldg(x4 + idx);
    int4 yv = __ldg(y4 + idx);
    int4 ov;

    half2* __restrict__ xp = reinterpret_cast<half2*>(&xv);
    half2* __restrict__ yp = reinterpret_cast<half2*>(&yv);
    half2* __restrict__ op = reinterpret_cast<half2*>(&ov);

    const half2 one = __float2half2_rn(1.0f);
#pragma unroll
    for (int j = 0; j < 4; ++j) {
        half2 g = xp[j];
        half2 den = __hadd2(one, h2exp(__hneg2(g)));
        op[j] = __hmul2(__h2div(g, den), yp[j]);
    }

    o4[idx] = ov;
}

void silu_mul(torch::Tensor input, torch::Tensor output) {
    CHECK_CUDA(input);
    CHECK_CUDA(output);

}
