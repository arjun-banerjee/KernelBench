import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__device__ float softplus(float x) {
    return logf(1 + expf(x));
}

__global__ void softplus_kernel(float* input, float* output, int num_elements) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < num_elements) {
        output[idx] = softplus(input[idx]);
    }
}

void launch_softplus_kernel(float* input, float* output, int num_elements, cudaStream_t stream) {
    int blockSize = 256;
    int numBlocks = (num_elements + blockSize - 1) / blockSize;
    softplus_kernel<<<numBlocks, blockSize, 0, stream>>>(input, output, num_elements);
}
'''

cpp_source = '''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>

void launch_softplus_kernel(float* input, float* output, int num_elements, cudaStream_t stream);

torch::Tensor softplus_cuda(torch::Tensor input) {
    auto output = torch::empty_like(input);
    int num_elements = input.numel();
    launch_softplus_kernel(input.data_ptr<float>(), output.data_ptr<float>(), num_elements, at::cuda::getCurrentCUDAStream());
    return output;
}
'''
