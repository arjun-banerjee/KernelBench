from torch.utils.cpp_extension import load_inline

# CUDA kernel for SELU activation
cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__device__ float selu(float x) {
    const float alpha = 1.6732632423543772848170429916717f;
    const float scale = 1.0507009873554804934193349852946f;
    return x > 0 ? scale * x : scale * alpha * (exp(x) - 1.0f);
}

__global__ void selu_kernel(float* output, const float* input, int num_elements) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < num_elements) {
        output[idx] = selu(input[idx]);
    }
}

torch::Tensor selu_forward(torch::Tensor input) {
    auto output = torch::empty_like(input);
    int num_elements = input.numel();
    int blockSize = 256;
    int numBlocks = (num_elements + blockSize - 1) / blockSize;
    selu_kernel<<<numBlocks, blockSize>>>(output.data_ptr<float>(), input.data_ptr<float>(), num_elements);
    return output;
}
'''
