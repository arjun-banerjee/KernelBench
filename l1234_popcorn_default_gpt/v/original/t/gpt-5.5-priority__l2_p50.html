import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Model that performs a 3D transposed convolution, scaling, average pooling, bias addition, and scaling.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale1, scale2, bias_shape):
        super(Model, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.scale1 = nn.Parameter(torch.tensor(scale1))
        self.avg_pool = nn.AvgPool3d(kernel_size=2)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scale2 = nn.Parameter(torch.tensor(scale2))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x * self.scale1
        x = self.avg_pool(x)
        x = x + self.bias
        x = x * self.scale2
        return x

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

// Precompute, for each input/output channel and each of the 8 input voxels
// touched by a 2x2x2 average-pooled transposed-convolution cell, the sum of
// the corresponding 3x3x3 transposed-convolution weights.
__global__ void combine_weights_kernel(const float* __restrict__ w,
                                       float* __restrict__ cw) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x; // 3*16*8 = 384
    if (idx >= 384) return;

    int m = idx;
    int ox = m & 1; m >>= 1;
    int oy = m & 1; m >>= 1;
    int oz = m & 1; m >>= 1;
    int oc = m & 15; m >>= 4;
    int ic = m;

    float s = 0.0f;
    // For an even transposed-conv output coordinate 2*p, only input p with
    // kernel index 1 contributes.  For odd coordinate 2*p+1, input p uses
    // kernel index 2 and input p+1 uses kernel index 0.  Summing the two
    // pooled coordinates per dimension gives sets {1,2} for offset 0 and
    // {0} for offset 1.
    int z0 = (oz == 0) ? 1 : 0;
    int z1 = (oz == 0) ? 2 : 0;
    int y0 = (oy == 0) ? 1 : 0;
    int y1 = (oy == 0) ? 2 : 0;
    int x0 = (ox == 0) ? 1 : 0;
    int x1 = (ox == 0) ? 2 : 0;

    #pragma unroll
    for (int kz = z0; kz <= z1; ++kz) {
        #pragma unroll
        for (int ky = y0; ky <= y1; ++ky) {
            #pragma unroll
            for (int kx = x0; kx <= x1; ++kx) {
                int wi = (((ic * 16 + oc) * 3 + kz) * 3 + ky) * 3 + kx;
                s += __ldg(w + wi);
            }
        }
    }
    cw[idx] = s;
}

__global__ void fused_kernel(const float* __restrict__ x,
                             const float* __restrict__ cw,
                             const float* __restrict__ conv_bias,
                             const float* __restrict__ scale1,
                             const float* __restrict__ add_bias,
                             const float* __restrict__ scale2,
                             float* __restrict__ out,
                             int total) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total) return;

    int t = idx;
    int ow = t % 31; t /= 31;
    int oh = t % 31; t /= 31;
    int od = t % 15; t /= 15;
    int oc = t & 15; t >>= 4;
    int n = t;

    float acc = 0.0f;

    #pragma unroll
    for (int ic = 0; ic < 3; ++ic) {
        #pragma unroll
        for (int oz = 0; oz < 2; ++oz) {
            int iz = od + oz;
            #pragma unroll
            for (int oy = 0; oy < 2; ++oy) {
                int iy = oh + oy;
                #pragma unroll
                for (int ox = 0; ox < 2; ++ox) {
                    int ix = ow + ox;
                    int xi = (((n * 3 + ic) * 16 + iz) * 32 + iy) * 32 + ix;
                    int ci = (((ic * 16 + oc) * 2 + oz) * 2 + oy) * 2 + ox;
                    acc = fmaf(__ldg(x + xi), __ldg(cw + ci), acc);
                }
            }
        }
    }

    float s1 = __ldg(scale1);
    float s2 = __ldg(scale2);
    float v = acc * (s1 * 0.125f) + __ldg(conv_bias + oc) * s1 + __ldg(add_bias + oc);
    out[idx] = v * s2;
}

void launch_fused_deconv_pool3d(const float* x,
                                const float* w,
                                const float* conv_bias,
                                const float* scale1,
                                const float* add_bias,
                                const float* scale2,
                                float* cw,
                                float* out,
                                int N) {
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    combine_weights_kernel<<<2, 256, 0, stream>>>(w, cw);
    int total = N * 16 * 15 * 31 * 31;
    int block = 256;
    int grid = (total + block - 1) / block;
    fused_kernel<<<grid, block, 0, stream>>>(x, cw, conv_bias, scale1, add_bias, scale2, out, total);
}
'''
