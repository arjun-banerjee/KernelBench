# From https://github.com/karpathy/minGPT/blob/master/mingpt/model.py

class NewGELU(nn.Module):
    """
    Implementation of the GELU activation function currently in Google BERT repo (identical to OpenAI GPT).
    Reference: Gaussian Error Linear Units (GELU) paper: https://arxiv.org/abs/1606.08415
    """
    def __init__(self):
        super(NewGELU, self).__init__()
    
    def forward(self, x):
        return 0.5 * x * (1.0 + torch.tanh(math.sqrt(2.0 / math.pi) * (x + 0.044715 * torch.pow(x, 3.0))))

class Model(nn.Module):
    """
    A multi-head masked self-attention layer with a projection at the end that uses ReLU instead of Softmax.
    It is possible to use torch.nn.MultiheadAttention here but I am including an
    explicit implementation here to show that there is nothing too scary here.
    """

    def __init__(self, n_embd, n_head, max_seqlen):
        super().__init__()
        assert n_embd % n_head == 0
        # key, query, value projections for all heads, but in a batch
        self.c_attn = nn.Linear(n_embd, 3 * n_embd)
        # output projection
        self.c_proj = nn.Linear(n_embd, n_embd)
        # causal mask to ensure that attention is only applied to the left in the input sequence
        self.register_buffer("bias", torch.tril(torch.ones(max_seqlen, max_seqlen))
                                     .view(1, 1, max_seqlen, max_seqlen))
        self.n_head = n_head
        self.n_embd = n_embd

    def forward(self, x):
        B, T, C = x.size() # batch size, sequence length, embedding dimensionality (n_embd)

        # calculate query, key, values for all heads in batch and move head forward to be the batch dim
        q, k ,v  = self.c_attn(x).split(self.n_embd, dim=2)
        k = k.view(B, T, self.n_head, C // self.n_head).transpose(1, 2) # (B, nh, T, hs)
        q = q.view(B, T, self.n_head, C // self.n_head).transpose(1, 2) # (B, nh, T, hs)
        v = v.view(B, T, self.n_head, C // self.n_head).transpose(1, 2) # (B, nh, T, hs)

        # causal self-attention; Self-attend: (B, nh, T, hs) x (B, nh, hs, T) -> (B, nh, T, T)
        att = (q @ k.transpose(-2, -1)) * (1.0 / math.sqrt(k.size(-1)))
        att = att.masked_fill(self.bias[:,:,:T,:T] == 0, float('-inf'))
        att = F.relu(att)

        y = att @ v # (B, nh, T, T) x (B, nh, T, hs) -> (B, nh, T, hs)
        y = y.transpose(1, 2).contiguous().view(B, T, C) # re-assemble all head outputs side by side

        return y

    const scalar_t* __restrict__ x,
    const scalar_t* __restrict__ w_qkv,

    scalar_t* __restrict__ qkv_out,

    // Compute att[b, h, t1, t2]
    scalar_t score = 0;
    for (int k_idx = 0; k_idx < hs; k_idx++) {
        score += q[b * nh * T * hs + h * T * hs + t1 * hs + k_idx] *
                 k[b * nh * T * hs + h * T * hs + t2 * hs + k_idx];
    }
    score *= scale;
    
    // Apply causal mask: only t2 <= t1 is valid
    if (t2 > t1) {
        score = -INFINITY;
    } else {
        // ReLU
        score = max(score, (scalar_t)0);
    }
    
    att[b * nh * T * T + h * T * T + t1 * T + t2] = score;
}

template <typename scalar_t>
__global__ void attention_value_kernel(
    const scalar_t* __restrict__ att,
    const scalar_t* __restrict__ v,
    scalar_t* __restrict__ out,
    int B, int nh, int T, int hs) {
    
    // att: (B, nh, T, T), v: (B, nh, T, hs), out: (B, nh, T, hs)
    int b = blockIdx.x;
    int h = blockIdx.y;
    int t1 = blockIdx.z;
    int hs_idx = threadIdx.x;
    
    if (b >= B || h >= nh || t1 >= T || hs_idx >= hs) return;
    
    scalar_t sum = 0;
    for (int t2 = 0; t2 < T; t2++) {
        scalar_t a = att[b * nh * T * T + h * T * T + t1 * T + t2];
        if (a > 0) {  // ReLU already applied
            sum += a * v[b * nh * T * hs + h * T * hs + t2 * hs + hs_idx];
        }
    }
    
    out[b * nh * T * hs + h * T * hs + t1 * hs + hs_idx] = sum;
}

torch::Tensor fused_attention_forward(
    torch::Tensor x,
    torch::Tensor w_qkv,
    torch::Tensor b_qkv,
    torch::Tensor w_proj,
    torch::Tensor b_proj,
    int n_head) {
    
    int B = x.size(0);
    int T = x.size(1);
    int C = x.size(2);
    int hs = C / n_head;
    
    auto options = x.options();
    
    // Step 1: Fused QKV projection
    auto qkv = torch::empty({B, T, 3 * C}, options);
    
    dim3 grid_qkv(B, T);
    dim3 block_qkv(3 * C);
    if (3 * C > 1024) block_qkv = dim3(1024);
    
    AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "fused_qkv_proj", ([&] {
        fused_qkv_proj_kernel<scalar_t><<<grid_qkv, block_qkv>>>(
            x.data_ptr<scalar_t>(),
            w_qkv.data_ptr<scalar_t>(),
            b_qkv.data_ptr<scalar_t>(),
            qkv.data_ptr<scalar_t>(),
            B, T, C);
    }));
    
    // Split and reshape - using PyTorch for these fast operations
    auto qkv_split = qkv.split(C, 2);
    auto q = qkv_split[0].view({B, T, n_head, hs}).transpose(1, 2).contiguous();
    auto k = qkv_split[1].view({B, T, n_head, hs}).transpose(1, 2).contiguous();
    auto v = qkv_split[2].view({B, T, n_head, hs}).transpose(1, 2).contiguous();
    
    // Step 2: Compute attention scores
    auto att = torch::empty({B, n_head, T, T}, options);
    float scale = 1.0f / std::sqrt(static_cast<float>(hs));
    
    dim3 grid_att(B, n_head, T);
    dim3 block_att(T);
    if (T > 1024) {
        // Need to handle this case differently
        // For now, use multiple kernels
        block_att = dim3(1024);
    }
    
    AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "compute_attention_scores", ([&] {
        compute_attention_scores_kernel<scalar_t><<<grid_att, block_att>>>(
            q.data_ptr<scalar_t>(),
            k.data_ptr<scalar_t>(),
            att.data_ptr<scalar_t>(),
            B, n_head, T, hs, scale);
    }));
    
    // Step 3: Attention @ V
    auto attn_out = torch::empty({B, n_head, T, hs}, options);
    
    dim3 grid_av(B, n_head, T);
    dim3 block_av(hs);
    if (hs > 1024) block_av = dim3(1024);
    
    AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "attention_value", ([&] {
        attention_value_kernel<scalar_t><<<grid_av, block_av>>>(
            att.data_ptr<scalar_t>(),
            v.data_ptr<scalar_t>(),
            attn_out.data_ptr<scalar_t>(),
            B, n_head, T, hs);
    }));
    
    // Reshape - using PyTorch
    auto y = attn_out.transpose(1, 2).contiguous().view({B, T, C});
    
    // Step 4: Output projection
    auto final_out = torch::empty({B, T, C}, options);
    
    dim3 grid_proj(B, T);
    dim3 block_proj(C);
    if (C > 1024) block_proj = dim3(1024);
    
    AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "output_proj", ([&] {
        output_proj_kernel<scalar_t><<<grid_proj, block_proj>>>(
            y.data_ptr<scalar_t>(),
            w_proj.data_ptr<scalar_t>(),
            b_proj.data_ptr<scalar_t>(),
            final_out.data_ptr<scalar_t>(),
            B, T, C);
    }));
    
    return final_out;
}

class ModelNew(nn.Module):
    def __init__(self, n_embd, n_head, max_seqlen):
        super().__init__()
        assert n_embd % n_head == 0
        # key, query, value projections for all heads, but in a batch
        self.c_attn = nn.Linear(n_embd, 3 * n_embd)
        # output projection
        self.c_proj = nn.Linear(n_embd, n_embd)
        self.n_head = n_head
        self.n_embd = n_embd
        self.max_seqlen = max_seqlen
        
    def forward(self, x):
        # Use custom CUDA kernel for the entire forward pass
        return fused_attention.fused_attention_forward(
            x,
            self.c_attn.weight,
            self.c_attn.bias,
            self.c_proj.weight,
            self.c_proj.bias,
