import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__device__ __forceinline__ float gelu(float x) {
    return 0.5f * x * (1.0f + tanhf(0.7978845608f * (x + 0.044715f * x * x)));
}

__global__ void fused_pool_gelu_scale_max_kernel(
    const float* __restrict__ matmul_out,
    float* __restrict__ output,
    int batch_size,
    int in_features,  // 8192
    int pool_kernel_size,  // 16
    float scale_factor
) {
    int b = blockIdx.x;
    if (b >= batch_size) return;

    const float* row = matmul_out + b * in_features;
    float max_val = -INFINITY;
    int num_pools = in_features / pool_kernel_size;  // 512

    for (int p = 0; p < num_pools; ++p) {
        float sum = 0.0f;
        int base = p * pool_kernel_size;
        #pragma unroll
        for (int k = 0; k < pool_kernel_size; ++k) {
            sum += row[base + k];
        }
        float avg = sum / pool_kernel_size;
        float gelu_val = gelu(avg);
        float scaled = gelu_val * scale_factor;
        if (scaled > max_val) max_val = scaled;
    }
    output[b] = max_val;
}
