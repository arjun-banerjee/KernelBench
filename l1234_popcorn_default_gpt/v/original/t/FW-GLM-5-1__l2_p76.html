import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Fused GEMM + Bias + ReLU CUDA kernel
cuda_source = """
#include <cuda.h>
#include <cuda_runtime.h>

#define TILE_SIZE 32

__global__ void gemm_bias_relu_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int M, int K, int N) {
    
    // Shared memory for tiled matrix multiplication
    __shared__ float s_input[TILE_SIZE][TILE_SIZE];
    __shared__ float s_weight[TILE_SIZE][TILE_SIZE];
    
    int row = blockIdx.y * TILE_SIZE + threadIdx.y;
    int col = blockIdx.x * TILE_SIZE + threadIdx.x;
    
    float val = 0.0f;
    
    // Tiled matrix multiplication
    for (int t = 0; t < (K + TILE_SIZE - 1) / TILE_SIZE; ++t) {
        // Load tiles into shared memory with boundary checking
        int input_col = t * TILE_SIZE + threadIdx.x;
        if (row < M && input_col < K) {
            s_input[threadIdx.y][threadIdx.x] = input[row * K + input_col];
        } else {
            s_input[threadIdx.y][threadIdx.x] = 0.0f;
        }
        
        int weight_row = t * TILE_SIZE + threadIdx.y;
        if (weight_row < K && col < N) {
            s_weight[threadIdx.y][threadIdx.x] = weight[weight_row * N + col];
        } else {
            s_weight[threadIdx.y][threadIdx.x] = 0.0f;
        }
        
        __syncthreads();
        
        // Compute partial dot product for this tile
        #pragma unroll
        for (int k = 0; k < TILE_SIZE; ++k) {
            val += s_input[threadIdx.y][k] * s_weight[k][threadIdx.x];
        }
        
        __syncthreads();
    }
    
    // Apply bias and ReLU activation
    if (row < M && col < N) {
        val += bias[col];
        output[row * N + col] = fmaxf(0.0f, val);
    }
}

void gemm_bias_relu_cuda(
    const torch::Tensor& input,
    const torch::Tensor& weight,
    const torch::Tensor& bias,
    torch::Tensor& output,
    int M, int K, int N) {

    dim3 gridDim((N + TILE_SIZE - 1) / TILE_SIZE, (M + TILE_SIZE - 1) / TILE_SIZE);

        input.data_ptr<float>(),

        bias.data_ptr<float>(),
        output.data_ptr<float>(),
        M, K, N
    );
}
