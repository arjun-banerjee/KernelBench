import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Simple model that performs a matrix multiplication, scales the result, and applies batch normalization.
    """
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(Model, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.randn(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.gemm(x)
        x = x * self.scale
        x = self.bn(x)
        return x

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>

void scale_bn_eval_cuda(torch::Tensor z,
                        torch::Tensor scale,
                        torch::Tensor bn_weight,
                        torch::Tensor bn_bias,
                        torch::Tensor running_mean,
                        torch::Tensor running_var,
                        double eps);

torch::Tensor scale_bn_eval(torch::Tensor z,
                            torch::Tensor scale,
                            torch::Tensor bn_weight,
                            torch::Tensor bn_bias,
                            torch::Tensor running_mean,
                            torch::Tensor running_var,
                            double eps) {
    scale_bn_eval_cuda(z, scale, bn_weight, bn_bias, running_mean, running_var, eps);
    return z;
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>

__global__ void scale_bn_eval_kernel(float* __restrict__ z,
                                     const float* __restrict__ scale,
                                     const float* __restrict__ w,
                                     const float* __restrict__ b,
                                     const float* __restrict__ mean,
                                     const float* __restrict__ var,
                                     int N, int C, float eps) {
    // 32 columns x 8 rows per block: coalesced full-warp row accesses and no integer modulo/divide.
    int c = blockIdx.x * 32 + threadIdx.x;
    int r = blockIdx.y * 8 + threadIdx.y;
    if (c < C && r < N) {
        int idx = r * C + c;
        float invstd = rsqrtf(var[c] + eps);
        float v = (z[idx] * scale[c] - mean[c]) * invstd;
        z[idx] = v * w[c] + b[c];
    }
}

void scale_bn_eval_cuda(torch::Tensor z,
                        torch::Tensor scale,
                        torch::Tensor bn_weight,
                        torch::Tensor bn_bias,
                        torch::Tensor running_mean,
                        torch::Tensor running_var,
                        double eps) {
    const at::cuda::OptionalCUDAGuard device_guard(device_of(z));
    int N = (int)z.size(0);
    int C = (int)z.size(1);
    dim3 block(32, 8, 1);
    dim3 grid((C + 31) / 32, (N + 7) / 8, 1);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    scale_bn_eval_kernel<<<grid, block, 0, stream>>>(
        z.data_ptr<float>(),
        scale.data_ptr<float>(),
        bn_weight.data_ptr<float>(),
        bn_bias.data_ptr<float>(),
        running_mean.data_ptr<float>(),
        running_var.data_ptr<float>(),
        N, C, (float)eps);
}
"""

    """
    Drop-in replacement for the given model.  In eval/inference mode it keeps the
    cuBLAS linear layer and fuses the following scale and BatchNorm affine into a
    single in-place CUDA pass over the GEMM output.
    """
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.randn(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        z = F.linear(x, self.gemm.weight, self.gemm.bias)
