import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Model that performs a 3D transposed convolution, LogSumExp, HardSwish, subtraction, clamp operations.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias_shape):
        super(Model, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.bias = nn.Parameter(torch.randn(1, 1, 1, 1)) 

    def forward(self, x):
        x = self.conv_transpose(x)
        x = torch.logsumexp(x, dim=1, keepdim=True)
        x = x * torch.sigmoid(x + 3) / 6
        x = x - self.bias
        x = torch.clamp(x, min=-1, max=1)
        return x

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void fused_logsumexp_hardswish_kernel(
    const float* __restrict__ input,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int B, int C, int D, int H, int W) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D * H * W;
    if (idx >= total) return;

    int b = idx / (D * H * W);
    int spatial = idx % (D * H * W);

    float max_val = -INFINITY;
    float sum_exp = 0.0f;
    for (int c = 0; c < C; ++c) {
        float val = input[b * C * D * H * W + c * D * H * W + spatial];
        if (val > max_val) max_val = val;
    }
    for (int c = 0; c < C; ++c) {
        float val = input[b * C * D * H * W + c * D * H * W + spatial];
        sum_exp += expf(val - max_val);
    }
    float lse = max_val + logf(sum_exp);

    float hs = lse * (1.0f / (1.0f + expf(-(lse + 3.0f)))) / 6.0f;
    float res = hs - bias[0];
    res = fminf(fmaxf(res, -1.0f), 1.0f);
    output[idx] = res;
}
