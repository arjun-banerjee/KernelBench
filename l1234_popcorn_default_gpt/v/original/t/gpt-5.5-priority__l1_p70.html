                        int w_base_ci = (ci * Cout + co0) * 27 + kz * 9 + ky * 3;
                        for (int kx = kx0; kx <= kx1; ++kx) {
                            float xv = __ldg(x + in_base + (ox - kx));
                            int wb = w_base_ci + kx;
                            if (fullc) {
                                s0 = fmaf(xv, __ldg(w + wb + 0*27), s0);
                                s1 = fmaf(xv, __ldg(w + wb + 1*27), s1);
                                s2 = fmaf(xv, __ldg(w + wb + 2*27), s2);
                                s3 = fmaf(xv, __ldg(w + wb + 3*27), s3);
                                s4 = fmaf(xv, __ldg(w + wb + 4*27), s4);
                                s5 = fmaf(xv, __ldg(w + wb + 5*27), s5);
                                s6 = fmaf(xv, __ldg(w + wb + 6*27), s6);
                                s7 = fmaf(xv, __ldg(w + wb + 7*27), s7);
                            } else {
                                if (co0+0 < Cout) s0 = fmaf(xv, __ldg(w + wb + 0*27), s0);
                                if (co0+1 < Cout) s1 = fmaf(xv, __ldg(w + wb + 1*27), s1);
                                if (co0+2 < Cout) s2 = fmaf(xv, __ldg(w + wb + 2*27), s2);
                                if (co0+3 < Cout) s3 = fmaf(xv, __ldg(w + wb + 3*27), s3);
                                if (co0+4 < Cout) s4 = fmaf(xv, __ldg(w + wb + 4*27), s4);
                                if (co0+5 < Cout) s5 = fmaf(xv, __ldg(w + wb + 5*27), s5);
                                if (co0+6 < Cout) s6 = fmaf(xv, __ldg(w + wb + 6*27), s6);
                                if (co0+7 < Cout) s7 = fmaf(xv, __ldg(w + wb + 7*27), s7);
                            }
                        }
                    }
                }
            }
        }

        long long out_base = ((long long)n * Cout * Dout + (long long)co0 * Dout + oz) * Hout * Wout + (long long)oy * Wout + ox;
        long long cstride = (long long)Dout * Hout * Wout;
        if (co0+0 < Cout) y[out_base + 0*cstride] = s0;
        if (co0+1 < Cout) y[out_base + 1*cstride] = s1;
        if (co0+2 < Cout) y[out_base + 2*cstride] = s2;
        if (co0+3 < Cout) y[out_base + 3*cstride] = s3;
        if (co0+4 < Cout) y[out_base + 4*cstride] = s4;
        if (co0+5 < Cout) y[out_base + 5*cstride] = s5;
        if (co0+6 < Cout) y[out_base + 6*cstride] = s6;
        if (co0+7 < Cout) y[out_base + 7*cstride] = s7;
    }
}

__global__ __launch_bounds__(256, 2)
void convt3d_generic_k3_kernel(const float* __restrict__ x,
                               const float* __restrict__ w,
                               const float* __restrict__ b,
                               float* __restrict__ y,
                               int N, int Cin, int Cout,
                               int D, int H, int W,
                               int Dout, int Hout, int Wout,
                               int stride, int padding, int dilation,
                               bool has_bias) {
    long long total = (long long)N * Cout * Dout * Hout * Wout;
    long long tid = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    long long step = (long long)blockDim.x * gridDim.x;
    for (long long idx = tid; idx < total; idx += step) {
        long long t = idx;
        int ox = (int)(t % Wout); t /= Wout;
        int oy = (int)(t % Hout); t /= Hout;
        int oz = (int)(t % Dout); t /= Dout;
        int co = (int)(t % Cout); t /= Cout;
        int n = (int)t;
        float sum = has_bias ? __ldg(b + co) : 0.0f;
        for (int ci = 0; ci < Cin; ++ci) {
            #pragma unroll
            for (int kz = 0; kz < 3; ++kz) {
                int tz = oz + padding - kz * dilation;
                if (tz % stride != 0) continue;
                int iz = tz / stride;
                if ((unsigned)iz >= (unsigned)D) continue;
                #pragma unroll
                for (int ky = 0; ky < 3; ++ky) {
                    int ty = oy + padding - ky * dilation;
                    if (ty % stride != 0) continue;
                    int iy = ty / stride;
                    if ((unsigned)iy >= (unsigned)H) continue;
                    #pragma unroll
                    for (int kx = 0; kx < 3; ++kx) {
                        int tx = ox + padding - kx * dilation;
                        if (tx % stride != 0) continue;
                        int ix = tx / stride;
                        if ((unsigned)ix >= (unsigned)W) continue;
                        float xv = __ldg(x + (((n * Cin + ci) * D + iz) * H + iy) * W + ix);
                        float wv = __ldg(w + (((ci * Cout + co) * 3 + kz) * 3 + ky) * 3 + kx);
                        sum = fmaf(xv, wv, sum);
                    }
                }
            }
        }
        y[idx] = sum;
    }
}
