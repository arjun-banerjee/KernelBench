class Model(nn.Module):

    A model that performs a GEMM, BiasAdd, Hardtanh, Mish, and GroupNorm operations in sequence.
    """

        super(Model, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.hardtanh = nn.Hardtanh()
        self.mish = nn.Mish()
        self.groupnorm = nn.GroupNorm(num_groups=num_groups, num_channels=out_features)

    def forward(self, x):
        """

        x = x + self.bias
        x = self.hardtanh(x)
        x = self.mish(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Define the CUDA kernel source code
cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__device__ float mish(float x) {
    float exp_x = expf(x);
    float exp_2x = expf(2.0f * x);
    return x * (exp_2x + 2.0f * exp_x + 2.0f) / (exp_2x + 2.0f * exp_x + 2.0f + 2.0f * exp_x * exp_x);
}

__global__ void fused_kernel(float *input, float *weight, float *bias, float *output, float *mean, float *var, int batch_size, int in_features, int out_features, int num_groups) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= batch_size * out_features) return;

    int batch_idx = idx / out_features;
    int feature_idx = idx % out_features;
    int group_idx = feature_idx / (out_features / num_groups);

    // GEMM
    float gemm_out = 0.0f;
    for (int i = 0; i < in_features; i++) {
        gemm_out += input[batch_idx * in_features + i] * weight[feature_idx * in_features + i];
    }

    // BiasAdd
    float bias_add_out = gemm_out + bias[feature_idx];

    // HardTanh
    float hardtanh_out = fmaxf(-1.0f, fminf(1.0f, bias_add_out));

    // Mish
    float mish_out = mish(hardtanh_out);

    // Store output for GroupNorm
    output[idx] = mish_out;

    // GroupNorm
    __syncthreads();
    if (threadIdx.x == 0) {
        float sum = 0.0f;
        float sum_sq = 0.0f;
        for (int i = 0; i < out_features / num_groups; i++) {
            int feature_offset = (group_idx * (out_features / num_groups) + i) * batch_size + batch_idx;
            sum += output[feature_offset];
            sum_sq += output[feature_offset] * output[feature_offset];
        }
        mean[group_idx * batch_size + batch_idx] = sum / (out_features / num_groups);
        var[group_idx * batch_size + batch_idx] = sum_sq / (out_features / num_groups) - mean[group_idx * batch_size + batch_idx] * mean[group_idx * batch_size + batch_idx];
    }
    __syncthreads();

    // Normalize
    float group_mean = mean[group_idx * batch_size + batch_idx];
    float group_var = var[group_idx * batch_size + batch_idx];
    output[idx] = (mish_out - group_mean) / sqrtf(group_var + 1e-5f);
}
'''
