class Model(nn.Module):

    Model that performs a transposed convolution, multiplies by a scalar, applies global average pooling,

    """

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Compile the CUDA kernel at module load time
cpp_source = """
torch::Tensor fused_mul_avgpool_cuda(torch::Tensor input, float multiplier);
"""

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void fused_mul_avgpool_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    float multiplier,
    int N, int C, int H, int W
) {
    int nc = blockIdx.x;
    int n = nc / C;
    int c = nc % C;
    
    int tid = threadIdx.x;
    int total_elements = H * W;
    
    float local_sum = 0.0f;
    
    // Strided loop over all elements in this (n, c) slice
    for (int i = tid; i < total_elements; i += blockDim.x) {
        int h = i / W;
        int w = i % W;
        int idx = ((n * C + c) * H + h) * W + w;
        local_sum += input[idx] * multiplier;
    }
    
    // Block reduction using warp shuffle
    __shared__ float sdata[32];
    
    int lane = tid % 32;
    int warp_id = tid / 32;
    int num_warps = blockDim.x / 32;
    
    // Warp reduction using shuffle
    for (int offset = 16; offset > 0; offset >>= 1) {
        local_sum += __shfl_down_sync(0xffffffff, local_sum, offset);
    }
    
    if (lane == 0) {
        sdata[warp_id] = local_sum;
    }
    __syncthreads();
    
    // Final reduction across warps (first warp does it)
    if (warp_id == 0) {
        local_sum = (tid < num_warps) ? sdata[tid] : 0.0f;
        for (int offset = 16; offset > 0; offset >>= 1) {
            local_sum += __shfl_down_sync(0xffffffff, local_sum, offset);
        }
    }
    
    if (tid == 0) {
        output[nc] = local_sum / total_elements;
    }
}

torch::Tensor fused_mul_avgpool_cuda(torch::Tensor input, float multiplier) {
    int N = input.size(0);
    int C = input.size(1);
    int H = input.size(2);
    int W = input.size(3);
    
    auto output = torch::zeros({N, C, 1, 1}, input.options());
    
    int blocks = N * C;
    int threads = 256;
    
    fused_mul_avgpool_kernel<<<blocks, threads>>>(
        input.data_ptr<float>(),
        output.data_ptr<float>(),
        multiplier,
        N, C, H, W
    );
    
    return output;
}
"""
