from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void group_norm_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int N, int C, int HW, int G, int C_per_group,
    float eps
) {
    extern __shared__ float sdata[];
    float* shared_sum = sdata;
    float* shared_sq_sum = sdata + blockDim.x;

    int nc = blockIdx.x; // n * G + g
    int n = nc / G;
    int g = nc % G;
    int c_start = g * C_per_group;
    int c_end = c_start + C_per_group;

    float sum = 0.0f;
    float sq_sum = 0.0f;
    for (int c = c_start + threadIdx.x; c < c_end; c += blockDim.x) {
        for (int hw = 0; hw < HW; hw++) {
            float val = input[((n * C + c) * HW) + hw];
            sum += val;
            sq_sum += val * val;
        }
    }

    shared_sum[threadIdx.x] = sum;
    shared_sq_sum[threadIdx.x] = sq_sum;
    __syncthreads();

    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (threadIdx.x < s) {
            shared_sum[threadIdx.x] += shared_sum[threadIdx.x + s];
            shared_sq_sum[threadIdx.x] += shared_sq_sum[threadIdx.x + s];
        }
        __syncthreads();
    }

    float mean = shared_sum[0] / (C_per_group * HW);
    float var = shared_sq_sum[0] / (C_per_group * HW) - mean * mean;
    float inv_std = rsqrtf(var + eps);

    for (int c = c_start + threadIdx.x; c < c_end; c += blockDim.x) {
        float w = weight[c];
        float b = bias[c];
        for (int hw = 0; hw < HW; hw++) {
            int idx = ((n * C + c) * HW) + hw;
            float val = input[idx];
            float out_val = (val - mean) * inv_std * w + b;
            output[idx] = out_val;
        }
    }
}

torch::Tensor group_norm_cuda(torch::Tensor input, torch::Tensor weight, torch::Tensor bias, int num_groups, float eps) {
    auto N = input.size(0);
    auto C = input.size(1);
    auto HW = input.size(2) * input.size(3);
    auto G = num_groups;
    auto C_per_group = C / G;

    auto output = torch::empty_like(input);

    int threads = 256;
    int blocks = N * G;
    size_t shared_mem = 2 * threads * sizeof(float);

    group_norm_kernel<<<blocks, threads, shared_mem>>>(
        input.data_ptr<float>(),
        weight.data_ptr<float>(),
        bias.data_ptr<float>(),
        output.data_ptr<float>(),
        N, C, HW, G, C_per_group,
        eps
    );

    return output;
}
