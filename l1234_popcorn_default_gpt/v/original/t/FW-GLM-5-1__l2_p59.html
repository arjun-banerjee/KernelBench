import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Simple model that performs a matrix multiplication, applies Swish activation, and scales the result.
    """
    def __init__(self, in_features, out_features, scaling_factor):
        super(Model, self).__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.matmul(x)
        x = x * torch.sigmoid(x)  # Swish activation
        x = x * self.scaling_factor
        return x

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <algorithm>

#define TILE_SIZE 32
#define K_TILE_SIZE 128

__global__ void gemm_swish_scale_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int M, int K, int N,
    float scaling_factor
) {
    const int row = blockIdx.y * TILE_SIZE + threadIdx.y;
    const int col = blockIdx.x * TILE_SIZE + threadIdx.x;
    
    __shared__ float s_input[TILE_SIZE][K_TILE_SIZE + 1];
    __shared__ float s_weight[TILE_SIZE][K_TILE_SIZE + 1];
    
    float sum = 0.0f;
    
    const int num_k_tiles = (K + K_TILE_SIZE - 1) / K_TILE_SIZE;
    const int linear_tid = threadIdx.y * TILE_SIZE + threadIdx.x;
    
    for (int kt = 0; kt < num_k_tiles; kt++) {
        const int k_base = kt * K_TILE_SIZE;
        const int k_size = min(K_TILE_SIZE, K - k_base);
        
        // Cooperative loading of input tile
        #pragma unroll
        for (int i = 0; i < 4; i++) {
            const int idx = linear_tid + i * (TILE_SIZE * TILE_SIZE);
            const int local_row = idx / K_TILE_SIZE;
            const int local_k = idx % K_TILE_SIZE;
            const int global_row = blockIdx.y * TILE_SIZE + local_row;
            const int global_k = k_base + local_k;
            
            if (global_row < M && global_k < K) {
                s_input[local_row][local_k] = input[global_row * K + global_k];
            } else {
                s_input[local_row][local_k] = 0.0f;
            }
        }
        
        // Cooperative loading of weight tile
        #pragma unroll
        for (int i = 0; i < 4; i++) {
            const int idx = linear_tid + i * (TILE_SIZE * TILE_SIZE);
            const int local_col = idx / K_TILE_SIZE;
            const int local_k = idx % K_TILE_SIZE;
            const int global_col = blockIdx.x * TILE_SIZE + local_col;
            const int global_k = k_base + local_k;
            
            if (global_col < N && global_k < K) {
                s_weight[local_col][local_k] = weight[global_col * K + global_k];
            } else {
                s_weight[local_col][local_k] = 0.0f;
            }
        }
        
        __syncthreads();
        
        // Compute partial dot product
        for (int k = 0; k < k_size; k++) {
            sum += s_input[threadIdx.y][k] * s_weight[threadIdx.x][k];
        }
        
        __syncthreads();
    }
    
    // Apply bias, swish, and scaling
    if (row < M && col < N) {
        sum += bias[col];
        const float sig = __frcp_rn(1.0f + __expf(-sum));
        output[row * N + col] = sum * sig * scaling_factor;
    }
}

torch::Tensor gemm_swish_scale_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias,
    double scaling_factor
) {
    TORCH_CHECK(input.is_contiguous(), "Input must be contiguous");
    TORCH_CHECK(weight.is_contiguous(), "Weight must be contiguous");
    TORCH_CHECK(bias.is_contiguous(), "Bias must be contiguous");
    TORCH_CHECK(input.scalar_type() == torch::kFloat32, "Input must be float32");
    TORCH_CHECK(weight.scalar_type() == torch::kFloat32, "Weight must be float32");
    TORCH_CHECK(bias.scalar_type() == torch::kFloat32, "Bias must be float32");
    
    const int M = input.size(0);
    const int K = input.size(1);
    const int N = weight.size(0);
    
    auto output = torch::empty({M, N}, input.options());
    
    const dim3 blockDim(TILE_SIZE, TILE_SIZE);
    const dim3 gridDim((N + TILE_SIZE - 1) / TILE_SIZE, (M + TILE_SIZE - 1) / TILE_SIZE);
    
    gemm_swish_scale_kernel<<<gridDim, blockDim>>>(
        input.data_ptr<float>(),
        weight.data_ptr<float>(),
        bias.data_ptr<float>(),
        output.data_ptr<float>(),
        M, K, N,
        static_cast<float>(scaling_factor)
    );
    
    return output;
}
