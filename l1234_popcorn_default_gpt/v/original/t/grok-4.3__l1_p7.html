import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

#define TILE_M 64
#define TILE_N 64
#define TILE_K 64
#define THREADS_PER_BLOCK 256

__global__ void matmul_kernel(const float* __restrict__ A, const float* __restrict__ B, float* __restrict__ C, int M, int K, int N) {
    __shared__ float As[TILE_M][TILE_K];
    __shared__ float Bs[TILE_K][TILE_N];

    int bx = blockIdx.x;
    int by = blockIdx.y;
    int tx = threadIdx.x;

    int row = by * TILE_M + (tx / (TILE_N / 4)) * 4;  // each thread handles 4 rows partially
    int col = bx * TILE_N + (tx % (TILE_N / 4)) * 4;

    float sum[4][4] = {0.0f};

    for (int t = 0; t < (K + TILE_K - 1) / TILE_K; t++) {
        // Load A tile
        for (int i = tx; i < TILE_M * TILE_K; i += THREADS_PER_BLOCK) {
            int r = i / TILE_K;
            int c = i % TILE_K;
            int global_r = by * TILE_M + r;
            int global_c = t * TILE_K + c;
            As[r][c] = (global_r < M && global_c < K) ? A[global_r * K + global_c] : 0.0f;
        }

        // Load B tile
        for (int i = tx; i < TILE_K * TILE_N; i += THREADS_PER_BLOCK) {
            int r = i / TILE_N;
            int c = i % TILE_N;
            int global_r = t * TILE_K + r;
            int global_c = bx * TILE_N + c;
            Bs[r][c] = (global_r < K && global_c < N) ? B[global_r * N + global_c] : 0.0f;
        }

        __syncthreads();

        // Compute
        #pragma unroll
        for (int k = 0; k < TILE_K; k++) {
            float a_vals[4];
            float b_vals[4];
            #pragma unroll
            for (int i = 0; i < 4; i++) {
                a_vals[i] = (row + i < M) ? As[(tx / (TILE_N / 4)) * 4 + i][k] : 0.0f;
                b_vals[i] = (col + i < N) ? Bs[k][(tx % (TILE_N / 4)) * 4 + i] : 0.0f;
            }
            #pragma unroll
            for (int i = 0; i < 4; i++) {
                #pragma unroll
                for (int j = 0; j < 4; j++) {
                    sum[i][j] += a_vals[i] * b_vals[j];
                }
            }
        }

        __syncthreads();
    }

    // Write results
    #pragma unroll
    for (int i = 0; i < 4; i++) {
        #pragma unroll
        for (int j = 0; j < 4; j++) {
            int global_row = row + i;
            int global_col = col + j;
            if (global_row < M && global_col < N) {
                C[global_row * N + global_col] = sum[i][j];
            }
        }
    }
}

torch::Tensor matmul_cuda(torch::Tensor A, torch::Tensor B) {
    int M = A.size(0);
    int K = A.size(1);
    int N = B.size(1);

    auto C = torch::zeros({M, N}, A.options());

    dim3 blocks((N + TILE_N - 1) / TILE_N, (M + TILE_M - 1) / TILE_M);
    dim3 threads(THREADS_PER_BLOCK);

    matmul_kernel<<<blocks, threads>>>(A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(), M, K, N);

    return C;
}
'''
