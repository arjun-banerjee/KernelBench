import torch.utils.cpp_extension

# Define CUDA kernel and its wrapper
cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__device__ float sigmoid(float x) {
    return 1.0f / (1.0f + expf(-x));
}

__global__ void fused_kernel(float *output, const float *input, const float *weight, const float *bias, const float *gamma, const float *beta, const float *multiply_weight, int batch_size, int out_features) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= batch_size * out_features) return;

    int batch_idx = idx / out_features;
    int feature_idx = idx % out_features;

    // GEMM output
    float gemm_out = output[idx];

    // GroupNorm
    float group_norm_mean = 0.0f;
    float group_norm_var = 0.0f;
    int group_size = out_features / 256;
    int group_idx = feature_idx / group_size;
    for (int i = batch_idx * out_features + group_idx * group_size; i < batch_idx * out_features + (group_idx + 1) * group_size; ++i) {
        group_norm_mean += output[i];
    }
    group_norm_mean /= group_size;
    for (int i = batch_idx * out_features + group_idx * group_size; i < batch_idx * out_features + (group_idx + 1) * group_size; ++i) {
        float diff = output[i] - group_norm_mean;
        group_norm_var += diff * diff;
    }
    group_norm_var /= group_size;
    float group_norm_std = sqrtf(group_norm_var + 1e-5f);
    float group_norm_out = (gemm_out - group_norm_mean) / group_norm_std * gamma[feature_idx] + beta[feature_idx];

    // Swish
    float swish_out = group_norm_out * sigmoid(group_norm_out);

    // Multiply
    float multiply_out = swish_out * multiply_weight[feature_idx];

    // Swish again
    output[idx] = multiply_out * sigmoid(multiply_out);
}

void wrapper(float *output, const float *input, const float *weight, const float *bias, const float *gamma, const float *beta, const float *multiply_weight, int batch_size, int in_features, int out_features) {

    cublasCreate(&handle);

    cudaFree(gemm_out);
}
'''
