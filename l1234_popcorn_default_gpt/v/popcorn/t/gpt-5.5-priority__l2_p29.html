import torch
import torch.nn as nn
import torch.nn.functional as F
import math


class Model(nn.Module):
    """
    Row-wise gated self-attention over Multiple Sequence Alignment (MSA)
    representations.  Each row of the MSA is an independent sequence; for
    each row, standard multi-head attention is performed across residue
    positions, with an additive bias from the pair representation.
    """

    def __init__(self, msa_dim, pair_dim, num_heads, dropout=0.0):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        assert msa_dim % num_heads == 0

        self.layer_norm = nn.LayerNorm(msa_dim)
        self.query = nn.Linear(msa_dim, msa_dim, bias=False)
        self.key = nn.Linear(msa_dim, msa_dim, bias=False)
        self.value = nn.Linear(msa_dim, msa_dim, bias=False)
        self.gate = nn.Linear(msa_dim, msa_dim)
        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)
        self.out_proj = nn.Linear(msa_dim, msa_dim)
        self.dropout = nn.Dropout(dropout)

    def forward(
        self,
        msa_repr: torch.Tensor,
        pair_repr: torch.Tensor,
        msa_mask: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            msa_repr:  (B, S, N, msa_dim)
            pair_repr: (B, N, N, pair_dim)
            msa_mask:  (B, S, N)
        Returns:
            updated MSA (B, S, N, msa_dim)
        """
        B, S, N, D = msa_repr.shape
        h = self.num_heads
        d = self.head_dim

        x = self.layer_norm(msa_repr)
        q = self.query(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        k = self.key(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = self.value(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        g = torch.sigmoid(self.gate(x)).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)

        # Pair bias: shared across MSA rows
        bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2)  # (B, h, N, N)

        attn = torch.einsum("bshid,bshjd->bshij", q, k) / math.sqrt(d)
        attn = attn + bias.unsqueeze(1)

        mask = msa_mask.view(B, S, 1, 1, N).expand_as(attn)
        attn = attn.masked_fill(mask == 0, float("-inf"))
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)

        out = torch.einsum("bshij,bshjd->bshid", attn, v)
        out = g * out
        out = out.permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        return self.out_proj(out)

        float s = 0.0f;
        #pragma unroll
        for (int j = 0; j < 32; ++j) {
            float e = expf(scores[j] - m);
            scores[j] = e;
            s += e;
        }
        denom = 1.0f / s;
    }
    __syncthreads();

    if (tid < HD) {
        int l = tid;
        float acc = 0.0f;
        #pragma unroll
        for (int j = 0; j < 32; ++j) {
            float a = scores[j] * denom;
            int vj = (((t * H + h) * N + j) * HD + l);
            acc += a * v[vj];
        }
        int gi = (((t * H + h) * N + i) * HD + l);
        float out = acc * g[gi];
        tmp_out[(t * N + i) * D + h * HD + l] = out;
    }
}

__global__ void out_proj_kernel(
    const float* __restrict__ tmp,
    const float* __restrict__ out_w,
    const float* __restrict__ out_b,
    float* __restrict__ out,
    int total_tokens, int D)
{
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = total_tokens * D;
    if (idx >= total) return;
    int c = idx & 63;
    int tok = idx >> 6;
    const float* x = tmp + tok * D;
    const float* w = out_w + c * D;
    float acc = out_b[c];
    #pragma unroll
    for (int r = 0; r < 64; ++r) acc += x[r] * w[r];
    out[idx] = acc;
}

torch::Tensor msa_forward(
    torch::Tensor msa, torch::Tensor pair, torch::Tensor mask,
    torch::Tensor ln_w, torch::Tensor ln_b,
    torch::Tensor q_w, torch::Tensor k_w, torch::Tensor v_w,
    torch::Tensor gate_w, torch::Tensor gate_b,
    torch::Tensor pair_w,
    torch::Tensor out_w, torch::Tensor out_b,
    int64_t num_heads)
{
    CHECK_CUDA(msa); CHECK_CUDA(pair); CHECK_CUDA(mask);
    CHECK_FLOAT(msa); CHECK_FLOAT(pair); CHECK_FLOAT(mask);
    const int B = (int)msa.size(0);
    const int S = (int)msa.size(1);
    const int N = (int)msa.size(2);
    const int D = (int)msa.size(3);
    const int H = (int)num_heads;
    const int HD = D / H;
    const int P = (int)pair.size(3);
    const int tokens = B * S * N;
    auto opts = msa.options();
    auto q = torch::empty({B*S, H, N, HD}, opts);
    auto k = torch::empty({B*S, H, N, HD}, opts);
    auto v = torch::empty({B*S, H, N, HD}, opts);
    auto g = torch::empty({B*S, H, N, HD}, opts);
    auto pb = torch::empty({B, H, N, N}, opts);
    auto tmp = torch::empty({B, S, N, D}, opts);
    auto out = torch::empty({B, S, N, D}, opts);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    ln_proj_kernel<<<tokens, 256, 0, stream>>>(
        msa.data_ptr<float>(), ln_w.data_ptr<float>(), ln_b.data_ptr<float>(),
        q_w.data_ptr<float>(), k_w.data_ptr<float>(), v_w.data_ptr<float>(),
        gate_w.data_ptr<float>(), gate_b.data_ptr<float>(),
        q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(), g.data_ptr<float>(),
        tokens, N, D, H, HD);
    int pb_total = B * H * N * N;
    pair_bias_kernel<<<(pb_total + 255) / 256, 256, 0, stream>>>(
        pair.data_ptr<float>(), pair_w.data_ptr<float>(), pb.data_ptr<float>(), B, N, P, H);
    attention_kernel<<<B * S * H * N, 32, 0, stream>>>(
        q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(), g.data_ptr<float>(),
        pb.data_ptr<float>(), mask.data_ptr<float>(), tmp.data_ptr<float>(),
        B, S, N, D, H, HD);
    int total = tokens * D;
    out_proj_kernel<<<(total + 255) / 256, 256, 0, stream>>>(
        tmp.data_ptr<float>(), out_w.data_ptr<float>(), out_b.data_ptr<float>(),
        out.data_ptr<float>(), tokens, D);
    return out;
}
