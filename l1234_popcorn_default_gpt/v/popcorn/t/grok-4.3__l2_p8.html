import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Grounded top-2 MoE router.

    Router logits are biased by token-expert similarity against learned
    grounding embeddings, then normalized over the selected top-2 experts.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        token_hidden: torch.Tensor,
        router_logits: torch.Tensor,
        expert_ground: torch.Tensor,
        alpha: float,
    ) -> torch.Tensor:
        grounded = router_logits + alpha * (token_hidden @ expert_ground.t())
        top_vals, top_idx = torch.topk(grounded, k=2, dim=-1)
        top_weights = torch.softmax(top_vals, dim=-1)
        return torch.stack((top_idx.to(torch.float32), top_weights), dim=-1)


num_tokens = 4096
hidden_dim = 128
num_experts = 16
alpha = 0.35


def get_inputs():

    router_logits = torch.randn(num_tokens, num_experts, dtype=torch.float32)
    expert_ground = torch.randn(num_experts, hidden_dim, dtype=torch.float32)
    return [token_hidden, router_logits, expert_ground, alpha]

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void grounded_top2_kernel(
    const float* __restrict__ token_hidden,
    const float* __restrict__ router_logits,
    const float* __restrict__ expert_ground,
    float* __restrict__ output,
    int num_tokens,
    int hidden_dim,
    int num_experts,
    float alpha
) {
    int token_idx = blockIdx.x;
    if (token_idx >= num_tokens) return;

    extern __shared__ float s_grounded[];

    const float* token = token_hidden + token_idx * hidden_dim;

    // Compute grounded scores for this token
    for (int e = threadIdx.x; e < num_experts; e += blockDim.x) {
        float dot = 0.0f;
        const float* expert = expert_ground + e * hidden_dim;
        for (int d = 0; d < hidden_dim; ++d) {
            dot += token[d] * expert[d];
        }
        s_grounded[e] = router_logits[token_idx * num_experts + e] + alpha * dot;
    }
    __syncthreads();

    // Find top-2 (only thread 0 does this since num_experts small)
    if (threadIdx.x == 0) {
        float max1 = -1e30f, max2 = -1e30f;
        int idx1 = -1, idx2 = -1;
        for (int e = 0; e < num_experts; ++e) {
            float v = s_grounded[e];
            if (v > max1) {
                max2 = max1; idx2 = idx1;
                max1 = v; idx1 = e;
            } else if (v > max2) {
                max2 = v; idx2 = e;
            }
        }
        float m = fmaxf(max1, max2);
        float sum_exp = expf(max1 - m) + expf(max2 - m);
        float w1 = expf(max1 - m) / sum_exp;
        float w2 = expf(max2 - m) / sum_exp;

        // Output: [token, 0, :] = idx1, w1; [token, 1, :] = idx2, w2 ? Wait, order by top
        // Actually topk is descending, so idx1 > idx2 in value
        output[token_idx * 4 + 0] = static_cast<float>(idx1);
        output[token_idx * 4 + 1] = w1;
        output[token_idx * 4 + 2] = static_cast<float>(idx2);
        output[token_idx * 4 + 3] = w2;
    }
}

torch::Tensor grounded_top2_cuda(
    torch::Tensor token_hidden,
    torch::Tensor router_logits,
    torch::Tensor expert_ground,
    double alpha
) {
    int num_tokens = token_hidden.size(0);
    int hidden_dim = token_hidden.size(1);
    int num_experts = router_logits.size(1);

    auto options = torch::TensorOptions().dtype(torch::kFloat32).device(token_hidden.device());
    auto output = torch::empty({num_tokens, 2, 2}, options);

    int threads = 32; // enough since experts=16
    int shared_mem = num_experts * sizeof(float);
    grounded_top2_kernel<<<num_tokens, threads, shared_mem>>>(
        token_hidden.data_ptr<float>(),
        router_logits.data_ptr<float>(),
        expert_ground.data_ptr<float>(),
        output.data_ptr<float>(),
        num_tokens, hidden_dim, num_experts, static_cast<float>(alpha)
    );

    return output;
}
