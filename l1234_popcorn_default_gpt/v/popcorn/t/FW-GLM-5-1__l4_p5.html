    const float* __restrict__ weight,
    float* __restrict__ output,
    float epsilon,
    int hidden_size,
    int num_rows
) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    
    if (row >= num_rows) return;
    
    extern __shared__ float shared_mem[];
    
    const float* row_in = input + row * hidden_size;
    float* row_out = output + row * hidden_size;
    
    // Compute sum of squares
    float sum_sq = 0.0f;
    for (int i = tid; i < hidden_size; i += blockDim.x) {
        float val = row_in[i];
        sum_sq += val * val;
    }
    shared_mem[tid] = sum_sq;
    __syncthreads();
    
    // Block reduction
    for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) {
            shared_mem[tid] += shared_mem[tid + stride];
        }
        __syncthreads();
    }
    
    const float rms_inv = rsqrtf(shared_mem[0] / hidden_size + epsilon);
    
    // Apply normalization and weight
    for (int i = tid; i < hidden_size; i += blockDim.x) {
        row_out[i] = row_in[i] * rms_inv * weight[i];
    }
}

torch::Tensor rms_norm_cuda(torch::Tensor input, torch::Tensor weight, double epsilon) {
    input = input.contiguous();
    weight = weight.contiguous();
    
    const int hidden_size = input.size(-1);
    const int num_rows = input.numel() / hidden_size;
    
    auto output = torch::empty_like(input);
    
    const int threads = hidden_size < 1024 ? hidden_size : 1024;
    const int shared_mem = threads * sizeof(float);
    
    rms_norm_kernel<<<num_rows, threads, shared_mem>>>(
        input.data_ptr<float>(),
        weight.data_ptr<float>(),
        output.data_ptr<float>(),
        static_cast<float>(epsilon),
        hidden_size,
        num_rows
    );
    
    return output;
}

// Fused SwiGLU: SiLU(gate) * up_proj
__global__ void swiglu_kernel(
    const float* __restrict__ gate,
    const float* __restrict__ up,
    float* __restrict__ output,
    int n
) {
    const int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        const float g = gate[idx];
        const float u = up[idx];
        // SiLU(x) = x * sigmoid(x) = x / (1 + exp(-x))
        const float silu = g / (1.0f + __expf(-g));
        output[idx] = silu * u;
    }
}

torch::Tensor swiglu_cuda(torch::Tensor gate, torch::Tensor up) {
    gate = gate.contiguous();
    up = up.contiguous();
    
    auto output = torch::empty_like(gate);
    const int n = gate.numel();
    const int threads = 256;
    const int blocks = (n + threads - 1) / threads;
    
    swiglu_kernel<<<blocks, threads>>>(
        gate.data_ptr<float>(),
        up.data_ptr<float>(),
        output.data_ptr<float>(),
        n
    );
    
    return output;
}

// Fused residual add with layer norm input preparation
__global__ void residual_add_kernel(
    const float* __restrict__ residual,
    const float* __restrict__ hidden,
    float* __restrict__ output,
    int n
) {
    const int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        output[idx] = residual[idx] + hidden[idx];
    }
}

torch::Tensor residual_add_cuda(torch::Tensor residual, torch::Tensor hidden) {
    residual = residual.contiguous();
    hidden = hidden.contiguous();
    
    auto output = torch::empty_like(residual);
    const int n = residual.numel();
    const int threads = 256;
    const int blocks = (n + threads - 1) / threads;
    
    residual_add_kernel<<<blocks, threads>>>(
        residual.data_ptr<float>(),
        hidden.data_ptr<float>(),
        output.data_ptr<float>(),
        n
    );
    
    return output;
}

// Fused final layer norm and projection to logits
__global__ void fused_norm_lmhead_kernel(
    const float* __restrict__ hidden_states,
    const float* __restrict__ norm_weight,
    const float* __restrict__ lm_head_weight,
    float* __restrict__ logits,

    extern __shared__ float shared_data[];

    const float* token_hidden = hidden_states + token_idx * hidden_size;
    
    // Compute RMS norm
    float sum_sq = 0.0f;
    for (int i = tid; i < hidden_size; i += blockDim.x) {
        const float val = token_hidden[i];
        sum_sq += val * val;
    }
    
    // Warp reduce for sum_sq
    for (int mask = 16; mask > 0; mask >>= 1) {
        sum_sq += __shfl_xor_sync(0xffffffff, sum_sq, mask);
    }
    
    __shared__ float s_rms_inv;
    if (tid == 0) {
        s_rms_inv = rsqrtf(sum_sq / hidden_size + epsilon);
    }
    __syncthreads();
    
    // Normalize into shared memory
    for (int i = tid; i < hidden_size; i += blockDim.x) {
        normed[i] = token_hidden[i] * s_rms_inv * norm_weight[i];
    }
    __syncthreads();

            sum += normed[h] * lm_head_weight[v * hidden_size + h];
        }

    }
}
