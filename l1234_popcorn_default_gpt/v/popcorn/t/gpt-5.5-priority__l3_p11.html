import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Fixed dimensions from the reference model
B = 16
H0 = 48
H1 = 64
H2 = 40
H3 = 24

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor mlp_forward_cuda(torch::Tensor h0, torch::Tensor w1, torch::Tensor w2, torch::Tensor w3);

torch::Tensor mlp_forward(torch::Tensor h0, torch::Tensor w1, torch::Tensor w2, torch::Tensor w3) {
    return mlp_forward_cuda(h0, w1, w2, w3);
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define H0 48
#define H1 64
#define H2 40
#define H3 24

__global__ __launch_bounds__(64, 4)
void mlp_kernel(const float* __restrict__ h0,
                const float* __restrict__ w1,
                const float* __restrict__ w2,
                const float* __restrict__ w3,
                float* __restrict__ out) {
    const int b = blockIdx.x;
    const int t = threadIdx.x;

    __shared__ float h1[H1];
    __shared__ float h2[H2];

    const float* x = h0 + b * H0;

    // h1 = relu(h0 @ W1.T).  One thread computes one output neuron.
    if (t < H1) {
        const float* w = w1 + t * H0;
        float s = 0.0f;
        #pragma unroll
        for (int k = 0; k < H0; ++k) {
            s = fmaf(x[k], w[k], s);
        }
        h1[t] = s > 0.0f ? s : 0.0f;
    }
    __syncthreads();

    // h2 = relu(h1 @ W2.T)
    if (t < H2) {
        const float* w = w2 + t * H1;
        float s = 0.0f;
        #pragma unroll
        for (int k = 0; k < H1; ++k) {
            s = fmaf(h1[k], w[k], s);
        }
        h2[t] = s > 0.0f ? s : 0.0f;
    }
    __syncthreads();

    // out = h2 @ W3.T
    if (t < H3) {
        const float* w = w3 + t * H2;
        float s = 0.0f;
        #pragma unroll
        for (int k = 0; k < H2; ++k) {
            s = fmaf(h2[k], w[k], s);
        }
        out[b * H3 + t] = s;
    }
}

torch::Tensor mlp_forward_cuda(torch::Tensor h0, torch::Tensor w1, torch::Tensor w2, torch::Tensor w3) {
    auto out = torch::empty({16, 24}, h0.options());
    mlp_kernel<<<16, 64>>>(
        h0.data_ptr<float>(),
        w1.data_ptr<float>(),
        w2.data_ptr<float>(),
        w3.data_ptr<float>(),
        out.data_ptr<float>());
    return out;
}
