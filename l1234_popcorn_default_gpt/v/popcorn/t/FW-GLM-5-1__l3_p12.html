import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Stack of planar normalizing flow layers.  Each layer applies the
    invertible transform f(z) = z + u * tanh(w^T z + b), with a
    log-determinant-Jacobian correction.  Used to transform a simple
    base distribution (e.g. standard Gaussian) into a complex posterior.
    """

    def __init__(self, dim, num_flows):
        super().__init__()
        self.dim = dim
        self.num_flows = num_flows
        self.w = nn.ParameterList([nn.Parameter(torch.randn(dim)) for _ in range(num_flows)])
        self.u = nn.ParameterList([nn.Parameter(torch.randn(dim)) for _ in range(num_flows)])
        self.b = nn.ParameterList([nn.Parameter(torch.zeros(1)) for _ in range(num_flows)])

    def _constrain_u(self, w, u):
        """Ensure invertibility: w^T u >= -1."""
        wtu = (w * u).sum()
        m = -1 + F.softplus(wtu)
        u_hat = u + (m - wtu) * w / (w ** 2).sum()
        return u_hat

    def forward(self, z: torch.Tensor) -> tuple:
        """
        Args:
            z: (B, dim) – samples from base distribution
        Returns:
            (z_out, sum_log_det):
                z_out:       (B, dim) – transformed samples
                sum_log_det: (B,)     – total log |det df/dz|
        """
        sum_log_det = torch.zeros(z.shape[0], device=z.device)

        for i in range(self.num_flows):
            w = self.w[i]
            u = self._constrain_u(w, self.u[i])
            b = self.b[i]

            linear = z @ w + b  # (B,)
            h = torch.tanh(linear)
            h_prime = 1.0 - h ** 2

            z = z + u.unsqueeze(0) * h.unsqueeze(-1)
            psi = h_prime.unsqueeze(-1) * w.unsqueeze(0)  # (B, dim)
            log_det = torch.log(torch.abs(1.0 + psi @ u) + 1e-8)
            sum_log_det = sum_log_det + log_det

        return z, sum_log_det

            int d = tid + i * block_dim;

                wtu_local += w_ptr[d] * u_ptr[d];
                w_sq_local += w_ptr[d] * w_ptr[d];
            }
        }
        
        float wtu = warp_reduce_sum(wtu_local);
        float w_sq_sum = warp_reduce_sum(w_sq_local);
        
        // Warp-level reduction to shared memory
        int warp_id = tid / 32;
        int lane_id = tid % 32;
        
        if (lane_id == 0) {
            s_linear[warp_id] = wtu;
            s_psi_u[warp_id] = w_sq_sum;
        }
        __syncthreads();
        
        // First warp does final reduction
        if (warp_id == 0) {
            float wtu_val = (lane_id < num_warps) ? s_linear[lane_id] : 0.0f;
            float w_sq_val = (lane_id < num_warps) ? s_psi_u[lane_id] : 0.0f;
            wtu_val = warp_reduce_sum(wtu_val);
            w_sq_val = warp_reduce_sum(w_sq_val);
            if (lane_id == 0) {
                s_linear[0] = wtu_val;
                s_psi_u[0] = w_sq_val;
            }
        }
        __syncthreads();
        
        wtu = s_linear[0];
        w_sq_sum = s_psi_u[0] + 1e-8f;  // avoid division by zero
        
        // Compute constrained u
        // m = -1 + softplus(wtu) = -1 + log(1 + exp(wtu))
        float softplus_wtu;
        if (wtu > 20.0f) {
            softplus_wtu = wtu;  // avoid overflow
        } else if (wtu < -20.0f) {
            softplus_wtu = 0.0f;  // avoid underflow
        } else {
            softplus_wtu = logf(1.0f + expf(wtu));
        }
        float m = -1.0f + softplus_wtu;
        
        // u_hat = u + (m - wtu) * w / w_sq_sum
        float u_scale = (m - wtu) / w_sq_sum;
        
        // Step 2: Compute linear = z @ w + b
        linear = 0.0f;
        for (int i = 0; i < stride; i++) {
            int d = tid + i * block_dim;
            if (d < dim) {
                int idx = batch_idx * dim + d;
                float z_val = z_out[idx];
                float u_hat = u_ptr[d] + u_scale * w_ptr[d];
                linear += z_val * w_ptr[d];
            }
        }
        
        // Reduce linear across threads
        linear = warp_reduce_sum(linear);
        
        if (lane_id == 0) {
            s_linear[warp_id] = linear;
        }
        __syncthreads();
        
        if (warp_id == 0) {
            float lin_val = (lane_id < num_warps) ? s_linear[lane_id] : 0.0f;
            lin_val = warp_reduce_sum(lin_val);
            if (lane_id == 0) {
                s_linear[0] = lin_val;
            }
        }
        __syncthreads();
        
        float linear_full = s_linear[0] + b;
        
        // Step 3: Compute h = tanh(linear), h_prime = 1 - h^2
        float h = tanhf(linear_full);
        float h_prime = 1.0f - h * h;
        
        // Step 4: Update z and compute psi @ u_hat
        psi_u = 0.0f;
        for (int i = 0; i < stride; i++) {
            int d = tid + i * block_dim;
            if (d < dim) {
                int idx = batch_idx * dim + d;
                float z_val = z_out[idx];
                float u_hat = u_ptr[d] + u_scale * w_ptr[d];
                // Update z
                z_out[idx] = z_val + u_hat * h;
                // Accumulate psi @ u_hat
                psi_u += h_prime * w_ptr[d] * u_hat;
            }
        }
        
        // Reduce psi_u across threads
        psi_u = warp_reduce_sum(psi_u);
        
        if (lane_id == 0) {
            s_psi_u[warp_id] = psi_u;
        }
        __syncthreads();
        
        if (warp_id == 0) {
            float psi_val = (lane_id < num_warps) ? s_psi_u[lane_id] : 0.0f;
            psi_val = warp_reduce_sum(psi_val);
            if (lane_id == 0) {
                s_psi_u[0] = psi_val;
            }
        }
        __syncthreads();
        
        float psi_u_full = s_psi_u[0];
        
        // Compute log determinant
        float log_det = logf(fabsf(1.0f + psi_u_full) + 1e-8f);
        
        // Accumulate log determinant (only thread 0 does this)
        if (tid == 0) {
            log_det_out[batch_idx] = (flow == 0) ? log_det : log_det_out[batch_idx] + log_det;
        }
        __syncthreads();
    }
}

class ModelNew(torch.nn.Module):
    def __init__(self, dim, num_flows):
        super().__init__()
        self.dim = dim
        self.num_flows = num_flows
        self.w = torch.nn.ParameterList([
            torch.nn.Parameter(torch.randn(dim)) for _ in range(num_flows)
        ])
        self.u = torch.nn.ParameterList([
            torch.nn.Parameter(torch.randn(dim)) for _ in range(num_flows)
        ])
        self.b = torch.nn.ParameterList([
            torch.nn.Parameter(torch.zeros(1)) for _ in range(num_flows)
        ])
    
    def _constrain_u(self, w, u):
        """Ensure invertibility: w^T u >= -1."""
        wtu = (w * u).sum()
        m = -1 + torch.nn.functional.softplus(wtu)
        u_hat = u + (m - wtu) * w / (w ** 2).sum()
        return u_hat
    
    def forward(self, z):
        # Pack parameters into contiguous tensors
        w_packed = torch.stack([self.w[i] for i in range(self.num_flows)]).contiguous()
        u_packed = torch.stack([self.u[i] for i in range(self.num_flows)]).contiguous()
        b_packed = torch.stack([self.b[i].squeeze() for i in range(self.num_flows)]).contiguous()
        
        # Ensure same device and dtype
        w_packed = w_packed.to(z.dtype).to(z.device)
        u_packed = u_packed.to(z.dtype).to(z.device)
        b_packed = b_packed.to(z.dtype).to(z.device)
        
        # Call CUDA kernel
        z_out, log_det = flow_cuda.planar_flow_forward_cuda(
            z.contiguous(),
            w_packed,
            u_packed,
            b_packed,
            self.num_flows
        )
        
        return z_out, log_det
