import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

void rope_inplace(torch::Tensor q, torch::Tensor k,
                  torch::Tensor cos, torch::Tensor sin,
                  int64_t rot_dim);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("rope_inplace", &rope_inplace, "Hopper optimized rotary embedding (in-place)");
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

__global__ __launch_bounds__(256, 3)
void rope_kernel_h100(half* __restrict__ q,
                      half* __restrict__ k,
                      const half* __restrict__ cos,
                      const half* __restrict__ sin,
                      const int rot_dim,
                      const int num_heads,
                      const int num_kv_heads,
                      const int head_size) {
  const int token = blockIdx.x;
  const int tid = threadIdx.x;
  const int half_dim = rot_dim >> 1;

  extern __shared__ half smem[];
  half* sh_cos = smem;
  half* sh_sin = smem + rot_dim;

  // The same rotary table row is used by every head of this token.  Put it in
  // shared memory once; on H100 this is cheap and avoids re-reading the 64
  // values for all 40 Q/KV heads.
  for (int i = tid; i < rot_dim; i += blockDim.x) {
    sh_cos[i] = cos[token * rot_dim + i];
    sh_sin[i] = sin[token * rot_dim + i];
  }
  __syncthreads();

  const int q_pairs = num_heads * half_dim;
  const int k_pairs = num_kv_heads * half_dim;
  const int total_pairs = q_pairs + k_pairs;

  for (int p = tid; p < total_pairs; p += blockDim.x) {
    if (p < q_pairs) {
      const int h = p / half_dim;
      const int r = p - h * half_dim;
      half* base = q + ((token * num_heads + h) * head_size);

      const half xh = base[r];
      const half yh = base[r + half_dim];
      const float x = __half2float(xh);
      const float y = __half2float(yh);

      const float c0 = __half2float(sh_cos[r]);
      const float s0 = __half2float(sh_sin[r]);
      const float c1 = __half2float(sh_cos[r + half_dim]);
      const float s1 = __half2float(sh_sin[r + half_dim]);

      base[r] = __float2half_rn(x * c0 - y * s0);
      base[r + half_dim] = __float2half_rn(y * c1 + x * s1);
    } else {
      const int kp = p - q_pairs;
      const int h = kp / half_dim;
      const int r = kp - h * half_dim;
      half* base = k + ((token * num_kv_heads + h) * head_size);

      const half xh = base[r];
      const half yh = base[r + half_dim];
      const float x = __half2float(xh);
      const float y = __half2float(yh);

      const float c0 = __half2float(sh_cos[r]);
      const float s0 = __half2float(sh_sin[r]);
      const float c1 = __half2float(sh_cos[r + half_dim]);
      const float s1 = __half2float(sh_sin[r + half_dim]);

      base[r] = __float2half_rn(x * c0 - y * s0);
      base[r + half_dim] = __float2half_rn(y * c1 + x * s1);
    }
  }
}

void rope_inplace(torch::Tensor q, torch::Tensor k,
                  torch::Tensor cos, torch::Tensor sin,
                  int64_t rot_dim64) {
  const int tokens = static_cast<int>(q.size(0));
  const int num_heads = static_cast<int>(q.size(1));
  const int head_size = static_cast<int>(q.size(2));
  const int num_kv_heads = static_cast<int>(k.size(1));
  const int rot_dim = static_cast<int>(rot_dim64);

  const int threads = 256;
  const size_t shmem = static_cast<size_t>(2 * rot_dim) * sizeof(half);
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();

  rope_kernel_h100<<<tokens, threads, shmem, stream>>>(
      reinterpret_cast<half*>(q.data_ptr<at::Half>()),
      reinterpret_cast<half*>(k.data_ptr<at::Half>()),
      reinterpret_cast<const half*>(cos.data_ptr<at::Half>()),
      reinterpret_cast<const half*>(sin.data_ptr<at::Half>()),
      rot_dim, num_heads, num_kv_heads, head_size);
}
"""
