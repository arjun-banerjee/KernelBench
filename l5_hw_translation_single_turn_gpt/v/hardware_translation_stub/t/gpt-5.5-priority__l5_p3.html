// One CTA computes one 4096-wide row.  Each thread handles two 128-bit vectors
// for the fixed benchmark shape (4096 half values), keeping the kernel bandwidth
// oriented while using only a small amount of shared memory.
__global__ __launch_bounds__(256, 2)
void rms_norm_fp16_vec8_kernel(
        __half* __restrict__ out,
        const __half* __restrict__ inp,
        const __half* __restrict__ weight,
        const float eps,
        const int hidden_size) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int n_vec = hidden_size >> 3;  // eight fp16s per uint4

    const __half* __restrict__ xrow = inp + (size_t)row * hidden_size;
    __half* __restrict__ orow = out + (size_t)row * hidden_size;
    const uint4* __restrict__ x4 = reinterpret_cast<const uint4*>(xrow);

    float ss = 0.0f;
#pragma unroll 2
    for (int v = tid; v < n_vec; v += 256) {
        const uint4 p = x4[v];
        const __half2 h0 = h2_from_u32(p.x);
        const __half2 h1 = h2_from_u32(p.y);
        const __half2 h2 = h2_from_u32(p.z);
        const __half2 h3 = h2_from_u32(p.w);
        const float2 f0 = __half22float2(h0);
        const float2 f1 = __half22float2(h1);
        const float2 f2 = __half22float2(h2);
        const float2 f3 = __half22float2(h3);
        ss = fmaf(f0.x, f0.x, ss); ss = fmaf(f0.y, f0.y, ss);
        ss = fmaf(f1.x, f1.x, ss); ss = fmaf(f1.y, f1.y, ss);
        ss = fmaf(f2.x, f2.x, ss); ss = fmaf(f2.y, f2.y, ss);
        ss = fmaf(f3.x, f3.x, ss); ss = fmaf(f3.y, f3.y, ss);
    }

    __shared__ float smem[8];
    ss = warp_sum(ss);
    if ((tid & 31) == 0) smem[tid >> 5] = ss;

        const __half2 w0h = h2_from_u32(pw.x);
        const __half2 w1h = h2_from_u32(pw.y);
        const __half2 w2h = h2_from_u32(pw.z);
        const __half2 w3h = h2_from_u32(pw.w);

        const float2 x0 = __half22float2(x0h); const float2 w0 = __half22float2(w0h);
        const float2 x1 = __half22float2(x1h); const float2 w1 = __half22float2(w1h);
        const float2 x2 = __half22float2(x2h); const float2 w2 = __half22float2(w2h);
