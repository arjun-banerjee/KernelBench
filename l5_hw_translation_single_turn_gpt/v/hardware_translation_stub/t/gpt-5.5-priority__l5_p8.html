                               half* __restrict__ y,

    const int warps_per_block = blockDim.x >> 5;
    const int out_idx = blockIdx.x * warps_per_block + warp;

    const int n = out_idx - m * N;
    const half* xp = x + (size_t)m * K;

    int k2_lim = K & ~1;
