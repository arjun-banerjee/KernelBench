void dq_kernel(const __nv_bfloat16* __restrict__ q,
               const __nv_bfloat16* __restrict__ k,
               const __nv_bfloat16* __restrict__ v,
               const __nv_bfloat16* __restrict__ dout,
               const float* __restrict__ lse,
               const float* __restrict__ delta,
               __nv_bfloat16* __restrict__ dq,
               int Hq, int Hkv) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int group = Hq / Hkv;
    const int i = row % S_LEN;
    const int hq = (row / S_LEN) % Hq;
    const int b = row / (Hq * S_LEN);
    const int kvh = hq / group;
    const float scale = 0.08838834764831845f;

    const __nv_bfloat16* qptr = q + (((b * Hq + hq) * S_LEN + i) * D_DIM);
    const __nv_bfloat16* doptr = dout + (((b * Hq + hq) * S_LEN + i) * D_DIM);
    const __nv_bfloat16* kbase = k + ((b * Hkv + kvh) * S_LEN * D_DIM);
    const __nv_bfloat16* vbase = v + ((b * Hkv + kvh) * S_LEN * D_DIM);
    __shared__ float ds[S_LEN];
    const float li = lse[row];
    const float di = delta[row];

    for (int j = tid; j < S_LEN; j += blockDim.x) {
        const __nv_bfloat16* kptr = kbase + j * D_DIM;
        const __nv_bfloat16* vptr = vbase + j * D_DIM;
        float qk = 0.f, dp = 0.f;
#pragma unroll 4
        for (int d = 0; d < D_DIM; ++d) {
            qk += bf16_to_f(qptr[d]) * bf16_to_f(kptr[d]);
            dp += bf16_to_f(doptr[d]) * bf16_to_f(vptr[d]);
        }
        float p = expf(qk * scale - li);
        ds[j] = p * (dp - di) * scale;
    }
    __syncthreads();

    if (tid < D_DIM) {
        float acc = 0.f;
#pragma unroll 4
        for (int j = 0; j < S_LEN; ++j) {
            acc += ds[j] * bf16_to_f(kbase[j * D_DIM + tid]);
        }
        dq[((b * Hq + hq) * S_LEN + i) * D_DIM + tid] = __float2bfloat16_rn(acc);
    }
}

__global__ __launch_bounds__(256, 2)
void dkv_kernel(const __nv_bfloat16* __restrict__ q,
                const __nv_bfloat16* __restrict__ k,
                const __nv_bfloat16* __restrict__ v,
                const __nv_bfloat16* __restrict__ dout,
                const float* __restrict__ lse,
                const float* __restrict__ delta,
                float* __restrict__ dk,
                float* __restrict__ dv,
                int Hq, int Hkv) {
    const int tid = threadIdx.x;
    const int kt = blockIdx.x;
    const int bhq = blockIdx.y;
    const int hq = bhq % Hq;
    const int b = bhq / Hq;
    const int group = Hq / Hkv;
    const int kvh = hq / group;
    const int k_start = kt * KTILE;
    const float scale = 0.08838834764831845f;

    extern __shared__ float sh[];
    float* acc_dk = sh;                         // KTILE * D_DIM
    float* acc_dv = acc_dk + KTILE * D_DIM;     // KTILE * D_DIM
    float* psh = acc_dv + KTILE * D_DIM;        // QTILE * KTILE
    float* dssh = psh + QTILE * KTILE;          // QTILE * KTILE

    for (int idx = tid; idx < KTILE * D_DIM; idx += blockDim.x) {
        acc_dk[idx] = 0.f;
        acc_dv[idx] = 0.f;
    }
    __syncthreads();

    const __nv_bfloat16* qbase = q + (b * Hq + hq) * S_LEN * D_DIM;
    const __nv_bfloat16* dobase = dout + (b * Hq + hq) * S_LEN * D_DIM;
    const __nv_bfloat16* kbase = k + (b * Hkv + kvh) * S_LEN * D_DIM;
    const __nv_bfloat16* vbase = v + (b * Hkv + kvh) * S_LEN * D_DIM;

    for (int qt = 0; qt < S_LEN; qt += QTILE) {
        if (tid < QTILE * KTILE) {
            int qi = tid / KTILE;
            int kj = tid - qi * KTILE;
            int qrow = qt + qi;
            int krow = k_start + kj;
            const __nv_bfloat16* qptr = qbase + qrow * D_DIM;
            const __nv_bfloat16* doptr = dobase + qrow * D_DIM;
            const __nv_bfloat16* kptr = kbase + krow * D_DIM;
            const __nv_bfloat16* vptr = vbase + krow * D_DIM;
            float qk = 0.f, dp = 0.f;
#pragma unroll 4
            for (int d = 0; d < D_DIM; ++d) {
                qk += bf16_to_f(qptr[d]) * bf16_to_f(kptr[d]);
                dp += bf16_to_f(doptr[d]) * bf16_to_f(vptr[d]);
            }
            int ridx = (b * Hq + hq) * S_LEN + qrow;
            float p = expf(qk * scale - lse[ridx]);
            psh[tid] = p;
            dssh[tid] = p * (dp - delta[ridx]) * scale;
        }
        __syncthreads();

        for (int idx = tid; idx < KTILE * D_DIM; idx += blockDim.x) {
            int kj = idx / D_DIM;
            int d = idx - kj * D_DIM;
            float av = 0.f, ak = 0.f;
#pragma unroll
            for (int qi = 0; qi < QTILE; ++qi) {
                int qrow = qt + qi;
                float pp = psh[qi * KTILE + kj];
                float ss = dssh[qi * KTILE + kj];
                av += pp * bf16_to_f(dobase[qrow * D_DIM + d]);
                ak += ss * bf16_to_f(qbase[qrow * D_DIM + d]);
            }
            acc_dv[idx] += av;
            acc_dk[idx] += ak;
        }
        __syncthreads();
    }

    for (int idx = tid; idx < KTILE * D_DIM; idx += blockDim.x) {
        int kj = idx / D_DIM;
        int d = idx - kj * D_DIM;
        int krow = k_start + kj;
        int out_idx = ((b * Hkv + kvh) * S_LEN + krow) * D_DIM + d;
        atomicAdd(dv + out_idx, acc_dv[idx]);
        atomicAdd(dk + out_idx, acc_dk[idx]);
    }
}

__global__ void cast_kernel(const float* __restrict__ dkf, const float* __restrict__ dvf,
                            __nv_bfloat16* __restrict__ dk, __nv_bfloat16* __restrict__ dv,
                            int n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        dk[idx] = __float2bfloat16_rn(dkf[idx]);
        dv[idx] = __float2bfloat16_rn(dvf[idx]);
    }
}

std::vector<torch::Tensor> flash_bwd(torch::Tensor q, torch::Tensor k, torch::Tensor v, torch::Tensor dout,
                                     int64_t Hq, int64_t Hkv) {
    auto dq = torch::empty_like(q);
    auto dk = torch::empty_like(k);
    auto dv = torch::empty_like(v);
    auto opts_f = torch::TensorOptions().device(q.device()).dtype(torch::kFloat32);
    auto lse = torch::empty({q.size(0), Hq, S_LEN}, opts_f);
    auto delta = torch::empty({q.size(0), Hq, S_LEN}, opts_f);
    auto dkf = torch::zeros({k.size(0), Hkv, S_LEN, D_DIM}, opts_f);
    auto dvf = torch::zeros({v.size(0), Hkv, S_LEN, D_DIM}, opts_f);

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    int rows = (int)(q.size(0) * Hq * S_LEN);
    lse_delta_kernel<<<rows, 256, 0, stream>>>((const __nv_bfloat16*)q.data_ptr<at::BFloat16>(),
        (const __nv_bfloat16*)k.data_ptr<at::BFloat16>(), (const __nv_bfloat16*)v.data_ptr<at::BFloat16>(),
        (const __nv_bfloat16*)dout.data_ptr<at::BFloat16>(), lse.data_ptr<float>(), delta.data_ptr<float>(),
        (int)Hq, (int)Hkv);
    dq_kernel<<<rows, 256, 0, stream>>>((const __nv_bfloat16*)q.data_ptr<at::BFloat16>(),
        (const __nv_bfloat16*)k.data_ptr<at::BFloat16>(), (const __nv_bfloat16*)v.data_ptr<at::BFloat16>(),
        (const __nv_bfloat16*)dout.data_ptr<at::BFloat16>(), lse.data_ptr<float>(), delta.data_ptr<float>(),
        (__nv_bfloat16*)dq.data_ptr<at::BFloat16>(), (int)Hq, (int)Hkv);
    dim3 grid_dkv(S_LEN / KTILE, (unsigned)(q.size(0) * Hq));
    size_t shmem = (2 * KTILE * D_DIM + 2 * QTILE * KTILE) * sizeof(float);
    dkv_kernel<<<grid_dkv, 256, shmem, stream>>>((const __nv_bfloat16*)q.data_ptr<at::BFloat16>(),
        (const __nv_bfloat16*)k.data_ptr<at::BFloat16>(), (const __nv_bfloat16*)v.data_ptr<at::BFloat16>(),
        (const __nv_bfloat16*)dout.data_ptr<at::BFloat16>(), lse.data_ptr<float>(), delta.data_ptr<float>(),
        dkf.data_ptr<float>(), dvf.data_ptr<float>(), (int)Hq, (int)Hkv);
    int n = (int)(k.size(0) * Hkv * S_LEN * D_DIM);
    cast_kernel<<<(n + 255) / 256, 256, 0, stream>>>(dkf.data_ptr<float>(), dvf.data_ptr<float>(),
        (__nv_bfloat16*)dk.data_ptr<at::BFloat16>(), (__nv_bfloat16*)dv.data_ptr<at::BFloat16>(), n);
    return {dq, dk, dv};
}
