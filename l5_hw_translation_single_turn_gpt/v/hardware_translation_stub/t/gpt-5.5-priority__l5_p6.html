    mark_done_kernel<<<1, 32, 0, stream>>>(p[rank], data_bytes, (unsigned long long)seq);
    AT_CUDA_CHECK(cudaGetLastError());
}
'''

    _ext = load_inline(
        name="h100_fp16_ipc_allreduce_ext",
        cpp_sources=cpp_src,
        cuda_sources=cuda_src,
        functions=["allocate_ipc_buffer", "open_ipc_handle", "free_ipc_buffer", "allreduce_ipc"],
        extra_cflags=["-O3"],
        extra_cuda_cflags=["-O3", "--use_fast_math"],
        with_cuda=True,
        verbose=False,
    )
    return _ext


class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.ext = _get_ext()
        if is_distributed_run():
            maybe_init_process_group()
        self.world_size = get_world_size() if (dist.is_available() and dist.is_initialized()) else 1
        self.rank = dist.get_rank() if (dist.is_available() and dist.is_initialized()) else 0
        self.seq = 0
        self.local_ptr = 0
        self.ptrs = []
        if self.world_size > 1:
            data_bytes = 512 * 4096 * 2
            self.local_ptr, handle = self.ext.allocate_ipc_buffer(data_bytes)
            hb = bytes(handle.cpu().tolist())
            gathered = [None for _ in range(self.world_size)]
            dist.all_gather_object(gathered, hb)
            ptrs = []
            for r, hbytes in enumerate(gathered):
                if r == self.rank:
                    ptrs.append(int(self.local_ptr))
                else:
                    ht = torch.tensor(list(hbytes), dtype=torch.uint8, device="cpu")
                    ptrs.append(int(self.ext.open_ipc_handle(ht)))
            self.ptrs = ptrs
            dist.barrier()

    def forward(self, x):
        if self.world_size <= 1:
            return x
        out = torch.empty_like(x)
        self.seq += 1
        self.ext.allreduce_ipc(x.contiguous(), out, self.ptrs, int(self.rank), int(self.seq))
        return out
