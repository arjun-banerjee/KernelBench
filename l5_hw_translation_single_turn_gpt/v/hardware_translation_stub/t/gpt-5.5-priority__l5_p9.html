#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_bf16.h>
#include <cmath>

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif
#ifndef CHECK_CONTIGUOUS

static constexpr int BATCH_C = 4;
static constexpr int H_Q_C = 32;
static constexpr int H_KV_C = 8;
static constexpr int SEQ_C = 2048;
static constexpr int DIM_C = 128;
static constexpr float SCALE_C = 0.08838834764831845f; // 1 / sqrt(128)

__inline__ __device__ float warp_sum(float v) {
    v += __shfl_down_sync(0xffffffff, v, 16);
    v += __shfl_down_sync(0xffffffff, v, 8);
    v += __shfl_down_sync(0xffffffff, v, 4);
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

__inline__ __device__ float block_sum_128(float v) {
    __shared__ float partial[4];
    const int lane = threadIdx.x & 31;
    const int wid = threadIdx.x >> 5;
    v = warp_sum(v);
    if (lane == 0) partial[wid] = v;
    __syncthreads();
    v = (threadIdx.x < 4) ? partial[threadIdx.x] : 0.0f;
    if (wid == 0) v = warp_sum(v);
    return v;
}

// One CTA computes one complete attention row.  This is a streaming softmax
// implementation: K and V are read once, logits are never materialized, and all
// reductions are in FP32.  The shape constants match the benchmark contract.
__global__ __launch_bounds__(128, 4)
void mha_fwd_bf16_hdim128_kernel(const __nv_bfloat16* __restrict__ q,
                                 const __nv_bfloat16* __restrict__ k,
                                 const __nv_bfloat16* __restrict__ v,
                                 __nv_bfloat16* __restrict__ out) {
    const int tid = threadIdx.x; // 0..127, also the head-dim element owned by this thread
    const int row = blockIdx.x;

    const int s_q = row & (SEQ_C - 1);          // SEQ_C is 2048
    const int tmp = row >> 11;
    const int h = tmp & (H_Q_C - 1);            // H_Q_C is 32
    const int b = tmp >> 5;
    const int h_kv = h >> 2;                    // 32 query heads / 8 kv heads

    const long long q_off = (((long long)b * H_Q_C + h) * SEQ_C + s_q) * DIM_C;
    const long long kv_base = (((long long)b * H_KV_C + h_kv) * SEQ_C) * DIM_C;

    const float q_reg = __bfloat162float(q[q_off + tid]);
    float acc = 0.0f;

    __shared__ float sh_score;
    __shared__ float sh_alpha;
    __shared__ float sh_beta;
    __shared__ float sh_l;

    float m = -CUDART_INF_F; // only thread 0's copy is authoritative
    float l = 0.0f;          // only thread 0's copy is authoritative

    #pragma unroll 1
    for (int j = 0; j < SEQ_C; ++j) {
        const long long ko = kv_base + (long long)j * DIM_C + tid;
        float prod = q_reg * __bfloat162float(k[ko]);
        float dot = block_sum_128(prod);

        if (tid == 0) {
            float score = dot * SCALE_C;
            float m_new = fmaxf(m, score);
            float alpha = (m == -CUDART_INF_F) ? 0.0f : __expf(m - m_new);
            float beta = __expf(score - m_new);
            l = l * alpha + beta;
            m = m_new;
            sh_alpha = alpha;
            sh_beta = beta;
            sh_l = l;
        }
        __syncthreads();

        const float vv = __bfloat162float(v[kv_base + (long long)j * DIM_C + tid]);
        acc = acc * sh_alpha + vv * sh_beta;
    }

    out[q_off + tid] = __float2bfloat16(acc / sh_l);
}
