With 65,536 blocks each computing 512*128, it seems like there’s a lot of multiplication involved and that could lead to significant overhead. However, the GPU is designed to handle high workloads, so I wonder if the performance might still be acceptable despite the overhead from this method. **Evaluating softmax normalization** I’m considering the combination of softmax normalization with the use of logits inverse. The current output leverages exponential and inverse functions. While it might have a trade-off in speed compared to PyTorch, I think it could still be acceptable. The computation involves 17G scalar operations rather than using tensor cores, and factoring in memory usage, it seems there are notable loads — around 256KB load adjustments for q and k. #include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_fp16.h>
#include <mma.h>

using namespace nvcuda;

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

static __device__ __forceinline__ float warp_reduce_sum(float v) {
    #pragma unroll
    for (int mask = 16; mask > 0; mask >>= 1) {
        v += __shfl_xor_sync(0xffffffff, v, mask);
    }
    return v;
}

static __device__ __forceinline__ float warp_reduce_max(float v) {
    #pragma unroll
    for (int mask = 16; mask > 0; mask >>= 1) {
        float u = __shfl_xor_sync(0xffffffff, v, mask);
        v = fmaxf(v, u);
    }
    return v;
}

// Fixed problem shape from the contract:
// q: [4, 32, 512, 128], k/v: [4, 8, 512, 128].
// One CTA computes a 16-query tile for one (batch, query-head).
// The QK^T and PV products use tensor cores; the full 16x512 score tile
// remains in shared memory for an exact row-wise softmax.
__global__ __launch_bounds__(256, 4)
void gqa_attn_kernel(const half* __restrict__ q,
                     const half* __restrict__ k,
                     const half* __restrict__ v,
                     half* __restrict__ out) {
    constexpr int B = 4;
    constexpr int H = 32;
    constexpr int HKV = 8;
    constexpr int S = 512;
    constexpr int D = 128;
    constexpr int QT = 16;
    constexpr int WARPS = 8;
    constexpr float SCALE = 0.08838834764831845f; // 1/sqrt(128)

    const int q_tile = blockIdx.x;   // 0..31
    const int h = blockIdx.y;        // 0..31
    const int b = blockIdx.z;        // 0..3
    const int tid = threadIdx.x;
    const int warp_id = tid >> 5;
    const int lane = tid & 31;
    const int q0 = q_tile * QT;
    const int kvh = h >> 2;          // 32 query heads / 8 kv heads = 4

    const half* __restrict__ q_base = q + (((b * H + h) * S + q0) * D);
    const half* __restrict__ k_base = k + ((b * HKV + kvh) * S * D);
    const half* __restrict__ v_base = v + ((b * HKV + kvh) * S * D);
    half* __restrict__ o_base = out + (((b * H + h) * S + q0) * D);

    __shared__ float logits[QT * S];      // [16, 512]
    __shared__ half probs[QT * S];        // [16, 512]
    __shared__ float out_frag[WARPS * 16 * 16]; // one 16x16 output tile per warp

    // Compute logits = Q[16,128] * K[512,128]^T.  Each warp owns one 16-column
    // K tile at a time, so 8 warps cover 128 keys per wave.
    for (int kt = warp_id * 16; kt < S; kt += WARPS * 16) {
        wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
        wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::col_major> b_frag;
        wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;
        wmma::fill_fragment(c_frag, 0.0f);

        #pragma unroll
        for (int kk = 0; kk < D; kk += 16) {
            wmma::load_matrix_sync(a_frag, q_base + kk, D);
            // Treat the row-major K tile as a column-major [128, 16] K^T tile.
            wmma::load_matrix_sync(b_frag, k_base + kt * D + kk, D);
            wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);
        }
        wmma::store_matrix_sync(logits + kt, c_frag, S, wmma::mem_row_major);
    }

    __syncthreads();

    // Row-wise stable softmax.  Eight warps process all 16 rows in two passes.
    for (int row = warp_id; row < QT; row += WARPS) {
        float m = -CUDART_INF_F;
        #pragma unroll
        for (int col = lane; col < S; col += WARP_SIZE) {
            m = fmaxf(m, logits[row * S + col] * SCALE);
        }
        m = warp_reduce_max(m);

        float s = 0.0f;
        #pragma unroll
        for (int col = lane; col < S; col += WARP_SIZE) {
            float e = __expf(logits[row * S + col] * SCALE - m);
            logits[row * S + col] = e;
            s += e;
        }
        s = warp_reduce_sum(s);
        float inv_s = 1.0f / s;

        #pragma unroll
        for (int col = lane; col < S; col += WARP_SIZE) {
            probs[row * S + col] = __float2half_rn(logits[row * S + col] * inv_s);
        }
    }

    __syncthreads();

    // Output = softmax(logits) * V.  One warp computes one 16-column output tile.
    const int od = warp_id * 16;
    {
        wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
        wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b_frag;
        wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;
        wmma::fill_fragment(c_frag, 0.0f);

        #pragma unroll
        for (int kk = 0; kk < S; kk += 16) {
            wmma::load_matrix_sync(a_frag, probs + kk, S);
            wmma::load_matrix_sync(b_frag, v_base + kk * D + od, D);
            wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);
        }
        wmma::store_matrix_sync(out_frag + warp_id * 256, c_frag, 16, wmma::mem_row_major);
    }

    __syncthreads();

    // Convert the 8 accumulator tiles to FP16 and write the contiguous output tile.
    for (int idx = tid; idx < QT * D; idx += blockDim.x) {
        const int row = idx >> 7;          // /128
        const int d = idx & 127;           // %128
        const int tile = d >> 4;           // /16
        const int col = d & 15;
        float val = out_frag[tile * 256 + row * 16 + col];
        o_base[row * D + d] = __float2half_rn(val);
    }
}

torch::Tensor gqa_attention(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
    auto out = torch::empty_like(q);
    dim3 block(256);
    dim3 grid(512 / 16, 32, 4);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    gqa_attn_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()));
    return out;
}
