    v += __shfl_down_sync(0xffffffff, v, 8);
    v += __shfl_down_sync(0xffffffff, v, 4);
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

__global__ __launch_bounds__(128, 2)
void linear128_kernel(const float* __restrict__ x,
                      const float* __restrict__ w,
                      const float* __restrict__ b,
                      float* __restrict__ y,
                      int64_t rows) {
    int row = blockIdx.x;
    int tile = blockIdx.y;
    int tid = threadIdx.x;
    int lane = tid & 31;
    int warp = tid >> 5;
    int out0 = tile * 32 + warp * OUTS_PER_WARP;
    if (row >= rows) return;

    const float* xp = x + ((int64_t)row) * FEAT;
    float s0 = 0.0f, s1 = 0.0f, s2 = 0.0f, s3 = 0.0f;
    float s4 = 0.0f, s5 = 0.0f, s6 = 0.0f, s7 = 0.0f;

    #pragma unroll
    for (int q = 0; q < 4; ++q) {
        int k = lane + q * 32;
        float xv = xp[k];
        const float* wp = w + k;
        s0 = fmaf(xv, wp[(out0 + 0) * FEAT], s0);
        s1 = fmaf(xv, wp[(out0 + 1) * FEAT], s1);
        s2 = fmaf(xv, wp[(out0 + 2) * FEAT], s2);
        s3 = fmaf(xv, wp[(out0 + 3) * FEAT], s3);
        s4 = fmaf(xv, wp[(out0 + 4) * FEAT], s4);
        s5 = fmaf(xv, wp[(out0 + 5) * FEAT], s5);
        s6 = fmaf(xv, wp[(out0 + 6) * FEAT], s6);
        s7 = fmaf(xv, wp[(out0 + 7) * FEAT], s7);
    }

    s0 = warp_sum(s0); s1 = warp_sum(s1); s2 = warp_sum(s2); s3 = warp_sum(s3);
    s4 = warp_sum(s4); s5 = warp_sum(s5); s6 = warp_sum(s6); s7 = warp_sum(s7);

    if (lane == 0) {
        float* yp = y + ((int64_t)row) * FEAT + out0;
        yp[0] = s0 + b[out0 + 0];
        yp[1] = s1 + b[out0 + 1];
        yp[2] = s2 + b[out0 + 2];
        yp[3] = s3 + b[out0 + 3];
        yp[4] = s4 + b[out0 + 4];
        yp[5] = s5 + b[out0 + 5];
        yp[6] = s6 + b[out0 + 6];
        yp[7] = s7 + b[out0 + 7];
    }
}

torch::Tensor linear_forward(torch::Tensor x, torch::Tensor w, torch::Tensor b) {
    auto y = torch::empty({x.size(0), (int64_t)FEAT}, x.options());
    int64_t rows = x.size(0);
    if (rows == 0) return y;
    dim3 block(128);
    dim3 grid((unsigned int)rows, FEAT / (4 * OUTS_PER_WARP));
    linear128_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>(
        x.data_ptr<float>(), w.data_ptr<float>(), b.data_ptr<float>(), y.data_ptr<float>(), rows);
    return y;
}
