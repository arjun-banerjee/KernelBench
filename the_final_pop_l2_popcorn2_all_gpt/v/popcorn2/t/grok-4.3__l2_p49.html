import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cmath>

__device__ __forceinline__ float gelu(float x) {
    const float kAlpha = M_SQRT1_2;
    return 0.5f * x * (1.0f + erf(x * kAlpha));
}

__global__ void gelu_sum_kernel(const float* __restrict__ input, float* __restrict__ output, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx < total) {
        float sum = 0.0f;
        for (int r = 0; r < R; r++) {
            float val = input[r * total + idx];
            sum += gelu(val);
        }
        output[idx] = sum;
    }
}

torch::Tensor gelu_sum_cuda(torch::Tensor x) {
    auto R = x.size(0);
    auto B = x.size(1);
    auto D = x.size(2);
    auto output = torch::empty({B, D}, x.options());
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    gelu_sum_kernel<<<blocks, threads>>>(x.data_ptr<float>(), output.data_ptr<float>(), R, B, D);
    return output;
}
