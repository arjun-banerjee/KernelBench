import torch
import torch.nn as nn
import math

class Model(nn.Module):
    """
    Batched Metropolis-Hastings MCMC step.  Given a current state and a
    log-probability function (here a Gaussian target for benchmarking),
    proposes a new state via a Gaussian random walk, computes the
    acceptance ratio, and stochastically accepts or rejects.
    """

    def __init__(self, dim, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))
        self.register_buffer('sigma', torch.ones(dim))

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (((x - self.mu) / self.sigma) ** 2).sum(dim=-1)

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, dim) – current chain states
        Returns:
            (x_new, accepted):
                x_new:    (B, dim) – states after accept/reject
                accepted: (B,)    – boolean mask of accepted proposals
        """
        proposal = x + self.step_size * torch.randn_like(x)
        log_alpha = self._log_prob(proposal) - self._log_prob(x)
        log_u = torch.log(torch.rand(x.shape[0], device=x.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        x_new = torch.where(accepted.unsqueeze(-1), proposal, x)
        return (x_new, accepted)

// Metropolis-Hastings step kernel - highly optimized version
// Uses warp-level parallelism for efficient memory access and reduction
__global__ void mh_step_kernel(
    const float* __restrict__ x,        // (B, dim) - current states
    const float* __restrict__ mu,       // (dim,) - target mean
    const float* __restrict__ sigma,    // (dim,) - target std
    const float* __restrict__ randn,    // (B, dim) - Gaussian random numbers
    const float* __restrict__ randu,    // (B,) - Uniform random numbers
    float* __restrict__ x_new,          // (B, dim) - output states
    bool* __restrict__ accepted,        // (B,) - acceptance mask
    int batch_size,
    int dim,
    float step_size
) {
    // Each warp handles one chain
    int warp_id = (blockIdx.x * blockDim.x + threadIdx.x) / 32;
    int lane_id = threadIdx.x % 32;
    
    if (warp_id >= batch_size) return;
    
    // Compute base pointers
    const float* x_base = x + warp_id * dim;
    const float* randn_base = randn + warp_id * dim;
    float* x_new_base = x_new + warp_id * dim;
    
    // Fused computation of log_prob(x) and proposal
    // This reduces memory traffic by reading x only once
    float log_prob_x = 0.0f;
    float log_prob_proposal = 0.0f;
    
    // Strided loop - lane i processes elements i, i+32, i+64, etc.
    // This ensures contiguous memory access within a warp
    for (int d = lane_id; d < dim; d += 32) {
        float x_val = x_base[d];
        float mu_val = __ldg(&mu[d]);
        float sigma_val = __ldg(&sigma[d]);
        float rand_val = randn_base[d];
        
        // Compute log_prob contribution for x
        float diff_x = (x_val - mu_val) / sigma_val;
        log_prob_x += diff_x * diff_x;
        
        // Compute proposal
        float proposal_val = x_val + step_size * rand_val;
        x_new_base[d] = proposal_val;
        
        // Compute log_prob contribution for proposal
        float diff_p = (proposal_val - mu_val) / sigma_val;
        log_prob_proposal += diff_p * diff_p;
    }
    
    // Warp reduce for both log_probs
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        log_prob_x += __shfl_down_sync(0xffffffff, log_prob_x, offset);
        log_prob_proposal += __shfl_down_sync(0xffffffff, log_prob_proposal, offset);
    }

        }
    }
}
