"""
Reference: all_to_all_single with equal splits (NCCL all_to_all family).

With identical inputs on every rank, a symmetric equal-split exchange is an identity on the
concatenated layout; we multiply by a learnable scalar so the module is non-trivial.
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_world_size, is_distributed_run, maybe_init_process_group
flat_dim = 64

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(1.25))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        (b, d) = x.shape
        assert d == flat_dim, f'expected last dim {flat_dim}, got {d}'
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunk = d // ws
        assert chunk * ws == d, 'flat_dim must be divisible by world_size'
        flat = x.reshape(b * d)
        out_flat = torch.empty_like(flat)
        sizes = [chunk * b] * ws
        dist.all_to_all_single(out_flat, flat, output_split_sizes=sizes, input_split_sizes=sizes)
        return out_flat.reshape(b, d) * self.scale

def get_inputs():
    p = popcorn_pri
    mode = p.sample_input_mode()
    dev = default_device()
    return [torch.randn(p.trial_dim(2, '_', mode=mode), flat_dim, device=dev)]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void __launch_bounds__(256) scale_vec4_kernel(const float4* __restrict__ x, float4* __restrict__ out, float scale, int64_t n4) {
    int64_t idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n4) {
        float4 v = x[idx];
        v.x *= scale; v.y *= scale; v.z *= scale; v.w *= scale;
        out[idx] = v;
    }
}

torch::Tensor scale_forward(torch::Tensor x, torch::Tensor scale) {
    auto out = torch::empty_like(x);
    int64_t n = x.numel();
    int64_t n4 = n / 4;
    const int block = 256;
    const int grid = (n4 + block - 1) / block;
    scale_vec4_kernel<<<grid, block>>>(
        reinterpret_cast<const float4*>(x.data_ptr<float>()),
        reinterpret_cast<float4*>(out.data_ptr<float>()),
        scale.item<float>(), n4);
    // handle tail if any (n % 4 != 0) - but flat_dim=64 ensures multiple of 4
    return out;
}
'''
