import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Pack tokens into expert-major order for MoE execution.

    Each token is assigned to one expert and one slot within that expert's
    buffer. The kernel writes token activations into the packed dispatch layout.
    """

    def __init__(self):
        super().__init__()

    def forward(self, token_hidden: torch.Tensor, expert_idx: torch.Tensor, slot_idx: torch.Tensor, expert_offsets: torch.Tensor) -> torch.Tensor:
        num_rows = int(expert_offsets[-1].item())
        out = torch.zeros(num_rows, token_hidden.shape[1], dtype=token_hidden.dtype, device=token_hidden.device)
        for token in range(token_hidden.shape[0]):
            expert = int(expert_idx[token].item())
            row = int(expert_offsets[expert].item() + slot_idx[token].item())
            out[row] = token_hidden[token]
        return out

template <typename scalar_t, typename expert_t, typename slot_t, typename offset_t>
__global__ void moe_dispatch_kernel(const scalar_t* __restrict__ token_hidden,
                                    const expert_t* __restrict__ expert_idx,

    // One warp copies one token row.  This keeps both the source and destination
    // row accesses coalesced while avoiding one CUDA block per token.

    int64_t expert = static_cast<int64_t>(expert_idx[token]);
    int64_t row = static_cast<int64_t>(expert_offsets[expert]) + static_cast<int64_t>(slot_idx[token]);

    const scalar_t* __restrict__ src = token_hidden + static_cast<int64_t>(token) * hidden_dim;
    scalar_t* __restrict__ dst = out + row * hidden_dim;

    #pragma unroll
    for (int col = lane; col < hidden_dim; col += 32) {
        dst[col] = src[col];
    }
}

template <typename scalar_t, typename expert_t, typename slot_t, typename offset_t>
static inline void launch_moe_dispatch(const torch::Tensor& token_hidden,
                                       const torch::Tensor& expert_idx,
                                       const torch::Tensor& slot_idx,
                                       const torch::Tensor& expert_offsets,
                                       torch::Tensor& out,
                                       int num_tokens,
                                       int hidden_dim,
                                       cudaStream_t stream) {
    constexpr int threads = 256; // 8 warps/block
    int blocks = (num_tokens + 7) / 8;
    moe_dispatch_kernel<scalar_t, expert_t, slot_t, offset_t><<<blocks, threads, 0, stream>>>(
        token_hidden.data_ptr<scalar_t>(),
        expert_idx.data_ptr<expert_t>(),
        slot_idx.data_ptr<slot_t>(),
        expert_offsets.data_ptr<offset_t>(),
        out.data_ptr<scalar_t>(),
        num_tokens,
        hidden_dim);
}

static int64_t read_last_offset(const torch::Tensor& expert_offsets) {
    TORCH_CHECK(expert_offsets.numel() > 0, "expert_offsets must be non-empty");
    int64_t n = expert_offsets.numel();
    auto st = expert_offsets.scalar_type();

    if (expert_offsets.is_cuda()) {
        c10::cuda::CUDAGuard guard(expert_offsets.device());
        if (st == at::kLong) {
            int64_t v = 0;
            C10_CUDA_CHECK(cudaMemcpy(&v, expert_offsets.data_ptr<int64_t>() + n - 1,
                                      sizeof(int64_t), cudaMemcpyDeviceToHost));
            return v;
        } else if (st == at::kInt) {
            int32_t v = 0;
            C10_CUDA_CHECK(cudaMemcpy(&v, expert_offsets.data_ptr<int32_t>() + n - 1,
                                      sizeof(int32_t), cudaMemcpyDeviceToHost));
            return static_cast<int64_t>(v);
        }
    } else {
        if (st == at::kLong) {
            return expert_offsets.data_ptr<int64_t>()[n - 1];
        } else if (st == at::kInt) {
            return static_cast<int64_t>(expert_offsets.data_ptr<int32_t>()[n - 1]);
        }
    }
    TORCH_CHECK(false, "expert_offsets must be int32 or int64");
}

torch::Tensor moe_dispatch_forward(torch::Tensor token_hidden,
                                   torch::Tensor expert_idx,
                                   torch::Tensor slot_idx,
                                   torch::Tensor expert_offsets) {
    TORCH_CHECK(token_hidden.is_cuda(), "token_hidden must be CUDA");
    TORCH_CHECK(expert_idx.is_cuda(), "expert_idx must be CUDA");
    TORCH_CHECK(slot_idx.is_cuda(), "slot_idx must be CUDA");
    TORCH_CHECK(expert_offsets.is_cuda(), "expert_offsets must be CUDA");
    TORCH_CHECK(token_hidden.dim() == 2, "token_hidden must be 2D");

    c10::cuda::CUDAGuard device_guard(token_hidden.device());

    const int num_tokens = static_cast<int>(token_hidden.size(0));
    const int hidden_dim = static_cast<int>(token_hidden.size(1));
    const int64_t num_rows = read_last_offset(expert_offsets);

    auto out = torch::empty({num_rows, static_cast<int64_t>(hidden_dim)}, token_hidden.options());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if (out.numel() > 0) {
        C10_CUDA_CHECK(cudaMemsetAsync(out.data_ptr(), 0, out.numel() * out.element_size(), stream));
    }

    auto et = expert_idx.scalar_type();
    auto st = slot_idx.scalar_type();
    auto ot = expert_offsets.scalar_type();

    AT_DISPATCH_FLOATING_TYPES_AND_HALF(token_hidden.scalar_type(), "moe_dispatch_forward", [&] {
        if (et == at::kLong && st == at::kLong && ot == at::kLong) {
            launch_moe_dispatch<scalar_t, int64_t, int64_t, int64_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kLong && st == at::kLong && ot == at::kInt) {
            launch_moe_dispatch<scalar_t, int64_t, int64_t, int32_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kLong && st == at::kInt && ot == at::kLong) {
            launch_moe_dispatch<scalar_t, int64_t, int32_t, int64_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kLong && st == at::kInt && ot == at::kInt) {
            launch_moe_dispatch<scalar_t, int64_t, int32_t, int32_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kInt && st == at::kLong && ot == at::kLong) {
            launch_moe_dispatch<scalar_t, int32_t, int64_t, int64_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kInt && st == at::kLong && ot == at::kInt) {
            launch_moe_dispatch<scalar_t, int32_t, int64_t, int32_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kInt && st == at::kInt && ot == at::kLong) {
            launch_moe_dispatch<scalar_t, int32_t, int32_t, int64_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else if (et == at::kInt && st == at::kInt && ot == at::kInt) {
            launch_moe_dispatch<scalar_t, int32_t, int32_t, int32_t>(token_hidden, expert_idx, slot_idx, expert_offsets, out, num_tokens, hidden_dim, stream);
        } else {
            TORCH_CHECK(false, "index tensors must be int32 or int64");
        }
    });

    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
