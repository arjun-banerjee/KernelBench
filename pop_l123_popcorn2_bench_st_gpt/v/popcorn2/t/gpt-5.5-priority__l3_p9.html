import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Bayesian linear regression with a conjugate Gaussian prior.
    Given features X and targets y, computes the posterior distribution
    over weights: p(w | X, y) = N(mu_post, Sigma_post) in closed form
    via the matrix inversion lemma / Cholesky decomposition.
    """

    def __init__(self, input_dim, prior_precision=1.0, noise_precision=1.0):
        super().__init__()
        self.input_dim = input_dim
        self.prior_precision = prior_precision
        self.noise_precision = noise_precision

    def forward(self, X: torch.Tensor, y: torch.Tensor, X_test: torch.Tensor) -> tuple:
        """
        Args:
            X:      (B, N, D) – training features
            y:      (B, N, 1) – training targets
            X_test: (B, M, D) – test features
        Returns:
            (pred_mean, pred_var):
                pred_mean: (B, M, 1)
                pred_var:  (B, M)
        """
        D = X.shape[-1]
        eye = torch.eye(D, device=X.device, dtype=X.dtype).unsqueeze(0)
        Lambda_post = self.prior_precision * eye + self.noise_precision * (X.transpose(-1, -2) @ X)
        L = torch.linalg.cholesky(Lambda_post)
        rhs = self.noise_precision * (X.transpose(-1, -2) @ y)
        mu_post = torch.cholesky_solve(rhs, L)
        pred_mean = X_test @ mu_post
        Sigma_post = torch.cholesky_inverse(L)
        pred_var = (X_test @ Sigma_post * X_test).sum(-1) + 1.0 / self.noise_precision
        return (pred_mean, pred_var)

    const float* Xb = X + ((size_t)b) * N * D_FIXED;
    const float* yb = y + ((size_t)b) * N;

    // Compute posterior precision matrix: prior*I + noise*X^T X.
    // One thread owns one 16x16 entry; deterministic simple fp32 accumulation.
    if (tid < D_FIXED * D_FIXED) {
        int i = tid >> 4;
        int j = tid & 15;
        float s = 0.0f;
        #pragma unroll 4
        for (int n = 0; n < N; ++n) {
            const float* row = Xb + ((size_t)n) * D_FIXED;
            s += row[i] * row[j];
        }
        A[tid] = noise * s + ((i == j) ? prior : 0.0f);
    }

    // Compute rhs = noise * X^T y.
    if (tid < D_FIXED) {
        float s = 0.0f;
        #pragma unroll 4
        for (int n = 0; n < N; ++n) {
            s += Xb[((size_t)n) * D_FIXED + tid] * yb[n];
        }
        rhs[tid] = noise * s;
    }

    __syncthreads();

    // Factor, solve, and invert the tiny 16x16 SPD system in one thread per batch.
    if (tid == 0) {
        #pragma unroll
        for (int i = 0; i < D_FIXED * D_FIXED; ++i) {
            L[i] = 0.0f;
            Sigma[i] = 0.0f;
        }

        // Lower Cholesky A = L L^T.
        for (int i = 0; i < D_FIXED; ++i) {
            for (int j = 0; j <= i; ++j) {
                float s = A[i * D_FIXED + j];
                for (int k = 0; k < j; ++k) {
                    s -= L[i * D_FIXED + k] * L[j * D_FIXED + k];
                }
                if (i == j) {
                    L[i * D_FIXED + j] = sqrtf(fmaxf(s, 1.0e-20f));
                } else {
                    L[i * D_FIXED + j] = s / L[j * D_FIXED + j];
                }
            }
        }

        // mu = A^{-1} rhs using the Cholesky factors.
        float z[D_FIXED];
        for (int i = 0; i < D_FIXED; ++i) {
            float s = rhs[i];
            for (int k = 0; k < i; ++k) s -= L[i * D_FIXED + k] * z[k];
            z[i] = s / L[i * D_FIXED + i];
        }
        for (int i = D_FIXED - 1; i >= 0; --i) {
            float s = z[i];
            for (int k = i + 1; k < D_FIXED; ++k) s -= L[k * D_FIXED + i] * mu[k];
            mu[i] = s / L[i * D_FIXED + i];
        }

        // Full inverse Sigma = A^{-1}; solve once for each basis vector.
        for (int col = 0; col < D_FIXED; ++col) {
            float zz[D_FIXED];
            float xx[D_FIXED];
            for (int i = 0; i < D_FIXED; ++i) {
                float s = (i == col) ? 1.0f : 0.0f;
                for (int k = 0; k < i; ++k) s -= L[i * D_FIXED + k] * zz[k];
                zz[i] = s / L[i * D_FIXED + i];
            }
            for (int i = D_FIXED - 1; i >= 0; --i) {
                float s = zz[i];
                for (int k = i + 1; k < D_FIXED; ++k) s -= L[k * D_FIXED + i] * xx[k];
                xx[i] = s / L[i * D_FIXED + i];
            }
            for (int i = 0; i < D_FIXED; ++i) Sigma[i * D_FIXED + col] = xx[i];
        }
    }

    __syncthreads();

    const float* Xtb = Xt + ((size_t)b) * M * D_FIXED;
    // Predictions: mean = x*mu, variance = x*Sigma*x + noise^{-1}.
    for (int m = tid; m < M; m += blockDim.x) {
        const float* x = Xtb + ((size_t)m) * D_FIXED;
        float mean = 0.0f;
        #pragma unroll
        for (int i = 0; i < D_FIXED; ++i) mean += x[i] * mu[i];

        float var = 0.0f;
        #pragma unroll
        for (int i = 0; i < D_FIXED; ++i) {
            float t = 0.0f;
            #pragma unroll
            for (int j = 0; j < D_FIXED; ++j) {
                t += x[j] * Sigma[j * D_FIXED + i];
            }
            var += t * x[i];
        }
        out_mean[((size_t)b) * M + m] = mean;
        out_var[((size_t)b) * M + m] = var + 1.0f / noise;
    }
}

                                        double prior_precision, double noise_precision) {

    const int N = (int)X.size(1);
    const int M = (int)X_test.size(1);

    auto mean = torch::empty({B, M, 1}, X.options());
    auto var = torch::empty({B, M}, X.options());

        mean.data_ptr<float>(), var.data_ptr<float>(),

    return {mean, var};
}
