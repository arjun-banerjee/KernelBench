import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Multi-head row-wise softmax over CSR edge scores.

    This extends scalar graph edge softmax to multiple attention heads while
    preserving sparse CSR layout. Each row is normalized independently for each
    head.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, edge_scores: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        out = torch.empty_like(edge_scores)
        for node in range(num_nodes):
            start = int(row_ptr[node].item())
            end = int(row_ptr[node + 1].item())
            if end > start:
                out[start:end] = torch.softmax(edge_scores[start:end], dim=0)
        return out

    if (row >= num_nodes || lane >= num_heads) return;

    index_t start = row_ptr[row];
    index_t end = row_ptr[row + 1];
    index_t deg = end - start;
    if (deg <= 0) return;

    const int h = lane;
    float m = -CUDART_INF_F;
    for (index_t e = start; e < end; ++e) {
        float v = static_cast<float>(scores[(long long)e * num_heads + h]);
        m = fmaxf(m, v);
    }

    float s = 0.0f;
    for (index_t e = start; e < end; ++e) {
        s += expf(static_cast<float>(scores[(long long)e * num_heads + h]) - m);
    }
    float inv_s = 1.0f / s;

    for (index_t e = start; e < end; ++e) {
        float v = expf(static_cast<float>(scores[(long long)e * num_heads + h]) - m) * inv_s;
        out[(long long)e * num_heads + h] = static_cast<scalar_t>(v);
    }
}
