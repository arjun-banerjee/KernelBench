import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Sparse message passing with CSR adjacency.

    For each destination node, aggregate weighted source-node features from the
    incoming edges in its CSR row. This is a core primitive in sparse graph
    neural network layers and graph convolution operators.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, col_idx: torch.Tensor, edge_weight: torch.Tensor, node_feat: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        feat_dim = node_feat.shape[1]
        out = torch.zeros(num_nodes, feat_dim, dtype=node_feat.dtype, device=node_feat.device)
        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end > start:
                src = col_idx[start:end].long()
                weights = edge_weight[start:end].unsqueeze(-1)
                out[dst] = (node_feat[src] * weights).sum(dim=0)
        return out

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor csr_spmm_cuda(torch::Tensor row_ptr, torch::Tensor col_idx, torch::Tensor edge_weight, torch::Tensor node_feat);

torch::Tensor csr_spmm(torch::Tensor row_ptr, torch::Tensor col_idx, torch::Tensor edge_weight, torch::Tensor node_feat) {
    return csr_spmm_cuda(row_ptr, col_idx, edge_weight, node_feat);
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cstdint>

template <typename row_t, typename col_t>
__global__ void csr_spmm_kernel(const row_t* __restrict__ row_ptr,
                                const col_t* __restrict__ col_idx,
                                const float* __restrict__ edge_weight,
                                const float* __restrict__ node_feat,
                                float* __restrict__ out,
                                int num_nodes,
                                int feat_dim) {
    int dst = blockIdx.x;
    int tid = threadIdx.x;
    if (dst >= num_nodes) return;

    row_t start = row_ptr[dst];
    row_t end = row_ptr[dst + 1];

    for (int f = tid; f < feat_dim; f += blockDim.x) {
        float acc = 0.0f;
        #pragma unroll 1
        for (row_t e = start; e < end; ++e) {
            col_t src = col_idx[e];
            acc += edge_weight[e] * node_feat[(int64_t)src * feat_dim + f];
        }
        out[(int64_t)dst * feat_dim + f] = acc;
    }
}

template <typename row_t, typename col_t>
static inline void launch_csr(torch::Tensor row_ptr, torch::Tensor col_idx, torch::Tensor edge_weight,
                              torch::Tensor node_feat, torch::Tensor out, int num_nodes, int feat_dim) {
    const int threads = 128;
    const dim3 blocks(num_nodes);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    csr_spmm_kernel<row_t, col_t><<<blocks, threads, 0, stream>>>(
        row_ptr.data_ptr<row_t>(), col_idx.data_ptr<col_t>(),
        edge_weight.data_ptr<float>(), node_feat.data_ptr<float>(), out.data_ptr<float>(),
        num_nodes, feat_dim);
}

torch::Tensor csr_spmm_cuda(torch::Tensor row_ptr, torch::Tensor col_idx, torch::Tensor edge_weight, torch::Tensor node_feat) {
    const int num_nodes = (int)row_ptr.numel() - 1;
    const int feat_dim = (int)node_feat.size(1);
    auto out = torch::empty({num_nodes, feat_dim}, node_feat.options());

    if (num_nodes <= 0 || feat_dim <= 0) return out;

    if (row_ptr.scalar_type() == at::kLong) {
        if (col_idx.scalar_type() == at::kLong) {
            launch_csr<int64_t, int64_t>(row_ptr, col_idx, edge_weight, node_feat, out, num_nodes, feat_dim);
        } else {
            launch_csr<int64_t, int32_t>(row_ptr, col_idx, edge_weight, node_feat, out, num_nodes, feat_dim);
        }
    } else {
        if (col_idx.scalar_type() == at::kLong) {
            launch_csr<int32_t, int64_t>(row_ptr, col_idx, edge_weight, node_feat, out, num_nodes, feat_dim);
        } else {
            launch_csr<int32_t, int32_t>(row_ptr, col_idx, edge_weight, node_feat, out, num_nodes, feat_dim);
        }
    }
    return out;
}
"""
