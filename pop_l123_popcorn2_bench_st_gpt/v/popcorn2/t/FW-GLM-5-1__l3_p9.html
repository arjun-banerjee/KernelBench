class Model(nn.Module):

    Bayesian linear regression with a conjugate Gaussian prior.
    Given features X and targets y, computes the posterior distribution
    over weights: p(w | X, y) = N(mu_post, Sigma_post) in closed form
    via the matrix inversion lemma / Cholesky decomposition.
    """

        super().__init__()

        self.prior_precision = prior_precision
        self.noise_precision = noise_precision

    def forward(self, X: torch.Tensor, y: torch.Tensor, X_test: torch.Tensor) -> tuple:
        """
        Args:
            X:      (B, N, D) – training features
            y:      (B, N, 1) – training targets
            X_test: (B, M, D) – test features
        Returns:
            (pred_mean, pred_var):
                pred_mean: (B, M, 1)
                pred_var:  (B, M)
        """
        D = X.shape[-1]
        eye = torch.eye(D, device=X.device, dtype=X.dtype).unsqueeze(0)
        Lambda_post = self.prior_precision * eye + self.noise_precision * (X.transpose(-1, -2) @ X)

        rhs = self.noise_precision * (X.transpose(-1, -2) @ y)
        mu_post = torch.cholesky_solve(rhs, L)
        pred_mean = X_test @ mu_post
        Sigma_post = torch.cholesky_inverse(L)
        pred_var = (X_test @ Sigma_post * X_test).sum(-1) + 1.0 / self.noise_precision
        return (pred_mean, pred_var)

    }
    __syncthreads();
    
    // Forward substitution column by column

// Backward substitution: solve L^T @ x = b for lower triangular L

// Cholesky inverse: compute inverse of L where Lambda = L @ L^T
// inv(Lambda) = inv(L^T) @ inv(L)
__global__ void cholesky_inverse_kernel(const float* __restrict__ L, float* __restrict__ inv_out, int n, int batch_size) {
    int batch = blockIdx.x;
    if (batch >= batch_size) return;
    
    __shared__ float L_shared[256];   // n x n, max 16x16
    __shared__ float inv_shared[256];
    
    // Load L into shared memory and initialize inv to identity
    for (int i = threadIdx.x; i < n * n; i += blockDim.x) {
        L_shared[i] = L[batch * n * n + i];
        inv_shared[i] = 0.0f;
    }
    __syncthreads();
    
    // Compute inv(L) column by column
    // inv(L) is also lower triangular
    for (int j = 0; j < n; j++) {
        // Diagonal element: inv[j,j] = 1 / L[j,j]
        if (threadIdx.x == 0) {
            inv_shared[j * n + j] = 1.0f / L_shared[j * n + j];
        }
        __syncthreads();
        
        // Off-diagonal: inv[i,j] = -sum_{k=j}^{i-1} L[i,k] * inv[k,j] / L[i,i]
        for (int i = threadIdx.x; i < n; i += blockDim.x) {
            if (i > j) {
                float sum = 0.0f;
                for (int k = j; k < i; k++) {
                    sum += L_shared[i * n + k] * inv_shared[k * n + j];
                }
                inv_shared[i * n + j] = -sum / L_shared[i * n + i];
            }
        }
        __syncthreads();
    }
    
    // Compute Sigma = inv(L)^T @ inv(L)
    // This gives the inverse of Lambda = L @ L^T
    for (int i = threadIdx.x; i < n * n; i += blockDim.x) {
        int row = i / n;
        int col = i % n;
        if (row >= col) {  // Lower triangular part
            float sum = 0.0f;
            for (int k = row; k < n; k++) {  // inv(L) is lower triangular
                sum += inv_shared[k * n + row] * inv_shared[k * n + col];
            }
            inv_out[batch * n * n + i] = sum;
        } else {
            inv_out[batch * n * n + i] = 0.0f;  // Will be filled by symmetry
        }
    }
    __syncthreads();
    
    // Fill upper triangular part by symmetry
    for (int i = threadIdx.x; i < n * n; i += blockDim.x) {
        int row = i / n;
        int col = i % n;
        if (row < col) {
            inv_out[batch * n * n + i] = inv_out[batch * n * n + col * n + row];
        }
    }
}

// Prediction mean and variance computation
__global__ void predict_kernel(
    const float* __restrict__ X_test,
    const float* __restrict__ mu_post,
    const float* __restrict__ Sigma,
    float* __restrict__ pred_mean,
    float* __restrict__ pred_var,
    int M, int D, int batch_size,
    float noise_var
) {
    int batch = blockIdx.x;
    if (batch >= batch_size) return;
    
    __shared__ float mu_shared[16];     // Max D = 16
    __shared__ float Sigma_shared[256]; // Max D*D = 256
    __shared__ float Xtest_row[16];     // One row of X_test
    
    // Load mu_post into shared memory
    if (threadIdx.x < D) {
        mu_shared[threadIdx.x] = mu_post[batch * D + threadIdx.x];
    }
    
    // Load Sigma into shared memory
    for (int i = threadIdx.x; i < D * D; i += blockDim.x) {
        Sigma_shared[i] = Sigma[batch * D * D + i];
    }
    __syncthreads();
    
    // Compute predictions for each test point
    for (int m = 0; m < M; m++) {
        // Load one row of X_test
        if (threadIdx.x < D) {
            Xtest_row[threadIdx.x] = X_test[batch * M * D + m * D + threadIdx.x];
        }
        __syncthreads();
        
        // Compute pred_mean = X_test[m] @ mu_post
        float mean_val = 0.0f;
        if (threadIdx.x == 0) {
            for (int d = 0; d < D; d++) {
                mean_val += Xtest_row[d] * mu_shared[d];
            }
            pred_mean[batch * M + m] = mean_val;
        }
        
        // Compute pred_var = X_test[m] @ Sigma @ X_test[m]^T + noise_var
        // = sum_d1 sum_d2 X[d1] * Sigma[d1,d2] * X[d2]
        float var_val = 0.0f;
        for (int d1 = threadIdx.x; d1 < D; d1 += blockDim.x) {
            float row_sum = 0.0f;
            for (int d2 = 0; d2 < D; d2++) {
                row_sum += Sigma_shared[d1 * D + d2] * Xtest_row[d2];
            }
            var_val += Xtest_row[d1] * row_sum;
        }
        
        // Reduce variance across threads
        __shared__ float var_shared[256];
        var_shared[threadIdx.x] = var_val;
        __syncthreads();
        
        // Simple reduction
        for (int s = blockDim.x / 2; s > 0; s >>= 1) {
            if (threadIdx.x < s) {
                var_shared[threadIdx.x] += var_shared[threadIdx.x + s];
            }
            __syncthreads();
        }
        
        if (threadIdx.x == 0) {
            pred_var[batch * M + m] = var_shared[0] + noise_var;
        }
        __syncthreads();
    }
}

        B
    );
    cudaDeviceSynchronize();
    
    // Cholesky decomposition
    cholesky_kernel<<<B, BLOCK_SIZE>>>(Lambda_post, L, D, B);
    cudaDeviceSynchronize();
    
    // Forward solve: L @ temp = rhs
    forward_solve_kernel<<<B, BLOCK_SIZE>>>(L, rhs, temp, D, B);
    cudaDeviceSynchronize();
    
    // Backward solve: L^T @ mu_post = temp
    backward_solve_kernel<<<B, BLOCK_SIZE>>>(L, temp, mu_post, D, B);
    cudaDeviceSynchronize();
    
    // Cholesky inverse: Sigma = inv(L^T) @ inv(L)
    cholesky_inverse_kernel<<<B, BLOCK_SIZE>>>(L, Sigma, D, B);
    cudaDeviceSynchronize();
    
    // Prediction
    float noise_var = 1.0f / noise_precision;
    predict_kernel<<<B, BLOCK_SIZE>>>(X_test, mu_post, Sigma, pred_mean, pred_var, M, D, B, noise_var);
    cudaDeviceSynchronize();
}
"""

    name='bayesian_linear_regression_ext',
    cpp_sources=cpp_source,

class ModelNew(nn.Module):
    """
    Bayesian linear regression with a conjugate Gaussian prior.
    Custom CUDA implementation for improved performance.
    """

    def __init__(self, input_dim, prior_precision=1.0, noise_precision=1.0):
        super().__init__()
        self.input_dim = input_dim
        self.prior_precision = prior_precision
        self.noise_precision = noise_precision

    def forward(self, X: torch.Tensor, y: torch.Tensor, X_test: torch.Tensor) -> tuple:
        """
        Args:
            X:      (B, N, D) - training features
            y:      (B, N, 1) - training targets
            X_test: (B, M, D) - test features
        Returns:
            (pred_mean, pred_var):
                pred_mean: (B, M, 1)
                pred_var:  (B, M)
        """
        B = X.shape[0]
        N = X.shape[1]
        D = X.shape[2]
        M = X_test.shape[1]
        
        # Ensure contiguous float32 tensors
        X = X.contiguous().float()
        y = y.contiguous().float()
        X_test = X_test.contiguous().float()
        
        # Allocate output tensors
        pred_mean = torch.empty(B, M, 1, dtype=torch.float32, device=X.device)
        pred_var = torch.empty(B, M, dtype=torch.float32, device=X.device)
        
        # Allocate intermediate tensors
        Lambda_post = torch.empty(B, D, D, dtype=torch.float32, device=X.device)
        L = torch.empty(B, D, D, dtype=torch.float32, device=X.device)
        rhs = torch.empty(B, D, 1, dtype=torch.float32, device=X.device)
        mu_post = torch.empty(B, D, 1, dtype=torch.float32, device=X.device)
        Sigma = torch.empty(B, D, D, dtype=torch.float32, device=X.device)
        temp = torch.empty(B, D, 1, dtype=torch.float32, device=X.device)
        
        # Call CUDA kernel
        bayesian_ext.bayesian_linear_regression_cuda(
            X.data_ptr(),
            y.data_ptr(),
            X_test.data_ptr(),
            pred_mean.data_ptr(),
            pred_var.data_ptr(),
            B, N, D, M,
            self.prior_precision,
            self.noise_precision,
            Lambda_post.data_ptr(),
            L.data_ptr(),
            rhs.data_ptr(),
            mu_post.data_ptr(),
            Sigma.data_ptr(),
            temp.data_ptr()
        )
        
        return (pred_mean, pred_var)
