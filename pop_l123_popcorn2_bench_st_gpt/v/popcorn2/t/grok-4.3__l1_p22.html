import torch
import torch.nn as nn

class Model(nn.Module):
    """
    SE(3)-invariant linear layer.  Maps node features that are augmented
    with 3-D coordinate information through a linear transform that
    respects rotational and translational invariance.  The layer operates
    on (scalar_features, vector_features) pairs where scalar features
    transform trivially and vector features transform as 3-D vectors.
    Used in equivariant GNNs (EGNN, PaiNN, TFN).
    """

    def __init__(self, scalar_in, scalar_out, vector_in, vector_out):
        super().__init__()
        self.scalar_linear = nn.Linear(scalar_in + vector_in, scalar_out)
        self.vector_weight = nn.Parameter(torch.randn(vector_out, vector_in))
        nn.init.xavier_uniform_(self.vector_weight)
        self.vector_gate = nn.Linear(scalar_in + vector_in, vector_out)

    def forward(self, scalar_feat: torch.Tensor, vector_feat: torch.Tensor) -> tuple:
        """
        Args:
            scalar_feat: (B, N, scalar_in) – invariant node features
            vector_feat: (B, N, vector_in, 3) – equivariant vector features
        Returns:
            (scalar_out, vector_out):
                scalar_out: (B, N, scalar_out)
                vector_out: (B, N, vector_out, 3)
        """
        vec_norm = vector_feat.norm(dim=-1)
        combined = torch.cat([scalar_feat, vec_norm], dim=-1)
        s_out = self.scalar_linear(combined)
        v_out = torch.einsum('oi,bnid->bnod', self.vector_weight, vector_feat)
        gate = torch.sigmoid(self.vector_gate(combined)).unsqueeze(-1)
        v_out = gate * v_out
        return (s_out, v_out)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void se3_inv_linear_kernel(
    const float* __restrict__ scalar_feat,
    const float* __restrict__ vector_feat,
    const float* __restrict__ scalar_weight,
    const float* __restrict__ scalar_bias,
    const float* __restrict__ vector_weight,
    const float* __restrict__ gate_weight,
    const float* __restrict__ gate_bias,
    float* __restrict__ s_out,
    float* __restrict__ v_out,
    int B, int N, int scalar_in, int scalar_out, int vector_in, int vector_out) {

    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * N;
    if (idx >= total) return;

    int b = idx / N;
    int n = idx % N;

    const float* v_ptr = vector_feat + (b * N + n) * vector_in * 3;
    float norms[32];
    for (int i = 0; i < vector_in; i++) {
        float x = v_ptr[i*3 + 0];
        float y = v_ptr[i*3 + 1];
        float z = v_ptr[i*3 + 2];
        norms[i] = sqrtf(x*x + y*y + z*z);
    }

    int comb_dim = scalar_in + vector_in;

    for (int o = 0; o < scalar_out; o++) {
        float val = scalar_bias[o];
        for (int i = 0; i < scalar_in; i++) {
            val += scalar_feat[(b*N + n)*scalar_in + i] * scalar_weight[o * comb_dim + i];
        }
        for (int i = 0; i < vector_in; i++) {
            val += norms[i] * scalar_weight[o * comb_dim + scalar_in + i];
        }
        s_out[(b*N + n)*scalar_out + o] = val;
    }

    float vtemp[32*3];
    for (int o = 0; o < vector_out; o++) {
        for (int d = 0; d < 3; d++) {
            float val = 0.0f;
            for (int i = 0; i < vector_in; i++) {
                val += vector_weight[o * vector_in + i] * v_ptr[i*3 + d];
            }
            vtemp[o*3 + d] = val;
        }
    }

    for (int o = 0; o < vector_out; o++) {
        float val = gate_bias[o];
        for (int i = 0; i < scalar_in; i++) {
            val += scalar_feat[(b*N + n)*scalar_in + i] * gate_weight[o * comb_dim + i];
        }
        for (int i = 0; i < vector_in; i++) {
            val += norms[i] * gate_weight[o * comb_dim + scalar_in + i];
        }
        float g = 1.0f / (1.0f + expf(-val));
        for (int d = 0; d < 3; d++) {
            v_out[(b*N + n)*vector_out*3 + o*3 + d] = g * vtemp[o*3 + d];
        }
    }
}

    torch::Tensor scalar_feat,
    torch::Tensor vector_feat,
    torch::Tensor scalar_weight,
    torch::Tensor scalar_bias,
    torch::Tensor vector_weight,
    torch::Tensor gate_weight,
    torch::Tensor gate_bias) {

    int B = scalar_feat.size(0);
    int N = scalar_feat.size(1);
