import torch
import torch.nn as nn

class Model(nn.Module):
    """
    RBF (Squared Exponential) covariance matrix computation for Gaussian
    Processes.  Given two sets of input points, computes the full
    kernel matrix K[i,j] = sigma^2 * exp(-||x_i - x_j||^2 / (2 * l^2)).
    The dominant cost in GP inference and a natural target for fusion.
    """

    def __init__(self, input_dim, lengthscale=1.0, variance=1.0):
        super().__init__()
        self.log_lengthscale = nn.Parameter(torch.tensor(lengthscale).log())
        self.log_variance = nn.Parameter(torch.tensor(variance).log())

    def forward(self, X1: torch.Tensor, X2: torch.Tensor) -> torch.Tensor:
        """
        Args:
            X1: (B, N, D) – first set of inputs
            X2: (B, M, D) – second set of inputs
        Returns:
            K: (B, N, M) – covariance matrix
        """
        l2 = self.log_lengthscale.exp() ** 2
        var = self.log_variance.exp()
        X1_sq = (X1 ** 2).sum(-1, keepdim=True)
        X2_sq = (X2 ** 2).sum(-1, keepdim=True)
        dist_sq = X1_sq - 2.0 * X1 @ X2.transpose(-1, -2) + X2_sq.transpose(-1, -2)
        dist_sq = dist_sq.clamp(min=0.0)
        return var * torch.exp(-0.5 * dist_sq / l2)

class ModelNew(nn.Module):
    """
    Optimized CUDA kernel for RBF (Squared Exponential) covariance matrix.
    
    Uses a tiled 2D thread block approach where each thread computes one 
    output element K[b,i,j]. The kernel fuses norm computation, distance
    calculation, and exponential application into a single pass.
    
    Memory access patterns are optimized:
    - X1 accessed with stride D (coalesced within warps across i dimension)
    - X2 accessed with stride D (coalesced within warps across j dimension)
    - Output written with stride M (coalesced)
    """

    def __init__(self, input_dim, lengthscale=1.0, variance=1.0):
        super().__init__()
        self.log_lengthscale = nn.Parameter(torch.tensor(lengthscale).log())
        self.log_variance = nn.Parameter(torch.tensor(variance).log())
        self._input_dim = input_dim

    def forward(self, X1: torch.Tensor, X2: torch.Tensor) -> torch.Tensor:
        """
        Args:
            X1: (B, N, D) - first set of inputs
            X2: (B, M, D) - second set of inputs
        Returns:
            K: (B, N, M) - covariance matrix
        """
        # Ensure inputs are contiguous float32 on GPU
        X1_c = X1.contiguous()
        X2_c = X2.contiguous()
        
        if X1_c.dtype != torch.float32:
            X1_c = X1_c.float()
        if X2_c.dtype != torch.float32:
            X2_c = X2_c.float()
        
        assert X1_c.is_cuda and X2_c.is_cuda, "Inputs must be on CUDA device"
        
        batch_size, N, D = X1_c.shape
        _, M, _ = X2_c.shape
