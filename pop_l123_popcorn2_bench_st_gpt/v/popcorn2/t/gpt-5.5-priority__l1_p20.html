import torch
import torch.nn as nn
import math

class Model(nn.Module):
    """
    Rotary Position Embedding (RoPE) applied to protein residue sequences.
    Used in ESM-2 and similar protein language models.  Encodes absolute
    position via rotation of query/key pairs in 2-D subspaces, preserving
    relative-position information in dot-product attention.
    """

    def __init__(self, dim, max_seq_len=2048, base=10000.0):
        super().__init__()
        inv_freq = 1.0 / base ** (torch.arange(0, dim, 2).float() / dim)
        self.register_buffer('inv_freq', inv_freq)
        self.max_seq_len = max_seq_len
        self.dim = dim

    def _build_cos_sin(self, seq_len: int, device: torch.device):
        t = torch.arange(seq_len, device=device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        emb = torch.cat([freqs, freqs], dim=-1)
        return (emb.cos(), emb.sin())

    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        (x1, x2) = x.chunk(2, dim=-1)
        return torch.cat([-x2, x1], dim=-1)

    def forward(self, q: torch.Tensor, k: torch.Tensor) -> tuple:
        """
        Args:
            q: (B, H, N, D)  – queries
            k: (B, H, N, D)  – keys
        Returns:
            (q_rot, k_rot) each (B, H, N, D) with RoPE applied
        """
        N = q.shape[2]
        (cos, sin) = self._build_cos_sin(N, q.device)
        cos = cos.unsqueeze(0).unsqueeze(0)
        sin = sin.unsqueeze(0).unsqueeze(0)
        q_rot = q * cos + self._rotate_half(q) * sin
        k_rot = k * cos + self._rotate_half(k) * sin
        return (q_rot, k_rot)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r'''
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> rope_forward(torch::Tensor q,
                                       torch::Tensor k,
                                       torch::Tensor cos_cached,
                                       torch::Tensor sin_cached,
                                       torch::Tensor inv_freq);
'''

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

__global__ void rope_cached_kernel(const float* __restrict__ q,
                                   const float* __restrict__ k,
                                   const float* __restrict__ cos_t,
                                   const float* __restrict__ sin_t,
                                   const float* __restrict__ inv_freq,
                                   float* __restrict__ q_out,
                                   float* __restrict__ k_out,
                                   long long total_pairs,
                                   int D,
                                   int half,
                                   int N,
                                   int max_seq) {
    long long idx = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total_pairs) return;

    int i = (int)(idx % half);
    long long token = idx / half;          // flattened (B,H,N) index
    int n = (int)(token % N);
    long long base = token * (long long)D;

    float c, s;
    if (n < max_seq) {
        long long cs_idx = (long long)n * half + i;
        c = cos_t[cs_idx];
        s = sin_t[cs_idx];
    } else {
        float angle = (float)n * inv_freq[i];
        sincosf(angle, &s, &c);
    }

    long long off0 = base + i;
    long long off1 = off0 + half;

    float q0 = q[off0];
    float q1 = q[off1];
    float k0 = k[off0];
    float k1 = k[off1];

    q_out[off0] = q0 * c - q1 * s;
    q_out[off1] = q1 * c + q0 * s;
    k_out[off0] = k0 * c - k1 * s;
    k_out[off1] = k1 * c + k0 * s;
}

std::vector<torch::Tensor> rope_forward(torch::Tensor q,
                                       torch::Tensor k,
                                       torch::Tensor cos_cached,
                                       torch::Tensor sin_cached,
                                       torch::Tensor inv_freq) {
    auto q_out = torch::empty_like(q);
    auto k_out = torch::empty_like(k);

    const int D = (int)q.size(3);
    const int half = D >> 1;
    const int N = (int)q.size(2);
    const int max_seq = (int)cos_cached.size(0);
    const long long tokens = q.numel() / D;
    const long long total_pairs = tokens * (long long)half;

    const int threads = 256;
    const int blocks = (int)((total_pairs + threads - 1) / threads);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    rope_cached_kernel<<<blocks, threads, 0, stream>>>(
        q.data_ptr<float>(), k.data_ptr<float>(),
        cos_cached.data_ptr<float>(), sin_cached.data_ptr<float>(), inv_freq.data_ptr<float>(),
        q_out.data_ptr<float>(), k_out.data_ptr<float>(),
        total_pairs, D, half, N, max_seq);

    return {q_out, k_out};
}
'''

_rope_ext = load_inline(
    name='rope_popcorn2_cached_ext',
    cpp_sources=_cpp_src,
    cuda_sources=_cuda_src,
    functions=['rope_forward'],
    extra_cflags=['-O3'],
    extra_cuda_cflags=['-O3'],
    verbose=False,
)


class ModelNew(nn.Module):
    def __init__(self, dim, max_seq_len=2048, base=10000.0):
        super().__init__()
        inv_freq = 1.0 / (base ** (torch.arange(0, dim, 2).float() / dim))
        self.register_buffer('inv_freq', inv_freq)
        self.max_seq_len = max_seq_len
        self.dim = dim
        t = torch.arange(max_seq_len, dtype=inv_freq.dtype)
        freqs = torch.outer(t, inv_freq)
        self.register_buffer('cos_cached', freqs.cos())
        self.register_buffer('sin_cached', freqs.sin())

    def forward(self, q: torch.Tensor, k: torch.Tensor) -> tuple:
        # Buffers normally move with the module; this handles harnesses that only move inputs.
        if self.cos_cached.device != q.device:
            self.cos_cached = self.cos_cached.to(device=q.device)
            self.sin_cached = self.sin_cached.to(device=q.device)
            self.inv_freq = self.inv_freq.to(device=q.device)
        outs = _rope_ext.rope_forward(q, k, self.cos_cached, self.sin_cached, self.inv_freq)
        return (outs[0], outs[1])
