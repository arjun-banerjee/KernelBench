import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Full Mamba block: RMSNorm -> linear expansion into two branches -> causal depthwise Conv1d ->
    SiLU -> selective scan (input-dependent SSM) -> elementwise gate -> linear projection -> residual.
    This is the complete Mamba architecture block as described in Gu & Dao (2023), used in Mamba,
    Caduceus (bidirectional DNA), and various protein/genomic language models.

    Args:
        d_model (int): Model dimension.
        d_state (int): SSM state dimension (N).
        d_conv (int): Causal convolution kernel size.
        expand (int): Expansion factor for inner dimension.
    """

    def __init__(self, d_model: int, d_state: int, d_conv: int, expand: int):
        super(Model, self).__init__()
        self.d_model = d_model
        self.d_inner = d_model * expand
        self.d_state = d_state
        self.d_conv = d_conv
        self.norm = nn.RMSNorm(d_model)
        self.in_proj = nn.Linear(d_model, self.d_inner * 2, bias=False)
        self.conv1d = nn.Conv1d(self.d_inner, self.d_inner, d_conv, padding=d_conv - 1, groups=self.d_inner, bias=True)
        self.activation = nn.SiLU()
        self.A_log = nn.Parameter(torch.randn(self.d_inner, d_state))
        self.proj_B = nn.Linear(self.d_inner, d_state, bias=False)
        self.proj_C = nn.Linear(self.d_inner, d_state, bias=False)
        self.proj_delta = nn.Linear(self.d_inner, self.d_inner, bias=True)
        self.D = nn.Parameter(torch.ones(self.d_inner))
        self.out_proj = nn.Linear(self.d_inner, d_model, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        (B, L, D) = x.shape
        residual = x
        x = self.norm(x)
        xz = self.in_proj(x)
        (x_branch, z) = xz.chunk(2, dim=-1)
        x_branch = x_branch.transpose(1, 2)
        x_branch = self.conv1d(x_branch)[..., :L]
        x_branch = x_branch.transpose(1, 2)
        x_branch = self.activation(x_branch)
        N = self.d_state
        delta = torch.nn.functional.softplus(self.proj_delta(x_branch))
        B_t = self.proj_B(x_branch)
        C_t = self.proj_C(x_branch)
        A = -torch.exp(self.A_log)
        dA = torch.exp(delta.unsqueeze(-1) * A.unsqueeze(0).unsqueeze(0))
        dB = delta.unsqueeze(-1) * B_t.unsqueeze(2)
        x_db = x_branch.unsqueeze(-1) * dB
        h = torch.zeros(B, self.d_inner, N, device=x.device, dtype=x.dtype)
        outputs = []
        for t in range(L):
            h = dA[:, t] * h + x_db[:, t]
            y_t = (h * C_t[:, t].unsqueeze(1)).sum(dim=-1)
            outputs.append(y_t)
        y = torch.stack(outputs, dim=1)
        y = y + x_branch * self.D.unsqueeze(0).unsqueeze(0)
        z = self.activation(z)
        y = y * z
        y = self.out_proj(y)
        y = y + residual
        return y

        scalar_t sum_sq = scalar_t(0.0);
        for (int d = tid; d < D; d += blockDim.x) {
            scalar_t x_val = x_input[(batch_idx * L + t) * D + d];
            sum_sq += x_val * x_val;
        }
        
        shared_buf[tid] = sum_sq;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) shared_buf[tid] += shared_buf[tid + stride];
            __syncthreads();
        }
        
        scalar_t rms_norm = rsqrt(shared_buf[0] / D + norm_eps);
        
        // Compute in_proj contributions
        scalar_t x_branch_contrib = scalar_t(0.0);
        scalar_t z_contrib = scalar_t(0.0);
        for (int d = tid; d < D; d += blockDim.x) {
            scalar_t x_val = x_input[(batch_idx * L + t) * D + d];
            scalar_t x_norm = x_val * rms_norm * norm_weight[d];
            x_branch_contrib += x_norm * in_proj_weight[inner_idx * D + d];
            z_contrib += x_norm * in_proj_weight[(d_inner + inner_idx) * D + d];
        }
        
        shared_buf[tid] = x_branch_contrib;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) shared_buf[tid] += shared_buf[tid + stride];
            __syncthreads();
        }
        scalar_t x_branch_val = shared_buf[0];
        
        shared_buf[tid] = z_contrib;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) shared_buf[tid] += shared_buf[tid + stride];
            __syncthreads();
        }
        scalar_t z_val = shared_buf[0];
        
        // Conv1d (causal)
        for (int i = d_conv - 2; i > 0; i--) {
            conv_history[i] = conv_history[i-1];
        }
        conv_history[0] = x_branch_val;
        
        scalar_t conv_out = conv_b;
        for (int i = 0; i < d_conv; i++) {
            conv_out += conv_w[i] * conv_history[i];
        }
        
        // SiLU
        scalar_t x_silu = silu(conv_out);
        
        // Project delta (needs reduction)
        scalar_t delta_contrib = scalar_t(0.0);
        for (int d = tid; d < d_inner; d += blockDim.x) {
            delta_contrib += x_silu * proj_delta_weight[d * d_inner + inner_idx];
        }
        shared_buf[tid] = delta_contrib;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) shared_buf[tid] += shared_buf[tid + stride];
            __syncthreads();
        }
        scalar_t delta_val = softplus(shared_buf[0] + proj_delta_bias[inner_idx]);
        
        // Project B and C for this thread's state
        scalar_t B_t = scalar_t(0.0), C_t = scalar_t(0.0);
        if (tid < N) {
            B_t = x_silu * proj_B_weight[inner_idx * N + tid];
            C_t = x_silu * proj_C_weight[inner_idx * N + tid];
        }
        
        // SSM update
        if (tid < N) {
            scalar_t dA = exp(delta_val * A_val);
            scalar_t dB = delta_val * B_t;
            scalar_t x_db = x_silu * dB;
            h_val = dA * h_val + x_db;
        }
        
        // Compute y_t = sum(h * C_t) across state dimension
        scalar_t y_contrib = (tid < N) ? h_val * C_t : scalar_t(0.0);
        shared_buf[tid] = y_contrib;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) shared_buf[tid] += shared_buf[tid + stride];
            __syncthreads();
        }
        
        if (tid == 0) {
            scalar_t y_t = shared_buf[0] + x_silu * D_val;
            scalar_t z_act = silu(z_val);
            scalar_t gated = y_t * z_act;
            intermediate[(batch_idx * L + t) * d_inner + inner_idx] = gated;
        }
    }
}

// Output projection kernel
template<typename scalar_t>
__global__ void out_proj_kernel(
    const scalar_t* __restrict__ intermediate,
    const scalar_t* __restrict__ out_proj_weight,
    scalar_t* __restrict__ output,
    const scalar_t* __restrict__ residual,
    int B, int L, int D, int d_inner) {
    
    const int batch_idx = blockIdx.y;
    const int t_idx = blockIdx.x;
    const int d_idx = threadIdx.x;
    
    if (batch_idx >= B || t_idx >= L || d_idx >= D) return;
    
    scalar_t sum = scalar_t(0.0);
    for (int i = 0; i < d_inner; i++) {
        sum += intermediate[(batch_idx * L + t_idx) * d_inner + i] * out_proj_weight[i * D + d_idx];
    }
    
    int out_idx = (batch_idx * L + t_idx) * D + d_idx;
    output[out_idx] = sum + residual[out_idx];
}

class ModelNew(nn.Module):
    def __init__(self, d_model: int, d_state: int, d_conv: int, expand: int):
        super(ModelNew, self).__init__()
        self.d_model = d_model
        self.d_inner = d_model * expand
        self.d_state = d_state
        self.d_conv = d_conv
        
        # Initialize parameters matching the reference
        self.norm = nn.RMSNorm(d_model)
        self.in_proj = nn.Linear(d_model, self.d_inner * 2, bias=False)
        self.conv1d = nn.Conv1d(self.d_inner, self.d_inner, d_conv, 
                                padding=d_conv - 1, groups=self.d_inner, bias=True)
        self.A_log = nn.Parameter(torch.randn(self.d_inner, d_state))
        self.proj_B = nn.Linear(self.d_inner, d_state, bias=False)
        self.proj_C = nn.Linear(self.d_inner, d_state, bias=False)
        self.proj_delta = nn.Linear(self.d_inner, self.d_inner, bias=True)
        self.D = nn.Parameter(torch.ones(self.d_inner))
        self.out_proj = nn.Linear(self.d_inner, d_model, bias=False)
        
        self.extension = mamba_extension

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Ensure input is contiguous
        x = x.contiguous()
        
        # Extract parameters in the right format
        # norm weights
        norm_weight = self.norm.weight.contiguous()
        
        # in_proj weight: [d_inner*2, d_model] -> we need [d_model, d_inner*2] for the kernel
        in_proj_weight = self.in_proj.weight.t().contiguous()
        
        # conv1d weight: [d_inner, 1, d_conv] -> [d_inner, d_conv]
        conv_weight = self.conv1d.weight.squeeze(1).contiguous()
        conv_bias = self.conv1d.bias.contiguous()
        
        # A_log: [d_inner, d_state]
        A_log = self.A_log.contiguous()
        
        # proj_B weight: [d_state, d_inner] -> we need [d_inner, d_state]
        proj_B_weight = self.proj_B.weight.t().contiguous()
        
        # proj_C weight: [d_state, d_inner] -> we need [d_inner, d_state]
        proj_C_weight = self.proj_C.weight.t().contiguous()
        
        # proj_delta weight: [d_inner, d_inner]
        proj_delta_weight = self.proj_delta.weight.t().contiguous()
        proj_delta_bias = self.proj_delta.bias.contiguous()
        
        # D: [d_inner]
        D_param = self.D.contiguous()
        
        # out_proj weight: [d_model, d_inner] -> we need [d_inner, d_model]
        out_proj_weight = self.out_proj.weight.t().contiguous()
        
        norm_eps = self.norm.eps
        
        result = self.extension.mamba_forward_cuda(
            x, norm_weight, in_proj_weight,
            conv_weight, conv_bias,
            A_log, proj_B_weight, proj_C_weight,
            proj_delta_weight, proj_delta_bias,
            D_param, out_proj_weight,
            self.d_model, self.d_inner, self.d_state, self.d_conv,
            norm_eps
        )
        
        return result
