import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Conditional Variational Autoencoder (CVAE) with full encoder–decoder
    forward pass and loss computation.  The encoder conditions on both
    input x and label y; the decoder conditions on z and y.  Returns
    reconstruction loss + KL divergence.
    """

    def __init__(self, input_dim, label_dim, hidden_dim, latent_dim):
        super().__init__()
        self.enc = nn.Sequential(nn.Linear(input_dim + label_dim, hidden_dim), nn.ReLU())
        self.enc_mu = nn.Linear(hidden_dim, latent_dim)
        self.enc_logvar = nn.Linear(hidden_dim, latent_dim)
        self.dec = nn.Sequential(nn.Linear(latent_dim + label_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim))

    def forward(self, x: torch.Tensor, y: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim)  – input data
            y: (B, label_dim)  – conditioning label (one-hot)
        Returns:
            (loss, recon_loss, kl_loss):
                loss:       scalar – total CVAE loss
                recon_loss: scalar
                kl_loss:    scalar
        """
        h = self.enc(torch.cat([x, y], dim=-1))
        mu = self.enc_mu(h)
        logvar = self.enc_logvar(h)
        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)
        x_recon = self.dec(torch.cat([z, y], dim=-1))
        recon_loss = F.mse_loss(x_recon, x, reduction='sum') / x.shape[0]
        kl_loss = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum() / x.shape[0]
        loss = recon_loss + kl_loss
        return (loss, recon_loss, kl_loss)

        float sum = enc_mu_bias[j];
        for (int i = 0; i < hidden_dim; ++i) {
            sum += h[i] * enc_mu_weight[j * hidden_dim + i];
        }
        mu[j] = sum;
    }

    // logvar
    float logvar[32];
    for (int j = 0; j < latent_dim; ++j) {
        float sum = enc_logvar_bias[j];
        for (int i = 0; i < hidden_dim; ++i) {
            sum += h[i] * enc_logvar_weight[j * hidden_dim + i];
        }
        logvar[j] = sum;
    }

    // std and z
    float kl = 0.0f;
    for (int j = 0; j < latent_dim; ++j) {
        float std = expf(0.5f * logvar[j]);
        float eps = curand_normal(&state);
        z[j] = mu[j] + std * eps;
        kl += -0.5f * (1.0f + logvar[j] - mu[j]*mu[j] - expf(logvar[j]));
    }
    kl /= batch_size;

    // dec: concat z y -> linear1 + relu -> linear2
    float hdec[160]; // assume hidden <=160
    for (int j = 0; j < hidden_dim; ++j) {
        float sum = dec_bias1[j];
        for (int i = 0; i < latent_dim; ++i) {
            sum += z[i] * dec_weight1[j * (latent_dim + label_dim) + i];
        }
        for (int i = 0; i < label_dim; ++i) {
            sum += y[b * label_dim + i] * dec_weight1[j * (latent_dim + label_dim) + latent_dim + i];
        }
        hdec[j] = fmaxf(sum, 0.0f);
    }

    float recon_sum = 0.0f;
    for (int j = 0; j < input_dim; ++j) {
        float sum = dec_bias2[j];
        for (int i = 0; i < hidden_dim; ++i) {
            sum += hdec[i] * dec_weight2[j * hidden_dim + i];
        }
        float diff = sum - x[b * input_dim + j];
        recon_sum += diff * diff;
        x_recon[j] = sum; // not needed
    }
    float recon = recon_sum / batch_size;

    float total = recon + kl;

    if (threadIdx.x == 0) {
        atomicAdd(loss_out, total);
        atomicAdd(recon_out, recon);
        atomicAdd(kl_out, kl);
    }
}

class ModelNew(nn.Module):
    def __init__(self, input_dim, label_dim, hidden_dim, latent_dim):
        super().__init__()
        self.input_dim = input_dim
        self.label_dim = label_dim
        self.hidden_dim = hidden_dim
        self.latent_dim = latent_dim
        self.enc = nn.Sequential(nn.Linear(input_dim + label_dim, hidden_dim), nn.ReLU())
        self.enc_mu = nn.Linear(hidden_dim, latent_dim)
        self.enc_logvar = nn.Linear(hidden_dim, latent_dim)
        self.dec = nn.Sequential(nn.Linear(latent_dim + label_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim))

    def forward(self, x: torch.Tensor, y: torch.Tensor):
        enc_w = self.enc[0].weight
        enc_b = self.enc[0].bias
        mu_w = self.enc_mu.weight
        mu_b = self.enc_mu.bias
        logvar_w = self.enc_logvar.weight
        logvar_b = self.enc_logvar.bias
        dec_w1 = self.dec[0].weight
        dec_b1 = self.dec[0].bias
        dec_w2 = self.dec[2].weight
        dec_b2 = self.dec[2].bias

        outs = module.cvae_forward_cuda(
            x, y,
            enc_w, enc_b,
            mu_w, mu_b,
            logvar_w, logvar_b,
            dec_w1, dec_b1,
            dec_w2, dec_b2,
            self.input_dim, self.label_dim, self.hidden_dim, self.latent_dim
        )
        return tuple(outs)
