import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Multi-head row-wise softmax over CSR edge scores.

    This extends scalar graph edge softmax to multiple attention heads while
    preserving sparse CSR layout. Each row is normalized independently for each
    head.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, edge_scores: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        out = torch.empty_like(edge_scores)
        for node in range(num_nodes):
            start = int(row_ptr[node].item())
            end = int(row_ptr[node + 1].item())
            if end > start:
                out[start:end] = torch.softmax(edge_scores[start:end], dim=0)
        return out

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cfloat>

__global__ void edge_softmax_kernel(
    const int* __restrict__ row_ptr,
    const float* __restrict__ edge_scores,
    float* __restrict__ out,
    int num_nodes
) {
    int node = blockIdx.x;
    if (node >= num_nodes) return;

    int start = row_ptr[node];
    int end = row_ptr[node + 1];
    int length = end - start;

    if (length <= 0) return;

    extern __shared__ float sdata[];

    // Step 1: find max
    float local_max = -FLT_MAX;
    for (int i = threadIdx.x; i < length; i += blockDim.x) {
        float val = edge_scores[start + i];
        local_max = fmaxf(local_max, val);
    }
    sdata[threadIdx.x] = local_max;
    __syncthreads();

    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (threadIdx.x < s) {
            sdata[threadIdx.x] = fmaxf(sdata[threadIdx.x], sdata[threadIdx.x + s]);
        }
        __syncthreads();
    }
    float max_val = sdata[0];
    __syncthreads();

    // Step 2: sum of exp
    float local_sum = 0.0f;
    for (int i = threadIdx.x; i < length; i += blockDim.x) {
        float val = expf(edge_scores[start + i] - max_val);
        sdata[threadIdx.x + blockDim.x] = val;  // store exp temporarily
        local_sum += val;
    }
    // reuse sdata[0:blockDim.x] for sum reduction
    sdata[threadIdx.x] = local_sum;
    __syncthreads();

    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (threadIdx.x < s) {
            sdata[threadIdx.x] += sdata[threadIdx.x + s];
        }
        __syncthreads();
    }
    float sum_exp = sdata[0];
    __syncthreads();

    // Step 3: normalize
    float inv_sum = 1.0f / sum_exp;
    for (int i = threadIdx.x; i < length; i += blockDim.x) {
        out[start + i] = sdata[blockDim.x + i] * inv_sum;
    }
}

torch::Tensor edge_softmax_cuda(torch::Tensor row_ptr, torch::Tensor edge_scores) {
    int num_nodes = row_ptr.size(0) - 1;
    auto out = torch::empty_like(edge_scores);

    int threads = 256;
    dim3 blocks(num_nodes);
    size_t smem = 2 * threads * sizeof(float);

    edge_softmax_kernel<<<blocks, threads, smem>>>(
        row_ptr.data_ptr<int>(),
        edge_scores.data_ptr<float>(),
        out.data_ptr<float>(),
        num_nodes
    );

    return out;
}
