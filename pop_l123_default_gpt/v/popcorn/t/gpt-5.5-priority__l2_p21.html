import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Triangular multiplicative update – incoming edges.
    Mirror of the outgoing variant: updates pair (i,j) by aggregating over
    the shared index k via edges (k,i) and (k,j).
    """

    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            pair_repr: (B, N, N, pair_dim)
            pair_mask: (B, N, N)
        Returns:
            updated pair representation (B, N, N, pair_dim)
        """
        mask = pair_mask.unsqueeze(-1)
        x = self.layer_norm_in(pair_repr)

        a = self.proj_a(x) * torch.sigmoid(self.gate_a(x)) * mask  # (B, N, N, H)
        b = self.proj_b(x) * torch.sigmoid(self.gate_b(x)) * mask

        # Incoming: aggregate over k with a[k,i] * b[k,j]
        out = torch.einsum("bkih,bkjh->bijh", a, b)

        out = self.layer_norm_out(out)
        out = self.out_proj(out)
        out = torch.sigmoid(self.out_gate(pair_repr)) * out
        return out

            pb += xv * pbwh[c];
            gb += xv * gbwh[c];

__global__ void einsum_kernel(const float* __restrict__ A, const float* __restrict__ Bv,
                              float* __restrict__ T, int Bsz, int N, int H)
{
    long long total = (long long)Bsz * N * N * H;
    long long p = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    if (p >= total) return;
    int h = p & 31;
    long long q = p >> 5;
    int j = q & 31;
    q >>= 5;
    int i = q & 31;
    int b = q >> 5;
    float s = 0.0f;
    long long bbase = (long long)b * 32768;
    #pragma unroll
    for (int k = 0; k < 32; ++k) {
        s += A[bbase + ((long long)k * 32 + i) * 32 + h] * Bv[bbase + ((long long)k * 32 + j) * 32 + h];
    }
    T[p] = s;
}

__global__ void finish_kernel(
    const float* __restrict__ T, const float* __restrict__ x0, float* __restrict__ Y,
    const float* __restrict__ ln_w, const float* __restrict__ ln_b,
    const float* __restrict__ opw, const float* __restrict__ opb,
    const float* __restrict__ ogw, const float* __restrict__ ogb,
    int M, int C, int H)
{
    int idx = blockIdx.x;
    int tid = threadIdx.x;
    __shared__ float red[64];
    __shared__ float normh[32];
    const float* tb = T + (long long)idx * 32;
    const float* xb = x0 + (long long)idx * 64;

    float sum = (tid < 32) ? tb[tid] : 0.0f;
    red[tid] = sum; __syncthreads();
    for (int s = 32; s > 0; s >>= 1) { if (tid < s) red[tid] += red[tid+s]; __syncthreads(); }
    float mean = red[0] * (1.0f/32.0f);
    float ss = 0.0f;
    if (tid < 32) { float d = tb[tid] - mean; ss = d*d; }
    red[tid] = ss; __syncthreads();
    for (int s = 32; s > 0; s >>= 1) { if (tid < s) red[tid] += red[tid+s]; __syncthreads(); }
    float inv = rsqrtf(red[0] * (1.0f/32.0f) + 1.0e-5f);
    if (tid < 32) normh[tid] = (tb[tid] - mean) * inv * ln_w[tid] + ln_b[tid];
    __syncthreads();

    const float* w = opw + (long long)c * 32;
    #pragma unroll

    float g = ogb[c];
    const float* gw = ogw + (long long)c * 64;
    #pragma unroll
    for (int cc = 0; cc < 64; ++cc) g += xb[cc] * gw[cc];
    Y[(long long)idx * 64 + c] = sigmoidf_dev(g) * o;
}
