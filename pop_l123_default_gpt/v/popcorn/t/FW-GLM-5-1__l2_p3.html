import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Single-layer multi-head cross-attention with RoPE on Q and K, attention
    dropout, and output projection—intended as a fusion target for robotics
    stacks (e.g. proprio / action queries attending to vision or memory).

    Pipeline (conceptually one fused kernel): linear Q from query stream,
    linear K/V from key–value stream → reshape heads → apply RoPE separately
    along query positions (T_q) and key positions (T_k) → scaled dot-product,
    softmax, dropout on weights → aggregate values → output projection.
    """

    def __init__(
        self,
        dim: int,
        num_heads: int,
        dropout: float = 0.0,
        rope_base: float = 10000.0,
    ):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        assert self.head_dim % 2 == 0, "RoPE requires an even head dimension"

        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.out_proj = nn.Linear(dim, dim, bias=False)
        self.dropout = nn.Dropout(dropout)
        self.scale = 1.0 / math.sqrt(self.head_dim)

        inv_freq = 1.0 / (
            rope_base ** (torch.arange(0, self.head_dim, 2).float() / self.head_dim)
        )
        self.register_buffer("inv_freq", inv_freq, persistent=False)

    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        x1, x2 = x.chunk(2, dim=-1)
        return torch.cat([-x2, x1], dim=-1)

    def _rope(self, x: torch.Tensor, seq_len: int) -> torch.Tensor:
        """Apply RoPE along the sequence dimension (last-but-one after head). x: (B, H, L, d)."""
        t = torch.arange(seq_len, device=x.device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        emb = torch.cat([freqs, freqs], dim=-1)
        cos = emb.cos().view(1, 1, seq_len, self.head_dim)
        sin = emb.sin().view(1, 1, seq_len, self.head_dim)
        return x * cos + self._rotate_half(x) * sin

    def forward(self, x_query: torch.Tensor, x_kv: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x_query: (B, T_q, dim) — e.g. robot or decoder tokens
            x_kv:    (B, T_k, dim) — e.g. encoder / memory tokens
        Returns:
            (B, T_q, dim)
        """
        B, T_q, _ = x_query.shape
        _, T_k, _ = x_kv.shape
        h, d = self.num_heads, self.head_dim

        q = self.q_proj(x_query).view(B, T_q, h, d).transpose(1, 2)
        k = self.k_proj(x_kv).view(B, T_k, h, d).transpose(1, 2)
        v = self.v_proj(x_kv).view(B, T_k, h, d).transpose(1, 2)

        q = self._rope(q, T_q)
        k = self._rope(k, T_k)

        attn = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        out = torch.matmul(attn, v)
        out = out.transpose(1, 2).contiguous().view(B, T_q, self.dim)
        return self.out_proj(out)

        for (int k_idx = 0; k_idx < T_k; k_idx++) {
            float val = attn_ptr[row_offset + k_idx] * scale;
            if (val > max_val) max_val = val;
        }
        
        // Compute exp and sum
        float sum = 0.0f;
        for (int k_idx = 0; k_idx < T_k; k_idx++) {
            float val = expf(attn_ptr[row_offset + k_idx] * scale - max_val);
            attn_ptr[row_offset + k_idx] = val;
            sum += val;
        }
        
        // Normalize
        float inv_sum = 1.0f / sum;
        for (int k_idx = 0; k_idx < T_k; k_idx++) {
            attn_ptr[row_offset + k_idx] *= inv_sum;
        }
    }
}

// Attention matrix multiplication: Q @ K^T
__global__ void qk_kernel(
    const float* __restrict__ q_ptr,
    const float* __restrict__ k_ptr,
    float* __restrict__ attn_ptr,
    int B, int H, int T_q, int T_k, int d
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * H * T_q * T_k;
    
    if (idx < total) {
        int b = idx / (H * T_q * T_k);
        int h = (idx / (T_q * T_k)) % H;
        int q_idx = (idx / T_k) % T_q;
        int kv_idx = idx % T_k;
        
        int q_offset = b * H * T_q * d + h * T_q * d + q_idx * d;
        int k_offset = b * H * T_k * d + h * T_k * d + kv_idx * d;
        
        float sum = 0.0f;
        for (int i = 0; i < d; i++) {
            sum += q_ptr[q_offset + i] * k_ptr[k_offset + i];
        }
        
        attn_ptr[idx] = sum;
    }
}

// Attention-value multiplication: attn @ V
__global__ void av_kernel(
    const float* __restrict__ attn_ptr,
    const float* __restrict__ v_ptr,
    float* __restrict__ out_ptr,
    int B, int H, int T_q, int T_k, int d
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * H * T_q * d;
    
    if (idx < total) {
        int b = idx / (H * T_q * d);
        int h = (idx / (T_q * d)) % H;
        int q_idx = (idx / d) % T_q;
        int i = idx % d;
        
        int attn_offset = b * H * T_q * T_k + h * T_q * T_k + q_idx * T_k;
        
        float sum = 0.0f;
        for (int kv_idx = 0; kv_idx < T_k; kv_idx++) {
            int v_offset = b * H * T_k * d + h * T_k * d + kv_idx * d + i;
            sum += attn_ptr[attn_offset + kv_idx] * v_ptr[v_offset];
        }
        
        out_ptr[idx] = sum;
    }
}

class Model(nn.Module):
    """
    Single-layer multi-head cross-attention with RoPE on Q and K, attention
    dropout, and output projection—intended as a fusion target for robotics
    stacks (e.g. proprio / action queries attending to vision or memory).
    """

    def __init__(
        self,
        dim: int,
        num_heads: int,
        dropout: float = 0.0,
        rope_base: float = 10000.0,
    ):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        assert self.head_dim % 2 == 0, "RoPE requires an even head dimension"

        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.out_proj = nn.Linear(dim, dim, bias=False)
        self.dropout = nn.Dropout(dropout)
        self.scale = 1.0 / math.sqrt(self.head_dim)

        inv_freq = 1.0 / (
            rope_base ** (torch.arange(0, self.head_dim, 2).float() / self.head_dim)
        )
        self.register_buffer("inv_freq", inv_freq, persistent=False)

    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        x1, x2 = x.chunk(2, dim=-1)
        return torch.cat([-x2, x1], dim=-1)

    def _rope(self, x: torch.Tensor, seq_len: int) -> torch.Tensor:
        """Apply RoPE along the sequence dimension (last-but-one after head). x: (B, H, L, d)."""
        t = torch.arange(seq_len, device=x.device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        emb = torch.cat([freqs, freqs], dim=-1)
        cos = emb.cos().view(1, 1, seq_len, self.head_dim)
        sin = emb.sin().view(1, 1, seq_len, self.head_dim)
        return x * cos + self._rotate_half(x) * sin

    def forward(self, x_query: torch.Tensor, x_kv: torch.Tensor) -> torch.Tensor:
        B, T_q, _ = x_query.shape
        _, T_k, _ = x_kv.shape
        h, d = self.num_heads, self.head_dim

        q = self.q_proj(x_query).view(B, T_q, h, d).transpose(1, 2)
        k = self.k_proj(x_kv).view(B, T_k, h, d).transpose(1, 2)
        v = self.v_proj(x_kv).view(B, T_k, h, d).transpose(1, 2)

        q = self._rope(q, T_q)
        k = self._rope(k, T_k)

        attn = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        out = torch.matmul(attn, v)
        out = out.transpose(1, 2).contiguous().view(B, T_q, self.dim)
        return self.out_proj(out)


class ModelNew(nn.Module):
    """
    Custom CUDA implementation of multi-head cross-attention with RoPE.
    """

    def __init__(
        self,
        dim: int,
        num_heads: int,
        dropout: float = 0.0,
        rope_base: float = 10000.0,
    ):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        assert self.head_dim % 2 == 0, "RoPE requires an even head dimension"

        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.out_proj = nn.Linear(dim, dim, bias=False)
        self.dropout = nn.Dropout(dropout)
        self.scale = 1.0 / math.sqrt(self.head_dim)

        inv_freq = 1.0 / (
            rope_base ** (torch.arange(0, self.head_dim, 2).float() / self.head_dim)
        )
        self.register_buffer("inv_freq", inv_freq, persistent=False)

    def _compute_rope_emb(self, seq_len: int, device):
        """Compute RoPE embeddings for a sequence length."""
        t = torch.arange(seq_len, device=device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        cos = freqs.cos()  # (seq_len, head_dim/2)
        sin = freqs.sin()  # (seq_len, head_dim/2)
        return cos, sin

    def forward(self, x_query: torch.Tensor, x_kv: torch.Tensor) -> torch.Tensor:
        B, T_q, _ = x_query.shape
        _, T_k, _ = x_kv.shape
        h, d = self.num_heads, self.head_dim

        # Project and reshape
        q = self.q_proj(x_query).view(B, T_q, h, d).transpose(1, 2).contiguous()
        k = self.k_proj(x_kv).view(B, T_k, h, d).transpose(1, 2).contiguous()
        v = self.v_proj(x_kv).view(B, T_k, h, d).transpose(1, 2).contiguous()

        # Compute RoPE embeddings
        cos_q, sin_q = self._compute_rope_emb(T_q, x_query.device)
        cos_k, sin_k = self._compute_rope_emb(T_k, x_kv.device)

        # Apply RoPE using custom CUDA kernel
        q_rope = torch.empty_like(q)
        k_rope = torch.empty_like(k)
        attention_ext.rope_cuda(q, cos_q, sin_q, q_rope)
        attention_ext.rope_cuda(k, cos_k, sin_k, k_rope)

        # Compute attention scores using custom CUDA kernel
        attn = torch.empty(B, h, T_q, T_k, device=x_query.device, dtype=x_query.dtype)
        attention_ext.qk_cuda(q_rope, k_rope, attn, B, h, T_q, T_k, d)

        # Apply softmax using custom CUDA kernel
        attention_ext.softmax_cuda(attn, B, h, T_q, T_k, self.scale)

        # Apply dropout (p=0 in test, so no-op)
        attn = self.dropout(attn)

        # Compute attention output using custom CUDA kernel
        attn_out = torch.empty(B, h, T_q, d, device=x_query.device, dtype=x_query.dtype)
        attention_ext.av_cuda(attn, v, attn_out, B, h, T_q, T_k, d)

        # Transpose and reshape for output projection
        out = attn_out.transpose(1, 2).contiguous().view(B, T_q, self.dim)

        return self.out_proj(out)
