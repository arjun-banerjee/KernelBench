import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Differentiable Smith-Waterman-style local sequence alignment scoring.
    Uses a soft-max relaxation of the DP recurrence so gradients can flow
    through the alignment matrix.  
    """
    def __init__(self, gap_open=-1.0, temperature=1.0):
        super().__init__()
        self.gap_open = gap_open
        self.temperature = temperature

    def forward(self, score_matrix: torch.Tensor) -> torch.Tensor:
        """
        Args:
            score_matrix: (B, M, N) – per-position similarity scores
        Returns:
            alignment_score: (B,) – soft local-alignment score (log-sum-exp over DP cells)
        """
        B, M, N = score_matrix.shape
        neg_inf = torch.tensor(float("-inf"), device=score_matrix.device, dtype=score_matrix.dtype)

        # H[i][j] = max(0, H[i-1][j-1]+S[i][j], H[i-1][j]+gap, H[i][j-1]+gap)
        H = score_matrix.new_full((B, M + 1, N + 1), 0.0)

        for i in range(1, M + 1):
            for j in range(1, N + 1):
                match = H[:, i - 1, j - 1] + score_matrix[:, i - 1, j - 1]
                delete = H[:, i - 1, j] + self.gap_open
                insert = H[:, i, j - 1] + self.gap_open
                zero = torch.zeros(B, device=score_matrix.device)
                stack = torch.stack([zero, match, delete, insert], dim=-1)
                H[:, i, j] = self.temperature * torch.logsumexp(stack / self.temperature, dim=-1)

        return H.view(B, -1).max(dim=-1).values

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cfloat>

__global__ void soft_smith_waterman_kernel(
    const float* __restrict__ score_matrix,
    float* __restrict__ output,
    int B, int M, int N,
    float gap_open, float temperature
) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b >= B) return;

    const float* scores = score_matrix + b * M * N;
    float max_score = 0.0f;
    const float inv_temp = 1.0f / temperature;

    // Use two rows: prev and curr, size N+1
    extern __shared__ float shared[];
    float* prev_row = shared + threadIdx.x * (N + 1);
    float* curr_row = shared + (blockDim.x * (N + 1)) + threadIdx.x * (N + 1);

    // Initialize first row and col to 0
    for (int j = 0; j <= N; j++) {
        prev_row[j] = 0.0f;
    }
    curr_row[0] = 0.0f;

    for (int i = 1; i <= M; i++) {
        curr_row[0] = 0.0f;
        for (int j = 1; j <= N; j++) {
            float match = prev_row[j-1] + scores[(i-1)*N + (j-1)];
            float delete_ = prev_row[j] + gap_open;
            float insert = curr_row[j-1] + gap_open;
            float zero = 0.0f;

            // logsumexp with temperature
            float m = fmaxf(fmaxf(zero, match), fmaxf(delete_, insert));
            float sum = expf((zero - m) * inv_temp) +
                        expf((match - m) * inv_temp) +
                        expf((delete_ - m) * inv_temp) +
                        expf((insert - m) * inv_temp);
            float val = temperature * logf(sum) + m;
            curr_row[j] = val;
            if (val > max_score) max_score = val;
        }
        // swap rows
        float* temp = prev_row;
        prev_row = curr_row;
        curr_row = temp;
    }

    output[b] = max_score;
}

torch::Tensor soft_smith_waterman_cuda(torch::Tensor score_matrix, float gap_open, float temperature) {
    int B = score_matrix.size(0);
    int M = score_matrix.size(1);
    int N = score_matrix.size(2);

    auto output = torch::zeros({B}, score_matrix.options());

    int threads = 16;
    int blocks = (B + threads - 1) / threads;
    int shared_size = 2 * threads * (N + 1) * sizeof(float);

    soft_smith_waterman_kernel<<<blocks, threads, shared_size>>>(
        score_matrix.data_ptr<float>(),
        output.data_ptr<float>(),
        B, M, N, gap_open, temperature
    );

    return output;
}
