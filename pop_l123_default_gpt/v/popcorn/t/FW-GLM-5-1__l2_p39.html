// Each thread computes one output element (b, oc, t)
__global__ void fused_conv1d_silu_pw_kernel(
    const float* __restrict__ input,
    const float* __restrict__ dw_weight,
    const float* __restrict__ pw_weight,
    float* __restrict__ output,
    const int B, const int C_in, const int C_out, const int T,
    const int kernel_size, const int dilation, const int padding
) {
    int b = blockIdx.x;
    int oc = blockIdx.y * blockDim.y + threadIdx.y;
    int t = blockIdx.z * blockDim.x + threadIdx.x;
    
    if (oc >= C_out || t >= T) return;
    
    float out_val = 0.0f;
    
    for (int ic = 0; ic < C_in; ic++) {
        // Compute depthwise conv for channel ic at position t
        float dw_sum = 0.0f;
