import torch

from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void permute_kernel(const float* __restrict__ x, const int64_t* __restrict__ perm, float* __restrict__ y, int B, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < B * N) {
        int b = idx / N;
        int j = idx % N;
        int64_t p = perm[j];
        y[b * N + j] = x[b * N + p];
    }
}

torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm) {
    auto B = x.size(0);
    auto N = x.size(1);
    auto y = torch::empty({B, N}, x.options());
    int threads = 256;
    int blocks = (B * N + threads - 1) / threads;
    permute_kernel<<<blocks, threads>>>(x.data_ptr<float>(), perm.data_ptr<int64_t>(), y.data_ptr<float>(), B, N);
    return y;
}
