template <int KERNEL_SIZE>
__global__ void depthwise_conv1d_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size, int channels, int seq_len,
    int stride, int padding, int output_len, bool has_bias) {
    
    int batch = blockIdx.y;
    int channel = blockIdx.x;
    
    int tid = threadIdx.x;
    int block_dim = blockDim.x;
    
    // Input and output pointers for this batch and channel
    long long input_offset = (long long)batch * channels * seq_len + channel * seq_len;
    long long output_offset = (long long)batch * channels * output_len + channel * output_len;
    
    const float* input_ptr = input + input_offset;
    float* output_ptr = output + output_offset;
    
    // Load filter weights into registers
    float filter[KERNEL_SIZE];
    for (int k = 0; k < KERNEL_SIZE; k++) {
        filter[k] = weight[channel * KERNEL_SIZE + k];
    }
