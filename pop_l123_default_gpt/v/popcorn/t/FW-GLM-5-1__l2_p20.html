import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Scalar-gated delta-rule linear attention recurrence.

    A recurrent state matrix is decayed by a scalar gate per head, updated by a
    rank-1 key/value outer product, then read out by a query vector.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
        beta: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-1).unsqueeze(-1)
            beta_t = beta[:, t].unsqueeze(-1).unsqueeze(-1)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + beta_t * kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template <typename scalar_t>
__global__ void linear_attention_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ gate,
    const scalar_t* __restrict__ beta,
    scalar_t* __restrict__ output,
    int batch_size, int seq_len, int num_heads, int head_dim) {
    
    // Map block to (batch, head)
    int batch = blockIdx.x / num_heads;
    int head = blockIdx.x % num_heads;
    
    int tid = threadIdx.x;
    int block_size = blockDim.x;
    
    // Shared memory for state (head_dim x head_dim)
    extern __shared__ char shared_mem[];
    scalar_t* state = reinterpret_cast<scalar_t*>(shared_mem);
    
    // Initialize state to zero
    for (int i = tid; i < head_dim * head_dim; i += block_size) {
        state[i] = scalar_t(0);
    }
    __syncthreads();
    
    // Iterate through sequence
    for (int t = 0; t < seq_len; t++) {
        // Load gate and beta scalars for this (batch, head, t)
        // gate shape: (batch, seq_len, num_heads)
        int gate_idx = batch * seq_len * num_heads + t * num_heads + head;
        scalar_t gate_t = gate[gate_idx];
        scalar_t beta_t = beta[gate_idx];
        
        // Get pointers to k[t], v[t], q[t] for this (batch, head)
        // q/k/v shape: (batch, seq_len, num_heads, head_dim)
        int kvq_base = batch * seq_len * num_heads * head_dim + t * num_heads * head_dim + head * head_dim;
        const scalar_t* k_t = k + kvq_base;
        const scalar_t* v_t = v + kvq_base;
        const scalar_t* q_t = q + kvq_base;
        
        // Update state: state = gate_t * state + beta_t * k[t] * v[t]^T
        // state[i, j] = gate_t * state[i, j] + beta_t * k[i] * v[j]
        for (int idx = tid; idx < head_dim * head_dim; idx += block_size) {
            int row = idx / head_dim;
            int col = idx % head_dim;
            state[idx] = gate_t * state[idx] + beta_t * k_t[row] * v_t[col];
        }
        __syncthreads();
        
        // Compute output: y[t] = q[t]^T @ state -> sum over rows
        // y[j] = sum_i q[i] * state[i, j]
        scalar_t* y_t = output + kvq_base;

            for (int i = 0; i < head_dim; i++) {
                sum += q_t[i] * state[i * head_dim + j];
            }
            y_t[j] = sum;
        }
        __syncthreads();
    }
}
