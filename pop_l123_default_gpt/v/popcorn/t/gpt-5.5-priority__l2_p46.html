import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Stein Variational Gradient Descent (SVGD) update step.
    Maintains a set of particles approximating a posterior and updates
    them via a kernel-smoothed gradient that balances fitting the target
    with repulsion between particles.  The RBF kernel bandwidth is set
    via the median heuristic.
    """

    def __init__(self, dim):
        super().__init__()
        self.dim = dim
        # Target: N(0, I) for benchmarking
        self.register_buffer("target_mean", torch.zeros(dim))

    def _score_fn(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -x

    def forward(self, particles: torch.Tensor) -> torch.Tensor:
        """
        Args:
            particles: (N, dim) – current particle positions
        Returns:
            phi: (N, dim) – SVGD update direction
        """
        N = particles.shape[0]
        score = self._score_fn(particles)  # (N, dim)

        # Pairwise squared distances
        diff = particles.unsqueeze(0) - particles.unsqueeze(1)  # (N, N, dim)
        dist_sq = (diff ** 2).sum(-1)  # (N, N)

        # Median heuristic for bandwidth
        median_sq = torch.median(dist_sq.view(-1)).clamp(min=1e-5)
        h = median_sq / max(torch.log(torch.tensor(float(N))).item(), 1.0)

        K = torch.exp(-dist_sq / (2.0 * h))  # (N, N)
        grad_K = -diff / h * K.unsqueeze(-1)  # (N, N, dim)

        # SVGD update: phi(x_i) = (1/N) sum_j [ K(x_j, x_i) * score(x_j) + grad_K(x_j, x_i) ]
        phi = (K @ score + grad_K.sum(dim=0)) / N
        return phi


dim = 32
num_particles = 1024


def get_inputs():
    return [torch.randn(num_particles, dim)]


def get_init_inputs():
    return [dim]

__global__ void dist_upper_kernel(const float* __restrict__ x,
                                  float* __restrict__ dist,
                                  float* __restrict__ upper,
                                  int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = N * N;
    if (idx >= total) return;

    int i = idx / N;
    int j = idx - i * N;
    const float4* xi = reinterpret_cast<const float4*>(x + i * 32);
    const float4* xj = reinterpret_cast<const float4*>(x + j * 32);

    float s = 0.0f;
    #pragma unroll
    for (int q = 0; q < 8; ++q) {
        float4 a = xi[q];
        float4 b = xj[q];
        float dx = b.x - a.x; s = fmaf(dx, dx, s);
        float dy = b.y - a.y; s = fmaf(dy, dy, s);
        float dz = b.z - a.z; s = fmaf(dz, dz, s);
        float dw = b.w - a.w; s = fmaf(dw, dw, s);
    }
    dist[idx] = s;

    // The full NxN distance matrix contains N diagonal zeros and two copies of
    // each off-diagonal distance.  Store one triangular copy for a half-sized
    // kthvalue selection yielding the same PyTorch lower median.
    if (j > i) {
        int pos = (i * (2 * N - i - 1)) / 2 + (j - i - 1);
        upper[pos] = s;
    }
}

__global__ void phi_kernel(const float* __restrict__ x,
                           const float* __restrict__ dist,
                           const float* __restrict__ median_ptr,
                           float* __restrict__ out,
                           int N,
                           float logN) {
    int i = blockIdx.x;
    int tid = threadIdx.x;
    constexpr int TPB = 128;
    __shared__ float sh[TPB * 32];

    float med = median_ptr[0];
    med = med < 1.0e-5f ? 1.0e-5f : med;
    float h = med / (logN > 1.0f ? logN : 1.0f);
    float invh = 1.0f / h;
    float inv2h = 0.5f * invh;
    float coef_xj = invh - 1.0f;

    const float* xi = x + i * 32;
    float xid[32];
    float acc[32];
    #pragma unroll
    for (int d = 0; d < 32; ++d) {
        xid[d] = xi[d];
        acc[d] = 0.0f;
    }

    for (int j = tid; j < N; j += TPB) {
        float k = __expf(-dist[i * N + j] * inv2h);
        const float* xj = x + j * 32;
        #pragma unroll
        for (int d = 0; d < 32; ++d) {
            acc[d] += k * (coef_xj * xj[d] - invh * xid[d]);
        }
    }

    #pragma unroll
    for (int d = 0; d < 32; ++d) sh[tid * 32 + d] = acc[d];
    __syncthreads();

    for (int stride = 64; stride > 0; stride >>= 1) {
        if (tid < stride) {
            #pragma unroll

        }

        #pragma unroll

    }
}
