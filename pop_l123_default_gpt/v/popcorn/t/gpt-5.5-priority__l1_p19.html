import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues
        Returns:
            dist: (B, N, N) – pairwise Euclidean distance matrix
        """
        diff = coords.unsqueeze(2) - coords.unsqueeze(1)  # (B, N, N, 3)
        dist = torch.sqrt((diff ** 2).sum(dim=-1) + 1e-8)
        return dist

#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define TILE 32
#define ROW_THREADS 8

__global__ void pairwise_sym32_kernel(const float* __restrict__ coords, float* __restrict__ out,
                                      int B, int N) {
    int tile_j = blockIdx.x;
    int tile_i = blockIdx.y;
    if (tile_i > tile_j) return;

    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int b = blockIdx.z;
    const float* c = coords + ((long long)b * N * 3);

    __shared__ float si[TILE][3];
    __shared__ float sj[TILE][3];
    __shared__ float dsh[TILE][TILE + 1];

    if (ty < 3) {
        int ii = tile_i * TILE + tx;
        int jj = tile_j * TILE + tx;
        si[tx][ty] = (ii < N) ? c[ii * 3 + ty] : 0.0f;
        sj[tx][ty] = (jj < N) ? c[jj * 3 + ty] : 0.0f;
    }
    __syncthreads();

    #pragma unroll
    for (int rr = 0; rr < 4; ++rr) {
        int lr = ty + rr * ROW_THREADS;
        int i = tile_i * TILE + lr;
        int j = tile_j * TILE + tx;
        float val = 0.0f;
        if (i < N && j < N) {
            float dx = si[lr][0] - sj[tx][0];
            float dy = si[lr][1] - sj[tx][1];
            float dz = si[lr][2] - sj[tx][2];
            val = sqrtf(dx * dx + dy * dy + dz * dz + 1.0e-8f);
            out[((long long)b * N + i) * N + j] = val;
        }
        dsh[lr][tx] = val;
    }

    if (tile_i != tile_j) {
        __syncthreads();

            if (jrow < N && icol < N) {
                out[((long long)b * N + jrow) * N + icol] = dsh[tx][lr];
            }
        }
    }
}
