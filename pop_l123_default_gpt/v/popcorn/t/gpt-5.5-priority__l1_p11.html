import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Degree-normalized neighbor aggregation.

    Computes a GCN-style sparse aggregation where each incoming message is
    scaled by 1 / sqrt(deg(dst) * deg(src)). This captures a common spectral
    graph normalization pattern over CSR adjacency.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        row_ptr: torch.Tensor,
        col_idx: torch.Tensor,
        node_feat: torch.Tensor,
        degrees: torch.Tensor,
    ) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        feat_dim = node_feat.shape[1]
        out = torch.zeros(num_nodes, feat_dim, dtype=node_feat.dtype, device=node_feat.device)

        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end <= start:
                continue
            src = col_idx[start:end].long()
            norm = 1.0 / torch.sqrt(degrees[dst] * degrees[src]).unsqueeze(-1)
            out[dst] = (node_feat[src] * norm).sum(dim=0)

        return out


num_nodes = 640
avg_degree = 16
feat_dim = 80


def _make_graph():
    degree = torch.full((num_nodes,), avg_degree, dtype=torch.int32)
    degree = torch.clamp(degree + ((torch.arange(num_nodes, dtype=torch.int32) % 11) - 5), min=1)
    row_ptr = torch.zeros(num_nodes + 1, dtype=torch.int32)
    row_ptr[1:] = torch.cumsum(degree, dim=0)
    num_edges = int(row_ptr[-1].item())
    col_idx = torch.randint(0, num_nodes, (num_edges,), dtype=torch.int32)
    node_feat = torch.randn(num_nodes, feat_dim, dtype=torch.float32)

    # Source-side degrees derived from column occurrences, plus one for safety.
    degrees = torch.bincount(col_idx.long(), minlength=num_nodes).to(torch.float32) + 1.0
    return row_ptr, col_idx, node_feat, degrees


def get_inputs():
    return list(_make_graph())


def get_init_inputs():
    return []

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_runtime.h>

void gcn_forward_launcher(const int* row_ptr, const int* col_idx, const float* node_feat, const float* degrees, float* out, int num_nodes, int feat_dim, cudaStream_t stream);

torch::Tensor gcn_forward(torch::Tensor row_ptr, torch::Tensor col_idx, torch::Tensor node_feat, torch::Tensor degrees) {
    const int num_nodes = row_ptr.numel() - 1;
    const int feat_dim = node_feat.size(1);
    auto out = torch::empty({num_nodes, feat_dim}, node_feat.options());
    gcn_forward_launcher(row_ptr.data_ptr<int>(), col_idx.data_ptr<int>(), node_feat.data_ptr<float>(), degrees.data_ptr<float>(), out.data_ptr<float>(), num_nodes, feat_dim, at::cuda::getCurrentCUDAStream());
    return out;
}
"""

_cuda_src = r"""
#include <cuda.h>
#include <cuda_runtime.h>
#include <torch/extension.h>

__global__ void gcn_kernel_80(const int* __restrict__ row_ptr,
                           const int* __restrict__ col_idx,
                           const float* __restrict__ node_feat,
                           const float* __restrict__ degrees,
                           float* __restrict__ out) {
    int dst = blockIdx.x;
    int tid = threadIdx.x;
    int lane = tid & 31;
    int warp = tid >> 5;
    int f = (warp << 5) + lane;
    int start = row_ptr[dst];
    int end = row_ptr[dst + 1];
    float deg_dst = degrees[dst];
    float acc = 0.0f;
    #pragma unroll 1
    for (int e = start; e < end; ++e) {
        int src = 0;
        float norm = 0.0f;
        if (lane == 0) {
            src = col_idx[e];
            norm = rsqrtf(deg_dst * degrees[src]);
        }
        src = __shfl_sync(0xffffffff, src, 0);
        norm = __shfl_sync(0xffffffff, norm, 0);
        if (f < 80) acc += node_feat[src * 80 + f] * norm;
    }
    if (f < 80) out[dst * 80 + f] = acc;
}

__global__ void gcn_kernel_generic(const int* __restrict__ row_ptr,
                           const int* __restrict__ col_idx,
                           const float* __restrict__ node_feat,
                           const float* __restrict__ degrees,
                           float* __restrict__ out,
                           int num_nodes, int feat_dim) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = num_nodes * feat_dim;
    if (idx >= total) return;
    int dst = idx / feat_dim;
    int f = idx - dst * feat_dim;
    int start = row_ptr[dst];
    int end = row_ptr[dst + 1];
    float deg_dst = degrees[dst];
    float acc = 0.0f;
    for (int e = start; e < end; ++e) {
        int src = col_idx[e];
        float norm = rsqrtf(deg_dst * degrees[src]);
        acc += node_feat[src * feat_dim + f] * norm;
    }
    out[idx] = acc;
}

void gcn_forward_launcher(const int* row_ptr, const int* col_idx, const float* node_feat, const float* degrees, float* out, int num_nodes, int feat_dim, cudaStream_t stream) {
    if (feat_dim == 80) {
        gcn_kernel_80<<<num_nodes, 96, 0, stream>>>(row_ptr, col_idx, node_feat, degrees, out);
    } else {
        int total = num_nodes * feat_dim;
        int threads = 256;
        int blocks = (total + threads - 1) / threads;
        gcn_kernel_generic<<<blocks, threads, 0, stream>>>(row_ptr, col_idx, node_feat, degrees, out, num_nodes, feat_dim);
    }
}
"""

_ext = load_inline(name="gcn_norm_ext_submission", cpp_sources=_cpp_src, cuda_sources=_cuda_src, functions=["gcn_forward"], verbose=False, extra_cuda_cflags=["-O3", "--use_fast_math"], extra_cflags=["-O3"])

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
    def forward(self, row_ptr: torch.Tensor, col_idx: torch.Tensor, node_feat: torch.Tensor, degrees: torch.Tensor) -> torch.Tensor:
        if row_ptr.dtype is not torch.int32:
            row_ptr = row_ptr.to(torch.int32)
        if col_idx.dtype is not torch.int32:
            col_idx = col_idx.to(torch.int32)
        return _ext.gcn_forward(row_ptr, col_idx, node_feat, degrees)
