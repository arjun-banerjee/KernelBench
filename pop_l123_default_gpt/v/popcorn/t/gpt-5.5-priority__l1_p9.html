import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a parallel associative scan (linear recurrence) over a 1D sequence.
    Given coefficient a and input b at each timestep, computes:
        h_0 = b_0
        h_t = a_t * h_{t-1} + b_t  for t > 0
    This is the core parallelizable primitive underlying Mamba's selective scan and
    other SSM architectures. Sequential in naive PyTorch but can be parallelized
    via the associative scan algorithm in O(L log L) span.

    Args:
        d_model (int): Number of independent recurrence channels.
    """
    def __init__(self, d_model: int):
        super(Model, self).__init__()
        self.d_model = d_model

    def forward(self, a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
        B, L, D = a.shape
        h = torch.zeros(B, D, device=a.device, dtype=a.dtype)
        outputs = []
        for t in range(L):
            h = a[:, t] * h + b[:, t]
            outputs.append(h)
        return torch.stack(outputs, dim=1)

batch_size = 32
d_model = 256
seq_len = 4096

def get_inputs():
    a = torch.sigmoid(torch.randn(batch_size, seq_len, d_model))
    b = torch.randn(batch_size, seq_len, d_model)
    return [a, b]

def get_init_inputs():
    return [d_model]

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define CHUNK 128
#define NSEGS 32
#define TILE 128

__global__ void segment_summary_kernel(const float* __restrict__ a,
                                       const float* __restrict__ b,
                                       float* __restrict__ segA,
                                       float* __restrict__ segB,
                                       int B, int L, int D) {
    int d = blockIdx.x * TILE + threadIdx.x;
    int batch = blockIdx.y;
    int seg = blockIdx.z;
    if (d >= D) return;

    int base = batch * L * D + seg * CHUNK * D + d;
    float h = 0.0f;
    float p = 1.0f;

    #pragma unroll 1
    for (int i = 0; i < CHUNK; ++i) {
        int idx = base + i * D;
        float at = a[idx];
        p *= at;
        h = fmaf(at, h, b[idx]);
    }

    int si = (batch * NSEGS + seg) * D + d;
    segA[si] = p;
    segB[si] = h;
}

__global__ void scan_segments_kernel(float* __restrict__ segA,
                                     const float* __restrict__ segB,
                                     int B, int D) {
    int d = blockIdx.x * TILE + threadIdx.x;
    int batch = blockIdx.y;
    if (d >= D) return;

    float h = 0.0f;
    #pragma unroll
    for (int s = 0; s < NSEGS; ++s) {
        int si = (batch * NSEGS + s) * D + d;
        float p = segA[si];
        segA[si] = h;                 // reuse segA as the per-segment initial state
        h = fmaf(p, h, segB[si]);
    }
}

__global__ void emit_segments_kernel(const float* __restrict__ a,
                                     const float* __restrict__ b,
                                     const float* __restrict__ init,
                                     float* __restrict__ out,
                                     int B, int L, int D) {
    int d = blockIdx.x * TILE + threadIdx.x;
    int batch = blockIdx.y;
    int seg = blockIdx.z;
    if (d >= D) return;

    int si = (batch * NSEGS + seg) * D + d;
    float h = init[si];
    int base = batch * L * D + seg * CHUNK * D + d;

    #pragma unroll 1
    for (int i = 0; i < CHUNK; ++i) {
        int idx = base + i * D;
        h = fmaf(a[idx], h, b[idx]);
        out[idx] = h;
    }
}

torch::Tensor scan_forward(torch::Tensor a, torch::Tensor b) {
    int B = (int)a.size(0);
    int L = (int)a.size(1);
    int D = (int)a.size(2);

    auto out = torch::empty_like(a);
    auto segA = torch::empty({B, NSEGS, D}, a.options());
    auto segB = torch::empty({B, NSEGS, D}, a.options());

    dim3 block(TILE);
    dim3 gridSeg((D + TILE - 1) / TILE, B, NSEGS);
    dim3 gridScan((D + TILE - 1) / TILE, B);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    segment_summary_kernel<<<gridSeg, block, 0, stream>>>(
        a.data_ptr<float>(), b.data_ptr<float>(),
        segA.data_ptr<float>(), segB.data_ptr<float>(), B, L, D);

        segA.data_ptr<float>(), segB.data_ptr<float>(), B, D);

    emit_segments_kernel<<<gridSeg, block, 0, stream>>>(
        a.data_ptr<float>(), b.data_ptr<float>(), segA.data_ptr<float>(),
        out.data_ptr<float>(), B, L, D);

    return out;
}
