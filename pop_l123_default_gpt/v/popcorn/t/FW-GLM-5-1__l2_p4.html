class Model(nn.Module):
    """
    Graph-Fused VLA (GF-VLA) style **scene graph relation update** in one forward.

    Objects are nodes with vision-language aligned embeddings; directed pairs
    (i → j) carry pairwise conditioning (geometry, language grounding, or fused
    VLA features). This reference fuses:

      LayerNorm(nodes) → multi-head attention over objects with **pairwise
      relation MLP bias** on edges → residual write-back.

    Intended as a fusion target for navigation / manipulation stacks that maintain
    explicit scene graphs and update them from multimodal observations.
    """

    def __init__(self, dim: int, pair_dim: int, num_heads: int):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.scale = 1.0 / math.sqrt(self.head_dim)

        self.norm = nn.LayerNorm(dim)
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.rel_mlp = nn.Sequential(
            nn.Linear(pair_dim, pair_dim),
            nn.SiLU(),
            nn.Linear(pair_dim, num_heads),
        )
        self.out_proj = nn.Linear(dim, dim, bias=False)

    def forward(
        self,
        nodes: torch.Tensor,
        pair_feats: torch.Tensor,
        edge_mask: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            nodes: (B, N, D) object / region embeddings.
            pair_feats: (B, N, N, P) pairwise VLA or geometric conditioning.
            edge_mask: (B, N, N) float in {0, 1}; 1 keeps attention over column j.
        Returns:
            (B, N, D) updated node embeddings (residual around attention block).
        """
        B, N, D = nodes.shape
        _, N2, N3, P = pair_feats.shape
        assert N == N2 == N3 and P == self.pair_dim
        h = self.norm(nodes)

        q = self.q_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)

        logits = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        rel_bias = self.rel_mlp(pair_feats).permute(0, 3, 1, 2)
        logits = logits + rel_bias

        logits = logits.masked_fill(edge_mask.unsqueeze(1) < 0.5, float("-inf"))
        attn = F.softmax(logits, dim=-1)
        ctx = torch.matmul(attn, v).transpose(1, 2).reshape(B, N, D)
        return nodes + self.out_proj(ctx)

}

template <typename scalar_t>
__device__ __forceinline__ scalar_t blockReduceMax(scalar_t val) {
    __shared__ scalar_t shared[32];
    int lane = threadIdx.x % WARP_SIZE;
    int wid = threadIdx.x / WARP_SIZE;
    
    val = warpReduceMax(val);
    
    if (lane == 0) {
        shared[wid] = val;
    }
    __syncthreads();
    
    if (wid == 0) {
        val = (threadIdx.x < (blockDim.x + WARP_SIZE - 1) / WARP_SIZE) ? shared[lane] : scalar_t(-INFINITY);
        val = warpReduceMax(val);
    }
    return val;
}

template <typename scalar_t>
__device__ __forceinline__ scalar_t warpReduceSum(scalar_t val) {
    for (int offset = WARP_SIZE / 2; offset > 0; offset /= 2) {
        val += __shfl_down_sync(0xffffffff, val, offset);
    }
    return val;
}

template <typename scalar_t>
__device__ __forceinline__ scalar_t blockReduceSum(scalar_t val) {
    __shared__ scalar_t shared[32];
    int lane = threadIdx.x % WARP_SIZE;
    int wid = threadIdx.x / WARP_SIZE;
    
    val = warpReduceSum(val);
    
    if (lane == 0) {
        shared[wid] = val;
    }
    __syncthreads();
    
    if (wid == 0) {
        val = (threadIdx.x < (blockDim.x + WARP_SIZE - 1) / WARP_SIZE) ? shared[lane] : scalar_t(0);
        val = warpReduceSum(val);
    }
    return val;
}

// Attention kernel with bias - one thread block per (batch, head, query_position)
template <typename scalar_t>
__global__ void attention_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ rel_bias,
    const float* __restrict__ edge_mask,
    scalar_t* __restrict__ ctx,
    int B, int N, int num_heads, int head_dim, float scale) {
    
    int b = blockIdx.z;
    int h = blockIdx.y;
    int q_idx = blockIdx.x;
    int tid = threadIdx.x;
    int block_size = blockDim.x;
    
    const scalar_t* q_ptr = q + ((b * num_heads + h) * N + q_idx) * head_dim;
    const scalar_t* k_ptr = k + (b * num_heads + h) * N * head_dim;
    const scalar_t* v_ptr = v + (b * num_heads + h) * N * head_dim;
    const scalar_t* bias_ptr = rel_bias + ((b * num_heads + h) * N + q_idx) * N;
    const float* mask_ptr = edge_mask + (b * N + q_idx) * N;
    scalar_t* out_ptr = ctx + ((b * num_heads + h) * N + q_idx) * head_dim;
    
    // Shared memory layout: Q[head_dim], scores[N], attn[N]
    extern __shared__ char smem[];
    scalar_t* s_q = (scalar_t*)smem;
    scalar_t* s_scores = s_q + head_dim;
    scalar_t* s_attn = s_scores + N;
    
    // Load Q into shared memory (all threads cooperate)
    for (int d = tid; d < head_dim; d += block_size) {
        s_q[d] = q_ptr[d];
    }
    __syncthreads();
    
    // Compute attention scores - each thread handles multiple KV positions
    for (int kv_idx = tid; kv_idx < N; kv_idx += block_size) {
        scalar_t score = 0;
        const scalar_t* k_row = k_ptr + kv_idx * head_dim;
        for (int d = 0; d < head_dim; d++) {
            score += s_q[d] * k_row[d];
        }
        score = score * scale + bias_ptr[kv_idx];
        
        // Apply mask
        if (mask_ptr[kv_idx] < 0.5f) {
            s_scores[kv_idx] = scalar_t(-INFINITY);
        } else {
            s_scores[kv_idx] = score;
        }
    }
    __syncthreads();
    
    // Softmax: find max for stability
    scalar_t max_score = scalar_t(-INFINITY);
    for (int i = tid; i < N; i += block_size) {
        max_score = (s_scores[i] > max_score) ? s_scores[i] : max_score;
    }
    max_score = blockReduceMax<scalar_t>(max_score);
    __shared__ scalar_t s_max;
    if (tid == 0) s_max = max_score;
    __syncthreads();
    
    // Compute exp(score - max) and sum
    scalar_t exp_sum = 0;
    for (int i = tid; i < N; i += block_size) {
        if (s_scores[i] > scalar_t(-1e30f)) {  // Not masked
            s_attn[i] = expf(s_scores[i] - s_max);
            exp_sum += s_attn[i];
        } else {
            s_attn[i] = 0;
        }
    }
    exp_sum = blockReduceSum<scalar_t>(exp_sum);
    __shared__ scalar_t s_sum;
    if (tid == 0) s_sum = exp_sum;
    __syncthreads();
    
    // Normalize attention weights
    scalar_t inv_sum = scalar_t(1) / s_sum;
    for (int i = tid; i < N; i += block_size) {
        s_attn[i] = s_attn[i] * inv_sum;
    }
    __syncthreads();
    
    // Compute weighted sum of V - each thread handles some output dimensions
    for (int d = tid; d < head_dim; d += block_size) {
        scalar_t val = 0;
        for (int i = 0; i < N; i++) {
            const scalar_t* v_row = v_ptr + i * head_dim;
            val += s_attn[i] * v_row[d];
        }
        out_ptr[d] = val;
    }
}

// Rel MLP kernel: Linear -> SiLU -> Linear
template <typename scalar_t>
__global__ void rel_mlp_kernel(
    const scalar_t* __restrict__ pair_feats,
    const scalar_t* __restrict__ weight1,
    const scalar_t* __restrict__ weight2,
    scalar_t* __restrict__ output,
    int B, int N, int P, int num_heads) {
    
    int idx = blockIdx.x;
    int tid = threadIdx.x;
    int block_size = blockDim.x;
    
    int b = idx / (N * N);
    int rem = idx % (N * N);
    int i = rem / N;
    int j = rem % N;
    
    if (b >= B) return;
    
    const scalar_t* feat_ptr = pair_feats + idx * P;
    
    // First linear: P -> P with SiLU
    extern __shared__ char shared_mem[];
    scalar_t* intermediate = (scalar_t*)shared_mem;
    
    for (int p = tid; p < P; p += block_size) {
        scalar_t sum = 0;
        for (int k = 0; k < P; k++) {
            sum += feat_ptr[k] * weight1[p * P + k];
        }
        // SiLU activation: x * sigmoid(x)
        scalar_t sigmoid_val = scalar_t(1) / (scalar_t(1) + expf(-sum));
        intermediate[p] = sum * sigmoid_val;
    }
    __syncthreads();
    
    // Second linear: P -> num_heads
    for (int h = tid; h < num_heads; h += block_size) {
        scalar_t sum = 0;
        for (int k = 0; k < P; k++) {
            sum += intermediate[k] * weight2[h * P + k];
        }
        output[((b * num_heads + h) * N + i) * N + j] = sum;
    }
}

class ModelNew(nn.Module):
    def __init__(self, dim: int, pair_dim: int, num_heads: int):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.scale = 1.0 / math.sqrt(self.head_dim)

        self.norm = nn.LayerNorm(dim)
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.rel_mlp = nn.Sequential(
            nn.Linear(pair_dim, pair_dim),
            nn.SiLU(),
            nn.Linear(pair_dim, num_heads),
        )
        self.out_proj = nn.Linear(dim, dim, bias=False)

    def forward(
        self,
        nodes: torch.Tensor,
        pair_feats: torch.Tensor,
        edge_mask: torch.Tensor,
    ) -> torch.Tensor:
        B, N, D = nodes.shape
        
        h = self.norm(nodes)
        
        q = self.q_proj(h).view(B, N, self.num_heads, self.head_dim).permute(0, 2, 1, 3)
        k = self.k_proj(h).view(B, N, self.num_heads, self.head_dim).permute(0, 2, 1, 3)
        v = self.v_proj(h).view(B, N, self.num_heads, self.head_dim).permute(0, 2, 1, 3)
        
        rel_weight1 = self.rel_mlp[0].weight
        rel_weight2 = self.rel_mlp[2].weight
        
        ctx = gf_vla_ops.forward_cuda(
            h.contiguous(),
            q.contiguous(),
            k.contiguous(),
            v.contiguous(),
            pair_feats.contiguous(),
            edge_mask.contiguous(),
            rel_weight1.contiguous(),
            rel_weight2.contiguous(),
            self.num_heads,
            self.scale
        )
        
        ctx = ctx.permute(0, 2, 1, 3).reshape(B, N, D)
        out = self.out_proj(ctx)
        
        return nodes + out
