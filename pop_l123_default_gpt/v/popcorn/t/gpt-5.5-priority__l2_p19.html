import torch
import torch.nn as nn

class Model(nn.Module):
    """
    1D depthwise separable convolution block: pointwise Conv1d (channel mixing), depthwise Conv1d (spatial),
    GELU activation, and residual addition. Used in genomic foundation models (Enformer) for efficient
    sequence processing of DNA/RNA.
    """
    def __init__(self, channels, expanded_channels, kernel_size):
        super(Model, self).__init__()
        self.pointwise = nn.Conv1d(channels, expanded_channels, 1)
        self.depthwise = nn.Conv1d(expanded_channels, expanded_channels, kernel_size,
                                   padding=kernel_size // 2, groups=expanded_channels)
        self.activation = nn.GELU()
        self.project_back = nn.Conv1d(expanded_channels, channels, 1)

    def forward(self, x):
        residual = x
        x = self.pointwise(x)
        x = self.depthwise(x)
        x = self.activation(x)
        x = self.project_back(x)
        x = x + residual
        return x

__global__ __launch_bounds__(256) void depthwise_gelu_tiled_kernel(const float* __restrict__ z,
                                      const float* __restrict__ pw_b,
                                      const float* __restrict__ dw_w,
                                      const float* __restrict__ dw_b,
                                      float* __restrict__ a,
                                      int B, int E, int L) {
    __shared__ float sh[264];
    int tx = threadIdx.x;
    int tile = blockIdx.x;
    int e = blockIdx.y;
    int b = blockIdx.z;
    int start = tile * 256;
    int l = start + tx;
    const float* zbase = z + ((long long)b * E + e) * (long long)L;
    sh[tx + 4] = (l < L) ? zbase[l] : 0.0f;
    if (tx < 4) {
        int ll = start + tx - 4;
        sh[tx] = (ll >= 0) ? zbase[ll] : 0.0f;
        int rr = start + 256 + tx;
        sh[260 + tx] = (rr < L) ? zbase[rr] : 0.0f;
    }
    __syncthreads();
    if (l >= L) return;
    const float* w = dw_w + e * 9;
    float pbi = pw_b[e];
    float sum = dw_b[e];
    int ntiles = (L + 255) >> 8;
    if (tile != 0 && tile != ntiles - 1) {
        sum = fmaf(sh[tx+0] + pbi, w[0], sum);
        sum = fmaf(sh[tx+1] + pbi, w[1], sum);
        sum = fmaf(sh[tx+2] + pbi, w[2], sum);
        sum = fmaf(sh[tx+3] + pbi, w[3], sum);
        sum = fmaf(sh[tx+4] + pbi, w[4], sum);
        sum = fmaf(sh[tx+5] + pbi, w[5], sum);
        sum = fmaf(sh[tx+6] + pbi, w[6], sum);
        sum = fmaf(sh[tx+7] + pbi, w[7], sum);
        sum = fmaf(sh[tx+8] + pbi, w[8], sum);
    } else {
        #pragma unroll
        for (int kk = 0; kk < 9; ++kk) {
            int ll = l + kk - 4;
            if ((unsigned)ll < (unsigned)L) sum = fmaf(sh[tx + kk] + pbi, w[kk], sum);
        }
    }
    float y = 0.5f * sum * (1.0f + erff(sum * 0.70710678118654752440f));
    a[((long long)b * E + e) * (long long)L + l] = y;
}

__global__ __launch_bounds__(256) void init_out_kernel(const float* __restrict__ x, const float* __restrict__ pb_b, float* __restrict__ out, int B, int C, int L) {
    long long idx = blockIdx.x * (long long)blockDim.x + threadIdx.x;
    long long total = (long long)B * C * L;
    if (idx >= total) return;
    int c = (idx / L) & 255;
    out[idx] = x[idx] + pb_b[c];
}

torch::Tensor forward(torch::Tensor x, torch::Tensor pw_w, torch::Tensor pw_b, torch::Tensor dw_w, torch::Tensor dw_b, torch::Tensor pb_w, torch::Tensor pb_b) {

    c10::cuda::CUDAGuard device_guard(x.device());
    int B=(int)x.size(0), C=(int)x.size(1), L=(int)x.size(2), E=(int)pw_w.size(0);
    auto z=torch::empty({B,E,L}, x.options());
    auto a=torch::empty({B,E,L}, x.options());
    auto out=torch::empty({B,C,L}, x.options());
    cublasHandle_t handle=at::cuda::getCurrentCUDABlasHandle();
    cudaStream_t stream=at::cuda::getCurrentCUDAStream();

    cublas_check(cublasSetMathMode(handle, CUBLAS_TF32_TENSOR_OP_MATH));
    const float alpha=1.0f, beta0=0.0f, beta1=1.0f;
    cublas_check(cublasGemmStridedBatchedEx(handle, CUBLAS_OP_N, CUBLAS_OP_N, L, E, C, &alpha,
        x.data_ptr<float>(), CUDA_R_32F, L, (long long)C*L,
        pw_w.data_ptr<float>(), CUDA_R_32F, C, 0, &beta0,
        z.data_ptr<float>(), CUDA_R_32F, L, (long long)E*L, B,
        CUBLAS_COMPUTE_32F_FAST_TF32, CUBLAS_GEMM_DEFAULT_TENSOR_OP));
    depthwise_gelu_tiled_kernel<<<dim3((L+255)/256, E, B), 256, 0, stream>>>(z.data_ptr<float>(), pw_b.data_ptr<float>(), dw_w.data_ptr<float>(), dw_b.data_ptr<float>(), a.data_ptr<float>(), B, E, L);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    long long total_o=(long long)B*C*L;
    init_out_kernel<<<(int)((total_o+255)/256), 256, 0, stream>>>(x.data_ptr<float>(), pb_b.data_ptr<float>(), out.data_ptr<float>(), B, C, L);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    cublas_check(cublasGemmStridedBatchedEx(handle, CUBLAS_OP_N, CUBLAS_OP_N, L, C, E, &alpha,
        a.data_ptr<float>(), CUDA_R_32F, L, (long long)E*L,
        pb_w.data_ptr<float>(), CUDA_R_32F, E, 0, &beta1,
        out.data_ptr<float>(), CUDA_R_32F, L, (long long)C*L, B,
        CUBLAS_COMPUTE_32F_FAST_TF32, CUBLAS_GEMM_DEFAULT_TENSOR_OP));
    return out;
}
