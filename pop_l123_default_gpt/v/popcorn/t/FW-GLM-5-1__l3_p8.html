import torch
import torch.nn as nn
import torch.nn.functional as F
import math


class MSARowSelfAttention(nn.Module):
    def __init__(self, msa_dim, num_heads):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        self.norm = nn.LayerNorm(msa_dim)
        self.q = nn.Linear(msa_dim, msa_dim, bias=False)
        self.k = nn.Linear(msa_dim, msa_dim, bias=False)
        self.v = nn.Linear(msa_dim, msa_dim, bias=False)
        self.out = nn.Linear(msa_dim, msa_dim)

    def forward(self, msa):
        B, S, N, D = msa.shape
        h, d = self.num_heads, self.head_dim
        x = self.norm(msa)
        q = self.q(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        k = self.k(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = self.v(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        attn = (q @ k.transpose(-1, -2)) / math.sqrt(d)
        attn = F.softmax(attn, dim=-1)
        out = (attn @ v).permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        return self.out(out)


class OuterProductMeanSmall(nn.Module):
    def __init__(self, msa_dim, pair_dim, hidden=32):
        super().__init__()
        self.norm = nn.LayerNorm(msa_dim)
        self.a = nn.Linear(msa_dim, hidden)
        self.b = nn.Linear(msa_dim, hidden)
        self.out = nn.Linear(hidden * hidden, pair_dim)

    def forward(self, msa):
        x = self.norm(msa)
        a = self.a(x)
        b = self.b(x)
        outer = torch.einsum("bsih,bsjk->bijhk", a, b)
        S = msa.shape[1]
        outer = outer / max(S, 1)
        B, N1, N2, H1, H2 = outer.shape
        return self.out(outer.reshape(B, N1, N2, H1 * H2))


class TriMulOut(nn.Module):
    def __init__(self, pair_dim, hidden):
        super().__init__()
        self.norm = nn.LayerNorm(pair_dim)
        self.pa = nn.Linear(pair_dim, hidden)
        self.pb = nn.Linear(pair_dim, hidden)
        self.ga = nn.Linear(pair_dim, hidden)
        self.gb = nn.Linear(pair_dim, hidden)
        self.norm_out = nn.LayerNorm(hidden)
        self.out = nn.Linear(hidden, pair_dim)
        self.gate_out = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair):
        x = self.norm(pair)
        a = self.pa(x) * torch.sigmoid(self.ga(x))
        b = self.pb(x) * torch.sigmoid(self.gb(x))
        o = torch.einsum("bikh,bjkh->bijh", a, b)
        o = self.out(self.norm_out(o))
        return torch.sigmoid(self.gate_out(pair)) * o

        h, d = self.num_heads, self.head_dim
        
        # Ensure tensors are contiguous and on GPU
        msa_repr = msa_repr.contiguous()
        pair_repr = pair_repr.contiguous()
        
        # === MSA Row Self Attention ===
        msa_input = msa_repr
        
        # Layer norm on MSA
        msa_flat = msa_input.reshape(B * S * N, D)
        msa_normed = torch.empty_like(msa_flat)
        self.ext.launch_layernorm(msa_normed, msa_flat, self.msa_norm.weight, self.msa_norm.bias)
        msa_normed = msa_normed.reshape(B, S, N, D)
        
        # Q, K, V projections
        q = F.linear(msa_normed, self.q.weight)
        k = F.linear(msa_normed, self.k.weight)
        v = F.linear(msa_normed, self.v.weight)
        
        # Reshape for attention
        q = q.reshape(B, S, N, h, d).permute(0, 1, 3, 2, 4)  # (B, S, h, N, d)
        k = k.reshape(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = v.reshape(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        
        # Attention scores
        scale = 1.0 / math.sqrt(d)
        attn = (q @ k.transpose(-1, -2)) * scale  # (B, S, h, N, N)
        
        # Softmax
        attn_flat = attn.reshape(-1, N)
        attn_out = torch.empty_like(attn_flat)
        self.ext.launch_softmax(attn_out, attn_flat)
        attn = attn_out.reshape(B, S, h, N, N)
        
        # Attention output
        out = attn @ v  # (B, S, h, N, d)
        out = out.permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        
        # Output projection
        msa_attn_out = F.linear(out, self.msa_out.weight, self.msa_out.bias)
        
        # Residual add
        msa_repr = msa_input + msa_attn_out
        
        # === Outer Product Mean ===
        pair_input = pair_repr
        
        # Layer norm on MSA
        msa_flat = msa_repr.reshape(B * S * N, D)
        msa_normed = torch.empty_like(msa_flat)
        self.ext.launch_layernorm(msa_normed, msa_flat, self.opm_norm.weight, self.opm_norm.bias)
        msa_normed = msa_normed.reshape(B, S, N, D)
        
        # a and b projections
        a = F.linear(msa_normed, self.a.weight, self.a.bias)  # (B, S, N, 32)
        b = F.linear(msa_normed, self.b.weight, self.b.bias)
        
        # Outer product: einsum("bsih,bsjk->bijhk", a, b)
        outer = torch.einsum("bsih,bsjk->bijhk", a, b)
        outer = outer / max(S, 1)
        
        # Reshape and output projection
        B_out, N1, N2, H1, H2 = outer.shape
        outer_flat = outer.reshape(B_out, N1, N2, H1 * H2)
        opm_out = F.linear(outer_flat, self.opm_out.weight, self.opm_out.bias)
        
        # Residual add
        pair_repr = pair_input + opm_out
        
        # === Triangular Multiplicative Update ===
        pair_input2 = pair_repr
        
        # Layer norm
        pair_flat = pair_repr.reshape(B * N * N, self.pair_dim)
        pair_normed = torch.empty_like(pair_flat)
        self.ext.launch_layernorm(pair_normed, pair_flat, self.tri_norm.weight, self.tri_norm.bias)
        pair_normed = pair_normed.reshape(B, N, N, self.pair_dim)
        
        # Projections with gating
        pa_out = F.linear(pair_normed, self.pa.weight, self.pa.bias)
        ga_out = F.linear(pair_normed, self.ga.weight, self.ga.bias)
        pb_out = F.linear(pair_normed, self.pb.weight, self.pb.bias)
        gb_out = F.linear(pair_normed, self.gb.weight, self.gb.bias)
        
        # Apply sigmoid gates
        a = torch.empty_like(pa_out)
        self.ext.launch_sigmoid_multiply(a, pa_out, ga_out)
        
        b = torch.empty_like(pb_out)
        self.ext.launch_sigmoid_multiply(b, pb_out, gb_out)
        
        # einsum("bikh,bjkh->bijh", a, b)
        o = torch.einsum("bikh,bjkh->bijh", a, b)
        
        # Layer norm output
        o_flat = o.reshape(B * N * N, self.tri_hidden)
        o_normed = torch.empty_like(o_flat)
        self.ext.launch_layernorm(o_normed, o_flat, self.tri_norm_out.weight, self.tri_norm_out.bias)
        o_normed = o_normed.reshape(B, N, N, self.tri_hidden)
        
        # Output projection
        tri_out = F.linear(o_normed, self.tri_out.weight, self.tri_out.bias)
        
        # Gate output
        gate = torch.sigmoid(F.linear(pair_input2, self.gate_out.weight, self.gate_out.bias))
        tri_out = gate * tri_out
        
        # Residual add
        pair_repr = pair_input2 + tri_out
        
        return msa_repr, pair_repr
