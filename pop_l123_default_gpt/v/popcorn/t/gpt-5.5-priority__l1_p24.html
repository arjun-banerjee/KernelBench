import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Voxel-grid pooling for 3-D molecular point clouds.  Discretizes
    continuous 3-D coordinates into a regular grid and pools per-atom
    features into each voxel.  Used in 3-D-CNN approaches for molecular
    property prediction and ligand docking (e.g., PointVoxel-CNN,
    3D-Infomax).
    """

    def __init__(self, feat_dim, grid_size=32, voxel_size=1.0):
        super().__init__()
        self.feat_dim = feat_dim
        self.grid_size = grid_size
        self.voxel_size = voxel_size
        self.linear = nn.Linear(feat_dim, feat_dim)

    def forward(
        self, coords: torch.Tensor, features: torch.Tensor
    ) -> torch.Tensor:
        """
        Args:
            coords:   (B, N, 3) – atom coordinates
            features: (B, N, feat_dim) – per-atom features
        Returns:
            voxel_grid: (B, feat_dim, G, G, G) – pooled voxel features
        """
        B, N, _ = coords.shape
        G = self.grid_size

        features = self.linear(features)

        # Center coordinates and discretize
        center = coords.mean(dim=1, keepdim=True)
        shifted = (coords - center) / self.voxel_size + G / 2.0
        indices = shifted.long().clamp(0, G - 1)  # (B, N, 3)

        voxel_grid = coords.new_zeros(B, self.feat_dim, G, G, G)
        ix = indices[:, :, 0]
        iy = indices[:, :, 1]
        iz = indices[:, :, 2]

        batch_idx = torch.arange(B, device=coords.device).unsqueeze(1).expand(-1, N)
        for d in range(self.feat_dim):
            feat_d = features[:, :, d]
            voxel_grid[:, d].index_put_(
                (batch_idx.reshape(-1), ix.reshape(-1), iy.reshape(-1), iz.reshape(-1)),
                feat_d.reshape(-1),
                accumulate=True,
            )

        return voxel_grid


feat_dim = 32
grid_size = 32
voxel_size = 1.0
num_atoms = 256
batch_size = 4


def get_inputs():
    coords = torch.randn(batch_size, num_atoms, 3) * 5.0
    features = torch.randn(batch_size, num_atoms, feat_dim)
    return [coords, features]


def get_init_inputs():
    return [feat_dim, grid_size, voxel_size]

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void center_kernel(const float* __restrict__ coords, float* __restrict__ centers, int B, int N) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    float sx = 0.f, sy = 0.f, sz = 0.f;
    for (int n = tid; n < N; n += blockDim.x) {
        const float* p = coords + ((b * N + n) * 3);
        sx += p[0]; sy += p[1]; sz += p[2];
    }
    __shared__ float shx[256], shy[256], shz[256];
    shx[tid] = sx; shy[tid] = sy; shz[tid] = sz;
    __syncthreads();
    for (int s = blockDim.x >> 1; s > 0; s >>= 1) {
        if (tid < s) {
            shx[tid] += shx[tid+s];
            shy[tid] += shy[tid+s];
            shz[tid] += shz[tid+s];
        }
        __syncthreads();
    }
    if (tid == 0) {
        float invN = 1.0f / (float)N;
        centers[b*3+0] = shx[0] * invN;
        centers[b*3+1] = shy[0] * invN;
        centers[b*3+2] = shz[0] * invN;
    }
}

__device__ __forceinline__ int clamp_index(float v, int G) {
    int idx = (int)v; // truncates toward zero, matching torch.Tensor.long()
    if (idx < 0) idx = 0;
    if (idx >= G) idx = G - 1;
    return idx;
}

__global__ void scatter_linear_kernel(const float* __restrict__ coords,
                                      const float* __restrict__ features,
                                      const float* __restrict__ weight,
                                      const float* __restrict__ bias,
                                      const float* __restrict__ centers,
                                      float* __restrict__ out,
                                      int B, int N, int D, int G, float inv_voxel) {
    int atom = blockIdx.x;
    int d = threadIdx.x;
    if (atom >= B * N || d >= D) return;
    int b = atom / N;
    int n = atom - b * N;

    const float* cp = coords + ((b * N + n) * 3);
    float gx = (cp[0] - centers[b*3+0]) * inv_voxel + 0.5f * (float)G;
    float gy = (cp[1] - centers[b*3+1]) * inv_voxel + 0.5f * (float)G;
    float gz = (cp[2] - centers[b*3+2]) * inv_voxel + 0.5f * (float)G;
    int ix = clamp_index(gx, G);
    int iy = clamp_index(gy, G);
    int iz = clamp_index(gz, G);

    const float* feat = features + (atom * D);
    const float* wrow = weight + (d * D);
    float acc = bias[d];
    for (int k = 0; k < D; ++k) {
        acc = fmaf(feat[k], wrow[k], acc);
    }
    long long offset = (((long long)b * D + d) * G + ix);
    offset = (offset * G + iy) * G + iz;
    atomicAdd(out + offset, acc);
}

torch::Tensor voxel_pool_forward_cuda(torch::Tensor coords, torch::Tensor features, torch::Tensor weight, torch::Tensor bias, int64_t grid_size, double voxel_size) {
    const int B = (int)coords.size(0);
    const int N = (int)coords.size(1);
    const int D = (int)features.size(2);
    const int G = (int)grid_size;

    auto out = torch::empty({B, D, G, G, G}, coords.options());
    auto centers = torch::empty({B, 3}, coords.options());

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    size_t bytes = (size_t)B * D * G * G * G * sizeof(float);
    cudaMemsetAsync(out.data_ptr<float>(), 0, bytes, stream);

    center_kernel<<<B, 256, 0, stream>>>(coords.data_ptr<float>(), centers.data_ptr<float>(), B, N);

    int threads = 32;
    if (D > 32) threads = 64;
    if (D > 64) threads = 128;
    if (D > 128) threads = 256;
    scatter_linear_kernel<<<B * N, threads, 0, stream>>>(coords.data_ptr<float>(), features.data_ptr<float>(), weight.data_ptr<float>(), bias.data_ptr<float>(), centers.data_ptr<float>(), out.data_ptr<float>(), B, N, D, G, (float)(1.0 / voxel_size));
    return out;
}
