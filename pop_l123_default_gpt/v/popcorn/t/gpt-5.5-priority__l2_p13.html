def _fake_quant_symmetric_int8(x: torch.Tensor, scale: torch.Tensor) -> torch.Tensor:
    """Symmetric int8 fake-quant: round(x/scale) clamped to [-127, 127], times scale."""
    q = torch.clamp(torch.round(x / scale), -127.0, 127.0)
    return q * scale


class Model(nn.Module):
    """
    **QuantVLA selective layout fusion** (reference target for a fused CUDA/Triton kernel).

    Vision–language–action stacks often use **token-wise mixed precision**: some
    timesteps or modalities tolerate aggressive quantization while others stay in
    full precision. This module fuses in **one forward**:

      1. **Layout routing** — ``sigmoid(layout_logits)`` yields a per-token mix
         coefficient in ``(0, 1)`` (quant vs FP residual).
      2. **Quantitative path** — global symmetric **int8-style fake quantization**
         (scale from the tensor max, ``/ 127``).
      3. **Selective blend** — ``α · x_quant + (1 - α) · x``.
      4. **Head** — linear projection on the mixed activations.

    This matches a “quantitative selective layout” block: the layout logits (from a
    router, modality tags, or calibration) choose how strongly each position uses the
    quantized layout vs the identity layout.
    """

    def __init__(self, dim: int):
        super().__init__()
        self.dim = dim
        self.out_proj = nn.Linear(dim, dim, bias=True)

    def forward(self, x: torch.Tensor, layout_logits: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, T, D) VLA token activations (fused modalities).
            layout_logits: (B, T) unbounded logits; larger → more mass on quantized path.
        Returns:
            (B, T, D) updated activations.
        """
        B, T, D = x.shape
        assert D == self.dim
        assert layout_logits.shape == (B, T)

        alpha = torch.sigmoid(layout_logits).unsqueeze(-1)
        with torch.no_grad():
            amax = x.detach().abs().max().clamp(min=1e-8)
        scale = amax / 127.0
        x_q = _fake_quant_symmetric_int8(x, scale)
        mixed = alpha * x_q + (1.0 - alpha) * x
        return self.out_proj(mixed)


class ModelNew(nn.Module):
    """KernelBench candidate entry point; default delegates to ``Model``."""

    def __init__(self, dim: int):
        super().__init__()
        self._impl = Model(dim)

    def forward(self, x: torch.Tensor, layout_logits: torch.Tensor) -> torch.Tensor:
        return self._impl(x, layout_logits)

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void reduce_amax_kernel(const float* __restrict__ x, float* __restrict__ scale, int n) {
    __shared__ float s[256];
    int tid = threadIdx.x;
    float m = 0.0f;
    for (int i = tid; i < n; i += 256) {
        m = fmaxf(m, fabsf(x[i]));
    }
    s[tid] = m;
    __syncthreads();
    for (int st = 128; st > 0; st >>= 1) {
        if (tid < st) s[tid] = fmaxf(s[tid], s[tid + st]);
        __syncthreads();
    }
    if (tid == 0) scale[0] = fmaxf(s[0], 1.0e-8f) * (1.0f / 127.0f);
}

__global__ void preprocess_rows_kernel(const float* __restrict__ x,
                                        const float* __restrict__ logits,
                                        const float* __restrict__ scale,
                                        float* __restrict__ mixed,
                                        int M) {
    __shared__ float a, sc;
    int row = blockIdx.x;
    int k = threadIdx.x;
    if (k == 0) {
        float l = logits[row];
        a = 1.0f / (1.0f + expf(-l));
        sc = scale[0];
    }
    __syncthreads();
    int idx = row * 128 + k;
    float xv = x[idx];
    float q = nearbyintf(xv / sc);
    q = fminf(127.0f, fmaxf(-127.0f, q)) * sc;
    mixed[idx] = xv + a * (q - xv);
}

#define TILE 16
__global__ void matmul_kernel(const float* __restrict__ A,
                              const float* __restrict__ W,
                              const float* __restrict__ bias,
                              float* __restrict__ C,
                              int M) {
    __shared__ float As[TILE][TILE];
    __shared__ float Bs[TILE][TILE];
    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int row = blockIdx.y * TILE + ty;
    int col = blockIdx.x * TILE + tx;
    float acc = 0.0f;

    #pragma unroll
    for (int t = 0; t < 128; t += TILE) {
        As[ty][tx] = (row < M) ? A[row * 128 + t + tx] : 0.0f;
        int wcol = blockIdx.x * TILE + ty;
        Bs[tx][ty] = W[wcol * 128 + t + tx];
        __syncthreads();
        #pragma unroll
        for (int k = 0; k < TILE; ++k) {
            acc = fmaf(As[ty][k], Bs[k][tx], acc);
        }
        __syncthreads();
    }
    if (row < M) C[row * 128 + col] = acc + bias[col];
}

torch::Tensor quantvla_forward(torch::Tensor x, torch::Tensor layout_logits, torch::Tensor weight, torch::Tensor bias) {
    int M = x.size(0) * x.size(1);
    auto out = torch::empty_like(x);
    auto mixed = torch::empty_like(x);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    // Store the scalar scale in the first element of out; it is overwritten by matmul later.
    reduce_amax_kernel<<<1, 256, 0, stream>>>(x.data_ptr<float>(), out.data_ptr<float>(), M * 128);
    preprocess_rows_kernel<<<M, 128, 0, stream>>>(x.data_ptr<float>(), layout_logits.data_ptr<float>(), out.data_ptr<float>(), mixed.data_ptr<float>(), M);

    dim3 threads(TILE, TILE);
    dim3 grid(8, (M + TILE - 1) / TILE);
    matmul_kernel<<<grid, threads, 0, stream>>>(mixed.data_ptr<float>(), weight.data_ptr<float>(), bias.data_ptr<float>(), out.data_ptr<float>(), M);
    return out;
}
