import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Dilated convolution tower with exponentially increasing dilation rates and residual connections.
    Each block applies LayerNorm -> GELU -> DilatedConv1d, then adds back the original input as a residual.
    This architecture is a core building block of genomic foundation models (Enformer, Basenji, Caduceus)
    for capturing multi-scale sequence dependencies in DNA/RNA.
    """
    def __init__(self, channels, seq_len, kernel_size, max_dilation_power):
        super(Model, self).__init__()
        self.norms = nn.ModuleList()
        self.convs = nn.ModuleList()
        self.activation = nn.GELU()
        for i in range(max_dilation_power + 1):
            dilation = 2 ** i
            self.norms.append(nn.LayerNorm([channels, seq_len]))
            self.convs.append(nn.Conv1d(channels, channels, kernel_size, padding=dilation, dilation=dilation))

    def forward(self, x):
        residual = x
        for norm, conv in zip(self.norms, self.convs):
            x = norm(x)
            x = self.activation(x)
            x = conv(x)
            x = x + residual
        return x

def get_inputs():

def get_init_inputs():
    return [channels, seq_len, kernel_size, max_dilation_power]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r'''
#include <torch/extension.h>
#include <vector>

void add_launch(const torch::Tensor& a, const torch::Tensor& b, torch::Tensor& out, int64_t total);

torch::Tensor forward_ext(torch::Tensor x,
                          std::vector<torch::Tensor> norm_ws,
                          std::vector<torch::Tensor> norm_bs,
                          std::vector<torch::Tensor> conv_ws,
                          std::vector<torch::Tensor> conv_bs,
                          int64_t max_dilation_power) {
    const int64_t B = x.size(0);
    const int64_t C = x.size(1);
    const int64_t L = x.size(2);
    const int64_t total = B * C * L;

    torch::Tensor residual = x;
    torch::Tensor cur = x;
    torch::Tensor sum, norm, act, conv_out;

    // Keep the whole tower inside one C++ call to avoid repeated Python dispatch.
    // The residual add is delayed from the end of block i to the input of block i+1,
    // which is algebraically identical and lets us use one simple CUDA add kernel.
    for (int64_t i = 0; i <= max_dilation_power; ++i) {
        if (i == 0) {
            sum = cur;
        } else {
            sum = torch::empty_like(cur);
            add_launch(cur, residual, sum, total);
        }
        norm = torch::layer_norm(sum, {C, L}, norm_ws[i], norm_bs[i], 1.0e-5, false);
        act = torch::gelu(norm);
        const int64_t dilation = 1LL << i;
        conv_out = torch::conv1d(act, conv_ws[i], conv_bs[i], {1}, {dilation}, {dilation}, 1);
        cur = conv_out;
    }

    torch::Tensor out = torch::empty_like(conv_out);
    add_launch(conv_out, residual, out, total);
    return out;
}
'''
