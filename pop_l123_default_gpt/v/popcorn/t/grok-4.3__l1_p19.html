import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues
        Returns:
            dist: (B, N, N) – pairwise Euclidean distance matrix
        """
        diff = coords.unsqueeze(2) - coords.unsqueeze(1)  # (B, N, N, 3)
        dist = torch.sqrt((diff ** 2).sum(dim=-1) + 1e-8)
        return dist

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void pairwise_dist_kernel(const float* coords, float* dist, int B, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * N * N;
    if (idx < total) {
        int b = idx / (N * N);
        int rem = idx % (N * N);
        int i = rem / N;
        int j = rem % N;
        float dx = coords[b * N * 3 + i * 3 + 0] - coords[b * N * 3 + j * 3 + 0];
        float dy = coords[b * N * 3 + i * 3 + 1] - coords[b * N * 3 + j * 3 + 1];
        float dz = coords[b * N * 3 + i * 3 + 2] - coords[b * N * 3 + j * 3 + 2];
        float d2 = dx*dx + dy*dy + dz*dz;
        dist[b * N * N + i * N + j] = sqrtf(d2 + 1e-8f);
    }
}

torch::Tensor pairwise_dist_cuda(torch::Tensor coords) {
    int B = coords.size(0);
    int N = coords.size(1);
    auto dist = torch::empty({B, N, N}, coords.options());
    int total = B * N * N;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    pairwise_dist_kernel<<<blocks, threads>>>(
        coords.data_ptr<float>(),
        dist.data_ptr<float>(),
        B, N
    );
    return dist;
}
'''
