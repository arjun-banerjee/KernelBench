import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a depthwise 1D convolution where each input channel is convolved with its own filter.
    Fundamental primitive in Hyena, Mamba, and genomic sequence models (Enformer, HyenaDNA).

    Args:
        channels (int): Number of input/output channels.
        kernel_size (int): Size of the convolution kernel.
        stride (int, optional): Stride of the convolution. Defaults to 1.
        padding (int, optional): Padding applied to the input. Defaults to 0.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """
    def __init__(self, channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(Model, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.conv1d(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void depthwise_conv1d_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size,
    int channels,
    int seq_len,
    int kernel_size,
    int stride,
    int padding
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * channels * seq_len;
    if (idx >= total) return;

    int out_pos = idx % seq_len;
    int ch = (idx / seq_len) % channels;
    int b = idx / (seq_len * channels);

    float sum = 0.0f;
    for (int k = 0; k < kernel_size; ++k) {
        int in_pos = out_pos * stride + k - padding;
        if (in_pos >= 0 && in_pos < seq_len) {
            sum += input[b * channels * seq_len + ch * seq_len + in_pos] * weight[ch * kernel_size + k];
        }
    }
    if (bias != nullptr) {
        sum += bias[ch];
    }
    output[idx] = sum;
}

torch::Tensor depthwise_conv1d_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias,
    int stride,
    int padding
) {
    int batch_size = input.size(0);
    int channels = input.size(1);
    int seq_len = input.size(2);
    int kernel_size = weight.size(2);

    auto output = torch::empty({batch_size, channels, seq_len}, input.options());

    int total = batch_size * channels * seq_len;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;

    const float* bias_ptr = nullptr;
    if (bias.defined() && bias.numel() > 0) {
        bias_ptr = bias.data_ptr<float>();
    }

    depthwise_conv1d_kernel<<<blocks, threads>>>(
        input.data_ptr<float>(),

class ModelNew(nn.Module):
    def __init__(self, channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(ModelNew, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        weight = self.conv1d.weight  # shape (channels, 1, kernel_size)
        bias = self.conv1d.bias if self.conv1d.bias is not None else torch.empty(0, device=x.device)
        return module.depthwise_conv1d_cuda(x, weight, bias, self.conv1d.stride[0], self.conv1d.padding[0])
