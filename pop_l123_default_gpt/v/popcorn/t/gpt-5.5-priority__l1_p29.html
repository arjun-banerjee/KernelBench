import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>
torch::Tensor logsumexp_forward(torch::Tensor x);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void logsumexp4_kernel(const float* __restrict__ x, float* __restrict__ out) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < 320) {
        float a0 = x[idx];
        float a1 = x[idx + 320];
        float a2 = x[idx + 640];
        float a3 = x[idx + 960];
        float m = fmaxf(fmaxf(a0, a1), fmaxf(a2, a3));
        float s = expf(a0 - m) + expf(a1 - m) + expf(a2 - m) + expf(a3 - m);
        out[idx] = m + logf(s);
    }
}

torch::Tensor logsumexp_forward(torch::Tensor x) {
    auto out = torch::empty({10, 32}, x.options());
    logsumexp4_kernel<<<2, 256, 0, at::cuda::getCurrentCUDAStream()>>>(x.data_ptr<float>(), out.data_ptr<float>());
    return out;
}
"""
