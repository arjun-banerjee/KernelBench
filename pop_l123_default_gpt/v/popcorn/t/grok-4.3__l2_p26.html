import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Apply rotary position embedding to incoming keys and update the KV cache.

    This isolates a common decode-time kernel: rotate the fresh key vectors for
    the current positions and store both keys and values into the running cache.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        k_new: torch.Tensor,
        v_new: torch.Tensor,
        cos: torch.Tensor,
        sin: torch.Tensor,
        cache_k: torch.Tensor,
        cache_v: torch.Tensor,
        positions: torch.Tensor,
    ) -> torch.Tensor:
        updated_k = cache_k.clone()
        updated_v = cache_v.clone()
        rot_dim = k_new.shape[-1]

        for b in range(k_new.shape[0]):
            for t in range(k_new.shape[1]):
                pos = int(positions[b, t].item())
                c = cos[pos]
                s = sin[pos]
                x = k_new[b, t]
                x_even = x[..., 0::2]
                x_odd = x[..., 1::2]
                rot_even = x_even * c - x_odd * s
                rot_odd = x_even * s + x_odd * c
                rotated = torch.empty_like(x)
                rotated[..., 0::2] = rot_even
                rotated[..., 1::2] = rot_odd
                updated_k[b, pos] = rotated
                updated_v[b, pos] = v_new[b, t]

        return torch.stack((updated_k, updated_v), dim=0)

def _rope_tables(cache_len: int, half_dim: int):
    positions = torch.arange(cache_len, dtype=torch.float32).unsqueeze(1)
    freqs = torch.pow(10000.0, -torch.arange(half_dim, dtype=torch.float32) / half_dim).unsqueeze(0)
    angles = positions * freqs
    return torch.cos(angles), torch.sin(angles)


def get_inputs():
    k_new = torch.randn(batch_size, update_len, num_heads, head_dim, dtype=torch.float32)
    v_new = torch.randn(batch_size, update_len, num_heads, head_dim, dtype=torch.float32)
    cos, sin = _rope_tables(cache_len, head_dim // 2)
    cache_k = torch.randn(batch_size, cache_len, num_heads, head_dim, dtype=torch.float32)
    cache_v = torch.randn(batch_size, cache_len, num_heads, head_dim, dtype=torch.float32)
    base = torch.arange(update_len, dtype=torch.int32).unsqueeze(0).repeat(batch_size, 1)
    positions = (base + torch.tensor([[7], [31]], dtype=torch.int32)) % cache_len
    return [k_new, v_new, cos, sin, cache_k, cache_v, positions]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void rope_kv_update_kernel(
    const float* __restrict__ k_new,
    const float* __restrict__ v_new,
    const float* __restrict__ cos,
    const float* __restrict__ sin,
    float* __restrict__ updated_k,
    float* __restrict__ updated_v,
    const int* __restrict__ positions,
    int B, int T_new, int L_cache, int H, int D
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total_new = B * T_new * H * D;
    if (idx >= total_new) return;

    int d = idx % D;
    int h = (idx / D) % H;
    int t = (idx / (D * H)) % T_new;
    int b = idx / (D * H * T_new);

    int pos = positions[b * T_new + t];
    int half_D = D / 2;
    int pair_idx = d / 2;
    float c = cos[pos * half_D + pair_idx];
    float s = sin[pos * half_D + pair_idx];

    float x = k_new[idx];
    float rot;
    if (d % 2 == 0) {
        int odd_idx = idx + 1;
        float x_odd = k_new[odd_idx];
        rot = x * c - x_odd * s;
    } else {
        int even_idx = idx - 1;
        float x_even = k_new[even_idx];
        rot = x_even * s + x * c;
    }
    int out_idx = ((b * L_cache + pos) * H + h) * D + d;
    updated_k[out_idx] = rot;
    int v_idx = ((b * T_new + t) * H + h) * D + d;
    updated_v[out_idx] = v_new[v_idx];
}

torch::Tensor rope_kv_update(
    torch::Tensor k_new,
    torch::Tensor v_new,
    torch::Tensor cos,
    torch::Tensor sin,
    torch::Tensor cache_k,
    torch::Tensor cache_v,
    torch::Tensor positions
) {
    auto B = k_new.size(0);
    auto T_new = k_new.size(1);
    auto H = k_new.size(2);
    auto D = k_new.size(3);
    auto L = cache_k.size(1);

    auto updated_k = cache_k.clone();
    auto updated_v = cache_v.clone();

    int total = B * T_new * H * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;

    rope_kv_update_kernel<<<blocks, threads>>>(
        k_new.data_ptr<float>(),
        v_new.data_ptr<float>(),
        cos.data_ptr<float>(),
        sin.data_ptr<float>(),
        updated_k.data_ptr<float>(),
        updated_v.data_ptr<float>(),
        positions.to(torch::kInt32).data_ptr<int>(),
        B, T_new, L, H, D
    );

    auto result = torch::stack({updated_k, updated_v}, 0);
    return result;
}
