class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        kernel_size: int,
        dilation: int = 1,
    ):
        super().__init__()
        assert kernel_size % 2 == 1, "odd kernel_size keeps T unchanged with symmetric padding"
        assert dilation >= 1
        self.padding = ((kernel_size - 1) * dilation) // 2
        self.dw = nn.Conv1d(
            in_channels,
            in_channels,
            kernel_size,
            padding=self.padding,
            dilation=dilation,
            groups=in_channels,
            bias=False,
        )
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

__global__ void depthwise_silu_kernel(const float* __restrict__ x,
                                      const float* __restrict__ w,
                                      float* __restrict__ tmp,
                                      int B, int C, int T, int K,
                                      int dilation, int padding) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * C * T;
    if (idx >= total) return;

    int t = idx % T;
    int q = idx / T;
    int c = q % C;
    int b = q / C;

    float sum = 0.0f;
    #pragma unroll
    for (int kk = 0; kk < 16; ++kk) {
        if (kk < K) {
            int it = t + kk * dilation - padding;
            if ((unsigned)it < (unsigned)T) {
                sum = fmaf(x[(b * C + c) * T + it], w[c * K + kk], sum);
            }
        }
    }
    tmp[idx] = sum / (1.0f + expf(-sum));
}

__global__ void pointwise_kernel_16x128(const float* __restrict__ tmp,
                                        const float* __restrict__ pw,

    __shared__ float Bs[16][128];

    int ty = threadIdx.y;

    int nbase = blockIdx.x * 128;
