import torch
import torch.nn as nn
import math


class Model(nn.Module):
    """
    Radial Basis Function (RBF) expansion of interatomic distances.
    Standard featurization in molecular graph neural networks (SchNet,
    DimeNet, GemNet) that converts scalar distances into a feature vector
    of Gaussian-basis evaluations.
    """

    def __init__(self, num_rbf=64, cutoff=5.0):
        super().__init__()
        self.num_rbf = num_rbf
        self.cutoff = cutoff
        offsets = torch.linspace(0.0, cutoff, num_rbf)
        self.register_buffer("offsets", offsets)
        coeff = -0.5 / (offsets[1] - offsets[0]).item() ** 2
        self.coeff = coeff

    def forward(self, distances: torch.Tensor) -> torch.Tensor:
        """
        Args:
            distances: (B, N, N) – pairwise distance matrix (or (B, E) edge list)
        Returns:
            rbf_features: (*distances.shape, num_rbf) – expanded features
        """
        # Cosine cutoff envelope
        d_scaled = distances.unsqueeze(-1)  # (..., 1)
        envelope = 0.5 * (torch.cos(math.pi * d_scaled / self.cutoff) + 1.0)
        envelope = envelope * (d_scaled < self.cutoff).float()

        rbf = torch.exp(self.coeff * (d_scaled - self.offsets) ** 2)
        return rbf * envelope


num_atoms = 256
num_rbf = 64
cutoff = 5.0
batch_size = 8


def get_inputs():
    coords = torch.randn(batch_size, num_atoms, 3)
    diff = coords.unsqueeze(2) - coords.unsqueeze(1)
    distances = torch.sqrt((diff ** 2).sum(-1) + 1e-8)
    return [distances]


def get_init_inputs():
    return [num_rbf, cutoff]

    const float* __restrict__ offsets,
    float coeff,
    float cutoff,
    float* __restrict__ output,
    int total_distances,
    int num_rbf
) {
    // Use shared memory for offsets
    extern __shared__ float s_offsets[];
    
    // Cooperative load of offsets into shared memory
    for (int j = threadIdx.x; j < num_rbf; j += blockDim.x) {
        s_offsets[j] = offsets[j];
    }
    __syncthreads();
    
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = blockDim.x * gridDim.x;
    
    const float pi_div_cutoff = M_PI / cutoff;
    
    for (int i = idx; i < total_distances; i += stride) {
        float d = distances[i];
        
        // Cosine cutoff envelope
        float envelope;
        if (d < cutoff) {
            envelope = 0.5f * (cosf(pi_div_cutoff * d) + 1.0f);
        } else {
            envelope = 0.0f;
        }
        
        // RBF expansion - compute and store
        float* out_ptr = output + i * num_rbf;
        
        #pragma unroll 8
        for (int j = 0; j < num_rbf; j++) {
            float diff = d - s_offsets[j];
            float rbf = expf(coeff * diff * diff);
            out_ptr[j] = rbf * envelope;
        }
    }
}

torch::Tensor rbf_cuda_forward(
    torch::Tensor distances,
    torch::Tensor offsets,
    float coeff,
    float cutoff
) {
    auto sizes = distances.sizes().vec();
    int total_distances = distances.numel();
    int num_rbf = offsets.size(0);
    
    sizes.push_back(num_rbf);
    auto output = torch::empty(sizes, distances.options());
    
    int threads = 256;
    int blocks = (total_distances + threads - 1) / threads;
    
    // Shared memory size for offsets
    size_t shared_mem_size = num_rbf * sizeof(float);
    
    rbf_kernel_vector<<<blocks, threads, shared_mem_size>>>(
        distances.data_ptr<float>(),
        offsets.data_ptr<float>(),
        coeff,
        cutoff,
        output.data_ptr<float>(),
        total_distances,
        num_rbf
    );
    
    return output;
}
