import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Log-likelihood under the Dirichlet-Multinomial (Polya) distribution.
    Given count vectors n and concentration parameters alpha, computes
    log p(n | alpha) using the log-gamma formulation.  Core computation
    in Bayesian topic models and mixed-membership models.
    """

    def __init__(self, vocab_size):
        super().__init__()
        self.vocab_size = vocab_size
        self.log_alpha = nn.Parameter(torch.zeros(vocab_size))

    def forward(self, counts: torch.Tensor) -> torch.Tensor:
        """
        Args:
            counts: (B, V) – integer count vectors (as float for differentiability)
        Returns:
            log_lik: (B,) – log p(counts | alpha) for each sample
        """
        alpha = torch.exp(self.log_alpha).unsqueeze(0)  # (1, V)
        N = counts.sum(-1)  # (B,)
        alpha_sum = alpha.sum(-1)  # (1,)

        # log p(n | alpha) = log Gamma(alpha_sum) - log Gamma(N + alpha_sum)
        #                   + sum_v [ log Gamma(n_v + alpha_v) - log Gamma(alpha_v) ]
        log_lik = (
            torch.lgamma(alpha_sum)
            - torch.lgamma(N + alpha_sum)
            + (torch.lgamma(counts + alpha) - torch.lgamma(alpha)).sum(-1)
        )
        return log_lik


vocab_size = 256
batch_size = 512


def get_inputs():
    counts = torch.randint(0, 10, (batch_size, vocab_size)).float()
    return [counts]


def get_init_inputs():
    return [vocab_size]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cmath>

// Use texture cache for read-only data
__device__ __forceinline__ float load_float(const float* ptr) {
    return __ldg(ptr);
}

// Kernel: each thread handles one batch element
__global__ void dirichlet_multinomial_loglik_kernel(
    const float* __restrict__ counts,
    const float* __restrict__ alpha,
    const float* __restrict__ lgamma_alpha,
    float lgamma_alpha_sum,
    float alpha_sum,
    float* __restrict__ log_lik,
    int batch_size,
    int vocab_size
) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= batch_size) return;
    
    float N = 0.0f;
    float sum_lgamma_n_alpha = 0.0f;
    
    const float* counts_row = counts + i * vocab_size;
    
    // Use read-only cache for alpha and lgamma_alpha (broadcast pattern)
    for (int v = 0; v < vocab_size; v++) {
        float n_v = counts_row[v];
        float a_v = load_float(alpha + v);
        N += n_v;
        sum_lgamma_n_alpha += lgammaf(n_v + a_v) - load_float(lgamma_alpha + v);
    }
    
    log_lik[i] = lgamma_alpha_sum - lgammaf(N + alpha_sum) + sum_lgamma_n_alpha;
}

torch::Tensor dirichlet_multinomial_forward(
    torch::Tensor counts,
    torch::Tensor log_alpha
) {
    counts = counts.contiguous();
    log_alpha = log_alpha.contiguous();
    
    const int batch_size = counts.size(0);
    const int vocab_size = counts.size(1);
    
    auto alpha_tensor = log_alpha.exp();
    auto lgamma_alpha_tensor = torch::lgamma(alpha_tensor);
    
    float alpha_sum = alpha_tensor.sum().item<float>();
    float lgamma_alpha_sum = std::lgamma(alpha_sum);
    
    auto log_lik = torch::empty({batch_size}, counts.options());
    
    const int threads = 256;
    const int blocks = (batch_size + threads - 1) / threads;
    
    dirichlet_multinomial_loglik_kernel<<<blocks, threads>>>(
        counts.data_ptr<float>(),
        alpha_tensor.data_ptr<float>(),
        lgamma_alpha_tensor.data_ptr<float>(),
        lgamma_alpha_sum,
        alpha_sum,
        log_lik.data_ptr<float>(),
        batch_size,
        vocab_size
    );
    
    return log_lik;
}
"""
