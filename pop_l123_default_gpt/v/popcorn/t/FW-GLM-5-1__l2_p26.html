class Model(nn.Module):
    """
    Apply rotary position embedding to incoming keys and update the KV cache.

    This isolates a common decode-time kernel: rotate the fresh key vectors for
    the current positions and store both keys and values into the running cache.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        k_new: torch.Tensor,
        v_new: torch.Tensor,
        cos: torch.Tensor,
        sin: torch.Tensor,
        cache_k: torch.Tensor,
        cache_v: torch.Tensor,
        positions: torch.Tensor,
    ) -> torch.Tensor:
        updated_k = cache_k.clone()
        updated_v = cache_v.clone()
        rot_dim = k_new.shape[-1]

        for b in range(k_new.shape[0]):
            for t in range(k_new.shape[1]):
                pos = int(positions[b, t].item())
                c = cos[pos]
                s = sin[pos]
                x = k_new[b, t]
                x_even = x[..., 0::2]
                x_odd = x[..., 1::2]
                rot_even = x_even * c - x_odd * s
                rot_odd = x_even * s + x_odd * c
                rotated = torch.empty_like(x)
                rotated[..., 0::2] = rot_even
                rotated[..., 1::2] = rot_odd
                updated_k[b, pos] = rotated
                updated_v[b, pos] = v_new[b, t]

        return torch.stack((updated_k, updated_v), dim=0)

def _rope_tables(cache_len: int, half_dim: int):
    positions = torch.arange(cache_len, dtype=torch.float32).unsqueeze(1)
    freqs = torch.pow(10000.0, -torch.arange(half_dim, dtype=torch.float32) / half_dim).unsqueeze(0)
    angles = positions * freqs
    return torch.cos(angles), torch.sin(angles)

// Optimized update kernel - each thread handles one (b, t, h) tuple
// Processes all dimensions for that head
__global__ void rope_update_positions_kernel(
    float* __restrict__ updated_k,
    float* __restrict__ updated_v,
    const float* __restrict__ k_new,
    const float* __restrict__ v_new,
    const float* __restrict__ cos,
    const float* __restrict__ sin,
    const int32_t* __restrict__ positions,
    int batch_size, int update_len, int num_heads, int head_dim, int cache_len
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total_threads = batch_size * update_len * num_heads;
    if (idx >= total_threads) return;
    
    int h = idx % num_heads;
    idx /= num_heads;
    int t = idx % update_len;
    int b = idx / update_len;
    
    int pos = positions[b * update_len + t];
    int half_dim = head_dim / 2;
    
    // Offsets
    int cache_offset = ((b * cache_len + pos) * num_heads + h) * head_dim;
    int new_offset = ((b * update_len + t) * num_heads + h) * head_dim;
    int cos_offset = pos * half_dim;
    
    // Apply RoPE rotation
    #pragma unroll 4
    for (int d = 0; d < half_dim; d++) {
        float c = cos[cos_offset + d];
        float s = sin[cos_offset + d];
        float x_even = k_new[new_offset + d * 2];
        float x_odd = k_new[new_offset + d * 2 + 1];
        updated_k[cache_offset + d * 2] = x_even * c - x_odd * s;
        updated_k[cache_offset + d * 2 + 1] = x_even * s + x_odd * c;
    }
    
    // Copy new value
    #pragma unroll 4
    for (int d = 0; d < head_dim; d++) {
        updated_v[cache_offset + d] = v_new[new_offset + d];
    }
}

    torch::Tensor cos,
    torch::Tensor sin,
