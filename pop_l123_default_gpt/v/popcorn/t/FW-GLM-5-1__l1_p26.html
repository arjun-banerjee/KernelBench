import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Self-normalized importance sampling estimate.
    Given samples from a proposal distribution q, computes importance
    weights w_i = p(x_i)/q(x_i), normalizes them, and returns both
    the weighted estimate of E_p[f(x)] and the effective sample size.

    Target p is a mixture of Gaussians; proposal q is a single Gaussian.
    """

    def __init__(self, dim, num_components=3):
        super().__init__()
        self.dim = dim
        self.num_components = num_components
        # Target mixture components
        self.register_buffer("mix_weights", torch.ones(num_components) / num_components)
        mus = torch.randn(num_components, dim) * 2.0
        self.register_buffer("mix_mus", mus)
        self.register_buffer("mix_stds", torch.ones(num_components, dim))
        # Proposal
        self.register_buffer("q_mu", torch.zeros(dim))
        self.register_buffer("q_std", torch.ones(dim) * 3.0)

    def _log_p(self, x):
        """Log density of target mixture."""
        diff = x.unsqueeze(1) - self.mix_mus.unsqueeze(0)  # (B, K, D)
        log_comp = -0.5 * (diff / self.mix_stds.unsqueeze(0)) ** 2 - self.mix_stds.log().unsqueeze(0)
        log_comp = log_comp.sum(-1)  # (B, K)
        return torch.logsumexp(log_comp + self.mix_weights.log().unsqueeze(0), dim=-1)

    def _log_q(self, x):
        """Log density of proposal."""
        return (-0.5 * ((x - self.q_mu) / self.q_std) ** 2 - self.q_std.log()).sum(-1)

    def forward(self, samples: torch.Tensor) -> tuple:
        """
        Args:
            samples: (N, dim) – samples drawn from q
        Returns:
            (weighted_mean, ess):
                weighted_mean: (dim,) – importance-weighted mean estimate of E_p[x]
                ess: scalar – effective sample size
        """
        log_w = self._log_p(samples) - self._log_q(samples)
        log_w = log_w - torch.logsumexp(log_w, dim=0)
        w = torch.exp(log_w)

        weighted_mean = (w.unsqueeze(-1) * samples).sum(dim=0)
        ess = 1.0 / (w ** 2).sum()
        return weighted_mean, ess

    int tid = threadIdx.x;
    
    if (tid < dim && tid < 32) {
        float sum = 0.0f;
        for (int b = 0; b < num_blocks; b++) {
            sum += weighted_mean_partial[b * dim + tid];
        }
        s_mean[tid] = sum;
    }
    
    if (tid == 0) {
        float sum = 0.0f;
        for (int b = 0; b < num_blocks; b++) {
            sum += ess_partial[b];
        }
        s_ess = sum;
    }
    __syncthreads();
    
    if (tid < dim && tid < 32) {
        weighted_mean_out[tid] = s_mean[tid];
    }
    if (tid == 0) {
        ess_out[0] = 1.0f / s_ess;
    }
}

// Alternative single-kernel approach for better performance
__global__ void importance_sampling_kernel(
    const float* __restrict__ samples,
    const float* __restrict__ mix_mus,
    const float* __restrict__ mix_stds,
    const float* __restrict__ mix_weights,
    const float* __restrict__ q_mu,
    const float* __restrict__ q_std,
    float* __restrict__ log_w,
    float* __restrict__ block_maxes,
    float* __restrict__ block_sums,
    int N, int dim, int num_components) {
    
    extern __shared__ float shared_mem[];
    float* s_max = shared_mem;  // BLOCK_SIZE / WARP_SIZE elements
    float* s_sum = s_max + (BLOCK_SIZE / WARP_SIZE);  // BLOCK_SIZE / WARP_SIZE elements
    
    int tid = threadIdx.x;
    int warp_id = tid / WARP_SIZE;
    int lane_id = tid % WARP_SIZE;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    // Stage 1: Compute log weights
    float my_log_w = -1e30f;
    if (idx < N) {
        const float* sample = samples + idx * dim;
        float log_p = compute_log_p_single(sample, mix_mus, mix_stds, mix_weights, dim, num_components);
        float log_q = compute_log_q_single(sample, q_mu, q_std, dim);
        my_log_w = log_p - log_q;
        log_w[idx] = my_log_w;
    }
    
    // Stage 2: Block-level logsumexp
    // Warp-level max reduction
    float warp_max = warp_reduce_max(my_log_w);
    
    if (lane_id == 0) {
        s_max[warp_id] = warp_max;
    }
    __syncthreads();
    
    float block_max;
    if (warp_id == 0) {
        block_max = (lane_id < (BLOCK_SIZE / WARP_SIZE)) ? s_max[lane_id] : -1e30f;
        block_max = warp_reduce_max(block_max);
        if (lane_id == 0) {
            block_maxes[blockIdx.x] = block_max;
        }
    }
    __syncthreads();
    
    // Broadcast block_max (read from shared or global depending on timing)
    __shared__ float s_block_max;
    if (tid == 0) {
        s_block_max = block_maxes[blockIdx.x];
    }
    __syncthreads();
    block_max = s_block_max;
    
    // Compute exp(log_w - block_max)
    float my_exp = (idx < N) ? expf(my_log_w - block_max) : 0.0f;
    
    // Warp-level sum reduction
    float warp_sum = warp_reduce_sum(my_exp);
    
    if (lane_id == 0) {
        s_sum[warp_id] = warp_sum;
    }
    __syncthreads();
    
    if (warp_id == 0) {
        float block_sum = (lane_id < (BLOCK_SIZE / WARP_SIZE)) ? s_sum[lane_id] : 0.0f;
        block_sum = warp_reduce_sum(block_sum);
        if (lane_id == 0) {
            block_sums[blockIdx.x] = block_sum;
        }
    }
}

std::tuple<torch::Tensor, torch::Tensor> importance_sampling_cuda(
    torch::Tensor samples,
    torch::Tensor mix_mus,
    torch::Tensor mix_stds,
    torch::Tensor mix_weights,
    torch::Tensor q_mu,
    torch::Tensor q_std) {
    
    int N = samples.size(0);
    int dim = samples.size(1);
    int num_components = mix_mus.size(0);
    
    int num_blocks = (N + BLOCK_SIZE - 1) / BLOCK_SIZE;
    
    // Allocate intermediate buffers
    auto options = samples.options();
    auto log_w = torch::empty({N}, options);
    auto block_maxes = torch::empty({num_blocks}, options);
    auto block_sums = torch::empty({num_blocks}, options);
    
    // Launch combined kernel
    size_t shared_mem_size = 2 * (BLOCK_SIZE / WARP_SIZE) * sizeof(float);
    
    importance_sampling_kernel<<<num_blocks, BLOCK_SIZE, shared_mem_size>>>(
        samples.data_ptr<float>(),
        mix_mus.data_ptr<float>(),
        mix_stds.data_ptr<float>(),
        mix_weights.data_ptr<float>(),
        q_mu.data_ptr<float>(),
        q_std.data_ptr<float>(),
        log_w.data_ptr<float>(),
        block_maxes.data_ptr<float>(),
        block_sums.data_ptr<float>(),
        N, dim, num_components
    );
    
    // For now, do the final logsumexp reduction on CPU (small data)
    torch::Tensor block_maxes_cpu = block_maxes.cpu();
    torch::Tensor block_sums_cpu = block_sums.cpu();
    
    float global_max = block_maxes_cpu[0].item<float>();
    for (int i = 1; i < num_blocks; i++) {
        float val = block_maxes_cpu[i].item<float>();
        if (val > global_max) global_max = val;
    }
    
    float sum_exp = 0.0f;
    for (int i = 0; i < num_blocks; i++) {
        float max_i = block_maxes_cpu[i].item<float>();
        float sum_i = block_sums_cpu[i].item<float>();
        sum_exp += expf(max_i - global_max) * sum_i;
    }
    
    float global_logsumexp = global_max + logf(sum_exp);
    
    // Third pass: normalize and compute weighted mean and ESS
    auto weighted_mean_partial = torch::zeros({num_blocks, dim}, options);
    auto ess_partial = torch::zeros({num_blocks}, options);
    
    normalize_and_reduce_kernel<<<num_blocks, BLOCK_SIZE>>>(
        samples.data_ptr<float>(),
        log_w.data_ptr<float>(),
        global_logsumexp,
        weighted_mean_partial.data_ptr<float>(),
        ess_partial.data_ptr<float>(),
        N, dim
    );
    
    // Final reduction
    auto weighted_mean_out = torch::empty({dim}, options);
    auto ess_out = torch::empty({1}, options);
    
    final_reduce_kernel<<<1, dim>>>(
        weighted_mean_partial.data_ptr<float>(),
        ess_partial.data_ptr<float>(),
        weighted_mean_out.data_ptr<float>(),
        ess_out.data_ptr<float>(),
        num_blocks, dim
    );
    
    return std::make_tuple(weighted_mean_out, ess_out);
}
