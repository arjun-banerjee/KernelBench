import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues
        Returns:
            dist: (B, N, N) – pairwise Euclidean distance matrix
        """
        diff = coords.unsqueeze(2) - coords.unsqueeze(1)  # (B, N, N, 3)
        dist = torch.sqrt((diff ** 2).sum(dim=-1) + 1e-8)
        return dist

#include <cuda.h>
#include <cuda_runtime.h>
#include <cmath>

#define TILE_SIZE 32

__global__ void distance_kernel_tiled(const float* __restrict__ coords, float* __restrict__ dist, int N) {
    // blockIdx.x: which tile column (j direction)
    // blockIdx.y: which tile row (i direction)
    // blockIdx.z: which batch
    
    int tile_i = blockIdx.y;
    int tile_j = blockIdx.x;
    int b = blockIdx.z;
    
    int tx = threadIdx.x;  // 0 to TILE_SIZE-1
    int ty = threadIdx.y;  // 0 to TILE_SIZE-1
    
    int i = tile_i * TILE_SIZE + ty;  // Global row index
    int j = tile_j * TILE_SIZE + tx;  // Global col index
    
    // Shared memory for the two tiles of coordinates
    // Tile_i coords: TILE_SIZE * 3 floats
    // Tile_j coords: TILE_SIZE * 3 floats
    __shared__ float s_coords_i[TILE_SIZE * 3];
    __shared__ float s_coords_j[TILE_SIZE * 3];
    
    // Cooperative load of tile_i coordinates
    // Each thread loads one float, we need TILE_SIZE*3 floats per tile
    int load_idx = ty * TILE_SIZE + tx;  // Linear thread index
    
    // Load tile_i coordinates (the rows we're computing)
    if (load_idx < TILE_SIZE * 3) {
        int row = load_idx / 3;
        int comp = load_idx % 3;
        int global_row = tile_i * TILE_SIZE + row;
        if (global_row < N) {
            s_coords_i[load_idx] = coords[b * N * 3 + global_row * 3 + comp];
        }
    }
    
    // Load tile_j coordinates (the columns we're computing)
    if (load_idx < TILE_SIZE * 3) {
        int col = load_idx / 3;
        int comp = load_idx % 3;
        int global_col = tile_j * TILE_SIZE + col;
        if (global_col < N) {
            s_coords_j[load_idx] = coords[b * N * 3 + global_col * 3 + comp];
        }
    }
    
    __syncthreads();
    
    // Load the i-th coordinate into registers
    float xi = s_coords_i[ty * 3 + 0];
    float yi = s_coords_i[ty * 3 + 1];
    float zi = s_coords_i[ty * 3 + 2];
    
    // Load the j-th coordinate into registers
    float xj = s_coords_j[tx * 3 + 0];
    float yj = s_coords_j[tx * 3 + 1];
    float zj = s_coords_j[tx * 3 + 2];
    
    // Compute distance
    float dx = xi - xj;
    float dy = yi - yj;
    float dz = zi - zj;
    
    float dist_val = sqrtf(dx*dx + dy*dy + dz*dz + 1e-8f);
    
    // Write result
    if (i < N && j < N) {
        dist[b * N * N + i * N + j] = dist_val;
    }
}

torch::Tensor compute_distances(torch::Tensor coords) {

}
