import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
__global__ void mean_kernel(const float* input, float* output, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx < total) {
        float sum = 0.0f;
        for(int r=0; r<R; r++) {
            sum += input[r * total + idx];
        }
        output[idx] = sum / R;
    }
}

void launch_mean(const float* input, float* output, int R, int B, int D) {
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    mean_kernel<<<blocks, threads>>>(input, output, R, B, D);
}
'''

cpp_source = '''
void launch_mean(const float* input, float* output, int R, int B, int D);

torch::Tensor mean_reduce(torch::Tensor x) {
    int R = x.size(0);
    int B = x.size(1);
    int D = x.size(2);
    auto output = torch::empty({B, D}, x.options());
    launch_mean(x.data_ptr<float>(), output.data_ptr<float>(), R, B, D);
    return output;
}
'''

module = load_inline(
