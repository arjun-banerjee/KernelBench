import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Evidence Lower Bound (ELBO) computation for a Variational Autoencoder.
    Combines a reconstruction term (binary cross-entropy) with the
    KL divergence between the approximate posterior q(z|x) = N(mu, sigma^2)
    and the prior p(z) = N(0, I).  The core loss function in VAE training.
    """

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
        )
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)

        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
            nn.Sigmoid(),
        )

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim) – input data in [0, 1]
        Returns:
            (elbo, recon_loss, kl_div):
                elbo:       (B,)
                recon_loss: (B,)
                kl_div:     (B,)
        """
        h = self.encoder(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)

        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)

        x_recon = self.decoder(z)
        recon = F.binary_cross_entropy(x_recon, x, reduction="none").sum(-1)
        kl = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(-1)
        elbo = -(recon + kl)
        return elbo, recon, kl

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

__global__ void latent_kernel(const float* __restrict__ mu,
                              const float* __restrict__ logvar,
                              const float* __restrict__ eps,
                              float* __restrict__ z,
                              float* __restrict__ kl,
                              int L) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    int base = b * L;
    float sum = 0.0f;

    for (int j = tid; j < L; j += blockDim.x) {
        float m = mu[base + j];
        float lv = logvar[base + j];
        float ev = expf(lv);
        z[base + j] = m + expf(0.5f * lv) * eps[base + j];
        sum += 1.0f + lv - m * m - ev;
    }

    __shared__ float smem[128];
    smem[tid] = sum;
    __syncthreads();

    #pragma unroll
    for (int s = 64; s > 0; s >>= 1) {
        if (tid < s) smem[tid] += smem[tid + s];
        __syncthreads();
    }

    if (tid == 0) kl[b] = -0.5f * smem[0];
}

__global__ void loss_kernel(const float* __restrict__ logits,
                            const float* __restrict__ x,
                            const float* __restrict__ kl,
                            float* __restrict__ elbo,
                            float* __restrict__ recon,
                            int D) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    int base = b * D;
    float sum = 0.0f;

    for (int i = tid; i < D; i += blockDim.x) {
        float t = logits[base + i];
        float xi = x[base + i];
        // Equivalent to binary cross entropy after a sigmoid for the logit ranges
        // produced by this network, while avoiding materializing the sigmoid output.
        sum += fmaxf(t, 0.0f) - xi * t + log1pf(expf(-fabsf(t)));
    }

    __shared__ float smem[256];
    smem[tid] = sum;
    __syncthreads();

    #pragma unroll
    for (int s = 128; s > 0; s >>= 1) {
        if (tid < s) smem[tid] += smem[tid + s];
        __syncthreads();
    }

    if (tid == 0) {
        float r = smem[0];
        recon[b] = r;
        elbo[b] = -(r + kl[b]);
    }
}

std::vector<torch::Tensor> latent_forward_cuda(torch::Tensor mu, torch::Tensor logvar, torch::Tensor eps) {
    int B = mu.size(0);
    int L = mu.size(1);
    auto z = torch::empty_like(mu);
    auto kl = torch::empty({B}, mu.options());
    latent_kernel<<<B, 128>>>(mu.data_ptr<float>(), logvar.data_ptr<float>(), eps.data_ptr<float>(),
                              z.data_ptr<float>(), kl.data_ptr<float>(), L);
    return {z, kl};
}

}
