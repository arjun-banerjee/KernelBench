// Conv1d + residual kernel
__global__ void conv_residual_kernel(
    const float* __restrict__ input,
    const float* __restrict__ residual,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size,
    int channels,
    int seq_len,
    int kernel_size,
    int dilation,
    int padding
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * channels * seq_len;
    if (idx >= total) return;
    
    int b = idx / (channels * seq_len);
    int rem = idx % (channels * seq_len);
    int out_c = rem / seq_len;
    int out_s = rem % seq_len;
    
    float conv_sum = 0.0f;
    for (int in_c = 0; in_c < channels; in_c++) {
        for (int k = 0; k < kernel_size; k++) {
            int in_s = out_s - padding + k * dilation;
            if (in_s >= 0 && in_s < seq_len) {
                int input_idx = b * channels * seq_len + in_c * seq_len + in_s;
                int weight_idx = out_c * channels * kernel_size + in_c * kernel_size + k;
                conv_sum += input[input_idx] * weight[weight_idx];
            }
        }
    }
    conv_sum += bias[out_c];
    conv_sum += residual[b * channels * seq_len + out_c * seq_len + out_s];
    output[idx] = conv_sum;
}
