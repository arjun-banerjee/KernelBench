class Model(nn.Module):
    """
    Fused GroupNorm + SiLU (Swish), a common post-conv / residual-branch pattern in
    ResNet-family backbones (SiLU instead of ReLU) and in spatial blocks of diffusion
    transformers (DiT) where features are (N, C, H, W).

    Reference pipeline for a single fused kernel: normalize channels with
    `GroupNorm`, then apply `SiLU` elementwise (`x * sigmoid(x)`).
    """

    def __init__(
        self,
        num_channels: int,
        num_groups: int,
        eps: float = 1e-5,
        affine: bool = True,
    ):
        super().__init__()
        assert num_channels % num_groups == 0
        self.norm = nn.GroupNorm(
            num_groups=num_groups,
            num_channels=num_channels,
            eps=eps,
            affine=affine,
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C, H, W) feature map
        Returns:
            Same shape as x
        """
        return F.silu(self.norm(x))

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

// Fused GroupNorm + SiLU kernel
// Each thread block processes one (batch, group) pair
// Threads within block cooperatively compute mean, variance, then apply normalization + SiLU

template <int BLOCK_SIZE>
__global__ void fused_groupnorm_silu_kernel(
    const float* __restrict__ input,
    float* __restrict__ output,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    int N, int C, int H, int W, int G, int channels_per_group, float eps
) {
    // Each block handles one (batch_item, group) pair
    int batch_idx = blockIdx.x / G;
    int group_idx = blockIdx.x % G;
    
    int group_start_channel = group_idx * channels_per_group;
    int group_size = channels_per_group * H * W;
    
    // Index for this thread's element within the group
    int tid = threadIdx.x;
    int total_elements = group_size;
    
    // Pointers to the start of this batch and group in input/output
    int batch_offset = batch_idx * C * H * W;
    int group_offset = group_start_channel * H * W;
    const float* __restrict__ input_ptr = input + batch_offset + group_offset;
    float* __restrict__ output_ptr = output + batch_offset + group_offset;
    
    // Shared memory for reduction
    __shared__ float s_mean[BLOCK_SIZE];
    __shared__ float s_var[BLOCK_SIZE];
    
    // Compute mean using parallel reduction
    float local_sum = 0.0f;
    float local_sq_sum = 0.0f;
    
    // Grid-stride loop to process all elements in the group
    for (int i = tid; i < total_elements; i += BLOCK_SIZE) {
        float val = input_ptr[i];
        local_sum += val;
        local_sq_sum += val * val;
    }
    
    s_mean[tid] = local_sum;
    s_var[tid] = local_sq_sum;
    __syncthreads();
    
    // Parallel reduction in shared memory
    for (int stride = BLOCK_SIZE / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_mean[tid] += s_mean[tid + stride];
            s_var[tid] += s_var[tid + stride];
        }
        __syncthreads();
    }
    
    // Compute mean and variance
    float mean = 0.0f;
    float inv_std = 0.0f;
    if (tid == 0) {
        mean = s_mean[0] / total_elements;
        float variance = s_var[0] / total_elements - mean * mean;
        inv_std = 1.0f / sqrtf(variance + eps);
        s_mean[0] = mean;
        s_var[0] = inv_std;
    }
    __syncthreads();
    
    mean = s_mean[0];
    inv_std = s_var[0];
    
    // Apply normalization, affine, and SiLU
    for (int i = tid; i < total_elements; i += BLOCK_SIZE) {
        int c_local = i / (H * W);
        int c_global = group_start_channel + c_local;
        
        float val = input_ptr[i];
        float norm_val = (val - mean) * inv_std;
        
        // Apply affine transformation if weight/bias are provided
        if (weight != nullptr && bias != nullptr) {
            norm_val = weight[c_global] * norm_val + bias[c_global];
        }
        
        // Apply SiLU: x * sigmoid(x) = x / (1 + exp(-x))
        float silu_val = norm_val / (1.0f + expf(-norm_val));
        
        output_ptr[i] = silu_val;
    }
}
