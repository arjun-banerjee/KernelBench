    const half2 c0 = *reinterpret_cast<const half2*>(cosp + cs);
    const half2 s0 = *reinterpret_cast<const half2*>(sinp + cs);
    const half2 c1 = *reinterpret_cast<const half2*>(cosp + cs + 32);
