__inline__ __device__ float warp_reduce_sum(float v) {
    unsigned mask = 0xffffffffu;
    v += __shfl_down_sync(mask, v, 16);
    v += __shfl_down_sync(mask, v, 8);
    v += __shfl_down_sync(mask, v, 4);
    v += __shfl_down_sync(mask, v, 2);
    v += __shfl_down_sync(mask, v, 1);
    return v;

__global__ __launch_bounds__(256, 2)

                          const half* __restrict__ w,
                          half* __restrict__ y,
                          float eps) {
    constexpr int H = 4096;
    constexpr int VEC = 8;
    constexpr int VECS = H / VEC;
    const int row = blockIdx.x;
    const half* __restrict__ row_x = x + (size_t)row * H;
    half* __restrict__ row_y = y + (size_t)row * H;
    const float4* __restrict__ vx = reinterpret_cast<const float4*>(row_x);

    float sum = 0.0f;
    for (int i = threadIdx.x; i < VECS; i += blockDim.x) {
        float4 v = vx[i];
        const half2* h2 = reinterpret_cast<const half2*>(&v);

            sum = fmaf(f.y, f.y, sum);
        }
