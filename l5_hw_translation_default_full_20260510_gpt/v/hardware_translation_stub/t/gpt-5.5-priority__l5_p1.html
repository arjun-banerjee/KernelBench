#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <math.h>

__device__ __forceinline__ float warp_sum0(float x) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) x += __shfl_down_sync(0xffffffff, x, o);
    return x;
}
__device__ __forceinline__ float warp_sum_all(float x) { x = warp_sum0(x); return __shfl_sync(0xffffffff, x, 0); }
__device__ __forceinline__ float warp_max0(float x) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) x = fmaxf(x, __shfl_down_sync(0xffffffff, x, o));
    return x;
}
__device__ __forceinline__ float warp_max_all(float x) { x = warp_max0(x); return __shfl_sync(0xffffffff, x, 0); }

__global__ __launch_bounds__(1024, 1)
void causal_gqa_kernel(const half* __restrict__ q, const half* __restrict__ k, const half* __restrict__ v, half* __restrict__ out) {
    constexpr int H=32, KVH=8, S=512, D=128, ROWS=32, D2=64;
    extern __shared__ float smem[];
    int lane = threadIdx.x & 31;
    int wid = threadIdx.x >> 5;
    int blk = blockIdx.x;
    int tile = blk & 15;
    int h = (blk >> 4) & 31;
    int b = blk >> 9;
    int t = tile * ROWS + wid;
    int kvh = h >> 2;

    float* mylog = smem + wid * S;
    const half2* q2 = reinterpret_cast<const half2*>(q + (((b * H + h) * S + t) * D));
    const half2* k2base = reinterpret_cast<const half2*>(k + ((b * KVH + kvh) * S * D));
    const float2 qf0 = __half22float2(q2[lane]);
    const float2 qf1 = __half22float2(q2[lane + 32]);
    constexpr float scale = 0.08838834764831845f;

    for (int j = 0; j <= t; ++j) {
        const half2* k2 = k2base + j * D2;
        float2 kf0 = __half22float2(k2[lane]);
        float2 kf1 = __half22float2(k2[lane + 32]);
        float acc = 0.0f;
        acc = fmaf(qf0.x, kf0.x, acc);
        acc = fmaf(qf0.y, kf0.y, acc);
        acc = fmaf(qf1.x, kf1.x, acc);
        acc = fmaf(qf1.y, kf1.y, acc);
        acc = warp_sum0(acc);
        if (lane == 0) mylog[j] = acc * scale;
    }
    __syncwarp();

    float m = -INFINITY;
    for (int j = lane; j <= t; j += 32) m = fmaxf(m, mylog[j]);
    m = warp_max_all(m);

    float denom = 0.0f;
    for (int j = lane; j <= t; j += 32) {
        float e = expf(mylog[j] - m);
        mylog[j] = e;
        denom += e;
    }
    denom = warp_sum_all(denom);
    float inv_denom = 1.0f / denom;

    const half2* v2base = reinterpret_cast<const half2*>(v + ((b * KVH + kvh) * S * D));
    half2* o2 = reinterpret_cast<half2*>(out + (((b * H + h) * S + t) * D));
    float ax0 = 0.0f, ay0 = 0.0f, ax1 = 0.0f, ay1 = 0.0f;
    for (int j = 0; j <= t; ++j) {
        float w = mylog[j] * inv_denom;
        float2 vf0 = __half22float2(v2base[j * D2 + lane]);
        float2 vf1 = __half22float2(v2base[j * D2 + lane + 32]);
        ax0 = fmaf(w, vf0.x, ax0);
        ay0 = fmaf(w, vf0.y, ay0);
        ax1 = fmaf(w, vf1.x, ax1);
        ay1 = fmaf(w, vf1.y, ay1);
    }
    o2[lane] = __floats2half2_rn(ax0, ay0);
    o2[lane + 32] = __floats2half2_rn(ax1, ay1);
}

torch::Tensor mha_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, int, int, int) {
    auto out = torch::empty_like(q);
    cudaFuncSetAttribute(causal_gqa_kernel, cudaFuncAttributeMaxDynamicSharedMemorySize, 65536);
    causal_gqa_kernel<<<dim3(2048), dim3(1024), 65536, at::cuda::getCurrentCUDAStream()>>>(
        reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()));
    return out;
}
