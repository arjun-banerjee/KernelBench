__global__ void fp32_to_fp16_kernel(const float* __restrict__ inp, half* __restrict__ out, int n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) out[i] = __float2half_rn(inp[i]);
}

torch::Tensor convert_forward(torch::Tensor x) {
