    """

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

module = load_inline(
    name="scatter_add_kernel",
    cpp_sources="",
    cuda_sources=r"""
#include <cuda_runtime.h>
#include <device_atomic_functions.h>

// Atomic add for float2 - splits into two separate float atomics
__device__ void atomicAddFloat2(float2* addr, float2 val) {
    atomicAdd(&(addr->x), val.x);
    atomicAdd(&(addr->y), val.y);
}

extern "C" __global__ void scatter_add_kernel(
    const int* __restrict__ dst_idx,
    const float* __restrict__ edge_feat,
    float* __restrict__ out,
    int num_edges,
    int feat_dim
) {
    int edge_idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (edge_idx >= num_edges) return;
    
    int dst = dst_idx[edge_idx];
    float* out_row = out + dst * feat_dim;
    const float* feat_row = edge_feat + edge_idx * feat_dim;
    
    // Process in float2 chunks (2 floats at a time) for better memory throughput
    int dim2 = feat_dim / 2;
    for (int i = 0; i < dim2; i++) {
        float2 val = reinterpret_cast<const float2*>(feat_row)[i];
        atomicAddFloat2(reinterpret_cast<float2*>(out_row) + i, val);
    }
    
    // Handle remaining odd dimension
    if (feat_dim % 2 == 1) {
        atomicAdd(&out_row[feat_dim - 1], feat_row[feat_dim - 1]);
    }
}
""",
    functions=["scatter_add_kernel"],
    extra_cuda_cflags=["-O3"],

)
