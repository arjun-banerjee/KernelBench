import torch
import torch.nn as nn
import torch.nn.functional as F
import math


class Model(nn.Module):
    """
    Invariant Point Attention (IPA) from AlphaFold2's structure module.
    Combines sequence-space attention with 3-D geometric queries/keys/values
    expressed in local residue frames.  The attention logits incorporate
    Euclidean distances in global frame, making it SE(3)-equivariant.
    """

    def __init__(self, node_dim, pair_dim, num_heads, num_query_points, num_value_points):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = node_dim // num_heads
        self.num_query_points = num_query_points
        self.num_value_points = num_value_points
        assert node_dim % num_heads == 0

        d = self.head_dim
        self.q_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.k_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.v_proj = nn.Linear(node_dim, num_heads * d, bias=False)

        self.q_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.k_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.v_pts = nn.Linear(node_dim, num_heads * num_value_points * 3, bias=False)

        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)

        self.head_weights = nn.Parameter(torch.zeros(num_heads))

        out_features = num_heads * d + num_heads * num_value_points * 3
        self.out_proj = nn.Linear(out_features, node_dim)

        self.layer_norm = nn.LayerNorm(node_dim)

    def _apply_frame(self, pts, rotations, translations):
        """Transform local-frame points to global frame."""
        # pts: (B, N, P, 3), rotations: (B, N, 3, 3), translations: (B, N, 3)
        global_pts = torch.einsum("bnpc,bncd->bnpd", pts, rotations) + translations.unsqueeze(2)
        return global_pts

    def _apply_inv_frame(self, pts, rotations, translations):
        """Transform global-frame points to local frame."""
        centered = pts - translations.unsqueeze(2)
        local_pts = torch.einsum("bnpc,bndc->bnpd", centered, rotations)
        return local_pts

    def forward(
        self,
        node_repr: torch.Tensor,
        pair_repr: torch.Tensor,
        translations: torch.Tensor,
        rotations: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            node_repr:    (B, N, node_dim)
            pair_repr:    (B, N, N, pair_dim)
            translations: (B, N, 3)
            rotations:    (B, N, 3, 3)
        Returns:
            updated node representation (B, N, node_dim)
        """
        B, N, D = node_repr.shape
        h = self.num_heads
        d = self.head_dim
        Qp = self.num_query_points
        Vp = self.num_value_points

        x = self.layer_norm(node_repr)

        q = self.q_proj(x).view(B, N, h, d)
        k = self.k_proj(x).view(B, N, h, d)
        v = self.v_proj(x).view(B, N, h, d)

        # Points in local frame, then transform to global
        q_pts_local = self.q_pts(x).view(B, N, h * Qp, 3)
        k_pts_local = self.k_pts(x).view(B, N, h * Qp, 3)
        v_pts_local = self.v_pts(x).view(B, N, h * Vp, 3)

        q_pts = self._apply_frame(q_pts_local, rotations, translations).view(B, N, h, Qp, 3)
        k_pts = self._apply_frame(k_pts_local, rotations, translations).view(B, N, h, Qp, 3)
        v_pts = self._apply_frame(v_pts_local, rotations, translations).view(B, N, h, Vp, 3)

        # Scalar attention logits
        attn_scalar = torch.einsum("bihd,bjhd->bhij", q, k) / math.sqrt(d)

        # Point attention: negative squared distance between query and key points
        # q_pts: (B, N_i, h, Qp, 3), k_pts: (B, N_j, h, Qp, 3)
        q_expand = q_pts.unsqueeze(2)  # (B, N, 1, h, Qp, 3)
        k_expand = k_pts.unsqueeze(1)  # (B, 1, N, h, Qp, 3)
        pt_dist = ((q_expand - k_expand) ** 2).sum(dim=(-1, -2))  # (B, N, N, h)

        w = F.softplus(self.head_weights)
        pt_attn = -(w.view(1, 1, 1, h) * pt_dist) / 2.0
        pt_attn = pt_attn.permute(0, 3, 1, 2)  # (B, h, N, N)

        # Pair bias
        pair_bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2)  # (B, h, N, N)

        attn = attn_scalar + pt_attn + pair_bias
        attn = F.softmax(attn, dim=-1)  # (B, h, N_i, N_j)

        # Aggregate scalar values
        out_scalar = torch.einsum("bhij,bjhd->bihd", attn, v).reshape(B, N, h * d)

        # Aggregate point values in global frame, then transform back to local
        out_pts = torch.einsum("bhij,bjhpc->bihpc", attn, v_pts)  # (B, N, h, Vp, 3)
        out_pts = out_pts.reshape(B, N, h * Vp, 3)
        out_pts_local = self._apply_inv_frame(out_pts, rotations, translations)
        out_pts_flat = out_pts_local.reshape(B, N, h * Vp * 3)

        out = torch.cat([out_scalar, out_pts_flat], dim=-1)
        return self.out_proj(out)

// Fused kernel: compute attention scores (scalar + point + pair_bias) and softmax
// Grid: (B, H, N), Block: (N_j) - each block handles one query position
__global__ void fused_attention_kernel(

    float* __restrict__ attn,           // (B, H, N, N)
    int B, int N, int H, int D, int Qp, float inv_sqrt_d
) {

    int h = blockIdx.y;  // head
    int i = blockIdx.z;  // query position

    }

// Optimized attention computation with better memory access
__global__ void attention_forward_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ q_pts,
    const float* __restrict__ k_pts,
    const float* __restrict__ pair_bias,
    const float* __restrict__ head_weights,
    float* __restrict__ attn_out,
    int B, int N, int H, int D, int Qp, float inv_sqrt_d
) {
    // Each block handles one (batch, head, query_i) combination
    int b = blockIdx.x;
    int h = blockIdx.y;
    int i = blockIdx.z;
    
    if (b >= B || h >= H || i >= N) return;
    
    int tid = threadIdx.x;
    int lane = tid & 31;
    
    // Shared memory for query and query points
    extern __shared__ char smem_buf[];
    float* q_smem = reinterpret_cast<float*>(smem_buf);
    float* q_pts_smem = q_smem + D;
    float* scores_smem = q_pts_smem + Qp * 3;
    
    // Load q and q_pts into shared memory
    for (int idx = tid; idx < D; idx += blockDim.x) {
        q_smem[idx] = q[b * N * H * D + i * H * D + h * D + idx];
    }
    for (int idx = tid; idx < Qp * 3; idx += blockDim.x) {
        q_pts_smem[idx] = q_pts[b * N * H * Qp * 3 + i * H * Qp * 3 + h * Qp * 3 + idx];
    }
    __syncthreads();
    
    float hw = head_weights[h];
    
    // Compute attention scores for all j
    float local_max = -1e30f;
    
    for (int j = tid; j < N; j += blockDim.x) {
        // Scalar attention
        float scalar = 0.0f;
        for (int d = 0; d < D; d++) {
            scalar += q_smem[d] * k[b * N * H * D + j * H * D + h * D + d];
        }
        scalar *= inv_sqrt_d;
        
        // Point attention
        float pt_dist = 0.0f;
        for (int p = 0; p < Qp; p++) {
            for (int c = 0; c < 3; c++) {
                float diff = q_pts_smem[p * 3 + c] - 
                            k_pts[b * N * H * Qp * 3 + j * H * Qp * 3 + h * Qp * 3 + p * 3 + c];
                pt_dist += diff * diff;
            }
        }
        float pt_attn = -hw * pt_dist / 2.0f;
        
        // Pair bias
        float pb = pair_bias[b * H * N * N + h * N * N + i * N + j];
        
        float score = scalar + pt_attn + pb;
        scores_smem[j] = score;
        local_max = fmaxf(local_max, score);
    }
    
    // Block-level max reduction
    __shared__ float global_max;
    float reduced_max = blockReduceSum(local_max, tid);  // Actually need max, not sum
    // Use separate max reduction
    for (int offset = blockDim.x / 2; offset > 0; offset /= 2) {
        float other = __shfl_xor_sync(0xffffffff, local_max, offset);
        local_max = fmaxf(local_max, other);
    }
    if (tid == 0) global_max = local_max;
    __syncthreads();
    
    // Compute exp and sum
    float local_sum = 0.0f;
    for (int j = tid; j < N; j += blockDim.x) {
        float exp_val = expf(scores_smem[j] - global_max);
        scores_smem[j] = exp_val;
        local_sum += exp_val;
    }
    
    __shared__ float global_sum;
    float reduced_sum = blockReduceSum(local_sum, tid);
    if (tid == 0) global_sum = reduced_sum;
    __syncthreads();
    
    // Normalize and store
    for (int j = tid; j < N; j += blockDim.x) {
        attn_out[b * H * N * N + h * N * N + i * N + j] = scores_smem[j] / global_sum;
    }
}

class ModelNew(nn.Module):
    """
    Invariant Point Attention (IPA) with custom CUDA kernels.
    """

    def __init__(self, node_dim, pair_dim, num_heads, num_query_points, num_value_points):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = node_dim // num_heads
        self.num_query_points = num_query_points
        self.num_value_points = num_value_points
        assert node_dim % num_heads == 0

        d = self.head_dim
        self.q_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.k_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.v_proj = nn.Linear(node_dim, num_heads * d, bias=False)

        self.q_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.k_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.v_pts = nn.Linear(node_dim, num_heads * num_value_points * 3, bias=False)

        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)

        self.head_weights = nn.Parameter(torch.zeros(num_heads))

        out_features = num_heads * d + num_heads * num_value_points * 3
        self.out_proj = nn.Linear(out_features, node_dim)

        self.layer_norm = nn.LayerNorm(node_dim)

    def forward(
        self,
        node_repr: torch.Tensor,
        pair_repr: torch.Tensor,
        translations: torch.Tensor,
        rotations: torch.Tensor,
    ) -> torch.Tensor:
        B, N, D = node_repr.shape
        h = self.num_heads
        d = self.head_dim
        Qp = self.num_query_points
        Vp = self.num_value_points

        x = self.layer_norm(node_repr)

        # Projections
        q = self.q_proj(x).view(B, N, h, d)
        k = self.k_proj(x).view(B, N, h, d)
        v = self.v_proj(x).view(B, N, h, d)

        # Point projections (local frame)
        q_pts_local = self.q_pts(x).view(B, N, h * Qp, 3)
        k_pts_local = self.k_pts(x).view(B, N, h * Qp, 3)
        v_pts_local = self.v_pts(x).view(B, N, h * Vp, 3)

        # Transform points to global frame using custom CUDA kernel
        if x.is_cuda:
            q_pts_global = ipa_ops.apply_frame_cuda(
                q_pts_local.contiguous().float(),
                rotations.contiguous().float(),
                translations.contiguous().float(),
                h * Qp
            )
            k_pts_global = ipa_ops.apply_frame_cuda(
                k_pts_local.contiguous().float(),
                rotations.contiguous().float(),
                translations.contiguous().float(),
                h * Qp
            )
            v_pts_global = ipa_ops.apply_frame_cuda(
                v_pts_local.contiguous().float(),
                rotations.contiguous().float(),
                translations.contiguous().float(),
                h * Vp
            )
            
            q_pts = q_pts_global.view(B, N, h, Qp, 3)
            k_pts = k_pts_global.view(B, N, h, Qp, 3)
            v_pts = v_pts_global.view(B, N, h, Vp, 3)
            
            # Compute pair bias
            pair_bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2)
            
            # Compute head weights
            w = F.softplus(self.head_weights)
            
            # Fused attention computation
            inv_sqrt_d = 1.0 / math.sqrt(d)
            attn = ipa_ops.attention_forward_cuda(
                q.contiguous().float(),
                k.contiguous().float(),
                q_pts.contiguous().float(),
                k_pts.contiguous().float(),
                pair_bias.contiguous().float(),
                w.contiguous().float(),
                B, N, h, d, Qp, inv_sqrt_d
            )
            
            # Value aggregation
            out_combined = ipa_ops.value_aggregate_cuda(
                attn.contiguous().float(),
                v.contiguous().float(),
                v_pts.contiguous().float(),
                B, N, h, d, Vp
            )
            
            # Split scalar and point outputs
            out_scalar = out_combined[..., :h * d].view(B, N, h, d)
            out_pts_global = out_combined[..., h * d:].view(B, N, h * Vp, 3)
            
            # Transform points back to local frame
            out_pts_local = ipa_ops.inverse_frame_cuda(
                out_pts_global.contiguous().float(),
                rotations.contiguous().float(),
                translations.contiguous().float(),
                h * Vp
            )
            out_pts_flat = out_pts_local.view(B, N, h * Vp * 3)
            
            out = torch.cat([out_scalar.reshape(B, N, h * d), out_pts_flat], dim=-1)
            return self.out_proj(out)
        else:

            q_pts = self._apply_frame(q_pts_local, rotations, translations).view(B, N, h, Qp, 3)
            k_pts = self._apply_frame(k_pts_local, rotations, translations).view(B, N, h, Qp, 3)
            v_pts = self._apply_frame(v_pts_local, rotations, translations).view(B, N, h, Vp, 3)

            attn_scalar = torch.einsum("bihd,bjhd->bhij", q, k) / math.sqrt(d)

            q_expand = q_pts.unsqueeze(2)
            k_expand = k_pts.unsqueeze(1)

            out_scalar = torch.einsum("bhij,bjhd->bihd", attn, v).reshape(B, N, h * d)

            out_pts_flat = out_pts_local.reshape(B, N, h * Vp * 3)

            out = torch.cat([out_scalar, out_pts_flat], dim=-1)
            return self.out_proj(out)

        global_pts = torch.einsum("bnpc,bncd->bnpd", pts, rotations) + translations.unsqueeze(2)
        return global_pts

    def _apply_inv_frame(self, pts, rotations, translations):

        local_pts = torch.einsum("bnpc,bndc->bnpd", centered, rotations)
        return local_pts
