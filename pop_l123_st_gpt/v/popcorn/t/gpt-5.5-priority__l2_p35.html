class Model(nn.Module):
    """
    Fused 1D convolution + GroupNorm for temporal sequences (N, C, T).

    Typical in time-series heads, audio / motor trajectories, and sequence models
    that use strided or padded Conv1d followed by per-channel normalization without
    collapsing the time dimension (GroupNorm over `out_channels` on (N, C_out, T')).

    Reference fusion target: `Conv1d` → `GroupNorm` (no activation), matching a
    common stem or residual branch in temporal backbones.
    """

    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        num_groups: int,
        kernel_size: int = 3,
        stride: int = 1,
        padding: int = 1,
        dilation: int = 1,
        eps: float = 1e-5,
        affine: bool = True,
    ):
        super().__init__()
        assert out_channels % num_groups == 0
        self.conv = nn.Conv1d(
            in_channels,
            out_channels,
            kernel_size,
            stride=stride,
            padding=padding,
            dilation=dilation,
            bias=False,
        )
        self.norm = nn.GroupNorm(
            num_groups=num_groups,
            num_channels=out_channels,
            eps=eps,
            affine=affine,
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C_in, T) temporal feature map
        Returns:
            (N, C_out, T') with T' determined by conv stride / padding / kernel
        """
        return self.norm(self.conv(x))

__global__ void conv1d_k3_s1p1_kernel(const float* __restrict__ x,
                                      const float* __restrict__ w,
                                      float* __restrict__ tmp,
                                      int N, int Cin, int Cout, int T) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = N * Cout * T;
    if (idx >= total) return;
    int t = idx % T;
    int co = (idx / T) % Cout;
    int n = idx / (T * Cout);

    float sum = 0.0f;
    const float* __restrict__ wco = w + (size_t)co * Cin * 3;
    const float* __restrict__ xn = x + (size_t)n * Cin * T;

    if (t > 0 && t < T - 1) {
        for (int ci = 0; ci < Cin; ++ci) {
            const float* __restrict__ xp = xn + (size_t)ci * T + t;
            const float* __restrict__ wp = wco + ci * 3;
            sum = fmaf(xp[-1], wp[0], sum);
            sum = fmaf(xp[ 0], wp[1], sum);

__global__ void conv1d_generic_kernel(const float* __restrict__ x,
                                      const float* __restrict__ w,
                                      float* __restrict__ tmp,
                                      int N, int Cin, int Tin, int Cout, int K,

    float sum = 0.0f;

__global__ void groupnorm_from_tmp_kernel(const float* __restrict__ tmp,
                                          const float* __restrict__ gamma,
                                          const float* __restrict__ beta,
                                          float* __restrict__ y,
                                          int N, int Cout, int T, int G,
                                          float eps, bool affine) {
    extern __shared__ float smem[];
    float* ssum = smem;
    float* ssq  = smem + blockDim.x;

    int ng = blockIdx.x;
    int n = ng / G;
    int g = ng - n * G;
    int cpg = Cout / G;
    int elems = cpg * T;
    int c0 = g * cpg;

    float sum = 0.0f;
    float sq = 0.0f;
    for (int i = threadIdx.x; i < elems; i += blockDim.x) {
        int lc = i / T;
        int t = i - lc * T;
        int c = c0 + lc;
        float v = tmp[((size_t)n * Cout + c) * T + t];
        sum += v;
        sq = fmaf(v, v, sq);
    }
    ssum[threadIdx.x] = sum;
    ssq[threadIdx.x] = sq;
    __syncthreads();

    for (int offset = blockDim.x >> 1; offset > 0; offset >>= 1) {
        if (threadIdx.x < offset) {
            ssum[threadIdx.x] += ssum[threadIdx.x + offset];
            ssq[threadIdx.x]  += ssq[threadIdx.x + offset];
        }
        __syncthreads();
    }

    float mean = ssum[0] / (float)elems;
    float var = ssq[0] / (float)elems - mean * mean;
    if (var < 0.0f) var = 0.0f;
    float invstd = 1.0f / sqrtf(var + eps);

    for (int i = threadIdx.x; i < elems; i += blockDim.x) {

        super().__init__()
        assert out_channels % num_groups == 0
        self.conv = nn.Conv1d(in_channels, out_channels, kernel_size,
                              stride=stride, padding=padding, dilation=dilation,
                              bias=False)
        self.norm = nn.GroupNorm(num_groups=num_groups, num_channels=out_channels,
                                 eps=eps, affine=affine)
        self.num_groups = num_groups
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.eps = eps
        self.affine = affine

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if self.affine:
            gamma = self.norm.weight
            beta = self.norm.bias
        else:
            gamma = torch.empty(0, device=x.device, dtype=x.dtype)
            beta = torch.empty(0, device=x.device, dtype=x.dtype)
        return _ext.fused_conv1d_groupnorm_forward(
            x, self.conv.weight, gamma, beta,
            self.num_groups, self.stride, self.padding, self.dilation,
            float(self.eps), bool(self.affine)
        )
