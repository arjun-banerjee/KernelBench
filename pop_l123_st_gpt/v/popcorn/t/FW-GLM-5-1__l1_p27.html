"""
Fixed permutation along the feature axis: y[b, j] = x[b, perm[j]].
Custom CUDA kernel implementation.
"""

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Generate permutation with same seed as reference
_perm_generator = torch.Generator()
_perm_generator.manual_seed(10)
PERM = torch.randperm(36, generator=_perm_generator)

# Compile CUDA kernel at module load time
_cuda_source = """
__global__ void permute_kernel(const float* __restrict__ x, float* __restrict__ y,
                               const int* __restrict__ perm, int B, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * N;
    
    if (idx < total) {
        int j = idx % N;
        int b = idx / N;
        int src_j = perm[j];
        y[idx] = x[b * N + src_j];
    }
}

torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm) {
    TORCH_CHECK(x.dim() == 2, "Input must be 2D");
    TORCH_CHECK(x.scalar_type() == torch::kFloat32, "Input must be float");
    TORCH_CHECK(perm.scalar_type() == torch::kInt32, "Perm must be int32");
    
    int B = x.size(0);
    int N = x.size(1);
    
    auto y = torch::empty_like(x);
    
    int total = B * N;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    
    permute_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(),
        y.data_ptr<float>(),
        perm.data_ptr<int>(),
        B, N
    );
    
    return y;
}
"""

_cpp_source = "torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm);"

_permute_module = load_inline(
    name="permute_ext",
    cpp_sources=_cpp_source,
    cuda_sources=_cuda_source,
    functions=["permute_cuda"],
    verbose=False,
)


class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self._perm_device = None
        self._perm_int = None
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Handle 1D input by unsqueezing
        if x.dim() == 1:
            x_2d = x.unsqueeze(0)
            out_2d = self._forward_2d(x_2d)
            return out_2d.squeeze(0)
        return self._forward_2d(x)
    
    def _forward_2d(self, x: torch.Tensor) -> torch.Tensor:
        B, N = x.shape
        
        # Ensure contiguous input for kernel
        if not x.is_contiguous():
            x = x.contiguous()
        
        # Ensure perm is on correct device and int32
        if self._perm_device != x.device:
            self._perm_int = PERM.to(device=x.device, dtype=torch.int32)
            self._perm_device = x.device
        
        return _permute_module.permute_cuda(x, self._perm_int)
