import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Evidence Lower Bound (ELBO) computation for a Variational Autoencoder.
    Combines a reconstruction term (binary cross-entropy) with the
    KL divergence between the approximate posterior q(z|x) = N(mu, sigma^2)
    and the prior p(z) = N(0, I).  The core loss function in VAE training.
    """

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
        )
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)

        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
            nn.Sigmoid(),
        )

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim) – input data in [0, 1]
        Returns:
            (elbo, recon_loss, kl_div):
                elbo:       (B,)
                recon_loss: (B,)
                kl_div:     (B,)
        """
        h = self.encoder(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)

        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)

        x_recon = self.decoder(z)
        recon = F.binary_cross_entropy(x_recon, x, reduction="none").sum(-1)
        kl = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(-1)
        elbo = -(recon + kl)
        return elbo, recon, kl

    float kl_sum = 0.0f;
    
    // Compute reconstruction loss (BCE)
    for (int i = threadIdx.x; i < input_dim; i += blockDim.x) {
        int idx = batch_idx * input_dim + i;
        // Binary cross entropy: -(x * log(x_recon) + (1-x) * log(1-x_recon))
        // Add small epsilon for numerical stability
        float xr = x_recon[idx];
        float xi = x[idx];
        float eps = 1e-7f;
        xr = fmaxf(eps, fminf(1.0f - eps, xr));
        recon_sum += -(xi * logf(xr) + (1.0f - xi) * logf(1.0f - xr));
    }
    
    // Compute KL divergence
    for (int i = threadIdx.x; i < latent_dim; i += blockDim.x) {
        int idx = batch_idx * latent_dim + i;
        float m = mu[idx];
        float lv = logvar[idx];
        // KL = -0.5 * (1 + logvar - mu^2 - exp(logvar))
        kl_sum += -0.5f * (1.0f + lv - m * m - expf(lv));
    }
    
    // Reduce within block
    __shared__ float s_recon[BLOCK_SIZE];
    __shared__ float s_kl[BLOCK_SIZE];
    
    s_recon[threadIdx.x] = recon_sum;
    s_kl[threadIdx.x] = kl_sum;
    __syncthreads();
    
    // Tree reduction
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (threadIdx.x < s) {
            s_recon[threadIdx.x] += s_recon[threadIdx.x + s];
            s_kl[threadIdx.x] += s_kl[threadIdx.x + s];
        }
        __syncthreads();
    }
    
    if (threadIdx.x == 0) {
        recon_loss[batch_idx] = s_recon[0];
        kl_div[batch_idx] = s_kl[0];
        elbo[batch_idx] = -(s_recon[0] + s_kl[0]);
    }
}

// Add bias kernel
__global__ void add_bias_kernel(float* data, const float* bias, int batch_size, int features) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * features;
    if (idx < total) {
        int feature_idx = idx % features;
        data[idx] += bias[feature_idx];
    }
}

// Global handles (initialized once)
static cublasHandle_t cublas_handle = nullptr;
static bool handles_initialized = false;

void init_handles() {
    if (!handles_initialized) {
        cublasCreate(&cublas_handle);
        handles_initialized = true;
    }

// Main forward function
std::tuple<torch::Tensor, torch::Tensor, torch::Tensor> vae_forward(
    torch::Tensor x,
    torch::Tensor encoder_weight, torch::Tensor encoder_bias,
    torch::Tensor fc_mu_weight, torch::Tensor fc_mu_bias,
    torch::Tensor fc_logvar_weight, torch::Tensor fc_logvar_bias,
    torch::Tensor decoder1_weight, torch::Tensor decoder1_bias,
    torch::Tensor decoder2_weight, torch::Tensor decoder2_bias) {
    
    init_handles();
    
    const int batch_size = x.size(0);
    const int input_dim = x.size(1);
    const int hidden_dim = encoder_weight.size(0);
    const int latent_dim = fc_mu_weight.size(0);
    
    // Create output tensors
    auto options = x.options();
    
    // Intermediate tensors
    auto h = torch::empty({batch_size, hidden_dim}, options);
    auto h_relu = torch::empty({batch_size, hidden_dim}, options);
    auto mu = torch::empty({batch_size, latent_dim}, options);
    auto logvar = torch::empty({batch_size, latent_dim}, options);
    auto std = torch::empty({batch_size, latent_dim}, options);
    auto z = torch::empty({batch_size, latent_dim}, options);
    auto noise = torch::empty({batch_size, latent_dim}, options);
    auto decoder_h = torch::empty({batch_size, hidden_dim}, options);
    auto decoder_h_relu = torch::empty({batch_size, hidden_dim}, options);
    auto x_pre_sigmoid = torch::empty({batch_size, input_dim}, options);
    auto x_recon = torch::empty({batch_size, input_dim}, options);
    
    // Output tensors
    auto elbo = torch::empty({batch_size}, options);
    auto recon_loss = torch::empty({batch_size}, options);
    auto kl_div = torch::empty({batch_size}, options);
    
    float alpha = 1.0f, beta = 0.0f;
    
    // Encoder: Linear(input_dim, hidden_dim)
    // h = x @ encoder_weight.T
    cublasSgemm(cublas_handle, 
                CUBLAS_OP_T, CUBLAS_OP_N,
                hidden_dim, batch_size, input_dim,
                &alpha,
                encoder_weight.data_ptr<float>(), input_dim,
                x.data_ptr<float>(), input_dim,
                &beta,
                h.data_ptr<float>(), hidden_dim);
    
    // Add bias and ReLU
    int h_size = batch_size * hidden_dim;
    int blocks = (h_size + BLOCK_SIZE - 1) / BLOCK_SIZE;
    add_bias_kernel<<<blocks, BLOCK_SIZE>>>(h.data_ptr<float>(), encoder_bias.data_ptr<float>(), 
                                             batch_size, hidden_dim);
    relu_kernel<<<blocks, BLOCK_SIZE>>>(h.data_ptr<float>(), h_relu.data_ptr<float>(), h_size);
    
    // fc_mu: Linear(hidden_dim, latent_dim)
    cublasSgemm(cublas_handle,
                CUBLAS_OP_T, CUBLAS_OP_N,
                latent_dim, batch_size, hidden_dim,
                &alpha,
                fc_mu_weight.data_ptr<float>(), hidden_dim,
                h_relu.data_ptr<float>(), hidden_dim,
                &beta,
                mu.data_ptr<float>(), latent_dim);
    
    int mu_size = batch_size * latent_dim;
    blocks = (mu_size + BLOCK_SIZE - 1) / BLOCK_SIZE;
    add_bias_kernel<<<blocks, BLOCK_SIZE>>>(mu.data_ptr<float>(), fc_mu_bias.data_ptr<float>(),
                                            batch_size, latent_dim);
    
    // fc_logvar: Linear(hidden_dim, latent_dim)
    cublasSgemm(cublas_handle,
                CUBLAS_OP_T, CUBLAS_OP_N,
                latent_dim, batch_size, hidden_dim,
                &alpha,
                fc_logvar_weight.data_ptr<float>(), hidden_dim,
                h_relu.data_ptr<float>(), hidden_dim,
                &beta,
                logvar.data_ptr<float>(), latent_dim);
    
    add_bias_kernel<<<blocks, BLOCK_SIZE>>>(logvar.data_ptr<float>(), fc_logvar_bias.data_ptr<float>(),
                                            batch_size, latent_dim);
    
    // Generate random noise using torch (faster than cuRAND for this size)
    auto noise_tensor = torch::randn({batch_size, latent_dim}, options);
    
    // Sampling: z = mu + std * noise
    sampling_kernel<<<blocks, BLOCK_SIZE>>>(mu.data_ptr<float>(), logvar.data_ptr<float>(),
                                            noise_tensor.data_ptr<float>(),
                                            z.data_ptr<float>(), std.data_ptr<float>(), mu_size);
    
    // Decoder layer 1: Linear(latent_dim, hidden_dim)
    cublasSgemm(cublas_handle,
                CUBLAS_OP_T, CUBLAS_OP_N,
                hidden_dim, batch_size, latent_dim,
                &alpha,
                decoder1_weight.data_ptr<float>(), latent_dim,
                z.data_ptr<float>(), latent_dim,
                &beta,
                decoder_h.data_ptr<float>(), hidden_dim);
    
    add_bias_kernel<<<blocks, BLOCK_SIZE>>>(decoder_h.data_ptr<float>(), decoder1_bias.data_ptr<float>(),
                                            batch_size, hidden_dim);
    relu_kernel<<<blocks, BLOCK_SIZE>>>(decoder_h.data_ptr<float>(), decoder_h_relu.data_ptr<float>(), h_size);
    
    // Decoder layer 2: Linear(hidden_dim, input_dim)
    cublasSgemm(cublas_handle,
                CUBLAS_OP_T, CUBLAS_OP_N,
                input_dim, batch_size, hidden_dim,
                &alpha,
                decoder2_weight.data_ptr<float>(), hidden_dim,
                decoder_h_relu.data_ptr<float>(), hidden_dim,
                &beta,
                x_pre_sigmoid.data_ptr<float>(), input_dim);
    
    int x_size = batch_size * input_dim;
    blocks = (x_size + BLOCK_SIZE - 1) / BLOCK_SIZE;
    add_bias_kernel<<<blocks, BLOCK_SIZE>>>(x_pre_sigmoid.data_ptr<float>(), decoder2_bias.data_ptr<float>(),
                                            batch_size, input_dim);
    sigmoid_kernel<<<blocks, BLOCK_SIZE>>>(x_pre_sigmoid.data_ptr<float>(), x_recon.data_ptr<float>(), x_size);
    
    // Compute losses
    loss_kernel<<<batch_size, BLOCK_SIZE>>>(x_recon.data_ptr<float>(), x.data_ptr<float>(),
                                            mu.data_ptr<float>(), logvar.data_ptr<float>(),
                                            elbo.data_ptr<float>(), recon_loss.data_ptr<float>(),
                                            kl_div.data_ptr<float>(),
                                            batch_size, input_dim, latent_dim);
    
    return std::make_tuple(elbo, recon_loss, kl_div);
}

class ModelNew(nn.Module):
    """
    Evidence Lower Bound (ELBO) computation for a Variational Autoencoder.
    Fused CUDA implementation for improved performance.
    """

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        
        # Initialize weights and biases directly (no Sequential wrappers)
        # Encoder: Linear(input_dim, hidden_dim)
        self.encoder_weight = nn.Parameter(torch.empty(hidden_dim, input_dim))
        self.encoder_bias = nn.Parameter(torch.empty(hidden_dim))
        
        # fc_mu: Linear(hidden_dim, latent_dim)
        self.fc_mu_weight = nn.Parameter(torch.empty(latent_dim, hidden_dim))
        self.fc_mu_bias = nn.Parameter(torch.empty(latent_dim))
        
        # fc_logvar: Linear(hidden_dim, latent_dim)
        self.fc_logvar_weight = nn.Parameter(torch.empty(latent_dim, hidden_dim))
        self.fc_logvar_bias = nn.Parameter(torch.empty(latent_dim))
        
        # Decoder layer 1: Linear(latent_dim, hidden_dim)
        self.decoder1_weight = nn.Parameter(torch.empty(hidden_dim, latent_dim))
        self.decoder1_bias = nn.Parameter(torch.empty(hidden_dim))
        
        # Decoder layer 2: Linear(hidden_dim, input_dim)
        self.decoder2_weight = nn.Parameter(torch.empty(input_dim, hidden_dim))
        self.decoder2_bias = nn.Parameter(torch.empty(input_dim))
        
        # Initialize weights
        self._init_weights()
        
        # Store dimensions
        self.input_dim = input_dim
        self.hidden_dim = hidden_dim
        self.latent_dim = latent_dim
    
    def _init_weights(self):
        # Use same initialization as nn.Linear
        for name, param in self.named_parameters():
            if 'weight' in name:
                nn.init.kaiming_uniform_(param, a=5 ** 0.5)
            elif 'bias' in name:
                fan_in, _ = nn.init._calculate_fan_in_and_fan_out(param)
                bound = 1 / (fan_in ** 0.5) if fan_in > 0 else 0
                nn.init.uniform_(param, -bound, bound)
    
    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim) - input data in [0, 1]
        Returns:
            (elbo, recon_loss, kl_div):
                elbo:       (B,)
                recon_loss: (B,)
                kl_div:     (B,)
        """
        # Ensure input is contiguous and on same device as parameters
        x = x.contiguous()
        
        # Call the fused CUDA kernel
        elbo, recon_loss, kl_div = vae_extension.vae_forward(
            x,
            self.encoder_weight, self.encoder_bias,
            self.fc_mu_weight, self.fc_mu_bias,
            self.fc_logvar_weight, self.fc_logvar_bias,
            self.decoder1_weight, self.decoder1_bias,
            self.decoder2_weight, self.decoder2_bias
        )
        
        return elbo, recon_loss, kl_div
