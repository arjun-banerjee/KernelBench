import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Compile the CUDA extension at module load time
gelu_sum_module = load_inline(
    name='gelu_sum_module',
    cpp_sources='''
#include <torch/extension.h>
torch::Tensor gelu_sum_cuda(torch::Tensor x);
''',
    cuda_sources='''
#include <torch/extension.h>
#include <cuda_runtime.h>

// Exact GELU activation: gelu(x) = x * 0.5 * (1 + erf(x / sqrt(2)))
__device__ __forceinline__ float gelu_exact(float x) {
    return x * 0.5f * (1.0f + erff(x * 0.7071067811865475f));
}

__global__ void gelu_sum_kernel(const float* __restrict__ x, 
                                 float* __restrict__ out, 
                                 int R, int BD_total) {
    const int bd = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (bd < BD_total) {
        float sum = 0.0f;
        for (int r = 0; r < R; ++r) {
            sum += gelu_exact(x[r * BD_total + bd]);
        }
        out[bd] = sum;
    }
}

    const int R = x.size(0);

    const int blocks = (BD_total + threads - 1) / threads;
    
    gelu_sum_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(), out.data_ptr<float>(), R, BD_total

}
