import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Differentiable Smith-Waterman-style local sequence alignment scoring.
    Uses a soft-max relaxation of the DP recurrence so gradients can flow
    through the alignment matrix.  
    """
    def __init__(self, gap_open=-1.0, temperature=1.0):
        super().__init__()
        self.gap_open = gap_open
        self.temperature = temperature

    def forward(self, score_matrix: torch.Tensor) -> torch.Tensor:
        """
        Args:
            score_matrix: (B, M, N) – per-position similarity scores
        Returns:
            alignment_score: (B,) – soft local-alignment score (log-sum-exp over DP cells)
        """
        B, M, N = score_matrix.shape
        neg_inf = torch.tensor(float("-inf"), device=score_matrix.device, dtype=score_matrix.dtype)

        # H[i][j] = max(0, H[i-1][j-1]+S[i][j], H[i-1][j]+gap, H[i][j-1]+gap)
        H = score_matrix.new_full((B, M + 1, N + 1), 0.0)

        for i in range(1, M + 1):
            for j in range(1, N + 1):
                match = H[:, i - 1, j - 1] + score_matrix[:, i - 1, j - 1]
                delete = H[:, i - 1, j] + self.gap_open
                insert = H[:, i, j - 1] + self.gap_open
                zero = torch.zeros(B, device=score_matrix.device)
                stack = torch.stack([zero, match, delete, insert], dim=-1)
                H[:, i, j] = self.temperature * torch.logsumexp(stack / self.temperature, dim=-1)

        return H.view(B, -1).max(dim=-1).values

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor sw_forward_cuda(torch::Tensor score, double gap_open, double temperature);

torch::Tensor sw_forward(torch::Tensor score, double gap_open, double temperature) {
    return sw_forward_cuda(score, gap_open, temperature);
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void sw32_kernel(const float* __restrict__ s,
                            float* __restrict__ out,
                            int B, int M, int N,
                            float gap, float temp) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b >= B) return;

    // Rolling DP row. The benchmark uses M=N=32; support up to 32 without
    // dynamic allocation or repeated kernel launches.
    float prev[33];
#pragma unroll
    for (int j = 0; j < 33; ++j) prev[j] = 0.0f;

    float best = 0.0f;
    const float* sb = s + ((long long)b) * M * N;

    for (int i = 1; i <= M; ++i) {
        float left = 0.0f;   // H[i][j-1]
        float diag = 0.0f;   // old H[i-1][j-1]
        int base = (i - 1) * N;
        for (int j = 1; j <= N; ++j) {
            float up = prev[j];
            float match = diag + sb[base + (j - 1)];
            float delv  = up + gap;
            float insv  = left + gap;
            float m = fmaxf(fmaxf(match, delv), fmaxf(insv, 0.0f));
            float invt = 1.0f / temp;
            float sum = expf((0.0f  - m) * invt)
                      + expf((match - m) * invt)
                      + expf((delv  - m) * invt)
                      + expf((insv  - m) * invt);
            float val = m + temp * logf(sum);
            prev[j] = val;
            diag = up;
            left = val;
            best = fmaxf(best, val);
        }
    }
    out[b] = best;
}

// A slightly more general, slower fallback kernel for unusual sizes up to 64.
__global__ void sw64_kernel(const float* __restrict__ s,
                            float* __restrict__ out,
                            int B, int M, int N,
                            float gap, float temp) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b >= B) return;
    float prev[65];
#pragma unroll
    for (int j = 0; j < 65; ++j) prev[j] = 0.0f;
    float best = 0.0f;
    const float* sb = s + ((long long)b) * M * N;
    float invt = 1.0f / temp;
    for (int i = 1; i <= M; ++i) {
        float left = 0.0f, diag = 0.0f;
        int base = (i - 1) * N;
        for (int j = 1; j <= N; ++j) {
            float up = prev[j];
            float match = diag + sb[base + (j - 1)];
            float delv = up + gap;
            float insv = left + gap;
            float m = fmaxf(fmaxf(match, delv), fmaxf(insv, 0.0f));
            float sum = expf((0.0f-m)*invt) + expf((match-m)*invt) + expf((delv-m)*invt) + expf((insv-m)*invt);
            float val = m + temp * logf(sum);
            prev[j] = val;
            diag = up;
            left = val;
            best = fmaxf(best, val);
        }
    }
    out[b] = best;
}

torch::Tensor sw_forward_cuda(torch::Tensor score, double gap_open, double temperature) {
    const int B = (int)score.size(0);
    const int M = (int)score.size(1);
    const int N = (int)score.size(2);
    auto out = torch::empty({B}, score.options());
    const int threads = 32;
    const int blocks = (B + threads - 1) / threads;
    float gap = (float)gap_open;
    float temp = (float)temperature;
    if (M <= 32 && N <= 32) {
        sw32_kernel<<<blocks, threads>>>(score.data_ptr<float>(), out.data_ptr<float>(), B, M, N, gap, temp);
    } else {
        sw64_kernel<<<blocks, threads>>>(score.data_ptr<float>(), out.data_ptr<float>(), B, M, N, gap, temp);
    }
    return out;
}
"""
