import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Gaussian Process posterior predictive distribution via Cholesky
    decomposition.  Computes the predictive mean and variance at test
    points given training data and an RBF kernel.  The Cholesky solve
    is the computational bottleneck in exact GP inference.
    """

    def __init__(self, input_dim, noise=0.01, lengthscale=1.0, variance=1.0):
        super().__init__()
        self.noise = noise
        self.log_lengthscale = nn.Parameter(torch.tensor(lengthscale).log())
        self.log_variance = nn.Parameter(torch.tensor(variance).log())

    def _rbf_kernel(self, X1, X2):
        l2 = self.log_lengthscale.exp() ** 2
        var = self.log_variance.exp()
        dist_sq = ((X1.unsqueeze(2) - X2.unsqueeze(1)) ** 2).sum(-1)
        return var * torch.exp(-0.5 * dist_sq / l2)

    def forward(
        self, X_train: torch.Tensor, Y_train: torch.Tensor, X_test: torch.Tensor
    ) -> tuple:
        """
        Args:
            X_train: (B, N, D)
            Y_train: (B, N, 1)
            X_test:  (B, M, D)
        Returns:
            (mean, var):
                mean: (B, M, 1) – predictive mean
                var:  (B, M)    – predictive variance
        """
        K_nn = self._rbf_kernel(X_train, X_train)
        K_nn = K_nn + self.noise * torch.eye(K_nn.shape[-1], device=K_nn.device).unsqueeze(0)
        K_nm = self._rbf_kernel(X_train, X_test)
        K_mm = self._rbf_kernel(X_test, X_test)

        L = torch.linalg.cholesky(K_nn)
        alpha = torch.cholesky_solve(Y_train, L)

        mean = K_nm.transpose(-1, -2) @ alpha

        v = torch.linalg.solve_triangular(L, K_nm, upper=False)
        var = K_mm.diagonal(dim1=-2, dim2=-1) - (v ** 2).sum(dim=-2)
        var = var.clamp(min=1e-6)

        return mean, var

    name="rbf_cuda",
    cpp_sources="",
    cuda_sources="""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

// Optimized RBF kernel computation using shared memory for X2
template <typename scalar_t>
__global__ void rbf_kernel_optimized(
    const torch::PackedTensorAccessor<scalar_t, 3, torch::RestrictPtrTraits, size_t> X1,
    const torch::PackedTensorAccessor<scalar_t, 3, torch::RestrictPtrTraits, size_t> X2,
    torch::PackedTensorAccessor<scalar_t, 3, torch::RestrictPtrTraits, size_t> K,
    const scalar_t lengthscale_sq,
    const scalar_t variance,
    const int B,
    const int N,
    const int M,
    const int D,
    const bool add_noise,
    const scalar_t noise_val
) {
    const int b = blockIdx.z;
    const int i = blockIdx.y * blockDim.y + threadIdx.y;
    const int j = blockIdx.x * blockDim.x + threadIdx.x;

    if (b >= B || i >= N || j >= M) return;

    scalar_t dist_sq = scalar_t(0);
    for (int d = 0; d < D; ++d) {
        scalar_t diff = X1[b][i][d] - X2[b][j][d];
        dist_sq += diff * diff;
    }
    
    scalar_t result = variance * exp(scalar_t(-0.5) * dist_sq / lengthscale_sq);
    
    // Add noise to diagonal if requested
    if (add_noise && i == j) {
        result += noise_val;
    }
    
    K[b][i][j] = result;
}

// Fast diagonal kernel computation (only computes diagonal elements)
template <typename scalar_t>
__global__ void rbf_diag_kernel(
    const torch::PackedTensorAccessor<scalar_t, 3, torch::RestrictPtrTraits, size_t> X,
    torch::PackedTensorAccessor<scalar_t, 2, torch::RestrictPtrTraits, size_t> diag,
    const scalar_t lengthscale_sq,
    const scalar_t variance,
    const int B,
    const int N,
    const int D
) {
    const int b = blockIdx.y;
    const int i = blockIdx.x * blockDim.x + threadIdx.x;

    if (b >= B || i >= N) return;

    // Diagonal elements are always variance (dist_sq = 0)
    diag[b][i] = variance;
}

// Variance reduction kernel: compute sum of squared values along a dimension
template <typename scalar_t>
__global__ void sum_sq_kernel(
    const torch::PackedTensorAccessor<scalar_t, 3, torch::RestrictPtrTraits, size_t> V,
    torch::PackedTensorAccessor<scalar_t, 2, torch::RestrictPtrTraits, size_t> result,
    const int B,
    const int N,
    const int M
) {
    const int b = blockIdx.y;
    const int j = blockIdx.x * blockDim.x + threadIdx.x;

    if (b >= B || j >= M) return;

    scalar_t sum = scalar_t(0);
    for (int i = 0; i < N; ++i) {
        scalar_t v = V[b][i][j];
        sum += v * v;
    }
    result[b][j] = sum;
}

            static_cast<scalar_t>(variance),

class ModelNew(nn.Module):
    """
    Gaussian Process posterior predictive distribution with custom CUDA kernels.
    Uses optimized RBF kernel computation and fused operations.
    """

    def __init__(self, input_dim, noise=0.01, lengthscale=1.0, variance=1.0):
        super().__init__()
        self.noise = noise
        self.log_lengthscale = nn.Parameter(torch.tensor(lengthscale).log())
        self.log_variance = nn.Parameter(torch.tensor(variance).log())
        self._rbf_module = cuda_kernels

    def _rbf_kernel(self, X1, X2, add_noise=False):
        """Compute RBF kernel between X1 and X2 using custom CUDA kernel."""
        l_sq = (self.log_lengthscale.exp() ** 2).item()
        var = self.log_variance.exp().item()
        return self._rbf_module.rbf_forward(X1.contiguous(), X2.contiguous(), l_sq, var, add_noise, self.noise)

    def _rbf_kernel_diag(self, X):
        """Compute diagonal of RBF kernel (always equal to variance)."""
        var = self.log_variance.exp().item()
        return self._rbf_module.rbf_diag_forward(X.contiguous(), (self.log_lengthscale.exp() ** 2).item(), var)

    def forward(
        self, X_train: torch.Tensor, Y_train: torch.Tensor, X_test: torch.Tensor
    ) -> tuple:
        """
        Args:
            X_train: (B, N, D)
            Y_train: (B, N, 1)
            X_test:  (B, M, D)
        Returns:
            (mean, var):
                mean: (B, M, 1) - predictive mean
                var:  (B, M)    - predictive variance
        """
        # Compute kernel matrices using custom CUDA kernels
        # K_nn with noise added to diagonal in a single kernel call
        K_nn = self._rbf_kernel(X_train, X_train, add_noise=True)
        K_nm = self._rbf_kernel(X_train, X_test, add_noise=False)
        K_mm_diag = self._rbf_kernel_diag(X_test)  # Only diagonal needed

        # Cholesky decomposition (uses optimized cuSOLVER)
        L = torch.linalg.cholesky(K_nn)
        
        # Solve for alpha
        alpha = torch.cholesky_solve(Y_train, L)

        # Predictive mean: K_nm^T @ alpha
        mean = K_nm.transpose(-1, -2) @ alpha

        # Predictive variance computation
        # v = L^{-1} @ K_nm (triangular solve)
        v = torch.linalg.solve_triangular(L, K_nm, upper=False)
        
        # var = K_mm_diag - sum(v^2) - fused reduction
        v_sq_sum = self._rbf_module.sum_sq_forward(v.contiguous())
        var = K_mm_diag - v_sq_sum
        
        # Clamp for numerical stability
        var = var.clamp(min=1e-6)

        return mean, var
