import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Stein Variational Gradient Descent (SVGD) update step.
    Maintains a set of particles approximating a posterior and updates
    them via a kernel-smoothed gradient that balances fitting the target
    with repulsion between particles.  The RBF kernel bandwidth is set
    via the median heuristic.
    """

    def __init__(self, dim):
        super().__init__()
        self.dim = dim
        # Target: N(0, I) for benchmarking
        self.register_buffer("target_mean", torch.zeros(dim))

    def _score_fn(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -x

    def forward(self, particles: torch.Tensor) -> torch.Tensor:
        """
        Args:
            particles: (N, dim) – current particle positions
        Returns:
            phi: (N, dim) – SVGD update direction
        """
        N = particles.shape[0]
        score = self._score_fn(particles)  # (N, dim)

        # Pairwise squared distances
        diff = particles.unsqueeze(0) - particles.unsqueeze(1)  # (N, N, dim)
        dist_sq = (diff ** 2).sum(-1)  # (N, N)

        # Median heuristic for bandwidth
        median_sq = torch.median(dist_sq.view(-1)).clamp(min=1e-5)
        h = median_sq / max(torch.log(torch.tensor(float(N))).item(), 1.0)

        K = torch.exp(-dist_sq / (2.0 * h))  # (N, N)
        grad_K = -diff / h * K.unsqueeze(-1)  # (N, N, dim)

        # SVGD update: phi(x_i) = (1/N) sum_j [ K(x_j, x_i) * score(x_j) + grad_K(x_j, x_i) ]
        phi = (K @ score + grad_K.sum(dim=0)) / N
        return phi


dim = 32
num_particles = 1024


def get_inputs():
    return [torch.randn(num_particles, dim)]


def get_init_inputs():
    return [dim]

#include <ATen/ATen.h>
#include <ATen/cuda/CUDAContext.h>

#include <cuda.h>
#include <cuda_runtime.h>
#include <cmath>

__global__ void pairwise_dist_kernel(const float* __restrict__ x,
                                     float* __restrict__ dist,
                                     int N, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = N * N;
    if (idx >= total) return;
    int i = idx / N;
    int j = idx - i * N;
    const float* xi = x + i * D;
    const float* xj = x + j * D;
    float s = 0.0f;
    if (D == 32) {
        #pragma unroll
        for (int d = 0; d < 32; ++d) {
            float v = xj[d] - xi[d];
            s = fmaf(v, v, s);
        }
    } else {
        for (int d = 0; d < D; ++d) {
            float v = xj[d] - xi[d];
            s = fmaf(v, v, s);
        }
    }
    dist[idx] = s;
}

__global__ void make_kernel_and_rowsum_kernel(float* __restrict__ K,
                                              float* __restrict__ rowsum,
                                              const float* __restrict__ hptr,
                                              int N) {
    int i = blockIdx.x;
    int tid = threadIdx.x;
    float h = hptr[0];
    float inv_2h = -0.5f / h;
    float local = 0.0f;
    for (int j = tid; j < N; j += blockDim.x) {
        int idx = i * N + j;
        float kval = expf(K[idx] * inv_2h);
        K[idx] = kval;
        local += kval;
    }
    extern __shared__ float smem[];
    smem[tid] = local;
    __syncthreads();
    for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) smem[tid] += smem[tid + stride];
        __syncthreads();
    }
    if (tid == 0) rowsum[i] = smem[0];
}

__global__ void combine_kernel(const float* __restrict__ x,
                               const float* __restrict__ kx,
                               const float* __restrict__ rowsum,
                               const float* __restrict__ hptr,
                               float* __restrict__ out,
                               int N, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = N * D;
    if (idx >= total) return;
    int i = idx / D;
    float invh = 1.0f / hptr[0];
    float val = (-kx[idx] + (kx[idx] - x[idx] * rowsum[i]) * invh) / (float)N;
    out[idx] = val;
}

torch::Tensor svgd_cuda_forward(torch::Tensor particles) {
    TORCH_CHECK(particles.is_cuda(), "particles must be a CUDA tensor");
    TORCH_CHECK(particles.scalar_type() == at::kFloat, "only float32 is supported");
    c10::cuda::CUDAGuard device_guard(particles.device());

    auto x = particles.contiguous();
    const int64_t N64 = x.size(0);
    const int64_t D64 = x.size(1);
    TORCH_CHECK(N64 <= INT_MAX && D64 <= INT_MAX, "tensor too large");
    int N = (int)N64;
    int D = (int)D64;

    auto opts = x.options();
    auto K = torch::empty({N, N}, opts);
    auto rowsum = torch::empty({N}, opts);

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const int threads = 256;
    int total_pairs = N * N;
    int blocks_pairs = (total_pairs + threads - 1) / threads;
    pairwise_dist_kernel<<<blocks_pairs, threads, 0, stream>>>(
        x.data_ptr<float>(), K.data_ptr<float>(), N, D);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    // Exact median heuristic, matching torch.median(dist_sq.view(-1)).clamp(min=1e-5)
    auto flat = K.reshape({N64 * N64});
    auto med = at::median(flat);
    auto med_clamped = at::clamp_min(med, 1e-5);
    float denom_f = logf((float)N);
    if (denom_f < 1.0f) denom_f = 1.0f;
    auto h = med_clamped / (double)denom_f;

    make_kernel_and_rowsum_kernel<<<N, threads, threads * sizeof(float), stream>>>(
        K.data_ptr<float>(), rowsum.data_ptr<float>(), h.data_ptr<float>(), N);
    C10_CUDA_KERNEL_LAUNCH_CHECK();

    // Kx = K @ particles; then phi = (-Kx + (Kx - x * rowsum) / h) / N
    auto kx = at::mm(K, x);
    auto out = torch::empty_like(x);
    int total = N * D;
    int blocks = (total + threads - 1) / threads;
    combine_kernel<<<blocks, threads, 0, stream>>>(
        x.data_ptr<float>(), kx.data_ptr<float>(), rowsum.data_ptr<float>(),
        h.data_ptr<float>(), out.data_ptr<float>(), N, D);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
