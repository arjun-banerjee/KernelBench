import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Channel-wise gated delta attention recurrence.

    The recurrent state decays independently for each output channel, which
    makes the update closer to channel-gated delta attention variants used in
    fast long-context baselines.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-2)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor gated_delta_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, torch::Tensor gate);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void gated_delta_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    float* __restrict__ out,
    int B, int T, int H, int D)
{
    // One warp computes one independent output channel/column j for one (batch, head).
    // lane i owns recurrent state[row=i, col=j].
    const int lane = threadIdx.x & 31;
    const int idx = blockIdx.x;
    const int j = idx % D;
    const int tmp = idx / D;
    const int h = tmp % H;
    const int b = tmp / H;

    float s = 0.0f;
    const int bh_base = (b * T * H + h) * D;
    const int step_stride = H * D;

    #pragma unroll 1
    for (int t = 0; t < T; ++t) {
        const int base = bh_base + t * step_stride;
        const float g = __ldg(gate + base + j);
        const float vv = __ldg(v + base + j);

        float prod = 0.0f;
        if (lane < D) {
            const float kk = __ldg(k + base + lane);
            const float qq = __ldg(q + base + lane);
            // Match PyTorch's separate multiply then add more closely than an FMA.
            const float decayed = __fmul_rn(g, s);
            const float kv = __fmul_rn(kk, vv);
            s = __fadd_rn(decayed, kv);
            prod = __fmul_rn(qq, s);
        }

        // Warp reduction over row dimension to form y[t, j].
        prod = __fadd_rn(prod, __shfl_down_sync(0xffffffff, prod, 16));
        prod = __fadd_rn(prod, __shfl_down_sync(0xffffffff, prod, 8));
        prod = __fadd_rn(prod, __shfl_down_sync(0xffffffff, prod, 4));
        prod = __fadd_rn(prod, __shfl_down_sync(0xffffffff, prod, 2));
        prod = __fadd_rn(prod, __shfl_down_sync(0xffffffff, prod, 1));

        if (lane == 0) {
            out[base + j] = prod;
        }
    }
}

torch::Tensor gated_delta_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, torch::Tensor gate) {
    const c10::cuda::CUDAGuard device_guard(q.device());
    auto out = torch::empty_like(q);

    const int B = (int)q.size(0);
    const int T = (int)q.size(1);
    const int H = (int)q.size(2);
    const int D = (int)q.size(3);

    const int blocks = B * H * D;
    const dim3 block(32);
    const dim3 grid(blocks);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    gated_delta_kernel<<<grid, block, 0, stream>>>(
        q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(), gate.data_ptr<float>(),
        out.data_ptr<float>(), B, T, H, D);

    return out;
}
"""
