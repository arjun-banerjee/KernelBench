import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a causal 1D convolution: left-padded so each output only depends on current and past inputs.
    Used in autoregressive sequence models (Mamba, Hyena, WaveNet, TCN) for causal sequence processing.

    Args:
        in_channels (int): Number of input channels.
        out_channels (int): Number of output channels.
        kernel_size (int): Size of the convolution kernel.
        dilation (int, optional): Dilation rate. Defaults to 1.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int = 1, bias: bool = False):
        super(Model, self).__init__()
        self.causal_padding = (kernel_size - 1) * dilation
        self.conv1d = nn.Conv1d(in_channels, out_channels, kernel_size, dilation=dilation, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = torch.nn.functional.pad(x, (self.causal_padding, 0))
        return self.conv1d(x)

#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define BM 32
#define BN 32
#define BK 32

__global__ __launch_bounds__(256, 2)
void causal_conv1d_kernel(const float* __restrict__ x,
                          const float* __restrict__ w,
                          const float* __restrict__ bias,
                          float* __restrict__ y,
                          int B, int C, int O, int L, int K, int dilation,
                          int has_bias) {
    __shared__ float As[BM][BK];
    __shared__ float Bs[BK][BN + 1];

    const int tx = threadIdx.x; // row within half tile
    const int ty = threadIdx.y; // col within half tile
    const int tid = ty * blockDim.x + tx;

    const int m_base = blockIdx.x * BM;
    const int o_base = blockIdx.y * BN;
    const int total_k = C * K;
    const int pad = (K - 1) * dilation;

    const int row0 = tx;
    const int row1 = tx + 16;
    const int col0 = ty;
    const int col1 = ty + 16;

    float acc00 = 0.0f, acc01 = 0.0f, acc10 = 0.0f, acc11 = 0.0f;

    for (int kb = 0; kb < total_k; kb += BK) {
        // Load A tile (implicit im2col) with rows contiguous for coalesced reads over time.
        #pragma unroll
        for (int p = 0; p < 4; ++p) {
            int idx = tid + p * 256;
            int kk = idx / BM;
            int rr = idx - kk * BM;
            int r = kb + kk;
            float v = 0.0f;
            if (r < total_k) {
                int ic = r / K;
                int kval = r - ic * K;
                int m = m_base + rr;
                if (m < B * L) {
                    int b = m / L;
                    int t = m - b * L;
                    int xi = t + kval * dilation - pad;
                    if ((unsigned)xi < (unsigned)L) {
                        v = x[((b * C + ic) * L) + xi];
                    }
                }
            }
            As[rr][kk] = v;
        }

        // Load B/weight tile.  Each group reads contiguous kernel/channel elements for one output channel.
        #pragma unroll
        for (int p = 0; p < 4; ++p) {
            int idx = tid + p * 256;
            int jj = idx / BK;
            int kk = idx - jj * BK;
            int r = kb + kk;

            float v = 0.0f;

                int ic = r / K;
                int kval = r - ic * K;

            float b1 = Bs[kk][col1];
            acc00 = fmaf(a0, b0, acc00);
            acc01 = fmaf(a0, b1, acc01);
            acc10 = fmaf(a1, b0, acc10);
            acc11 = fmaf(a1, b1, acc11);
        }
