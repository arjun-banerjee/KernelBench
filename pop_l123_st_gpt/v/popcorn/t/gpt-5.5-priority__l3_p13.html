import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Conditional Variational Autoencoder (CVAE) with full encoder–decoder
    forward pass and loss computation.  The encoder conditions on both
    input x and label y; the decoder conditions on z and y.  Returns
    reconstruction loss + KL divergence.
    """

    def __init__(self, input_dim, label_dim, hidden_dim, latent_dim):
        super().__init__()
        # Encoder: q(z | x, y)
        self.enc = nn.Sequential(
            nn.Linear(input_dim + label_dim, hidden_dim),
            nn.ReLU(),
        )
        self.enc_mu = nn.Linear(hidden_dim, latent_dim)
        self.enc_logvar = nn.Linear(hidden_dim, latent_dim)

        # Decoder: p(x | z, y)
        self.dec = nn.Sequential(
            nn.Linear(latent_dim + label_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
        )

    def forward(self, x: torch.Tensor, y: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim)  – input data
            y: (B, label_dim)  – conditioning label (one-hot)
        Returns:
            (loss, recon_loss, kl_loss):
                loss:       scalar – total CVAE loss
                recon_loss: scalar
                kl_loss:    scalar
        """
        h = self.enc(torch.cat([x, y], dim=-1))
        mu = self.enc_mu(h)
        logvar = self.enc_logvar(h)

        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)

        x_recon = self.dec(torch.cat([z, y], dim=-1))
        recon_loss = F.mse_loss(x_recon, x, reduction="sum") / x.shape[0]
        kl_loss = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum() / x.shape[0]
        loss = recon_loss + kl_loss
        return loss, recon_loss, kl_loss


input_dim = 256
label_dim = 10
hidden_dim = 128
latent_dim = 32
batch_size = 128


def get_inputs():
    x = torch.randn(batch_size, input_dim)
    y = torch.zeros(batch_size, label_dim)
    y.scatter_(1, torch.randint(0, label_dim, (batch_size, 1)), 1.0)
    return [x, y]


def get_init_inputs():
    return [input_dim, label_dim, hidden_dim, latent_dim]

    #pragma unroll 4
    for (int k = 0; k < H; ++k) {
        float hv = hr[k];
        sm = fmaf(hv, wm[k], sm);
        sl = fmaf(hv, wl[k], sl);
    }
    mu[row * L + j] = sm;
    logvar[row * L + j] = sl;
}

__global__ void z_kl_kernel(const float* __restrict__ mu, const float* __restrict__ logvar,
                            const float* __restrict__ eps, float* __restrict__ z,
                            float* __restrict__ row_kl, int B, int L) {
    int row = blockIdx.x;
    int t = threadIdx.x;
    __shared__ float buf[64];
    float v = 0.0f;
    if (row < B && t < L) {
        int idx = row * L + t;
        float m = mu[idx];
        float lv = logvar[idx];
        z[idx] = m + expf(0.5f * lv) * eps[idx];
        v = -0.5f * (1.0f + lv - m * m - expf(lv));
    }
    buf[t] = v;
    __syncthreads();
    for (int off = 32; off > 0; off >>= 1) {
        if (t < off) buf[t] += buf[t + off];
        __syncthreads();
    }
    if (t == 0) row_kl[row] = buf[0];
}

__global__ void dec1_kernel(const float* __restrict__ z, const float* __restrict__ y,
                            const float* __restrict__ w, const float* __restrict__ b,
                            float* __restrict__ dh,
                            int B, int L, int label_dim, int H) {
    int row = blockIdx.x;
    int j = threadIdx.x;
    if (row >= B || j >= H) return;
    int in_total = L + label_dim;
    const float* wj = w + j * in_total;
    float s = b[j];
    const float* zr = z + row * L;
    #pragma unroll
    for (int k = 0; k < L; ++k) s = fmaf(zr[k], wj[k], s);
    const float* yr = y + row * label_dim;
    #pragma unroll
    for (int k = 0; k < label_dim; ++k) s = fmaf(yr[k], wj[L + k], s);
    dh[row * H + j] = s > 0.0f ? s : 0.0f;
}

__global__ void dec2_loss_kernel(const float* __restrict__ dh, const float* __restrict__ x,
                                 const float* __restrict__ w, const float* __restrict__ b,
                                 float* __restrict__ row_recon,
                                 int B, int H, int input_dim) {
    int row = blockIdx.x;
    int j = threadIdx.x;
    __shared__ float buf[256];
    float val = 0.0f;
    if (row < B && j < input_dim) {
        const float* hr = dh + row * H;
        const float* wj = w + j * H;
        float s = b[j];
        #pragma unroll 4
        for (int k = 0; k < H; ++k) s = fmaf(hr[k], wj[k], s);
        float d = s - x[row * input_dim + j];
        val = d * d;
    }
    buf[j] = val;
    __syncthreads();
    for (int off = 128; off > 0; off >>= 1) {
        if (j < off) buf[j] += buf[j + off];
        __syncthreads();
    }
    if (j == 0) row_recon[row] = buf[0];
}

__global__ void final_reduce_kernel(const float* __restrict__ row_recon, const float* __restrict__ row_kl,
                                    float* __restrict__ loss, float* __restrict__ recon, float* __restrict__ kl,
                                    int B) {
    int t = threadIdx.x;
    __shared__ float sr[256];
    __shared__ float sk[256];
    float r = 0.0f, ksum = 0.0f;
    for (int i = t; i < B; i += blockDim.x) {
        r += row_recon[i];
        ksum += row_kl[i];
    }
    sr[t] = r;
    sk[t] = ksum;
    __syncthreads();
    for (int off = 128; off > 0; off >>= 1) {
        if (t < off) {
            sr[t] += sr[t + off];
            sk[t] += sk[t + off];
        }
        __syncthreads();
    }
    if (t == 0) {
        float invB = 1.0f / (float)B;
        float rr = sr[0] * invB;
        float kk = sk[0] * invB;
        recon[0] = rr;
        kl[0] = kk;
        loss[0] = rr + kk;
    }
}
