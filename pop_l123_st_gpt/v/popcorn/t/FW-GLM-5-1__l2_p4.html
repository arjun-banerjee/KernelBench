class Model(nn.Module):
    """
    Graph-Fused VLA (GF-VLA) style **scene graph relation update** in one forward.

    Objects are nodes with vision-language aligned embeddings; directed pairs
    (i → j) carry pairwise conditioning (geometry, language grounding, or fused
    VLA features). This reference fuses:

      LayerNorm(nodes) → multi-head attention over objects with **pairwise
      relation MLP bias** on edges → residual write-back.

    Intended as a fusion target for navigation / manipulation stacks that maintain
    explicit scene graphs and update them from multimodal observations.
    """

    def __init__(self, dim: int, pair_dim: int, num_heads: int):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.scale = 1.0 / math.sqrt(self.head_dim)

        self.norm = nn.LayerNorm(dim)
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.rel_mlp = nn.Sequential(
            nn.Linear(pair_dim, pair_dim),
            nn.SiLU(),
            nn.Linear(pair_dim, num_heads),
        )
        self.out_proj = nn.Linear(dim, dim, bias=False)

    def forward(
        self,
        nodes: torch.Tensor,
        pair_feats: torch.Tensor,
        edge_mask: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            nodes: (B, N, D) object / region embeddings.
            pair_feats: (B, N, N, P) pairwise VLA or geometric conditioning.
            edge_mask: (B, N, N) float in {0, 1}; 1 keeps attention over column j.
        Returns:
            (B, N, D) updated node embeddings (residual around attention block).
        """
        B, N, D = nodes.shape
        _, N2, N3, P = pair_feats.shape
        assert N == N2 == N3 and P == self.pair_dim
        h = self.norm(nodes)

        q = self.q_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)

        logits = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        rel_bias = self.rel_mlp(pair_feats).permute(0, 3, 1, 2)
        logits = logits + rel_bias

        logits = logits.masked_fill(edge_mask.unsqueeze(1) < 0.5, float("-inf"))
        attn = F.softmax(logits, dim=-1)
        ctx = torch.matmul(attn, v).transpose(1, 2).reshape(B, N, D)
        return nodes + self.out_proj(ctx)

__global__ void layernorm_kernel(
    const scalar_t* __restrict__ x,

    int block_size = blockDim.x;

    scalar_t* out_ptr = out + (b * N + n) * D;
    
    // Compute mean
    scalar_t sum = 0;
    for (int d = tid; d < D; d += block_size) {
        sum += x_ptr[d];
    }

    // Compute variance
    scalar_t sum_sq = 0;
    for (int d = tid; d < D; d += block_size) {
        scalar_t diff = x_ptr[d] - mean;
        sum_sq += diff * diff;
    }
    s_sum_sq[tid] = sum_sq;
    __syncthreads();
    
    for (int s = block_size / 2; s > 0; s >>= 1) {
        if (tid < s) s_sum_sq[tid] += s_sum_sq[tid + s];

    }
    scalar_t inv_std = rsqrt(s_sum_sq[0] / D + eps);
    
    // Write normalized output
    for (int d = tid; d < D; d += block_size) {
        out_ptr[d] = (x_ptr[d] - mean) * inv_std * w[d] + b[d];
    }
}

// Matrix multiplication kernel for QKV projections
template <typename scalar_t>
__global__ void matmul_kernel(
    const scalar_t* __restrict__ x,
    const scalar_t* __restrict__ w,
    scalar_t* __restrict__ out,
    int B, int N, int in_dim, int out_dim) {
    
    int b = blockIdx.x / N;
    int n = blockIdx.x % N;
    int o = blockIdx.y;
    int tid = threadIdx.x;
    
    if (b >= B || o >= out_dim) return;
    
    const scalar_t* x_ptr = x + (b * N + n) * in_dim;
    scalar_t* out_ptr = out + (b * N + n) * out_dim;
    
    scalar_t val = 0;
    for (int i = tid; i < in_dim; i += blockDim.x) {
        val += x_ptr[i] * w[o * in_dim + i];
    }
    
    // Reduce across threads
    __shared__ scalar_t s_data[256];
    s_data[tid] = val;
    __syncthreads();
    
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) s_data[tid] += s_data[tid + s];
        __syncthreads();
    }
    
    if (tid == 0) {
        out_ptr[o] = s_data[0];
    }
}

// Relation MLP kernel: Linear -> SiLU -> Linear
template <typename scalar_t>
__global__ void rel_mlp_kernel(
    const scalar_t* __restrict__ pair_feats,
    const scalar_t* __restrict__ w1,
    const scalar_t* __restrict__ b1,
    const scalar_t* __restrict__ w2,
    const scalar_t* __restrict__ b2,
    scalar_t* __restrict__ out,
    int B, int N, int P, int H) {
    
    int b = blockIdx.x;
    int i = blockIdx.y;
    int j = blockIdx.z;
    int h = threadIdx.x;
    
    if (h >= H) return;
    
    const scalar_t* feat_ptr = pair_feats + (b * N * N + i * N + j) * P;
    
    // First linear: P -> P with SiLU activation
    // Store intermediate in registers
    scalar_t intermediate[16];  // Assuming P <= 16
    for (int p = 0; p < P; p++) {
        scalar_t val = b1[p];
        for (int k = 0; k < P; k++) {
            val += feat_ptr[k] * w1[p * P + k];
        }
        intermediate[p] = val / (1.0f + expf(-val));  // SiLU approximation
    }
    
    // Second linear: P -> H
    scalar_t out_val = b2[h];
    for (int p = 0; p < P; p++) {
        out_val += intermediate[p] * w2[h * P + p];
    }
    
    out[b * H * N * N + h * N * N + i * N + j] = out_val;
}

// Fused attention kernel
template <typename scalar_t>
__global__ void attention_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ rel_bias,
    const scalar_t* __restrict__ edge_mask,
    const scalar_t* __restrict__ out_proj_w,
    const scalar_t* __restrict__ nodes,
    scalar_t* __restrict__ out,
    int B, int N, int D, int H, int head_dim, scalar_t scale) {
    
    extern __shared__ char smem[];
    scalar_t* s_q = reinterpret_cast<scalar_t*>(smem);
    scalar_t* s_k = reinterpret_cast<scalar_t*>(smem + head_dim * sizeof(scalar_t));
    scalar_t* s_v = s_k + head_dim;
    scalar_t* s_logits = s_v + head_dim;
    scalar_t* s_out = s_logits + N;
    
    int b = blockIdx.x / (H * N);
    int h = (blockIdx.x / N) % H;
    int i = blockIdx.x % N;
    int tid = threadIdx.x;
    
    if (b >= B) return;
    
    // Load Q[i] into shared memory
    if (tid < head_dim) {
        s_q[tid] = q[b * H * N * head_dim + h * N * head_dim + i * head_dim + tid];
    }
    
    // Compute attention scores for all j
    scalar_t max_score = -FLT_MAX;
    scalar_t sum_exp = 0;
    
    for (int j = 0; j < N; j++) {
        // Load K[j] into shared memory
        if (tid < head_dim) {
            s_k[tid] = k[b * H * N * head_dim + h * N * head_dim + j * head_dim + tid];
        }
        __syncthreads();
        
        // Compute Q[i] @ K[j]
        scalar_t score = 0;
        for (int d = 0; d < head_dim; d++) {
            score += s_q[d] * s_k[d];
        }
        score *= scale;
        
        // Add relation bias
        score += rel_bias[b * H * N * N + h * N * N + i * N + j];
        
        // Check edge mask
        scalar_t mask = edge_mask[b * N * N + i * N + j];
        if (mask < 0.5f) {
            score = -FLT_MAX;
        }
        
        s_logits[j] = score;
        if (score > max_score) max_score = score;
    }
    
    // Softmax
    for (int j = 0; j < N; j++) {
        scalar_t exp_score = expf(s_logits[j] - max_score);
        s_logits[j] = exp_score;
        sum_exp += exp_score;
    }
    
    scalar_t inv_sum = 1.0f / sum_exp;
    for (int j = 0; j < N; j++) {
        s_logits[j] *= inv_sum;
    }
    
    // Compute attention output
    for (int d = 0; d < head_dim; d++) {
        s_out[d] = 0;
    }
    
    for (int j = 0; j < N; j++) {
        // Load V[j]
        if (tid < head_dim) {
            s_v[tid] = v[b * H * N * head_dim + h * N * head_dim + j * head_dim + tid];
        }
        __syncthreads();
        
        scalar_t attn = s_logits[j];
        for (int d = 0; d < head_dim; d++) {
            s_out[d] += attn * s_v[d];
        }
    }
    
    // Store attention output for this head
    // Later combine all heads and project output
    if (tid < head_dim) {
        // Store intermediate: [B, H, N, head_dim]
        // We'll write to a temporary buffer
    }
}

// Actually let me use a simpler tiled approach for the full forward
template <typename scalar_t>
__global__ void full_forward_kernel(
    const scalar_t* __restrict__ h_norm,      // [B, N, D] normalized input
    const scalar_t* __restrict__ qkv,         // [B, N, 3D] Q, K, V stacked
    const scalar_t* __restrict__ rel_bias,    // [B, H, N, N]
    const scalar_t* __restrict__ edge_mask,   // [B, N, N]
    const scalar_t* __restrict__ out_proj_w,  // [D, D]
    const scalar_t* __restrict__ nodes,       // [B, N, D]
    scalar_t* __restrict__ attn_out,          // [B, N, D] temporary
    scalar_t* __restrict__ out,               // [B, N, D] final output
    int B, int N, int D, int H, int head_dim, scalar_t scale) {
    
    // Each thread block handles one (batch, head, query_idx)
    int total_blocks = B * H * N;
    int block_idx = blockIdx.x;
    
    if (block_idx >= total_blocks) return;
    
    int b = block_idx / (H * N);
    int h = (block_idx / N) % H;
    int q_idx = block_idx % N;
    int tid = threadIdx.x;
    int block_size = blockDim.x;
    
    // Shared memory for attention
    extern __shared__ char smem[];
    scalar_t* s_q = reinterpret_cast<scalar_t*>(smem);              // [head_dim]
    scalar_t* s_scores = s_q + head_dim;                            // [N]
    scalar_t* s_attn_out = s_scores + N;                            // [head_dim]
    scalar_t* s_v_cache = s_attn_out + head_dim;                    // [head_dim]
    
    // Load Q for this query position
    int q_offset = b * N * D + q_idx * D + h * head_dim;
    for (int d = tid; d < head_dim; d += block_size) {
        s_q[d] = qkv[q_offset + d];  // Q is first D elements
    }
    __syncthreads();
    
    // Compute attention scores for all keys
    scalar_t max_score = -FLT_MAX;
    
    for (int k_idx = tid; k_idx < N; k_idx += block_size) {
        int k_offset = b * N * D + k_idx * D + h * head_dim;
        scalar_t score = 0;
        for (int d = 0; d < head_dim; d++) {
            score += s_q[d] * qkv[k_offset + d];  // K is second D elements
        }
        score *= scale;
        score += rel_bias[b * H * N * N + h * N * N + q_idx * N + k_idx];
        
        scalar_t mask = edge_mask[b * N * N + q_idx * N + k_idx];
        if (mask < 0.5f) score = -FLT_MAX;
        
        s_scores[k_idx] = score;
    }
    __syncthreads();
    
    // Find max for numerical stability
    for (int k_idx = 0; k_idx < N; k_idx++) {
        if (s_scores[k_idx] > max_score) max_score = s_scores[k_idx];
    }
    
    // Compute softmax
    scalar_t sum_exp = 0;
    for (int k_idx = tid; k_idx < N; k_idx += block_size) {
        scalar_t exp_val = expf(s_scores[k_idx] - max_score);
        s_scores[k_idx] = exp_val;
        sum_exp += exp_val;
    }
    
    // Reduce sum_exp
    __shared__ scalar_t s_sum_exp[256];
    s_sum_exp[tid] = (tid < N) ? sum_exp : 0;
    __syncthreads();
    for (int s = block_size / 2; s > 0; s >>= 1) {
        if (tid < s) s_sum_exp[tid] += s_sum_exp[tid + s];
        __syncthreads();
    }
    scalar_t inv_sum = 1.0f / s_sum_exp[0];
    
    for (int k_idx = tid; k_idx < N; k_idx += block_size) {
        s_scores[k_idx] *= inv_sum;
    }
    __syncthreads();
    
    // Compute weighted sum of V
    for (int d = tid; d < head_dim; d += block_size) {
        s_attn_out[d] = 0;
    }
    __syncthreads();
    
    for (int k_idx = 0; k_idx < N; k_idx++) {
        scalar_t attn_w = s_scores[k_idx];
        int v_offset = b * N * D + k_idx * D + D + h * head_dim;  // V is third D elements
        
        if (tid < head_dim) {
            for (int d = 0; d < head_dim; d++) {
                scalar_t v_val = qkv[v_offset + d];
                // This needs all threads to participate
            }
        }
        
        // Load V for this k_idx into cache
        for (int d = tid; d < head_dim; d += block_size) {
            s_v_cache[d] = qkv[v_offset + d];
        }
        __syncthreads();
        
        for (int d = tid; d < head_dim; d += block_size) {
            s_attn_out[d] += attn_w * s_v_cache[d];
        }
        __syncthreads();
    }
    
    // Write attention output for this head
    // Combine heads in output projection kernel
    for (int d = tid; d < head_dim; d += block_size) {
        // Store in a temporary buffer [B, H, N, head_dim]
        // Then another kernel will combine and project
        int out_offset = b * H * N * head_dim + h * N * head_dim + q_idx * head_dim + d;
        attn_out[out_offset] = s_attn_out[d];
    }
}

// Output projection kernel
template <typename scalar_t>
__global__ void output_proj_kernel(
    const scalar_t* __restrict__ attn_out,    // [B, H, N, head_dim]
    const scalar_t* __restrict__ out_proj_w,  // [D, D]
    const scalar_t* __restrict__ nodes,       // [B, N, D]
    scalar_t* __restrict__ out,               // [B, N, D]
    int B, int N, int D, int H, int head_dim) {
    
    int b = blockIdx.x / N;
    int n = blockIdx.x % N;
    int d_out = blockIdx.y;
    int tid = threadIdx.x;
    
    if (b >= B || d_out >= D) return;
    
    // Concatenate heads
    scalar_t val = 0;
    for (int h = 0; h < H; h++) {
        for (int d_in = tid; d_in < head_dim; d_in += blockDim.x) {
            int attn_offset = b * H * N * head_dim + h * N * head_dim + n * head_dim + d_in;
            int w_offset = d_out * D + h * head_dim + d_in;
            val += attn_out[attn_offset] * out_proj_w[w_offset];
        }
    }
    
    __shared__ scalar_t s_val[256];
    s_val[tid] = val;
    __syncthreads();
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) s_val[tid] += s_val[tid + s];
        __syncthreads();
    }
    
    if (tid == 0) {
        // Add residual
        out[b * N * D + n * D + d_out] = nodes[b * N * D + n * D + d_out] + s_val[0];
    }
}

// Complete fused forward kernel - simplified version for correctness
template <typename scalar_t>
__global__ void fused_forward_kernel(
    const scalar_t* __restrict__ h_norm,
    const scalar_t* __restrict__ q_w,
    const scalar_t* __restrict__ k_w,
    const scalar_t* __restrict__ v_w,
    const scalar_t* __restrict__ rel_bias,
    const scalar_t* __restrict__ edge_mask,
    const scalar_t* __restrict__ out_proj_w,
    const scalar_t* __restrict__ nodes,
    scalar_t* __restrict__ out,
    int B, int N, int D, int H, int head_dim, scalar_t scale) {
    
    // Each block handles one (batch, head, query_position)
    int block_idx = blockIdx.x;
    int total_blocks = B * H * N;
    if (block_idx >= total_blocks) return;
    
    int b = block_idx / (H * N);
    int rem = block_idx % (H * N);
    int h = rem / N;
    int q_idx = rem % N;
    int tid = threadIdx.x;
    int warp_size = 32;
    int lane_id = tid % warp_size;
    int warp_id = tid / warp_size;
    
    extern __shared__ char smem[];
    scalar_t* s_q = reinterpret_cast<scalar_t*>(smem);
    scalar_t* s_k = s_q + head_dim;
    scalar_t* s_v = s_k + head_dim;
    scalar_t* s_scores = s_v + head_dim;
    scalar_t* s_attn_out = s_scores + N;
    
    // Compute Q projection for this position
    // Q = h_norm[b, q_idx, :] @ q_w^T
    // But we only need the head h portion
    // Q_h = h_norm @ q_w[h*head_dim:(h+1)*head_dim, :]
    
    if (tid == 0) {
        for (int d = 0; d < head_dim; d++) {
            scalar_t val = 0;
            for (int k = 0; k < D; k++) {
                val += h_norm[b * N * D + q_idx * D + k] * q_w[(h * head_dim + d) * D + k];
            }
            s_q[d] = val;
        }
    }
    __syncthreads();
    
    // Compute attention scores Q @ K^T + rel_bias
    scalar_t max_score = -FLT_MAX;
    
    for (int k_idx = 0; k_idx < N; k_idx++) {
        // Compute K for position k_idx
        if (tid == 0) {
            for (int d = 0; d < head_dim; d++) {
                scalar_t val = 0;
                for (int k = 0; k < D; k++) {
                    val += h_norm[b * N * D + k_idx * D + k] * k_w[(h * head_dim + d) * D + k];
                }
                s_k[d] = val;
            }
        }
        __syncthreads();
        
        scalar_t score = 0;
        if (tid == 0) {
            for (int d = 0; d < head_dim; d++) {
                score += s_q[d] * s_k[d];
            }
            score *= scale;
            score += rel_bias[b * H * N * N + h * N * N + q_idx * N + k_idx];
            
            scalar_t mask = edge_mask[b * N * N + q_idx * N + k_idx];
            if (mask < 0.5f) score = -FLT_MAX;
            
            s_scores[k_idx] = score;
            if (score > max_score) max_score = score;
        }
    }
    __syncthreads();
    
    // Softmax - parallel reduction
    scalar_t sum_exp = 0;
    if (tid == 0) {
        for (int k_idx = 0; k_idx < N; k_idx++) {
            scalar_t exp_val = expf(s_scores[k_idx] - max_score);
            s_scores[k_idx] = exp_val;
            sum_exp += exp_val;
        }
        scalar_t inv_sum = 1.0f / (sum_exp + 1e-10f);
        for (int k_idx = 0; k_idx < N; k_idx++) {
            s_scores[k_idx] *= inv_sum;
        }
    }
    __syncthreads();
    
    // Attention @ V
    if (tid == 0) {
        for (int d = 0; d < head_dim; d++) {
            s_attn_out[d] = 0;
        }
        
        for (int k_idx = 0; k_idx < N; k_idx++) {
            scalar_t attn = s_scores[k_idx];
            
            // Compute V for this position
            for (int d = 0; d < head_dim; d++) {
                scalar_t v_val = 0;
                for (int k = 0; k < D; k++) {
                    v_val += h_norm[b * N * D + k_idx * D + k] * v_w[(h * head_dim + d) * D + k];
                }
                s_v[d] = v_val;
            }
            
            for (int d = 0; d < head_dim; d++) {
                s_attn_out[d] += attn * s_v[d];
            }
        }
    }
    __syncthreads();
    
    // Write attention output to global memory
    // [B, H, N, head_dim]
    for (int d = tid; d < head_dim; d += blockDim.x) {
        out[block_idx * head_dim + d] = s_attn_out[d];
    }
}

// Output projection + residual
template <typename scalar_t>
__global__ void output_proj_residual_kernel(
    const scalar_t* __restrict__ attn_out,    // [B, H, N, head_dim]
    const scalar_t* __restrict__ out_proj_w,  // [D, D]
    const scalar_t* __restrict__ nodes,       // [B, N, D]
    scalar_t* __restrict__ out,               // [B, N, D]
    int B, int N, int D, int H, int head_dim) {
    
    int b = blockIdx.x / N;
    int n = blockIdx.x % N;
    int d_out = threadIdx.x;
    
    if (d_out >= D) return;
    
    // Output projection: concat heads and project
    scalar_t proj_val = 0;
    for (int h = 0; h < H; h++) {
        for (int d_in = 0; d_in < head_dim; d_in++) {
            int attn_idx = b * H * N * head_dim + h * N * head_dim + n * head_dim + d_in;
            int w_idx = d_out * D + h * head_dim + d_in;
            proj_val += attn_out[attn_idx] * out_proj_w[w_idx];
        }
    }
    
    // Add residual
    out[b * N * D + n * D + d_out] = nodes[b * N * D + n * D + d_out] + proj_val;
}
