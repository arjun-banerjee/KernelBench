import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor mean4_forward(torch::Tensor x);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void mean4_float4_kernel(const float* __restrict__ x, float* __restrict__ y) {
    // Problem is fixed by the benchmark: x is (4,16,48), so output has 768 floats = 192 float4s.
    int tid = threadIdx.x;
    if (tid < 192) {
        const float4* __restrict__ r0 = reinterpret_cast<const float4*>(x);
        const float4* __restrict__ r1 = reinterpret_cast<const float4*>(x + 768);
        const float4* __restrict__ r2 = reinterpret_cast<const float4*>(x + 1536);
        const float4* __restrict__ r3 = reinterpret_cast<const float4*>(x + 2304);
        float4 a = r0[tid];
        float4 b = r1[tid];
        float4 c = r2[tid];
        float4 d = r3[tid];
        float4 o;
        o.x = (a.x + b.x + c.x + d.x) * 0.25f;
        o.y = (a.y + b.y + c.y + d.y) * 0.25f;
        o.z = (a.z + b.z + c.z + d.z) * 0.25f;
        o.w = (a.w + b.w + c.w + d.w) * 0.25f;
        reinterpret_cast<float4*>(y)[tid] = o;
    }
}

__global__ void mean4_scalar_kernel(const float* __restrict__ x, float* __restrict__ y, int n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) {
        y[i] = (x[i] + x[n + i] + x[2 * n + i] + x[3 * n + i]) * 0.25f;
    }
}

torch::Tensor mean4_forward(torch::Tensor x) {
    auto y = torch::empty({x.size(1), x.size(2)}, x.options());
    int n = (int)(x.size(1) * x.size(2));
    const float* xp = x.data_ptr<float>();
    float* yp = y.data_ptr<float>();
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if (n == 768) {
        mean4_float4_kernel<<<1, 256, 0, stream>>>(xp, yp);
    } else {
        int threads = 256;
        int blocks = (n + threads - 1) / threads;
        mean4_scalar_kernel<<<blocks, threads, 0, stream>>>(xp, yp, n);
    }
    return y;
}
"""
