import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Problem constants
R = 4
B = 12
H = 32
K = 56

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor vg_linear_cuda(torch::Tensor x, torch::Tensor weight, torch::Tensor bias);

torch::Tensor vg_linear(torch::Tensor x, torch::Tensor weight, torch::Tensor bias) {
    return vg_linear_cuda(x, weight, bias);
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

constexpr int R_ = 4;
constexpr int B_ = 12;
constexpr int H_ = 32;
constexpr int N_ = 128;
constexpr int K_ = 56;
constexpr int WARPS_PER_BLOCK = 8;

__global__ __launch_bounds__(256, 2)
void vg_linear_kernel(const float* __restrict__ x,
                      const float* __restrict__ w,
                      const float* __restrict__ bias,
                      float* __restrict__ y) {
    // One block computes up to 8 output columns for one batch row.
    const int b = blockIdx.x;
    const int k = blockIdx.y * WARPS_PER_BLOCK + (threadIdx.x >> 5);
    const int lane = threadIdx.x & 31;

    if (k >= K_) return;

    float sum = 0.0f;
    const float* wk = w + k * N_;

    // Dot over virtual concatenation [x[0,b,:], x[1,b,:], x[2,b,:], x[3,b,:]].
    // Each lane handles 4 elements (128 / 32).
    #pragma unroll
    for (int t = 0; t < 4; ++t) {
        int j = lane + 32 * t;
        int r = j >> 5;          // j / H_
        int h = j & 31;          // j % H_
        float xv = x[(r * B_ + b) * H_ + h];
        sum = fmaf(xv, wk[j], sum);
    }

    // Warp reduction.
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    }

    if (lane == 0) {
        y[b * K_ + k] = sum + bias[k];
    }
}

torch::Tensor vg_linear_cuda(torch::Tensor x, torch::Tensor weight, torch::Tensor bias) {
    auto y = torch::empty({B_, K_}, x.options());
    dim3 block(32 * WARPS_PER_BLOCK);
    dim3 grid(B_, (K_ + WARPS_PER_BLOCK - 1) / WARPS_PER_BLOCK);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    vg_linear_kernel<<<grid, block, 0, stream>>>(
        x.data_ptr<float>(), weight.data_ptr<float>(), bias.data_ptr<float>(), y.data_ptr<float>());
    return y;
}
"""

_ext = load_inline(
    name="virtual_gather_linear_ext_v1",

    functions=["vg_linear"],

    extra_cflags=["-O3"],
    verbose=False,
)
