import torch
import torch.nn as nn
import math


class Model(nn.Module):
    """
    Radial Basis Function (RBF) expansion of interatomic distances.
    Standard featurization in molecular graph neural networks (SchNet,
    DimeNet, GemNet) that converts scalar distances into a feature vector
    of Gaussian-basis evaluations.
    """

    def __init__(self, num_rbf=64, cutoff=5.0):
        super().__init__()
        self.num_rbf = num_rbf
        self.cutoff = cutoff
        offsets = torch.linspace(0.0, cutoff, num_rbf)
        self.register_buffer("offsets", offsets)
        coeff = -0.5 / (offsets[1] - offsets[0]).item() ** 2
        self.coeff = coeff

    def forward(self, distances: torch.Tensor) -> torch.Tensor:
        """
        Args:
            distances: (B, N, N) – pairwise distance matrix (or (B, E) edge list)
        Returns:
            rbf_features: (*distances.shape, num_rbf) – expanded features
        """
        # Cosine cutoff envelope
        d_scaled = distances.unsqueeze(-1)  # (..., 1)
        envelope = 0.5 * (torch.cos(math.pi * d_scaled / self.cutoff) + 1.0)
        envelope = envelope * (d_scaled < self.cutoff).float()

        rbf = torch.exp(self.coeff * (d_scaled - self.offsets) ** 2)
        return rbf * envelope

#include <cuda.h>

#ifndef M_PI_F
#define M_PI_F 3.14159265358979323846f
#endif

__global__ __launch_bounds__(256, 2)
void rbf64_kernel(const float* __restrict__ distances,
                  float* __restrict__ out,
                  int64_t m,
                  float cutoff) {
    // Four distance scalars per CTA, 64 threads/features per scalar.
    __shared__ float sd[4];
    __shared__ float se[4];

    const int tid = threadIdx.x;
    const int group = tid >> 6;       // 0..3
    const int k = tid & 63;           // rbf index
    const int64_t idx = (int64_t)blockIdx.x * 4 + group;

    if (k == 0) {
        float d = (idx < m) ? distances[idx] : 0.0f;
        sd[group] = d;
        float env = 0.0f;
        if (d < cutoff) {
            env = 0.5f * (cosf(M_PI_F * d / cutoff) + 1.0f);
        }
        se[group] = env;
    }
    __syncthreads();

    if (idx < m) {
        const float env = se[group];
        float val = 0.0f;
        if (env != 0.0f) {
            const float d = sd[group];
            const float step = cutoff * (1.0f / 63.0f);
            // coeff = -0.5 / step**2 for evenly spaced centers
            const float coeff = -0.5f / (step * step);
            const float x = d - step * (float)k;
            val = expf(coeff * x * x) * env;
        }
        out[idx * 64 + k] = val;
    }
}

__global__ __launch_bounds__(256, 2)
void rbf_generic_kernel(const float* __restrict__ distances,
                        float* __restrict__ out,
                        int64_t m,
                        int num_rbf,
                        float cutoff) {
    const int64_t total = m * (int64_t)num_rbf;
    const float step = (num_rbf > 1) ? cutoff / (float)(num_rbf - 1) : cutoff;
    const float coeff = -0.5f / (step * step);
    for (int64_t lin = (int64_t)blockIdx.x * blockDim.x + threadIdx.x;
         lin < total;
         lin += (int64_t)blockDim.x * gridDim.x) {
        int k = (int)(lin % num_rbf);
        int64_t idx = lin / num_rbf;
        float d = distances[idx];
        float val = 0.0f;
        if (d < cutoff) {
            float env = 0.5f * (cosf(M_PI_F * d / cutoff) + 1.0f);
            float x = d - step * (float)k;
            val = expf(coeff * x * x) * env;
        }
        out[lin] = val;
    }
}

torch::Tensor rbf_forward_cuda(torch::Tensor distances, int64_t num_rbf, double cutoff_d) {
    std::vector<int64_t> sizes = distances.sizes().vec();
    sizes.push_back(num_rbf);
    auto out = torch::empty(sizes, distances.options());

    const int64_t m = distances.numel();
    if (m == 0) return out;

    const float cutoff = static_cast<float>(cutoff_d);
    const float* dptr = distances.data_ptr<float>();
    float* optr = out.data_ptr<float>();

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

}
