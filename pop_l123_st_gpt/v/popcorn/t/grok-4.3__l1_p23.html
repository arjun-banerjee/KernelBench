import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void mean_dim0_kernel(const float* __restrict__ input, float* __restrict__ output, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx < total) {
        int b = idx / D;
        int d = idx % D;
        float sum = 0.0f;
        for (int r = 0; r < R; r++) {
            sum += input[(r * B + b) * D + d];
        }
        output[b * D + d] = sum / R;
    }
}

torch::Tensor mean_dim0_cuda(torch::Tensor x) {
    auto R = x.size(0);
    auto B = x.size(1);
    auto D = x.size(2);
    auto output = torch::empty({B, D}, x.options());
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    mean_dim0_kernel<<<blocks, threads>>>(x.data_ptr<float>(), output.data_ptr<float>(), R, B, D);
    return output;
}
