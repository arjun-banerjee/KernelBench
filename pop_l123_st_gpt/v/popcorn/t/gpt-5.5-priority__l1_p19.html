import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues
        Returns:
            dist: (B, N, N) – pairwise Euclidean distance matrix
        """
        diff = coords.unsqueeze(2) - coords.unsqueeze(1)  # (B, N, N, 3)
        dist = torch.sqrt((diff ** 2).sum(dim=-1) + 1e-8)
        return dist

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor pairwise_distance_forward(torch::Tensor coords);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define TILE 16

__global__ void pairwise_dist_sym_kernel(const float* __restrict__ coords,
                                         float* __restrict__ out,
                                         int B, int N) {
    const int tile_col = blockIdx.x;
    const int tile_row = blockIdx.y;
    const int b = blockIdx.z;

    // Only compute upper-triangular tiles; off-diagonal tiles also write the
    // transposed lower-triangular output. This halves the expensive sqrt work.
    if (tile_row > tile_col) return;

    const int tx = threadIdx.x;  // column within tile
    const int ty = threadIdx.y;  // row within tile
    const int row_base = tile_row * TILE;
    const int col_base = tile_col * TILE;

    __shared__ float r[TILE][3];
    __shared__ float c[TILE][3];

    const float* batch_coords = coords + ((long long)b * N * 3);

    if (ty == 0) {
        int rr = row_base + tx;
        if (rr < N) {
            const float* p = batch_coords + rr * 3;
            r[tx][0] = p[0];
            r[tx][1] = p[1];
            r[tx][2] = p[2];
        }
    }
    if (tx == 0) {
        int cc = col_base + ty;
        if (cc < N) {
            const float* p = batch_coords + cc * 3;
            c[ty][0] = p[0];
            c[ty][1] = p[1];
            c[ty][2] = p[2];
        }
    }
    __syncthreads();

    const int i = row_base + ty;
    const int j = col_base + tx;
    if (i < N && j < N) {
        float dx = r[ty][0] - c[tx][0];
        float dy = r[ty][1] - c[tx][1];
        float dz = r[ty][2] - c[tx][2];
        float d2 = dx * dx + dy * dy + dz * dz + 1.0e-8f;
        float d = sqrtf(d2);

        float* batch_out = out + ((long long)b * N * N);
        batch_out[(long long)i * N + j] = d;
        if (tile_row != tile_col) {
            batch_out[(long long)j * N + i] = d;
        }
    }
}

torch::Tensor pairwise_distance_forward(torch::Tensor coords) {
    TORCH_CHECK(coords.is_cuda(), "coords must be a CUDA tensor");
    TORCH_CHECK(coords.scalar_type() == at::kFloat, "coords must be float32");
    TORCH_CHECK(coords.dim() == 3 && coords.size(2) == 3, "coords must have shape (B, N, 3)");

    const int B = (int)coords.size(0);
    const int N = (int)coords.size(1);
    auto out = torch::empty({B, N, N}, coords.options());

    if (B == 0 || N == 0) return out;

    dim3 block(TILE, TILE, 1);
    dim3 grid((N + TILE - 1) / TILE, (N + TILE - 1) / TILE, B);
    pairwise_dist_sym_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>(
        coords.data_ptr<float>(), out.data_ptr<float>(), B, N);
    return out;
}
"""
