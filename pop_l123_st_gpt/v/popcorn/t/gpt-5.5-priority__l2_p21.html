import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Triangular multiplicative update – incoming edges.
    Mirror of the outgoing variant: updates pair (i,j) by aggregating over
    the shared index k via edges (k,i) and (k,j).
    """

    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            pair_repr: (B, N, N, pair_dim)
            pair_mask: (B, N, N)
        Returns:
            updated pair representation (B, N, N, pair_dim)
        """
        mask = pair_mask.unsqueeze(-1)
        x = self.layer_norm_in(pair_repr)

        a = self.proj_a(x) * torch.sigmoid(self.gate_a(x)) * mask  # (B, N, N, H)
        b = self.proj_b(x) * torch.sigmoid(self.gate_b(x)) * mask

        # Incoming: aggregate over k with a[k,i] * b[k,j]
        out = torch.einsum("bkih,bkjh->bijh", a, b)

        out = self.layer_norm_out(out)
        out = self.out_proj(out)
        out = torch.sigmoid(self.out_gate(pair_repr)) * out
        return out

    for (int c = tid; c < C; c += blockDim.x) xorig[c] = xrow[c];
    __syncthreads();

    for (int c = tid; c < C; c += blockDim.x) {
        float v = op_b[c];
        const float* ow = op_w + c * H;
        #pragma unroll
        for (int h = 0; h < 32; ++h) {
            if (h < H) v += ow[h] * yn[h];
        }

        float g = og_b[c];
        const float* gw = og_w + c * C;
        #pragma unroll
        for (int d = 0; d < 64; ++d) {
            if (d < C) g += gw[d] * xorig[d];
        }
        out[((long long)row) * C + c] = sigmoid_f(g) * v;
    }
}

torch::Tensor tri_forward(
    torch::Tensor pair_repr,
    torch::Tensor pair_mask,

    torch::Tensor ln_in_b,

    torch::Tensor ln_out_b,
    torch::Tensor out_proj_w,
    torch::Tensor out_proj_b,
    torch::Tensor out_gate_w,
    torch::Tensor out_gate_b)
{
    int Bsz = (int)pair_repr.size(0);
    int N = (int)pair_repr.size(1);
    int C = (int)pair_repr.size(3);
    int H = (int)proj_a_w.size(0);

    auto opts = pair_repr.options();
    auto A = torch::empty({Bsz, N, N, H}, opts);
    auto Bv = torch::empty({Bsz, N, N, H}, opts);
    auto M = torch::empty({Bsz, N, N, H}, opts);
    auto out = torch::empty_like(pair_repr);

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    int rows = Bsz * N * N;
    dim3 block1(128);
    make_ab_kernel<<<rows, block1, 0, stream>>>(
        pair_repr.data_ptr<float>(), pair_mask.data_ptr<float>(),
        ln_in_w.data_ptr<float>(), ln_in_b.data_ptr<float>(),

        gate_a_w.data_ptr<float>(), gate_a_b.data_ptr<float>(),
        proj_b_w.data_ptr<float>(), proj_b_b.data_ptr<float>(),
        gate_b_w.data_ptr<float>(), gate_b_b.data_ptr<float>(),
        A.data_ptr<float>(), Bv.data_ptr<float>(), Bsz, N, C, H);

    long long total = (long long)Bsz * N * N * H;
    dim3 block2(256);
    dim3 grid2((unsigned int)((total + block2.x - 1) / block2.x));
    incoming_einsum_kernel<<<grid2, block2, 0, stream>>>(
        A.data_ptr<float>(), Bv.data_ptr<float>(), M.data_ptr<float>(), Bsz, N, H);

    finish_kernel<<<rows, block1, 0, stream>>>(

        out.data_ptr<float>(), Bsz, N, C, H);

    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
