#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void fused_gather_linear_kernel(
    const float* __restrict__ input,    // (R, B, H)
    const float* __restrict__ weight,   // (K, R*H)
    const float* __restrict__ bias,     // (K)
    float* __restrict__ output,         // (B, K)
    int R, int B, int H, int K, int RH
) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    int k = blockIdx.y * blockDim.y + threadIdx.y;
    
    if (b >= B || k >= K) return;
    
    float sum = bias[k];
    const float* w_row = weight + k * RH;

        for (int h = 0; h < H; ++h) {
            sum += in_slice[h] * w_slice[h];
        }
    }
    
    output[b * K + k] = sum;
}
