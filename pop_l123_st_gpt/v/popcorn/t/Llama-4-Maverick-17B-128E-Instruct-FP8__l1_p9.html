import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a parallel associative scan (linear recurrence) over a 1D sequence.
    Given coefficient a and input b at each timestep, computes:
        h_0 = b_0
        h_t = a_t * h_{t-1} + b_t  for t > 0
    This is the core parallelizable primitive underlying Mamba's selective scan and
    other SSM architectures. Sequential in naive PyTorch but can be parallelized
    via the associative scan algorithm in O(L log L) span.

    Args:
        d_model (int): Number of independent recurrence channels.
    """
    def __init__(self, d_model: int):
        super(Model, self).__init__()
        self.d_model = d_model

    def forward(self, a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
        B, L, D = a.shape
        h = torch.zeros(B, D, device=a.device, dtype=a.dtype)
        outputs = []
        for t in range(L):
            h = a[:, t] * h + b[:, t]
            outputs.append(h)
        return torch.stack(outputs, dim=1)

seq_len = 4096

def get_inputs():

    b = torch.randn(batch_size, seq_len, d_model)
    return [a, b]

def get_init_inputs():
    return [d_model]

import torch
import torch.nn as nn
import torch.utils.cpp_extension as cpp_extension

# CUDA kernel for parallel associative scan
cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__device__ float sigmoid(float x) {
    return 1.0f / (1.0f + expf(-x));
}

__device__ float2 binary_operator(float2 a, float2 b) {
    return make_float2(a.x * b.x, a.y * b.x + b.y);
}

__global__ void associative_scan_forward(float *a, float *b, float *output, int B, int L, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int batch_idx = idx / D;
    int d_idx = idx % D;

    if (batch_idx >= B) return;

    float2 state = make_float2(1.0f, b[batch_idx * L * D + d_idx]);
    for (int t = 1; t < L; t++) {
        int index = batch_idx * L * D + t * D + d_idx;
        float a_val = sigmoid(a[index]);
        float2 a_state = make_float2(a_val, 0.0f);
        state = binary_operator(a_state, state);
        output[index] = state.y;
    }
}

__global__ void set_first_timestep(float *output, float *b, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int batch_idx = idx / D;
    int d_idx = idx % D;

    if (batch_idx >= B) return;

    output[batch_idx * D + d_idx] = b[batch_idx * D + d_idx];
}
'''
