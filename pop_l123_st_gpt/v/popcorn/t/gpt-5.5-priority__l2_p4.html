import math

import torch
import torch.nn as nn
import torch.nn.functional as F

# Default ModelNew is PyTorch-only. run_and_check: use check_kernel=False with backend=cuda
# until the candidate file contains a real CUDA kernel (see kernel_static_checker).


class Model(nn.Module):
    """
    Graph-Fused VLA (GF-VLA) style **scene graph relation update** in one forward.

    Objects are nodes with vision-language aligned embeddings; directed pairs
    (i → j) carry pairwise conditioning (geometry, language grounding, or fused
    VLA features). This reference fuses:

      LayerNorm(nodes) → multi-head attention over objects with **pairwise
      relation MLP bias** on edges → residual write-back.

    Intended as a fusion target for navigation / manipulation stacks that maintain
    explicit scene graphs and update them from multimodal observations.
    """

    def __init__(self, dim: int, pair_dim: int, num_heads: int):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.scale = 1.0 / math.sqrt(self.head_dim)

        self.norm = nn.LayerNorm(dim)
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.rel_mlp = nn.Sequential(
            nn.Linear(pair_dim, pair_dim),
            nn.SiLU(),
            nn.Linear(pair_dim, num_heads),
        )
        self.out_proj = nn.Linear(dim, dim, bias=False)

    def forward(
        self,
        nodes: torch.Tensor,
        pair_feats: torch.Tensor,
        edge_mask: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            nodes: (B, N, D) object / region embeddings.
            pair_feats: (B, N, N, P) pairwise VLA or geometric conditioning.
            edge_mask: (B, N, N) float in {0, 1}; 1 keeps attention over column j.
        Returns:
            (B, N, D) updated node embeddings (residual around attention block).
        """
        B, N, D = nodes.shape
        _, N2, N3, P = pair_feats.shape
        assert N == N2 == N3 and P == self.pair_dim
        h = self.norm(nodes)

        q = self.q_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)

        logits = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        rel_bias = self.rel_mlp(pair_feats).permute(0, 3, 1, 2)
        logits = logits + rel_bias

        logits = logits.masked_fill(edge_mask.unsqueeze(1) < 0.5, float("-inf"))
        attn = F.softmax(logits, dim=-1)
        ctx = torch.matmul(attn, v).transpose(1, 2).reshape(B, N, D)
        return nodes + self.out_proj(ctx)

    if (tid == 0) {
        float m = logits[0];
        for (int j = 1; j < N; ++j) m = fmaxf(m, logits[j]);
        float sum = 0.0f;
        for (int j = 0; j < N; ++j) {
            float a = expf(logits[j] - m);
            att[j] = a;
            sum += a;
        }
        float invsum = 1.0f / sum;
        for (int j = 0; j < N; ++j) att[j] *= invsum;
    }
    __syncthreads();

    if (tid < Hd) {
        float acc = 0.0f;
        for (int j = 0; j < N; ++j) {
            acc += att[j] * v[((b * H + h) * N + j) * Hd + tid];
        }
        ctx[(b * N + i) * D + h * Hd + tid] = acc;
    }
}

__global__ void out_kernel(
    const float* __restrict__ ctx,
    const float* __restrict__ nodes,
    const float* __restrict__ out_w,
    float* __restrict__ out,
    int B, int N, int D)
{
    int row = blockIdx.x;
    int tid = threadIdx.x;
    const float* cp = ctx + row * D;
    const float* np = nodes + row * D;
    for (int od = tid; od < D; od += blockDim.x) {
        float sum = 0.0f;
        const float* ww = out_w + od * D;
        #pragma unroll
        for (int id = 0; id < 96; ++id) {
            if (id < D) sum += cp[id] * ww[id];
        }
        out[row * D + od] = np[od] + sum;
    }
}

torch::Tensor gf_forward(
    torch::Tensor nodes,
    torch::Tensor pair_feats,
    torch::Tensor edge_mask,
    torch::Tensor norm_w,
    torch::Tensor norm_b,
    torch::Tensor q_w,
    torch::Tensor k_w,
    torch::Tensor v_w,
    torch::Tensor rel_w1,
    torch::Tensor rel_b1,
    torch::Tensor rel_w2,
    torch::Tensor rel_b2,
    torch::Tensor out_w,
    int64_t num_heads)
{
    int B = (int)nodes.size(0);
    int N = (int)nodes.size(1);
    int D = (int)nodes.size(2);
    int P = (int)pair_feats.size(3);
    int H = (int)num_heads;
    int Hd = D / H;

    auto opts = nodes.options();
    torch::Tensor q = torch::empty({B, H, N, Hd}, opts);
    torch::Tensor k = torch::empty({B, H, N, Hd}, opts);
    torch::Tensor v = torch::empty({B, H, N, Hd}, opts);
    torch::Tensor rel = torch::empty({B, N, N, H}, opts);
    torch::Tensor ctx = torch::empty({B, N, D}, opts);
    torch::Tensor out = torch::empty_like(nodes);

    const int threads = 128;
    ln_qkv_kernel<<<B * N, threads>>>(
        nodes.data_ptr<float>(), norm_w.data_ptr<float>(), norm_b.data_ptr<float>(),
        q_w.data_ptr<float>(), k_w.data_ptr<float>(), v_w.data_ptr<float>(),
        q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(), B, N, D, H, Hd);

    int total_edges = B * N * N;
    rel_kernel<<<(total_edges + 255) / 256, 256>>>(
        pair_feats.data_ptr<float>(), rel_w1.data_ptr<float>(), rel_b1.data_ptr<float>(),
        rel_w2.data_ptr<float>(), rel_b2.data_ptr<float>(), rel.data_ptr<float>(), B, N, P, H);

    float scale = rsqrtf((float)Hd);
    attn_kernel<<<B * H * N, 64>>>(
        q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(), rel.data_ptr<float>(),
        edge_mask.data_ptr<float>(), ctx.data_ptr<float>(), B, N, D, H, Hd, scale);

    out_kernel<<<B * N, threads>>>(
        ctx.data_ptr<float>(), nodes.data_ptr<float>(), out_w.data_ptr<float>(), out.data_ptr<float>(), B, N, D);

    return out;
}
