import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a depthwise 1D convolution where each input channel is convolved with its own filter.
    Fundamental primitive in Hyena, Mamba, and genomic sequence models (Enformer, HyenaDNA).

    Args:
        channels (int): Number of input/output channels.
        kernel_size (int): Size of the convolution kernel.
        stride (int, optional): Stride of the convolution. Defaults to 1.
        padding (int, optional): Padding applied to the input. Defaults to 0.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """
    def __init__(self, channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(Model, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.conv1d(x)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

// Fast path for the benchmark case: fp32 depthwise Conv1d, K=3, stride=1, padding=1.
// One CUDA block computes one (batch, channel) row.  The sequence dimension is contiguous,
// so all memory transactions are coalesced.  We keep the three filter taps in registers.
__global__ void dwconv1d_k3s1p1_kernel(const float* __restrict__ x,
                                        const float* __restrict__ w,
                                        const float* __restrict__ b,
                                        float* __restrict__ y,
                                        int rows, int C, int L,
                                        bool has_bias) {
    int row = blockIdx.x;
    if (row >= rows) return;
    int c = row % C;
    const float* __restrict__ xrow = x + (long long)row * L;
    float* __restrict__ yrow = y + (long long)row * L;

    const float w0 = w[(long long)c * 3 + 0];
    const float w1 = w[(long long)c * 3 + 1];
    const float w2 = w[(long long)c * 3 + 2];
    const float bv = has_bias ? b[c] : 0.0f;

    // Left boundary (only x[0], x[1])
    if (threadIdx.x == 0) {
        yrow[0] = xrow[0] * w1 + xrow[1] * w2 + bv;
    }

    // Interior points.  The extra branch avoidance in the loop is materially faster than
    // checking boundaries for every output element.
    for (int i = threadIdx.x + 1; i < L - 1; i += blockDim.x) {
        float xm = xrow[i - 1];
        float xc = xrow[i];
        float xp = xrow[i + 1];
        yrow[i] = xm * w0 + xc * w1 + xp * w2 + bv;
    }

    // Right boundary (only x[L-2], x[L-1])
    if (threadIdx.x == 0) {
        yrow[L - 1] = xrow[L - 2] * w0 + xrow[L - 1] * w1 + bv;
    }
}

// Generic fp32 fallback for other constructor arguments.  It is still a direct depthwise
// implementation, but the fixed K=3 path above is what is used for the supplied workload.
__global__ void dwconv1d_generic_kernel(const float* __restrict__ x,
                                        const float* __restrict__ w,
                                        const float* __restrict__ b,
                                        float* __restrict__ y,
                                        int rows, int C, int L, int K,
                                        int Lout, int stride, int padding,
                                        bool has_bias) {
    int row = blockIdx.x;
    if (row >= rows) return;
    int c = row % C;
    const float* __restrict__ xrow = x + (long long)row * L;
    float* __restrict__ yrow = y + (long long)row * Lout;
    const float* __restrict__ wc = w + (long long)c * K;
    float bv = has_bias ? b[c] : 0.0f;

    for (int out = threadIdx.x; out < Lout; out += blockDim.x) {
        int base = out * stride - padding;
        float acc = bv;
        #pragma unroll
        for (int k = 0; k < 16; ++k) {
            if (k < K) {
                int ix = base + k;
                if ((unsigned)ix < (unsigned)L) acc += xrow[ix] * wc[k];
            }
        }
        if (K > 16) {
            for (int k = 16; k < K; ++k) {
                int ix = base + k;
                if ((unsigned)ix < (unsigned)L) acc += xrow[ix] * wc[k];
            }
        }
        yrow[out] = acc;
    }
}

    TORCH_CHECK(x.is_cuda(), "x must be CUDA");
    TORCH_CHECK(weight.is_cuda(), "weight must be CUDA");
    TORCH_CHECK(x.scalar_type() == torch::kFloat32, "only float32 supported");
    TORCH_CHECK(weight.scalar_type() == torch::kFloat32, "only float32 supported");

    const int C = (int)x.size(1);
    const int L = (int)x.size(2);
