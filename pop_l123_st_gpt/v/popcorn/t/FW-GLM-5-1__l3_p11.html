"""
Three-stage MLP on packed activations: h1 = relu(h0 @ W1), h2 = relu(h1 @ W2), out = h2 @ W3.
Optimized with fused CUDA kernels for matmul + ReLU.
"""

B = 16
H0 = 48
H1 = 64
H2 = 40
H3 = 24

#include <cuda.h>
#include <cuda_runtime.h>
#include <cstdio>

// Fused matrix multiplication with ReLU activation
// Computes: output[i,j] = max(0, sum_k(input[i,k] * weight[k,j]))
// weight is stored transposed: weight[j,k] where j is output dim, k is input dim
template<int TILE_SIZE>
__global__ void matmul_relu_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    float* __restrict__ output,
    int batch_size, int in_dim, int out_dim)
{
    int i = blockIdx.x;
    int j = blockIdx.y * TILE_SIZE + threadIdx.x;
    
    if (i >= batch_size) return;
    
    float sum = 0.0f;
    
    for (int k_tile = 0; k_tile < in_dim; k_tile += TILE_SIZE) {
        __shared__ float weight_tile[TILE_SIZE * TILE_SIZE];
        __shared__ float input_tile[TILE_SIZE];
        
        // Load weight tile (transposed access for coalescing)
        int k_idx = k_tile + threadIdx.x;
        int j_idx = blockIdx.y * TILE_SIZE + threadIdx.y;
        if (k_idx < in_dim && j_idx < out_dim) {
            weight_tile[threadIdx.y * TILE_SIZE + threadIdx.x] = weight[j_idx * in_dim + k_idx];
        }
        
        // Load input tile
        if (threadIdx.y == 0 && k_idx < in_dim) {
            input_tile[threadIdx.x] = input[i * in_dim + k_idx];
        }
        
        __syncthreads();
        
        // Compute partial dot product
        int k_limit = min(TILE_SIZE, in_dim - k_tile);
        for (int k = 0; k < k_limit; k++) {
            int j_local = threadIdx.x;
            if (j < out_dim) {
                sum += input_tile[k] * weight_tile[j_local * TILE_SIZE + k];
            }
        }
        
        __syncthreads();
    }
    
    // Apply ReLU and store result
    if (j < out_dim) {
        output[i * out_dim + j] = fmaxf(0.0f, sum);
    }
}

// Matrix multiplication without activation (for final layer)
template<int TILE_SIZE>
__global__ void matmul_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    float* __restrict__ output,
    int batch_size, int in_dim, int out_dim)
{
    int i = blockIdx.x;
    int j = blockIdx.y * TILE_SIZE + threadIdx.x;
    
    if (i >= batch_size) return;
    
    float sum = 0.0f;
    
    for (int k_tile = 0; k_tile < in_dim; k_tile += TILE_SIZE) {
        __shared__ float weight_tile[TILE_SIZE * TILE_SIZE];
        __shared__ float input_tile[TILE_SIZE];
        
        int k_idx = k_tile + threadIdx.x;
        int j_idx = blockIdx.y * TILE_SIZE + threadIdx.y;
        if (k_idx < in_dim && j_idx < out_dim) {
            weight_tile[threadIdx.y * TILE_SIZE + threadIdx.x] = weight[j_idx * in_dim + k_idx];
        }
        
        if (threadIdx.y == 0 && k_idx < in_dim) {
            input_tile[threadIdx.x] = input[i * in_dim + k_idx];
        }
        
        __syncthreads();
        
        int k_limit = min(SHARED_MEM_SIZE, in_dim - k_tile);
        for (int k = 0; k < k_limit; k++) {
            int j_local = threadIdx.x;
            if (j < out_dim) {
                sum += input_tile[k] * weight_tile[j_local * TILE_SIZE + k];
            }
        }
        
        __syncthreads();
    }
    
    if (j < out_dim) {
        output[i * out_dim + j] = sum;
    }
}

// Simple non-tiled fused matmul + ReLU kernel for small matrices
__global__ void fused_matmul_relu_simple(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    float* __restrict__ output,
    int batch_size, int in_dim, int out_dim)
{
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    int j = blockIdx.y * blockDim.y + threadIdx.y;
    
    if (i >= batch_size || j >= out_dim) return;
    
    float sum = 0.0f;
    for (int k = 0; k < in_dim; k++) {
        sum += input[i * in_dim + k] * weight[j * in_dim + k];
    }
    output[i * out_dim + j] = fmaxf(0.0f, sum);
}

// Simple non-tiled matmul kernel for small matrices
__global__ void matmul_simple(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    float* __restrict__ output,
    int batch_size, int in_dim, int out_dim)
{

        sum += input[i * in_dim + k] * weight[j * in_dim + k];
    }

}

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.W1 = nn.Linear(H0, H1, bias=False)
        self.W2 = nn.Linear(H1, H2, bias=False)
        self.W3 = nn.Linear(H2, H3, bias=False)
    
    def forward(self, h0: torch.Tensor) -> torch.Tensor:
        # Get weight matrices (transposed for efficient access)
        W1 = self.W1.weight  # [H1, H0]
        W2 = self.W2.weight  # [H2, H1]
        W3 = self.W3.weight  # [H3, H2]
        
        # Ensure contiguous float32
        h0_f32 = h0.contiguous().float()
        W1_f32 = W1.contiguous().float()
        W2_f32 = W2.contiguous().float()
        W3_f32 = W3.contiguous().float()
        
        # Allocate output tensors
        h1 = torch.empty(B, H1, dtype=torch.float32, device=h0.device)
        h2 = torch.empty(B, H2, dtype=torch.float32, device=h0.device)
        out = torch.empty(B, H3, dtype=torch.float32, device=h0.device)
        
        # Stage 1: h1 = relu(h0 @ W1^T)
        mlp_cuda.fused_matmul_relu(
            h0_f32.data_ptr(), W1_f32.data_ptr(), h1.data_ptr(),
            B, H0, H1, torch.cuda.current_stream().cuda_stream
        )
        
        # Stage 2: h2 = relu(h1 @ W2^T)
        mlp_cuda.fused_matmul_relu(
            h1.data_ptr(), W2_f32.data_ptr(), h2.data_ptr(),
            B, H1, H2, torch.cuda.current_stream().cuda_stream
        )
        
        # Stage 3: out = h2 @ W3^T (no activation)
        mlp_cuda.matmul_final(
            h2.data_ptr(), W3_f32.data_ptr(), out.data_ptr(),
            B, H2, H3, torch.cuda.current_stream().cuda_stream
        )
        
        return out
