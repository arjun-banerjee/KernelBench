class Model(nn.Module):
    """
    Performs a depthwise 1D convolution where each input channel is convolved with its own filter.
    Fundamental primitive in Hyena, Mamba, and genomic sequence models (Enformer, HyenaDNA).

    Args:
        channels (int): Number of input/output channels.
        kernel_size (int): Size of the convolution kernel.
        stride (int, optional): Stride of the convolution. Defaults to 1.
        padding (int, optional): Padding applied to the input. Defaults to 0.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """
    def __init__(self, channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(Model, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.conv1d(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void depthwise_conv1d_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size,
    int channels,
    int in_seq_len,
    int out_seq_len,
    int kernel_size,
    int stride,
    int padding,
    bool has_bias
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * channels * out_seq_len;
    if (idx >= total) return;

    int o = idx % out_seq_len;
    int c = (idx / out_seq_len) % channels;
    int b = idx / (out_seq_len * channels);

    float sum = 0.0f;
    for (int k = 0; k < kernel_size; ++k) {
        int in_pos = o * stride + k - padding;
        if (in_pos >= 0 && in_pos < in_seq_len) {
            sum += input[b * channels * in_seq_len + c * in_seq_len + in_pos] * weight[c * kernel_size + k];
        }
    }
    if (has_bias) {
        sum += bias[c];
    }
    output[b * channels * out_seq_len + c * out_seq_len + o] = sum;
}

torch::Tensor depthwise_conv1d_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias,
    int stride,
    int padding
) {
    int batch_size = input.size(0);
    int channels = input.size(1);
    int in_seq_len = input.size(2);
    int kernel_size = weight.size(2);
    bool has_bias = bias.defined() && bias.numel() > 0;

    int out_seq_len = (in_seq_len + 2 * padding - kernel_size) / stride + 1;

    auto output = torch::empty({batch_size, channels, out_seq_len}, input.options());
