import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Dilated convolution tower with exponentially increasing dilation rates and residual connections.
    Each block applies LayerNorm -> GELU -> DilatedConv1d, then adds back the original input as a residual.
    This architecture is a core building block of genomic foundation models (Enformer, Basenji, Caduceus)
    for capturing multi-scale sequence dependencies in DNA/RNA.
    """
    def __init__(self, channels, seq_len, kernel_size, max_dilation_power):
        super(Model, self).__init__()
        self.norms = nn.ModuleList()
        self.convs = nn.ModuleList()
        self.activation = nn.GELU()
        for i in range(max_dilation_power + 1):
            dilation = 2 ** i
            self.norms.append(nn.LayerNorm([channels, seq_len]))
            self.convs.append(nn.Conv1d(channels, channels, kernel_size, padding=dilation, dilation=dilation))

    def forward(self, x):
        residual = x
        for norm, conv in zip(self.norms, self.convs):
            x = norm(x)
            x = self.activation(x)
            x = conv(x)
            x = x + residual
        return x

// Fused LayerNorm + GELU kernel
// Normalizes over the entire (channels, seq_len) for each batch sample
template <typename scalar_t>
__global__ void layernorm_gelu_kernel(
    const scalar_t* __restrict__ input,
    scalar_t* __restrict__ output,
    const int channels,
    const int seq_len) {
    
    const int batch_idx = blockIdx.x;
    const int total_elements = channels * seq_len;
    const int tid = threadIdx.x;
    const int block_size = blockDim.x;
    
    // Shared memory for reduction
    extern __shared__ char smem[];
    scalar_t* s_sum = reinterpret_cast<scalar_t*>(smem);
    scalar_t* s_sum_sq = s_sum + block_size;
    
    // Compute mean
    scalar_t local_sum = scalar_t(0.0);
    for (int i = tid; i < total_elements; i += block_size) {
        int idx = batch_idx * total_elements + i;
        local_sum += input[idx];
    }
    s_sum[tid] = local_sum;
    __syncthreads();
    
    for (int stride = block_size >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_sum[tid] += s_sum[tid + stride];
        }
        __syncthreads();
    }
    
    const scalar_t mean = s_sum[0] / total_elements;
    
    // Compute variance
    scalar_t local_sum_sq = scalar_t(0.0);
    for (int i = tid; i < total_elements; i += block_size) {
        int idx = batch_idx * total_elements + i;
        scalar_t diff = input[idx] - mean;
        local_sum_sq += diff * diff;
    }
    s_sum_sq[tid] = local_sum_sq;
    __syncthreads();
    
    for (int stride = block_size >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_sum_sq[tid] += s_sum_sq[tid + stride];
        }
        __syncthreads();
    }
    
    const scalar_t var = s_sum_sq[0] / total_elements;
    const scalar_t inv_std = rsqrt(var + scalar_t(1e-5));
    
    // Normalize and apply GELU
    for (int i = tid; i < total_elements; i += block_size) {
        int idx = batch_idx * total_elements + i;
        scalar_t val = (input[idx] - mean) * inv_std;
        // GELU approximation: 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
        const scalar_t sqrt_2_over_pi = scalar_t(0.7978845608028654);
        const scalar_t coeff = scalar_t(0.044715);
        scalar_t gelu = scalar_t(0.5) * val * (scalar_t(1.0) + tanh(sqrt_2_over_pi * (val + coeff * val * val * val)));
        output[idx] = gelu;
    }
}

// Optimized dilated Conv1d + Residual Add kernel
// Each thread handles one output element (batch, out_channel, seq_pos)

    const scalar_t* __restrict__ input,

    scalar_t* __restrict__ output,
    const int channels,
    const int seq_len,
    const int kernel_size,
    const int dilation,
    const int padding) {
    
    const int batch_idx = blockIdx.x;
    const int out_c = blockIdx.y;
    const int out_s = blockIdx.z * blockDim.x + threadIdx.x;
    
    if (out_s >= seq_len) return;
    
    const int total_elements = channels * seq_len;
    scalar_t sum = scalar_t(0.0);
    
    // Conv1d: each output channel connects to all input channels
    // Weight shape: [out_channels, in_channels, kernel_size]
    for (int in_c = 0; in_c < channels; in_c++) {
        for (int k = 0; k < kernel_size; k++) {
            const int in_s = out_s - padding + k * dilation;
            if (in_s >= 0 && in_s < seq_len) {
                const int in_idx = batch_idx * total_elements + in_c * seq_len + in_s;
                const int w_idx = out_c * channels * kernel_size + in_c * kernel_size + k;
                sum += input[in_idx] * weight[w_idx];
            }
        }
    }
    
    const int out_idx = batch_idx * total_elements + out_c * seq_len + out_s;
    output[out_idx] = sum + residual[out_idx];
}

// Multi-block fused operation - applies all blocks in sequence
void forward_cuda(
    torch::Tensor x,
    std::vector<torch::Tensor> weights,

    const int channels = x.size(1);
    const int seq_len = x.size(2);
    
    // Allocate temporary buffer
    auto temp = torch::empty_like(x);
    torch::Tensor* current = &x;
    torch::Tensor* next = &temp;
    
    const int threads = 256;
    const int shared_mem = 2 * threads * sizeof(float);
    
    for (int block_idx = 0; block_idx < num_blocks; block_idx++) {
        int dilation = 1 << block_idx;
        int kernel_size = weights[block_idx].size(2);
        int padding = dilation;
        
        // Save residual
        auto residual = *current;
        
        // LayerNorm + GELU
        AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "layernorm_gelu_cuda", ([&] {
            layernorm_gelu_kernel<scalar_t><<<batch_size, threads, shared_mem>>>(
                current->data_ptr<scalar_t>(),
                next->data_ptr<scalar_t>(),
                channels,
                seq_len);
        }));
        
        // Dilated Conv1d + Residual
