import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Fixed permutation
B = 10
N = 36
_g = torch.Generator()
_g.manual_seed(10)
PERM = torch.randperm(N, generator=_g)

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void permute_kernel(const float* __restrict__ x, float* __restrict__ y, const int* __restrict__ perm, int B, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < B * N) {
        int b = idx / N;
        int j = idx % N;
        int p = perm[j];
        y[b * N + j] = x[b * N + p];
    }
}

torch::Tensor permute_forward(torch::Tensor x, torch::Tensor perm) {
    auto B = x.size(0);
    auto N = x.size(1);
    auto y = torch::empty({B, N}, x.options());
    const int threads = 256;
    const int blocks = (B * N + threads - 1) / threads;
    permute_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(),
        y.data_ptr<float>(),
        perm.data_ptr<int>(),
        B,
        N
    );
    return y;
}
'''

cpp_source = r'''
torch::Tensor permute_forward(torch::Tensor x, torch::Tensor perm);
'''

# Compile the extension
permute_ext = load_inline(
    name='permute_ext',
    cpp_sources=[cpp_source],
    cuda_sources=[cuda_source],
    functions=['permute_forward'],
    verbose=False
)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.perm = PERM.cuda()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return permute_ext.permute_forward(x, self.perm)
