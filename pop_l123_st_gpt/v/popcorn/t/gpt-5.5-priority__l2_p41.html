import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    """
    Evidence Lower Bound (ELBO) computation for a Variational Autoencoder.
    Combines a reconstruction term (binary cross-entropy) with the
    KL divergence between the approximate posterior q(z|x) = N(mu, sigma^2)
    and the prior p(z) = N(0, I).  The core loss function in VAE training.
    """

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
        )
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)

        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
            nn.Sigmoid(),
        )

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim) – input data in [0, 1]
        Returns:
            (elbo, recon_loss, kl_div):
                elbo:       (B,)
                recon_loss: (B,)
                kl_div:     (B,)
        """
        h = self.encoder(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)

        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)

        x_recon = self.decoder(z)
        recon = F.binary_cross_entropy(x_recon, x, reduction="none").sum(-1)
        kl = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(-1)
        elbo = -(recon + kl)
        return elbo, recon, kl

def get_init_inputs():

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> vae_loss_forward(torch::Tensor logits,
                                            torch::Tensor x,
                                            torch::Tensor mu,
                                            torch::Tensor logvar);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

__global__ __launch_bounds__(256) void vae_loss_kernel(
    const float* __restrict__ logits,
    const float* __restrict__ x,
    const float* __restrict__ mu,
    const float* __restrict__ logvar,
    float* __restrict__ elbo,
    float* __restrict__ recon,
    float* __restrict__ kl,
    int D,
    int L) {

    const int b = blockIdx.x;
    const int tid = threadIdx.x;

    float rsum = 0.0f;
    const int baseD = b * D;
    for (int i = tid; i < D; i += blockDim.x) {
        float a = logits[baseD + i];
        float t = x[baseD + i];
        // Binary cross entropy of sigmoid(a), computed from logits to avoid
        // materializing the sigmoid and the elementwise BCE tensor.
        float aa = fabsf(a);
        float m = a > 0.0f ? a : 0.0f;
        rsum += m - a * t + log1pf(expf(-aa));
    }

    float ksum = 0.0f;
    const int baseL = b * L;
    for (int i = tid; i < L; i += blockDim.x) {
        float m = mu[baseL + i];
        float lv = logvar[baseL + i];
        ksum += -0.5f * (1.0f + lv - m * m - expf(lv));
    }

    __shared__ float sr[256];
    __shared__ float sk[256];
    sr[tid] = rsum;
    sk[tid] = ksum;
    __syncthreads();

    for (int stride = 128; stride > 0; stride >>= 1) {
        if (tid < stride) {
            sr[tid] += sr[tid + stride];
            sk[tid] += sk[tid + stride];
        }
        __syncthreads();
    }

    if (tid == 0) {
        float r = sr[0];
        float k = sk[0];
        recon[b] = r;
        kl[b] = k;
        elbo[b] = -(r + k);
    }
}

std::vector<torch::Tensor> vae_loss_forward(torch::Tensor logits,
                                            torch::Tensor x,
                                            torch::Tensor mu,
                                            torch::Tensor logvar) {
    const int B = (int)logits.size(0);
    const int D = (int)logits.size(1);
    const int L = (int)mu.size(1);

    auto opts = logits.options();
    torch::Tensor elbo = torch::empty({B}, opts);
    torch::Tensor recon = torch::empty({B}, opts);
    torch::Tensor kl = torch::empty({B}, opts);

    const dim3 block(256);
    const dim3 grid(B);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    vae_loss_kernel<<<grid, block, 0, stream>>>(
        logits.data_ptr<float>(), x.data_ptr<float>(), mu.data_ptr<float>(), logvar.data_ptr<float>(),
        elbo.data_ptr<float>(), recon.data_ptr<float>(), kl.data_ptr<float>(), D, L);

    return {elbo, recon, kl};
}
"""

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
        )
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
            nn.Sigmoid(),
        )

    def forward(self, x: torch.Tensor) -> tuple:
        # Keep the same stochastic operation as the reference so RNG behaviour
        # remains aligned; fuse only the terminal sigmoid/BCE/KL reductions.
        h = torch.relu(self.encoder[0](x))

        logvar = self.fc_logvar(h)

        logits = self.decoder[2](dh)
