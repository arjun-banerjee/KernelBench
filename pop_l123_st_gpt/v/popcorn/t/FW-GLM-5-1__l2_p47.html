"""
Virtual all-gather on the last dim: stack R tensors of shape (B, H) into (B, R*H), then Linear.
"""

import torch
import torch.nn as nn

from kernelbench.distributed_collectives import default_device

R = 4
B = 12
H = 32
K = 56


class Model(nn.Module):
    def __init__(self):
        super().__init__()
        self.proj = nn.Linear(R * H, K, bias=True)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        r, b, h = x.shape
        assert r == R and h == H
        parts = [x[i] for i in range(R)]
        gathered = torch.cat(parts, dim=-1)
        return self.proj(gathered)

    dev = default_device()
    g = torch.Generator(device=dev)

    return [torch.randn(R, B, H, device=dev, generator=g)]

    return []

    // Block maps to one B element, threads cover K dimension
    int b_idx = blockIdx.y;
    int k_idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (b_idx >= B || k_idx >= K) return;
    
    float sum = 0.0f;
    
    // Iterate over r and h
    for (int r = 0; r < R; r++) {
        // Load input[r, b, :] into shared memory for cache efficiency
        for (int h = threadIdx.x; h < H; h += blockDim.x) {
            // Prefetch hint for input
        }
        __syncthreads();
        
        for (int h = 0; h < H; h++) {
            // input[r, b, h]
            float in_val = input[r * B * H + b_idx * H + h];
            // weight[k, r*H + h]
            float w_val = weight[k_idx * R * H + r * H + h];
            sum += in_val * w_val;
        }
    }
    
    // Add bias
    sum += bias[k_idx];
    
    output[b_idx * K + k_idx] = sum;
}

// Optimized version with weight caching via shared memory
__global__ void fused_allgather_linear_cached_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int B, int K, int R, int H) {
    
    int b_idx = blockIdx.y;
    int k_idx = blockIdx.x * blockDim.x + threadIdx.x;
    int t_idx = threadIdx.x;
    
    if (b_idx >= B || k_idx >= K) return;
    
    // Shared memory for bias (small, K elements max)
    extern __shared__ float shared_mem[];
    float* s_bias = shared_mem;
    
    // Cooperatively load bias into shared memory
    for (int i = t_idx; i < K; i += blockDim.x) {
        s_bias[i] = bias[i];
    }
    __syncthreads();
    
    float sum = 0.0f;
    
    // Process each r and h
    // Input access pattern: for fixed (r, b), vary h -> coalesced
    // Weight access pattern: for fixed k, vary (r, h) -> coalesced within thread
    
    for (int r = 0; r < R; r++) {
        for (int h = 0; h < H; h++) {
            float in_val = input[r * B * H + b_idx * H + h];
            float w_val = weight[k_idx * R * H + r * H + h];
            sum += in_val * w_val;
        }
    }
    
    // Add bias from shared memory
    sum += s_bias[k_idx];
    
    output[b_idx * K + k_idx] = sum;
}

// Vectorized version using float4 for better memory throughput
__global__ void fused_allgather_linear_vec4_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int B, int K, int R, int H) {
    
    int b_idx = blockIdx.y;
    int k_idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (b_idx >= B || k_idx >= K) return;
    
    float sum = 0.0f;
    
    // Process in chunks of 4 when possible
    int H_vec4 = (H / 4) * 4;
    
    for (int r = 0; r < R; r++) {
        // Vectorized loads
        const float4* input_vec = reinterpret_cast<const float4*>(
            input + r * B * H + b_idx * H);
        const float4* weight_vec = reinterpret_cast<const float4*>(
            weight + k_idx * R * H + r * H);
        
        int h_vec = 0;
        for (; h_vec < H_vec4 / 4; h_vec++) {
            float4 in_vec = input_vec[h_vec];
            float4 w_vec = weight_vec[h_vec];
            sum += in_vec.x * w_vec.x + in_vec.y * w_vec.y + 
                   in_vec.z * w_vec.z + in_vec.w * w_vec.w;
        }
        
        // Handle remaining elements
        for (int h = h_vec * 4; h < H; h++) {
            sum += input[r * B * H + b_idx * H + h] * 
                   weight[k_idx * R * H + r * H + h];
        }
    }
    
    sum += bias[k_idx];
    output[b_idx * K + k_idx] = sum;
}

torch::Tensor fused_allgather_linear_cuda(
    torch::Tensor input,  // (R, B, H)
    torch::Tensor weight, // (K, R*H)
    torch::Tensor bias,   // (K,)
    int R, int B, int H, int K) {
    
    auto output = torch::empty({B, K}, input.options());
    
    // Use vectorized kernel for better memory throughput
    int threads_per_block = 64;
    int blocks_k = (K + threads_per_block - 1) / threads_per_block;
    
    dim3 grid(blocks_k, B);
    dim3 block(threads_per_block);
    
    // Choose kernel based on H alignment
    if (H % 4 == 0) {
        fused_allgather_linear_vec4_kernel<<<grid, block>>>(
            input.data_ptr<float>(),
            weight.data_ptr<float>(),
            bias.data_ptr<float>(),
            output.data_ptr<float>(),
            B, K, R, H);
    } else {
        // Allocate shared memory for bias
        int shared_mem_size = K * sizeof(float);
        fused_allgather_linear_cached_kernel<<<grid, block, shared_mem_size>>>(
            input.data_ptr<float>(),
            weight.data_ptr<float>(),
            bias.data_ptr<float>(),
            output.data_ptr<float>(),
            B, K, R, H);
    }
    
    return output;
}
