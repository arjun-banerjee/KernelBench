import torch
import torch.nn as nn


class Model(nn.Module):
    """
    SE(3)-invariant linear layer.  Maps node features that are augmented
    with 3-D coordinate information through a linear transform that
    respects rotational and translational invariance.  The layer operates
    on (scalar_features, vector_features) pairs where scalar features
    transform trivially and vector features transform as 3-D vectors.
    Used in equivariant GNNs (EGNN, PaiNN, TFN).
    """

    def __init__(self, scalar_in, scalar_out, vector_in, vector_out):
        super().__init__()
        self.scalar_linear = nn.Linear(scalar_in + vector_in, scalar_out)
        self.vector_weight = nn.Parameter(torch.randn(vector_out, vector_in))
        nn.init.xavier_uniform_(self.vector_weight)
        self.vector_gate = nn.Linear(scalar_in + vector_in, vector_out)

    def forward(
        self, scalar_feat: torch.Tensor, vector_feat: torch.Tensor
    ) -> tuple:
        """
        Args:
            scalar_feat: (B, N, scalar_in) – invariant node features
            vector_feat: (B, N, vector_in, 3) – equivariant vector features
        Returns:
            (scalar_out, vector_out):
                scalar_out: (B, N, scalar_out)
                vector_out: (B, N, vector_out, 3)
        """
        # Invariant scalars from vector norms
        vec_norm = vector_feat.norm(dim=-1)  # (B, N, vector_in)
        combined = torch.cat([scalar_feat, vec_norm], dim=-1)

        s_out = self.scalar_linear(combined)

        # Equivariant vector transform: linear combination of input vectors
        v_out = torch.einsum("oi,bnid->bnod", self.vector_weight, vector_feat)
        # Gating by scalar-derived signal
        gate = torch.sigmoid(self.vector_gate(combined)).unsqueeze(-1)  # (B, N, vector_out, 1)
        v_out = gate * v_out

        return s_out, v_out

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

__global__ void se3_fused_kernel(
    const float* __restrict__ scalar_feat,
    const float* __restrict__ vector_feat,
    const float* __restrict__ scalar_w,
    const float* __restrict__ scalar_b,
    const float* __restrict__ vector_w,
    const float* __restrict__ gate_w,
    const float* __restrict__ gate_b,
    float* __restrict__ scalar_out,
    float* __restrict__ vector_out,
    int rows, int scalar_in, int scalar_out_dim, int vector_in, int vector_out_dim) {

    int row = blockIdx.x;
    int tid = threadIdx.x;
    int C = scalar_in + vector_in;

    extern __shared__ float sh[];
    float* comb = sh;                 // C floats: scalar features + vector norms
    float* gates = sh + C;            // vector_out_dim floats

    // Materialize the small combined feature vector once per (B,N) row.
    for (int k = tid; k < scalar_in; k += blockDim.x) {
        comb[k] = scalar_feat[row * scalar_in + k];
    }
    for (int i = tid; i < vector_in; i += blockDim.x) {
        const float* vp = vector_feat + (row * vector_in + i) * 3;
        float x = vp[0];
        float y = vp[1];
        float z = vp[2];
        comb[scalar_in + i] = sqrtf(x * x + y * y + z * z);
    }
    __syncthreads();

    // Scalar linear layer: one thread per output channel (for the target sizes).
    if (tid < scalar_out_dim) {
        const float* w = scalar_w + tid * C;
        float acc = scalar_b[tid];
        #pragma unroll 4
        for (int k = 0; k < C; ++k) {
            acc = fmaf(w[k], comb[k], acc);
        }
        scalar_out[row * scalar_out_dim + tid] = acc;
    }

    // Gate linear + sigmoid.
    if (tid < vector_out_dim) {
        const float* w = gate_w + tid * C;
        float acc = gate_b[tid];
        #pragma unroll 4
        for (int k = 0; k < C; ++k) {
            acc = fmaf(w[k], comb[k], acc);
        }
        gates[tid] = 1.0f / (1.0f + expf(-acc));
    }
    __syncthreads();

    // Vector equivariant transform and apply gate.  One thread per (out_channel, xyz).
    int t = tid;
    int total_v = vector_out_dim * 3;
    if (t < total_v) {
        int o = t / 3;
        int d = t - o * 3;
        const float* vw = vector_w + o * vector_in;
        float acc = 0.0f;
        #pragma unroll
        for (int i = 0; i < 16; ++i) {
            if (i < vector_in) {
                acc = fmaf(vw[i], vector_feat[(row * vector_in + i) * 3 + d], acc);
            }
        }
        vector_out[(row * vector_out_dim + o) * 3 + d] = acc * gates[o];
    }
}

std::vector<torch::Tensor> se3_forward_cuda(
    torch::Tensor scalar_feat,
    torch::Tensor vector_feat,
    torch::Tensor scalar_w,
    torch::Tensor scalar_b,
    torch::Tensor vector_w,
    torch::Tensor gate_w,
    torch::Tensor gate_b) {

    const int B = (int)scalar_feat.size(0);
    const int N = (int)scalar_feat.size(1);
    const int scalar_in = (int)scalar_feat.size(2);
    const int vector_in = (int)vector_feat.size(2);
    const int scalar_out_dim = (int)scalar_w.size(0);
    const int vector_out_dim = (int)vector_w.size(0);
    const int rows = B * N;

    auto s_out = torch::empty({B, N, scalar_out_dim}, scalar_feat.options());
    auto v_out = torch::empty({B, N, vector_out_dim, 3}, vector_feat.options());

    const int threads = 128;
    const size_t shmem = (size_t)(scalar_in + vector_in + vector_out_dim) * sizeof(float);

    se3_fused_kernel<<<rows, threads, shmem>>>(
        scalar_feat.data_ptr<float>(),
        vector_feat.data_ptr<float>(),
        scalar_w.data_ptr<float>(),
        scalar_b.data_ptr<float>(),
        vector_w.data_ptr<float>(),
        gate_w.data_ptr<float>(),
        gate_b.data_ptr<float>(),
        s_out.data_ptr<float>(),
        v_out.data_ptr<float>(),
        rows, scalar_in, scalar_out_dim, vector_in, vector_out_dim);

    return {s_out, v_out};
}
