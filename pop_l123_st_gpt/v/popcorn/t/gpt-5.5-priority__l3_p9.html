import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Bayesian linear regression with a conjugate Gaussian prior.
    Given features X and targets y, computes the posterior distribution
    over weights: p(w | X, y) = N(mu_post, Sigma_post) in closed form
    via the matrix inversion lemma / Cholesky decomposition.
    """

    def __init__(self, input_dim, prior_precision=1.0, noise_precision=1.0):
        super().__init__()
        self.input_dim = input_dim
        self.prior_precision = prior_precision
        self.noise_precision = noise_precision

    def forward(
        self, X: torch.Tensor, y: torch.Tensor, X_test: torch.Tensor
    ) -> tuple:
        """
        Args:
            X:      (B, N, D) – training features
            y:      (B, N, 1) – training targets
            X_test: (B, M, D) – test features
        Returns:
            (pred_mean, pred_var):
                pred_mean: (B, M, 1)
                pred_var:  (B, M)
        """
        D = X.shape[-1]
        eye = torch.eye(D, device=X.device, dtype=X.dtype).unsqueeze(0)

        # Posterior precision: Lambda_post = alpha * I + beta * X^T X
        Lambda_post = self.prior_precision * eye + self.noise_precision * (X.transpose(-1, -2) @ X)
        L = torch.linalg.cholesky(Lambda_post)

        # Posterior mean: mu_post = Lambda_post^{-1} * beta * X^T y
        rhs = self.noise_precision * (X.transpose(-1, -2) @ y)
        mu_post = torch.cholesky_solve(rhs, L)

        # Predictive distribution
        pred_mean = X_test @ mu_post
        Sigma_post = torch.cholesky_inverse(L)
        pred_var = (X_test @ Sigma_post * X_test).sum(-1) + 1.0 / self.noise_precision

        return pred_mean, pred_var

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> blr_forward(torch::Tensor X, torch::Tensor y, torch::Tensor X_test, double alpha, double beta);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

#define MAX_D 16

__global__ void gram_kernel(const float* __restrict__ X,
                            const float* __restrict__ y,
                            float* __restrict__ A,
                            float* __restrict__ rhs,
                            int B, int N, int D,
                            float alpha, float beta) {
    int b = blockIdx.x;
    int col = threadIdx.x;
    int row = threadIdx.y;
    if (row >= D || col >= D) return;

    const float* Xb = X + ((long long)b * N * D);
    float s = 0.0f;
    #pragma unroll 4
    for (int n = 0; n < N; ++n) {
        float xr = Xb[n * D + row];
        float xc = Xb[n * D + col];
        s = fmaf(xr, xc, s);
    }
    A[((long long)b * D + row) * D + col] = beta * s + ((row == col) ? alpha : 0.0f);

    if (col == 0) {
        const float* yb = y + (long long)b * N;
        float r = 0.0f;
        #pragma unroll 4
        for (int n = 0; n < N; ++n) {
            r = fmaf(Xb[n * D + row], yb[n], r);
        }
        rhs[(long long)b * D + row] = beta * r;
    }
}

__global__ void chol_predict_kernel(const float* __restrict__ A,
                                    const float* __restrict__ rhs,
                                    const float* __restrict__ X_test,
                                    float* __restrict__ pred_mean,
                                    float* __restrict__ pred_var,
                                    int B, int M, int D,
                                    float inv_noise) {
    int b = blockIdx.x;
    __shared__ float L[MAX_D * MAX_D];
    __shared__ float W[MAX_D * MAX_D];
    __shared__ float Sigma[MAX_D * MAX_D];
    __shared__ float mu[MAX_D];
    __shared__ float z[MAX_D];

    if (threadIdx.x == 0) {
        // Cholesky factorization A = L L^T.  Only the lower triangle is used.
        for (int i = 0; i < D * D; ++i) {
            L[i] = 0.0f;
            W[i] = 0.0f;
            Sigma[i] = 0.0f;
        }

        const float* Ab = A + (long long)b * D * D;
        for (int i = 0; i < D; ++i) {
            for (int j = 0; j <= i; ++j) {
                float sum = Ab[i * D + j];
                for (int k = 0; k < j; ++k) {
                    sum -= L[i * D + k] * L[j * D + k];
                }
                if (i == j) {
                    // Matrix is SPD by construction; clamp only to avoid sqrt of tiny negative roundoff.
                    L[i * D + j] = sqrtf(fmaxf(sum, 1.0e-20f));
                } else {
                    L[i * D + j] = sum / L[j * D + j];
                }
            }
        }

        const float* rb = rhs + (long long)b * D;
        // Solve L z = rhs
        for (int i = 0; i < D; ++i) {
            float sum = rb[i];
            for (int k = 0; k < i; ++k) sum -= L[i * D + k] * z[k];
            z[i] = sum / L[i * D + i];
        }
        // Solve L^T mu = z
        for (int i = D - 1; i >= 0; --i) {
            float sum = z[i];
            for (int k = i + 1; k < D; ++k) sum -= L[k * D + i] * mu[k];
            mu[i] = sum / L[i * D + i];
        }

        // W = inv(L), by triangular solves against the identity.
        for (int c = 0; c < D; ++c) {
            for (int i = 0; i < D; ++i) {
                float sum = (i == c) ? 1.0f : 0.0f;
                for (int k = 0; k < i; ++k) sum -= L[i * D + k] * W[k * D + c];
                W[i * D + c] = sum / L[i * D + i];
            }
        }
        // Sigma = inv(A) = inv(L)^T inv(L)
        for (int i = 0; i < D; ++i) {
            for (int j = 0; j < D; ++j) {
                float sum = 0.0f;
                for (int k = 0; k < D; ++k) sum += W[k * D + i] * W[k * D + j];
                Sigma[i * D + j] = sum;
            }
        }
    }
    __syncthreads();

    int m = threadIdx.x;
    if (m < M) {
        const float* xt = X_test + ((long long)b * M + m) * D;
        float mean = 0.0f;
        for (int i = 0; i < D; ++i) mean = fmaf(xt[i], mu[i], mean);

        float var_quad = 0.0f;
        for (int i = 0; i < D; ++i) {
            float tmp = 0.0f;
            for (int j = 0; j < D; ++j) tmp = fmaf(Sigma[i * D + j], xt[j], tmp);
            var_quad = fmaf(xt[i], tmp, var_quad);
        }
        pred_mean[((long long)b * M + m)] = mean;
        pred_var[(long long)b * M + m] = var_quad + inv_noise;
    }
}

std::vector<torch::Tensor> blr_forward(torch::Tensor X, torch::Tensor y, torch::Tensor X_test, double alpha_d, double beta_d) {
    const int B = (int)X.size(0);
    const int N = (int)X.size(1);
    const int D = (int)X.size(2);
    const int M = (int)X_test.size(1);

    auto opts = X.options();
    auto A = torch::empty({B, D, D}, opts);
    auto rhs = torch::empty({B, D}, opts);
    auto pred_mean = torch::empty({B, M, 1}, opts);
    auto pred_var = torch::empty({B, M}, opts);

    float alpha = (float)alpha_d;
    float beta = (float)beta_d;
    dim3 block1(D, D, 1);
    gram_kernel<<<B, block1>>>(X.data_ptr<float>(), y.data_ptr<float>(), A.data_ptr<float>(), rhs.data_ptr<float>(), B, N, D, alpha, beta);

    int threads = 128;
    chol_predict_kernel<<<B, threads>>>(A.data_ptr<float>(), rhs.data_ptr<float>(), X_test.data_ptr<float>(), pred_mean.data_ptr<float>(), pred_var.data_ptr<float>(), B, M, D, 1.0f / beta);

    return {pred_mean, pred_var};
}
"""
