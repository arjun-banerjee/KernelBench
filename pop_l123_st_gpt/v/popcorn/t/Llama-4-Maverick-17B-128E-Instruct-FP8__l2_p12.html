import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Pre-norm 1D dilated convolution block with BatchNorm: BatchNorm1d, ReLU activation, dilated Conv1d, and residual addition.
    At high dilation rates, memory access becomes non-contiguous across large strides in the sequence dimension.
    """
    def __init__(self, channels, kernel_size, dilation):
        super(Model, self).__init__()
        self.norm = nn.BatchNorm1d(channels)
        self.activation = nn.ReLU()
        self.conv = nn.Conv1d(channels, channels, kernel_size, padding=dilation, dilation=dilation)

    def forward(self, x):
        residual = x
        x = self.norm(x)
        x = self.activation(x)
        x = self.conv(x)
        x = x + residual
        return x

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Define the CUDA kernel
cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void dilated_conv1d_kernel(const float* input, const float* weights, float* output, int batch_size, int channels, int seq_len, int kernel_size, int dilation) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = channels * seq_len;
    if (idx >= batch_size * stride) return;

    int batch_idx = idx / stride;
    int channel_idx = (idx % stride) / seq_len;
    int seq_idx = idx % seq_len;

    float sum = 0.0f;
    for (int i = 0; i < kernel_size; i++) {
        int seq_offset = seq_idx - dilation * (i - kernel_size / 2);
        if (seq_offset >= 0 && seq_offset < seq_len) {
            int input_idx = batch_idx * stride + channel_idx * seq_len + seq_offset;
            int weight_idx = channel_idx * kernel_size + i;
            sum += input[input_idx] * weights[weight_idx];
        }
    }
    output[idx] = sum;
}

__global__ void batchnorm_forward_kernel(float* input, float* running_mean, float* running_var, float* gamma, float* beta, int num_elements) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= num_elements) return;
    float x = input[idx];
    float mean = running_mean[idx % (num_elements / input[0].size(0))];
    float var = running_var[idx % (num_elements / input[0].size(0))];
    float gamma_val = gamma[idx % (num_elements / input[0].size(0))];
    float beta_val = beta[idx % (num_elements / input[0].size(0))];
    input[idx] = (x - mean) / sqrtf(var + 1e-5f) * gamma_val + beta_val;
}

__global__ void relu_kernel(float* input, int num_elements) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= num_elements) return;
    input[idx] = fmaxf(input[idx], 0.0f);
}

__global__ void add_residual_kernel(float* input, float* residual, int num_elements) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= num_elements) return;
    input[idx] += residual[idx];
}
'''
