import torch
import torch.nn as nn
import math


class Model(nn.Module):
    """
    Radial Basis Function (RBF) expansion of interatomic distances.
    Standard featurization in molecular graph neural networks (SchNet,
    DimeNet, GemNet) that converts scalar distances into a feature vector
    of Gaussian-basis evaluations.
    """

    def __init__(self, num_rbf=64, cutoff=5.0):
        super().__init__()
        self.num_rbf = num_rbf
        self.cutoff = cutoff
        offsets = torch.linspace(0.0, cutoff, num_rbf)
        self.register_buffer("offsets", offsets)
        coeff = -0.5 / (offsets[1] - offsets[0]).item() ** 2
        self.coeff = coeff

    def forward(self, distances: torch.Tensor) -> torch.Tensor:
        """
        Args:
            distances: (B, N, N) – pairwise distance matrix (or (B, E) edge list)
        Returns:
            rbf_features: (*distances.shape, num_rbf) – expanded features
        """
        # Cosine cutoff envelope
        d_scaled = distances.unsqueeze(-1)  # (..., 1)
        envelope = 0.5 * (torch.cos(math.pi * d_scaled / self.cutoff) + 1.0)
        envelope = envelope * (d_scaled < self.cutoff).float()

        rbf = torch.exp(self.coeff * (d_scaled - self.offsets) ** 2)
        return rbf * envelope


num_atoms = 256
num_rbf = 64
cutoff = 5.0
batch_size = 8


def get_inputs():
    coords = torch.randn(batch_size, num_atoms, 3)
    diff = coords.unsqueeze(2) - coords.unsqueeze(1)
    distances = torch.sqrt((diff ** 2).sum(-1) + 1e-8)
    return [distances]


def get_init_inputs():
    return [num_rbf, cutoff]

import torch
import torch.nn as nn
import math
from torch.utils.cpp_extension import load_inline

# Compile CUDA kernel at module load time
cuda_source = r'''
#include <cuda.h>
#include <cuda_runtime.h>

// Constant memory for offsets (64 floats easily fits in constant cache)
__constant__ float d_offsets[64];
__constant__ float d_coeff;
__constant__ float d_cutoff;
__constant__ float d_pi_over_cutoff;

// Optimized kernel: each thread computes one distance -> all RBF outputs
// Memory access pattern: strided reads (efficient with cache), coalesced writes
// Constant memory broadcasts offsets to all threads efficiently
__global__ void rbf_kernel(const float* __restrict__ distances,
                           float* __restrict__ output,
                           int total_distances,
                           int num_rbf) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (idx >= total_distances) return;
    
    float d = distances[idx];
    float d_scaled = d / d_cutoff;
    
    // Cosine envelope with cutoff
    float envelope = 0.0f;
    if (d < d_cutoff) {
        envelope = 0.5f * (cosf(d_pi_over_cutoff * d) + 1.0f);
    }
    
    // Precompute for RBF: coeff * d^2 and d
    float coeff_d_sq = d_coeff * d * d;
    float coeff_d = d_coeff * 2.0f * d;
    
    // Compute all RBF outputs with loop unrolling hint
    #pragma unroll 16
    for (int k = 0; k < num_rbf; ++k) {
        float offset = d_offsets[k];
        // Expand: coeff * (d - offset)^2 = coeff * d^2 - 2*coeff*d*offset + coeff*offset^2
        float rbf = expf(coeff_d_sq - coeff_d * offset + d_coeff * offset * offset);
        output[idx * num_rbf + k] = rbf * envelope;
    }
}

// Host wrapper
torch::Tensor rbf_forward(torch::Tensor distances, torch::Tensor offsets, 
                          float coeff, float cutoff, int num_rbf) {
    // Ensure inputs are contiguous and on GPU
    auto distances_cont = distances.contiguous();
    TORCH_CHECK(distances_cont.dim() >= 1, "distances must have at least 1 dimension");
    TORCH_CHECK(distances_cont.scalar_type() == torch::kFloat32, "distances must be float32");
    TORCH_CHECK(distances_cont.is_cuda(), "distances must be on CUDA");
    
    // Compute total number of distance elements
    int total_distances = 1;
    for (int i = 0; i < distances_cont.dim(); ++i) {
        total_distances *= distances_cont.size(i);
    }
    
    // Create output tensor with shape (*distances.shape, num_rbf)
    std::vector<int64_t> out_shape(distances_cont.sizes().vec());
    out_shape.push_back(num_rbf);
    auto output = torch::empty(out_shape, distances_cont.options());
    
    // Copy offsets to constant memory
    cudaMemcpyToSymbol(d_offsets, offsets.data_ptr<float>(), num_rbf * sizeof(float));
    cudaMemcpyToSymbol(d_coeff, &coeff, sizeof(float));
    cudaMemcpyToSymbol(d_cutoff, &cutoff, sizeof(float));
    float pi_over_cutoff = M_PI / cutoff;
    cudaMemcpyToSymbol(d_pi_over_cutoff, &pi_over_cutoff, sizeof(float));
    
    // Launch kernel
    int threads = 256;
    int blocks = (total_distances + threads - 1) / threads;
    
    rbf_kernel<<<blocks, threads>>>(
        distances_cont.data_ptr<float>(),
        output.data_ptr<float>(),
        total_distances,
        num_rbf
    );
    
    return output;
}
'''
