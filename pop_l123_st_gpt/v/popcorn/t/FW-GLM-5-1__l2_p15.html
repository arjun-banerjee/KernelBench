import math

import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Causal grouped-query attention with KV-head expansion.

    Query heads outnumber KV heads. Keys and values are expanded by repeating
    each KV head across its assigned query-head group before causal attention.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_q_heads, head_dim = q.shape
        num_kv_heads = k.shape[2]
        group_size = num_q_heads // num_kv_heads

        expanded_k = k.repeat_interleave(group_size, dim=2)
        expanded_v = v.repeat_interleave(group_size, dim=2)

        out = torch.zeros_like(q)
        scale = 1.0 / math.sqrt(head_dim)

        for b in range(batch_size):
            for h in range(num_q_heads):
                for t in range(seq_len):
                    q_t = q[b, t, h]
                    k_hist = expanded_k[b, : t + 1, h]
                    v_hist = expanded_v[b, : t + 1, h]
                    scores = (k_hist * q_t.unsqueeze(0)).sum(dim=-1) * scale
                    attn = torch.softmax(scores, dim=0)
                    out[b, t, h] = (attn.unsqueeze(-1) * v_hist).sum(dim=0)

        return out

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cmath>

__global__ void causal_gqa_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    float* __restrict__ out,
    int batch_size, int seq_len, int num_q_heads, int num_kv_heads, int head_dim,
    float scale
) {
    const int group_size = num_q_heads / num_kv_heads;
    
    const int idx = blockIdx.x * blockDim.x + threadIdx.x;
    const int total = batch_size * num_q_heads * seq_len;
    if (idx >= total) return;
    
    const int t = idx % seq_len;
    const int h = (idx / seq_len) % num_q_heads;
    const int b = idx / (seq_len * num_q_heads);
    const int kv_h = h / group_size;
    
    // Compute pointer offsets
    const int64_t q_offset = (int64_t)b * seq_len * num_q_heads * head_dim 
                           + (int64_t)t * num_q_heads * head_dim 
                           + (int64_t)h * head_dim;
    const int64_t kv_base_offset = (int64_t)b * seq_len * num_kv_heads * head_dim;
    const int64_t kv_stride_s = (int64_t)num_kv_heads * head_dim;
    
    const float* q_ptr = q + q_offset;
    const float* k_base = k + kv_base_offset;
    const float* v_base = v + kv_base_offset;
    
    // Load query vector into registers
    float q_vec[32];
    for (int d = 0; d < head_dim; d++) {
        q_vec[d] = q_ptr[d];
    }
    
    // Online softmax accumulation for numerical stability
    float max_score = -INFINITY;
    float sum_exp = 0.0f;
    float output[32];
    for (int d = 0; d < head_dim; d++) {
        output[d] = 0.0f;
    }
    
    for (int s = 0; s <= t; s++) {
        const int64_t kv_offset = (int64_t)s * kv_stride_s + (int64_t)kv_h * head_dim;
        const float* k_ptr = k_base + kv_offset;
        const float* v_ptr = v_base + kv_offset;
        
        // Compute attention score: q_t . k_s * scale
        float score = 0.0f;
        for (int d = 0; d < head_dim; d++) {
            score += q_vec[d] * k_ptr[d];
        }
        score *= scale;
        
        // Online softmax update (Flash Attention style)
        const float new_max = fmaxf(max_score, score);
        const float exp_old = expf(max_score - new_max);
        const float exp_new = expf(score - new_max);
        
        // Rescale previous accumulation and add new contribution
        for (int d = 0; d < head_dim; d++) {
            output[d] = output[d] * exp_old + exp_new * v_ptr[d];
        }
        
        sum_exp = sum_exp * exp_old + exp_new;
        max_score = new_max;
    }
    
    // Normalize output by sum of exp
    float* out_ptr = out + q_offset;
    for (int d = 0; d < head_dim; d++) {
        out_ptr[d] = output[d] / sum_exp;
    }
}

torch::Tensor causal_gqa_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v
) {
    TORCH_CHECK(q.is_cuda() && k.is_cuda() && v.is_cuda(), "Inputs must be CUDA tensors");
    TORCH_CHECK(q.is_contiguous() && k.is_contiguous() && v.is_contiguous(), "Inputs must be contiguous");
    TORCH_CHECK(q.dim() == 4 && k.dim() == 4 && v.dim() == 4, "Inputs must be 4D tensors");
    TORCH_CHECK(q.size(3) <= 32, "head_dim must be <= 32");
    
    const int batch_size = q.size(0);
    const int seq_len = q.size(1);
    const int num_q_heads = q.size(2);
    const int head_dim = q.size(3);
    const int num_kv_heads = k.size(2);
    const float scale = 1.0f / std::sqrt((float)head_dim);
    
    auto out = torch::zeros_like(q);
    
    const int total = batch_size * num_q_heads * seq_len;
    const int threads = 256;
    const int blocks = (total + threads - 1) / threads;
    
    causal_gqa_kernel<<<blocks, threads>>>(
        q.data_ptr<float>(),
        k.data_ptr<float>(),
        v.data_ptr<float>(),
        out.data_ptr<float>(),
        batch_size, seq_len, num_q_heads, num_kv_heads, head_dim,
        scale
    );
    
    return out;
}
