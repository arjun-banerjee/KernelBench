class Model(nn.Module):
    """
    **Batched forward kinematics (FK)** for a **planar serial** revolute chain.

    Fixed link lengths ``L_j`` (buffer), joint angles ``θ_j`` (batch ``B``). For each
    link, the heading is ``Σ_{k≤j} θ_k``; joint positions are the cumulative sum of
    ``L_j [cos(heading), sin(heading)]``—fully **vectorized** over ``B`` and ``J``.

    Returns positions of the **end of each link** in the base frame ``(B, J, 2)``
    (last row is the end-effector in the plane). Typical fusion target for
    batched geometry in calibration, whole-body control, and VLA motion heads.
    """

    def __init__(self, num_joints: int, link_scale: float = 1.0):
        super().__init__()
        assert num_joints >= 1
        self.num_joints = num_joints
        lengths = torch.ones(num_joints, dtype=torch.float32) * float(link_scale)
        self.register_buffer("link_lengths", lengths)

    def forward(self, joint_angles: torch.Tensor) -> torch.Tensor:
        """
        Args:
            joint_angles: (B, J) radians, one revolute per link.
        Returns:
            positions: (B, J, 2) world-frame (x, y) at each link end; ``[..., -1, :]`` is EE.
        """
        B, J = joint_angles.shape
        assert J == self.num_joints

        heading = torch.cumsum(joint_angles, dim=-1)
        c = torch.cos(heading)
        s = torch.sin(heading)
        direction = torch.stack((c, s), dim=-1)
        displacements = direction * self.link_lengths.view(1, J, 1)
        return torch.cumsum(displacements, dim=1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

void fk_forward_cuda_launcher(torch::Tensor angles, torch::Tensor out, float link_scale);

torch::Tensor fk_forward(torch::Tensor joint_angles, double link_scale) {
    const auto B = joint_angles.size(0);
    const auto J = joint_angles.size(1);
    auto out = torch::empty({B, J, 2}, joint_angles.options());
    fk_forward_cuda_launcher(joint_angles, out, (float)link_scale);
    return out;
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

// One CUDA thread handles one batch element.  J is tiny for this benchmark (7),
// so doing the dependent prefix sums in registers is faster than launching the
// several separate PyTorch kernels needed by cumsum/cos/sin/stack/mul/cumsum.
__global__ void fk_forward_kernel(const float* __restrict__ a,
                                  float* __restrict__ out,
                                  int B, int J,
                                  long long sB, long long sJ,
                                  float link_scale) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b >= B) return;

    const float* row = a + (long long)b * sB;
    float heading = 0.0f;
    float x = 0.0f;
    float y = 0.0f;
    long long base = ((long long)b * J) * 2LL;

    // Fast path for the benchmark's J=7.  Unrolling trims loop overhead and
    // gives the compiler a compact straight-line dependency chain.
    if (J == 7 && sJ == 1) {
        float sn, cs;
        heading += row[0]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 0] = x; out[base + 1] = y;
        heading += row[1]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 2] = x; out[base + 3] = y;
        heading += row[2]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 4] = x; out[base + 5] = y;
        heading += row[3]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 6] = x; out[base + 7] = y;
        heading += row[4]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 8] = x; out[base + 9] = y;
        heading += row[5]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 10] = x; out[base + 11] = y;
        heading += row[6]; __sincosf(heading, &sn, &cs); x += link_scale * cs; y += link_scale * sn; out[base + 12] = x; out[base + 13] = y;
        return;
    }

    for (int j = 0; j < J; ++j) {
        heading += row[(long long)j * sJ];
        float sn, cs;
        __sincosf(heading, &sn, &cs);
        x += link_scale * cs;
        y += link_scale * sn;
        long long o = base + (long long)j * 2LL;
        out[o] = x;
        out[o + 1] = y;
    }
}

void fk_forward_cuda_launcher(torch::Tensor angles, torch::Tensor out, float link_scale) {
    const int B = (int)angles.size(0);
    const int J = (int)angles.size(1);
    const int threads = 128;
    const int blocks = (B + threads - 1) / threads;
    fk_forward_kernel<<<blocks, threads, 0, at::cuda::getCurrentCUDAStream()>>>(
        angles.data_ptr<float>(), out.data_ptr<float>(), B, J,
        angles.stride(0), angles.stride(1), link_scale);
}
"""
