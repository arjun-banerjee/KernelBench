def _fake_quant_symmetric_int8(x: torch.Tensor, scale: torch.Tensor) -> torch.Tensor:
    """Symmetric int8 fake-quant: round(x/scale) clamped to [-127, 127], times scale."""
    q = torch.clamp(torch.round(x / scale), -127.0, 127.0)
    return q * scale


class Model(nn.Module):
    """
    **QuantVLA selective layout fusion** (reference target for a fused CUDA/Triton kernel).

    Vision–language–action stacks often use **token-wise mixed precision**: some
    timesteps or modalities tolerate aggressive quantization while others stay in
    full precision. This module fuses in **one forward**:

      1. **Layout routing** — ``sigmoid(layout_logits)`` yields a per-token mix
         coefficient in ``(0, 1)`` (quant vs FP residual).
      2. **Quantitative path** — global symmetric **int8-style fake quantization**
         (scale from the tensor max, ``/ 127``).
      3. **Selective blend** — ``α · x_quant + (1 - α) · x``.
      4. **Head** — linear projection on the mixed activations.

    This matches a “quantitative selective layout” block: the layout logits (from a
    router, modality tags, or calibration) choose how strongly each position uses the
    quantized layout vs the identity layout.
    """

    def __init__(self, dim: int):
        super().__init__()
        self.dim = dim
        self.out_proj = nn.Linear(dim, dim, bias=True)

    def forward(self, x: torch.Tensor, layout_logits: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, T, D) VLA token activations (fused modalities).
            layout_logits: (B, T) unbounded logits; larger → more mass on quantized path.
        Returns:
            (B, T, D) updated activations.
        """
        B, T, D = x.shape
        assert D == self.dim
        assert layout_logits.shape == (B, T)

        alpha = torch.sigmoid(layout_logits).unsqueeze(-1)
        with torch.no_grad():
            amax = x.detach().abs().max().clamp(min=1e-8)
        scale = amax / 127.0
        x_q = _fake_quant_symmetric_int8(x, scale)
        mixed = alpha * x_q + (1.0 - alpha) * x
        return self.out_proj(mixed)

        super().__init__()
        self._impl = Model(dim)

    def forward(self, x: torch.Tensor, layout_logits: torch.Tensor) -> torch.Tensor:
        return self._impl(x, layout_logits)

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.cpp_extension import load_inline


_cpp_src = r'''
#include <torch/extension.h>

torch::Tensor mixed_quant_blend(torch::Tensor x, torch::Tensor layout_logits);
'''

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif
#ifndef CHECK_CONTIGUOUS
#define CHECK_CONTIGUOUS(x) TORCH_CHECK(x.is_contiguous(), #x " must be contiguous")
#endif
#ifndef CHECK_FLOAT
#define CHECK_FLOAT(x) TORCH_CHECK(x.scalar_type() == at::kFloat, #x " must be float32")
#endif

__global__ void reduce_absmax_kernel(const float* __restrict__ x,
                                     float* __restrict__ partial,
                                     int n) {
    __shared__ float smem[256];
    int tid = threadIdx.x;
    int idx = blockIdx.x * blockDim.x + tid;
    int stride = blockDim.x * gridDim.x;
    float m = 0.0f;
    for (int i = idx; i < n; i += stride) {
        float v = fabsf(x[i]);
        m = fmaxf(m, v);
    }
    smem[tid] = m;
    __syncthreads();

    #pragma unroll
    for (int s = 128; s > 0; s >>= 1) {
        if (tid < s) smem[tid] = fmaxf(smem[tid], smem[tid + s]);
        __syncthreads();
    }
    if (tid == 0) partial[blockIdx.x] = smem[0];
}

__global__ void finalize_scale_kernel(const float* __restrict__ partial,
                                      float* __restrict__ scale,
                                      int nblocks) {
    __shared__ float smem[256];
    int tid = threadIdx.x;
    float m = 0.0f;
    for (int i = tid; i < nblocks; i += 256) {
        m = fmaxf(m, partial[i]);
    }
    smem[tid] = m;
    __syncthreads();

    #pragma unroll
    for (int s = 128; s > 0; s >>= 1) {
        if (tid < s) smem[tid] = fmaxf(smem[tid], smem[tid + s]);
        __syncthreads();
    }
    if (tid == 0) {
        float amax = fmaxf(smem[0], 1.0e-8f);
        scale[0] = amax * (1.0f / 127.0f);
    }
}

__global__ void blend_kernel(const float* __restrict__ x,
                             const float* __restrict__ logits,
                             const float* __restrict__ scale_ptr,
                             float* __restrict__ out,
                             int M, int D) {
    int row = blockIdx.x;
    int tid = threadIdx.x;
    if (row >= M) return;
    float scale = scale_ptr[0];
    float inv_scale = 1.0f / scale;
    float l = logits[row];
    float alpha = 1.0f / (1.0f + expf(-l));
    int base = row * D;
    for (int d = tid; d < D; d += blockDim.x) {
        float xv = x[base + d];
        // nearbyintf/rintf follows round-to-nearest-even, matching torch.round for fp32.
        float q = nearbyintf(xv * inv_scale);
        q = fminf(127.0f, fmaxf(-127.0f, q));
        float xq = q * scale;
        out[base + d] = fmaf(alpha, xq - xv, xv);
    }
}

torch::Tensor mixed_quant_blend(torch::Tensor x, torch::Tensor layout_logits) {
    CHECK_CUDA(x);
    CHECK_CUDA(layout_logits);
    CHECK_CONTIGUOUS(x);
    CHECK_CONTIGUOUS(layout_logits);
    CHECK_FLOAT(x);
    CHECK_FLOAT(layout_logits);
    TORCH_CHECK(x.dim() == 3, "x must be (B,T,D)");
    TORCH_CHECK(layout_logits.dim() == 2, "layout_logits must be (B,T)");
    const int64_t B = x.size(0);
    const int64_t T = x.size(1);
    const int64_t D64 = x.size(2);
    TORCH_CHECK(layout_logits.size(0) == B && layout_logits.size(1) == T, "bad layout_logits shape");
    TORCH_CHECK(D64 <= 4096, "D too large for this kernel");

    c10::cuda::CUDAGuard device_guard(x.device());
    auto out = torch::empty_like(x);
    const int n = (int)x.numel();
    const int M = (int)(B * T);
    const int D = (int)D64;

    const int threads = 256;
    int blocks = (n + threads - 1) / threads;
    if (blocks < 1) blocks = 1;
    if (blocks > 256) blocks = 256;

    auto opts = x.options();
    auto partial = torch::empty({blocks}, opts);
    auto scale = torch::empty({1}, opts);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    reduce_absmax_kernel<<<blocks, threads, 0, stream>>>(
        x.data_ptr<float>(), partial.data_ptr<float>(), n);
    finalize_scale_kernel<<<1, threads, 0, stream>>>(
        partial.data_ptr<float>(), scale.data_ptr<float>(), blocks);
    blend_kernel<<<M, threads, 0, stream>>>(
        x.data_ptr<float>(), layout_logits.data_ptr<float>(), scale.data_ptr<float>(),
        out.data_ptr<float>(), M, D);
    return out;
}
'''
