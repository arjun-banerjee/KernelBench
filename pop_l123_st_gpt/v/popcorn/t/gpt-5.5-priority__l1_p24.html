import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Voxel-grid pooling for 3-D molecular point clouds.  Discretizes
    continuous 3-D coordinates into a regular grid and pools per-atom
    features into each voxel.  Used in 3-D-CNN approaches for molecular
    property prediction and ligand docking (e.g., PointVoxel-CNN,
    3D-Infomax).
    """

    def __init__(self, feat_dim, grid_size=32, voxel_size=1.0):
        super().__init__()
        self.feat_dim = feat_dim
        self.grid_size = grid_size
        self.voxel_size = voxel_size
        self.linear = nn.Linear(feat_dim, feat_dim)

    def forward(
        self, coords: torch.Tensor, features: torch.Tensor
    ) -> torch.Tensor:
        """
        Args:
            coords:   (B, N, 3) – atom coordinates
            features: (B, N, feat_dim) – per-atom features
        Returns:
            voxel_grid: (B, feat_dim, G, G, G) – pooled voxel features
        """
        B, N, _ = coords.shape
        G = self.grid_size

        features = self.linear(features)

        # Center coordinates and discretize
        center = coords.mean(dim=1, keepdim=True)
        shifted = (coords - center) / self.voxel_size + G / 2.0
        indices = shifted.long().clamp(0, G - 1)  # (B, N, 3)

        voxel_grid = coords.new_zeros(B, self.feat_dim, G, G, G)
        ix = indices[:, :, 0]
        iy = indices[:, :, 1]
        iz = indices[:, :, 2]

        batch_idx = torch.arange(B, device=coords.device).unsqueeze(1).expand(-1, N)
        for d in range(self.feat_dim):
            feat_d = features[:, :, d]
            voxel_grid[:, d].index_put_(
                (batch_idx.reshape(-1), ix.reshape(-1), iy.reshape(-1), iz.reshape(-1)),
                feat_d.reshape(-1),
                accumulate=True,
            )

        return voxel_grid

__global__ void center_kernel(const float* __restrict__ coords,
                              float* __restrict__ centers,
                              int B, int N) {
    int b = blockIdx.x;
    int tid = threadIdx.x;

    float sx = 0.0f, sy = 0.0f, sz = 0.0f;
    for (int n = tid; n < N; n += blockDim.x) {
        const float* p = coords + ((b * N + n) * 3);
        sx += p[0];
        sy += p[1];
        sz += p[2];
    }

    __shared__ float shx[256];
    __shared__ float shy[256];
    __shared__ float shz[256];
    shx[tid] = sx;
    shy[tid] = sy;
    shz[tid] = sz;
    __syncthreads();

    for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) {
            shx[tid] += shx[tid + stride];
            shy[tid] += shy[tid + stride];
            shz[tid] += shz[tid + stride];
        }
        __syncthreads();
    }

    if (tid == 0) {
        float invN = 1.0f / (float)N;
        centers[b * 3 + 0] = shx[0] * invN;
        centers[b * 3 + 1] = shy[0] * invN;
        centers[b * 3 + 2] = shz[0] * invN;
    }
}

__global__ void scatter_linear_kernel(const float* __restrict__ coords,
                                      const float* __restrict__ features,
                                      const float* __restrict__ weight,
                                      const float* __restrict__ bias,
                                      const float* __restrict__ centers,
                                      float* __restrict__ out,
                                      int B, int N, int D, int G,
                                      float inv_voxel) {
    int atom = blockIdx.x;
    int b = atom / N;
    int n = atom - b * N;
    int tid = threadIdx.x;

    extern __shared__ float shfeat[];
    const float* fbase = features + (atom * D);
    for (int k = tid; k < D; k += blockDim.x) {
        shfeat[k] = fbase[k];
    }

    const float* c = coords + atom * 3;
    float gx_f = (c[0] - centers[b * 3 + 0]) * inv_voxel + ((float)G) * 0.5f;
    float gy_f = (c[1] - centers[b * 3 + 1]) * inv_voxel + ((float)G) * 0.5f;
    float gz_f = (c[2] - centers[b * 3 + 2]) * inv_voxel + ((float)G) * 0.5f;

    int ix = (int)gx_f;
    int iy = (int)gy_f;
    int iz = (int)gz_f;
    ix = ix < 0 ? 0 : (ix >= G ? G - 1 : ix);
    iy = iy < 0 ? 0 : (iy >= G ? G - 1 : iy);
    iz = iz < 0 ? 0 : (iz >= G ? G - 1 : iz);

    __syncthreads();

    long long voxel_offset = ((long long)ix * G + iy) * G + iz;
    long long channel_stride = (long long)G * G * G;
    long long batch_base = (long long)b * D * channel_stride;

    for (int d = tid; d < D; d += blockDim.x) {
        float acc = bias[d];
        const float* wrow = weight + d * D;
        #pragma unroll
        for (int k = 0; k < 32; ++k) {
            if (k < D) acc += shfeat[k] * wrow[k];
        }
        for (int k = 32; k < D; ++k) {
            acc += shfeat[k] * wrow[k];
        }
        atomicAdd(out + batch_base + (long long)d * channel_stride + voxel_offset, acc);
    }
}

torch::Tensor voxel_forward_cuda(torch::Tensor coords,
                                 torch::Tensor features,
                                 torch::Tensor weight,
                                 torch::Tensor bias,
                                 int64_t G64,
                                 double voxel_size) {
    const at::cuda::OptionalCUDAGuard device_guard(device_of(coords));
    int B = (int)coords.size(0);
    int N = (int)coords.size(1);
    int D = (int)features.size(2);
    int G = (int)G64;

    auto out = torch::empty({B, D, G, G, G}, coords.options());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    cudaMemsetAsync(out.data_ptr<float>(), 0, out.numel() * sizeof(float), stream);

    auto centers = torch::empty({B, 3}, coords.options());
