import torch
import torch.nn as nn
import math


class Model(nn.Module):
    """
    Batched Metropolis-Hastings MCMC step.  Given a current state and a
    log-probability function (here a Gaussian target for benchmarking),
    proposes a new state via a Gaussian random walk, computes the
    acceptance ratio, and stochastically accepts or rejects.
    """

    def __init__(self, dim, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.step_size = step_size
        # Target: N(mu, sigma^2 I)
        self.register_buffer("mu", torch.zeros(dim))
        self.register_buffer("sigma", torch.ones(dim))

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (((x - self.mu) / self.sigma) ** 2).sum(dim=-1)

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, dim) – current chain states
        Returns:
            (x_new, accepted):
                x_new:    (B, dim) – states after accept/reject
                accepted: (B,)    – boolean mask of accepted proposals
        """
        proposal = x + self.step_size * torch.randn_like(x)
        log_alpha = self._log_prob(proposal) - self._log_prob(x)
        log_u = torch.log(torch.rand(x.shape[0], device=x.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        x_new = torch.where(accepted.unsqueeze(-1), proposal, x)
        return x_new, accepted

from torch.utils.cpp_extension import load_inline

cuda_source = r"""
#include <cuda.h>
#include <cuda_runtime.h>
#include <curand_kernel.h>
#include <torch/extension.h>

__global__ void mcmc_step_kernel(
    const float* __restrict__ x,
    float* __restrict__ x_new,
    uint8_t* __restrict__ accepted,
    const float* __restrict__ mu,
    const float* __restrict__ sigma,
    int B, int dim, float step_size,
    unsigned long long seed
) {
    const int batch_idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (batch_idx >= B) return;
    
    // Initialize RNG state for this thread
    curandState state;
    curand_init(seed, batch_idx, 0, &state);
    
    // Compute log_prob(x) and generate proposal in one pass
    float sum_sq_x = 0.0f;  // sum of ((x - mu) / sigma)^2
    float sum_sq_p = 0.0f;  // sum of ((proposal - mu) / sigma)^2
    
    // Process elements in pairs using curand_normal2 for efficiency
    const int pairs = dim / 2;
    for (int p = 0; p < pairs; p++) {
        const int d0 = p * 2;
        const int d1 = d0 + 1;
        
        const float x0 = __ldg(&x[batch_idx * dim + d0]);
        const float x1 = __ldg(&x[batch_idx * dim + d1]);
        
        const float mu0 = __ldg(&mu[d0]);
        const float mu1 = __ldg(&mu[d1]);
        
        const float inv_sigma0 = __frcp_rn(__ldg(&sigma[d0]));
        const float inv_sigma1 = __frcp_rn(__ldg(&sigma[d1]));
        
        // Contribution from x
        const float diff_x0 = (x0 - mu0) * inv_sigma0;
        const float diff_x1 = (x1 - mu1) * inv_sigma1;
        sum_sq_x = __fmaf_rn(diff_x0, diff_x0, __fmaf_rn(diff_x1, diff_x1, sum_sq_x));
        
        // Generate proposal using two normal samples at once
        const float2 noise = curand_normal2(&state);
        const float prop0 = __fmaf_rn(step_size, noise.x, x0);
        const float prop1 = __fmaf_rn(step_size, noise.y, x1);
        x_new[batch_idx * dim + d0] = prop0;
        x_new[batch_idx * dim + d1] = prop1;
        
        // Contribution from proposal
        const float diff_p0 = (prop0 - mu0) * inv_sigma0;
        const float diff_p1 = (prop1 - mu1) * inv_sigma1;
        sum_sq_p = __fmaf_rn(diff_p0, diff_p0, __fmaf_rn(diff_p1, diff_p1, sum_sq_p));
    }
    
    // Handle remaining element if dim is odd
    if (dim & 1) {
        const int d = dim - 1;
        const float x_d = __ldg(&x[batch_idx * dim + d]);
        const float mu_d = __ldg(&mu[d]);
        const float inv_sigma_d = __frcp_rn(__ldg(&sigma[d]));
        
        const float diff_x = (x_d - mu_d) * inv_sigma_d;
        sum_sq_x = __fmaf_rn(diff_x, diff_x, sum_sq_x);
        
        const float noise = curand_normal(&state);
        const float prop_d = __fmaf_rn(step_size, noise, x_d);
        x_new[batch_idx * dim + d] = prop_d;
        
        const float diff_p = (prop_d - mu_d) * inv_sigma_d;
        sum_sq_p = __fmaf_rn(diff_p, diff_p, sum_sq_p);
    }
    
    // Compute log acceptance ratio
    // log_alpha = log_prob(proposal) - log_prob(x)
    //           = -0.5 * sum_sq_p - (-0.5 * sum_sq_x)
    //           = 0.5 * (sum_sq_x - sum_sq_p)
    const float log_alpha = 0.5f * (sum_sq_x - sum_sq_p);
    
    // Generate uniform and check acceptance
    const float u = curand_uniform(&state);
    const float log_u = logf(fmaxf(u, 1e-10f));
    
    const bool accept = log_u < log_alpha;
    accepted[batch_idx] = accept ? uint8_t(1) : uint8_t(0);
    
    // If rejected, restore original x
    if (!accept) {
        for (int d = 0; d < dim; d++) {
            x_new[batch_idx * dim + d] = __ldg(&x[batch_idx * dim + d]);
        }
    }
}

void mcmc_step_cuda(
    const float* x, float* x_new, uint8_t* accepted,
    const float* mu, const float* sigma,
    int B, int dim, float step_size, unsigned long long seed
) {
    const int threads = 256;
    const int blocks = (B + threads - 1) / threads;
    mcmc_step_kernel<<<blocks, threads>>>(
        x, x_new, accepted, mu, sigma, B, dim, step_size, seed
    );
}
"""

cpp_source = r"""
#include <torch/extension.h>

void mcmc_step_cuda(
    const float* x, float* x_new, uint8_t* accepted,
    const float* mu, const float* sigma,
    int B, int dim, float step_size, unsigned long long seed
);

void mcmc_step(
    torch::Tensor x, torch::Tensor x_new, torch::Tensor accepted,
    torch::Tensor mu, torch::Tensor sigma,
    int64_t B, int64_t dim, float step_size, unsigned long long seed
) {
    mcmc_step_cuda(
        x.data_ptr<float>(), x_new.data_ptr<float>(), accepted.data_ptr<uint8_t>(),
        mu.data_ptr<float>(), sigma.data_ptr<float>(),
        static_cast<int>(B), static_cast<int>(dim), step_size, seed
    );
}
"""

mcmc_module = load_inline(
    name='mcmc_cuda_v1',
    cpp_sources=cpp_source,
    cuda_sources=cuda_source,
    functions=['mcmc_step'],
    extra_cuda_cflags=['-O3', '--use_fast_math'],
    verbose=False
)

class ModelNew(nn.Module):
    """
    Batched Metropolis-Hastings MCMC step with custom CUDA kernel.
    Fuses random number generation, proposal computation, and acceptance
    into a single kernel launch for optimal performance.
    """
    
    def __init__(self, dim, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.step_size = step_size
        self.register_buffer("mu", torch.zeros(dim))
        self.register_buffer("sigma", torch.ones(dim))
        self._call_count = 0
    
    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, dim) - current chain states
        Returns:
            (x_new, accepted):
                x_new:    (B, dim) - states after accept/reject
                accepted: (B,)     - boolean mask of accepted proposals
        """
        B, dim = x.shape
        assert dim == self.dim, f"Expected dim={self.dim}, got {dim}"
        assert x.is_cuda, "Input must be on CUDA device"
        
        x_new = torch.empty_like(x)
        accepted = torch.empty(B, dtype=torch.uint8, device=x.device)
        
        # Use a different seed for each call to ensure independent samples
        seed = (self._call_count * 2654435761ULL + 12345) & 0xFFFFFFFFFFFFFFFF
        self._call_count += 1
        
        mcmc_module.mcmc_step(
            x.contiguous(), x_new, accepted,
            self.mu.contiguous(), self.sigma.contiguous(),
            B, dim, self.step_size, seed
        )
        
        return x_new, accepted.to(torch.bool)
