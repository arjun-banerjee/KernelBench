import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a causal 1D convolution: left-padded so each output only depends on current and past inputs.
    Used in autoregressive sequence models (Mamba, Hyena, WaveNet, TCN) for causal sequence processing.

    Args:
        in_channels (int): Number of input channels.
        out_channels (int): Number of output channels.
        kernel_size (int): Size of the convolution kernel.
        dilation (int, optional): Dilation rate. Defaults to 1.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int = 1, bias: bool = False):
        super(Model, self).__init__()
        self.causal_padding = (kernel_size - 1) * dilation
        self.conv1d = nn.Conv1d(in_channels, out_channels, kernel_size, dilation=dilation, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = torch.nn.functional.pad(x, (self.causal_padding, 0))
        return self.conv1d(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void causal_conv1d_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    float* __restrict__ output,
    int batch_size,
    int in_channels,
    int out_channels,
    int seq_len,
    int kernel_size,
    int dilation
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * out_channels * seq_len;
    if (idx >= total) return;

    int b = idx / (out_channels * seq_len);
    int rem = idx % (out_channels * seq_len);
    int oc = rem / seq_len;
    int t = rem % seq_len;

    float sum = 0.0f;
    for (int ic = 0; ic < in_channels; ++ic) {
        for (int k = 0; k < kernel_size; ++k) {
            int in_pos = t - (kernel_size - 1 - k) * dilation;
            float val = 0.0f;
            if (in_pos >= 0 && in_pos < seq_len) {
                val = input[b * in_channels * seq_len + ic * seq_len + in_pos];
            }
            sum += val * weight[oc * in_channels * kernel_size + ic * kernel_size + k];
        }
    }
    output[idx] = sum;
}

torch::Tensor causal_conv1d_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    int kernel_size,
    int dilation
) {
    int batch_size = input.size(0);
    int in_channels = input.size(1);
    int seq_len = input.size(2);
    int out_channels = weight.size(0);

    auto output = torch::empty({batch_size, out_channels, seq_len}, input.options());

    int total = batch_size * out_channels * seq_len;
    int threads = 256;

        output.data_ptr<float>(),

}
