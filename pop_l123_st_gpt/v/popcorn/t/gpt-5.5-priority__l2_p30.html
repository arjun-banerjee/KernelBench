import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Selective scan (Mamba's S6 layer): projects input to derive input-dependent SSM parameters
    (delta, B, C), discretizes a learned state matrix A, runs the linear recurrence, and
    applies a skip connection via D. The fusion opportunity is keeping all intermediates
    (discretized parameters, hidden states) in SRAM rather than materializing them to global memory.

    Args:
        d_model (int): Input feature dimension.
        d_state (int): Hidden state dimension per channel (N in Mamba, typically 16).
    """
    def __init__(self, d_model: int, d_state: int):
        super(Model, self).__init__()
        self.d_model = d_model
        self.d_state = d_state
        self.A_log = nn.Parameter(torch.randn(d_model, d_state))
        self.proj_B = nn.Linear(d_model, d_state, bias=False)
        self.proj_C = nn.Linear(d_model, d_state, bias=False)
        self.proj_delta = nn.Linear(d_model, d_model, bias=True)
        self.D = nn.Parameter(torch.ones(d_model))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        B, L, D = x.shape
        N = self.d_state

        delta = torch.nn.functional.softplus(self.proj_delta(x))
        B_t = self.proj_B(x)
        C_t = self.proj_C(x)

        A = -torch.exp(self.A_log)
        dA = torch.exp(delta.unsqueeze(-1) * A.unsqueeze(0).unsqueeze(0))
        dB = delta.unsqueeze(-1) * B_t.unsqueeze(2)
        x_db = x.unsqueeze(-1) * dB

        h = torch.zeros(B, D, N, device=x.device, dtype=x.dtype)
        outputs = []
        for t in range(L):
            h = dA[:, t] * h + x_db[:, t]
            y_t = (h * C_t[:, t].unsqueeze(1)).sum(dim=-1)
            outputs.append(y_t)

        y = torch.stack(outputs, dim=1)
        y = y + x * self.D.unsqueeze(0).unsqueeze(0)
        return y

import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.cpp_extension import load_inline

_cpp_source = r'''
#include <torch/extension.h>

torch::Tensor scan_forward(torch::Tensor x,
                           torch::Tensor delta,
                           torch::Tensor Bt,
                           torch::Tensor Ct,
                           torch::Tensor A_log,
                           torch::Tensor Dskip);
'''

_cuda_source = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void s6_scan_kernel(const float* __restrict__ x,
                               const float* __restrict__ delta,
                               const float* __restrict__ Bt,
                               const float* __restrict__ Ct,
                               const float* __restrict__ A_log,
                               const float* __restrict__ Dskip,
                               float* __restrict__ y,
                               int B, int L, int D, int N) {
    const int d = blockIdx.x;
    const int b = blockIdx.y;
    const int lane = threadIdx.x;  // one warp per (batch, channel)

    float h = 0.0f;
    float a = 0.0f;
    if (lane < N) {
        a = -expf(A_log[d * N + lane]);
    }

    const int bd_x_base = b * L * D + d;
    const int b_bn_base = b * L * N;
    const float dskip = Dskip[d];

    #pragma unroll 1
    for (int t = 0; t < L; ++t) {
        const int xidx = bd_x_base + t * D;
        const float del = delta[xidx];
        const float xv = x[xidx];
        float v = 0.0f;

        if (lane < N) {
            const int nidx = b_bn_base + t * N + lane;
            const float da = expf(del * a);
            h = da * h + (xv * del) * Bt[nidx];
            v = h * Ct[nidx];
        }

        // Warp reduction over N (N=16 in the benchmark; this also works for N<=32).
        v += __shfl_down_sync(0xffffffff, v, 16);
        v += __shfl_down_sync(0xffffffff, v, 8);
        v += __shfl_down_sync(0xffffffff, v, 4);
        v += __shfl_down_sync(0xffffffff, v, 2);
        v += __shfl_down_sync(0xffffffff, v, 1);

        if (lane == 0) {
            y[xidx] = v + xv * dskip;
        }
    }
}

torch::Tensor scan_forward(torch::Tensor x,
                           torch::Tensor delta,
                           torch::Tensor Bt,
                           torch::Tensor Ct,
                           torch::Tensor A_log,
                           torch::Tensor Dskip) {
    const int B = (int)x.size(0);
    const int L = (int)x.size(1);
    const int D = (int)x.size(2);
    const int N = (int)Bt.size(2);

    auto y = torch::empty_like(x);

    dim3 grid(D, B);
    dim3 block(32);
    s6_scan_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>(
        x.data_ptr<float>(),
        delta.data_ptr<float>(),
        Bt.data_ptr<float>(),
        Ct.data_ptr<float>(),
        A_log.data_ptr<float>(),
        Dskip.data_ptr<float>(),
        y.data_ptr<float>(),
        B, L, D, N);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return y;
}
'''

class ModelNew(nn.Module):
    def __init__(self, d_model: int, d_state: int):
        super(ModelNew, self).__init__()
        self.d_model = d_model
        self.d_state = d_state
        self.A_log = nn.Parameter(torch.randn(d_model, d_state))
        self.proj_B = nn.Linear(d_model, d_state, bias=False)
        self.proj_C = nn.Linear(d_model, d_state, bias=False)
        self.proj_delta = nn.Linear(d_model, d_model, bias=True)
        self.D = nn.Parameter(torch.ones(d_model))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Use cuBLAS/cuDNN-backed PyTorch linear kernels for the dense projections, then
        # replace the expensive materialized discretization + Python recurrence with one
        # fused CUDA scan kernel that keeps each channel's state in registers.
        if not x.is_contiguous():
            x = x.contiguous()
        delta = F.softplus(F.linear(x, self.proj_delta.weight, self.proj_delta.bias))
        Bt = F.linear(x, self.proj_B.weight, None)
        Ct = F.linear(x, self.proj_C.weight, None)
        return _s6_ext.scan_forward(x, delta, Bt, Ct, self.A_log, self.D)
