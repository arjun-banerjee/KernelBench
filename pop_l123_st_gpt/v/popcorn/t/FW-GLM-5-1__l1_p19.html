class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues

#include <cuda.h>
#include <cuda_runtime.h>
#include <cmath>

__global__ void __launch_bounds__(512) pairwise_distance_kernel(
    const float* __restrict__ coords,
    float* __restrict__ dist,
    int num_atoms
) {
    int b = blockIdx.x;
    
    extern __shared__ float s_coords[];
    
    // Cooperatively load all coordinates for this batch into shared memory
    for (int idx = threadIdx.x; idx < num_atoms * 3; idx += blockDim.x) {
        s_coords[idx] = coords[b * num_atoms * 3 + idx];
    }
    __syncthreads();
    
    int j = threadIdx.x;
    
    if (j >= num_atoms) return;
    
    // Load coords[j] into registers (reused for all i)
    float cj0 = s_coords[j * 3 + 0];
    float cj1 = s_coords[j * 3 + 1];
    float cj2 = s_coords[j * 3 + 2];
    
    // Compute distances for column j (all i values)
    // This gives coalesced writes since threads write to consecutive j values
    for (int i = 0; i < num_atoms; i++) {
        float ci0 = s_coords[i * 3 + 0];
        float ci1 = s_coords[i * 3 + 1];
        float ci2 = s_coords[i * 3 + 2];
        
        float dx = ci0 - cj0;
        float dy = ci1 - cj1;
        float dz = ci2 - cj2;
        
        float dist_sq = dx * dx + dy * dy + dz * dz;
        dist[b * num_atoms * num_atoms + i * num_atoms + j] = sqrtf(dist_sq + 1e-8f);
    }
}
