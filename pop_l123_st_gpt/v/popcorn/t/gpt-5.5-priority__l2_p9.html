import torch
import torch.nn as nn
import math


class Model(nn.Module):
    """
    Batched Metropolis-Hastings MCMC step.  Given a current state and a
    log-probability function (here a Gaussian target for benchmarking),
    proposes a new state via a Gaussian random walk, computes the
    acceptance ratio, and stochastically accepts or rejects.
    """

    def __init__(self, dim, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.step_size = step_size
        # Target: N(mu, sigma^2 I)
        self.register_buffer("mu", torch.zeros(dim))
        self.register_buffer("sigma", torch.ones(dim))

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (((x - self.mu) / self.sigma) ** 2).sum(dim=-1)

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, dim) – current chain states
        Returns:
            (x_new, accepted):
                x_new:    (B, dim) – states after accept/reject
                accepted: (B,)    – boolean mask of accepted proposals
        """
        proposal = x + self.step_size * torch.randn_like(x)
        log_alpha = self._log_prob(proposal) - self._log_prob(x)
        log_u = torch.log(torch.rand(x.shape[0], device=x.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        x_new = torch.where(accepted.unsqueeze(-1), proposal, x)
        return x_new, accepted

// One block handles one Markov chain row.  For the benchmark D=64, so this
// fuses proposal construction, both log-probability reductions, accept test,
// and final select into a single pass over x/noise.
__global__ void mh_kernel(const float* __restrict__ x,
                          const float* __restrict__ noise,
                          const float* __restrict__ u,
                          float* __restrict__ out,
                          unsigned char* __restrict__ accepted,
                          int B, int D, float step) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    extern __shared__ float smem[];

    float local = 0.0f;
    int base = b * D;

    // Compute sum(proposal^2 - x^2).  Also store proposal temporarily in out;
    // rejected rows are overwritten with x after the accept decision.
    for (int d = tid; d < D; d += blockDim.x) {
        float xv = x[base + d];
        float p = xv + step * noise[base + d];
        out[base + d] = p;
        local += p * p - xv * xv;
    }

    smem[tid] = local;
    __syncthreads();

    // blockDim is a power of two (128).  Reduction order need not be bitwise
    // identical to PyTorch; fp32 tolerance is 1e-4 and D is small.
    for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) smem[tid] += smem[tid + stride];
        __syncthreads();
    }

    __shared__ unsigned char acc_s;
    if (tid == 0) {
        float log_alpha = -0.5f * smem[0];
        float uv = u[b];
        if (uv < 1.0e-10f) uv = 1.0e-10f;
        float log_u = logf(uv);
        unsigned char acc = (log_u < log_alpha) ? 1 : 0;
        accepted[b] = acc;
        acc_s = acc;
    }
    __syncthreads();

    if (!acc_s) {
        for (int d = tid; d < D; d += blockDim.x) {
            out[base + d] = x[base + d];
        }
    }
}

std::vector<torch::Tensor> mh_forward(torch::Tensor x, torch::Tensor noise, torch::Tensor u, double step_size) {
    TORCH_CHECK(x.is_cuda(), "x must be a CUDA tensor");
    TORCH_CHECK(noise.is_cuda() && u.is_cuda(), "noise and u must be CUDA tensors");
    TORCH_CHECK(x.scalar_type() == at::kFloat, "only fp32 is supported");
    TORCH_CHECK(noise.scalar_type() == at::kFloat && u.scalar_type() == at::kFloat, "noise/u must be fp32");
    TORCH_CHECK(x.dim() == 2, "x must be 2D");

    TORCH_CHECK(u.dim() == 1 && u.size(0) == x.size(0), "u shape mismatch");

    x = x.contiguous();
    noise = noise.contiguous();
    u = u.contiguous();

    const int B = (int)x.size(0);
