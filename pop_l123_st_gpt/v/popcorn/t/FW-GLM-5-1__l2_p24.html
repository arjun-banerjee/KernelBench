import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Channel-wise gated delta attention recurrence.

    The recurrent state decays independently for each output channel, which
    makes the update closer to channel-gated delta attention variants used in
    fast long-context baselines.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-2)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

            v_local[i] = v_t[i];
            gate_local[i] = gate_t[i];
            q_local[i] = q_t[i];
        }
        
        // Need to broadcast k, v, gate, q to all threads in the block
        // Using shared memory for broadcast
        __shared__ float shared_k[24], shared_v[24], shared_gate[24], shared_q[24];
        
        for (int i = lane_id; i < head_dim; i += blockDim.x) {
            shared_k[i] = k_local[i];
            shared_v[i] = v_local[i];
            shared_gate[i] = gate_local[i];
            shared_q[i] = q_local[i];
        }
        __syncthreads();
        
        // Update state: state = gate * state + kv_outer
        // Cooperative update of state matrix
        for (int idx = lane_id; idx < head_dim * head_dim; idx += blockDim.x) {
            int row = idx / head_dim;
            int col = idx % head_dim;
            // gate[row] broadcasts across columns of state[row, :]
            // kv_outer[row, col] = k[row] * v[col]
            float gate_val = shared_gate[row];
            float kv_val = shared_k[row] * shared_v[col];
            state[idx] = gate_val * state[idx] + kv_val;
        }
        __syncthreads();
        
        // Compute output: y[i] = sum_j(q[i] * state[i, j])
        // Each thread computes one or more output elements
        for (int i = lane_id; i < head_dim; i += blockDim.x) {
            float q_val = shared_q[i];
            float sum = 0.0f;
            for (int j = 0; j < head_dim; j++) {
                sum += q_val * state[i * head_dim + j];
            }
            out_t[i] = sum;
        }
        __syncthreads();
    }
}

// Alternative: simple kernel with one thread per (batch, head) pair
__global__ void gated_delta_attention_simple_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    float* __restrict__ output,
    int batch_size, int seq_len, int num_heads, int head_dim
) {
    int batch_idx = blockIdx.y;
    int head_idx = blockIdx.x;
    
    if (batch_idx >= batch_size) return;
    if (head_idx >= num_heads) return;
    
    // State matrix stored in registers (head_dim x head_dim)
    // For head_dim=24, this is 576 floats - too large for registers
    // Use a smaller tile approach or shared memory
    
    extern __shared__ float state[];
    
    // Initialize state to zero
    for (int i = threadIdx.x; i < head_dim * head_dim; i += blockDim.x) {
        state[i] = 0.0f;
    }
    __syncthreads();
    
    // Strides
    long long qkv_stride_batch = (long long)seq_len * num_heads * head_dim;
    long long qkv_stride_seq = num_heads * head_dim;
    long long qkv_stride_head = head_dim;
    
    // Process timesteps
    for (int t = 0; t < seq_len; t++) {
        long long base_offset = (long long)batch_idx * qkv_stride_batch + 
                                (long long)t * qkv_stride_seq + 
                                (long long)head_idx * qkv_stride_head;
        
        const float* q_t = q + base_offset;
        const float* k_t = k + base_offset;
        const float* v_t = v + base_offset;
        const float* gate_t = gate + base_offset;
        float* out_t = output + base_offset;
        
        // Load k, v, gate, q cooperatively
        __shared__ float shared_k[24], shared_v[24], shared_gate[24], shared_q[24];
        
        for (int i = threadIdx.x; i < head_dim; i += blockDim.x) {
            shared_k[i] = k_t[i];
            shared_v[i] = v_t[i];
            shared_gate[i] = gate_t[i];
            shared_q[i] = q_t[i];
        }
        __syncthreads();
        
        // Update state
        for (int idx = threadIdx.x; idx < head_dim * head_dim; idx += blockDim.x) {
            int row = idx / head_dim;
            int col = idx % head_dim;
            state[idx] = shared_gate[row] * state[idx] + shared_k[row] * shared_v[col];
        }
        __syncthreads();
        
        // Compute output
        for (int i = threadIdx.x; i < head_dim; i += blockDim.x) {
            float sum = 0.0f;
            for (int j = 0; j < head_dim; j++) {
                sum += shared_q[i] * state[i * head_dim + j];
            }
            out_t[i] = sum;
        }
        __syncthreads();
    }
}

// Fused kernel optimized for the specific problem dimensions
// Uses warp-level cooperation with fixed head_dim=24
template <int HEAD_DIM>
__global__ void __launch_bounds__(128) gated_delta_attention_fused_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    float* __restrict__ output,
    int batch_size, int seq_len, int num_heads
) {
    // One block per (batch, head) pair
    int batch_idx = blockIdx.y;
    int head_idx = blockIdx.x;
    
    if (batch_idx >= batch_size) return;
    if (head_idx >= num_heads) return;
    
    const int head_dim = HEAD_DIM;
    
    // Shared memory for state (head_dim x head_dim) and broadcast buffers
    __shared__ float state[24 * 24];  // Fixed size for head_dim=24
    __shared__ float shared_k[24], shared_v[24], shared_gate[24], shared_q[24];
    
    int lane_id = threadIdx.x & 31;
    int warp_id = threadIdx.x >> 5;
    int num_threads = blockDim.x;
    
    // Initialize state to zero
    #pragma unroll
    for (int i = threadIdx.x; i < head_dim * head_dim; i += num_threads) {
        state[i] = 0.0f;
    }
    __syncthreads();
    
    // Strides
    long long qkv_stride_batch = (long long)seq_len * num_heads * head_dim;
    long long qkv_stride_seq = num_heads * head_dim;
    long long qkv_stride_head = head_dim;
    
    // Base offset for this (batch, head)
    long long batch_head_offset = (long long)batch_idx * qkv_stride_batch + 
                                   (long long)head_idx * qkv_stride_head;
    
    // Process timesteps
    for (int t = 0; t < seq_len; t++) {
        const float* q_t = q + batch_head_offset + (long long)t * qkv_stride_seq;
        const float* k_t = k + batch_head_offset + (long long)t * qkv_stride_seq;
        const float* v_t = v + batch_head_offset + (long long)t * qkv_stride_seq;
        const float* gate_t = gate + batch_head_offset + (long long)t * qkv_stride_seq;
        float* out_t = output + batch_head_offset + (long long)t * qkv_stride_seq;
        
        // Cooperative load into shared memory
        for (int i = threadIdx.x; i < head_dim; i += num_threads) {
            shared_k[i] = k_t[i];
            shared_v[i] = v_t[i];
            shared_gate[i] = gate_t[i];
            shared_q[i] = q_t[i];
        }
        __syncthreads();
        
        // Update state matrix: state[r,c] = gate[r]*state[r,c] + k[r]*v[c]
        // Process in tiles for better cache utilization
        for (int idx = threadIdx.x; idx < head_dim * head_dim; idx += num_threads) {
            int row = idx / head_dim;
            int col = idx % head_dim;
            state[idx] = shared_gate[row] * state[idx] + shared_k[row] * shared_v[col];
        }
        __syncthreads();
        
        // Compute output: out[i] = sum_j(q[i] * state[i,j])
        for (int i = threadIdx.x; i < head_dim; i += num_threads) {
            float q_val = shared_q[i];
            float sum = 0.0f;
            
            // Manual unroll for head_dim=24
            #pragma unroll 8
            for (int j = 0; j < head_dim; j++) {
                sum += q_val * state[i * head_dim + j];
            }
            out_t[i] = sum;
        }
        __syncthreads();
    }
}

// Host function
torch::Tensor gated_delta_attention_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    torch::Tensor gate
) {
    int batch_size = q.size(0);
    int seq_len = q.size(1);
    int num_heads = q.size(2);
    int head_dim = q.size(3);
    
    auto output = torch::empty_like(q);
    
    // Use the fused kernel with template parameter
    dim3 grid(num_heads, batch_size);
    int threads = 128;  // Multiple warps for cooperative loading
    
    size_t shared_mem_size = 24 * 24 * sizeof(float) + 4 * 24 * sizeof(float);
    
    gated_delta_attention_fused_kernel<24><<<grid, threads, shared_mem_size>>>(
        q.data_ptr<float>(),
        k.data_ptr<float>(),
        v.data_ptr<float>(),
        gate.data_ptr<float>(),
        output.data_ptr<float>(),
        batch_size, seq_len, num_heads
    );
    
    return output;
}
