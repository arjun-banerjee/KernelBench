import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Define and compile the CUDA kernel
cuda_src = r'''
#include <cuda.h>
#include <cuda_runtime.h>

// Warp-level all-reduce using shuffle
__device__ __inline__ float warp_reduce_sum(float val) {
    for (int offset = 16; offset > 0; offset >>= 1) {
        val += __shfl_down_sync(0xffffffff, val, offset);
    }
    return val;
}

// Optimized kernel: each warp handles one edge for better memory coalescing
__global__ void edge_matmul_kernel(
    const int* __restrict__ src_idx,
    const int* __restrict__ dst_idx,
    const float* __restrict__ src_feat,
    const float* __restrict__ dst_feat,
    float* __restrict__ output,
    int num_edges,
    int feat_dim
) {
    // Each warp handles one edge
    const int warp_id = (blockIdx.x * blockDim.x + threadIdx.x) / 32;
    const int lane_id = threadIdx.x % 32;
    
    // Grid-stride loop over edges
    for (int edge = warp_id; edge < num_edges; edge += gridDim.x * blockDim.x / 32) {
        // Get the node indices for this edge
        int src = src_idx[edge];
        int dst = dst_idx[edge];
        
        // Compute partial dot product
        // Each thread handles feat_dim/32 features (assuming feat_dim divisible by 32)
        float sum = 0.0f;
        
        // Strided access for coalescing: thread k accesses feature k, k+32, k+64, ...
        // This ensures consecutive threads access consecutive memory
        for (int k = lane_id; k < feat_dim; k += 32) {
            float s_val = src_feat[src * feat_dim + k];
            float d_val = dst_feat[dst * feat_dim + k];
            sum += s_val * d_val;
        }
        
        // Reduce across the warp
        sum = warp_reduce_sum(sum);
        
        // Lane 0 writes the result
        if (lane_id == 0) {
            output[edge] = sum;
        }
    }
}

// Alternative kernel for non-power-of-2 feature dimensions
__global__ void edge_matmul_general_kernel(
    const int* __restrict__ src_idx,
    const int* __restrict__ dst_idx,
    const float* __restrict__ src_feat,
    const float* __restrict__ dst_feat,
    float* __restrict__ output,
    int num_edges,
    int feat_dim
) {
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int num_threads = gridDim.x * blockDim.x;
    
    // Each thread handles one edge
    for (int edge = tid; edge < num_edges; edge += num_threads) {
        int src = src_idx[edge];
        int dst = dst_idx[edge];
        
        float sum = 0.0f;
        for (int k = 0; k < feat_dim; k++) {
            float s_val = src_feat[src * feat_dim + k];
            float d_val = dst_feat[dst * feat_dim + k];
            sum += s_val * d_val;
        }
        
        output[edge] = sum;
    }
}

// Host wrapper - choose optimal kernel based on problem size
torch::Tensor edge_matmul_forward(
    torch::Tensor src_idx,
    torch::Tensor dst_idx,
    torch::Tensor src_feat,
    torch::Tensor dst_feat
) {
    const int num_edges = src_idx.size(0);
    const int feat_dim = src_feat.size(1);
    
    auto output = torch::empty({num_edges}, src_feat.options());
    
    const int threads_per_block = 128;  // 4 warps per block
    
    if (feat_dim % 32 == 0 && feat_dim >= 32) {
        // Use warp-optimized kernel
        // Each warp handles one edge
        const int warps_per_block = threads_per_block / 32;
        const int num_blocks = (num_edges + warps_per_block - 1) / warps_per_block;
        const int max_blocks = 65535;
        const int actual_blocks = min(num_blocks, max_blocks);
        
        edge_matmul_kernel<<<actual_blocks, threads_per_block>>>(
            src_idx.data_ptr<int>(),
            dst_idx.data_ptr<int>(),
            src_feat.data_ptr<float>(),
            dst_feat.data_ptr<float>(),
            output.data_ptr<float>(),
            num_edges,
            feat_dim
        );

        // Use general kernel with one thread per edge

            dst_idx.data_ptr<int>(),
            src_feat.data_ptr<float>(),
            dst_feat.data_ptr<float>(),
            output.data_ptr<float>(),
            num_edges,
            feat_dim
        );
    }
    
    return output;
}
