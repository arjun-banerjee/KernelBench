import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Scalar-gated delta-rule linear attention recurrence.

    A recurrent state matrix is decayed by a scalar gate per head, updated by a
    rank-1 key/value outer product, then read out by a query vector.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
        beta: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-1).unsqueeze(-1)
            beta_t = beta[:, t].unsqueeze(-1).unsqueeze(-1)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + beta_t * kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

// One block computes one (batch, head) stream.  The state is only 24x24, so
// keeping it in shared memory and doing the whole recurrence in a single kernel
// avoids the 128-step PyTorch dispatch/materialization overhead.
__global__ void delta_rule_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    const float* __restrict__ beta,
    float* __restrict__ out,
    int B, int T, int H, int D)
{
    extern __shared__ float state[]; // D*D floats, row-major [key_dim][value_dim]
    const int bh = blockIdx.x;
    const int b = bh / H;
    const int h = bh - b * H;
    const int tid = threadIdx.x;
    const int DD = D * D;

    for (int idx = tid; idx < DD; idx += blockDim.x) {
        state[idx] = 0.0f;
    }
    __syncthreads();

    const int q_base_bh = ((b * T) * H + h) * D;
    const int gh_base = (b * T) * H + h;

    for (int t = 0; t < T; ++t) {
        const float g = gate[gh_base + t * H];
        const float be = beta[gh_base + t * H];
        const int vec_base = q_base_bh + t * H * D;

        // Update recurrent matrix: state[i,j] = g*state[i,j] + beta*k[i]*v[j]
        for (int idx = tid; idx < DD; idx += blockDim.x) {
            const int i = idx / D;
            const int j = idx - i * D;
            const float kval = k[vec_base + i];
            const float vval = v[vec_base + j];
            state[idx] = fmaf(be * kval, vval, g * state[idx]);
        }
        __syncthreads();

        // Readout y[j] = sum_i q[i] * state[i,j].  Use one thread per output
        // element for deterministic, reference-like accumulation over i.

            out[vec_base + j] = acc;

        __syncthreads();
