class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        kernel_size: int,
        dilation: int = 1,
    ):
        super().__init__()
        assert kernel_size % 2 == 1, "odd kernel_size keeps T unchanged with symmetric padding"
        assert dilation >= 1
        self.padding = ((kernel_size - 1) * dilation) // 2
        self.dw = nn.Conv1d(
            in_channels,
            in_channels,
            kernel_size,
            padding=self.padding,
            dilation=dilation,
            groups=in_channels,
            bias=False,
        )
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif

__global__ void dw_silu_kernel(const float* __restrict__ x,
                               const float* __restrict__ w,
                               float* __restrict__ tmp,
                               int B, int C, int T, int K, int padding, int dilation) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * C * T;
    if (idx >= total) return;
    int t = idx % T;
    int cc = (idx / T) % C;
    int b = idx / (C * T);
    float acc = 0.0f;
    const float* wx = w + cc * K;
    const float* xb = x + (b * C + cc) * T;
    // generic odd-k dilated depthwise convolution with zero padding
    #pragma unroll
    for (int kk = 0; kk < 16; ++kk) {
        if (kk < K) {
            int it = t + kk * dilation - padding;
            if ((unsigned)it < (unsigned)T) {
                acc += xb[it] * wx[kk];
            }
        }
    }
    float sig = 1.0f / (1.0f + expf(-acc));
    tmp[idx] = acc * sig;
}

// C = input channels, O = output channels, N = B*T.
// Computes y[b,o,t] = sum_c pw[o,c] * tmp[b,c,t]
__global__ void pointwise_kernel(const float* __restrict__ tmp,
                                 const float* __restrict__ pw,
                                 float* __restrict__ y,
                                 int B, int C, int O, int T) {
    constexpr int TN = 16;
    constexpr int TO = 16;
    constexpr int TK = 32;
    __shared__ float As[TN][TK + 1];
    __shared__ float Bs[TK][TO + 1];

    int tx = threadIdx.x; // output-channel inside tile
    int ty = threadIdx.y; // N inside tile
    int tid = ty * blockDim.x + tx;
    int n0 = blockIdx.x * TN;
    int o0 = blockIdx.y * TO;
    int n = n0 + ty;
    int o = o0 + tx;
    int b = 0, t = 0;
    bool valid_n = n < B * T;
    if (valid_n) {
        b = n / T;
        t = n - b * T;
    }

    float acc = 0.0f;
    for (int c0 = 0; c0 < C; c0 += TK) {
        // load tmp tile: TN x TK
        for (int idx = tid; idx < TN * TK; idx += TN * TO) {
            int rr = idx / TK;
            int kk = idx - rr * TK;
            int nn = n0 + rr;
            int cc = c0 + kk;
            float v = 0.0f;
            if (nn < B * T && cc < C) {
                int bb = nn / T;
                int tt = nn - bb * T;
                v = tmp[(bb * C + cc) * T + tt];
            }
            As[rr][kk] = v;
        }
        // load weight tile: TK x TO
        for (int idx = tid; idx < TK * TO; idx += TN * TO) {
            int kk = idx / TO;
            int oo = idx - kk * TO;
            int cc = c0 + kk;
            int outc = o0 + oo;
            float v = 0.0f;
            if (cc < C && outc < O) v = pw[outc * C + cc];
            Bs[kk][oo] = v;
        }
        __syncthreads();

        if (valid_n && o < O) {
            #pragma unroll
            for (int kk = 0; kk < TK; ++kk) {
                if (c0 + kk < C) acc += As[ty][kk] * Bs[kk][tx];
            }
        }
        __syncthreads();
    }
    if (valid_n && o < O) {
        y[(b * O + o) * T + t] = acc;
    }
}

                            int padding, int dilation) {
