import torch
import torch.nn as nn

class Model(nn.Module):
    """
    1D depthwise separable convolution block: pointwise Conv1d (channel mixing), depthwise Conv1d (spatial),
    GELU activation, and residual addition. Used in genomic foundation models (Enformer) for efficient
    sequence processing of DNA/RNA.
    """
    def __init__(self, channels, expanded_channels, kernel_size):
        super(Model, self).__init__()
        self.pointwise = nn.Conv1d(channels, expanded_channels, 1)
        self.depthwise = nn.Conv1d(expanded_channels, expanded_channels, kernel_size,
                                   padding=kernel_size // 2, groups=expanded_channels)
        self.activation = nn.GELU()
        self.project_back = nn.Conv1d(expanded_channels, channels, 1)

    def forward(self, x):
        residual = x
        x = self.pointwise(x)
        x = self.depthwise(x)
        x = self.activation(x)
        x = self.project_back(x)
        x = x + residual
        return x

#include <torch/extension.h>
#include <cuda.h>

__global__ void fused_depthwise_separable(
    const float* __restrict__ input,
    const float* __restrict__ pw_weight,
    const float* __restrict__ pw_bias,
    const float* __restrict__ dw_weight,
    const float* __restrict__ dw_bias,
    const float* __restrict__ pb_weight,
    const float* __restrict__ pb_bias,
    float* __restrict__ output,
    int batch_size,
    int channels,
    int expanded_channels,
    int seq_len,
    int kernel_size,
    int pad) {

    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * channels * seq_len;
    if (idx >= total) return;

    int c_out = idx / (seq_len);
    int b = c_out / channels;
    c_out = c_out % channels;
    int s = idx % seq_len;

    float sum = 0.0f;

    // pointwise -> depthwise -> gelu -> project_back
    for (int ec = 0; ec < expanded_channels; ec++) {
        // pointwise
        float pw_val = 0.0f;
        for (int ic = 0; ic < channels; ic++) {
            pw_val += input[((b * channels + ic) * seq_len) + s] * 
                      pw_weight[ec * channels + ic];
        }
        if (pw_bias) pw_val += pw_bias[ec];

        // depthwise
        float dw_val = 0.0f;
        for (int k = 0; k < kernel_size; k++) {
            int in_s = s + k - pad;
            if (in_s >= 0 && in_s < seq_len) {
                dw_val += pw_val * dw_weight[ec * kernel_size + k];
            }
        }
        if (dw_bias) dw_val += dw_bias[ec];

        // GELU
        float gelu_val = 0.5f * dw_val * (1.0f + tanhf(0.79788456f * (dw_val + 0.044715f * dw_val * dw_val * dw_val)));

        // project back
        sum += gelu_val * pb_weight[c_out * expanded_channels + ec];
    }
    if (pb_bias) sum += pb_bias[c_out];

    // residual
    sum += input[((b * channels + c_out) * seq_len) + s];

    output[((b * channels + c_out) * seq_len) + s] = sum;
}
