import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Voxel-grid pooling for 3-D molecular point clouds.  Discretizes
    continuous 3-D coordinates into a regular grid and pools per-atom
    features into each voxel.  Used in 3-D-CNN approaches for molecular
    property prediction and ligand docking (e.g., PointVoxel-CNN,
    3D-Infomax).
    """

    def __init__(self, feat_dim, grid_size=32, voxel_size=1.0):
        super().__init__()
        self.feat_dim = feat_dim
        self.grid_size = grid_size
        self.voxel_size = voxel_size
        self.linear = nn.Linear(feat_dim, feat_dim)

    def forward(
        self, coords: torch.Tensor, features: torch.Tensor
    ) -> torch.Tensor:
        """
        Args:
            coords:   (B, N, 3) – atom coordinates
            features: (B, N, feat_dim) – per-atom features
        Returns:
            voxel_grid: (B, feat_dim, G, G, G) – pooled voxel features
        """
        B, N, _ = coords.shape
        G = self.grid_size

        features = self.linear(features)

        # Center coordinates and discretize
        center = coords.mean(dim=1, keepdim=True)
        shifted = (coords - center) / self.voxel_size + G / 2.0
        indices = shifted.long().clamp(0, G - 1)  # (B, N, 3)

        voxel_grid = coords.new_zeros(B, self.feat_dim, G, G, G)
        ix = indices[:, :, 0]
        iy = indices[:, :, 1]
        iz = indices[:, :, 2]

        batch_idx = torch.arange(B, device=coords.device).unsqueeze(1).expand(-1, N)
        for d in range(self.feat_dim):
            feat_d = features[:, :, d]
            voxel_grid[:, d].index_put_(
                (batch_idx.reshape(-1), ix.reshape(-1), iy.reshape(-1), iz.reshape(-1)),
                feat_d.reshape(-1),
                accumulate=True,
            )

        return voxel_grid


feat_dim = 32
grid_size = 32
voxel_size = 1.0

    return [feat_dim, grid_size, voxel_size]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# CUDA kernel for voxel pooling
cuda_source = r"""
#include <cuda.h>
#include <cuda_runtime.h>
#include <ATen/ATen.h>

__global__ void voxel_pool_kernel(
    const float* __restrict__ features,
    const long* __restrict__ indices,
    float* __restrict__ voxel_grid,
    int B, int N, int feat_dim, int G
) {
    // Each thread handles one (batch, atom, feature_dim) element
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total_elements = B * N * feat_dim;
    
    if (idx < total_elements) {
        int d = idx % feat_dim;
        int atom_idx = (idx / feat_dim) % N;
        int b = idx / (feat_dim * N);
        
        int ix = indices[(b * N + atom_idx) * 3 + 0];
        int iy = indices[(b * N + atom_idx) * 3 + 1];
        int iz = indices[(b * N + atom_idx) * 3 + 2];
        
        float val = features[idx];
        
        // Linear index into voxel_grid[b, d, ix, iy, iz]
        // Shape is [B, feat_dim, G, G, G]
        long voxel_idx = ((((long)b * feat_dim + d) * G + ix) * G + iy) * G + iz;
        atomicAdd(&voxel_grid[voxel_idx], val);
    }
}

torch::Tensor voxel_pool_cuda(
    torch::Tensor features,
    torch::Tensor indices,
    int grid_size
) {
    int B = features.size(0);
    int N = features.size(1);
    int feat_dim = features.size(2);
    int G = grid_size;
    
    auto voxel_grid = torch::zeros({B, feat_dim, G, G, G}, features.options());
    
    int total_elements = B * N * feat_dim;
    const int threads = 256;
    const int blocks = (total_elements + threads - 1) / threads;
    
    voxel_pool_kernel<<<blocks, threads>>>(
        features.data_ptr<float>(),
        indices.data_ptr<long>(),
        voxel_grid.data_ptr<float>(),
        B, N, feat_dim, G
    );
    
    return voxel_grid;
}
"""
