import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor lse_forward_cuda(torch::Tensor x);

torch::Tensor lse_forward(torch::Tensor x) {
    return lse_forward_cuda(x);
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void lse4_kernel(const float* __restrict__ x, float* __restrict__ out, int n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        float a0 = x[idx];
        float a1 = x[n + idx];
        float a2 = x[2 * n + idx];
        float a3 = x[3 * n + idx];
        float m = fmaxf(fmaxf(a0, a1), fmaxf(a2, a3));
        float s = expf(a0 - m) + expf(a1 - m) + expf(a2 - m) + expf(a3 - m);
        out[idx] = m + logf(s);
    }
}

torch::Tensor lse_forward_cuda(torch::Tensor x) {
    const int B = (int)x.size(1);
    const int D = (int)x.size(2);
    const int n = B * D;
    auto out = torch::empty({B, D}, x.options());
    const int threads = 256;
    const int blocks = (n + threads - 1) / threads;
    lse4_kernel<<<blocks, threads>>>(x.data_ptr<float>(), out.data_ptr<float>(), n);
    return out;
}
"""

_ext = load_inline(

    cuda_sources=[_cuda_src],
    functions=["lse_forward"],
    verbose=False,
    extra_cflags=["-O3"],
    extra_cuda_cflags=["-O3"],
)
