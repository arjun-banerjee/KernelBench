class Model(nn.Module):
    """
    **Fused probability-flow ODE integration** (flow-matching style) in one ``forward``.

    Models the ODE ``dx/dt = v_θ(x, t)`` on ``t ∈ [0, 1]`` with a learned velocity
    field. A **single fused pass** unrolls **fixed-step Euler** integration for
    ``num_steps`` substeps—typical for **real-time trajectory / action generation**
    where latency is dominated by repeated ``v_θ`` evaluations.

    This is the continuous-time analogue of a rectified-flow / conditional-flow
    sampler: one ``forward`` returns an approximate **x(1)** from **x(0)** without a
    Python loop across separate module calls (fusion target for the integrator +
    field).
    """

    def __init__(self, dim: int, num_steps: int):
        super().__init__()
        assert dim >= 1 and num_steps >= 1
        self.dim = dim
        self.num_steps = num_steps
        self.register_buffer("dt", torch.tensor(1.0 / float(num_steps)))

        self.velocity = nn.Sequential(
            nn.Linear(dim + 1, dim * 4),
            nn.SiLU(),
            nn.Linear(dim * 4, dim * 4),
            nn.SiLU(),
            nn.Linear(dim * 4, dim),
        )

    def forward(self, x0: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x0: (B, dim) initial state at ``t = 0`` (e.g. noise or anchor trajectory).
        Returns:
            (B, dim) approximate state at ``t = 1`` after fused Euler integration.
        """
        B, d = x0.shape
        assert d == self.dim
        x = x0
        dt = self.dt.to(dtype=x0.dtype, device=x0.device)
        for k in range(self.num_steps):
            t = (float(k) + 0.5) / float(self.num_steps)
            t_b = torch.full((B, 1), t, device=x0.device, dtype=x0.dtype)
            inp = torch.cat([x, t_b], dim=-1)
            v = self.velocity(inp)
            x = x + dt * v
        return x

static __device__ __forceinline__ float silu_f(float x) {
    return x / (1.0f + expf(-x));
}

static __device__ __forceinline__ float reduce4(float v) {
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

__global__ void flow_kernel_group4(const float* __restrict__ x0,
                                   float* __restrict__ out,
                                   const float* __restrict__ w0,
                                   const float* __restrict__ b0,
                                   const float* __restrict__ w1,
                                   const float* __restrict__ b1,
                                   const float* __restrict__ w2,
                                   const float* __restrict__ b2,
                                   int B, int dim, int steps) {
    const int b = blockIdx.x;
    const int tid = threadIdx.x;
    const int lane = tid & 3;
    const int group = tid >> 2;
    const int ngroups = blockDim.x >> 2;
    const int hidden = dim * 4;

    extern __shared__ float smem[];
    float* x = smem;
    float* h1 = x + dim;
    float* h2 = h1 + hidden;

    for (int i = tid; i < dim; i += blockDim.x) {
        x[i] = x0[b * dim + i];
    }
    __syncthreads();

    const float inv_steps = 1.0f / (float)steps;

    for (int k = 0; k < steps; ++k) {
        const float t = ((float)k + 0.5f) * inv_steps;

        // Linear(dim+1 -> 4*dim) + SiLU. Four threads cooperatively compute one output.
        for (int j = group; j < hidden; j += ngroups) {
            const float* __restrict__ wr = w0 + (size_t)j * (dim + 1);
            float acc = 0.0f;
            for (int i = lane; i < dim; i += 4) {
                acc = fmaf(x[i], __ldg(wr + i), acc);
            }
            if (lane == 0) acc = fmaf(t, __ldg(wr + dim), acc);
            acc = reduce4(acc);
            if (lane == 0) {
                acc += __ldg(b0 + j);
                h1[j] = silu_f(acc);
            }
        }
        __syncthreads();

        // Linear(4*dim -> 4*dim) + SiLU.
        for (int j = group; j < hidden; j += ngroups) {
            const float* __restrict__ wr = w1 + (size_t)j * hidden;
            float acc = 0.0f;
            for (int i = lane; i < hidden; i += 4) {
                acc = fmaf(h1[i], __ldg(wr + i), acc);
            }
            acc = reduce4(acc);
            if (lane == 0) {
                acc += __ldg(b1 + j);
                h2[j] = silu_f(acc);
