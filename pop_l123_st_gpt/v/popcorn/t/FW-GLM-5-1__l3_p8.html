import torch
import torch.nn as nn
import torch.nn.functional as F
import math


class MSARowSelfAttention(nn.Module):
    def __init__(self, msa_dim, num_heads):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        self.norm = nn.LayerNorm(msa_dim)
        self.q = nn.Linear(msa_dim, msa_dim, bias=False)
        self.k = nn.Linear(msa_dim, msa_dim, bias=False)
        self.v = nn.Linear(msa_dim, msa_dim, bias=False)
        self.out = nn.Linear(msa_dim, msa_dim)

    def forward(self, msa):
        B, S, N, D = msa.shape
        h, d = self.num_heads, self.head_dim
        x = self.norm(msa)
        q = self.q(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        k = self.k(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = self.v(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        attn = (q @ k.transpose(-1, -2)) / math.sqrt(d)
        attn = F.softmax(attn, dim=-1)
        out = (attn @ v).permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        return self.out(out)


class OuterProductMeanSmall(nn.Module):
    def __init__(self, msa_dim, pair_dim, hidden=32):
        super().__init__()
        self.norm = nn.LayerNorm(msa_dim)
        self.a = nn.Linear(msa_dim, hidden)
        self.b = nn.Linear(msa_dim, hidden)
        self.out = nn.Linear(hidden * hidden, pair_dim)

    def forward(self, msa):
        x = self.norm(msa)
        a = self.a(x)
        b = self.b(x)
        outer = torch.einsum("bsih,bsjk->bijhk", a, b)
        S = msa.shape[1]
        outer = outer / max(S, 1)
        B, N1, N2, H1, H2 = outer.shape
        return self.out(outer.reshape(B, N1, N2, H1 * H2))


class TriMulOut(nn.Module):
    def __init__(self, pair_dim, hidden):
        super().__init__()
        self.norm = nn.LayerNorm(pair_dim)
        self.pa = nn.Linear(pair_dim, hidden)
        self.pb = nn.Linear(pair_dim, hidden)
        self.ga = nn.Linear(pair_dim, hidden)
        self.gb = nn.Linear(pair_dim, hidden)
        self.norm_out = nn.LayerNorm(hidden)
        self.out = nn.Linear(hidden, pair_dim)
        self.gate_out = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair):
        x = self.norm(pair)
        a = self.pa(x) * torch.sigmoid(self.ga(x))
        b = self.pb(x) * torch.sigmoid(self.gb(x))
        o = torch.einsum("bikh,bjkh->bijh", a, b)
        o = self.out(self.norm_out(o))
        return torch.sigmoid(self.gate_out(pair)) * o

class ModelNew(nn.Module):
    """
    Optimized Evoformer block with custom CUDA kernels for fused operations.
    """
    
    def __init__(self, msa_dim, pair_dim, num_heads, tri_hidden):
        super().__init__()
        self.msa_dim = msa_dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        self.tri_hidden = tri_hidden
        self.eps = 1e-5
        
        # MSA attention weights (LayerNorm + Q/K/V projections)
        self.msa_ln_weight = nn.Parameter(torch.ones(msa_dim))
        self.msa_ln_bias = nn.Parameter(torch.zeros(msa_dim))
        self.q_weight = nn.Parameter(torch.randn(msa_dim, msa_dim) * (1.0 / math.sqrt(msa_dim)))
        self.k_weight = nn.Parameter(torch.randn(msa_dim, msa_dim) * (1.0 / math.sqrt(msa_dim)))
        self.v_weight = nn.Parameter(torch.randn(msa_dim, msa_dim) * (1.0 / math.sqrt(msa_dim)))
        self.out_weight = nn.Parameter(torch.randn(msa_dim, msa_dim) * (1.0 / math.sqrt(msa_dim)))
        self.out_bias = nn.Parameter(torch.zeros(msa_dim))
        
        # OPM weights (LayerNorm + a/b projections)
        self.opm_ln_weight = nn.Parameter(torch.ones(msa_dim))
        self.opm_ln_bias = nn.Parameter(torch.zeros(msa_dim))
        self.a_weight = nn.Parameter(torch.randn(msa_dim, 32) * (1.0 / math.sqrt(msa_dim)))
        self.b_weight = nn.Parameter(torch.randn(msa_dim, 32) * (1.0 / math.sqrt(msa_dim)))
        self.opm_out_weight = nn.Parameter(torch.randn(32 * 32, pair_dim) * (1.0 / math.sqrt(32 * 32)))
        self.opm_out_bias = nn.Parameter(torch.zeros(pair_dim))
        
        # TriMul weights (LayerNorm + pa/pb/ga/gb projections)
        self.tri_ln_weight = nn.Parameter(torch.ones(pair_dim))
        self.tri_ln_bias = nn.Parameter(torch.zeros(pair_dim))
        self.pa_weight = nn.Parameter(torch.randn(pair_dim, tri_hidden) * (1.0 / math.sqrt(pair_dim)))
        self.pb_weight = nn.Parameter(torch.randn(pair_dim, tri_hidden) * (1.0 / math.sqrt(pair_dim)))
        self.ga_weight = nn.Parameter(torch.randn(pair_dim, tri_hidden) * (1.0 / math.sqrt(pair_dim)))
        self.gb_weight = nn.Parameter(torch.randn(pair_dim, tri_hidden) * (1.0 / math.sqrt(pair_dim)))
        self.tri_out_ln_weight = nn.Parameter(torch.ones(tri_hidden))
        self.tri_out_ln_bias = nn.Parameter(torch.zeros(tri_hidden))
        self.tri_out_weight = nn.Parameter(torch.randn(tri_hidden, pair_dim) * (1.0 / math.sqrt(tri_hidden)))
        self.tri_out_bias = nn.Parameter(torch.zeros(pair_dim))
        self.gate_weight = nn.Parameter(torch.randn(pair_dim, pair_dim) * (1.0 / math.sqrt(pair_dim)))
        self.gate_bias = nn.Parameter(torch.zeros(pair_dim))
    
    def _msa_attention(self, msa):
        """MSA row self-attention with fused LayerNorm + Q/K/V kernel."""
        B, S, N, D = msa.shape
        h, d = self.num_heads, self.head_dim
        
        # Reshape for fused kernel: (B*S*N, D)
        x = msa.reshape(-1, D)
        
        # Fused LayerNorm + Q/K/V projection
        qkv = fused_ops.layernorm_qkv_cuda(
            x, self.msa_ln_weight, self.msa_ln_bias,
            self.q_weight, self.k_weight, self.v_weight, D, self.eps
        )
        q = qkv[0].view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        k = qkv[1].view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = qkv[2].view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        
        # Scaled dot-product attention
        scale = 1.0 / math.sqrt(d)
        attn = (q @ k.transpose(-1, -2)) * scale
        attn = F.softmax(attn, dim=-1)
        out = (attn @ v).permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        
        # Output projection
        return F.linear(out, self.out_weight, self.out_bias)
    
    def _outer_product_mean(self, msa):
        """Outer product mean with fused LayerNorm + a/b kernel."""
        B, S, N, D = msa.shape
        H = 32  # hidden dimension
        
        # Reshape for fused kernel: (B*S*N, D)
        x = msa.reshape(-1, D)
        
        # Fused LayerNorm + a/b projection
        ab = fused_ops.layernorm_ab_cuda(
            x, self.opm_ln_weight, self.opm_ln_bias,
            self.a_weight, self.b_weight, H, self.eps
        )
        a = ab[0].view(B, S, N, H)
        b = ab[1].view(B, S, N, H)
        
        # Outer product: einsum("bsih,bsjk->bijhk", a, b)
        # Optimized using reshape + matmul
        outer = torch.einsum("bsih,bsjk->bijhk", a, b)
        outer = outer / max(S, 1)
        
        # Output projection
        B_out, N1, N2, H1, H2 = outer.shape
        outer_flat = outer.reshape(B_out, N1, N2, H1 * H2)
        return F.linear(outer_flat, self.opm_out_weight, self.opm_out_bias)
    
    def _tri_mul_out(self, pair):
        """Triangular multiplicative update with fused LayerNorm + gated projections."""
        B, N, _, D_pair = pair.shape
        H = self.tri_hidden
        
        # Reshape for fused kernel: (B*N*N, D_pair)
        x = pair.reshape(-1, D_pair)
        
        # Fused LayerNorm + gated pa/pb projections
        ab = fused_ops.layernorm_tri_cuda(
            x, self.tri_ln_weight, self.tri_ln_bias,
            self.pa_weight, self.pb_weight,
            self.ga_weight, self.gb_weight, H, self.eps
        )
        a = ab[0].view(B, N, N, H)
        b = ab[1].view(B, N, N, H)
        
        # Triangular multiplication: einsum("bikh,bjkh->bijh", a, b)
        o = torch.einsum("bikh,bjkh->bijh", a, b)
        
        # LayerNorm + output projection
        o = F.layer_norm(o, (H,), self.tri_out_ln_weight, self.tri_out_ln_bias)
        o = F.linear(o, self.tri_out_weight, self.tri_out_bias)
        
        # Gated residual connection
        gate = torch.sigmoid(F.linear(pair, self.gate_weight, self.gate_bias))
        return gate * o
    
    def forward(self, msa_repr: torch.Tensor, pair_repr: torch.Tensor) -> tuple:
        """
        Forward pass with fused CUDA kernels.
        
        Args:
            msa_repr:  (B, S, N, msa_dim)
            pair_repr: (B, N, N, pair_dim)
        Returns:
            (updated_msa, updated_pair)
        """
        # MSA row self-attention
        msa_repr = msa_repr + self._msa_attention(msa_repr)
        
        # Outer product mean
        pair_repr = pair_repr + self._outer_product_mean(msa_repr)
        
        # Triangular multiplicative update
        pair_repr = pair_repr + self._tri_mul_out(pair_repr)
        
        return msa_repr, pair_repr
