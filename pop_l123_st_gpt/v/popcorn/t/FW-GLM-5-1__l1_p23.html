"""
Virtual ranks: input packs R per-rank tensors along dim 0. Output is the mean over ranks (B, D).

Custom CUDA kernel implementation for computing mean along dimension 0.
"""

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Compile the CUDA kernel at module load time
cuda_source = r"""
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void mean_dim0_kernel(
    const float* __restrict__ x,
    float* __restrict__ out,
    const int R,
    const int B,
    const int D,
    const long long stride_r,
    const long long stride_b,
    const long long stride_d
) {
    // 2D indexing for output element (b, d)
    int b = blockIdx.y * blockDim.y + threadIdx.y;
    int d = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (b < B && d < D) {
        // Compute mean across dimension 0 (R values)
        float sum = 0.0f;
        
        #pragma unroll
        for (int r = 0; r < R; ++r) {
            sum += x[r * stride_r + b * stride_b + d * stride_d];
        }
        
        out[b * D + d] = sum / (float)R;
    }
}

// Optimized kernel for contiguous input with small, known R
__global__ void mean_dim0_contiguous_kernel(
    const float* __restrict__ x,
    float* __restrict__ out,
    const int R,
    const int B,
    const int D
) {
    // 2D indexing for output element (b, d)
    int b = blockIdx.y * blockDim.y + threadIdx.y;
    int d = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (b < B && d < D) {
        // Input is contiguous with layout [R][B][D]
        // For fixed (b, d), values across R are spaced B*D apart
        const float* base_ptr = x + b * D + d;
        const long long stride = static_cast<long long>(B) * D;
        
        float sum = 0.0f;
        
        // Unroll for small R values
        #pragma unroll 8
        for (int r = 0; r < R; ++r) {
            sum += base_ptr[r * stride];
        }
        
        out[b * D + d] = sum / (float)R;
    }
}

// Host function that launches the kernel
torch::Tensor mean_dim0_cuda(torch::Tensor x) {
    TORCH_CHECK(x.dim() == 3, "Input must be 3D tensor");
    
    const int R = x.size(0);
    const int B = x.size(1);
    const int D = x.size(2);
    
    TORCH_CHECK(R > 0, "R must be positive");
    
    // Create contiguous output tensor
    auto out = torch::empty({B, D}, x.options());

    if (x.is_contiguous()) {

        dim3 gridSize((D + blockSize.x - 1) / blockSize.x,
                      (B + blockSize.y - 1) / blockSize.y);

            x.data_ptr<float>(),
            out.data_ptr<float>(),
            R, B, D
        );

        // Generic kernel for non-contiguous input
        dim3 blockSize(16, 16);
        dim3 gridSize((D + blockSize.x - 1) / blockSize.x,
                      (B + blockSize.y - 1) / blockSize.y);

            x.data_ptr<float>(),
            out.data_ptr<float>(),
            R, B, D,
            x.stride(0),
            x.stride(1),
            x.stride(2)

}
