import torch.nn as nn

class Model(nn.Module):
    """
    Farthest Point Sampling (FPS) on 3D point coordinates (B, N, 3).

    Used in navigation / manipulation perception (PointNet++, PVCNN, voxelization
    pipelines) to subsample a dense cloud while spreading support across the shape.

    Greedy construction (reference fusion target): fix the first index to **0** for
    determinism, then repeatedly choose the point whose **minimum** squared distance
    to the set of already chosen points is **largest** (classical FPS).
    """

    def __init__(self, num_samples: int):
        super().__init__()
        assert num_samples >= 1
        self.num_samples = num_samples

    def forward(self, points: torch.Tensor) -> torch.Tensor:
        """
        Args:
            points: (B, N, 3) point coordinates.
        Returns:
            indices: (B, M) int64, each row is M chosen indices in [0, N).
        """
        B, N, _ = points.shape
        M = self.num_samples
        assert M <= N, "FPS cannot sample more points than exist"
        device = points.device
        dtype = points.dtype

        batch_idx = torch.arange(B, device=device)
        indices = torch.empty(B, M, dtype=torch.long, device=device)

        # min squared distance from each point to the current chosen set (large init)
        min_dist_sq = torch.full((B, N), float("inf"), device=device, dtype=dtype)
        # First center: point index 0 (deterministic)
        farthest = torch.zeros(B, dtype=torch.long, device=device)

        for i in range(M):
            indices[:, i] = farthest
            center = points[batch_idx, farthest]
            dist_sq = ((points - center.unsqueeze(1)) ** 2).sum(dim=-1)
            min_dist_sq = torch.minimum(min_dist_sq, dist_sq)
            # Exclude the just-chosen index from being selected again
            min_dist_sq[batch_idx, farthest] = -1.0
            farthest = min_dist_sq.argmax(dim=-1)

        return indices


class ModelNew(nn.Module):
    """KernelBench candidate entry point; default delegates to ``Model``."""

    def __init__(self, num_samples: int):
        super().__init__()
        self._impl = Model(num_samples)

    def forward(self, points: torch.Tensor) -> torch.Tensor:
        return self._impl(points)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cstdint>

void fps_launcher(const float* points, int64_t* indices, float* temp,
                  int B, int N, int M, cudaStream_t stream);

torch::Tensor fps_forward(torch::Tensor points, int64_t M) {
    TORCH_CHECK(points.is_cuda(), "points must be a CUDA tensor");
    TORCH_CHECK(points.scalar_type() == at::kFloat, "only float32 points are supported");
    TORCH_CHECK(points.dim() == 3 && points.size(2) == 3, "points must have shape (B,N,3)");
    TORCH_CHECK(points.is_contiguous(), "points must be contiguous");
    const int B = (int)points.size(0);
    const int N = (int)points.size(1);
    TORCH_CHECK(M >= 1 && M <= N, "invalid number of samples");

    auto out = torch::empty({B, (long long)M}, torch::TensorOptions().device(points.device()).dtype(torch::kInt64));
    auto temp = torch::empty({B, N}, torch::TensorOptions().device(points.device()).dtype(torch::kFloat32));
    fps_launcher(points.data_ptr<float>(), out.data_ptr<int64_t>(), temp.data_ptr<float>(),
                 B, N, (int)M, at::cuda::getCurrentCUDAStream());
    return out;
}
'''

_cuda_src = r'''
#include <cuda.h>
#include <cuda_runtime.h>
#include <stdint.h>
#include <math_constants.h>

#ifndef BLOCK_SIZE
#define BLOCK_SIZE 256
#endif

__global__ __launch_bounds__(BLOCK_SIZE, 4)
void fps_kernel(const float* __restrict__ points,
                int64_t* __restrict__ indices,
                float* __restrict__ temp,
                int B, int N, int M) {
    const int b = blockIdx.x;
    const int tid = threadIdx.x;
    const int pbase = b * N * 3;
    const int tbase = b * N;
    const int obase = b * M;

    __shared__ float s_val[BLOCK_SIZE];
    __shared__ int   s_idx[BLOCK_SIZE];
    __shared__ int farthest;

    if (tid == 0) farthest = 0;
    __syncthreads();

    for (int m = 0; m < M; ++m) {
        const int cur = farthest;
        if (tid == 0) indices[obase + m] = (int64_t)cur;

        const float cx = points[pbase + cur * 3 + 0];
        const float cy = points[pbase + cur * 3 + 1];
        const float cz = points[pbase + cur * 3 + 2];

        float best_val = -2.0f;
        int best_idx = 0;

        for (int j = tid; j < N; j += BLOCK_SIZE) {
            const float x = points[pbase + j * 3 + 0] - cx;
            const float y = points[pbase + j * 3 + 1] - cy;
            const float z = points[pbase + j * 3 + 2] - cz;
            const float d = x * x + y * y + z * z;

            float md = (m == 0) ? CUDART_INF_F : temp[tbase + j];
            md = (d < md) ? d : md;
            if (j == cur) md = -1.0f;
            temp[tbase + j] = md;

            if (md > best_val || (md == best_val && j < best_idx)) {
                best_val = md;
                best_idx = j;
            }
        }

        s_val[tid] = best_val;
        s_idx[tid] = best_idx;
        __syncthreads();

        // Deterministic block reduction: maximize value, choose lowest index on ties,
        // matching torch.argmax semantics for all practical tie cases.
        for (int stride = BLOCK_SIZE >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) {
                const float ov = s_val[tid + stride];
                const int oi = s_idx[tid + stride];
                const float cv = s_val[tid];
                const int ci = s_idx[tid];
                if (ov > cv || (ov == cv && oi < ci)) {
                    s_val[tid] = ov;
                    s_idx[tid] = oi;
                }
            }
            __syncthreads();
        }

        if (tid == 0) farthest = s_idx[0];
        __syncthreads();
    }
}

void fps_launcher(const float* points, int64_t* indices, float* temp,
                  int B, int N, int M, cudaStream_t stream) {
    fps_kernel<<<B, BLOCK_SIZE, 0, stream>>>(points, indices, temp, B, N, M);
}
'''
