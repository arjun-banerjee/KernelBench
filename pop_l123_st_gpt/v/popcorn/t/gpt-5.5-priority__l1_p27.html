import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Match the reference's fixed CPU-generated permutation exactly.
_g = torch.Generator()
_g.manual_seed(10)
_perm_values = ",".join(str(int(v)) for v in torch.randperm(36, generator=_g).tolist())

_cpp_src = r'''
#include <torch/extension.h>
torch::Tensor permute_forward(torch::Tensor x);
'''

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda.h>
#include <cuda_runtime.h>

__device__ __constant__ int kPerm[36] = {PERM_INIT};

__global__ void perm36_kernel(const float* __restrict__ x,
                              float* __restrict__ y,
                              long long B,
                              long long s0,
                              long long s1) {
    int tid = threadIdx.x;
    int total = (int)(B * 36LL);
    // Single block is sufficient for the benchmark size (10x36), and avoids
    // extra block scheduling overhead; loop keeps it correct for larger B too.
    for (int idx = tid; idx < total; idx += blockDim.x) {
        int j = idx % 36;
        int b = idx / 36;
        y[idx] = x[(long long)b * s0 + (long long)kPerm[j] * s1];
    }
}

torch::Tensor permute_forward(torch::Tensor x) {
    const auto B = x.size(0);
    auto y = torch::empty({B, 36}, x.options());
    const int threads = 256;
    perm36_kernel<<<1, threads, 0, at::cuda::getCurrentCUDAStream()>>>(
        x.data_ptr<float>(), y.data_ptr<float>(), (long long)B,
        (long long)x.stride(0), (long long)x.stride(1));
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return y;
}
'''.replace('PERM_INIT', _perm_values)

_perm_ext = load_inline(
    name='perm36_fixed_ext',
    cpp_sources=_cpp_src,
    cuda_sources=_cuda_src,
    functions=['permute_forward'],
    extra_cflags=['-O3'],
    extra_cuda_cflags=['-O3'],
    verbose=False,
)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return _perm_ext.permute_forward(x)
