__global__ void bn_stats_kernel(const float* __restrict__ x,
                                float* __restrict__ mean,
                                float* __restrict__ var,
                                float* __restrict__ running_mean,
                                float* __restrict__ running_var,
                                int N, int C, int L, float momentum) {
    int c = blockIdx.x;
    int M = N * L;
    float s = 0.0f;
    float ss = 0.0f;
    for (int i = threadIdx.x; i < M; i += blockDim.x) {
        int n = i / L;
        int l = i - n * L;
        float v = x[((n * C + c) * L) + l];
        s += v;
        ss = fmaf(v, v, ss);
    }
    __shared__ float sh_s[256];
    __shared__ float sh_ss[256];
    sh_s[threadIdx.x] = s;
    sh_ss[threadIdx.x] = ss;
    __syncthreads();
    for (int off = 128; off > 0; off >>= 1) {
        if (threadIdx.x < off) {
            sh_s[threadIdx.x] += sh_s[threadIdx.x + off];
            sh_ss[threadIdx.x] += sh_ss[threadIdx.x + off];
        }
        __syncthreads();
    }
    if (threadIdx.x == 0) {
        float m = sh_s[0] / (float)M;
        float v = sh_ss[0] / (float)M - m * m;
        v = v < 0.0f ? 0.0f : v;
        mean[c] = m;
        var[c] = v;
        // PyTorch normalizes with the biased variance, but updates running_var with
        // the unbiased estimate.
        float vu = v * ((float)M / (float)(M - 1));
        running_mean[c] = running_mean[c] * (1.0f - momentum) + momentum * m;
        running_var[c] = running_var[c] * (1.0f - momentum) + momentum * vu;
    }
}

__global__ void bn_relu_kernel(const float* __restrict__ x,
                               float* __restrict__ a,
                               const float* __restrict__ mean,
                               const float* __restrict__ var,
                               const float* __restrict__ gamma,
                               const float* __restrict__ beta,
                               int total, int C, int L, float eps) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = blockDim.x * gridDim.x;
    for (; idx < total; idx += stride) {
        int t = idx % L;
        int tmp = idx / L;
        int c = tmp % C;
        float y = (x[idx] - mean[c]) * rsqrtf(var[c] + eps);
        y = fmaf(y, gamma[c], beta[c]);
        a[idx] = y > 0.0f ? y : 0.0f;
    }
}

// Computes Conv1d(kernel=3,padding=dilation,dilation=dilation) + residual.
// A 16x16 output tile is evaluated by one CTA.  For every 8 input channels, the
// needed activations and weights are staged in shared memory, eliminating most
// redundant global loads in this small-kernel, many-channel convolution.
template<int OTILE, int TTILE, int CTILE>
__global__ void conv3_residual_kernel(const float* __restrict__ a,
                                      const float* __restrict__ x,
                                      const float* __restrict__ w,
                                      const float* __restrict__ b,
                                      float* __restrict__ out,
                                      int N, int C, int L, int dilation) {
    __shared__ float shA[3 * CTILE * TTILE];
    __shared__ float shW[OTILE * CTILE * 3];

    int tx = threadIdx.x;             // time within tile
    int oy = threadIdx.y;             // output channel within tile
    int tid = oy * TTILE + tx;
    int t = blockIdx.x * TTILE + tx;
    int oc = blockIdx.y * OTILE + oy;
    int n = blockIdx.z;

    float acc = 0.0f;
    if (oc < C && t < L) acc = b[oc];

    #pragma unroll 1
    for (int cb = 0; cb < C; cb += CTILE) {
        // Load activation tile: [k][ci][t]
        for (int idx = tid; idx < 3 * CTILE * TTILE; idx += OTILE * TTILE) {
            int lt = idx % TTILE;
            int q = idx / TTILE;
            int ci_rel = q % CTILE;
            int k = q / CTILE;
            int ci = cb + ci_rel;
            int tt = blockIdx.x * TTILE + lt + (k - 1) * dilation;
            float v = 0.0f;
            if (ci < C && tt >= 0 && tt < L) {
                v = a[((n * C + ci) * L) + tt];
            }
            shA[idx] = v;
        }
        // Load weight tile: [oc][ci][k]
        for (int idx = tid; idx < OTILE * CTILE * 3; idx += OTILE * TTILE) {
            int k = idx % 3;
            int q = idx / 3;
            int ci_rel = q % CTILE;
            int oo = q / CTILE;
            int o = blockIdx.y * OTILE + oo;
            int ci = cb + ci_rel;
            float v = 0.0f;
            if (o < C && ci < C) {
                v = w[((o * C + ci) * 3) + k];
            }
            shW[idx] = v;
        }
        __syncthreads();

        if (oc < C && t < L) {
            #pragma unroll
            for (int ci_rel = 0; ci_rel < CTILE; ++ci_rel) {
                #pragma unroll
                for (int k = 0; k < 3; ++k) {
                    float av = shA[(k * CTILE + ci_rel) * TTILE + tx];
                    float ww = shW[(oy * CTILE + ci_rel) * 3 + k];
                    acc = fmaf(ww, av, acc);

            }

        __syncthreads();

        int oi = ((n * C + oc) * L) + t;
        out[oi] = acc + x[oi];
    }
}
