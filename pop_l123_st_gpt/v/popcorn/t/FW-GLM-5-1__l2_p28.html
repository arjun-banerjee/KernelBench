class Model(nn.Module):
    """
    Fused GroupNorm + SiLU (Swish), a common post-conv / residual-branch pattern in
    ResNet-family backbones (SiLU instead of ReLU) and in spatial blocks of diffusion
    transformers (DiT) where features are (N, C, H, W).

    Reference pipeline for a single fused kernel: normalize channels with
    `GroupNorm`, then apply `SiLU` elementwise (`x * sigmoid(x)`).
    """

    def __init__(
        self,
        num_channels: int,
        num_groups: int,
        eps: float = 1e-5,
        affine: bool = True,
    ):
        super().__init__()
        assert num_channels % num_groups == 0
        self.norm = nn.GroupNorm(
            num_groups=num_groups,
            num_channels=num_channels,
            eps=eps,
            affine=affine,
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C, H, W) feature map
        Returns:
            Same shape as x
        """
        return F.silu(self.norm(x))

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template <typename scalar_t>
__global__ void groupnorm_silu_kernel(
    const scalar_t* __restrict__ x,
    const scalar_t* __restrict__ weight,
    const scalar_t* __restrict__ bias,
    scalar_t* __restrict__ out,
    int N, int C, int H, int W, int G,
    int channels_per_group, int elements_per_group,
    scalar_t eps) {
    
    int batch_idx = blockIdx.x / G;
    int group_idx = blockIdx.x % G;
    
    int group_start_channel = group_idx * channels_per_group;
    int base_idx = batch_idx * C * H * W + group_start_channel * H * W;
    
    int tid = threadIdx.x;
    int lane_id = tid & 31;
    int warp_id = tid >> 5;
    int block_size = blockDim.x;
    int num_warps = (block_size + 31) >> 5;
    
    // Compute sum and sum of squares
    scalar_t sum = scalar_t(0);
    scalar_t sum_sq = scalar_t(0);
    
    for (int i = tid; i < elements_per_group; i += block_size) {
        scalar_t val = x[base_idx + i];
        sum += val;
        sum_sq += val * val;
    }
    
    // Warp-level reduction using shuffle
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        sum += __shfl_down_sync(0xffffffff, sum, offset);
        sum_sq += __shfl_down_sync(0xffffffff, sum_sq, offset);
    }
    
    // Shared memory for cross-warp reduction
    __shared__ scalar_t s_sum[32];
    __shared__ scalar_t s_sum_sq[32];
    
    if (lane_id == 0) {
        s_sum[warp_id] = sum;
        s_sum_sq[warp_id] = sum_sq;
    }
    __syncthreads();
    
    // First warp reduces across warps
    if (warp_id == 0) {
        sum = (lane_id < num_warps) ? s_sum[lane_id] : scalar_t(0);
        sum_sq = (lane_id < num_warps) ? s_sum_sq[lane_id] : scalar_t(0);
        
        #pragma unroll
        for (int offset = 16; offset > 0; offset >>= 1) {
            sum += __shfl_down_sync(0xffffffff, sum, offset);
            sum_sq += __shfl_down_sync(0xffffffff, sum_sq, offset);
        }
        
        if (lane_id == 0) {
            scalar_t mean = sum / static_cast<scalar_t>(elements_per_group);
            scalar_t var = sum_sq / static_cast<scalar_t>(elements_per_group) - mean * mean;
            scalar_t inv_std = rsqrt(var + eps);
            s_sum[0] = mean;
            s_sum_sq[0] = inv_std;
        }
    }
    __syncthreads();
    
    scalar_t mean = s_sum[0];
    scalar_t inv_std = s_sum_sq[0];
    
    // Apply normalization, affine, and SiLU
    for (int i = tid; i < elements_per_group; i += block_size) {
        int c_local = i / (H * W);
        int c = group_start_channel + c_local;
        int idx = base_idx + i;
        
        scalar_t val = x[idx];
        scalar_t normalized = (val - mean) * inv_std;
        
        scalar_t affine_val = normalized;
        if (weight != nullptr) {
            affine_val = affine_val * weight[c];
        }
        if (bias != nullptr) {
            affine_val = affine_val + bias[c];
        }
        
        // SiLU: x * sigmoid(x)
        scalar_t sigmoid_val = scalar_t(1) / (scalar_t(1) + exp(-affine_val));
        out[idx] = affine_val * sigmoid_val;
    }
}
