class Model(nn.Module):
    """
    Fused GroupNorm + SiLU (Swish), a common post-conv / residual-branch pattern in
    ResNet-family backbones (SiLU instead of ReLU) and in spatial blocks of diffusion
    transformers (DiT) where features are (N, C, H, W).

    Reference pipeline for a single fused kernel: normalize channels with
    `GroupNorm`, then apply `SiLU` elementwise (`x * sigmoid(x)`).
    """

    def __init__(
        self,
        num_channels: int,
        num_groups: int,
        eps: float = 1e-5,
        affine: bool = True,
    ):
        super().__init__()
        assert num_channels % num_groups == 0
        self.norm = nn.GroupNorm(
            num_groups=num_groups,
            num_channels=num_channels,
            eps=eps,
            affine=affine,
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C, H, W) feature map
        Returns:
            Same shape as x
        """
        return F.silu(self.norm(x))

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

__global__ void groupnorm_silu_kernel(
    const float* __restrict__ x,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ y,
    int N, int C, int H, int W, int G,
    float eps, bool affine)
{
    // one block computes one (n, group)
    const int bid = blockIdx.x;
    const int g = bid % G;
    const int n = bid / G;
    const int tid = threadIdx.x;
    const int spatial = H * W;
    const int cpg = C / G;
    const int elems = cpg * spatial;
    const int c0 = g * cpg;
    const int base_n = n * C * spatial;

    float sum = 0.0f;
    float sumsq = 0.0f;

    for (int i = tid; i < elems; i += blockDim.x) {
        int cc = i / spatial;
        int s = i - cc * spatial;
        float v = x[base_n + (c0 + cc) * spatial + s];
        sum += v;
        sumsq += v * v;
    }

    __shared__ float sh_sum[256];
    __shared__ float sh_sumsq[256];
    sh_sum[tid] = sum;
    sh_sumsq[tid] = sumsq;
    __syncthreads();

    // blockDim is 256
    for (int stride = 128; stride > 0; stride >>= 1) {
        if (tid < stride) {
            sh_sum[tid] += sh_sum[tid + stride];
            sh_sumsq[tid] += sh_sumsq[tid + stride];
        }
        __syncthreads();
    }

    float mean = sh_sum[0] / (float)elems;
    float var = sh_sumsq[0] / (float)elems - mean * mean;
    // avoid tiny negative variance from roundoff
    var = var < 0.0f ? 0.0f : var;
    float inv_std = rsqrtf(var + eps);

    for (int i = tid; i < elems; i += blockDim.x) {
        int cc = i / spatial;
        int s = i - cc * spatial;
        int c = c0 + cc;
        int idx = base_n + c * spatial + s;
        float v = (x[idx] - mean) * inv_std;
        if (affine) {
            v = v * weight[c] + bias[c];
        }
        // SiLU: v * sigmoid(v)
        float sig = 1.0f / (1.0f + expf(-v));
        y[idx] = v * sig;
    }
}
