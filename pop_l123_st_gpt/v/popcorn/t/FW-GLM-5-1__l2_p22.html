class Model(nn.Module):
    """
    Fused radius (ball) neighborhood query over a 3D point map with per-query feature
    aggregation—common in navigation / manipulation perception (PointNet++-style set
    abstraction, local maps around end-effector or goal queries on LiDAR / depth).

    Single forward fusion target:
      1) pairwise Euclidean distances between each query center and every point,
      2) membership in the closed ball of radius ``r``,
      3) channel-wise **max** pooling of point features over in-ball neighbors.

    Points with no neighbors inside the radius contribute a **zero** feature vector
    for that query (instead of ``-inf`` max).
    """

    def __init__(self, in_channels: int, radius: float):
        super().__init__()
        self.in_channels = in_channels
        self.register_buffer("radius_sq", torch.tensor(float(radius) ** 2))

    def forward(
        self,
        points: torch.Tensor,
        queries: torch.Tensor,
        features: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            points: (B, P, 3) point coordinates (e.g. world or sensor frame).
            queries: (B, Q, 3) ball centers (e.g. navigation waypoints, grasp samples).
            features: (B, P, C) per-point descriptors (geometry or lifted RGB-D).
        Returns:
            (B, Q, C) max-pooled features over neighbors with ||p - q||_2 <= r.
        """
        B, P, _ = points.shape
        _, Q, _ = queries.shape
        _, P2, C = features.shape
        assert P == P2 and C == self.in_channels

        # Squared distances (B, Q, P)
        dist_sq = torch.cdist(queries, points, p=2.0) ** 2
        in_ball = dist_sq <= self.radius_sq

        # (B, 1, P, C) broadcasts with (B, Q, P) for masked max over P
        feats = features.unsqueeze(1)
        neg_large = torch.finfo(features.dtype).min
        masked = torch.where(
            in_ball.unsqueeze(-1),
            feats,
            torch.full_like(feats, neg_large),
        )
        pooled, _ = masked.max(dim=2)

        empty = in_ball.sum(dim=2) == 0
        if empty.any():
            pooled = pooled.masked_fill(empty.unsqueeze(-1), 0.0)

        return pooled

            float feat_val = __ldg(&features[feat_idx]);
            max_val = fmaxf(max_val, feat_val);
            has_neighbor = true;
        }
    }
    
    // Write result: zero if no neighbors, max value otherwise
    output[(b_idx * Q + q_idx) * C + c_idx] = has_neighbor ? max_val : 0.0f;
}

// Host wrapper function
torch::Tensor ball_query_maxpool_cuda(
    torch::Tensor points,
    torch::Tensor queries,
    torch::Tensor features,
    float radius_sq
) {
    // Get dimensions
    int B = points.size(0);
    int P = points.size(1);
    int Q = queries.size(1);
    int C = features.size(2);
    
    // Allocate output tensor
    auto output = torch::zeros({B, Q, C}, features.options());
    
    // Launch configuration
    int num_blocks = B * Q;
    int threads_per_block = C;
    
    // Ensure threads_per_block is sufficient
    if (threads_per_block < 32) {
        threads_per_block = 32;
    }
    
    dim3 grid_dim(num_blocks, 1, 1);
    dim3 block_dim(threads_per_block, 1, 1);
    
    ball_query_maxpool_kernel<<<grid_dim, block_dim>>>(
        points.data_ptr<float>(),
        queries.data_ptr<float>(),
        features.data_ptr<float>(),
        output.data_ptr<float>(),
        B, P, Q, C, radius_sq
    );
    
    // Check for launch errors
    cudaError_t err = cudaGetLastError();
    if (err != cudaSuccess) {
        throw std::runtime_error(cudaGetErrorString(err));
    }
    
    return output;
}

class ModelNew(nn.Module):
    """
    Optimized CUDA implementation of radius neighborhood query with max pooling.
    
    Each thread handles one channel of one query, iterating over all points to:
    1. Compute squared distance from query to point
    2. Update channel max if point is within radius
    3. Output zero if no neighbors found (instead of -inf)
    """
    
    def __init__(self, in_channels: int, radius: float):
        super().__init__()
        self.in_channels = in_channels
        self.radius_sq = float(radius) ** 2
    
    def forward(
        self,
        points: torch.Tensor,
        queries: torch.Tensor,
        features: torch.Tensor,
    ) -> torch.Tensor:
        """
        Args:
            points: (B, P, 3) point coordinates
            queries: (B, Q, 3) ball centers
            features: (B, P, C) per-point descriptors
        Returns:
            (B, Q, C) max-pooled features over neighbors within radius
        """
        B, P, _ = points.shape
        _, Q, _ = queries.shape
        _, P2, C = features.shape
        
        # Assert shapes match expected dimensions
        assert P == P2
        assert C == self.in_channels
        
        # Ensure contiguous memory layout for efficient GPU access
        points = points.contiguous()
        queries = queries.contiguous()
        features = features.contiguous()
        
        # Call the CUDA kernel
        return ball_query_ext.ball_query_maxpool_cuda(
            points, queries, features, self.radius_sq
        )
