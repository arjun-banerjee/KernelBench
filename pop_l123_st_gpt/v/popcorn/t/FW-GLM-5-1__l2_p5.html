# Load the CUDA extension
module = load_inline(
    name='dilated_conv_extension',
    cpp_sources=cpp_source,
    cuda_sources=cuda_source,
    functions=['layernorm_gelu_cuda', 'dilated_conv1d_residual_cuda'],
    extra_cuda_cflags=['-O3', '-use_fast_math'],
    verbose=False
)


class ModelNew(nn.Module):
    """
    Pre-norm 1D dilated convolution block: LayerNorm, GELU activation, dilated Conv1d, and residual addition.
    Optimized with custom CUDA kernels.
    """
    def __init__(self, channels, seq_len, kernel_size, dilation):
        super(ModelNew, self).__init__()
        
        # LayerNorm parameters
        self.channels = channels
        self.seq_len = seq_len
        
        # LayerNorm weights (gamma and beta)
        self.norm_weight = nn.Parameter(torch.ones(channels, seq_len))
        self.norm_bias = nn.Parameter(torch.zeros(channels, seq_len))
        
        # Conv1d weights
        self.conv_weight = nn.Parameter(torch.empty(channels, channels, kernel_size))
        self.conv_bias = nn.Parameter(torch.zeros(channels))
        self.dilation = dilation
        self.kernel_size = kernel_size
        
        # Initialize weights like Conv1d would
        nn.init.kaiming_uniform_(self.conv_weight, a=math.sqrt(5))
        
        # Register the loaded module
        self.cuda_module = module
    
    def forward(self, x):
        residual = x
        
        # Allocate output tensor for LayerNorm + GELU
        x_norm_gelu = torch.empty_like(x)
        
        # Fused LayerNorm + GELU
        self.cuda_module.layernorm_gelu_cuda(
            x, x_norm_gelu, self.norm_weight, self.norm_bias
        )
        
        # Allocate output tensor for Conv1d + Residual
        output = torch.empty_like(x)
        
        # Fused Dilated Conv1d + Residual Add
        self.cuda_module.dilated_conv1d_residual_cuda(
            x_norm_gelu, residual, output, self.conv_weight, self.conv_bias, self.dilation
        )
        
        return output
