import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Grounded top-2 MoE router.

    Router logits are biased by token-expert similarity against learned
    grounding embeddings, then normalized over the selected top-2 experts.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        token_hidden: torch.Tensor,
        router_logits: torch.Tensor,
        expert_ground: torch.Tensor,
        alpha: float,
    ) -> torch.Tensor:
        grounded = router_logits + alpha * (token_hidden @ expert_ground.t())
        top_vals, top_idx = torch.topk(grounded, k=2, dim=-1)
        top_weights = torch.softmax(top_vals, dim=-1)
        return torch.stack((top_idx.to(torch.float32), top_weights), dim=-1)


num_tokens = 4096
hidden_dim = 128
num_experts = 16
alpha = 0.35


def get_inputs():
    token_hidden = torch.randn(num_tokens, hidden_dim, dtype=torch.float32)
    router_logits = torch.randn(num_tokens, num_experts, dtype=torch.float32)
    expert_ground = torch.randn(num_experts, hidden_dim, dtype=torch.float32)
    return [token_hidden, router_logits, expert_ground, alpha]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor grounded_router_forward(torch::Tensor token_hidden,
                                      torch::Tensor router_logits,
                                      torch::Tensor expert_ground,
                                      double alpha);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <ATen/cuda/CUDAContext.h>
#include <cfloat>

// One block computes one token.  Four warps in the block compute the 16
// expert scores (four experts per warp, one at a time).  The small final
// top-2/softmax is done by a single thread in the same block.
__global__ __launch_bounds__(128, 4)
void grounded_router_kernel(const float* __restrict__ token_hidden,
                            const float* __restrict__ router_logits,
                            const float* __restrict__ expert_ground,
                            float* __restrict__ out,
                            int num_tokens,
                            float alpha) {
    const int token = blockIdx.x;
    const int tid = threadIdx.x;
    const int lane = tid & 31;
    const int warp = tid >> 5;       // 0..3

    __shared__ float scores[16];

    const float* __restrict__ th = token_hidden + ((long long)token) * 128;
    const float* __restrict__ rl = router_logits + ((long long)token) * 16;

    #pragma unroll
    for (int pass = 0; pass < 4; ++pass) {
        const int e = (pass << 2) + warp;
        const float* __restrict__ eg = expert_ground + e * 128;

        float sum = 0.0f;
        // hidden_dim is 128: each lane accumulates four products.
        #pragma unroll
        for (int k = 0; k < 4; ++k) {
            const int h = lane + (k << 5);
            sum = fmaf(th[h], eg[h], sum);
        }

        // Warp reduction.
        #pragma unroll
        for (int offset = 16; offset > 0; offset >>= 1) {
            sum += __shfl_down_sync(0xffffffff, sum, offset);
        }

        if (lane == 0) {
            scores[e] = rl[e] + alpha * sum;
        }
    }

    __syncthreads();

    if (tid == 0) {
        float best1 = -FLT_MAX;
        float best2 = -FLT_MAX;
        int idx1 = 0;
        int idx2 = 0;

        #pragma unroll
        for (int e = 0; e < 16; ++e) {
            float v = scores[e];
            if (v > best1) {
                best2 = best1;
                idx2 = idx1;
                best1 = v;
                idx1 = e;
            } else if (v > best2) {
                best2 = v;
                idx2 = e;
            }
        }

        // Stable 2-way softmax. Since best1 >= best2, exp(best2-best1) <= 1.
        float r = __expf(best2 - best1);
        float w1 = 1.0f / (1.0f + r);
        float w2 = r * w1;

        float* __restrict__ o = out + ((long long)token) * 4;
        o[0] = (float)idx1;
        o[1] = w1;
        o[2] = (float)idx2;
        o[3] = w2;
    }
}

torch::Tensor grounded_router_forward(torch::Tensor token_hidden,
                                      torch::Tensor router_logits,
                                      torch::Tensor expert_ground,
                                      double alpha) {
    const int num_tokens = (int)token_hidden.size(0);
    auto out = torch::empty({num_tokens, 2, 2}, token_hidden.options());

    const dim3 block(128);
    const dim3 grid(num_tokens);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    grounded_router_kernel<<<grid, block, 0, stream>>>(
        token_hidden.data_ptr<float>(),
        router_logits.data_ptr<float>(),
        expert_ground.data_ptr<float>(),
        out.data_ptr<float>(),
        num_tokens,
        (float)alpha);
    return out;
}
"""
