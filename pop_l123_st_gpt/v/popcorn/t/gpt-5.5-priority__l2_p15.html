import math

import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Causal grouped-query attention with KV-head expansion.

    Query heads outnumber KV heads. Keys and values are expanded by repeating
    each KV head across its assigned query-head group before causal attention.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_q_heads, head_dim = q.shape
        num_kv_heads = k.shape[2]
        group_size = num_q_heads // num_kv_heads

        expanded_k = k.repeat_interleave(group_size, dim=2)
        expanded_v = v.repeat_interleave(group_size, dim=2)

        out = torch.zeros_like(q)
        scale = 1.0 / math.sqrt(head_dim)

        for b in range(batch_size):
            for h in range(num_q_heads):
                for t in range(seq_len):
                    q_t = q[b, t, h]
                    k_hist = expanded_k[b, : t + 1, h]
                    v_hist = expanded_v[b, : t + 1, h]
                    scores = (k_hist * q_t.unsqueeze(0)).sum(dim=-1) * scale
                    attn = torch.softmax(scores, dim=0)
                    out[b, t, h] = (attn.unsqueeze(-1) * v_hist).sum(dim=0)

        return out

// One CUDA block computes one (batch, query head, query time) output vector.
__global__ __launch_bounds__(128, 4)
void causal_gqa_attention_kernel(const float* __restrict__ q,
                                 const float* __restrict__ k,
                                 const float* __restrict__ v,
                                 float* __restrict__ out) {
    constexpr int B = 2;
    constexpr int T = 80;
    constexpr int Hq = 8;
    constexpr int Hkv = 2;
    constexpr int D = 32;
    constexpr int GROUP = Hq / Hkv;
    constexpr float SCALE = 0.17677669529663687f; // 1/sqrt(32)

    const int tid = threadIdx.x;
    const int idx = blockIdx.x;
    const int t = idx % T;
    const int h = (idx / T) % Hq;
    const int b = idx / (T * Hq);
    const int kvh = h / GROUP;

    __shared__ float scores[T];
    __shared__ float red[128];

    // Compute all causal scores.  Each thread owns one or more source positions.
    const float* qptr = q + (((b * T + t) * Hq + h) * D);
    for (int s = tid; s <= t; s += blockDim.x) {
        const float* kptr = k + (((b * T + s) * Hkv + kvh) * D);
        float dot = 0.0f;
        #pragma unroll
        for (int d = 0; d < D; ++d) {
            dot = fmaf(qptr[d], kptr[d], dot);
        }
        scores[s] = dot * SCALE;
    }
    __syncthreads();

    // Block reduction for max(score).
    float local_max = -CUDART_INF_F;
    for (int s = tid; s <= t; s += blockDim.x) {
        local_max = fmaxf(local_max, scores[s]);
    }
    red[tid] = local_max;
    __syncthreads();
    for (int stride = 64; stride > 0; stride >>= 1) {
        if (tid < stride) red[tid] = fmaxf(red[tid], red[tid + stride]);
        __syncthreads();
    }
    const float m = red[0];

    // Exponentiate scores in-place and reduce denominator.
    float local_sum = 0.0f;
    for (int s = tid; s <= t; s += blockDim.x) {
        float e = expf(scores[s] - m);
        scores[s] = e;
        local_sum += e;
    }
    red[tid] = local_sum;
    __syncthreads();

        if (tid < stride) red[tid] += red[tid + stride];
        __syncthreads();
    }
    const float inv_sum = 1.0f / red[0];

}
