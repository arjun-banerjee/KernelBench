import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Fused sparse attention-value aggregation in CSR format.

    For each destination row, compute a softmax over edge logits and then use
    the normalized weights to aggregate source node values. This is a common
    fused graph-attention pattern.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        row_ptr: torch.Tensor,
        col_idx: torch.Tensor,
        edge_scores: torch.Tensor,
        node_value: torch.Tensor,
    ) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        feat_dim = node_value.shape[1]
        out = torch.zeros(num_nodes, feat_dim, dtype=node_value.dtype, device=node_value.device)

        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end > start:
                weights = torch.softmax(edge_scores[start:end], dim=0)
                src = col_idx[start:end].long()
                out[dst] = (node_value[src] * weights.unsqueeze(-1)).sum(dim=0)

        return out

// One CTA computes one destination row.  The benchmark problem has degree <= 21
// and feature dimension 64, so this maps naturally to a small amount of shared
// softmax work followed by one thread per output feature.
__global__ void csr_attn_value_kernel(
    const int* __restrict__ row_ptr,
    const int* __restrict__ col_idx,
    const float* __restrict__ edge_scores,
    const float* __restrict__ node_value,
    float* __restrict__ out,
    int num_nodes,
    int feat_dim)
{
    int dst = blockIdx.x;
    int tid = threadIdx.x;
    if (dst >= num_nodes) return;

    int start = row_ptr[dst];
    int end   = row_ptr[dst + 1];
    int deg   = end - start;

    // Enough for the supplied CSR rows (max degree 21).  If a larger row ever
    // appears, the strided loops below still compute the output directly after
    // using the first 1024 entries of shared scratch only for the normal case.
    __shared__ float smem[1024];
    __shared__ float smax;
    __shared__ float ssum;

    if (deg <= 0) {
        for (int f = tid; f < feat_dim; f += blockDim.x) {
            out[dst * feat_dim + f] = 0.0f;
        }
        return;
    }

    // Reduce max score across the row.
    float local_max = -FLT_MAX;
    for (int e = tid; e < deg; e += blockDim.x) {
        float v = edge_scores[start + e];
        local_max = fmaxf(local_max, v);
    }
    smem[tid] = local_max;
    __syncthreads();

    // blockDim is 128 in the launcher.
    if (tid < 64) smem[tid] = fmaxf(smem[tid], smem[tid + 64]);
    __syncthreads();
    if (tid < 32) {
        float v = smem[tid];
        v = fmaxf(v, smem[tid + 32]);
        v = fmaxf(v, smem[tid + 16]);
        v = fmaxf(v, smem[tid + 8]);
        v = fmaxf(v, smem[tid + 4]);
        v = fmaxf(v, smem[tid + 2]);
        v = fmaxf(v, smem[tid + 1]);
        if (tid == 0) smax = v;
    }
    __syncthreads();

    // Exponentiate once and keep normalized numerator in shared memory.
    float local_sum = 0.0f;
    for (int e = tid; e < deg; e += blockDim.x) {
        float w = __expf(edge_scores[start + e] - smax);
        if (e < 1024) smem[e] = w;
        local_sum += w;
    }
    __syncthreads();
    smem[tid] = local_sum;
    __syncthreads();

    if (tid < 64) smem[tid] += smem[tid + 64];
    __syncthreads();
    if (tid < 32) {
        float v = smem[tid];
        v += smem[tid + 32];
        v += smem[tid + 16];
        v += smem[tid + 8];
        v += smem[tid + 4];
        v += smem[tid + 2];
        v += smem[tid + 1];
        if (tid == 0) ssum = v;
    }
    __syncthreads();

    float inv_sum = 1.0f / ssum;

    // Compute aggregation.  For the target feat_dim=64, the first half-block
    // produces one output element each; this is coalesced over feature columns.
    for (int f = tid; f < feat_dim; f += blockDim.x) {
        float acc = 0.0f;
        for (int e = 0; e < deg; ++e) {
            float w;
            if (e < 1024) {
                w = smem[e] * inv_sum;
            } else {
                w = __expf(edge_scores[start + e] - smax) * inv_sum;
            }
            int src = col_idx[start + e];
            acc += w * node_value[src * feat_dim + f];
        }
        out[dst * feat_dim + f] = acc;
    }
}
