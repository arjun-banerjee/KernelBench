import torch.nn as nn

    name='fused_linear_kernel',

#include <torch/extension.h>
#include <cuda_runtime.h>

#define BLOCK_SIZE 256

__global__ void linear_kernel(
    const float* __restrict__ x,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ out,
    int batch_size,
    int in_features,
    int out_features
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * out_features;
    
    if (idx >= total) return;
    
    int batch_idx = idx / out_features;
    int out_feature_idx = idx % out_features;
    
    float sum = 0.0f;
    
    const float* x_ptr = x + batch_idx * in_features;
    const float* w_ptr = weight + out_feature_idx * in_features;
    
    #pragma unroll 4
    for (int k = 0; k < in_features; k++) {
        sum += x_ptr[k] * w_ptr[k];
    }
    
    out[idx] = sum + bias[out_feature_idx];
}
