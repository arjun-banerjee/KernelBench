import math
import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Graph-Fused VLA (GF-VLA) style **scene graph relation update** in one forward.

    Objects are nodes with vision-language aligned embeddings; directed pairs
    (i → j) carry pairwise conditioning (geometry, language grounding, or fused
    VLA features). This reference fuses:

      LayerNorm(nodes) → multi-head attention over objects with **pairwise
      relation MLP bias** on edges → residual write-back.

    Intended as a fusion target for navigation / manipulation stacks that maintain
    explicit scene graphs and update them from multimodal observations.
    """

    def __init__(self, dim: int, pair_dim: int, num_heads: int):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.pair_dim = pair_dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        self.scale = 1.0 / math.sqrt(self.head_dim)
        self.norm = nn.LayerNorm(dim)
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.rel_mlp = nn.Sequential(nn.Linear(pair_dim, pair_dim), nn.SiLU(), nn.Linear(pair_dim, num_heads))
        self.out_proj = nn.Linear(dim, dim, bias=False)

    def forward(self, nodes: torch.Tensor, pair_feats: torch.Tensor, edge_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            nodes: (B, N, D) object / region embeddings.
            pair_feats: (B, N, N, P) pairwise VLA or geometric conditioning.
            edge_mask: (B, N, N) float in {0, 1}; 1 keeps attention over column j.
        Returns:
            (B, N, D) updated node embeddings (residual around attention block).
        """
        (B, N, D) = nodes.shape
        (_, N2, N3, P) = pair_feats.shape
        assert N == N2 == N3 and P == self.pair_dim
        h = self.norm(nodes)
        q = self.q_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(h).view(B, N, self.num_heads, self.head_dim).transpose(1, 2)
        logits = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        rel_bias = self.rel_mlp(pair_feats).permute(0, 3, 1, 2)
        logits = logits + rel_bias
        logits = logits.masked_fill(edge_mask.unsqueeze(1) < 0.5, float('-inf'))
        attn = F.softmax(logits, dim=-1)
        ctx = torch.matmul(attn, v).transpose(1, 2).reshape(B, N, D)
        return nodes + self.out_proj(ctx)

    if (tid == 0) {
        s_var[0] = local_var / dim;
    }
    __syncthreads();
    
    float var = s_var[0];
    float inv_std = rsqrtf(var + eps);
    
    // Apply LayerNorm and store normalized values
    for (int i = tid; i < dim; i += blockDim.x) {
        s_xnorm[i] = (x[i] - mean) * inv_std * norm_weight[i] + norm_bias[i];
    }
    __syncthreads();
    
    // Compute Q, K, V projections with transpose for attention
    // Output layout: [batch, num_heads, N, head_dim]
    for (int h = 0; h < num_heads; h++) {
        for (int d = tid; d < head_dim; d += blockDim.x) {
            int out_idx = ((b * num_heads + h) * gridDim.y + n) * head_dim + d;
            int in_start = h * head_dim;
            
            float q_val = 0.0f, k_val = 0.0f, v_val = 0.0f;
            for (int i = 0; i < dim; i++) {
                float xnorm_val = s_xnorm[i];
                q_val += xnorm_val * q_weight[i * dim + in_start + d];
                k_val += xnorm_val * k_weight[i * dim + in_start + d];
                v_val += xnorm_val * v_weight[i * dim + in_start + d];
            }
            q_out[out_idx] = q_val;
            k_out[out_idx] = k_val;
            v_out[out_idx] = v_val;
        }
    }
}

// Fused attention logits computation: logits = Q @ K^T * scale + rel_bias
__global__ void attention_logits_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ rel_bias,
    const float* __restrict__ edge_mask,
    float* __restrict__ logits,
    int N, int head_dim, float scale) {
    
    int b = blockIdx.z;  // batch
    int h = blockIdx.y;  // head
    int q_idx = blockIdx.x;  // query index
    
    int k_idx = threadIdx.x;
    
    if (b >= gridDim.z || h >= gridDim.y || q_idx >= gridDim.x) return;
    
    int batch_offset_q = b * gridDim.y * N * head_dim + h * N * head_dim;
    int batch_offset_k = b * gridDim.y * N * head_dim + h * N * head_dim;
    
    // Compute Q @ K^T for this query position
    // Each thread computes one element (query_idx, key_idx)
    float qk = 0.0f;
    const float* q_ptr = q + batch_offset_q + q_idx * head_dim;
    
    for (int d = 0; d < head_dim; d++) {
        const float* k_ptr = k + batch_offset_k + k_idx * head_dim;
        qk += q_ptr[d] * k_ptr[d];
    }
    
    float val = qk * scale;
    
    // Add relative bias
    // rel_bias layout: [B, N, N, num_heads] after MLP
    val += rel_bias[((b * N + q_idx) * N + k_idx) * gridDim.y + h];
    
    // Apply edge mask
    float mask = edge_mask[b * N * N + q_idx * N + k_idx];
    if (mask < 0.5f) {
        val = -INFINITY;
    }
    
    logits[b * gridDim.y * N * N + h * N * N + q_idx * N + k_idx] = val;
}

// Fused Softmax + Context computation
__global__ void softmax_context_kernel(
    const float* __restrict__ logits,
    const float* __restrict__ v,
    float* __restrict__ ctx,
    int N, int head_dim, int num_heads) {
    
    int b = blockIdx.x;  // batch
    int h = blockIdx.y;  // head
    int q_idx = blockIdx.z;  // query index
    
    if (b >= gridDim.x || h >= gridDim.y || q_idx >= gridDim.z) return;
    
    int tid = threadIdx.x;
    int logits_offset = b * num_heads * N * N + h * N * N + q_idx * N;
    int v_offset = b * num_heads * N * head_dim + h * N * head_dim;
    
    // Find max for numerical stability
    __shared__ float s_max[1];
    __shared__ float s_sum[1];
    __shared__ float s_probs[32];  // max N=24
    
    float local_max = -INFINITY;
    for (int k_idx = tid; k_idx < N; k_idx += blockDim.x) {
        float val = logits[logits_offset + k_idx];
        if (val > local_max) local_max = val;
    }
    
    // Warp reduce for max
    for (int offset = 16; offset > 0; offset >>= 1) {
        float other = __shfl_down_sync(0xffffffff, local_max, offset);
        if (other > local_max) local_max = other;
    }
    
    if (tid == 0) {
        s_max[0] = local_max;
    }
    __syncthreads();
    
    float max_val = s_max[0];
    
    // Compute exp and sum
    float local_sum = 0.0f;
    for (int k_idx = tid; k_idx < N; k_idx += blockDim.x) {
        float val = logits[logits_offset + k_idx];
        if (isfinite(val)) {
            float exp_val = expf(val - max_val);
            s_probs[k_idx] = exp_val;
            local_sum += exp_val;
        } else {
            s_probs[k_idx] = 0.0f;
        }
    }
    
    // Warp reduce for sum
    for (int offset = 16; offset > 0; offset >>= 1) {
        local_sum += __shfl_down_sync(0xffffffff, local_sum, offset);
    }
    
    if (tid == 0) {
        s_sum[0] = local_sum;
    }
    __syncthreads();
    
    float sum_val = s_sum[0];
    float inv_sum = 1.0f / sum_val;
    
    // Normalize and store probabilities
    for (int k_idx = tid; k_idx < N; k_idx += blockDim.x) {
        s_probs[k_idx] *= inv_sum;
    }
    __syncthreads();
    
    // Compute context: weighted sum of V
    // ctx layout: [B, num_heads, N, head_dim]
    for (int d = tid; d < head_dim; d += blockDim.x) {
        float sum = 0.0f;
        for (int k_idx = 0; k_idx < N; k_idx++) {
            float prob = s_probs[k_idx];
            sum += prob * v[v_offset + k_idx * head_dim + d];
        }
        ctx[b * num_heads * N * head_dim + h * N * head_dim + q_idx * head_dim + d] = sum;
    }
}

// Output projection + residual kernel
__global__ void output_proj_residual_kernel(
    const float* __restrict__ ctx,
    const float* __restrict__ nodes,
    const float* __restrict__ weight,
    float* __restrict__ out,
    int N, int num_heads, int head_dim, int dim) {
    
    int b = blockIdx.x;  // batch
    int n = blockIdx.y;  // object index
    
    if (b >= gridDim.x || n >= gridDim.y) return;
    
    int tid = threadIdx.x;
    
    // ctx layout: [B, num_heads, N, head_dim]
    // Reshape to [B, N, D]
    
    for (int d = tid; d < dim; d += blockDim.x) {
        // Gather context value from multi-head layout
        int h = d / head_dim;
        int hd = d % head_dim;
        float ctx_val = ctx[b * num_heads * N * head_dim + h * N * head_dim + n * head_dim + hd];
        
        // Compute output projection
        float proj_sum = 0.0f;
        for (int i = 0; i < dim; i++) {
            int h_i = i / head_dim;
            int hd_i = i % head_dim;
            float ctx_i = ctx[b * num_heads * N * head_dim + h_i * N * head_dim + n * head_dim + hd_i];
            proj_sum += ctx_i * weight[i * dim + d];
        }
        
        // Add residual
        float node_val = nodes[b * N * dim + n * dim + d];
        out[b * N * dim + n * dim + d] = node_val + proj_sum;
    }
}

// Fused Relational MLP kernel: computes bias from pair features
__global__ void rel_mlp_kernel(
    const float* __restrict__ pair_feats,
    const float* __restrict__ w1,
    const float* __restrict__ b1,
    const float* __restrict__ w2,
    const float* __restrict__ b2,
    float* __restrict__ bias,
    int B, int N, int pair_dim, int num_heads) {
    
    int b = blockIdx.x;
    int i = blockIdx.y;
    int j = blockIdx.y;
    
    if (b >= gridDim.x || i >= gridDim.y) return;
    
    int tid = threadIdx.x;
    __shared__ float hidden[32];  // max pair_dim
    
    // First linear + SiLU
    for (int p = tid; p < pair_dim; p += blockDim.x) {
        float sum = b1[p];
        for (int pp = 0; pp < pair_dim; pp++) {
            sum += pair_feats[((b * N + i) * N + j) * pair_dim + pp] * w1[pp * pair_dim + p];
        }
        hidden[p] = silu(sum);
    }
    __syncthreads();
    
    // Second linear
    for (int h = tid; h < num_heads; h += blockDim.x) {
        float sum = b2[h];
        for (int p = 0; p < pair_dim; p++) {
            sum += hidden[p] * w2[p * num_heads + h];
        }
        bias[((b * N + i) * N + j) * num_heads + h] = sum;
    }
}
