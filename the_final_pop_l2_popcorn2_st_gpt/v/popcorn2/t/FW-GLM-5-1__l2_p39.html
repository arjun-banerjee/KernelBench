import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        assert kernel_size % 2 == 1, 'odd kernel_size keeps T unchanged with symmetric padding'
        assert dilation >= 1
        self.padding = (kernel_size - 1) * dilation // 2
        self.dw = nn.Conv1d(in_channels, in_channels, kernel_size, padding=self.padding, dilation=dilation, groups=in_channels, bias=False)
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

class ModelNew(nn.Module):
    """KernelBench candidate entry point; default delegates to ``Model``."""

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        self._impl = Model(in_channels, out_channels, kernel_size, dilation=dilation)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self._impl(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Define the CUDA kernel and C++ wrapper
cuda_source = r"""
#include <cuda_fp16.h>
#include <torch/extension.h>

__device__ __forceinline__ float sigmoid_func(float x) {
    return 1.0f / (1.0f + expf(-x));
}

__device__ __forceinline__ float silu_func(float x) {
    return x * sigmoid_func(x);
}

__global__ void fused_conv_kernel(
    const float* __restrict__ x,
    const float* __restrict__ dw_weight,
    const float* __restrict__ pw_weight,
    float* __restrict__ y,
    int B, int C_in, int C_out, int T, int kernel_size, int dilation, int padding
) {
    int b = blockIdx.x / T;
    int t_out = blockIdx.x % T;
    
    extern __shared__ float silu_out[];
    
    // Step 1: Cooperative depthwise + SiLU
    // All threads cooperate to compute depthwise conv for all input channels
    for (int c_in_base = 0; c_in_base < C_in; c_in_base += blockDim.x) {
        int c_in = c_in_base + threadIdx.x;
        float dw_out = 0.0f;
        if (c_in < C_in) {
            for (int k = 0; k < kernel_size; k++) {
                int t_in = t_out - padding + k * dilation;
                float in_val = (t_in >= 0 && t_in < T) ? 
                    __ldg(&x[b * C_in * T + c_in * T + t_in]) : 0.0f;
                // dw_weight shape: (C_in, 1, kernel_size), stride (kernel_size, kernel_size, 1)
                dw_out += __ldg(&dw_weight[c_in * kernel_size + k]) * in_val;
            }
            silu_out[c_in] = silu_func(dw_out);
        }
        __syncthreads();
    }
    
    // Step 2: Pointwise conv (each thread may handle multiple output channels)
    for (int c_out_base = 0; c_out_base < C_out; c_out_base += blockDim.x) {
        int c_out = c_out_base + threadIdx.x;
        if (c_out < C_out) {

            y[b * C_out * T + c_out * T + t_out] = pw_out;
        }
    }
}

    torch::Tensor x,

class ModelNew(nn.Module):
    """
    Fused depthwise-pointwise 1D temporal convolution with SiLU activation.
    
    Fuses:
    1. Temporal depthwise k-tap conv (per-channel along time)
    2. SiLU activation
    3. Pointwise 1x1 channel mixing
    
    This avoids intermediate tensor materialization and memory bandwidth.
    """

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int = 1):
        super().__init__()
        assert kernel_size % 2 == 1, 'odd kernel_size keeps T unchanged with symmetric padding'
        assert dilation >= 1
        self.padding = (kernel_size - 1) * dilation // 2
        self.kernel_size = kernel_size
        self.dilation = dilation

        self.dw = nn.Conv1d(in_channels, in_channels, kernel_size, 
                           padding=self.padding, dilation=dilation, groups=in_channels, bias=False)
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T) fused convolution output.
        """

        # dw.weight shape: (in_channels, 1, kernel_size)
        # pw.weight shape: (out_channels, in_channels, 1)
