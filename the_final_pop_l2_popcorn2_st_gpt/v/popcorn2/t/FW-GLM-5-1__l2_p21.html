        if (tid < s) reduction[tid] += reduction[tid + s];
        __syncthreads();
    }
    float mean = reduction[0] / hidden_dim;
    
    float var_sum = 0.0f;
    for (int h = tid; h < hidden_dim; h += blockDim.x) {
        float diff = einsum_vals[h] - mean;
        var_sum += diff * diff;
    }
    reduction[tid] = var_sum;
    __syncthreads();
    for (int s = blockDim.x / 2; s > 0; s >>= 1) {
        if (tid < s) reduction[tid] += reduction[tid + s];
        __syncthreads();
    }
    float inv_std = rsqrtf(reduction[0] / hidden_dim + 1e-5f);
    
    for (int h = tid; h < hidden_dim; h += blockDim.x) {
        ln_out[h] = (einsum_vals[h] - mean) * inv_std * ln2_w[h] + ln2_b[h];
    }
    __syncthreads();
    
    // Output projection and gating
    int out_offset = b * N * N * pair_dim + n1 * N * pair_dim + n2 * pair_dim;
    int pair_offset = out_offset;
    
    for (int d = tid; d < pair_dim; d += blockDim.x) {
        float proj_val = out_proj_b[d];
        for (int h = 0; h < hidden_dim; h++) {
            proj_val += ln_out[h] * out_proj_w[h * pair_dim + d];
        }
        
        float gate_val = out_gate_b[d];
        for (int p = 0; p < pair_dim; p++) {
            gate_val += pair_repr[pair_offset + p] * out_gate_w[p * pair_dim + d];
        }
        
        float sigmoid_gate = 1.0f / (1.0f + expf(-gate_val));
        final_out[out_offset + d] = sigmoid_gate * proj_val;
    }
}

class ModelNew(nn.Module):
    """
    Triangular multiplicative update - incoming edges.
    Optimized CUDA implementation.
    """

    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)
        
        self.cuda_op = module.triangular_multiplicative_update_cuda

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            pair_repr: (B, N, N, pair_dim)
            pair_mask: (B, N, N)
        Returns:
            updated pair representation (B, N, N, pair_dim)
        """
        # Ensure contiguous tensors
        pair_repr = pair_repr.contiguous()
        pair_mask = pair_mask.contiguous()
        
        # Get weights and biases
        ln1_w = self.layer_norm_in.weight.contiguous()
        ln1_b = self.layer_norm_in.bias.contiguous()
        
        # Linear weights are stored as (out_features, in_features) in PyTorch
        # We need them as (in_features, out_features) for our kernel
        proj_a_w = self.proj_a.weight.t().contiguous()
        proj_a_b = self.proj_a.bias.contiguous()
        gate_a_w = self.gate_a.weight.t().contiguous()
        gate_a_b = self.gate_a.bias.contiguous()
        
        proj_b_w = self.proj_b.weight.t().contiguous()
        proj_b_b = self.proj_b.bias.contiguous()
        gate_b_w = self.gate_b.weight.t().contiguous()
        gate_b_b = self.gate_b.bias.contiguous()
        
        ln2_w = self.layer_norm_out.weight.contiguous()
        ln2_b = self.layer_norm_out.bias.contiguous()
        
        out_proj_w = self.out_proj.weight.t().contiguous()
        out_proj_b = self.out_proj.bias.contiguous()
        out_gate_w = self.out_gate.weight.t().contiguous()
        out_gate_b = self.out_gate.bias.contiguous()
        
        result = self.cuda_op(
            pair_repr, pair_mask,
            ln1_w, ln1_b,
            proj_a_w, proj_a_b,
            gate_a_w, gate_a_b,
            proj_b_w, proj_b_b,
            gate_b_w, gate_b_b,
            ln2_w, ln2_b,
            out_proj_w, out_proj_b,
            out_gate_w, out_gate_b
        )
        
        return result[0]
