_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__device__ __forceinline__ float gelu_exact_f32(float v) {
    return 0.5f * v * (1.0f + erff(v * 0.707106781186547524400844362104849f));
}

__global__ void sum_gelu_r4_kernel(const float* __restrict__ x,
                                   float* __restrict__ out,
                                   int n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) {
        const float* p0 = x + i;
        float acc = gelu_exact_f32(p0[0]);
        acc += gelu_exact_f32(p0[n]);
        acc += gelu_exact_f32(p0[2 * n]);
        acc += gelu_exact_f32(p0[3 * n]);
        out[i] = acc;
    }
}

__global__ void sum_gelu_generic_kernel(const float* __restrict__ x,
                                        float* __restrict__ out,
                                        int R, int n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) {
        float acc = 0.0f;
        const float* p = x + i;
        #pragma unroll 1
        for (int r = 0; r < R; ++r) {
            acc += gelu_exact_f32(p[r * n]);
        }
        out[i] = acc;
    }
}

torch::Tensor sum_gelu_cuda(torch::Tensor x) {
    const int64_t R64 = x.size(0);
    const int64_t B64 = x.size(1);
    const int64_t D64 = x.size(2);
    auto out = torch::empty({B64, D64}, x.options());

    int n = (int)(B64 * D64);
    if (n == 0) return out;

    const int threads = 256;
    const int blocks = (n + threads - 1) / threads;
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    const float* xp = reinterpret_cast<const float*>(x.data_ptr<float>());
    float* op = reinterpret_cast<float*>(out.data_ptr<float>());

    if (R64 == 4) {
        sum_gelu_r4_kernel<<<blocks, threads, 0, stream>>>(xp, op, n);
    } else {
        sum_gelu_generic_kernel<<<blocks, threads, 0, stream>>>(xp, op, (int)R64, n);
    }
    return out;
}
'''
