import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Differentiable Smith-Waterman-style local sequence alignment scoring.
    Uses a soft-max relaxation of the DP recurrence so gradients can flow
    through the alignment matrix.  
    """

    def __init__(self, gap_open=-1.0, temperature=1.0):
        super().__init__()
        self.gap_open = gap_open
        self.temperature = temperature

    def forward(self, score_matrix: torch.Tensor) -> torch.Tensor:
        """
        Args:
            score_matrix: (B, M, N) – per-position similarity scores
        Returns:
            alignment_score: (B,) – soft local-alignment score (log-sum-exp over DP cells)
        """
        (B, M, N) = score_matrix.shape
        neg_inf = torch.tensor(float('-inf'), device=score_matrix.device, dtype=score_matrix.dtype)
        H = score_matrix.new_full((B, M + 1, N + 1), 0.0)
        for i in range(1, M + 1):
            for j in range(1, N + 1):
                match = H[:, i - 1, j - 1] + score_matrix[:, i - 1, j - 1]
                delete = H[:, i - 1, j] + self.gap_open
                insert = H[:, i, j - 1] + self.gap_open
                zero = torch.zeros(B, device=score_matrix.device)
                stack = torch.stack([zero, match, delete, insert], dim=-1)
                H[:, i, j] = self.temperature * torch.logsumexp(stack / self.temperature, dim=-1)
        return H.view(B, -1).max(dim=-1).values

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__device__ __forceinline__ float soft_cell(float match, float del, float ins, float gap, float temp) {
    float m = fmaxf(0.0f, fmaxf(match, fmaxf(del, ins)));
    float invt = 1.0f / temp;
    float s = expf((0.0f - m) * invt) + expf((match - m) * invt) + expf((del - m) * invt) + expf((ins - m) * invt);
    return m + temp * logf(s);
}

__global__ void sw32_kernel(const float* __restrict__ score,
                            float* __restrict__ out,
                            int B, int M, int N,
                            float gap, float temp) {
    int b = blockIdx.x;
    if (b >= B) return;
    int lane = threadIdx.x;  // launched with one warp
    int N1 = N + 1;
    int total = (M + 1) * (N + 1);

    extern __shared__ float H[];

    // Initialize DP matrix (including top row / left column) to zero.
    for (int idx = lane; idx < total; idx += 32) {
        H[idx] = 0.0f;
    }
    __syncwarp();

    const float* S = score + ((long long)b) * M * N;

    // Wavefront over anti-diagonals.  For the benchmark M=N=32, each diagonal
    // has at most 32 cells, so one warp computes a whole batch element.
    for (int diag = 2; diag <= M + N; ++diag) {
        int i_min = diag - N;
        if (i_min < 1) i_min = 1;
        int i_max = diag - 1;
        if (i_max > M) i_max = M;
        int len = i_max - i_min + 1;
        if (lane < len) {
            int i = i_min + lane;
            int j = diag - i;
            int idx = i * N1 + j;
            float match = H[(i - 1) * N1 + (j - 1)] + S[(i - 1) * N + (j - 1)];
            float delv  = H[(i - 1) * N1 + j] + gap;
            float insv  = H[i * N1 + (j - 1)] + gap;
            H[idx] = soft_cell(match, delv, insv, gap, temp);
        }
        __syncwarp();
    }

    float mx = -INFINITY;
    for (int idx = lane; idx < total; idx += 32) {
        float v = H[idx];
        mx = fmaxf(mx, v);
    }

    // Warp max reduction.
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        mx = fmaxf(mx, __shfl_down_sync(0xffffffff, mx, off));
    }
    if (lane == 0) out[b] = mx;
}

// Slightly more general fallback for dimensions whose anti-diagonal length is
// larger than one warp.  It is still a CUDA implementation, not a PyTorch fallback.
__global__ void sw_block_kernel(const float* __restrict__ score,
                                float* __restrict__ out,
                                int B, int M, int N,
                                float gap, float temp) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    int N1 = N + 1;
    int total = (M + 1) * (N + 1);
    extern __shared__ float H[];
    float* red = H + total;

    for (int idx = tid; idx < total; idx += blockDim.x) H[idx] = 0.0f;
    __syncthreads();

    const float* S = score + ((long long)b) * M * N;
    for (int diag = 2; diag <= M + N; ++diag) {
        int i_min = diag - N; if (i_min < 1) i_min = 1;
        int i_max = diag - 1; if (i_max > M) i_max = M;
        int len = i_max - i_min + 1;
        for (int k = tid; k < len; k += blockDim.x) {
            int i = i_min + k;
            int j = diag - i;
            int idx = i * N1 + j;
            float match = H[(i - 1) * N1 + (j - 1)] + S[(i - 1) * N + (j - 1)];
            float delv  = H[(i - 1) * N1 + j] + gap;
            float insv  = H[i * N1 + (j - 1)] + gap;
            H[idx] = soft_cell(match, delv, insv, gap, temp);
        }
        __syncthreads();
    }

    float mx = -INFINITY;
    for (int idx = tid; idx < total; idx += blockDim.x) mx = fmaxf(mx, H[idx]);
    red[tid] = mx;
    __syncthreads();
    for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) red[tid] = fmaxf(red[tid], red[tid + stride]);
        __syncthreads();
    }
    if (tid == 0) out[b] = red[0];
}

torch::Tensor sw_forward(torch::Tensor score, double gap_open, double temperature) {
    TORCH_CHECK(score.is_cuda(), "score_matrix must be a CUDA tensor");
    TORCH_CHECK(score.scalar_type() == at::kFloat, "score_matrix must be float32");
    TORCH_CHECK(score.dim() == 3, "score_matrix must have shape (B, M, N)");
    TORCH_CHECK(score.is_contiguous(), "score_matrix must be contiguous");

    int B = (int)score.size(0);
    int M = (int)score.size(1);
    int N = (int)score.size(2);
    auto out = torch::empty({B}, score.options());
    if (B == 0) return out;

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    float gap = (float)gap_open;
    float temp = (float)temperature;

        size_t shmem = (size_t)(M + 1) * (N + 1) * sizeof(float);
        sw32_kernel<<<B, 32, shmem, stream>>>(score.data_ptr<float>(), out.data_ptr<float>(), B, M, N, gap, temp);

        sw_block_kernel<<<B, threads, shmem, stream>>>(score.data_ptr<float>(), out.data_ptr<float>(), B, M, N, gap, temp);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
