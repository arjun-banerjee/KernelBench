        pw2_w = self.pw2_weight.squeeze(-1).contiguous()  # (channels, expanded_channels)
        pw2_b = self.pw2_bias.contiguous()  # (channels,)
        
        # Choose implementation based on sequence length
        # For large sequences, the fully fused kernel has too much redundant computation
        # because each depthwise position requires recomputing pointwise outputs
        
        if self.use_fused and seq_len <= 4096:
            # Use fully fused kernel
            output = torch.empty_like(x)
            
            total_elements = batch_size * self.channels * seq_len
            threads = 256
            blocks = (total_elements + threads - 1) // threads
            
            # Calculate shared memory size
            shared_mem = self.expanded_channels * sizeof_float
            
            depthwise_conv_ops.optimized_fused_kernel(
                blocks, threads,
                x.data_ptr(),
                pw1_w.data_ptr(),
                pw1_b.data_ptr(),
                dw_w.data_ptr(),
                dw_b.data_ptr(),
                pw2_w.data_ptr(),
                pw2_b.data_ptr(),
                output.data_ptr(),
                batch_size, self.channels, self.expanded_channels, seq_len,
                self.kernel_size, self.padding
            )
            
            return output
        else:
            # Use separate kernels (better for large sequences)
            # Allocate intermediate tensor
            intermediate1 = torch.empty(batch_size, self.expanded_channels, seq_len, 
                                        device=x.device, dtype=x.dtype)
            intermediate2 = torch.empty(batch_size, self.expanded_channels, seq_len,
                                        device=x.device, dtype=x.dtype)
            
            threads = 256
            
            # Step 1: Pointwise convolution
            total1 = batch_size * self.expanded_channels * seq_len
            blocks1 = (total1 + threads - 1) // threads
            depthwise_conv_ops.pointwise_conv_kernel(
                blocks1, threads,
                x.data_ptr(),
                pw1_w.data_ptr(),
                pw1_b.data_ptr(),
                intermediate1.data_ptr(),
                batch_size, self.channels, self.expanded_channels, seq_len
            )
            
            # Step 2: Depthwise convolution + GELU (fused)
            total2 = batch_size * self.expanded_channels * seq_len
            blocks2 = (total2 + threads - 1) // threads
            depthwise_conv_ops.depthwise_gelu_kernel(
                blocks2, threads,
                intermediate1.data_ptr(),
                dw_w.data_ptr(),
                dw_b.data_ptr(),
                intermediate2.data_ptr(),
                batch_size, self.expanded_channels, seq_len,
                self.kernel_size, self.padding
            )
            
            # Step 3: Pointwise projection + residual addition
            output = torch.empty_like(x)
            total3 = batch_size * self.channels * seq_len
            blocks3 = (total3 + threads - 1) // threads
            depthwise_conv_ops.pointwise_residual_kernel(
                blocks3, threads,
                intermediate2.data_ptr(),
                x.data_ptr(),
                pw2_w.data_ptr(),
                pw2_b.data_ptr(),
                output.data_ptr(),
                batch_size, self.expanded_channels, self.channels, seq_len
            )
            
            return output
