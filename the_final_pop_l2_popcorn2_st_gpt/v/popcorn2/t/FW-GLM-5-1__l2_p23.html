import torch
import torch.nn as nn

class Model(nn.Module):
    """
    One full sweep of Gibbs sampling for a multivariate Gaussian with
    known precision matrix.  Each coordinate is sampled from its full
    conditional distribution while holding all other coordinates fixed.
    Benchmarks the sequential-scan pattern common in Bayesian models
    with tractable conditionals (e.g., topic models, Boltzmann machines).
    """

    def __init__(self, dim):
        super().__init__()
        self.dim = dim
        L = torch.randn(dim, dim) * 0.1
        precision = L @ L.t() + torch.eye(dim) * 2.0
        self.register_buffer('precision', precision)
        self.register_buffer('mu', torch.zeros(dim))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, dim) – current state
        Returns:
            x_new: (B, dim) – state after one full Gibbs sweep
        """
        x = x.clone()
        for i in range(self.dim):
            prec_ii = self.precision[i, i]
            residual = x @ self.precision[i] - x[:, i] * prec_ii
            cond_mean = self.mu[i] - residual / prec_ii
            cond_var = 1.0 / prec_ii
            x[:, i] = cond_mean + torch.sqrt(cond_var) * torch.randn(x.shape[0], device=x.device)
        return x

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

// Optimized Gibbs sweep kernel with pre-generated random numbers
__global__ void gibbs_sweep_kernel(
    float* __restrict__ x,
    const float* __restrict__ precision,
    const float* __restrict__ mu,
    const float* __restrict__ rand_samples,
    const int dim,
    const int batch_size
) {
    int batch_idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (batch_idx >= batch_size) return;
    
    float* x_row = x + batch_idx * dim;
    const float* rand_row = rand_samples + batch_idx * dim;
    
    // Process each coordinate sequentially
    for (int i = 0; i < dim; i++) {
        float prec_ii = precision[i * dim + i];
        const float* prec_row = precision + i * dim;
        
        // Compute residual = x @ precision[i] - x[i] * prec_ii
        float dot = 0.0f;
        for (int j = 0; j < dim; j++) {
            dot += x_row[j] * prec_row[j];
        }
        float residual = dot - x_row[i] * prec_ii;
        
        // Conditional distribution parameters
        float cond_mean = mu[i] - residual / prec_ii;
        float cond_std = rsqrtf(prec_ii);  // 1/sqrt(prec_ii)
        
        // Update using pre-generated random sample
        x_row[i] = cond_mean + cond_std * rand_row[i];
    }
}

torch::Tensor gibbs_sweep_cuda(torch::Tensor x, torch::Tensor precision, torch::Tensor mu, torch::Tensor rand_samples) {
