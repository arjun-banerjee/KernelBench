import math
import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Fused MLA-style attention from compressed KV latents.

    The compressed latent representation is expanded into keys and values on the
    fly inside the attention computation rather than materializing a KV cache
    first. This isolates the main compute pattern behind fused MLA attention.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q: torch.Tensor, kv_latent: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        rank = kv_latent.shape[-1]
        out = torch.zeros_like(q)
        scale = 1.0 / math.sqrt(head_dim)
        for b in range(batch_size):
            for h in range(num_heads):
                for t in range(seq_len):
                    logits = []
                    values = []
                    for s in range(t + 1):
                        latent = kv_latent[b, s]
                        k = torch.matmul(latent, w_up_k[:, h])
                        v = torch.matmul(latent, w_up_v[:, h])
                        logits.append((q[b, t, h] * k).sum() * scale)
                        values.append(v)
                    attn = torch.softmax(torch.stack(logits, dim=0), dim=0)
                    value_stack = torch.stack(values, dim=0)
                    out[b, t, h] = (attn.unsqueeze(-1) * value_stack).sum(dim=0)
        return out

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void mla_attention_kernel(
    const float* __restrict__ q,
    const float* __restrict__ kv_latent,
    const float* __restrict__ w_up_k,
    const float* __restrict__ w_up_v,
    float* __restrict__ out,
    int batch_size,
    int seq_len,
    int num_heads,
    int head_dim,
    int rank,
    float scale
) {
    int b = blockIdx.x;
    int h = blockIdx.y;
    int t = threadIdx.x;

    if (t >= seq_len) return;

    extern __shared__ float sdata[];

    float* logits = sdata + threadIdx.x * (seq_len + head_dim);
    float* values = logits + seq_len;

    // Compute q vector pointer
    const float* q_ptr = q + ((b * seq_len + t) * num_heads + h) * head_dim;

    float max_logit = -1e30f;
    for (int s = 0; s <= t; ++s) {
        const float* latent = kv_latent + (b * seq_len + s) * rank;

        // Compute k = latent @ w_up_k[:, h]
        float dot = 0.0f;
        for (int d = 0; d < head_dim; ++d) {
            float k_val = 0.0f;
            for (int r = 0; r < rank; ++r) {
                float w = w_up_k[r * (num_heads * head_dim) + h * head_dim + d];
                k_val += latent[r] * w;
            }
            float qd = q_ptr[d];
            dot += qd * k_val;
        }
        logits[s] = dot * scale;
        if (logits[s] > max_logit) max_logit = logits[s];

        // Compute v = latent @ w_up_v[:, h] and store
        float* v_ptr = values + s * head_dim;
        for (int d = 0; d < head_dim; ++d) {
            float v_val = 0.0f;
            for (int r = 0; r < rank; ++r) {
                float w = w_up_v[r * (num_heads * head_dim) + h * head_dim + d];
                v_val += latent[r] * w;
            }
            v_ptr[d] = v_val;
        }
    }

    // Softmax
    float sum_exp = 0.0f;
    for (int s = 0; s <= t; ++s) {
        float e = expf(logits[s] - max_logit);
        logits[s] = e;
        sum_exp += e;
    }
    float inv_sum = 1.0f / sum_exp;
    for (int s = 0; s <= t; ++s) {
        logits[s] *= inv_sum;

        float acc = 0.0f;
        for (int s = 0; s <= t; ++s) {
            acc += logits[s] * values[s * head_dim + d];
        }
        out_ptr[d] = acc;
    }
}

    torch::Tensor kv_latent,
    torch::Tensor w_up_k,
    torch::Tensor w_up_v
) {
