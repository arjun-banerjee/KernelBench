"""
Reference: shard the last dimension per rank, then all_gather to reconstruct (tensor-parallel style).

Multi-GPU: each rank keeps one column block; output is concat along last dim (matches full input).
Single-GPU: returns x unchanged.
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_rank, get_world_size, is_distributed_run, maybe_init_process_group
dim = 80

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunks = x.chunk(ws, dim=-1)
        local = chunks[rank].contiguous() * self.scale
        gathered = [torch.empty_like(local) for _ in range(ws)]
        dist.all_gather(gathered, local)
        return torch.cat(gathered, dim=-1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void scale_kernel(const float* __restrict__ input, float* __restrict__ output, float scale, int64_t n) {
    int64_t idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        output[idx] = input[idx] * scale;
    }
}

torch::Tensor scale_cuda(torch::Tensor x, torch::Tensor scale) {
    auto output = torch::empty_like(x);
    int64_t n = x.numel();
    const int block_size = 256;
    const int num_blocks = (n + block_size - 1) / block_size;
    scale_kernel<<<num_blocks, block_size>>>(x.data_ptr<float>(), output.data_ptr<float>(), scale.item<float>(), n);
    return output;
}
'''

cpp_source = '''
torch::Tensor scale_cuda(torch::Tensor x, torch::Tensor scale);
'''

scale_module = load_inline(
    name='scale_module',
    cpp_sources=[cpp_source],
    cuda_sources=[cuda_source],
    functions=['scale_cuda'],
    verbose=False,
    with_cuda=True
)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if x.is_cuda and x.dtype == torch.float32:
            return scale_module.scale_cuda(x, self.scale)
        else:
            return x * self.scale
