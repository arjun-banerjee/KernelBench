#include <cuda.h>
#include <cuda_runtime.h>

template<int BLOCK_SIZE>
__global__ void fused_gather_linear_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int R, int B, int H, int K
) {
    // Each thread computes one output element
    // blockIdx.x: covers batch and K dimensions
    // threadIdx.x: offset within the block
    int total_threads = gridDim.x * blockDim.x;
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    
    const int RH = R * H;
    
    while (tid < B * K) {
        int b = tid / K;
        int k = tid % K;
        
        float sum = bias ? bias[k] : 0.0f;
        
        // Compute dot product: gather and multiply
        // input[r, b, h] * weight[k, r*H + h]
        #pragma unroll 4
        for (int i = 0; i < RH; i++) {

            int h = i % H;
            sum += input[r * B * H + b * H + h] * weight[k * RH + i];

        tid += total_threads;
    }
}

    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias,
