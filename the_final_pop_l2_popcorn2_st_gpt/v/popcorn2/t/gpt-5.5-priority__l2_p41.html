import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Evidence Lower Bound (ELBO) computation for a Variational Autoencoder.
    Combines a reconstruction term (binary cross-entropy) with the
    KL divergence between the approximate posterior q(z|x) = N(mu, sigma^2)
    and the prior p(z) = N(0, I).  The core loss function in VAE training.
    """

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(nn.Linear(input_dim, hidden_dim), nn.ReLU())
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)
        self.decoder = nn.Sequential(nn.Linear(latent_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim), nn.Sigmoid())

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim) – input data in [0, 1]
        Returns:
            (elbo, recon_loss, kl_div):
                elbo:       (B,)
                recon_loss: (B,)
                kl_div:     (B,)
        """
        h = self.encoder(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)
        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)
        x_recon = self.decoder(z)
        recon = F.binary_cross_entropy(x_recon, x, reduction='none').sum(-1)
        kl = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(-1)
        elbo = -(recon + kl)
        return (elbo, recon, kl)

static __inline__ __device__ float warp_sum(float v) {
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) v += __shfl_down_sync(0xffffffff, v, off);
    return v;
}

// One warp computes one output element of a linear layer. act: 0 none, 1 relu.
__global__ void linear_warp_kernel(const float* __restrict__ inp,
                                   const float* __restrict__ w,
                                   const float* __restrict__ b,
                                   float* __restrict__ out,
                                   int B, int IN, int OUT, int act) {
    const int lane = threadIdx.x & 31;
    const int warp = threadIdx.x >> 5;
    const int warps_per_block = blockDim.x >> 5;
    int idx = blockIdx.x * warps_per_block + warp;
    int total = B * OUT;
    if (idx >= total) return;
    int row = idx / OUT;
    int col = idx - row * OUT;
    const float* xp = inp + row * IN;
    const float* wp = w + col * IN;
    float s = 0.0f;
    for (int k = lane; k < IN; k += 32) s = fmaf(xp[k], wp[k], s);
    s = warp_sum(s);
    if (lane == 0) {
        s += b[col];
        if (act == 1 && s < 0.0f) s = 0.0f;
        out[row * OUT + col] = s;
    }
}

// For each batch row compute KL and sampled z. latent dim is small; one block per row.
__global__ void sample_kl_kernel(const float* __restrict__ mu,
                                 const float* __restrict__ lv,
                                 const float* __restrict__ eps,
                                 float* __restrict__ z,
                                 float* __restrict__ kl,
                                 int B, int L) {
    int row = blockIdx.x;
    int tid = threadIdx.x;
    float acc = 0.0f;
    for (int j = tid; j < L; j += blockDim.x) {
        int o = row * L + j;
        float m = mu[o];
        float l = lv[o];
        float el = expf(l);
        z[o] = m + expf(0.5f * l) * eps[o];
        acc += 1.0f + l - m * m - el;
    }
    __shared__ float sm[128];
    sm[tid] = acc;
    __syncthreads();
    for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) sm[tid] += sm[tid + stride];
        __syncthreads();
    }
    if (tid == 0) kl[row] = -0.5f * sm[0];
}

// Final decoder linear layer fused with BCE-with-logits accumulation.
// One warp computes one input feature/logit and atomically accumulates the per-row BCE sum.
__global__ void final_bce_warp_kernel(const float* __restrict__ h,
                                      const float* __restrict__ x,
                                      const float* __restrict__ w,
                                      const float* __restrict__ b,
                                      float* __restrict__ recon,
                                      int B, int H, int D) {
    const int lane = threadIdx.x & 31;
    const int warp = threadIdx.x >> 5;
    const int warps_per_block = blockDim.x >> 5;
    int idx = blockIdx.x * warps_per_block + warp;
    int total = B * D;
    if (idx >= total) return;
    int row = idx / D;
    int col = idx - row * D;
    const float* hp = h + row * H;
    const float* wp = w + col * H;
    float s = 0.0f;
    for (int k = lane; k < H; k += 32) s = fmaf(hp[k], wp[k], s);
    s = warp_sum(s);
    if (lane == 0) {
        float logit = s + b[col];
        float target = x[row * D + col];
        // BCE(sigmoid(logit), target), evaluated in the stable BCE-with-logits form.
        float a = fabsf(logit);
        float loss = fmaxf(logit, 0.0f) - logit * target + log1pf(expf(-a));
        atomicAdd(recon + row, loss);
    }
}

__global__ void elbo_kernel(const float* __restrict__ recon,
                            const float* __restrict__ kl,
                            float* __restrict__ elbo,
                            int B) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < B) elbo[i] = -(recon[i] + kl[i]);
}

    torch::Tensor x, torch::Tensor eps,
    torch::Tensor enc_w, torch::Tensor enc_b,
    torch::Tensor mu_w, torch::Tensor mu_b,
    torch::Tensor lv_w, torch::Tensor lv_b,
    torch::Tensor dec1_w, torch::Tensor dec1_b,
    torch::Tensor dec2_w, torch::Tensor dec2_b) {

    const int B = (int)x.size(0);
    const int D = (int)x.size(1);
    const int H = (int)enc_w.size(0);
    const int L = (int)mu_w.size(0);
    auto opts = x.options();
    auto h = torch::empty({B, H}, opts);
    auto mu = torch::empty({B, L}, opts);
    auto lv = torch::empty({B, L}, opts);
    auto z = torch::empty({B, L}, opts);
    auto dh = torch::empty({B, H}, opts);
