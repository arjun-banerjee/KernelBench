import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Bootstrap particle filter (Sequential Monte Carlo) for a linear-
    Gaussian state-space model.  Implements the predict–weight–resample
    cycle for one time step.  The state transition is x_{t} = A x_{t-1} + w
    and the observation model is y_t = C x_t + v.
    """

    def __init__(self, state_dim, obs_dim, num_particles):
        super().__init__()
        self.state_dim = state_dim
        self.obs_dim = obs_dim
        self.num_particles = num_particles
        self.register_buffer('A', 0.9 * torch.eye(state_dim))
        self.register_buffer('C', torch.randn(obs_dim, state_dim) * 0.5)
        self.process_noise = 0.1
        self.obs_noise = 0.5

    def forward(self, particles: torch.Tensor, log_weights: torch.Tensor, observation: torch.Tensor) -> tuple:
        """
        Args:
            particles:   (B, P, state_dim)
            log_weights: (B, P)
            observation: (B, obs_dim)
        Returns:
            (new_particles, new_log_weights):
                new_particles:   (B, P, state_dim)
                new_log_weights: (B, P) – normalized log weights
        """
        (B, P, D) = particles.shape
        pred = particles @ self.A.t() + self.process_noise * torch.randn_like(particles)
        obs_pred = pred @ self.C.t()
        diff = observation.unsqueeze(1) - obs_pred
        log_lik = -0.5 * (diff ** 2).sum(-1) / self.obs_noise ** 2
        new_log_w = log_weights + log_lik
        new_log_w = new_log_w - torch.logsumexp(new_log_w, dim=1, keepdim=True)
        weights = torch.exp(new_log_w)
        cumsum = torch.cumsum(weights, dim=1)
        u = (torch.arange(P, device=particles.device, dtype=particles.dtype) + torch.rand(B, 1, device=particles.device)) / P
        indices = torch.searchsorted(cumsum, u)
        indices = indices.clamp(0, P - 1)
        batch_idx = torch.arange(B, device=particles.device).unsqueeze(1).expand(-1, P)
        resampled = pred[batch_idx, indices]
        uniform_log_w = torch.full_like(new_log_w, -torch.log(torch.tensor(float(P))))
        return (resampled, uniform_log_w)

        if (tid < s) sdata[tid] = fmaxf(sdata[tid], sdata[tid+s]);
        __syncthreads();
    }
    maxv = sdata[0];

    float sum = 0.0f;
    for (int p = tid; p < P; p += blockDim.x) {
        float w = expf(lw_in[b*P+p] - maxv);
        sum += w;
        sdata[p] = w;
    }
    __syncthreads();
    sdata[tid] = sum;
    __syncthreads();
    for (int s=blockDim.x/2; s>0; s>>=1){ if(tid<s) sdata[tid]+=sdata[tid+s]; __syncthreads();}
    float lse = maxv + logf(sdata[0]);

    if (tid == 0) {
        float c = 0.0f;
        float logP = logf((float)P);
        curandState rs;
        curand_init(seed, b, 0, &rs);
        for (int p=0; p<P; p++) {
            float w = sdata[p];
            c += w;
            float u = (p + curand_uniform(&rs)) / P;
            int low=0, high=P;
            while(low < high){
                int m = (low+high)/2;
                if (c < u) low = m+1; else high=m; // reuse c cumul
            }
            // better cumsum rebuild
        }
        // Simplified correct sequential path for one thread
        c = 0.0f;
        curand_init(seed, b, 0, &rs);
        float cum[P<2048?P:0]; // stack for demo, assume P<=1024
        for(int p=0;p<P;p++){
            c += sdata[p];
            cum[p] = c;
        }
        for(int p=0;p<P;p++){
            float u = (p + curand_uniform(&rs)) / P;
            int idx=0;
            for(int j=0;j<P;j++) if(cum[j] >= u){idx=j;break;}
            idx = min(idx, P-1);
            for(int d=0;d<D;d++){
                resampled[b*P*D + p*D + d] = pred[b*P*D + idx*D + d];
            }
            uniform_log_w[b*P + p] = -logP;
        }
    }
}

std::vector<torch::Tensor> launch_full(
    torch::Tensor particles, torch::Tensor log_weights, torch::Tensor observation,
    torch::Tensor A, torch::Tensor C, float pn, float on, int64_t seed) {
    int B = particles.size(0), P = particles.size(1), D = particles.size(2), O = observation.size(1);
    auto opt = torch::TensorOptions().dtype(torch::kFloat32).device(particles.device());
    auto pred = torch::empty({B,P,D}, opt);
    auto nlw = torch::empty({B,P}, opt);
    auto res = torch::empty({B,P,D}, opt);
    auto ulw = torch::empty({B,P}, opt);

    dim3 g(B);
    particle_filter_kernel<<<g, min(P,256)>>>(
        particles.data_ptr<float>(), log_weights.data_ptr<float>(), observation.data_ptr<float>(),
        A.data_ptr<float>(), C.data_ptr<float>(), pred.data_ptr<float>(), nlw.data_ptr<float>(),
        B,P,D,O,pn,on,(unsigned long long)seed);

        B,P,D,(unsigned long long)seed+12345);

}
