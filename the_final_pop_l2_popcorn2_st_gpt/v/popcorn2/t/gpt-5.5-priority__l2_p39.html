import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        assert kernel_size % 2 == 1, 'odd kernel_size keeps T unchanged with symmetric padding'
        assert dilation >= 1
        self.padding = (kernel_size - 1) * dilation // 2
        self.dw = nn.Conv1d(in_channels, in_channels, kernel_size, padding=self.padding, dilation=dilation, groups=in_channels, bias=False)
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

class ModelNew(nn.Module):
    """KernelBench candidate entry point; default delegates to ``Model``."""

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        self._impl = Model(in_channels, out_channels, kernel_size, dilation=dilation)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self._impl(x)

__global__ void dw_silu_kernel(const float* __restrict__ x,
                               const float* __restrict__ dw,
                               float* __restrict__ tmp,
                               int B, int C, int T, int K, int dilation, int pad) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * C * T;
    if (idx >= total) return;
    int t = idx % T;
    int q = idx / T;
    int c = q % C;
    int b = q / C;

    float acc = 0.0f;
    const float* w = dw + c * K;
    const float* xb = x + (b * C + c) * T;
    #pragma unroll
    for (int kk = 0; kk < 16; ++kk) {
        if (kk < K) {
            int ti = t + kk * dilation - pad;
            if ((unsigned)ti < (unsigned)T) acc += xb[ti] * w[kk];
        }
    }
    // SiLU: x * sigmoid(x)
    float s = 1.0f / (1.0f + expf(-acc));
    tmp[idx] = acc * s;
}

// Computes output = A * W^T where A is logical (B*T, C), A[m,c]=tmp[b,c,t]
// W is (O,C), output layout is (B,O,T). 16x16 output tile, 32-wide K tile.
__global__ void pw_gemm_kernel(const float* __restrict__ tmp,
                               const float* __restrict__ pw,
                               float* __restrict__ out,
                               int B, int C, int T, int O) {
    constexpr int TM = 16;
    constexpr int TN = 16;
    constexpr int TK = 32;
    __shared__ float As[TM][TK + 1];
    __shared__ float Bs[TK][TN + 1];

    int tx = threadIdx.x; // output channel within tile
    int ty = threadIdx.y; // row within tile
    int m = blockIdx.y * TM + ty;
    int o = blockIdx.x * TN + tx;
    int NT = B * T;

    int b = 0, t = 0;
    if (m < NT) {
        b = m / T;
        t = m - b * T;
    }

    float acc = 0.0f;
    for (int k0 = 0; k0 < C; k0 += TK) {
        int k1 = k0 + tx;
        int k2 = k0 + tx + 16;
        if (m < NT && k1 < C) As[ty][tx] = tmp[(b * C + k1) * T + t];
        else As[ty][tx] = 0.0f;
        if (m < NT && k2 < C) As[ty][tx + 16] = tmp[(b * C + k2) * T + t];

        }

    }
}
