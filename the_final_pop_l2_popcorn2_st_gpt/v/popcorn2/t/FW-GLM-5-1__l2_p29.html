import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class Model(nn.Module):
    """
    Row-wise gated self-attention over Multiple Sequence Alignment (MSA)
    representations.  Each row of the MSA is an independent sequence; for
    each row, standard multi-head attention is performed across residue
    positions, with an additive bias from the pair representation.
    """

    def __init__(self, msa_dim, pair_dim, num_heads, dropout=0.0):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        assert msa_dim % num_heads == 0
        self.layer_norm = nn.LayerNorm(msa_dim)
        self.query = nn.Linear(msa_dim, msa_dim, bias=False)
        self.key = nn.Linear(msa_dim, msa_dim, bias=False)
        self.value = nn.Linear(msa_dim, msa_dim, bias=False)
        self.gate = nn.Linear(msa_dim, msa_dim)
        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)
        self.out_proj = nn.Linear(msa_dim, msa_dim)
        self.dropout = nn.Dropout(dropout)

    def forward(self, msa_repr: torch.Tensor, pair_repr: torch.Tensor, msa_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            msa_repr:  (B, S, N, msa_dim)
            pair_repr: (B, N, N, pair_dim)
            msa_mask:  (B, S, N)
        Returns:
            updated MSA (B, S, N, msa_dim)
        """
        (B, S, N, D) = msa_repr.shape
        h = self.num_heads
        d = self.head_dim
        x = self.layer_norm(msa_repr)
        q = self.query(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        k = self.key(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        v = self.value(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        g = torch.sigmoid(self.gate(x)).view(B, S, N, h, d).permute(0, 1, 3, 2, 4)
        bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2)
        attn = torch.einsum('bshid,bshjd->bshij', q, k) / math.sqrt(d)
        attn = attn + bias.unsqueeze(1)
        mask = msa_mask.view(B, S, 1, 1, N).expand_as(attn)
        attn = attn.masked_fill(mask == 0, float('-inf'))
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        out = torch.einsum('bshij,bshjd->bshid', attn, v)
        out = g * out
        out = out.permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        return self.out_proj(out)

from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template <typename scalar_t>
__global__ void attention_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ g,
    const scalar_t* __restrict__ bias,
    const bool* __restrict__ mask,
    scalar_t* __restrict__ out,
    int B, int S, int H, int N, int D,
    float scale) {
    
    int bsh = blockIdx.x;
    int tid = threadIdx.x;
    int block_size = blockDim.x;
    
    int b = bsh / (S * H);
    int rem = bsh % (S * H);
    int s = rem / H;
    int h = rem % H;
    
    extern __shared__ float shared_mem[];
    float* scores = shared_mem;
    
    int total_pairs = N * N;
    for (int idx = tid; idx < total_pairs; idx += block_size) {
        int i = idx / N;
        int j = idx % N;
        
        float score = 0.0f;
        int q_base = ((b * S + s) * H + h) * N * D;
        int k_base = q_base;
        
        for (int d = 0; d < D; d++) {
            score += (float)q[q_base + i * D + d] * (float)k[k_base + j * D + d];
        }
        score *= scale;
        
        int bias_offset = (b * H + h) * N * N + i * N + j;
        score += (float)bias[bias_offset];
        
        int mask_offset = (b * S + s) * N + j;
        if (!mask[mask_offset]) {
            score = -INFINITY;
        }
        
        scores[idx] = score;
    }
    __syncthreads();
    
    // Softmax per row
    for (int i = tid; i < N; i += block_size) {
        float max_val = -INFINITY;
        for (int j = 0; j < N; j++) {
            max_val = fmaxf(max_val, scores[i * N + j]);
        }
        
        float sum = 0.0f;
        for (int j = 0; j < N; j++) {
            if (scores[i * N + j] > -1e10f) {
                scores[i * N + j] = expf(scores[i * N + j] - max_val);
                sum += scores[i * N + j];
            } else {
                scores[i * N + j] = 0.0f;
            }
        }
        
        float inv_sum = 1.0f / (sum + 1e-10f);
        for (int j = 0; j < N; j++) {
            scores[i * N + j] *= inv_sum;
        }
    }
    __syncthreads();
    
    // Compute output
    for (int idx = tid; idx < N * D; idx += block_size) {
        int i = idx / D;
        int d = idx % D;
        
        float val = 0.0f;
        int v_base = ((b * S + s) * H + h) * N * D;
        for (int j = 0; j < N; j++) {
            val += scores[i * N + j] * (float)v[v_base + j * D + d];
        }
        
        int g_base = ((b * S + s) * H + h) * N * D;
        int out_base = g_base;
        out[out_base + i * D + d] = (scalar_t)(val * (float)g[g_base + i * D + d]);
    }
}

torch::Tensor attention_forward_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    torch::Tensor g,
    torch::Tensor bias,
    torch::Tensor mask) {
    
    int B = q.size(0);
    int S = q.size(1);
    int H = q.size(2);
    int N = q.size(3);
    int D = q.size(4);
    
    auto out = torch::zeros_like(q);
    float scale = 1.0f / sqrtf((float)D);
    
    int blocks = B * S * H;
    int threads = 256;
    
    size_t shared_mem_size = N * N * sizeof(float);
    
    AT_DISPATCH_FLOATING_TYPES(q.scalar_type(), "attention_kernel", ([&] {
        attention_kernel<scalar_t><<<blocks, threads, shared_mem_size>>>(
            q.data_ptr<scalar_t>(),
            k.data_ptr<scalar_t>(),
            v.data_ptr<scalar_t>(),
            g.data_ptr<scalar_t>(),
            bias.data_ptr<scalar_t>(),
            mask.data_ptr<bool>(),
            out.data_ptr<scalar_t>(),
            B, S, H, N, D,
            scale);
    }));
    
    return out;
}
"""

attention_ext = load_inline(
    name='attention_ext_v1',
    cpp_sources='',
    cuda_sources=cuda_source,
    functions=['attention_forward_cuda'],
    extra_cuda_cflags=['-O2'],
    verbose=False
)

class ModelNew(nn.Module):
    def __init__(self, msa_dim, pair_dim, num_heads, dropout=0.0):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = msa_dim // num_heads
        assert msa_dim % num_heads == 0
        self.layer_norm = nn.LayerNorm(msa_dim)
        self.query = nn.Linear(msa_dim, msa_dim, bias=False)
        self.key = nn.Linear(msa_dim, msa_dim, bias=False)
        self.value = nn.Linear(msa_dim, msa_dim, bias=False)
        self.gate = nn.Linear(msa_dim, msa_dim)
        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)
        self.out_proj = nn.Linear(msa_dim, msa_dim)
        self.dropout = nn.Dropout(dropout)
    
    def forward(self, msa_repr, pair_repr, msa_mask):
        (B, S, N, D) = msa_repr.shape
        h = self.num_heads
        d = self.head_dim
        
        x = self.layer_norm(msa_repr)
        q = self.query(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4).contiguous()
        k = self.key(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4).contiguous()
        v = self.value(x).view(B, S, N, h, d).permute(0, 1, 3, 2, 4).contiguous()
        g = torch.sigmoid(self.gate(x)).view(B, S, N, h, d).permute(0, 1, 3, 2, 4).contiguous()
        bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2).contiguous()
        
        mask_bool = (msa_mask > 0)
        out = attention_ext.attention_forward_cuda(q, k, v, g, bias, mask_bool)
        
        out = out.permute(0, 1, 3, 2, 4).reshape(B, S, N, D)
        return self.out_proj(out)
