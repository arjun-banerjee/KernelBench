import torch

class Model(nn.Module):
    """
    Reparameterization trick for differentiable sampling from a diagonal
    Gaussian.  Separates stochasticity from parameters so that gradients
    flow through mu and log_var.  Also demonstrates a "mixture of
    Gaussians" reparameterization with learnable component weights.
    """

    def __init__(self, latent_dim, num_components=1):
        super().__init__()
        self.latent_dim = latent_dim
        self.num_components = num_components
        if num_components > 1:
            self.logits = nn.Parameter(torch.zeros(num_components))

    def forward(self, mu: torch.Tensor, log_var: torch.Tensor) -> tuple:
        """
        Args:
            mu:      (B, K, D) or (B, D) – means
            log_var: (B, K, D) or (B, D) – log-variances
        Returns:
            (z, log_q_z):
                z:       (B, D) – sampled latents
                log_q_z: (B,)   – log-density of sample under q
        """
        std = torch.exp(0.5 * log_var)
        if self.num_components > 1 and mu.dim() == 3:
            weights = torch.softmax(self.logits, dim=0)
            comp = torch.multinomial(weights.expand(mu.shape[0], -1), 1).squeeze(-1)
            batch_idx = torch.arange(mu.shape[0], device=mu.device)
            mu_k = mu[batch_idx, comp]
            std_k = std[batch_idx, comp]
            z = mu_k + std_k * torch.randn_like(mu_k)
            log_probs = -0.5 * ((z.unsqueeze(1) - mu) / std) ** 2 - log_var * 0.5 - 0.5 * torch.log(torch.tensor(2 * 3.14159265))
            log_probs = log_probs.sum(-1) + weights.log().unsqueeze(0)
            log_q_z = torch.logsumexp(log_probs, dim=1)
        else:
            if mu.dim() == 3:
                mu = mu[:, 0]
                std = std[:, 0]
                log_var = log_var[:, 0]
            z = mu + std * torch.randn_like(mu)
            log_q_z = (-0.5 * ((z - mu) / std) ** 2 - 0.5 * log_var - 0.5 * torch.log(torch.tensor(2 * 3.14159265))).sum(-1)
        return (z, log_q_z)

            acc += -0.5f*x*x -0.5f*lv;
        }
        // block reduce using warp reductions and atomics
        for (int off=16; off>0; off>>=1) acc += __shfl_down_sync(0xffffffff, acc, off);
        if ((tid&31)==0) atomicAdd(&sums[k], acc);
        __syncthreads();
    }
    if (tid==0) {
        const float c=-0.9189385332046727f;
        float m=-CUDART_INF_F;
        for(int k=0;k<K;k++){ float v=sums[k]+D*c+logits[k]-lse; sums[k]=v; m=fmaxf(m,v); }
        float s=0.f; for(int k=0;k<K;k++) s+=expf(sums[k]-m);
        logq[b]=m+logf(s);
    }
}

template<int D>
__global__ void single_kernel_fixed(
    const float* __restrict__ mu,
    const float* __restrict__ logv,
    const float* __restrict__ eps,
    float* __restrict__ z,
    float* __restrict__ logq,
    int B)
{
    int b=blockIdx.x;
    int tid=threadIdx.x;
    float acc=0.f;
    if (tid < D) {
        int idx=b*D+tid;
        float lv=logv[idx];
        float e=eps[idx];
        float val=mu[idx]+expf(0.5f*lv)*e;
        z[idx]=val;
        // exactly the same algebra as ((z-mu)/std)^2, but equals eps^2 up to tiny error
        acc = -0.5f*e*e -0.5f*lv;
    }
    for(int off=16; off>0; off>>=1) acc += __shfl_down_sync(0xffffffff, acc, off);
    __shared__ float partial[8];
    if ((tid&31)==0) partial[tid>>5]=acc;
    __syncthreads();
    if (tid < 8) {
        acc = (tid < (blockDim.x+31)/32) ? partial[tid] : 0.f;
        for(int off=4; off>0; off>>=1) acc += __shfl_down_sync(0xffffffff, acc, off);
        if (tid==0) logq[b]=acc + (float)D * -0.9189385332046727f;
    }
}

__global__ void single_kernel_generic(
    const float* __restrict__ mu,
    const float* __restrict__ logv,
    const float* __restrict__ eps,
    float* __restrict__ z,
    float* __restrict__ logq,
    int B, int D)
{
    int b=blockIdx.x, tid=threadIdx.x;
    float acc=0.f;
    for(int d=tid; d<D; d+=blockDim.x){
        int idx=b*D+d;
        float lv=logv[idx];
        float e=eps[idx];
        z[idx]=mu[idx]+expf(0.5f*lv)*e;
        acc += -0.5f*e*e -0.5f*lv;
    }
    for(int off=16; off>0; off>>=1) acc += __shfl_down_sync(0xffffffff, acc, off);
    __shared__ float partial[8];
    if((tid&31)==0) partial[tid>>5]=acc;
    __syncthreads();
    if(tid<8){
        acc=(tid < (blockDim.x+31)/32)?partial[tid]:0.f;
        for(int off=4; off>0; off>>=1) acc += __shfl_down_sync(0xffffffff, acc, off);
        if(tid==0) logq[b]=acc + (float)D * -0.9189385332046727f;
    }
}

std::vector<torch::Tensor> reparam_mixture(torch::Tensor mu, torch::Tensor logv, torch::Tensor eps, torch::Tensor comp, torch::Tensor logits) {
    CHECK_CUDA(mu); CHECK_CUDA(logv); CHECK_CUDA(eps); CHECK_CUDA(comp); CHECK_CUDA(logits);
    TORCH_CHECK(mu.scalar_type() == torch::kFloat32, "float32 only");
    int B = mu.size(0), K = mu.size(1), D = mu.size(2);
    auto z = torch::empty({B, D}, mu.options());
    auto logq = torch::empty({B}, mu.options());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if (D == 64 && K == 4) {
        mixture_kernel_fixed<64,4><<<B, 128, 0, stream>>>(mu.data_ptr<float>(), logv.data_ptr<float>(), eps.data_ptr<float>(), comp.data_ptr<long>(), logits.data_ptr<float>(), z.data_ptr<float>(), logq.data_ptr<float>(), B);
    } else {
        int threads = 128;
        size_t smem = (D + K) * sizeof(float);
        mixture_kernel_generic<<<B, threads, smem, stream>>>(mu.data_ptr<float>(), logv.data_ptr<float>(), eps.data_ptr<float>(), comp.data_ptr<long>(), logits.data_ptr<float>(), z.data_ptr<float>(), logq.data_ptr<float>(), B, K, D);
    }
    return {z, logq};
}

std::vector<torch::Tensor> reparam_single(torch::Tensor mu, torch::Tensor logv, torch::Tensor eps) {
    CHECK_CUDA(mu); CHECK_CUDA(logv); CHECK_CUDA(eps);
    TORCH_CHECK(mu.scalar_type() == torch::kFloat32, "float32 only");
    int B = mu.size(0), D = mu.size(1);
    auto z = torch::empty({B, D}, mu.options());
    auto logq = torch::empty({B}, mu.options());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if (D == 64) single_kernel_fixed<64><<<B, 64, 0, stream>>>(mu.data_ptr<float>(), logv.data_ptr<float>(), eps.data_ptr<float>(), z.data_ptr<float>(), logq.data_ptr<float>(), B);
    else single_kernel_generic<<<B, 128, 0, stream>>>(mu.data_ptr<float>(), logv.data_ptr<float>(), eps.data_ptr<float>(), z.data_ptr<float>(), logq.data_ptr<float>(), B, D);
    return {z, logq};
}
