import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Hamiltonian Monte Carlo (HMC) proposal using leapfrog integration.
    Simulates Hamiltonian dynamics on a batched set of particles to
    produce proposals with high acceptance probability.  The target
    distribution is a standard Gaussian for benchmarking.
    """

    def __init__(self, dim, num_leapfrog_steps=10, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.num_leapfrog_steps = num_leapfrog_steps
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))

    def _grad_log_prob(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -(x - self.mu)

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (x ** 2).sum(dim=-1)

    def forward(self, q: torch.Tensor) -> tuple:
        """
        Args:
            q: (B, dim) – current positions
        Returns:
            (q_new, accepted):
                q_new:    (B, dim)
                accepted: (B,)
        """
        eps = self.step_size
        p = torch.randn_like(q)
        current_p = p.clone()
        current_q = q.clone()
        p = p + 0.5 * eps * self._grad_log_prob(q)
        for _ in range(self.num_leapfrog_steps - 1):
            q = q + eps * p
            p = p + eps * self._grad_log_prob(q)
        q = q + eps * p
        p = p + 0.5 * eps * self._grad_log_prob(q)
        p = -p
        current_H = -self._log_prob(current_q) + 0.5 * (current_p ** 2).sum(-1)
        proposed_H = -self._log_prob(q) + 0.5 * (p ** 2).sum(-1)
        log_alpha = current_H - proposed_H
        log_u = torch.log(torch.rand(q.shape[0], device=q.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        q_new = torch.where(accepted.unsqueeze(-1), q, current_q)
        return (q_new, accepted)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> hmc_forward(torch::Tensor q, torch::Tensor p, torch::Tensor u, int64_t L, double eps);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>
#include <cmath>

__global__ void hmc_kernel(const float* __restrict__ q,
                           const float* __restrict__ p0,
                           const float* __restrict__ u,
                           float* __restrict__ q_out,
                           bool* __restrict__ acc_out,
                           int B, int D, int L, float eps) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    int nthreads = blockDim.x;

    extern __shared__ float smem[];
    float* qprop = smem;                 // D floats
    float* scur  = qprop + D;            // nthreads floats
    float* spro  = scur + nthreads;      // nthreads floats

    float cur = 0.0f;
    float pro = 0.0f;
    int base = b * D;
    float half_eps = 0.5f * eps;

    for (int d = tid; d < D; d += nthreads) {
        float qq0 = q[base + d];
        float pp = p0[base + d];
        float qq = qq0;
        float pp0 = pp;

        // Leapfrog integration for grad log N(0,I) = -q.
        pp = pp - half_eps * qq;
        #pragma unroll 16
        for (int i = 0; i < L - 1; ++i) {
            qq = qq + eps * pp;
            pp = pp - eps * qq;
        }
        qq = qq + eps * pp;
        pp = pp - half_eps * qq;
        // Reference negates p here; p^2 is unchanged.

        qprop[d] = qq;
        cur += 0.5f * (qq0 * qq0 + pp0 * pp0);
        pro += 0.5f * (qq * qq + pp * pp);
    }

    scur[tid] = cur;
    spro[tid] = pro;
    __syncthreads();

    // Block reduction. 128 threads are used by the launcher.
    for (int offset = nthreads >> 1; offset > 0; offset >>= 1) {
        if (tid < offset) {
            scur[tid] += scur[tid + offset];
            spro[tid] += spro[tid + offset];
        }
        __syncthreads();
    }

    bool accepted = false;
    if (tid == 0) {
        float uu = u[b];
        uu = uu < 1.0e-10f ? 1.0e-10f : uu;
        float log_u = logf(uu);
        float log_alpha = scur[0] - spro[0];
        accepted = (log_u < log_alpha);
        acc_out[b] = accepted;
        scur[0] = accepted ? 1.0f : 0.0f; // broadcast through shared memory
    }
    __syncthreads();
    accepted = (scur[0] != 0.0f);

    for (int d = tid; d < D; d += nthreads) {
        q_out[base + d] = accepted ? qprop[d] : q[base + d];
    }
}

std::vector<torch::Tensor> hmc_forward(torch::Tensor q, torch::Tensor p, torch::Tensor u, int64_t L64, double eps64) {
    const int B = (int)q.size(0);
    const int D = (int)q.size(1);
    auto q_out = torch::empty_like(q);
    auto acc = torch::empty({B}, q.options().dtype(torch::kBool));

    const int threads = 128;
    const size_t shmem = (D + 2 * threads) * sizeof(float);
    hmc_kernel<<<B, threads, shmem, at::cuda::getCurrentCUDAStream()>>>(
        q.data_ptr<float>(), p.data_ptr<float>(), u.data_ptr<float>(),
        q_out.data_ptr<float>(), acc.data_ptr<bool>(), B, D, (int)L64, (float)eps64);
    return {q_out, acc};
}

class ModelNew(nn.Module):
    def __init__(self, dim, num_leapfrog_steps=10, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.num_leapfrog_steps = num_leapfrog_steps
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))

    def forward(self, q: torch.Tensor) -> tuple:
        # Keep the same RNG calls, in the same order, as the reference.
        p = torch.randn_like(q)
        u = torch.rand((q.shape[0],), device=q.device)
