import torch
import torch.nn as nn
import torch.nn.functional as F

def _quaternion_to_rotation_matrix(q: torch.Tensor) -> torch.Tensor:
    """
    Unit quaternion ``q = (w, x, y, z)`` (scalar-first) to rotation matrix ``R``,
    shape ``(B, 3, 3)``, with ``p' = p @ R.T`` for row vectors ``p`` (``N×3``).
    """
    q = F.normalize(q, dim=-1, eps=1e-08)
    (w, x, y, z) = q.unbind(dim=-1)
    xx = x * x
    yy = y * y
    zz = z * z
    xy = x * y
    xz = x * z
    yz = y * z
    wx = w * x
    wy = w * y
    wz = w * z
    m00 = 1.0 - 2.0 * (yy + zz)
    m01 = 2.0 * (xy - wz)
    m02 = 2.0 * (xz + wy)
    m10 = 2.0 * (xy + wz)
    m11 = 1.0 - 2.0 * (xx + zz)
    m12 = 2.0 * (yz - wx)
    m20 = 2.0 * (xz - wy)
    m21 = 2.0 * (yz + wx)
    m22 = 1.0 - 2.0 * (xx + yy)
    row0 = torch.stack((m00, m01, m02), dim=-1)
    row1 = torch.stack((m10, m11, m12), dim=-1)
    row2 = torch.stack((m20, m21, m22), dim=-1)
    return torch.stack((row0, row1, row2), dim=-2)

class Model(nn.Module):
    """
    **Fused vectorized pose transform**: rotation from unit quaternion ``q`` and
    translation ``t`` applied to a batch of 3D points.

    One forward fuses **quaternion → ``3×3``** (Hamilton, scalar-first ``w,x,y,z``)
    with **batched matmul** and **broadcast translation**—typical for LiDAR / camera
    extrinsics, hand-eye calibration, and world ↔ body frame changes in VLA stacks.

    ``points' = points @ R.T + t`` with ``points`` row-wise ``(B, N, 3)``.
    """

    def __init__(self):
        super().__init__()

    def forward(self, points: torch.Tensor, quat: torch.Tensor, trans: torch.Tensor) -> torch.Tensor:
        """
        Args:
            points: (B, N, 3) points in the **source** frame.
            quat: (B, 4) unit quaternion ``(w, x, y, z)`` mapping source → target rotation.
            trans: (B, 3) translation in the **target** frame (same convention as ``R``).
        Returns:
            (B, N, 3) points in the **target** frame.
        """
        (B, N, _) = points.shape
        assert quat.shape == (B, 4) and trans.shape == (B, 3)
        R = _quaternion_to_rotation_matrix(quat)
        return torch.matmul(points, R.transpose(-1, -2)) + trans.unsqueeze(1)

        return self._impl(points, quat, trans)

// Fused quaternion-to-rotation and point transform kernel
// Each thread processes one point
// Uses shared memory to cache rotation matrix and translation per batch block

template <typename scalar_t>
__global__ void pose_transform_kernel(
    const scalar_t* __restrict__ points,    // (B, N, 3)
    const scalar_t* __restrict__ quat,      // (B, 4)
    const scalar_t* __restrict__ trans,     // (B, 3)

    __shared__ scalar_t s_trans[3];

    const int point_id = blockIdx.x * blockDim.x + threadIdx.x;

    if (threadIdx.x < 4) {

        scalar_t qw = quat[batch_id * 4 + 0];
        scalar_t qx = quat[batch_id * 4 + 1];
        scalar_t qy = quat[batch_id * 4 + 2];
        scalar_t qz = quat[batch_id * 4 + 3];

        scalar_t norm = sqrtf(qw * qw + qx * qx + qy * qy + qz * qz);
        if (norm > scalar_t(0)) {
            scalar_t inv_norm = scalar_t(1.0) / norm;
            qw *= inv_norm;
            qx *= inv_norm;
            qy *= inv_norm;
            qz *= inv_norm;

        // Each thread computes specific elements

            scalar_t xz = qx * qz;
