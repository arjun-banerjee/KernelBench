    for (int i = tid; i < total; i += BLOCK_SIZE) {
        int n = i / seq_len;
        int s = i % seq_len;
        sum += input[n * channels * seq_len + c * seq_len + s];
    }
    float m = block_reduce_sum(sum) / total;
    if (tid == 0) mean[c] = m;
    __syncthreads();
    
    // Compute variance
    float sum_sq = 0.0f;
    for (int i = tid; i < total; i += BLOCK_SIZE) {
        int n = i / seq_len;
        int s = i % seq_len;
        float diff = input[n * channels * seq_len + c * seq_len + s] - m;
        sum_sq += diff * diff;
    }
    if (tid == 0) var[c] = block_reduce_sum(sum_sq) / total;
}

__global__ void fused_kernel(const float* __restrict__ input, const float* __restrict__ conv_weight,
    const float* __restrict__ conv_bias, const float* __restrict__ bn_weight,
    const float* __restrict__ bn_bias, const float* __restrict__ bn_mean,
    const float* __restrict__ bn_var, float* __restrict__ output,
    int batch_size, int channels, int seq_len, int kernel_size, int dilation, int padding, float eps) {
    
    int bc = blockIdx.x;
    int n = bc / channels;
    int c_out = bc % channels;
    int s = blockIdx.y * BLOCK_SIZE + threadIdx.x;
    
    if (n >= batch_size || c_out >= channels || s >= seq_len) return;
    
    float conv_out = conv_bias[c_out];
    
    #pragma unroll 3
    for (int c_in = 0; c_in < 320; c_in++) {
        float mean = bn_mean[c_in];
        float var = bn_var[c_in];
        float gamma = bn_weight[c_in];
        float beta = bn_bias[c_in];
        float inv_std = rsqrtf(var + eps);
        
        #pragma unroll
        for (int k = 0; k < 3; k++) {
            int input_s = s - padding + k * dilation;
            if (input_s >= 0 && input_s < seq_len) {
                int input_idx = n * channels * seq_len + c_in * seq_len + input_s;
                float val = input[input_idx];
                // BatchNorm + ReLU
                val = fmaxf(0.0f, (val - mean) * inv_std * gamma + beta);
                // Conv weight access
                int weight_idx = c_out * channels * kernel_size + c_in * kernel_size + k;
                conv_out += val * conv_weight[weight_idx];
            }
        }
    }
    
    int idx = n * channels * seq_len + c_out * seq_len + s;
    output[idx] = conv_out + input[idx];
}
