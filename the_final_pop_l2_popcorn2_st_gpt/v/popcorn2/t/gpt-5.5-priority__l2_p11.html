"""
Reference: broadcast tensor from rank 0 (NCCL broadcast), then apply a linear map.

Multi-GPU: rank != 0 starts from zeros; after broadcast all ranks match rank 0's shard.
Single-GPU: equivalent to linear(x).
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_rank, is_distributed_run, maybe_init_process_group
features = 128

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.lin = nn.Linear(features, features, bias=True)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        if is_distributed_run() and get_rank() != 0:
            x = torch.zeros_like(x)
        if is_distributed_run():
            dist.broadcast(x, src=0)
        return self.lin(x)

// C = A * W^T + b, where A is [M,128], W is [128,128].
// A simple tiled GEMM specialized for the fixed 128 feature dimension.
__global__ void linear128_kernel(const float* __restrict__ x,
                                 const float* __restrict__ w,
                                 const float* __restrict__ b,
                                 float* __restrict__ y,
                                 int M) {
    constexpr int BM = 16;
    constexpr int BN = 16;
    constexpr int BK = 32;

    __shared__ float As[BM][BK];
    __shared__ float Bs[BN][BK];   // stored as [output_col_within_tile][k]

    int tx = threadIdx.x;          // 0..15, output column within tile
    int ty = threadIdx.y;          // 0..15, row within tile
    int tid = ty * BN + tx;

    int row = blockIdx.y * BM + ty;
    int col = blockIdx.x * BN + tx;

    float acc = 0.0f;

    #pragma unroll
    for (int kt = 0; kt < 128; kt += BK) {
        // Load A tile: 16 x 32
        #pragma unroll
        for (int idx = tid; idx < BM * BK; idx += BM * BN) {
            int r = idx / BK;
            int k = idx - r * BK;
            int gr = blockIdx.y * BM + r;
            As[r][k] = (gr < M) ? x[gr * 128 + kt + k] : 0.0f;
        }

        // Load W tile, transposed into shared memory for contiguous global loads.
        #pragma unroll
        for (int idx = tid; idx < BN * BK; idx += BM * BN) {
            int c = idx / BK;
            int k = idx - c * BK;

// This avoids launching only 8 heavy tiled blocks for the common M~=16 case.

}
