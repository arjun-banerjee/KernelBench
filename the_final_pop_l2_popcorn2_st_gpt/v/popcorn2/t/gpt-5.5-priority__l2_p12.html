import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Pre-norm 1D dilated convolution block with BatchNorm: BatchNorm1d, ReLU activation, dilated Conv1d, and residual addition.
    At high dilation rates, memory access becomes non-contiguous across large strides in the sequence dimension.
    """

    def __init__(self, channels, kernel_size, dilation):
        super(Model, self).__init__()
        self.norm = nn.BatchNorm1d(channels)
        self.activation = nn.ReLU()
        self.conv = nn.Conv1d(channels, channels, kernel_size, padding=dilation, dilation=dilation)

    def forward(self, x):
        residual = x
        x = self.norm(x)
        x = self.activation(x)
        x = self.conv(x)
        x = x + residual
        return x

__global__ void transpose_w3_kernel(const float* __restrict__ w,
                                    float* __restrict__ wt,
                                    int C) {
    int total = 3 * C * C;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    for (; idx < total; idx += blockDim.x * gridDim.x) {
        int co = idx / (C * 3);
        int r = idx - co * C * 3;
        int ci = r / 3;
        int k = r - ci * 3;
        wt[(k * C + ci) * C + co] = w[idx];
    }
}

__global__ void bn_stats_kernel(const float* __restrict__ x,
                                float* __restrict__ mean,
                                float* __restrict__ invstd,
                                int N, int C, int L, float eps) {
    int c = blockIdx.x;
    int tid = threadIdx.x;
    const int BS = 256;
    __shared__ float ssum[BS];
    __shared__ float ssq[BS];
    float sum = 0.0f;
    float sq = 0.0f;
    int M = N * L;
    for (int m = tid; m < M; m += BS) {
        int n = m / L;
        int l = m - n * L;
        float v = x[(n * C + c) * L + l];
        sum += v;
        sq = fmaf(v, v, sq);
    }
    ssum[tid] = sum;
    ssq[tid] = sq;
    __syncthreads();
    for (int stride = BS >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) {
            ssum[tid] += ssum[tid + stride];
            ssq[tid] += ssq[tid + stride];
        }
        __syncthreads();
    }
    if (tid == 0) {
        float m = ssum[0] / (float)M;
        float var = ssq[0] / (float)M - m * m;
        if (var < 0.0f) var = 0.0f;
        mean[c] = m;
        invstd[c] = rsqrtf(var + eps);
    }
}

__global__ void fused_bn_relu_conv_res_kernel(
    const float* __restrict__ x,
    const float* __restrict__ wt,          // [3, C, C_out] contiguous, C_out==C
    const float* __restrict__ bias,
    const float* __restrict__ gamma,
    const float* __restrict__ beta,
    const float* __restrict__ mean,
    const float* __restrict__ var_or_inv,
    float* __restrict__ out,
    int N, int C, int L, int dilation, float eps, int training) {

    int l = blockIdx.x;
    int n = blockIdx.y;
    int t = threadIdx.x;
    extern __shared__ float sh[]; // 3*C normalized+relu input values for this (n,l)

    for (int idx = t; idx < 3 * C; idx += blockDim.x) {
        int k = idx / C;
        int ci = idx - k * C;
        int pos = l + (k - 1) * dilation;
        float a = 0.0f;
        if ((unsigned)pos < (unsigned)L) {
            float v = x[(n * C + ci) * L + pos];
            float inv = training ? var_or_inv[ci] : rsqrtf(var_or_inv[ci] + eps);
            a = (v - mean[ci]) * inv;
            a = fmaf(a, gamma[ci], beta[ci]);
            a = a > 0.0f ? a : 0.0f;
        }
        sh[idx] = a;
    }
    __syncthreads();

    if (t < C) {
        int co = t;
        float acc = bias[co];
        #pragma unroll 3
        for (int k = 0; k < 3; ++k) {
            int base = k * C;
            const float* wbase = wt + base * C + co;
            #pragma unroll 4
            for (int ci = 0; ci < C; ++ci) {
                acc = fmaf(sh[base + ci], wbase[ci * C], acc);
            }
        }
        out[(n * C + co) * L + l] = acc + x[(n * C + co) * L + l];
    }
}
