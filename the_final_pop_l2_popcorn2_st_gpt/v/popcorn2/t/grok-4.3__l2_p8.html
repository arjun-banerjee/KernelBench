import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Grounded top-2 MoE router.

    Router logits are biased by token-expert similarity against learned
    grounding embeddings, then normalized over the selected top-2 experts.
    """

    def __init__(self):
        super().__init__()

    def forward(self, token_hidden: torch.Tensor, router_logits: torch.Tensor, expert_ground: torch.Tensor, alpha: float) -> torch.Tensor:
        grounded = router_logits + alpha * (token_hidden @ expert_ground.t())
        (top_vals, top_idx) = torch.topk(grounded, k=2, dim=-1)
        top_weights = torch.softmax(top_vals, dim=-1)
        return torch.stack((top_idx.to(torch.float32), top_weights), dim=-1)
num_tokens = 8192
hidden_dim = 160
num_experts = 16
alpha = 0.35

def get_inputs():
    p = popcorn_pri
    mode = p.sample_input_mode()
    nt = p.trial_dim(num_tokens, 'num_tokens', mode=mode)
    hd = p.trial_dim(hidden_dim, 'hidden_dim', mode=mode, align=16)
    ne = p.trial_dim(num_experts, 'num_experts', mode=mode, minimum=2)
    token_hidden = torch.randn(nt, hd, dtype=torch.float32)
    router_logits = torch.randn(nt, ne, dtype=torch.float32)
    expert_ground = torch.randn(ne, hd, dtype=torch.float32)
    return [token_hidden, router_logits, expert_ground, alpha]

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cfloat>

__global__ void grounded_top2_kernel(
    const float* __restrict__ token_hidden,
    const float* __restrict__ router_logits,
    const float* __restrict__ expert_ground,
    float* __restrict__ output,
    int num_tokens,
    int hidden_dim,
    int num_experts,
    float alpha
) {
    int token_idx = blockIdx.x;
    if (token_idx >= num_tokens) return;

    extern __shared__ float sdata[];

    // Compute grounded scores: router + alpha * (token @ expert.t())
    // Each thread computes partial dot products or full since hidden=160 small
    float* scores = sdata;
    
    for (int e = threadIdx.x; e < num_experts; e += blockDim.x) {
        float dot = 0.0f;
        for (int h = 0; h < hidden_dim; h++) {
            dot += token_hidden[token_idx * hidden_dim + h] * expert_ground[e * hidden_dim + h];
        }
        scores[e] = router_logits[token_idx * num_experts + e] + alpha * dot;
    }
    __syncthreads();

    // Find top-2 (only first few threads needed)
    if (threadIdx.x == 0) {
        float max1 = -FLT_MAX, max2 = -FLT_MAX;
        int idx1 = -1, idx2 = -1;
        
        for (int e = 0; e < num_experts; e++) {
            float val = scores[e];
            if (val > max1) {
                max2 = max1; idx2 = idx1;
                max1 = val; idx1 = e;
            } else if (val > max2) {
                max2 = val; idx2 = e;
            }
        }
        
        // Softmax on top_vals
        float maxv = fmaxf(max1, max2);
        float sum_exp = expf(max1 - maxv) + expf(max2 - maxv);
        float w1 = expf(max1 - maxv) / sum_exp;
        float w2 = expf(max2 - maxv) / sum_exp;
        
        // Output: [idx1, w1, idx2, w2] but stacked as (top_idx, top_weights) -> shape [N, 2, 2]
        // torch.stack((top_idx.to(float32), top_weights), dim=-1) -> [N, 2, 2]
        float* out_ptr = output + token_idx * 4;
        out_ptr[0] = (float)idx1;
        out_ptr[1] = w1;
        out_ptr[2] = (float)idx2;
        out_ptr[3] = w2;
    }
}

torch::Tensor grounded_top2_cuda(
    torch::Tensor token_hidden,
    torch::Tensor router_logits,
    torch::Tensor expert_ground,
    float alpha
) {
    int num_tokens = token_hidden.size(0);
    int hidden_dim = token_hidden.size(1);
    int num_experts = router_logits.size(1);
    
    auto output = torch::empty({num_tokens, 2, 2}, token_hidden.options());
    
    int threads = 32;  // enough for num_experts<=32 probably
    int shared_mem = num_experts * sizeof(float);
    
    grounded_top2_kernel<<<num_tokens, threads, shared_mem>>>(
        token_hidden.data_ptr<float>(),
        router_logits.data_ptr<float>(),
        expert_ground.data_ptr<float>(),
        output.data_ptr<float>(),
        num_tokens, hidden_dim, num_experts, alpha
    );
    
    return output;
}
