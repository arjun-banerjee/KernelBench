# popcorn2: large-tier module centers (scripts/gen_popcorn2_centers.py).
# Source: KernelBench/level2/popcorn/36_InvariantPointAttention.py

import torch
import torch.nn as nn
import torch.nn.functional as F
import math

class Model(nn.Module):
    """
    Invariant Point Attention (IPA) from AlphaFold2's structure module.
    Combines sequence-space attention with 3-D geometric queries/keys/values
    expressed in local residue frames.  The attention logits incorporate
    Euclidean distances in global frame, making it SE(3)-equivariant.
    """

    def __init__(self, node_dim, pair_dim, num_heads, num_query_points, num_value_points):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = node_dim // num_heads
        self.num_query_points = num_query_points
        self.num_value_points = num_value_points
        assert node_dim % num_heads == 0
        d = self.head_dim
        self.q_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.k_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.v_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.q_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.k_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.v_pts = nn.Linear(node_dim, num_heads * num_value_points * 3, bias=False)
        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)
        self.head_weights = nn.Parameter(torch.zeros(num_heads))
        out_features = num_heads * d + num_heads * num_value_points * 3
        self.out_proj = nn.Linear(out_features, node_dim)
        self.layer_norm = nn.LayerNorm(node_dim)

    def _apply_frame(self, pts, rotations, translations):
        """Transform local-frame points to global frame."""
        global_pts = torch.einsum('bnpc,bncd->bnpd', pts, rotations) + translations.unsqueeze(2)
        return global_pts

    def _apply_inv_frame(self, pts, rotations, translations):
        """Transform global-frame points to local frame."""
        centered = pts - translations.unsqueeze(2)
        local_pts = torch.einsum('bnpc,bndc->bnpd', centered, rotations)
        return local_pts

    def forward(self, node_repr: torch.Tensor, pair_repr: torch.Tensor, translations: torch.Tensor, rotations: torch.Tensor) -> torch.Tensor:
        """
        Args:
            node_repr:    (B, N, node_dim)
            pair_repr:    (B, N, N, pair_dim)
            translations: (B, N, 3)
            rotations:    (B, N, 3, 3)
        Returns:
            updated node representation (B, N, node_dim)
        """
        (B, N, D) = node_repr.shape
        h = self.num_heads
        d = self.head_dim
        Qp = self.num_query_points
        Vp = self.num_value_points
        x = self.layer_norm(node_repr)
        q = self.q_proj(x).view(B, N, h, d)
        k = self.k_proj(x).view(B, N, h, d)
        v = self.v_proj(x).view(B, N, h, d)
        q_pts_local = self.q_pts(x).view(B, N, h * Qp, 3)
        k_pts_local = self.k_pts(x).view(B, N, h * Qp, 3)
        v_pts_local = self.v_pts(x).view(B, N, h * Vp, 3)
        q_pts = self._apply_frame(q_pts_local, rotations, translations).view(B, N, h, Qp, 3)
        k_pts = self._apply_frame(k_pts_local, rotations, translations).view(B, N, h, Qp, 3)
        v_pts = self._apply_frame(v_pts_local, rotations, translations).view(B, N, h, Vp, 3)
        attn_scalar = torch.einsum('bihd,bjhd->bhij', q, k) / math.sqrt(d)
        q_expand = q_pts.unsqueeze(2)
        k_expand = k_pts.unsqueeze(1)
        pt_dist = ((q_expand - k_expand) ** 2).sum(dim=(-1, -2))
        w = F.softplus(self.head_weights)
        pt_attn = -(w.view(1, 1, 1, h) * pt_dist) / 2.0
        pt_attn = pt_attn.permute(0, 3, 1, 2)
        pair_bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2)
        attn = attn_scalar + pt_attn + pair_bias
        attn = F.softmax(attn, dim=-1)
        out_scalar = torch.einsum('bhij,bjhd->bihd', attn, v).reshape(B, N, h * d)
        out_pts = torch.einsum('bhij,bjhpc->bihpc', attn, v_pts)
        out_pts = out_pts.reshape(B, N, h * Vp, 3)
        out_pts_local = self._apply_inv_frame(out_pts, rotations, translations)
        out_pts_flat = out_pts_local.reshape(B, N, h * Vp * 3)
        out = torch.cat([out_scalar, out_pts_flat], dim=-1)
        return self.out_proj(out)

num_value_points = 8
seq_len = 64
batch_size = 4

        scalar_t bias = pair_bias[b * h * N * N + head * N * N + i * N + j];
        
        attn[tid * N + j] = scalar_attn + pt_attn + bias;
    }
    
    // Softmax - find max
    scalar_t max_val = attn[tid * N];
    for (int j = 1; j < N; ++j) {
        if (attn[tid * N + j] > max_val) max_val = attn[tid * N + j];
    }
    
    // Compute exp and sum
    scalar_t sum_val = 0;
    for (int j = 0; j < N; ++j) {
        attn[tid * N + j] = std::exp(attn[tid * N + j] - max_val);
        sum_val += attn[tid * N + j];
    }
    
    // Normalize
    for (int j = 0; j < N; ++j) {
        attn[tid * N + j] /= sum_val;
    }
    
    // Aggregate scalar output
    for (int dd = 0; dd < d; ++dd) {
        scalar_t out_val = 0;
        for (int j = 0; j < N; ++j) {
            out_val += attn[tid * N + j] * v[b * N * h * d + j * h * d + head * d + dd];
        }
        out_scalar[b * N * h * d + i * h * d + head * d + dd] = out_val;
    }
    
    // Aggregate point output
    for (int vp = 0; vp < Vp; ++vp) {
        for (int c = 0; c < 3; ++c) {
            scalar_t out_val = 0;
            for (int j = 0; j < N; ++j) {
                out_val += attn[tid * N + j] * v_pts[b * N * h * Vp * 3 + j * h * Vp * 3 + head * Vp * 3 + vp * 3 + c];
            }
            out_pts[b * N * h * Vp * 3 + i * h * Vp * 3 + head * Vp * 3 + vp * 3 + c] = out_val;
        }
    }
}

template <typename scalar_t>
__global__ void apply_frame_kernel(
    const scalar_t* __restrict__ pts_local,
    const scalar_t* __restrict__ rotations,
    const scalar_t* __restrict__ translations,
    scalar_t* __restrict__ pts_global,
    const int B, const int N, const int P) {
    
    const int b = blockIdx.z;
    const int n = blockIdx.y;
    const int p = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (p >= P) return;
    
    for (int c = 0; c < 3; ++c) {
        scalar_t val = 0;
        for (int d = 0; d < 3; ++d) {
            val += pts_local[b * N * P * 3 + n * P * 3 + p * 3 + d] *
                   rotations[b * N * 3 * 3 + n * 3 * 3 + d * 3 + c];
        }
        pts_global[b * N * P * 3 + n * P * 3 + p * 3 + c] = val + translations[b * N * 3 + n * 3 + c];
    }
}

template <typename scalar_t>
__global__ void apply_inv_frame_kernel(
    const scalar_t* __restrict__ pts_global,
    const scalar_t* __restrict__ rotations,
    const scalar_t* __restrict__ translations,
    scalar_t* __restrict__ pts_local,
    const int B, const int N, const int P) {
    
    const int b = blockIdx.z;
    const int n = blockIdx.y;
    const int p = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (p >= P) return;
    
    // Center by translation
    scalar_t centered[3];
    for (int c = 0; c < 3; ++c) {
        centered[c] = pts_global[b * N * P * 3 + n * P * 3 + p * 3 + c] - 
                     translations[b * N * 3 + n * 3 + c];
    }
    
    // Apply rotation^T (rotation is orthogonal, so inv = transpose)
    for (int c = 0; c < 3; ++c) {
        scalar_t val = 0;
        for (int d = 0; d < 3; ++d) {
            val += centered[d] * rotations[b * N * 3 * 3 + n * 3 * 3 + c * 3 + d];
        }
        pts_local[b * N * P * 3 + n * P * 3 + p * 3 + c] = val;
    }
}

torch::Tensor attention_fused_cuda(
    torch::Tensor q, torch::Tensor k, torch::Tensor v,
    torch::Tensor q_pts, torch::Tensor k_pts, torch::Tensor v_pts,
    torch::Tensor pair_bias, torch::Tensor head_weights) {
    
    const int B = q.size(0);
    const int N = q.size(1);
    const int h = q.size(2);
    const int d = q.size(3);
    const int Qp = q_pts.size(3);
    const int Vp = v_pts.size(3);
    const float sqrt_d = std::sqrt(static_cast<float>(d));
    
    auto out_scalar = torch::zeros({B, N, h, d}, q.options());
    auto out_pts = torch::zeros({B, N, h, Vp, 3}, v_pts.options());
    
    const int threads = 256;
    const dim3 blocks((N + threads - 1) / threads, h, B);
    const int shared_mem_size = threads * N * sizeof(float) + 2 * threads * sizeof(float);
    
    AT_DISPATCH_FLOATING_TYPES(q.scalar_type(), "attention_fused_kernel", ([&] {
        attention_fused_kernel<scalar_t><<<blocks, threads, shared_mem_size>>>(
            q.data_ptr<scalar_t>(),
            k.data_ptr<scalar_t>(),
            v.data_ptr<scalar_t>(),
            q_pts.data_ptr<scalar_t>(),
            k_pts.data_ptr<scalar_t>(),
            v_pts.data_ptr<scalar_t>(),
            pair_bias.data_ptr<scalar_t>(),
            head_weights.data_ptr<scalar_t>(),
            out_scalar.data_ptr<scalar_t>(),
            out_pts.data_ptr<scalar_t>(),
            B, N, h, d, Qp, Vp, static_cast<scalar_t>(sqrt_d)
        );
    }));
    
    return torch::cat({out_scalar.reshape({B, N, h * d}), 
                       out_pts.reshape({B, N, h * Vp * 3})}, -1);
}

class ModelNew(nn.Module):
    """
    Invariant Point Attention (IPA) from AlphaFold2's structure module.
    Optimized with custom CUDA kernels for attention and frame transformations.
    """

    def __init__(self, node_dim, pair_dim, num_heads, num_query_points, num_value_points):
        super().__init__()
        self.num_heads = num_heads
        self.head_dim = node_dim // num_heads
        self.num_query_points = num_query_points
        self.num_value_points = num_value_points
        assert node_dim % num_heads == 0
        d = self.head_dim
        self.q_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.k_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.v_proj = nn.Linear(node_dim, num_heads * d, bias=False)
        self.q_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.k_pts = nn.Linear(node_dim, num_heads * num_query_points * 3, bias=False)
        self.v_pts = nn.Linear(node_dim, num_heads * num_value_points * 3, bias=False)
        self.pair_bias = nn.Linear(pair_dim, num_heads, bias=False)
        self.head_weights = nn.Parameter(torch.zeros(num_heads))
        out_features = num_heads * d + num_heads * num_value_points * 3
        self.out_proj = nn.Linear(out_features, node_dim)
        self.layer_norm = nn.LayerNorm(node_dim)

    def forward(self, node_repr: torch.Tensor, pair_repr: torch.Tensor, 
                translations: torch.Tensor, rotations: torch.Tensor) -> torch.Tensor:
        """
        Args:
            node_repr:    (B, N, node_dim)
            pair_repr:    (B, N, N, pair_dim)
            translations: (B, N, 3)
            rotations:    (B, N, 3, 3)
        Returns:
            updated node representation (B, N, node_dim)
        """
        (B, N, D) = node_repr.shape
        h = self.num_heads
        d = self.head_dim
        Qp = self.num_query_points
        Vp = self.num_value_points
        
        # Layer norm and projections
        x = self.layer_norm(node_repr)
        q = self.q_proj(x).view(B, N, h, d)
        k = self.k_proj(x).view(B, N, h, d)
        v = self.v_proj(x).view(B, N, h, d)
        
        # Point projections
        q_pts_local = self.q_pts(x).view(B, N, h * Qp, 3)
        k_pts_local = self.k_pts(x).view(B, N, h * Qp, 3)
        v_pts_local = self.v_pts(x).view(B, N, h * Vp, 3)
        
        # Apply frame transformations using custom CUDA kernels
        q_pts = ipa_ops.apply_frame_cuda(q_pts_local, rotations, translations).view(B, N, h, Qp, 3)
        k_pts = ipa_ops.apply_frame_cuda(k_pts_local, rotations, translations).view(B, N, h, Qp, 3)
        v_pts = ipa_ops.apply_frame_cuda(v_pts_local, rotations, translations).view(B, N, h, Vp, 3)
        
        # Pair bias
        pair_bias = self.pair_bias(pair_repr).permute(0, 3, 1, 2)
        
        # Head weights
        w = F.softplus(self.head_weights)
        
        # Fused attention computation using custom CUDA kernel
        # This computes: scalar_attn + point_attn + pair_bias, then softmax, then aggregation
        out_concat = ipa_ops.attention_fused_cuda(q, k, v, q_pts, k_pts, v_pts, pair_bias, w)
        
        # Split output
        out_scalar = out_concat[:, :, :h * d].view(B, N, h, d)
        out_pts_flat = out_concat[:, :, h * d:].view(B, N, h * Vp, 3)
        
        # Apply inverse frame transformation using custom CUDA kernel
        out_pts_local = ipa_ops.apply_inv_frame_cuda(out_pts_flat, rotations, translations)
        out_pts_final = out_pts_local.reshape(B, N, h * Vp * 3)
        
        # Concatenate and project
        out = torch.cat([out_scalar.reshape(B, N, h * d), out_pts_final], dim=-1)
        return self.out_proj(out)
