import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Stein Variational Gradient Descent (SVGD) update step.
    Maintains a set of particles approximating a posterior and updates
    them via a kernel-smoothed gradient that balances fitting the target
    with repulsion between particles.  The RBF kernel bandwidth is set
    via the median heuristic.
    """

    def __init__(self, dim):
        super().__init__()
        self.dim = dim
        self.register_buffer('target_mean', torch.zeros(dim))

    def _score_fn(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -x

    def forward(self, particles: torch.Tensor) -> torch.Tensor:
        """
        Args:
            particles: (N, dim) – current particle positions
        Returns:
            phi: (N, dim) – SVGD update direction
        """
        N = particles.shape[0]
        score = self._score_fn(particles)
        diff = particles.unsqueeze(0) - particles.unsqueeze(1)
        dist_sq = (diff ** 2).sum(-1)
        median_sq = torch.median(dist_sq.view(-1)).clamp(min=1e-05)
        h = median_sq / max(torch.log(torch.tensor(float(N))).item(), 1.0)
        K = torch.exp(-dist_sq / (2.0 * h))
        grad_K = -diff / h * K.unsqueeze(-1)
        phi = (K @ score + grad_K.sum(dim=0)) / N
        return phi

            for (int j = 0; j < N; j++) {
                float k = expf(-dist_sq[i * N + j] / (2.0f * h));
                sum_k_score += k * score[j * dim + d];
                float diff = particles[i * dim + d] - particles[j * dim + d];
                sum_grad_k += (-diff / h) * k;
            }
            phi[i * dim + d] = (sum_k_score + sum_grad_k) / N;
        }
    }
}

torch::Tensor svgd_update(torch::Tensor particles) {
    int N = particles.size(0);
    int dim = particles.size(1);
    auto options = torch::TensorOptions().device(particles.device()).dtype(torch::kFloat32);
    auto dist_sq = torch::empty({N, N}, options);
    
    dim3 block(16, 16);
    dim3 grid((N + 15) / 16, (N + 15) / 16);
    compute_dist_sq<<<grid, block>>>(particles.data_ptr<float>(), dist_sq.data_ptr<float>(), N, dim);
    
    auto dist_sq_flat = dist_sq.view(-1);
    auto dist_sq_copy = dist_sq_flat.clone();
    thrust::device_ptr<float> ptr(dist_sq_copy.data_ptr<float>());
    thrust::sort(ptr, ptr + N * N);
    float median_sq = dist_sq_copy[N * N / 2];
    if (median_sq < 1e-5f) median_sq = 1e-5f;
    float logN = logf((float)N);
    float denom = fmaxf(logN, 1.0f);
    float h = median_sq / denom;
    
    auto score = -particles;
    auto phi = torch::empty({N, dim}, options);
    
    int block_size = 256;
    int grid_size = (N + block_size - 1) / block_size;
    compute_phi<<<grid_size, block_size>>>(particles.data_ptr<float>(), score.data_ptr<float>(), 
                                           dist_sq.data_ptr<float>(), phi.data_ptr<float>(), h, N, dim);
    return phi;
}
