        Args:
            x: (N, C, H, W) feature map
        Returns:
            Same shape as x

            normalized = gamma[c] * normalized + beta[c];

        // SiLU: x * sigmoid(x)
        float sigmoid_out = 1.0f / (1.0f + expf(-normalized));
        output[idx] = normalized * sigmoid_out;
    }
}
