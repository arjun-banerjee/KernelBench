import math
import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Scaled low-precision attention with explicit dequantization semantics.

    The inputs are quantized tensors plus per-head scales. This mirrors the data
    path of FP8-style scaled matmul attention even though the reference computes
    everything through explicit dequantization in PyTorch.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q_q: torch.Tensor, k_q: torch.Tensor, v_q: torch.Tensor, q_scale: torch.Tensor, k_scale: torch.Tensor, v_scale: torch.Tensor) -> torch.Tensor:
        q = q_q.float() * q_scale.view(1, 1, -1, 1)
        k = k_q.float() * k_scale.view(1, 1, -1, 1)
        v = v_q.float() * v_scale.view(1, 1, -1, 1)
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        out = torch.zeros_like(q)
        scale = 1.0 / math.sqrt(head_dim)
        for b in range(batch_size):
            for h in range(num_heads):
                for t in range(seq_len):
                    q_t = q[b, t, h]
                    k_hist = k[b, :t + 1, h]
                    v_hist = v[b, :t + 1, h]
                    scores = (k_hist * q_t.unsqueeze(0)).sum(dim=-1) * scale
                    attn = torch.softmax(scores, dim=0)
                    out[b, t, h] = (attn.unsqueeze(-1) * v_hist).sum(dim=0)
        return out

def _quantize_per_head(x: torch.Tensor):
    scale = x.abs().amax(dim=(0, 1, 3)).clamp(min=0.0001) / 127.0
    q = torch.clamp(torch.round(x / scale.view(1, 1, -1, 1)), -127, 127).to(torch.int8)
    return (q, scale.to(torch.float32))

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <stdint.h>
#include <math.h>

static __device__ __forceinline__ float block_reduce_max(float v, float* red) {
    int tid = threadIdx.x;
    red[tid] = v;
    __syncthreads();
    for (int s = blockDim.x >> 1; s > 0; s >>= 1) {
        if (tid < s) {
            float o = red[tid + s];
            red[tid] = red[tid] > o ? red[tid] : o;
        }
        __syncthreads();
    }
    return red[0];
}

static __device__ __forceinline__ float block_reduce_sum(float v, float* red) {
    int tid = threadIdx.x;
    red[tid] = v;
    __syncthreads();
    for (int s = blockDim.x >> 1; s > 0; s >>= 1) {
        if (tid < s) red[tid] += red[tid + s];
        __syncthreads();
    }
    return red[0];
}

__global__ void fp8_attn_kernel(const int8_t* __restrict__ q,
                                const int8_t* __restrict__ k,
                                const int8_t* __restrict__ v,
                                const float* __restrict__ q_scale,
                                const float* __restrict__ k_scale,
                                const float* __restrict__ v_scale,
                                float* __restrict__ out,
                                int B, int S, int H, int D,
                                float inv_sqrt_d) {
    int b = blockIdx.x;
    int h = blockIdx.y;
    int t = blockIdx.z;
    int tid = threadIdx.x;

    extern __shared__ float smem[];
    float* score = smem;          // S floats
    float* red = smem + S;        // blockDim.x floats

    const float qk_scale = q_scale[h] * k_scale[h] * inv_sqrt_d;
    const int q_base = ((b * S + t) * H + h) * D;

    float local_max = -3.402823466e38f;
    for (int i = tid; i <= t; i += blockDim.x) {
        const int k_base = ((b * S + i) * H + h) * D;
        float dot = 0.0f;
        // D is small (normally 32, aligned to 8). Keep a deterministic order.
        for (int d = 0; d < D; ++d) {
            dot += (float)q[q_base + d] * (float)k[k_base + d];
        }
        float s = dot * qk_scale;
        score[i] = s;
        local_max = local_max > s ? local_max : s;
    }

    float m = block_reduce_max(local_max, red);

    float local_sum = 0.0f;
    for (int i = tid; i <= t; i += blockDim.x) {
        float e = expf(score[i] - m);
        score[i] = e;
        local_sum += e;
    }
    float denom = block_reduce_sum(local_sum, red);
    float inv_denom = 1.0f / denom;
    float vs = v_scale[h];

    int out_base = q_base;
    for (int d = tid; d < D; d += blockDim.x) {
        float acc = 0.0f;
        for (int i = 0; i <= t; ++i) {
            const int v_idx = ((b * S + i) * H + h) * D + d;
            acc += score[i] * (float)v[v_idx];
        }
        out[out_base + d] = acc * inv_denom * vs;
    }
}

// Slightly specialized D=32 version: one warp computes/stores the 32 output channels;
// otherwise it is the same algorithm. This avoids the D-loop condition in the common case.
__global__ void fp8_attn_kernel_d32(const int8_t* __restrict__ q,
                                    const int8_t* __restrict__ k,
                                    const int8_t* __restrict__ v,
                                    const float* __restrict__ q_scale,
                                    const float* __restrict__ k_scale,
                                    const float* __restrict__ v_scale,
                                    float* __restrict__ out,
                                    int B, int S, int H,
                                    float inv_sqrt_d) {
    int b = blockIdx.x;
    int h = blockIdx.y;
    int t = blockIdx.z;
    int tid = threadIdx.x;

    extern __shared__ float smem[];
    float* score = smem;
    float* red = smem + S;

    const float qk_scale = q_scale[h] * k_scale[h] * inv_sqrt_d;
    const int q_base = ((b * S + t) * H + h) * 32;

    float local_max = -3.402823466e38f;
    for (int i = tid; i <= t; i += blockDim.x) {
        const int k_base = ((b * S + i) * H + h) * 32;
        float dot = 0.0f;
        #pragma unroll
        for (int d = 0; d < 32; ++d) {
            dot += (float)q[q_base + d] * (float)k[k_base + d];
        }
        float s = dot * qk_scale;
        score[i] = s;
        local_max = local_max > s ? local_max : s;
    }
    float m = block_reduce_max(local_max, red);

    float local_sum = 0.0f;
    for (int i = tid; i <= t; i += blockDim.x) {
        float e = expf(score[i] - m);
        score[i] = e;
        local_sum += e;
    }
    float denom = block_reduce_sum(local_sum, red);
    float coef = v_scale[h] / denom;

    if (tid < 32) {
        float acc = 0.0f;
        for (int i = 0; i <= t; ++i) {
            const int v_idx = ((b * S + i) * H + h) * 32 + tid;
            acc += score[i] * (float)v[v_idx];
        }
        out[q_base + tid] = acc * coef;
    }
}

torch::Tensor fp8_attention_forward_cuda(torch::Tensor q_q, torch::Tensor k_q, torch::Tensor v_q,
                                         torch::Tensor q_scale, torch::Tensor k_scale, torch::Tensor v_scale) {
    const int B = (int)q_q.size(0);
    const int S = (int)q_q.size(1);
    const int H = (int)q_q.size(2);
    const int D = (int)q_q.size(3);

    auto out = torch::empty({B, S, H, D}, q_q.options().dtype(torch::kFloat32));

    const dim3 grid(B, H, S);
    const int threads = 128;
    const size_t shmem = (size_t)(S + threads) * sizeof(float);
    const float inv_sqrt_d = rsqrtf((float)D);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    if (D == 32) {
        fp8_attn_kernel_d32<<<grid, threads, shmem, stream>>>(
            (const int8_t*)q_q.data_ptr<int8_t>(), (const int8_t*)k_q.data_ptr<int8_t>(), (const int8_t*)v_q.data_ptr<int8_t>(),
            q_scale.data_ptr<float>(), k_scale.data_ptr<float>(), v_scale.data_ptr<float>(),
            out.data_ptr<float>(), B, S, H, inv_sqrt_d);
    } else {
        fp8_attn_kernel<<<grid, threads, shmem, stream>>>(
            (const int8_t*)q_q.data_ptr<int8_t>(), (const int8_t*)k_q.data_ptr<int8_t>(), (const int8_t*)v_q.data_ptr<int8_t>(),
            q_scale.data_ptr<float>(), k_scale.data_ptr<float>(), v_scale.data_ptr<float>(),
            out.data_ptr<float>(), B, S, H, D, inv_sqrt_d);
    }
    return out;
}
