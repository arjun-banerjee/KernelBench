import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Low-rank expansion of compressed KV activations into multi-head K/V tensors.

    This isolates the MLA-style compressed-cache pattern: a hidden activation is
    projected into a low-rank latent space and then expanded back into separate
    key and value tensors.
    """

    def __init__(self):
        super().__init__()

    def forward(self, hidden: torch.Tensor, w_down: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, model_dim) = hidden.shape
        rank = w_down.shape[1]
        kv_heads = 4
        head_dim = w_up_k.shape[1] // kv_heads
        latent = hidden.reshape(batch_size * seq_len, model_dim) @ w_down
        k = latent @ w_up_k
        v = latent @ w_up_v
        k = k.view(batch_size, seq_len, kv_heads, head_dim)
        v = v.view(batch_size, seq_len, kv_heads, head_dim)
        return torch.stack((k, v), dim=2)

torch::Tensor mla_forward(torch::Tensor hidden, torch::Tensor w_down, torch::Tensor w_up_k, torch::Tensor w_up_v);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>
#include <cuda.h>
#include <cuda_runtime.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

__inline__ __device__ float warp_reduce_sum(float v) {
    v += __shfl_down_sync(0xffffffff, v, 16);
    v += __shfl_down_sync(0xffffffff, v, 8);
    v += __shfl_down_sync(0xffffffff, v, 4);
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

// One CUDA block computes one token row.  The low-rank latent (rank=48 in the
// benchmark) is materialized only in shared memory, then immediately expanded
// to the interleaved stack output [B,S,2,4,D].
__global__ __launch_bounds__(256, 2)
void mla_fused_kernel(const float* __restrict__ hidden,
                      const float* __restrict__ w_down,
                      const float* __restrict__ w_up_k,
                      const float* __restrict__ w_up_v,
                      float* __restrict__ out,
                      int rows, int model_dim, int rank, int ncols) {
    int row = blockIdx.x;
    if (row >= rows) return;

    int tid = threadIdx.x;
    int lane = tid & 31;
    int warp = tid >> 5;
    constexpr int NWARPS = 8;

    extern __shared__ float latent[];
    const float* h = hidden + (long long)row * model_dim;

    // Compute latent[r] = sum_m hidden[m] * w_down[m,r].
    // Eight warps work on different rank columns concurrently.
    for (int r = warp; r < rank; r += NWARPS) {
        float acc = 0.0f;
        for (int m = lane; m < model_dim; m += WARP_SIZE) {
            acc = fmaf(h[m], w_down[(long long)m * rank + r], acc);
        }
        acc = warp_reduce_sum(acc);
        if (lane == 0) latent[r] = acc;
    }

    __syncthreads();

    // Expand to K and V.  Flat contiguous layout of [rows,2,ncols] is identical
    // to [B,S,2,4,head_dim] because ncols = 4*head_dim.
    float* o = out + (long long)row * (2 * ncols);
    for (int idx = tid; idx < 2 * ncols; idx += blockDim.x) {
        int is_v = idx >= ncols;
        int c = idx - is_v * ncols;
        const float* wu = is_v ? w_up_v : w_up_k;
        float acc = 0.0f;
#pragma unroll
        for (int r = 0; r < 48; ++r) {
            if (r < rank) acc = fmaf(latent[r], wu[(long long)r * ncols + c], acc);
        }
        o[idx] = acc;
    }
}

torch::Tensor mla_forward(torch::Tensor hidden, torch::Tensor w_down, torch::Tensor w_up_k, torch::Tensor w_up_v) {
    TORCH_CHECK(hidden.is_cuda(), "hidden must be CUDA");
    TORCH_CHECK(w_down.is_cuda() && w_up_k.is_cuda() && w_up_v.is_cuda(), "weights must be CUDA");
    TORCH_CHECK(hidden.scalar_type() == at::kFloat, "only fp32 supported");

    const int B = (int)hidden.size(0);
    const int S = (int)hidden.size(1);
    const int M = (int)hidden.size(2);
    const int R = (int)w_down.size(1);
    const int N = (int)w_up_k.size(1);
    const int rows = B * S;
    const int head_dim = N / 4;

    auto out = torch::empty({B, S, 2, 4, head_dim}, hidden.options());

    dim3 block(256);
    dim3 grid(rows);
    size_t smem = (size_t)R * sizeof(float);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    mla_fused_kernel<<<grid, block, smem, stream>>>(
        hidden.data_ptr<float>(), w_down.data_ptr<float>(), w_up_k.data_ptr<float>(), w_up_v.data_ptr<float>(),
        out.data_ptr<float>(), rows, M, R, N);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
"""

_mla_ext = load_inline(
    name="mla_lora_fused_ext_v1",
    cpp_sources=_cpp_src,
    cuda_sources=_cuda_src,
    functions=["mla_forward"],
    extra_cflags=["-O3"],
    extra_cuda_cflags=["-O3"],
    verbose=False,
)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, hidden: torch.Tensor, w_down: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        return _mla_ext.mla_forward(hidden, w_down, w_up_k, w_up_v)
