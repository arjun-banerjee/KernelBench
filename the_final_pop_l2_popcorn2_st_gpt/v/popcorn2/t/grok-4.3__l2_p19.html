import torch
import torch.nn as nn

class Model(nn.Module):
    """
    1D depthwise separable convolution block: pointwise Conv1d (channel mixing), depthwise Conv1d (spatial),
    GELU activation, and residual addition. Used in genomic foundation models (Enformer) for efficient
    sequence processing of DNA/RNA.
    """

    def __init__(self, channels, expanded_channels, kernel_size):
        super(Model, self).__init__()
        self.pointwise = nn.Conv1d(channels, expanded_channels, 1)
        self.depthwise = nn.Conv1d(expanded_channels, expanded_channels, kernel_size, padding=kernel_size // 2, groups=expanded_channels)
        self.activation = nn.GELU()
        self.project_back = nn.Conv1d(expanded_channels, channels, 1)

        residual = x
        x = self.pointwise(x)
        x = self.depthwise(x)
        x = self.activation(x)
        x = self.project_back(x)
        x = x + residual
        return x

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void fused_depthwise_gelu_project_add_kernel(
    const float* __restrict__ x,          // [B, C, L]
    const float* __restrict__ pw_weight,  // [E, C, 1]
    const float* __restrict__ pw_bias,    // [E]
    const float* __restrict__ dw_weight,  // [E, 1, K]
    const float* __restrict__ dw_bias,    // [E]
    const float* __restrict__ pb_weight,  // [C, E, 1]
    const float* __restrict__ pb_bias,    // [C]
    float* __restrict__ out,              // [B, C, L]
    int B, int C, int E, int L, int K, int pad
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * C * L;
    if (idx >= total) return;

    int l = idx % L;
    int c_out = (idx / L) % C;
    int b = idx / (L * C);

    // residual
    float res = x[b * C * L + c_out * L + l];

    // Pointwise conv for all expanded channels at this position
    float acc_e[512]; // assume E <=512
    for (int e = 0; e < E; ++e) {
        float sum = pw_bias[e];
        for (int c = 0; c < C; ++c) {
            sum += pw_weight[e * C + c] * x[b * C * L + c * L + l];
        }
        // depthwise at this e, position l
        float dw_sum = dw_bias[e];
        for (int k = 0; k < K; ++k) {
            int in_l = l + k - pad;
            if (in_l >= 0 && in_l < L) {
                float pw_val = pw_bias[e];
                for (int cc = 0; cc < C; ++cc) {
                    pw_val += pw_weight[e * C + cc] * x[b * C * L + cc * L + in_l];
                }
                dw_sum += dw_weight[e * K + k] * pw_val;
            }
        }
        // GELU
        float val = dw_sum;
        val = 0.5f * val * (1.0f + tanhf(0.79788456f * (val + 0.044715f * val * val * val)));
        acc_e[e] = val;
    }

    // project back
    float final = pb_bias[c_out];
    for (int e = 0; e < E; ++e) {
        final += pb_weight[c_out * E + e] * acc_e[e];
    }
    final += res;
    out[idx] = final;
}
