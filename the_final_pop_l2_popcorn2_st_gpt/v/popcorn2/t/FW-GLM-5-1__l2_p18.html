Multi-GPU: each rank keeps one column block; output is concat along last dim (matches full input).
Single-GPU: returns x unchanged.

import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_rank, get_world_size, is_distributed_run, maybe_init_process_group
dim = 80

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        if not is_distributed_run() or ws == 1:
            return x * self.scale

        gathered = [torch.empty_like(local) for _ in range(ws)]
        dist.all_gather(gathered, local)

import torch.distributed as dist
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_extract_scale_module = load_inline(

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template<typename scalar_t>
__global__ void extract_scale_kernel(
    const scalar_t* __restrict__ input,
    scalar_t* __restrict__ output,
    float scale,
    int64_t total_elements,
    int64_t input_last_dim,
    int64_t chunk_size,
    int64_t offset) {
    
    int64_t idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total_elements) return;
    
    // Compute position in output tensor
    int64_t last_dim_pos = idx % chunk_size;
    int64_t other_dims = idx / chunk_size;
    
    // Map to position in input tensor
    int64_t input_idx = other_dims * input_last_dim + offset + last_dim_pos;
    output[idx] = input[input_idx] * scalar_t(scale);
}

torch::Tensor extract_scale_cuda(torch::Tensor input, int rank, int world_size, float scale) {
    TORCH_CHECK(input.is_cuda(), "Input must be a CUDA tensor");
    int64_t last_dim = input.size(-1);
    TORCH_CHECK(last_dim % world_size == 0, "Last dimension must be divisible by world_size");
    
    int64_t chunk_size = last_dim / world_size;
    int64_t offset = rank * chunk_size;
    
    // Create output tensor with reduced last dimension
    auto sizes = input.sizes().vec();
    sizes.back() = chunk_size;
    auto output = torch::empty(sizes, input.options());
    
    int64_t total_elements = output.numel();
    if (total_elements == 0) return output;

            output.data_ptr<scalar_t>(),
            scale,
            total_elements,
            last_dim,
            chunk_size,
            offset

    }));
    
    return output;
}

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        
        # Single GPU or non-distributed: simple scaling
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        
        # Fused: extract local chunk from x and scale it
        local = _extract_scale_module.extract_scale_cuda(x, rank, ws, self.scale.item())
        
        # Pre-allocate output tensor (avoids torch.cat overhead)
        output = torch.empty(x.shape, dtype=x.dtype, device=x.device)
        output_chunks = list(output.chunk(ws, dim=-1))  # Contiguous views
        
        # All-gather into pre-allocated output (avoids concatenation)
        dist.all_gather(output_chunks, local)
        
        return output
