import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Sparse message passing with CSR adjacency.

    For each destination node, aggregate weighted source-node features from the
    incoming edges in its CSR row. This is a core primitive in sparse graph
    neural network layers and graph convolution operators.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, col_idx: torch.Tensor, edge_weight: torch.Tensor, node_feat: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        feat_dim = node_feat.shape[1]
        out = torch.zeros(num_nodes, feat_dim, dtype=node_feat.dtype, device=node_feat.device)
        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end > start:
                src = col_idx[start:end].long()
                weights = edge_weight[start:end].unsqueeze(-1)
                out[dst] = (node_feat[src] * weights).sum(dim=0)
        return out

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template <typename scalar_t, typename index_t>
__global__ void csr_spmm_kernel_vec(
    const index_t* __restrict__ row_ptr,
    const index_t* __restrict__ col_idx,
    const scalar_t* __restrict__ edge_weight,
    const scalar_t* __restrict__ node_feat,
    scalar_t* __restrict__ output,
    int num_nodes,
    int feat_dim) {
    
    const int VEC = 4;  // Process 4 features at a time
    const int dst = blockIdx.x;
    const int feat_offset = threadIdx.x * VEC;
    
    if (dst >= num_nodes || feat_offset >= feat_dim) return;
    
    index_t start = row_ptr[dst];
    index_t end = row_ptr[dst + 1];
    
    // Use registers for accumulation
    scalar_t sum[VEC] = {scalar_t(0), scalar_t(0), scalar_t(0), scalar_t(0)};
    
    // Accumulate contributions from incoming edges
    for (index_t e = start; e < end; e++) {
        index_t src = col_idx[e];
        scalar_t w = edge_weight[e];
        const scalar_t* feat_ptr = node_feat + src * feat_dim + feat_offset;
        
        #pragma unroll
        for (int v = 0; v < VEC; v++) {
            sum[v] += w * feat_ptr[v];
        }
    }
    
    // Write results
    scalar_t* out_ptr = output + dst * feat_dim + feat_offset;
    #pragma unroll
    for (int v = 0; v < VEC; v++) {
        out_ptr[v] = sum[v];
    }
}

torch::Tensor csr_spmm_cuda(
    torch::Tensor row_ptr,
    torch::Tensor col_idx,
    torch::Tensor edge_weight,
    torch::Tensor node_feat) {
    
    int num_nodes = row_ptr.numel() - 1;
    int feat_dim = node_feat.size(1);
    
    auto output = torch::zeros({num_nodes, feat_dim}, node_feat.options());
    
    // Each block handles one destination node, threads handle features
    const int threads = (feat_dim + 3) / 4;  // 4 features per thread
    const int blocks = num_nodes;

    auto col_idx_cont = col_idx.contiguous().to(torch::kInt64);

    auto node_feat_cont = node_feat.contiguous();
