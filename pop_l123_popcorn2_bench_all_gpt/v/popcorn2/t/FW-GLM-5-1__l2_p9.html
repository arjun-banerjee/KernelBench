import torch
import torch.nn as nn
import math

class Model(nn.Module):
    """
    Batched Metropolis-Hastings MCMC step.  Given a current state and a
    log-probability function (here a Gaussian target for benchmarking),
    proposes a new state via a Gaussian random walk, computes the
    acceptance ratio, and stochastically accepts or rejects.
    """

    def __init__(self, dim, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))
        self.register_buffer('sigma', torch.ones(dim))

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (((x - self.mu) / self.sigma) ** 2).sum(dim=-1)

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, dim) – current chain states
        Returns:
            (x_new, accepted):
                x_new:    (B, dim) – states after accept/reject
                accepted: (B,)    – boolean mask of accepted proposals
        """
        proposal = x + self.step_size * torch.randn_like(x)
        log_alpha = self._log_prob(proposal) - self._log_prob(x)
        log_u = torch.log(torch.rand(x.shape[0], device=x.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        x_new = torch.where(accepted.unsqueeze(-1), proposal, x)
        return (x_new, accepted)

__global__ void mh_step_kernel(
    const float* __restrict__ x,
    const float* __restrict__ mu,
    const float* __restrict__ sigma,
    const float* __restrict__ randn,
    const float* __restrict__ rand_uniform,
    float step_size,
    int dim,
    int num_chains,
    float* __restrict__ x_new,
    bool* __restrict__ accepted
) {
    int chain_idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (chain_idx >= num_chains) return;
    
    const float* x_chain = x + chain_idx * dim;
    const float* randn_chain = randn + chain_idx * dim;
    float* x_new_chain = x_new + chain_idx * dim;
    
    // Compute log_prob(x) and generate proposal simultaneously
    float log_prob_x = 0.0f;
    float log_prob_proposal = 0.0f;
    
    for (int i = 0; i < dim; i++) {
        // Generate proposal
        float proposal_val = x_chain[i] + step_size * randn_chain[i];
        
        // Compute log_prob contributions
        float diff_x = (x_chain[i] - mu[i]) / sigma[i];
        float diff_proposal = (proposal_val - mu[i]) / sigma[i];
        
        log_prob_x += diff_x * diff_x;
        log_prob_proposal += diff_proposal * diff_proposal;
        
        // Store proposal
        x_new_chain[i] = proposal_val;
    }
    
    log_prob_x = -0.5f * log_prob_x;
    log_prob_proposal = -0.5f * log_prob_proposal;
    
    // Compute log_alpha
    float log_alpha = log_prob_proposal - log_prob_x;
    
    // Get uniform random number for acceptance
    float u = rand_uniform[chain_idx];

    accepted[chain_idx] = accept;
    
    // If rejected, restore original x
    if (!accept) {

    }
}
