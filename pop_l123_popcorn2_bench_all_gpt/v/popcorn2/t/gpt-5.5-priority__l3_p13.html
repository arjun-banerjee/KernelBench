import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Conditional Variational Autoencoder (CVAE) with full encoder–decoder
    forward pass and loss computation.  The encoder conditions on both
    input x and label y; the decoder conditions on z and y.  Returns
    reconstruction loss + KL divergence.
    """

    def __init__(self, input_dim, label_dim, hidden_dim, latent_dim):
        super().__init__()
        self.enc = nn.Sequential(nn.Linear(input_dim + label_dim, hidden_dim), nn.ReLU())
        self.enc_mu = nn.Linear(hidden_dim, latent_dim)
        self.enc_logvar = nn.Linear(hidden_dim, latent_dim)
        self.dec = nn.Sequential(nn.Linear(latent_dim + label_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim))

    def forward(self, x: torch.Tensor, y: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim)  – input data
            y: (B, label_dim)  – conditioning label (one-hot)
        Returns:
            (loss, recon_loss, kl_loss):
                loss:       scalar – total CVAE loss
                recon_loss: scalar
                kl_loss:    scalar
        """
        h = self.enc(torch.cat([x, y], dim=-1))
        mu = self.enc_mu(h)
        logvar = self.enc_logvar(h)
        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)
        x_recon = self.dec(torch.cat([z, y], dim=-1))
        recon_loss = F.mse_loss(x_recon, x, reduction='sum') / x.shape[0]
        kl_loss = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum() / x.shape[0]
        loss = recon_loss + kl_loss
        return (loss, recon_loss, kl_loss)

    int warp = (blockIdx.x * blockDim.x + tid) >> 5;
    int N = B * H;
    if (warp >= N) return;
    int b = warp / H;
    int oh = warp - b * H;
    int K = L + Y;
    const float* wrow = w + oh * K;
    float sum = (lane == 0) ? bias[oh] : 0.0f;
    for (int k = lane; k < K; k += 32) {
        float a;
        if (k < L) {
            int ii = b * L + k;
            a = mu[ii] + expf(0.5f * lv[ii]) * eps[ii];
        } else {
            a = y[b * Y + (k - L)];
        }
        sum += a * wrow[k];
    }
    sum = warp_sum(sum);
    if (lane == 0) dh[warp] = sum > 0.0f ? sum : 0.0f;
}

__global__ void loss_kernel(const float* __restrict__ x,
                            const float* __restrict__ mu, const float* __restrict__ lv,
                            const float* __restrict__ dh,
                            const float* __restrict__ w2, const float* __restrict__ b2,
                            float* __restrict__ partial,
                            int B, int I, int H, int L, int total_items) {
    __shared__ float smse[8];
    __shared__ float skl[8];
    int tid = threadIdx.x;
    int lane = tid & 31;
    int wid = tid >> 5;
    int item = blockIdx.x * (blockDim.x >> 5) + wid;
    float mse = 0.0f, kl = 0.0f;
    if (item < B * I) {
        int b = item / I;
        int j = item - b * I;
        const float* drow = dh + b * H;
        const float* wrow = w2 + j * H;
        float out = (lane == 0) ? b2[j] : 0.0f;
        for (int k = lane; k < H; k += 32) out += drow[k] * wrow[k];
        out = warp_sum(out);
        if (lane == 0) { float diff = out - x[item]; mse = diff * diff; }
    } else if (item < total_items) {
        int kidx = item - B * I;
        if (lane == 0) {
            float m = mu[kidx];
            float v = lv[kidx];
            kl = -0.5f * (1.0f + v - m * m - expf(v));
        }
    }
    if (lane == 0) { smse[wid] = mse; skl[wid] = kl; }
    __syncthreads();
    if (wid == 0) {
        float m = (lane < 8) ? smse[lane] : 0.0f;
        float k = (lane < 8) ? skl[lane] : 0.0f;
        m = warp_sum(m); k = warp_sum(k);
        if (lane == 0) { partial[blockIdx.x] = m; partial[gridDim.x + blockIdx.x] = k; }
    }
}

__global__ void final_kernel(const float* __restrict__ partial, float* __restrict__ out, int nblocks, int B) {
    extern __shared__ float sh[];
    float* sm = sh;
    float* sk = sh + blockDim.x;
    int tid = threadIdx.x;
    float mse = 0.0f, kl = 0.0f;
    for (int i = tid; i < nblocks; i += blockDim.x) {
        mse += partial[i];
        kl += partial[nblocks + i];
    }
    sm[tid] = mse;
    sk[tid] = kl;
    __syncthreads();
    for (int s = blockDim.x >> 1; s > 0; s >>= 1) {
        if (tid < s) { sm[tid] += sm[tid + s]; sk[tid] += sk[tid + s]; }
        __syncthreads();
    }
    if (tid == 0) {
        float r = sm[0] / (float)B;
        float k = sk[0] / (float)B;
        out[0] = r + k;
        out[1] = r;
        out[2] = k;
    }
}
