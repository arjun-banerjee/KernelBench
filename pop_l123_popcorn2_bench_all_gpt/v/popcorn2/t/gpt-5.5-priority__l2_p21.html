import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Triangular multiplicative update – incoming edges.
    Mirror of the outgoing variant: updates pair (i,j) by aggregating over
    the shared index k via edges (k,i) and (k,j).
    """

    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            pair_repr: (B, N, N, pair_dim)
            pair_mask: (B, N, N)
        Returns:
            updated pair representation (B, N, N, pair_dim)
        """
        mask = pair_mask.unsqueeze(-1)
        x = self.layer_norm_in(pair_repr)
        a = self.proj_a(x) * torch.sigmoid(self.gate_a(x)) * mask
        b = self.proj_b(x) * torch.sigmoid(self.gate_b(x)) * mask
        out = torch.einsum('bkih,bkjh->bijh', a, b)
        out = self.layer_norm_out(out)
        out = self.out_proj(out)
        out = torch.sigmoid(self.out_gate(pair_repr)) * out
        return out

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cpp_src = r'''
#include <torch/extension.h>
torch::Tensor tri_incoming_cuda(torch::Tensor a, torch::Tensor b);
torch::Tensor tri_incoming(torch::Tensor a, torch::Tensor b) { return tri_incoming_cuda(a,b); }
'''

cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>

// Computes out[b,i,j,h] = sum_k a[b,k,i,h] * b[b,k,j,h]
// Optimized for the benchmark sizes (hidden_dim=48, N around 64).
__global__ void tri_kernel(const float* __restrict__ a,
                           const float* __restrict__ b,
                           float* __restrict__ out,
                           int B, int N, int H) {
    int h = blockIdx.x * blockDim.x + threadIdx.x;
    int j = blockIdx.y * blockDim.y + threadIdx.y;
    int bi = blockIdx.z;
    int batch = bi / N;
    int i = bi - batch * N;
    if (h >= H || j >= N) return;

    float acc = 0.0f;
    const long strideK = (long)N * H;
    const long baseA = ((long)batch * N * N + i) * H + h;
    const long baseB = ((long)batch * N * N + j) * H + h;

    #pragma unroll 8
    for (int k = 0; k < N; ++k) {
        acc = fmaf(a[baseA + (long)k * strideK], b[baseB + (long)k * strideK], acc);
    }
    out[((long)batch * N * N + (long)i * N + j) * H + h] = acc;
}

torch::Tensor tri_incoming_cuda(torch::Tensor a, torch::Tensor b) {
    const int B = (int)a.size(0);
    const int N = (int)a.size(1);
    const int H = (int)a.size(3);
    auto out = torch::empty({B, N, N, H}, a.options());
    dim3 block(16, 16);
    dim3 grid((H + 15) / 16, (N + 15) / 16, B * N);
    tri_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>(
        a.data_ptr<float>(), b.data_ptr<float>(), out.data_ptr<float>(), B, N, H);
    return out;
}
'''

class ModelNew(nn.Module):
    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        mask = pair_mask.unsqueeze(-1)
        x = self.layer_norm_in(pair_repr)
        a = self.proj_a(x) * torch.sigmoid(self.gate_a(x)) * mask
        b = self.proj_b(x) * torch.sigmoid(self.gate_b(x)) * mask
        out = _ext.tri_incoming(a.contiguous(), b.contiguous())
