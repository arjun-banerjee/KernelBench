template <int HEAD_SIZE>
__global__ void attention_kernel(
    const __half* __restrict__ q,
    const __half* __restrict__ k,
    const __half* __restrict__ v,
    __half* __restrict__ out,
    const int batch_size,
    const int num_heads,
    const int num_kv_heads,
    const int seq_len,
    const float scale
) {

    const int batch_idx = blockIdx.y;

    const int num_threads = blockDim.x;

    extern __shared__ char smem_raw[];
    __half* k_smem = reinterpret_cast<__half*>(smem_raw);
    __half* v_smem = k_smem + seq_len * HEAD_SIZE;

    const int k_offset = (batch_idx * num_kv_heads + kv_head_idx) * seq_len * HEAD_SIZE;
    const int out_offset = q_offset;

        v_smem[i] = v[k_offset + i];
