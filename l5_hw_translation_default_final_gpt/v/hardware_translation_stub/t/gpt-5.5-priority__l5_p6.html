#include <ATen/cuda/CUDAContext.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>
#include <cstdint>

__global__ __launch_bounds__(512, 2)
void scale_half2_vec_kernel(const half* __restrict__ x, half* __restrict__ y, int64_t n, float scale) {
    int64_t n2 = n >> 1;
    const half2* __restrict__ x2 = reinterpret_cast<const half2*>(x);
    half2* __restrict__ y2 = reinterpret_cast<half2*>(y);
    half2 s = __float2half2_rn(scale);
    int64_t tid = (int64_t)blockIdx.x * blockDim.x + threadIdx.x;
    int64_t stride = (int64_t)blockDim.x * gridDim.x;
    int64_t limit = n2 & ~3LL;
    for (int64_t i = tid * 4; i < limit; i += stride * 4) {
        half2 v0 = x2[i + 0];
        half2 v1 = x2[i + 1];
        half2 v2 = x2[i + 2];
        half2 v3 = x2[i + 3];
        y2[i + 0] = __hmul2(v0, s);
        y2[i + 1] = __hmul2(v1, s);
        y2[i + 2] = __hmul2(v2, s);
        y2[i + 3] = __hmul2(v3, s);
    }
    int64_t start = limit << 1;
    for (int64_t i = start + tid; i < n; i += stride) {
        y[i] = __float2half(__half2float(x[i]) * scale);
    }
}

torch::Tensor scale_half(torch::Tensor x, double scale_d) {
    TORCH_CHECK(x.is_cuda(), "input must be CUDA");
    TORCH_CHECK(x.scalar_type() == at::ScalarType::Half, "input must be float16");
    TORCH_CHECK(x.is_contiguous(), "input must be contiguous");
    auto y = torch::empty_like(x);
    const int64_t n = x.numel();
    if (n == 0) return y;
    const int threads = 512;

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

        reinterpret_cast<const half*>(x.data_ptr<at::Half>()),
        reinterpret_cast<half*>(y.data_ptr<at::Half>()),

}
