__global__ void matmul_16x4096_kernel(const half* __restrict__ x,
                                      const half* __restrict__ w,
                                      half* __restrict__ y,
                                      int K, int N) {
    int n = blockIdx.x * 16 + threadIdx.x;   // 0..15 in tile
    int m = threadIdx.y;                     // 0..15 rows
    if (m >= 16 || n >= N) return;

    const half* xrow = x + m * K;
    const half* wcol = w + n;
    float acc = 0.0f;

    // K is 4096 in the benchmark; unroll by 4 for ILP.  w is row-major [K,N].
    int k = 0;
#pragma unroll 4
    for (; k + 3 < K; k += 4) {
        acc = fmaf(__half2float(xrow[k + 0]), __half2float(wcol[(k + 0) * N]), acc);
        acc = fmaf(__half2float(xrow[k + 1]), __half2float(wcol[(k + 1) * N]), acc);
        acc = fmaf(__half2float(xrow[k + 2]), __half2float(wcol[(k + 2) * N]), acc);
        acc = fmaf(__half2float(xrow[k + 3]), __half2float(wcol[(k + 3) * N]), acc);
    }
    for (; k < K; ++k) acc = fmaf(__half2float(xrow[k]), __half2float(wcol[k * N]), acc);
    y[m * N + n] = __float2half_rn(acc);
}

torch::Tensor matmul_forward(torch::Tensor x, torch::Tensor w) {
    const int K = (int)x.size(1);
    const int N = (int)w.size(1);
    auto y = torch::empty({x.size(0), w.size(1)}, x.options());

}
