static __device__ __forceinline__ float sumsq_u4(uint4 v) {
    union CV { uint4 u; half2 h2[4]; } cv;
    cv.u = v;
    float s = 0.0f;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        float2 f = __half22float2(cv.h2[i]);
        s = fmaf(f.x, f.x, s);
        s = fmaf(f.y, f.y, s);
    }
    return s;
}

static __device__ __forceinline__ uint4 norm_u4(uint4 xv, uint4 wv, float scale) {
    union CV { uint4 u; half2 h2[4]; } x, w, o;
    x.u = xv; w.u = wv;
    #pragma unroll
    for (int i = 0; i < 4; ++i) {
        float2 xf = __half22float2(x.h2[i]);
        float2 wf = __half22float2(w.h2[i]);
        o.h2[i] = __floats2half2_rn((xf.x * scale) * wf.x, (xf.y * scale) * wf.y);
    }
    return o.u;
}

__global__ __launch_bounds__(256, 4)
void rms_norm_4096_fp16_kernel(const half* __restrict__ x,
                               const half* __restrict__ weight,
                               half* __restrict__ out,
                               float eps) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const half* __restrict__ row_x = x + ((size_t)row << 12);  // *4096
    half* __restrict__ row_o = out + ((size_t)row << 12);

    const uint4* __restrict__ x4 = reinterpret_cast<const uint4*>(row_x);
    const uint4* __restrict__ w4 = reinterpret_cast<const uint4*>(weight);
    uint4* __restrict__ o4 = reinterpret_cast<uint4*>(row_o);

    uint4 a0 = x4[tid];
    uint4 a1 = x4[tid + 256];
    float sum = sumsq_u4(a0) + sumsq_u4(a1);

    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        sum += __shfl_down_sync(0xffffffff, sum, off);
    }

    __shared__ float warp_sums[8];
    __shared__ float scale_s;
    const int lane = tid & 31;
    const int warp = tid >> 5;
    if (lane == 0) warp_sums[warp] = sum;
    __syncthreads();

    if (warp == 0) {
        float v = (lane < 8) ? warp_sums[lane] : 0.0f;
        #pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            v += __shfl_down_sync(0xffffffff, v, off);
