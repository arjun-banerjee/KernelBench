        super(ModelNew, self).__init__()
        self.batch = batch
        self.k = k
        self.n = n
        self.alpha = alpha
        self.beta = beta
        self.B = nn.Parameter(torch.randn(batch, k, n, device='cuda', dtype=torch.float16))
    
    def forward(self, A):
        A16 = A.to(torch.float16)
        return module.batched_gemm_cuda(A16, self.B, self.batch, self.k, self.n, self.alpha, self.beta)
