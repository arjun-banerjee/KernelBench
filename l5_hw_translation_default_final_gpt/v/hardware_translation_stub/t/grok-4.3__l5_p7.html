#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_bf16.h>
#include <c10/cuda/CUDAGuard.h>

#define TILE_M 16
#define TILE_N 128
#define TILE_K 64
#define WARP_SIZE 32

__global__ void __launch_bounds__(256) marlin_h100_kernel(
    const half* __restrict__ A,
    const int4* __restrict__ B,
    half* __restrict__ C,
    const half* __restrict__ scales,
    int M, int N, int K, int group_size) {

    __shared__ half As[TILE_M][TILE_K + 4];
    __shared__ half Bs[TILE_K][TILE_N + 4];

    int bx = blockIdx.x;
    int by = blockIdx.y;
    int tx = threadIdx.x;
    int ty = threadIdx.y;

    int row = by * TILE_M + ty;
    int col = bx * TILE_N + tx;

    half sum = __float2half(0.0f);

    for (int t = 0; t < (K + TILE_K - 1) / TILE_K; ++t) {
        // Load A tile
        if (row < M && (t * TILE_K + tx % TILE_K) < K) {
            As[ty][tx % TILE_K] = A[row * K + t * TILE_K + (tx % TILE_K)];
        } else {
            As[ty][tx % TILE_K] = __float2half(0.0f);
        }

        // Load B tile (quantized - simplified dequant path)
        int bk = t * TILE_K + ty;
        if (bk < K && col < N) {
            int group = bk / group_size;
            half w = __float2half(0.5f);
            Bs[ty][tx] = w * scales[group * N + col];
        } else {
            Bs[ty][tx] = __float2half(0.0f);
        }
        __syncthreads();

        #pragma unroll
        for (int k = 0; k < TILE_K; ++k) {
            sum = __hadd(sum, __hmul(As[ty][k], Bs[k][tx]));
        }
        __syncthreads();
    }

    if (row < M && col < N) {
        C[row * N + col] = sum;
    }
}

    torch::Tensor A,
    torch::Tensor B_q,
