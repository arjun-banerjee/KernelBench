template <typename T>
__device__ __forceinline__ float to_float(T val) {
  return static_cast<float>(val);
}

template <>
__device__ __forceinline__ float to_float<half>(half val) {
  return __half2float(val);
}

template <typename scalar_t, int VEC_SIZE, int NUM_DIMS>
__global__ void rms_norm_kernel(
    scalar_t* __restrict__ out,
    const scalar_t* __restrict__ input,
    const int64_t input_stride_d2,
    const int64_t input_stride_d3,
    const int64_t input_stride_d4,
    const int64_t input_shape_d2,
    const int64_t input_shape_d3,
    const scalar_t* __restrict__ weight,
    const float epsilon, const int num_tokens, const int hidden_size) {
  __shared__ float s_variance;
  float variance = 0.0f;
  const scalar_t* input_row;
  if constexpr (NUM_DIMS == 2) {
    input_row = input + blockIdx.x * input_stride_d2;
  } else if constexpr (NUM_DIMS == 3) {
    int batch_idx = blockIdx.x / input_shape_d2;
    int head_idx = blockIdx.x % input_shape_d2;
    input_row = input + batch_idx * input_stride_d3 + head_idx * input_stride_d2;
  } else if constexpr (NUM_DIMS == 4) {
    int batch_idx = blockIdx.x / (input_shape_d3 * input_shape_d2);
    int remaining = blockIdx.x % (input_shape_d3 * input_shape_d2);
    int seq_idx = remaining / input_shape_d2;
    int head_idx = remaining % input_shape_d2;
    input_row = input + batch_idx * input_stride_d4 +
                seq_idx * input_stride_d3 + head_idx * input_stride_d2;
  }

  for (int i = threadIdx.x; i < hidden_size; i += blockDim.x) {
    float x = to_float(input_row[i]);
    variance += x * x;
  }

  using BlockReduce = cub::BlockReduce<float, 256>;
  __shared__ typename BlockReduce::TempStorage reduceStore;
  variance = BlockReduce(reduceStore).Reduce(variance, cub::Sum{}, blockDim.x);

  if (threadIdx.x == 0) {
    s_variance = rsqrtf(variance / hidden_size + epsilon);
  }
  __syncthreads();

  scalar_t* out_row = out + blockIdx.x * hidden_size;
  const half* v_in = reinterpret_cast<const half*>(input_row);
  const half* v_w = reinterpret_cast<const half*>(weight);
  half* v_out = reinterpret_cast<half*>(out_row);
  const float scale = s_variance;
  
  for (int i = threadIdx.x; i < hidden_size / 8; i += blockDim.x) {
    float4 in4 = *reinterpret_cast<const float4*>(v_in + i * 8);
    float4 w4 = *reinterpret_cast<const float4*>(v_w + i * 8);
    half2* in_h2 = reinterpret_cast<half2*>(&in4);
    half2* w_h2 = reinterpret_cast<half2*>(&w4);
    half2 out_h2[4];

    for (int j = 0; j < 4; j++) {
      float x = __half2float(in_h2[j].x);
      float w = __half2float(w_h2[j].x);

      out_h2[j].y = __float2half(x * scale * w);
    }
    *reinterpret_cast<float4*>(v_out + i * 8) = *reinterpret_cast<float4*>(out_h2);
  }
}
