#include <torch/extension.h>

#define WARP_SIZE 32

template <int HEAD_SIZE>
__global__ void attention_kernel(
    const __half* __restrict__ q,
    const __half* __restrict__ k,
    const __half* __restrict__ v,
    __half* __restrict__ out,
    const int batch_size,
    const int num_heads,
    const int num_kv_heads,
    const int seq_len,
    const float scale) {

    const int head_idx = blockIdx.x;
    const int batch_idx = blockIdx.y;
    const int tid = threadIdx.x;

    extern __shared__ __half smem[];
    __half* k_smem = smem;
    __half* v_smem = smem + seq_len * HEAD_SIZE;

    const int kv_head_idx = head_idx / (num_heads / num_kv_heads);
    const int q_offset = (batch_idx * num_heads + head_idx) * seq_len * HEAD_SIZE;
    const int kv_offset = (batch_idx * num_kv_heads + kv_head_idx) * seq_len * HEAD_SIZE;

    for (int i = tid; i < seq_len * HEAD_SIZE; i += blockDim.x) {
        k_smem[i] = k[kv_offset + i];
        v_smem[i] = v[kv_offset + i];
    }
    __syncthreads();

    for (int q_pos = tid; q_pos < seq_len; q_pos += blockDim.x) {
        float scores[1024];
        float max_score = -INFINITY;

            float sum = 0.0f;
            for (int d = 0; d < HEAD_SIZE; d++) {

            max_score = fmaxf(max_score, scores[k_pos]);
