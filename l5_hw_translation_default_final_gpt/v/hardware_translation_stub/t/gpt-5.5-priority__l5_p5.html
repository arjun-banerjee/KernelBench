import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <ATen/cuda/CUDAContext.h>

__global__ __launch_bounds__(512, 2)
void rotary_kernel(__half* __restrict__ q,
                   __half* __restrict__ k,
                   const __half* __restrict__ cos,
                   const __half* __restrict__ sin) {
    const int token = blockIdx.x;
    const int tid = threadIdx.x;
    constexpr int NUM_HEADS = 32;
    constexpr int NUM_KV_HEADS = 8;
    constexpr int HEAD_SIZE = 128;
    constexpr int ROT_DIM = 64;
    constexpr int HALF_DIM = 32;
    constexpr int Q_PAIRS = NUM_HEADS * HALF_DIM;
    constexpr int TOTAL_PAIRS = (NUM_HEADS + NUM_KV_HEADS) * HALF_DIM;

    const __half* __restrict__ cos_t = cos + token * ROT_DIM;
    const __half* __restrict__ sin_t = sin + token * ROT_DIM;

    #pragma unroll
    for (int idx = tid; idx < TOTAL_PAIRS; idx += 512) {
        const bool is_q = idx < Q_PAIRS;
        const int j = is_q ? idx : idx - Q_PAIRS;
        const int head = j >> 5;
        const int off = j & 31;
        __half* __restrict__ base = is_q
            ? q + ((token * NUM_HEADS + head) * HEAD_SIZE)
            : k + ((token * NUM_KV_HEADS + head) * HEAD_SIZE);

        const float x0 = __half2float(base[off]);
        const float x1 = __half2float(base[off + HALF_DIM]);
        const float c0 = __half2float(cos_t[off]);
        const float s0 = __half2float(sin_t[off]);
        const float c1 = __half2float(cos_t[off + HALF_DIM]);
        const float s1 = __half2float(sin_t[off + HALF_DIM]);

        base[off] = __float2half_rn(x0 * c0 - x1 * s0);
        base[off + HALF_DIM] = __float2half_rn(x1 * c1 + x0 * s1);
    }
}

void rotary(torch::Tensor q, torch::Tensor k, torch::Tensor cos, torch::Tensor sin) {
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    rotary_kernel<<<512, 512, 0, stream>>>(
        reinterpret_cast<__half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<__half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const __half*>(cos.data_ptr<at::Half>()),
        reinterpret_cast<const __half*>(sin.data_ptr<at::Half>()));
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("rotary", &rotary, "H100 rotary embedding");
}
