#include <torch/extension.h>
#include <cuda_fp16.h>
#include <c10/cuda/CUDAStream.h>

constexpr int WARP_SIZE = 32;
constexpr int HEAD_SIZE = 128;

__device__ __forceinline__ float warp_reduce_max(float val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        val = max(val, __shfl_xor_sync(0xffffffff, val, offset));
    }
    return val;
}

__device__ __forceinline__ float warp_reduce_sum(float val) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        val += __shfl_xor_sync(0xffffffff, val, offset));
    }
    return val;
}

// Each warp handles one query position (one head)
// Thread block has 4 warps = 128 threads, processing 4 query positions
__global__ void attention_kernel(
    at::Half* __restrict__ out,
    const at::Half* __restrict__ q,
    const at::Half* __restrict__ k,
    const at::Half* __restrict__ v,
    int num_heads, int num_kv_heads, int seq_len, float scale
) {
    // Shared memory for accumulator output
    __shared__ float acc_shared[4][HEAD_SIZE];  // 4 warps, each accumulates HEAD_SIZE floats
    
    const int batch_idx = blockIdx.z;
    const int head_idx = blockIdx.y;
    const int q_block_idx = blockIdx.x;
    
    const int warp_id = threadIdx.x / WARP_SIZE;
    const int lane_id = threadIdx.x % WARP_SIZE;
    const int q_pos = q_block_idx * 4 + warp_id;
    
    const int kv_head_idx = head_idx / (num_heads / num_kv_heads);
    
    const int64_t q_stride = num_heads * seq_len * HEAD_SIZE;
    const int64_t kv_stride = num_kv_heads * seq_len * HEAD_SIZE;
    
    const half* q_ptr = reinterpret_cast<const half*>(q);
    const half* k_ptr = reinterpret_cast<const half*>(k);
    const half* v_ptr = reinterpret_cast<const half*>(v);
    half* out_ptr = reinterpret_cast<half*>(out);
    
    // Initialize accumulator
    if (q_pos < seq_len) {
        for (int h = lane_id; h < HEAD_SIZE; h += WARP_SIZE) {
            acc_shared[warp_id][h] = 0.0f;
        }
    }
    
    // Each warp loads its query vector
    float q_local[4];
    if (q_pos < seq_len) {
        const half* q_row = q_ptr + batch_idx * q_stride + head_idx * seq_len * HEAD_SIZE + q_pos * HEAD_SIZE;
        for (int h = lane_id; h < HEAD_SIZE; h += WARP_SIZE) {
            q_local[(h / WARP_SIZE)] = __half2float(q_row[h]);
        }
    }
    
    // Online softmax state
    float max_logit = -INFINITY;
    float exp_sum = 0.0f;
    
    const half* k_head = k_ptr + batch_idx * kv_stride + kv_head_idx * seq_len * HEAD_SIZE;
    const half* v_head = v_ptr + batch_idx * kv_stride + kv_head_idx * seq_len * HEAD_SIZE;
    
    // Iterate over KV positions
    for (int kv_pos = 0; kv_pos <= q_pos && kv_pos < seq_len; kv_pos++) {
        // Compute Q*K dot product
        float dot = 0.0f;
        const half* k_row = k_head + kv_pos * HEAD_SIZE;
        const half* v_row = v_head + kv_pos * HEAD_SIZE;
        
        for (int h = lane_id; h < HEAD_SIZE; h += WARP_SIZE) {
            dot += q_local[(h / WARP_SIZE)] * __half2float(k_row[h]);
        }
        
        // Reduce across warp
        dot = warp_reduce_sum(dot);
        
        // Thread 0 in warp does softmax and V accumulation
        if (lane_id == 0) {
            float score = dot * scale;
            float old_max = max_logit;
            max_logit = max(max_logit, score);
            
            // Rescale previous accumulator if max changed
            if (old_max < max_logit && exp_sum > 0) {
                float rescale = __expf(old_max - max_logit);
                exp_sum *= rescale;
                for (int h = 0; h < HEAD_SIZE; h++) {
                    acc_shared[warp_id][h] *= rescale;
                }
            }
            
            float exp_val = __expf(score - max_logit);
            exp_sum += exp_val;
            
            // Accumulate weighted V
            for (int h = 0; h < HEAD_SIZE; h++) {
                acc_shared[warp_id][h] += exp_val * __half2float(v_row[h]);
            }
        }
    }
    
    __syncthreads();
    
    // Write output
    if (q_pos < seq_len && exp_sum > 0) {
        half* out_row = out_ptr + batch_idx * q_stride + head_idx * seq_len * HEAD_SIZE + q_pos * HEAD_SIZE;
        if (lane_id == 0) {
            float inv_sum = 1.0f / exp_sum;
            for (int h = 0; h < HEAD_SIZE; h++) {
                out_row[h] = __float2half(acc_shared[warp_id][h] * inv_sum);
            }
        }
    }
}

torch::Tensor flash_attention_forward_cuda(
    torch::Tensor q, torch::Tensor k, torch::Tensor v,
    int64_t num_heads, int64_t num_kv_heads, int64_t head_size) {
    
    int batch_size = q.size(0);
    int seq_len = q.size(2);
    
    auto out = torch::empty_like(q);
    float scale = 1.0f / sqrtf(static_cast<float>(head_size));
    
    int num_q_blocks = (seq_len + 3) / 4;  // Each block handles 4 query positions
    dim3 grid(num_q_blocks, num_heads, batch_size);
    dim3 block(128);  // 4 warps = 128 threads
    
    auto stream = at::cuda::getCurrentCUDAStream();
    
    TORCH_CHECK(head_size == HEAD_SIZE, "Only head_size=128 supported");
    
    attention_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<at::Half*>(out.data_ptr<at::Half>()),
        reinterpret_cast<const at::Half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const at::Half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const at::Half*>(v.data_ptr<at::Half>()),
        num_heads, num_kv_heads, seq_len, scale);
    
    C10_CUDA_CHECK(cudaGetLastError());
    return out;
}
