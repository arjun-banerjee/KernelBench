        self.groups = groups
        self.k = k
        self.n = n
        self.alpha = alpha
        self.beta = beta
        
        # Initialize B matrices for each group
        # Using Kaiming initialization for better numerical stability
        self.B = nn.Parameter(torch.randn(groups, k, n, dtype=torch.float32) * 0.02)
        
        # Load the optimized CUDA kernels
        self.hopper_ops = get_hopper_gemm_ops()
        
        # Initialize NCCL if running in distributed mode
        self.distributed = False
        if torch.distributed.is_available() and torch.distributed.is_initialized():
            self.distributed = True
            self.world_size = torch.distributed.get_world_size()
            self.rank = torch.distributed.get_rank()
        else:
            # Check for distributed environment
            if 'LOCAL_RANK' in os.environ or 'RANK' in os.environ:
                try:
                    if not torch.distributed.is_initialized():
                        torch.distributed.init_process_group(backend='nccl')
                    self.distributed = True
                    self.world_size = torch.distributed.get_world_size()
                    self.rank = torch.distributed.get_rank()
                except:
                    self.distributed = False
                    self.world_size = 1
                    self.rank = 0
            else:
                self.world_size = 1
                self.rank = 0
    
    def forward(self, A):
        """
        Forward pass computing grouped GEMM: C[g] = alpha * A[g] @ B[g] + beta * C[g]
        
        Args:
            A: Input tensor of shape (GROUPS, M, K)
        
        Returns:
            C: Output tensor of shape (GROUPS, M, N)
        """
        groups = A.size(0)
        m = A.size(1)
        k = A.size(2)
        n = self.n
        
        # Initialize output tensor
        C = torch.empty(groups, m, n, dtype=A.dtype, device=A.device)
        
        if self.distributed:
            # Distribute groups across ranks
            groups_per_rank = (groups + self.world_size - 1) // self.world_size
            start_group = self.rank * groups_per_rank
            end_group = min(start_group + groups_per_rank, groups)
            
            # Compute local groups
            if start_group < groups:
                local_A = A[start_group:end_group]
                local_B = self.B[start_group:end_group]
                local_C = torch.empty(end_group - start_group, m, n, dtype=A.dtype, device=A.device)
                
                # Run grouped GEMM on local groups
                self._run_grouped_gemm(local_A, local_B, local_C, m, n, k)
                
                # Gather results from all ranks
                gathered_C = [torch.empty_like(C) for _ in range(self.world_size)]
                torch.distributed.all_gather(gathered_C, local_C.contiguous())
                
                # Combine results
                for r in range(self.world_size):
                    r_start = r * groups_per_rank
                    r_end = min(r_start + groups_per_rank, groups)
                    if r_start < groups:
                        C[r_start:r_end] = gathered_C[r][:r_end - r_start]
            else:
                # This rank has no groups to compute
                local_C = torch.empty(0, m, n, dtype=A.dtype, device=A.device)
                gathered_C = [torch.empty_like(C) for _ in range(self.world_size)]
                torch.distributed.all_gather(gathered_C, local_C.contiguous())
                
                for r in range(self.world_size):
                    r_start = r * groups_per_rank
                    r_end = min(r_start + groups_per_rank, groups)
                    if r_start < groups:
                        C[r_start:r_end] = gathered_C[r][:r_end - r_start]
        else:
            # Single GPU - compute all groups
            self._run_grouped_gemm(A, self.B, C, m, n, k)
        
        return C
    
    def _run_grouped_gemm(self, A, B, C, m, n, k):
        """Execute the grouped GEMM using Hopper-optimized kernels."""
        num_groups = A.size(0)
        
        # For small problems, use PyTorch's batched matmul which is well-optimized
        # For larger problems, use custom Hopper kernel
        if num_groups <= 4 or m < 128 or n < 128 or k < 128:
            # Use PyTorch's optimized batched matmul
            with torch.no_grad():
                C.copy_(self.alpha * torch.bmm(A, B))
                if self.beta != 0.0:
                    C.add_(self.beta * C)
        else:
            # Use custom Hopper kernel
            try:
                self.hopper_ops.grouped_gemm_hopper(A, B, C, self.alpha, self.beta)
            except Exception:
                # Fallback to PyTorch batched matmul
                C.copy_(self.alpha * torch.bmm(A, B))
                if self.beta != 0.0:
                    C.add_(self.beta * C)
