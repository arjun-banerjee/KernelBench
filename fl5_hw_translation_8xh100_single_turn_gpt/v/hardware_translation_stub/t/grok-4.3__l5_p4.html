import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

__device__ __forceinline__ float silu(float x) {
    return x / (1.0f + __expf(-x));
}

__global__ void silu_and_mul_kernel(
    __half* __restrict__ out,
    const __half* __restrict__ input,
    const int d,
    const int num_tokens) {
    
    const int token_idx = blockIdx.x;
    if (token_idx >= num_tokens) return;
    
    const __half* gate_ptr = input + token_idx * 2 * d;
    const __half* up_ptr = gate_ptr + d;
    __half* out_ptr = out + token_idx * d;
    
    using vec_t = float4;
    constexpr int VEC_SIZE = 8;
    
    const int num_vecs = d / VEC_SIZE;
    const vec_t* gate_vec = reinterpret_cast<const vec_t*>(gate_ptr);
    const vec_t* up_vec = reinterpret_cast<const vec_t*>(up_ptr);
    vec_t* out_vec = reinterpret_cast<vec_t*>(out_ptr);
    
    for (int i = threadIdx.x; i < num_vecs; i += blockDim.x) {
        vec_t g4 = __ldg(&gate_vec[i]);
        vec_t u4 = __ldg(&up_vec[i]);
        __half* gpb = reinterpret_cast<__half*>(&g4);
        __half* upb = reinterpret_cast<__half*>(&u4);
        
        #pragma unroll
        for (int j = 0; j < VEC_SIZE; j++) {
            float gf = __half2float(gpb[j]);
            float uf = __half2float(upb[j]);
            gpb[j] = __float2half(silu(gf) * uf);
        }
        out_vec[i] = g4;
    }
    
    const int tail_start = num_vecs * VEC_SIZE;
    for (int i = tail_start + threadIdx.x; i < d; i += blockDim.x) {
        float g = __half2float(gate_ptr[i]);
        float u = __half2float(up_ptr[i]);
        out_ptr[i] = __float2half(silu(g) * u);
    }
}

torch::Tensor silu_and_mul_cuda(torch::Tensor input) {
    auto options = torch::TensorOptions().dtype(torch::kFloat16).device(input.device());
    auto out = torch::empty({input.size(0), input.size(1)/2}, options);

}
