                                                                    (k_tile + kk) / GROUP_SIZE * TILE_N + local_n]) : 0.0f;

                        sum += a_val * w_val;
                    }
                    
                    // Accumulate
                    acc[local_m][local_n] += sum;
                }
            }
        }
        
        // Advance pipeline
        pipeline_stage = (pipeline_stage + 1) % STAGES;
        __syncthreads();
    }
    
    // Write results to global memory
    #pragma unroll
    for (int m = 0; m < TILE_M; m++) {
        #pragma unroll
        for (int n = 0; n < TILE_N; n++) {
            const int global_m = start_m + m;
            const int global_n = start_n + n;
            
            if (global_m < size_m && global_n < size_n) {
                // Use lane_id to distribute writes
                if ((m * TILE_N + n) % THREADS_PER_BLOCK == tid) {

                }
            }
        }
    }
    
    __syncthreads();
    
    // Final reduction and write using all threads
    for (int offset = 0; offset < TILE_M * TILE_N; offset += THREADS_PER_BLOCK) {
        const int idx = offset + tid;
        const int m = idx / TILE_N;
        const int n = idx % TILE_N;
        const int global_m = start_m + m;
        const int global_n = start_n + n;
        
        if (global_m < size_m && global_n < size_n) {
            C[global_m * size_n + global_n] = float_to_half(acc[m][n]);
        }
    }
}

        # Number of quantization groups
        self.num_groups = (K + group_size - 1) // group_size
        
        # Initialize quantized weight matrix (K x N/8 int4 packed)
        self.register_buffer(
            'qweight',
            torch.zeros(K, N // self.pack_factor, dtype=torch.int32)
        )
        
        # Initialize scales (num_groups x N)
        self.register_buffer(
            'scales',
            torch.ones(self.num_groups, N, dtype=torch.float16)
        )
        
        # Initialize zero points (num_groups x N)
        self.register_buffer(
            'zeros',
            torch.zeros(self.num_groups, N, dtype=torch.float16)
        )
        
        # Group indices for activation ordering (optional)
        self.register_buffer(
            'g_idx',
            torch.zeros(K, dtype=torch.int32)
        )
        
        # Workspace for permutation (if needed)
        self.register_buffer(
            'workspace',
            torch.empty(0, dtype=torch.float16)
        )
        
        # Flag for activation ordering
        self.has_act_order = False
        self.has_zp = True
        
    def quantize_weight(self, weight: torch.Tensor) -> None:
        """
        Quantize FP16 weight matrix to INT4 with group-wise scales and zeros.
        
        Args:
            weight: FP16 weight tensor of shape [K, N]
        """
        assert weight.shape == (self.K, self.N), f"Weight shape mismatch: {weight.shape}"
        assert weight.dtype == torch.float16, f"Expected FP16 weight, got {weight.dtype}"
        
        weight = weight.cuda()
        
        # Process each group
        for g in range(self.num_groups):
            start_k = g * self.group_size
            end_k = min(start_k + self.group_size, self.K)
            
            group_weight = weight[start_k:end_k, :]  # [group_size, N]
            
            # Compute min/max for symmetric quantization with zero point
            w_min = group_weight.min(dim=0)[0]  # [N]
            w_max = group_weight.max(dim=0)[0]  # [N]
            
            # Zero point
            zero = (w_min + w_max) / 2.0
            self.zeros[g, :] = zero.half()
            
            # Scale
            scale = (w_max - w_min) / 15.0  # 15 = max int4 value
            scale = scale.clamp(min=1e-8)  # Avoid division by zero
            self.scales[g, :] = scale.half()
            
            # Quantize to INT4
            # Shift to centered range [-7, 7] or [0, 15] depending on implementation
            # Here we use symmetric quantization: value = (weight - zero) / scale
            group_weight_centered = group_weight - zero.unsqueeze(0)
            group_weight_scaled = group_weight_centered / scale.unsqueeze(0)
            
            # Clamp to [-8, 7] for signed int4
            group_int4 = torch.clamp(torch.round(group_weight_scaled), -8, 7).to(torch.int8)
            group_int4 = (group_int4 + 8).to(torch.uint8)  # Shift to [0, 15]
            
            # Pack into int32
            for k in range(end_k - start_k):
                packed = 0
                for n in range(0, self.N, self.pack_factor):
                    packed = 0
                    for p in range(self.pack_factor):
                        if n + p < self.N:
                            val = group_int4[k, n + p].item()
                            packed |= (val << (p * 4))
                    self.qweight[start_k + k, n // self.pack_factor] = packed
        
        # Update g_idx for activation ordering
        for g in range(self.num_groups):
            start_k = g * self.group_size
            end_k = min(start_k + self.group_size, self.K)
            self.g_idx[start_k:end_k] = g
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Forward pass: INT4 GEMM with on-the-fly dequantization.
        
        Args:
            x: Input activation tensor of shape [M, K]
            
        Returns:
            Output tensor of shape [M, N]
        """
        M = x.shape[0]
        K = x.shape[1]
        
        assert K == self.K, f"Input dimension mismatch: {K} != {self.K}"
        assert x.dtype == torch.float16, f"Expected FP16 input, got {x.dtype}"
        
        x = x.cuda()
        qweight = self.qweight.cuda()
        scales = self.scales.cuda()
        zeros = self.zeros.cuda()
        g_idx = self.g_idx.cuda()
        
        # Call optimized kernel
        output = marlin_ops.marlin_gemm_h100(
            x, qweight, scales, zeros, g_idx,
            M, self.N, self.K,
            self.has_act_order, self.has_zp
        )
        
        return output
    
    def to(self, *args, **kwargs):
        """Override to ensure all buffers are moved to GPU."""
        super().to(*args, **kwargs)
        self.qweight = self.qweight.cuda()
        self.scales = self.scales.cuda()
        self.zeros = self.zeros.cuda()
        self.g_idx = self.g_idx.cuda()
        return self
