import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void int4_gemm_kernel(const half* __restrict__ A,
                                 const unsigned char* __restrict__ Q,
                                 const half* __restrict__ S,
                                 half* __restrict__ C,
                                 int M, int N, int K, int group_size) {
    int n = blockIdx.x * 16 + threadIdx.x;
    int m = blockIdx.y * 16 + threadIdx.y;
    if (m >= M || n >= N) return;

    float acc = 0.0f;
    int halfN = N >> 1;
    int q_col = n >> 1;
    bool high = (n & 1) != 0;

    #pragma unroll 4
    for (int k = 0; k < K; ++k) {
        float av = __half2float(A[m * K + k]);
        unsigned char p = Q[k * halfN + q_col];
        int qi = high ? ((int)(p >> 4) & 15) : ((int)p & 15);
        qi -= 8;
        float sc = __half2float(S[(k / group_size) * N + n]);
        acc = fmaf(av, ((float)qi) * sc, acc);
    }
    C[m * N + n] = __float2half_rn(acc);
}

torch::Tensor int4_gemm(torch::Tensor A, torch::Tensor Q, torch::Tensor S, int64_t group_size) {
    TORCH_CHECK(A.is_cuda() && Q.is_cuda() && S.is_cuda(), "all tensors must be CUDA");
    TORCH_CHECK(A.scalar_type() == at::ScalarType::Half, "A must be fp16");
    TORCH_CHECK(S.scalar_type() == at::ScalarType::Half, "scales must be fp16");
    TORCH_CHECK(Q.scalar_type() == at::ScalarType::Byte, "qweight must be uint8");
    const int M = (int)A.size(0);
    const int K = (int)A.size(1);
    const int N = (int)S.size(1);
    auto C = torch::empty({M, N}, A.options());
    dim3 block(16, 16, 1);
    dim3 grid((N + 15) / 16, (M + 15) / 16, 1);
    int4_gemm_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>(
        reinterpret_cast<const half*>(A.data_ptr<at::Half>()),
        reinterpret_cast<const unsigned char*>(Q.data_ptr<unsigned char>()),
        reinterpret_cast<const half*>(S.data_ptr<at::Half>()),
        reinterpret_cast<half*>(C.data_ptr<at::Half>()),
        M, N, K, (int)group_size);
    return C;
}
'''

_ext = None

def _get_ext():
    global _ext
    if _ext is None:
        _ext = load_inline(
            name="h100_simple_int4_gemm_ext",
            cpp_sources="",
            cuda_sources=_cuda_src,
            functions=["int4_gemm"],
            with_cuda=True,
            extra_cuda_cflags=["-O3", "--use_fast_math"],
            verbose=False,
        )
    return _ext

class ModelNew(nn.Module):
    def __init__(self, K: int, N: int, GROUP_SIZE: int):
        super().__init__()
        self.K = int(K)
        self.N = int(N)
        self.GROUP_SIZE = int(GROUP_SIZE)
        # Deterministic int4, per-group symmetric quantized weight.  The full
        # fp16 weight is deliberately not kept, matching the memory footprint of
        # the Marlin-style operator while giving forward a true custom CUDA path.
        w = torch.randn(self.K, self.N, dtype=torch.float16)
        ng = (self.K + self.GROUP_SIZE - 1) // self.GROUP_SIZE
        wv = w.view(ng, self.GROUP_SIZE, self.N)
        scales = (wv.float().abs().amax(dim=1) / 7.0).clamp_min(1.0e-6).to(torch.float16)
        q = torch.round(wv.float() / scales[:, None, :].float()).clamp(-8, 7).to(torch.int16) + 8
        q = q.view(self.K, self.N).to(torch.uint8)
        qlo = q[:, 0::2]
        qhi = q[:, 1::2]
        packed = (qlo | (qhi << 4)).contiguous()
        self.register_buffer("qweight", packed)
        self.register_buffer("scales", scales.contiguous())

    def forward(self, a: torch.Tensor) -> torch.Tensor:
        return _get_ext().int4_gemm(a.contiguous(), self.qweight, self.scales, self.GROUP_SIZE)
