import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void gemm_permute_kernel(const half* __restrict__ A, const half* __restrict__ B,
                                    half* __restrict__ D, int batch, int m, int k, int n,
                                    int d1, float alpha, float beta) {
    int bm = blockIdx.x * blockDim.x + threadIdx.x;
    int bn = blockIdx.y * blockDim.y + threadIdx.y;
    int b_batch = blockIdx.z;

    if (bm >= m || bn >= n || b_batch >= batch) return;

    float sum = 0.0f;
    for (int i = 0; i < k; ++i) {
        sum += __half2float(A[b_batch * m * k + bm * k + i]) * __half2float(B[i * n + bn]);
    }
    float val = alpha * sum + beta * __half2float(D[b_batch * m * n + bm * n + bn]);
    D[b_batch * m * n + bm * n + bn] = __float2half(val);
}

torch::Tensor gemm_permute_cuda(torch::Tensor A, torch::Tensor B, int batch, int m, int k, int n,
                                int d1, float alpha, float beta) {
    auto options = torch::TensorOptions().dtype(torch::kFloat16).device(A.device());
    auto D = torch::zeros({batch, m, n}, options);

    dim3 threads(16, 16);
    dim3 blocks((m + 15) / 16, (n + 15) / 16, batch);

    gemm_permute_kernel<<<blocks, threads>>>(
        reinterpret_cast<half*>(A.data_ptr<at::Half>()),
        reinterpret_cast<half*>(B.data_ptr<at::Half>()),
        reinterpret_cast<half*>(D.data_ptr<at::Half>()),
        batch, m, k, n, d1, alpha, beta
    );

    // Apply Tensor4DPermuteBMM0213 permute on D: reshape [batch, m, n] -> [batch/d1, d1, m, n] -> permute [0,2,1,3]
    int new_batch = batch / d1;
    auto D_reshaped = D.view({new_batch, d1, m, n}).permute({0, 2, 1, 3}).contiguous();
    return D_reshaped.view({new_batch, m, d1 * n});
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gemm_permute_cuda", &gemm_permute_cuda, "GEMM Permute CUDA");
}
