#include <cuda.h>

#include <cuda_fp16.h>
#include <math_constants.h>

#define NUM_HEADS 32
#define NUM_KV_HEADS 8
#define HEAD_SIZE 128
#define SEQ_LEN 1024
#define GROUP_SIZE 4

__inline__ __device__ float warp_reduce_sum(float v) {
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) v += __shfl_down_sync(0xffffffff, v, off);

// Simple exact scaled-dot-product GQA attention.  One CTA computes one query row.
// It is intentionally conservative numerically: fp32 logits/softmax accumulation.
__global__ __launch_bounds__(128, 2)
void gqa_attention_kernel(const half* __restrict__ q,
                          const half* __restrict__ k,
                          const half* __restrict__ v,
                          half* __restrict__ out) {
    extern __shared__ float logits[]; // 1024 floats
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int s_q = row & (SEQ_LEN - 1);
    const int tmp = row >> 10;
    const int h = tmp & 31;
    const int b = tmp >> 5;
    const int kvh = h >> 2;

    const half* qptr = q + (((b * NUM_HEADS + h) * SEQ_LEN + s_q) * HEAD_SIZE);
    const half* kbase = k + (((b * NUM_KV_HEADS + kvh) * SEQ_LEN) * HEAD_SIZE);
    const half* vbase = v + (((b * NUM_KV_HEADS + kvh) * SEQ_LEN) * HEAD_SIZE);
    constexpr float scale = 0.08838834764831845f; // 1/sqrt(128)

    float local_max = -CUDART_INF_F;
    for (int sk = 0; sk < SEQ_LEN; ++sk) {
        const half* kptr = kbase + sk * HEAD_SIZE;
        float sum = 0.0f;
        // 128 threads, one element each: coalesced q/k loads.
        half qh = qptr[tid];
        half kh = kptr[tid];
        sum = __half2float(qh) * __half2float(kh);
        sum = block_reduce_sum(sum);
        if (tid == 0) {
            float lg = sum * scale;
            logits[sk] = lg;
            local_max = fmaxf(local_max, lg);
        }
        __syncthreads();
    }
    // broadcast row max from logits pass: recompute via shared logits in parallel
    local_max = -CUDART_INF_F;
    for (int i = tid; i < SEQ_LEN; i += blockDim.x) local_max = fmaxf(local_max, logits[i]);
    float m = block_reduce_max(local_max);
    __syncthreads();

    float local_sum = 0.0f;
    for (int i = tid; i < SEQ_LEN; i += blockDim.x) {
        float e = __expf(logits[i] - m);
        logits[i] = e;
        local_sum += e;
    }
    float denom = block_reduce_sum(local_sum);
    float inv = 1.0f / denom;
    __syncthreads();

    // Compute all 128 output dims; thread tid owns dim tid.
    float acc = 0.0f;
    for (int sk = 0; sk < SEQ_LEN; ++sk) {
        float p = logits[sk] * inv;
        acc += p * __half2float(vbase[sk * HEAD_SIZE + tid]);
    }
    out[(((b * NUM_HEADS + h) * SEQ_LEN + s_q) * HEAD_SIZE) + tid] = __float2half_rn(acc);
}

torch::Tensor gqa_attention(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
