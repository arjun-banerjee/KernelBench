// One warp computes one 16x16 output tile.  A is row-major [M,K], B is row-major [K,N],
// output is row-major [M,N], columns are scattered through indices[j].
__global__ void gemm_scatter_wmma_kernel(const half* __restrict__ A,
                                         const half* __restrict__ B,
                                         const int* __restrict__ indices,
                                         float* __restrict__ O,
                                         int M, int K, int N, int J,

  int lane = threadIdx.x & 31;

  int tiles_j = (J + 15) >> 4;
  int total_tiles = tiles_m * tiles_j;

  for (int tile = warp_global; tile < total_tiles; tile += ((gridDim.x * blockDim.x) >> 5)) {
    int tm = tile / tiles_j;
    int tj = tile - tm * tiles_j;

    int j0 = tj * 16;

    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;
    wmma::fill_fragment(c_frag, 0.0f);

      for (int t = lane; t < 256; t += 32) {

  CHECK_CUDA(A); CHECK_CUDA(B); CHECK_CUDA(indices);
  TORCH_CHECK(A.dtype() == torch::kFloat16, "A must be fp16");
  TORCH_CHECK(B.dtype() == torch::kFloat16, "B must be fp16");
  TORCH_CHECK(indices.dtype() == torch::kInt32 || indices.dtype() == torch::kInt64, "indices must be int32/int64");
  TORCH_CHECK(indices.dtype() == torch::kInt32, "internal indices must be int32");

  int K = (int)A.size(1);
