class ModelNew(nn.Module):
    def __init__(self, batch, K, N, alpha, beta):
        super().__init__()
        self.batch = batch
        self.K = K
        self.N = N
        self.alpha = alpha
        self.beta = beta
        self.world_size = int(os.environ.get('WORLD_SIZE', '1'))
        self.rank = int(os.environ.get('RANK', '0'))
        if self.world_size > 1 and not dist.is_initialized():
            dist.init_process_group(backend='nccl')
        
    def forward(self, A):
        batch_per_rank = (A.shape[0] + self.world_size - 1) // self.world_size
        start = self.rank * batch_per_rank
