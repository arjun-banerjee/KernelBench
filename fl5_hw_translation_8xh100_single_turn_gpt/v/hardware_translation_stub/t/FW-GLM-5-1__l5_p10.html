            V.data_ptr<scalar_t>(),
            dO.data_ptr<scalar_t>(),
            L.data_ptr<float>(),
            D_new.data_ptr<float>(),
            dQ.data_ptr<scalar_t>(),
            dK.data_ptr<scalar_t>(),
            dV.data_ptr<scalar_t>(),
            batch_size, num_heads, num_kv_heads, seq_len, head_dim, scale
        );
    });
    
    C10_CUDA_CHECK(cudaGetLastError());
    
    return {dQ, dK, dV};
}
"""

flash_attn_bwd_cpp = """
std::vector<torch::Tensor> flash_attn_bwd_cuda(
    torch::Tensor Q,
    torch::Tensor K,
    torch::Tensor V,

# Load the CUDA extension
flash_attn_module = load_inline(
    name='flash_attn_bwd_hopper',
    cpp_sources=flash_attn_bwd_cpp,
    cuda_sources=flash_attn_bwd_source,
    functions=['flash_attn_bwd_cuda'],
    with_cuda=True,
    extra_cuda_cflags=[
        '-O3',
        '-arch=sm_90',  # Target Hopper architecture
        '--ptxas-options=-v',
        '-lineinfo',
        '--use_fast_math',
    ],
    extra_cflags=['-O3'],
)


class ModelNew(nn.Module):
    """
    Flash Attention 2 Backward Pass - H100 Hopper Optimized
    
    Optimizations for Hopper architecture:
    1. Larger tile sizes (TILE_M=128, TILE_N=128) leveraging 228KB shared memory
    2. Warp-level reductions for efficient softmax computation
    3. Atomic accumulation for dK and dV gradients
    4. BF16/FP16 support with float accumulation for numerical stability
    5. Optimized memory access patterns for H100's 3.35 TB/s bandwidth
    """
    
    def __init__(self, num_heads, num_kv_heads, head_dim):
        super().__init__()
        self.num_heads = num_heads
        self.num_kv_heads = num_kv_heads
        self.head_dim = head_dim
        self.scale = 1.0 / (head_dim ** 0.5)
        
    def forward(self, q, k, v, dout):
        """
        Forward pass computes gradients for Flash Attention backward.
        
        Args:
            q: Query tensor [batch, num_heads, seq_len, head_dim]
            k: Key tensor [batch, num_kv_heads, seq_len, head_dim]
            v: Value tensor [batch, num_kv_heads, seq_len, head_dim]
            dout: Gradient of output [batch, num_heads, seq_len, head_dim]
            
        Returns:
            dq: Gradient for query
            dk: Gradient for key
            dv: Gradient for value
        """
        # Use the custom Hopper-optimized kernel
        dq, dk, dv = flash_attn_module.flash_attn_bwd_cuda(q, k, v, dout)
        
        return dq, dk, dv
