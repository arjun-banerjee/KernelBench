import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>

void rotary_inplace(torch::Tensor q, torch::Tensor k,
                    torch::Tensor cos, torch::Tensor sin,
                    int64_t rot_dim);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("rotary_inplace", &rotary_inplace, "Hopper optimized rotary embedding (in-place)");
}
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ __launch_bounds__(256, 2)
void rotary_neox_kernel(__half* __restrict__ q,
                        __half* __restrict__ k,
                        const __half* __restrict__ cos,
                        const __half* __restrict__ sin,
                        int num_tokens,
                        int num_heads,
                        int num_kv_heads,
                        int head_size,
                        int rot_dim) {
  extern __shared__ __half smem[];
  __half* scos = smem;
  __half* ssin = smem + rot_dim;

  const int tok = blockIdx.x;
  const int tid = threadIdx.x;

  // One block owns one token.  The sin/cos row is reused by all Q/KV heads,
  // so stage it once in shared memory (H100 has ample SMEM and very fast LDS).
  for (int i = tid; i < rot_dim; i += blockDim.x) {
    scos[i] = cos[tok * rot_dim + i];
    ssin[i] = sin[tok * rot_dim + i];
  }
  __syncthreads();

  const int half_dim = rot_dim >> 1;
  const int total = (num_heads + num_kv_heads) * half_dim;

  for (int idx = tid; idx < total; idx += blockDim.x) {
    const int off = idx % half_dim;
    const int h_all = idx / half_dim;
    const bool is_q = h_all < num_heads;
    const int h = is_q ? h_all : (h_all - num_heads);

    __half* __restrict__ base = is_q
        ? q + ((tok * num_heads + h) * head_size)
        : k + ((tok * num_kv_heads + h) * head_size);

    const float x = __half2float(base[off]);
    const float y = __half2float(base[off + half_dim]);

    // GPT-NeoX/LLaMA-style rotate_half over the first rot_dim coordinates.
    const float c0 = __half2float(scos[off]);
    const float s0 = __half2float(ssin[off]);
    const float c1 = __half2float(scos[off + half_dim]);
    const float s1 = __half2float(ssin[off + half_dim]);

    base[off]            = __float2half_rn(x * c0 - y * s0);
    base[off + half_dim] = __float2half_rn(y * c1 + x * s1);
  }
}

void rotary_inplace(torch::Tensor q, torch::Tensor k,
                    torch::Tensor cos, torch::Tensor sin,
                    int64_t rot_dim64) {
  const int num_tokens = (int)q.size(0);
  const int num_heads = (int)q.size(1);
  const int head_size = (int)q.size(2);
  const int num_kv_heads = (int)k.size(1);
  const int rot_dim = (int)rot_dim64;

  const int threads = 256;
  const size_t shmem = (size_t)rot_dim * 2u * sizeof(__half);
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();

  rotary_neox_kernel<<<num_tokens, threads, shmem, stream>>>(
      reinterpret_cast<__half*>(q.data_ptr<at::Half>()),
      reinterpret_cast<__half*>(k.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(cos.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(sin.data_ptr<at::Half>()),
      num_tokens, num_heads, num_kv_heads, head_size, rot_dim);
}
"""

_ext = load_inline(

    cuda_sources=[_cuda_src],
    extra_cflags=["-O3"],

class ModelNew(nn.Module):
    def __init__(self, rot_dim: int):
        super().__init__()
        self.rot_dim = int(rot_dim)

    def forward(self, q: torch.Tensor, k: torch.Tensor,
                cos: torch.Tensor, sin: torch.Tensor):
        _ext.rotary_inplace(q, k, cos, sin, self.rot_dim)
        return q, k
