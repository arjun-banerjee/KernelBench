#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <stdint.h>

#ifndef __CUDA_ARCH__
#define __CUDA_ARCH__ 900
#endif

static __device__ __forceinline__ int8_t cvt_rni_s8(float x) {
    unsigned int r;
    asm volatile("cvt.rni.sat.s8.f32 %0, %1;" : "=r"(r) : "f"(x));
    return reinterpret_cast<int8_t&>(r);
}

// Fast deterministic FP16 GEMM for M=16, K=N=4096.  The weight is stored
// transposed (K,N) half for coalesced loads across N.  One warp computes a
// 16x16 output tile: lanes map to (m,n) and each lane accumulates one element.
__global__ __launch_bounds__(256, 2)
void hgemm_16x4096_kernel(const half* __restrict__ x,
                          const half* __restrict__ w_t,
                          half* __restrict__ y,
                          int K, int N) {
    const int tile_n = blockIdx.x * 16;
    const int lane = threadIdx.x & 31;
    const int warp = threadIdx.x >> 5;
    const int m = lane >> 1;               // 0..15
    const int n0 = (lane & 1) + warp * 2;  // 0..15 over 8 warps
    const int n = tile_n + n0;

    float acc = 0.0f;
    if (n < N) {
        #pragma unroll 4
        for (int k = 0; k < K; ++k) {
            float a = __half2float(x[m * K + k]);
            float b = __half2float(w_t[k * N + n]);
            acc = fmaf(a, b, acc);
        }
        y[m * N + n] = __float2half_rn(acc);
    }
}

// Optional int8-weight/activation path.  It is intentionally conservative and
// mirrors the usual W8A8 definition: dynamic per-row activation scale and
// per-tensor weight scale, accumulating in int32 then dequantizing to fp16.
// This path is used only if the Python wrapper selects it.
__global__ __launch_bounds__(256, 2)
void w8a8_gemm_kernel(const half* __restrict__ x,
                      const int8_t* __restrict__ wq_t,
                      const float* __restrict__ w_scale,
                      half* __restrict__ y,
                      int K, int N) {
    const int tile_n = blockIdx.x * 16;
    const int lane = threadIdx.x & 31;
    const int warp = threadIdx.x >> 5;
    const int m = lane >> 1;
    const int n0 = (lane & 1) + warp * 2;
    const int n = tile_n + n0;

    // row max, one warp pair redundantly computes per row; cheap for M=16
    float amax = 0.f;
    for (int k = threadIdx.x; k < K * 16; k += blockDim.x) {
        int row = k / K;
        if (row == m) {
            float v = fabsf(__half2float(x[k]));
            amax = fmaxf(amax, v);
        }
    }
    // reduce within all lanes that share this output lane; simpler full-block max
    __shared__ float sm[256];
    sm[threadIdx.x] = amax;
    __syncthreads();
    for (int s = 128; s > 0; s >>= 1) { if (threadIdx.x < s) sm[threadIdx.x] = fmaxf(sm[threadIdx.x], sm[threadIdx.x+s]); __syncthreads(); }
    float scale_a = sm[0] / 127.f;
    float inv_a = (sm[0] == 0.f) ? 0.f : 127.f / sm[0];

    int acc = 0;
    if (n < N) {
        #pragma unroll 4
        for (int k = 0; k < K; ++k) {
            int aq = (int)cvt_rni_s8(__half2float(x[m*K+k]) * inv_a);
            int bq = (int)wq_t[k*N+n];
            acc += aq * bq;
        }
        float out = (float)acc * scale_a * (*w_scale);
        y[m*N+n] = __float2half_rn(out);
    }
}

torch::Tensor hgemm_forward(torch::Tensor x, torch::Tensor w_t) {
    auto y = torch::empty({16, w_t.size(1)}, x.options());
    const int K = (int)x.size(1);
    const int N = (int)w_t.size(1);
    dim3 block(256);
    dim3 grid((N + 15) / 16);

        reinterpret_cast<const half*>(w_t.data_ptr<at::Half>()),

    return y;
}
