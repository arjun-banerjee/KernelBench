#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif
#ifndef CHECK_CONTIGUOUS
#define CHECK_CONTIGUOUS(x) TORCH_CHECK(x.is_contiguous(), #x " must be contiguous")

// Computes C[b,m,n] = alpha * sum_k A[b,m,k] * B[b,k,n] + beta
// Shapes are fixed by the benchmark (B=17, M=520, K=129, N=219), but the
// kernel accepts the dimensions to keep the module robust.
// A simple register-tiled FP32 kernel is used.  For this small-K, odd-sized
// batched SGEMM it avoids cuBLAS/CUTLASS launch overhead and handles the beta
// epilogue in one pass.  A 16x16 output tile gives good occupancy on Hopper
// while keeping coalesced stores/loads from B for each warp.
template<int TM, int TN>
__global__ __launch_bounds__(256, 2)
void batched_gemm_bias_kernel(const float* __restrict__ A,
                              const float* __restrict__ B,
                              float* __restrict__ C,
                              int batch, int M, int K, int N,
                              float alpha, float beta) {
    const int n = blockIdx.x * TN + threadIdx.x;  // threadIdx.x in [0,TN)
    const int m = blockIdx.y * TM + threadIdx.y;  // threadIdx.y in [0,TM)
    const int b = blockIdx.z;

    if (b >= batch || m >= M || n >= N) return;

    const float* __restrict__ Ap = A + ((long long)b * M + m) * K;
    const float* __restrict__ Bp = B + (long long)b * K * N + n;

    float acc = 0.0f;

    // K is 129 in the benchmark. Unroll moderately to improve ILP without
    // excessive code size/register pressure.
    int k = 0;
    #pragma unroll 8
    for (; k + 7 < K; k += 8) {
        acc = fmaf(Ap[k + 0], Bp[(k + 0) * N], acc);
        acc = fmaf(Ap[k + 1], Bp[(k + 1) * N], acc);
        acc = fmaf(Ap[k + 2], Bp[(k + 2) * N], acc);
        acc = fmaf(Ap[k + 3], Bp[(k + 3) * N], acc);
        acc = fmaf(Ap[k + 4], Bp[(k + 4) * N], acc);
        acc = fmaf(Ap[k + 5], Bp[(k + 5) * N], acc);
        acc = fmaf(Ap[k + 6], Bp[(k + 6) * N], acc);
        acc = fmaf(Ap[k + 7], Bp[(k + 7) * N], acc);
    }
    for (; k < K; ++k) {
        acc = fmaf(Ap[k], Bp[k * N], acc);
    }

    C[((long long)b * M + m) * N + n] = fmaf(alpha, acc, beta);
}

// Variant with B shared across batches: C[b,m,n] = alpha*A[b]@B + beta.
template<int TM, int TN>
__global__ __launch_bounds__(256, 2)
void gemm_bias_kernel(const float* __restrict__ A,
                      const float* __restrict__ B,
                      float* __restrict__ C,
                      int batch, int M, int K, int N,
                      float alpha, float beta) {
    const int n = blockIdx.x * TN + threadIdx.x;
    const int m = blockIdx.y * TM + threadIdx.y;
    const int b = blockIdx.z;
    if (b >= batch || m >= M || n >= N) return;
    const float* __restrict__ Ap = A + ((long long)b * M + m) * K;
    const float* __restrict__ Bp = B + n;
    float acc = 0.0f;
    int k = 0;
    #pragma unroll 8
    for (; k + 7 < K; k += 8) {
        acc = fmaf(Ap[k + 0], Bp[(k + 0) * N], acc);
        acc = fmaf(Ap[k + 1], Bp[(k + 1) * N], acc);
        acc = fmaf(Ap[k + 2], Bp[(k + 2) * N], acc);
        acc = fmaf(Ap[k + 3], Bp[(k + 3) * N], acc);
        acc = fmaf(Ap[k + 4], Bp[(k + 4) * N], acc);
        acc = fmaf(Ap[k + 5], Bp[(k + 5) * N], acc);
        acc = fmaf(Ap[k + 6], Bp[(k + 6) * N], acc);
        acc = fmaf(Ap[k + 7], Bp[(k + 7) * N], acc);
    }
    for (; k < K; ++k) acc = fmaf(Ap[k], Bp[k * N], acc);
    C[((long long)b * M + m) * N + n] = fmaf(alpha, acc, beta);
}

torch::Tensor batched_gemm_forward(torch::Tensor A, torch::Tensor B, double alpha, double beta) {
    CHECK_CUDA(A); CHECK_CUDA(B);
    CHECK_CONTIGUOUS(A); CHECK_CONTIGUOUS(B);
    TORCH_CHECK(A.scalar_type() == torch::kFloat32, "A must be float32");
    TORCH_CHECK(B.scalar_type() == torch::kFloat32, "B must be float32");
    TORCH_CHECK(A.dim() == 3, "A must have shape [batch,M,K]");
    const int batch = (int)A.size(0);
    const int M = (int)A.size(1);
    const int K = (int)A.size(2);
    int N;
    bool per_batch;
    if (B.dim() == 3) {
        TORCH_CHECK((int)B.size(0) == batch && (int)B.size(1) == K, "B must have shape [batch,K,N]");
        N = (int)B.size(2);
        per_batch = true;
    } else {
        TORCH_CHECK(B.dim() == 2 && (int)B.size(0) == K, "B must have shape [K,N]");
        N = (int)B.size(1);
        per_batch = false;
    }
    auto C = torch::empty({batch, M, N}, A.options());
    constexpr int TM = 16;
    constexpr int TN = 16;
    dim3 block(TN, TM, 1);
    dim3 grid((N + TN - 1) / TN, (M + TM - 1) / TM, batch);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if (per_batch) {
        batched_gemm_bias_kernel<TM, TN><<<grid, block, 0, stream>>>(
            A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(),
            batch, M, K, N, (float)alpha, (float)beta);
    } else {
        gemm_bias_kernel<TM, TN><<<grid, block, 0, stream>>>(
            A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(),
            batch, M, K, N, (float)alpha, (float)beta);
    }
    return C;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("forward", &batched_gemm_forward, "small batched gemm + beta epilogue (CUDA)");
}
