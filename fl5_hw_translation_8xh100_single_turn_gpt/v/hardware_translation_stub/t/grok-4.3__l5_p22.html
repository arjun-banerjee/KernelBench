import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <mma.h>

using namespace nvcuda;

__global__ void gemm_fp16_kernel(const half* A, const half* B, float* C, int M, int N, int K, float alpha, float beta) {
    __shared__ half As[32][33];
    __shared__ half Bs[32][33];
    
    int row = blockIdx.y * 32 + threadIdx.y;
    int col = blockIdx.x * 32 + threadIdx.x;
    
    float sum = 0.0f;
    
    for (int t = 0; t < (K + 31) / 32; t++) {
        int k_idx = t * 32 + threadIdx.x;
        if (row < M && k_idx < K) {
            As[threadIdx.y][threadIdx.x] = A[row * K + k_idx];
        } else {
            As[threadIdx.y][threadIdx.x] = __float2half(0.0f);
        }
        
        k_idx = t * 32 + threadIdx.y;
        if (col < N && k_idx < K) {
            Bs[threadIdx.y][threadIdx.x] = B[k_idx * N + col];
        } else {
            Bs[threadIdx.y][threadIdx.x] = __float2half(0.0f);
        }
        
        __syncthreads();
        
        for (int k = 0; k < 32; k++) {
            sum += __half2float(As[threadIdx.y][k]) * __half2float(Bs[k][threadIdx.x]);
        }
        
        __syncthreads();
    }
    
    if (row < M && col < N) {
        C[row * N + col] = alpha * sum + beta * C[row * N + col];
    }
}

torch::Tensor gemm_fp16(torch::Tensor A, torch::Tensor B, float alpha, float beta) {
    auto M = A.size(0);
    auto K = A.size(1);
    auto N = B.size(1);
    
    auto C = torch::zeros({M, N}, torch::TensorOptions().dtype(torch::kFloat32).device(A.device()));
    
    auto A_half = A.to(torch::kFloat16);
    auto B_half = B.to(torch::kFloat16);
    
    dim3 block(32, 32);
    dim3 grid((N + 31) / 32, (M + 31) / 32);

        reinterpret_cast<half*>(B_half.data_ptr<at::Half>()),
        C.data_ptr<float>(),
        M, N, K, alpha, beta

}
