        thread_max = fmaxf(thread_max, __shfl_down_sync(0xffffffff, thread_max, offset));
    }
    
    // Block-level reduction using shared memory
    __shared__ float block_max;
    if (lane_id == 0) {
        atomicMax(reinterpret_cast<int*>(&block_max), __float_as_int(thread_max));
    }
    __syncthreads();
    
    if (tid == 0) {
        float absmax = block_max;
        scales[token_idx] = absmax / 127.0f;
        block_max = absmax;
    }
    __syncthreads();
    
    // Quantize with computed scale
    float scale = block_max / 127.0f;
    float inv_scale = (scale > 0.0f) ? 127.0f / block_max : 0.0f;
    
    for (int i = tid; i < hidden_size; i += stride) {
        float val = __half2float(row_in[i]) * inv_scale;
        row_out[i] = float_to_int8_rn(val);
    }
}

// Weight quantization kernel (static, per-tensor)
__global__ void quantize_weights_kernel(
    const half* __restrict__ weights,
    int8_t* __restrict__ weights_int8,
    float* __restrict__ scale,
    const int num_elements) {
    
    const int tid = threadIdx.x;
    const int stride = blockDim.x;
    
    // Find max absolute value
    float thread_max = 0.0f;
    for (int i = tid; i < num_elements; i += stride) {
        float val = fabsf(__half2float(weights[i]));
        thread_max = fmaxf(thread_max, val);
    }
    
    // Block reduction
    for (int offset = blockDim.x / 2; offset > 0; offset /= 2) {
        thread_max = fmaxf(thread_max, __shfl_down_sync(0xffffffff, thread_max, offset));
    }
    
    __shared__ float absmax;
    if (tid == 0) {
        absmax = thread_max;
        *scale = absmax / 127.0f;
    }
    __syncthreads();
    
    // Quantize
    float inv_scale = 127.0f / absmax;
    for (int i = tid; i < num_elements; i += stride) {
        float val = __half2float(weights[i]) * inv_scale;
        weights_int8[i] = float_to_int8_rn(val);
    }
}

// Optimized batched INT8 GEMM using tensor cores when possible
// For small batch sizes (M=16), use optimized single-kernel approach

__global__ void int8_gemm_small_batch_kernel(
    const half* __restrict__ A,
    const int8_t* __restrict__ B,
    half* __restrict__ C,
    const float scale_a,
    const float scale_b,
    const int M, const int N, const int K) {
    
    // For small M, each block handles one row
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int stride = blockDim.x;

    const half* a_row = A + row * K;
    half* c_row = C + row * N;
    
    // Process N dimension

            sum += a_val * b_val;
        }
        
        c_row[col] = __float2half(sum * scale_a * scale_b);
    }
}

        weight_fp16 = torch.randn(K, N, dtype=torch.float16, device='cuda')

        K = x.shape[1]
