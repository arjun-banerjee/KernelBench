import os
import torch
import torch.nn as nn
import torch.distributed as dist
from torch.utils.cpp_extension import load_inline


_CPP_SRC = r"""
#include <torch/extension.h>

torch::Tensor gemm_tf32(torch::Tensor A, torch::Tensor B, int64_t n_start, int64_t n_cols, double alpha);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("gemm_tf32", &gemm_tf32, "TF32 GEMM shard using cuBLAS (row-major A*B[:, shard])");
}
"""

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cublas_v2.h>
#include <cuda_runtime.h>

namespace {

static cublasHandle_t g_handle = nullptr;

inline void check_cublas(cublasStatus_t st) {
  TORCH_CHECK(st == CUBLAS_STATUS_SUCCESS, "cuBLAS call failed with status ", int(st));
}

inline cublasHandle_t get_handle() {
  if (g_handle == nullptr) {
    check_cublas(cublasCreate(&g_handle));
    check_cublas(cublasSetMathMode(g_handle, CUBLAS_TF32_TENSOR_OP_MATH));
  }
  check_cublas(cublasSetStream(g_handle, at::cuda::getCurrentCUDAStream()));
  return g_handle;
}

}  // namespace

// Computes C = alpha * A @ B[:, n_start:n_start+n_cols]
// A is row-major [M,K], B is row-major contiguous [K,N], C is row-major [M,n_cols].
// The call is expressed to cuBLAS as column-major C^T = B_shard^T * A^T.
torch::Tensor gemm_tf32(torch::Tensor A, torch::Tensor B, int64_t n_start, int64_t n_cols, double alpha_d) {
  TORCH_CHECK(A.is_cuda() && B.is_cuda(), "A and B must be CUDA tensors");
  TORCH_CHECK(A.scalar_type() == torch::kFloat32 && B.scalar_type() == torch::kFloat32,
              "A and B must be float32 tensors");
  TORCH_CHECK(A.dim() == 2 && B.dim() == 2, "A and B must be matrices");
  TORCH_CHECK(A.is_contiguous(), "A must be contiguous");
  TORCH_CHECK(B.is_contiguous(), "B must be contiguous");

  const int64_t M64 = A.size(0);
  const int64_t K64 = A.size(1);
  const int64_t KB64 = B.size(0);
  const int64_t N64 = B.size(1);
  TORCH_CHECK(K64 == KB64, "inner dimensions do not match");
  TORCH_CHECK(n_start >= 0 && n_cols > 0 && n_start + n_cols <= N64, "invalid N shard");
  TORCH_CHECK(M64 <= INT_MAX && K64 <= INT_MAX && N64 <= INT_MAX && n_cols <= INT_MAX,
              "problem too large for cuBLAS int interface");

  auto C = torch::empty({M64, n_cols}, A.options());

  const int M = static_cast<int>(M64);
  const int K = static_cast<int>(K64);
  const int N = static_cast<int>(N64);
  const int NC = static_cast<int>(n_cols);

  const float alpha = static_cast<float>(alpha_d);
  const float beta = 0.0f;

  const float* A_ptr = A.data_ptr<float>();
  const float* B_ptr = B.data_ptr<float>() + n_start;
  float* C_ptr = C.data_ptr<float>();

  cublasHandle_t h = get_handle();

  check_cublas(cublasGemmEx(
      h,
      CUBLAS_OP_N, CUBLAS_OP_N,
      NC, M, K,
      &alpha,
      B_ptr, CUDA_R_32F, N,
      A_ptr, CUDA_R_32F, K,
      &beta,
      C_ptr, CUDA_R_32F, NC,
      CUBLAS_COMPUTE_32F_FAST_TF32,
      CUBLAS_GEMM_DEFAULT_TENSOR_OP));

  return C;
}
"""

    def __init__(self, K, N, alpha=1.0, beta=0.0):
        super().__init__()

            if local_rank < torch.cuda.device_count():

        self.K = int(K)

        self.alpha = float(alpha)
        self.beta = float(beta)

        # Hidden reference initializes the GEMM right operand in the module.  Keep it
        # as fp32 and use Hopper TF32 tensor cores in the custom operator.
        self.register_buffer("B", torch.randn(self.K, self.N, device="cuda", dtype=torch.float32))

            # Use one consistent copy of the model weights for tensor-parallel execution.
            dist.broadcast(self.B, src=0)

    def forward(self, x):
