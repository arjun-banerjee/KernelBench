            }

            float scale_factor = expf(m_acc - m_prime);
            exp_sum = warp_reduce_sum(exp_sum);
            float l_prime = scale_factor * l_acc + exp_sum;
            
            // Update output accumulator
            #pragma unroll
            for (int k = 0; k < HEAD_DIM; ++k) {
                float v_sum = 0.0f;
                #pragma unroll
                for (int i = 0; i < BLOCK_N / WARP_SIZE; ++i) {
                    int n_idx = i * WARP_SIZE + lane_id;
                    if (n_idx < num_n_rows) {
                        v_sum += row_exp[i] * half_to_float(V_smem[n_idx * HEAD_DIM + k]);
                    }
                }
                v_sum = warp_reduce_sum(v_sum);
                if (lane_id == 0) {
                    O_acc[k] = scale_factor * O_acc[k] + v_sum;
                }
            }
            
            if (lane_id == 0) {
                m_acc = m_prime;
                l_acc = l_prime;
            }
        }
        
        __syncthreads();
    }
    
    // Normalize and write output
    if (warp_id < num_m_rows / 4 || (warp_id * 4 < num_m_rows)) {
        const int m_row_start = warp_id * 4;
        const int m_row_end = min(m_row_start + 4, num_m_rows);
        
        for (int m_row = m_row_start; m_row < m_row_end; ++m_row) {
            if (lane_id < HEAD_DIM / 8) {
                const int col = lane_id * 8;
                const int actual_row = m_block * BLOCK_M + m_row;
                
                #pragma unroll
                for (int c = 0; c < 8; ++c) {
                    if (col + c < HEAD_DIM) {
                        float val = O_acc[col + c] / l_acc;
                        O_smem[m_row * HEAD_DIM + col + c] = __float2half(val);
                    }
                }
            }
            
            if (lane_id == 0) {
                const int actual_row = m_block * BLOCK_M + m_row;
                L_smem[m_row] = m_acc + logf(l_acc);  // Store log-sum-exp
            }
        }
    }
    
    __syncthreads();
    
    // Write output to global memory
    for (int row = tid; row < num_m_rows; row += THREADS_PER_BLOCK) {
        const int actual_row = m_block * BLOCK_M + row;
        half* o_ptr = O + batch_idx * batch_stride + head_idx * head_stride 
                    + actual_row * HEAD_DIM;
        #pragma unroll
        for (int col = 0; col < HEAD_DIM; col += 8) {
            if (col + 8 <= HEAD_DIM) {
                float4 vec = *reinterpret_cast<float4*>(&O_smem[row * HEAD_DIM + col]);
                *reinterpret_cast<float4*>(o_ptr + col) = vec;
            }
        }
    }
}

// Fused attention kernel - simplified version for correctness
__global__ void flash_attn_fwd_simple_kernel(
    const half* __restrict__ Q,
    const half* __restrict__ K,
    const half* __restrict__ V,
    half* __restrict__ O,
    float* __restrict__ LSE,
    int batch_size, int num_heads, int num_kv_heads,
    int seq_len, int head_dim, int group_size
) {
    const int batch_idx = blockIdx.z;
    const int head_idx = blockIdx.y;
    const int seq_idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    const int kv_head_idx = head_idx / group_size;
    
    if (seq_idx >= seq_len) return;
    
    const int batch_stride = num_heads * seq_len * HEAD_DIM;
    const int head_stride = seq_len * HEAD_DIM;
    
    const half* q_ptr = Q + batch_idx * batch_stride + head_idx * head_stride + seq_idx * HEAD_DIM;
    const half* k_base = K + batch_idx * batch_stride + kv_head_idx * head_stride;
    const half* v_base = V + batch_idx * batch_stride + kv_head_idx * head_stride;
    
    // Compute attention scores
    float scores[2048];  // Max seq_len
    float max_score = -INFINITY;
    
    for (int kv_seq = 0; kv_seq < seq_len; ++kv_seq) {
        const half* k_ptr = k_base + kv_seq * HEAD_DIM;
        float score = 0.0f;
        
        #pragma unroll 8
        for (int d = 0; d < HEAD_DIM; ++d) {
            score += __half2float(q_ptr[d]) * __half2float(k_ptr[d]);
        }
        score *= softmax_scale;
        scores[kv_seq] = score;
        max_score = fmaxf(max_score, score);
    }
    
    // Softmax
    float exp_sum = 0.0f;
    for (int kv_seq = 0; kv_seq < seq_len; ++kv_seq) {
        exp_sum += expf(scores[kv_seq] - max_score);
    }
    
    float log_sum_exp = max_score + logf(exp_sum);
    
    // Compute weighted sum
    float output[HEAD_DIM] = {0.0f};
    for (int kv_seq = 0; kv_seq < seq_len; ++kv_seq) {
        float weight = expf(scores[kv_seq] - max_score) / exp_sum;
        const half* v_ptr = v_base + kv_seq * HEAD_DIM;
        
        #pragma unroll 8
        for (int d = 0; d < HEAD_DIM; ++d) {
            output[d] += weight * __half2float(v_ptr[d]);
        }
    }
    
    // Write output
    half* o_ptr = O + batch_idx * batch_stride + head_idx * head_stride + seq_idx * HEAD_DIM;
    for (int d = 0; d < HEAD_DIM; ++d) {
        o_ptr[d] = __float2half(output[d]);
    }
    
    // Write LSE
    float* lse_ptr = LSE + batch_idx * num_heads * seq_len + head_idx * seq_len + seq_idx;
    *lse_ptr = log_sum_exp;
}

// Optimized flash attention kernel for Hopper with better parallelism
__global__ void flash_attn_fwd_opt_kernel(
    const half* __restrict__ Q,
    const half* __restrict__ K,
    const half* __restrict__ V,
    half* __restrict__ O,
    float* __restrict__ LSE,
    int batch_size, int num_heads, int num_kv_heads,
    int seq_len, int head_dim, int group_size
) {
    extern __shared__ char smem[];
    
    const int batch_idx = blockIdx.z;
    const int head_idx = blockIdx.y;
    const int m_block = blockIdx.x;
    const int tid = threadIdx.x;
    const int lane = tid & 31;
    const int warp = tid >> 5;
    
    const int kv_head_idx = head_idx / group_size;
    const int m_start = m_block * BLOCK_M;
    const int m_end = min(m_start + BLOCK_M, seq_len);
    const int num_m = m_end - m_start;
    
    const int batch_stride = num_heads * seq_len * HEAD_DIM;
    const int head_stride = seq_len * HEAD_DIM;
    
    // Shared memory pointers
    half* Q_smem = (half*)smem;
    half* K_smem = Q_smem + BLOCK_M * HEAD_DIM;
    half* V_smem = K_smem + BLOCK_N * HEAD_DIM;
    float* O_smem = (float*)(V_smem + BLOCK_N * HEAD_DIM);
    float* L_smem = O_smem + BLOCK_M * HEAD_DIM;
    
    // Load Q block
    for (int i = tid; i < num_m * HEAD_DIM; i += THREADS_PER_BLOCK) {
        int row = i / HEAD_DIM;
        int col = i % HEAD_DIM;
        int global_row = m_start + row;
        int q_idx = batch_idx * batch_stride + head_idx * head_stride + global_row * HEAD_DIM + col;
        Q_smem[i] = Q[q_idx];
    }
    
    __syncthreads();
    
    // Initialize accumulators
    float o_acc[HEAD_DIM] = {0.0f};
    float l_acc = 0.0f;
    float m_acc = -INFINITY;
    
    // Process K/V in blocks
    const int num_n_blocks = (seq_len + BLOCK_N - 1) / BLOCK_N;
    
    for (int n_block = 0; n_block < num_n_blocks; ++n_block) {
        const int n_start = n_block * BLOCK_N;
        const int n_end = min(n_start + BLOCK_N, seq_len);
        const int num_n = n_end - n_start;
        
        // Load K/V block
        for (int i = tid; i < num_n * HEAD_DIM; i += THREADS_PER_BLOCK) {
            int row = i / HEAD_DIM;
            int col = i % HEAD_DIM;
            int global_row = n_start + row;
            int kv_idx = batch_idx * batch_stride + kv_head_idx * head_stride + global_row * HEAD_DIM + col;
            K_smem[i] = K[kv_idx];
            V_smem[i] = V[kv_idx];
        }
        
        __syncthreads();
        
        // Compute attention for each Q row assigned to this warp
        const int q_rows_per_warp = (num_m + NUM_WARPS - 1) / NUM_WARPS;
        const int q_row_start = warp * q_rows_per_warp;
        const int q_row_end = min(q_row_start + q_rows_per_warp, num_m);
        
        for (int q_row = q_row_start; q_row < q_row_end; ++q_row) {
            float row_max = -INFINITY;
            float row_sum = 0.0f;
            
            // Compute scores for this Q row against all K rows
            for (int k_row = lane; k_row < num_n; k_row += WARP_SIZE) {
                float score = 0.0f;
                for (int d = 0; d < HEAD_DIM; d += 4) {
                    float q0 = __half2float(Q_smem[q_row * HEAD_DIM + d]);
                    float q1 = __half2float(Q_smem[q_row * HEAD_DIM + d + 1]);
                    float q2 = __half2float(Q_smem[q_row * HEAD_DIM + d + 2]);
                    float q3 = __half2float(Q_smem[q_row * HEAD_DIM + d + 3]);
                    
                    float k0 = __half2float(K_smem[k_row * HEAD_DIM + d]);
                    float k1 = __half2float(K_smem[k_row * HEAD_DIM + d + 1]);
                    float k2 = __half2float(K_smem[k_row * HEAD_DIM + d + 2]);
                    float k3 = __half2float(K_smem[k_row * HEAD_DIM + d + 3]);
                    
                    score += (q0 * k0 + q1 * k1) + (q2 * k2 + q3 * k3);
                }
                score *= softmax_scale;
                
                O_smem[q_row * BLOCK_N + k_row] = score;
                row_max = fmaxf(row_max, score);
            }
            
            row_max = warp_reduce_max(row_max);
            float new_m = fmaxf(m_acc, row_max);
            float scale = expf(m_acc - new_m);
            
            for (int k_row = lane; k_row < num_n; k_row += WARP_SIZE) {
                float exp_score = expf(O_smem[q_row * BLOCK_N + k_row] - row_max);
                O_smem[q_row * BLOCK_N + k_row] = exp_score;
                row_sum += exp_score;
            }
            
            row_sum = warp_reduce_sum(row_sum);
            float new_l = scale * l_acc + row_sum;
            
            // Update output with weighted V
            for (int d = lane; d < HEAD_DIM; d += WARP_SIZE) {
                float v_sum = 0.0f;
                for (int k_row = 0; k_row < num_n; ++k_row) {
                    float weight = O_smem[q_row * BLOCK_N + k_row];
                    v_sum += weight * __half2float(V_smem[k_row * HEAD_DIM + d]);
                }
                v_sum = warp_reduce_sum(v_sum);
                if (lane == 0) {
                    o_acc[d] = scale * o_acc[d] + v_sum;
                }
            }
            
            if (lane == 0) {
                m_acc = new_m;
                l_acc = new_l;
            }
        }
        
        __syncthreads();
    }
    
    // Store output
    const int q_rows_per_warp = (num_m + NUM_WARPS - 1) / NUM_WARPS;
    const int q_row_start = warp * q_rows_per_warp;
    const int q_row_end = min(q_row_start + q_rows_per_warp, num_m);
    
    for (int q_row = q_row_start; q_row < q_row_end; ++q_row) {
        for (int d = lane; d < HEAD_DIM; d += WARP_SIZE) {
            float val = o_acc[d] / l_acc;
            int global_row = m_start + q_row;
            int o_idx = batch_idx * batch_stride + head_idx * head_stride + global_row * HEAD_DIM + d;
            O[o_idx] = __float2half(val);
        }
        
        if (lane == 0) {
            int global_row = m_start + q_row;
            int lse_idx = batch_idx * num_heads * seq_len + head_idx * seq_len + global_row;
            LSE[lse_idx] = m_acc + logf(l_acc);
        }
    }
}

std::vector<torch::Tensor> flash_attn_fwd_cuda(
    torch::Tensor Q,
    torch::Tensor K,
    torch::Tensor V,
    int num_heads,
    int num_kv_heads,
    int head_dim
) {
    const int batch_size = Q.size(0);
    const int seq_len = Q.size(2);
    const int group_size = num_heads / num_kv_heads;
    
    auto O = torch::empty_like(Q);
    auto LSE = torch::empty({batch_size, num_heads, seq_len}, 
                           torch::TensorOptions().dtype(torch::kFloat32).device(Q.device()));
    
    const int num_m_blocks = (seq_len + BLOCK_M - 1) / BLOCK_M;
    
    dim3 grid(num_m_blocks, num_heads, batch_size);
    dim3 block(THREADS_PER_BLOCK);
    
    // Shared memory size: Q + K + V + O (float for scores) + L
    const size_t smem_size = (BLOCK_M + BLOCK_N + BLOCK_N) * HEAD_DIM * sizeof(half) 
                           + BLOCK_M * BLOCK_N * sizeof(float)
                           + BLOCK_M * HEAD_DIM * sizeof(float);
    
    flash_attn_fwd_opt_kernel<<<grid, block, smem_size>>>(
        reinterpret_cast<half*>(Q.data_ptr<at::Half>()),
        reinterpret_cast<half*>(K.data_ptr<at::Half>()),
        reinterpret_cast<half*>(V.data_ptr<at::Half>()),
        reinterpret_cast<half*>(O.data_ptr<at::Half>()),
        LSE.data_ptr<float>(),
        batch_size, num_heads, num_kv_heads, seq_len, head_dim, group_size
    );
    
    C10_CUDA_CHECK(cudaGetLastError());
    
    return {O, LSE};
}

            output: Attention output [batch, num_heads, seq_len, head_dim]
        """

        seq_len = q.size(2)

        q = q.contiguous()
        k = k.contiguous()
        v = v.contiguous()
        
        # Check dtype and call appropriate kernel
        if q.dtype == torch.float16:
            output, lse = self.flash_attn.flash_attn_fwd_cuda(
                q, k, v, self.num_heads, self.num_kv_heads, self.head_dim
            )
        elif q.dtype == torch.bfloat16:
            output, lse = self.flash_attn.flash_attn_fwd_bf16_cuda(

            )
