    const float sin_f,

        x_idx = rot_offset;

    const int64_t pos = positions[token_idx];
    
    // Shared memory for cos/sin values - cached for entire block
    // Using float for precision during computation
    extern __shared__ float shared_cos_sin[];
    float* cos_shared = shared_cos_sin;
    float* sin_shared = shared_cos_sin + embed_dim;
    
    // Cooperative load of cos/sin values into shared memory
    // Each thread loads multiple values if blockDim.x < embed_dim
    const half* cache_ptr = cos_sin_cache + pos * rot_dim;
    
    for (int i = threadIdx.x; i < embed_dim; i += blockDim.x) {
        cos_shared[i] = __half2float(cache_ptr[i]);
        sin_shared[i] = __half2float(cache_ptr[embed_dim + i]);
    }
    
    __syncthreads();
    
    // Process query heads
    const int nq = num_heads * embed_dim;
    
    for (int i = threadIdx.x; i < nq; i += blockDim.x) {
        const int head_idx = i / embed_dim;
        const int rot_offset = i % embed_dim;
        
        const int64_t token_head = token_idx * query_stride + 
                                   head_idx * head_stride + 
                                   rope_dim_offset;
        
        float cos_f = cos_shared[rot_offset];
        float sin_f = inverse ? -sin_shared[rot_offset] : sin_shared[rot_offset];
        
        apply_rotary_pair<IS_NEOX>(query + token_head, cos_f, sin_f, rot_offset, embed_dim);
    }
    
    // Process key heads if present
    if (key != nullptr) {
        const int nk = num_kv_heads * embed_dim;
        
        for (int i = threadIdx.x; i < nk; i += blockDim.x) {
            const int head_idx = i / embed_dim;
            const int rot_offset = i % embed_dim;
            
            const int64_t token_head = token_idx * key_stride + 
                                       head_idx * head_stride + 
                                       rope_dim_offset;
            
            float cos_f = cos_shared[rot_offset];
            float sin_f = inverse ? -sin_shared[rot_offset] : sin_shared[rot_offset];
            
            apply_rotary_pair<IS_NEOX>(key + token_head, cos_f, sin_f, rot_offset, embed_dim);
        }
    }
}

// Vectorized version using half2 for better memory throughput
template <bool IS_NEOX>
__global__ void rotary_embedding_hopper_vec2(
    const int64_t* __restrict__ positions,
    half* __restrict__ query,
    half* __restrict__ key,
    const half* __restrict__ cos_sin_cache,
    const int rot_dim,
    const int64_t query_stride,
    const int64_t key_stride,
    const int64_t head_stride,
    const int num_heads,
    const int num_kv_heads,
    const int head_size,
    const int64_t rope_dim_offset,

    const int token_idx = blockIdx.x;
    const int embed_dim = rot_dim / 2;
    const int64_t pos = positions[token_idx];
    
    // Shared memory for cos/sin with padding to avoid bank conflicts
    extern __shared__ float shared_mem[];
    float* cos_shared = shared_mem;
    float* sin_shared = shared_mem + embed_dim;
    
    const half* cache_ptr = cos_sin_cache + pos * rot_dim;
    
    // Vectorized load of cos/sin using half2
    const int vec_embed_dim = embed_dim / 2;
    const half2* cache_vec = reinterpret_cast<const half2*>(cache_ptr);
    const half2* cache_sin_vec = reinterpret_cast<const half2*>(cache_ptr + embed_dim);
    float2* cos_shared_vec = reinterpret_cast<float2*>(cos_shared);
    float2* sin_shared_vec = reinterpret_cast<float2*>(sin_shared);
    
    for (int i = threadIdx.x; i < vec_embed_dim; i += blockDim.x) {
        half2 cos_h2 = cache_vec[i];
        half2 sin_h2 = cache_sin_vec[i];
        float2 cos_f2 = __half22float2(cos_h2);
        float2 sin_f2 = __half22float2(sin_h2);
        cos_shared_vec[i] = cos_f2;
        sin_shared_vec[i] = sin_f2;
    }
    
    __syncthreads();
    
    // Process query with vectorization
    const int nq = num_heads * embed_dim;
    
    for (int i = threadIdx.x; i < nq; i += blockDim.x) {
        const int head_idx = i / embed_dim;
        const int rot_offset = i % embed_dim;
        
        const int64_t token_head = token_idx * query_stride + 
                                   head_idx * head_stride + 
                                   rope_dim_offset;
        
        float cos_f = cos_shared[rot_offset];
        float sin_f = inverse ? -sin_shared[rot_offset] : sin_shared[rot_offset];
        
        apply_rotary_pair<IS_NEOX>(query + token_head, cos_f, sin_f, rot_offset, embed_dim);
    }
    
    if (key != nullptr) {
        const int nk = num_kv_heads * embed_dim;
        
        for (int i = threadIdx.x; i < nk; i += blockDim.x) {
            const int head_idx = i / embed_dim;

            const int64_t token_head = token_idx * key_stride + 
                                       head_idx * head_stride + 
                                       rope_dim_offset;
            
            float cos_f = cos_shared[rot_offset];
            float sin_f = inverse ? -sin_shared[rot_offset] : sin_shared[rot_offset];
            
            apply_rotary_pair<IS_NEOX>(key + token_head, cos_f, sin_f, rot_offset, embed_dim);
        }
    }
}

    int num_kv_heads = key.has_value() ? key_hidden_size / head_size : num_heads;

    // Shared memory size: 2 * embed_dim floats for cos/sin

                    query.data_ptr<at::Half>(),

                    positions.data_ptr<int64_t>(),

class ModelNew(nn.Module):
    def __init__(self, rot_dim):
        super().__init__()
        self.rot_dim = rot_dim
        self.head_size = 128  # Standard head size
        
    def forward(self, query, key, cos, sin):
        """
        Apply rotary position embedding to query and key tensors.
        
        Args:
            query: [num_tokens, num_heads, head_size] tensor
            key: [num_tokens, num_kv_heads, head_size] tensor  
            cos: [num_tokens, rot_dim] tensor
            sin: [num_tokens, rot_dim] tensor (combined with cos in cache format)
        
        Returns:
            query_out, key_out: Rotated tensors with same shape as input
        """
        num_tokens = query.size(0)
        num_heads = query.size(1)
        num_kv_heads = key.size(1)
        head_size = query.size(2)
        
        # Create positions tensor [0, 1, 2, ..., num_tokens-1]
        positions = torch.arange(num_tokens, dtype=torch.int64, device=query.device)
        
        # Combine cos and sin into cache format [num_tokens, rot_dim]
        # First half is cos, second half is sin
        cos_sin_cache = torch.cat([cos, sin], dim=-1)

        query_out = query.clone()
        key_out = key.clone()
        
        # Call optimized CUDA kernel
        rotary_embedding_module.rotary_embedding_cuda(
            positions,
            query_out,
            key_out,
            self.head_size,
            cos_sin_cache,
            True,  # is_neox style
            0,     # rope_dim_offset
            False  # inverse
        )
        
        return query_out, key_out
