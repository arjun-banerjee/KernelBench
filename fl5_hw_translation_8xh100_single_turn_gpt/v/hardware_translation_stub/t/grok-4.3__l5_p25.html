import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>
#include <cooperative_groups.h>

__global__ void grouped_gemm_fp16_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    half* __restrict__ C,
    int groups, int M, int N, int K,
    float alpha, float beta) {
    
    int g = blockIdx.z;
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (g >= groups || row >= M || col >= N) return;
    
    const half* A_g = A + g * M * K;
    const half* B_g = B + g * K * N;
    half* C_g = C + g * M * N;
    
    float sum = 0.0f;
    for (int k = 0; k < K; ++k) {
        sum += __half2float(A_g[row * K + k]) * __half2float(B_g[k * N + col]);
    }
    float c_val = (beta == 0.0f) ? 0.0f : __half2float(C_g[row * N + col]);
    C_g[row * N + col] = __float2half(alpha * sum + beta * c_val);
}

torch::Tensor grouped_gemm_cuda(torch::Tensor A, torch::Tensor B, float alpha, float beta) {
    int groups = A.size(0);
    int M = A.size(1);
    int K = A.size(2);
    int N = B.size(2);
    
    auto C = torch::zeros({groups, M, N}, A.options().dtype(torch::kHalf));
    
    dim3 threads(16, 16);
    dim3 blocks((N + 15) / 16, (M + 15) / 16, groups);
    
    grouped_gemm_fp16_kernel<<<blocks, threads>>>(
        reinterpret_cast<half*>(A.data_ptr<at::Half>()),
        reinterpret_cast<half*>(B.data_ptr<at::Half>()),
        reinterpret_cast<half*>(C.data_ptr<at::Half>()),
        groups, M, N, K, alpha, beta);
    
    return C;
}
