import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void gather_gemm_scatter_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    const int* __restrict__ indices,
    float* __restrict__ D,
    int M, int K, int N, int IDX, float alpha, float beta) {
    
    int row = blockIdx.x * blockDim.x + threadIdx.x;
    int col_idx = blockIdx.y * blockDim.y + threadIdx.y;
    
    if (row < M && col_idx < IDX) {
        int col = indices[col_idx];
        float sum = 0.0f;
        for (int k = 0; k < K; ++k) {
            sum += __half2float(A[row * K + k]) * __half2float(B[k * N + col]);
        }
        D[row * N + col] = alpha * sum + beta * D[row * N + col];
    }
}

torch::Tensor gather_gemm_scatter_cuda(
    torch::Tensor A,
    torch::Tensor B,
    torch::Tensor indices,
    float alpha, float beta) {
    
    int M = A.size(0);
    int K = A.size(1);
    int N = B.size(1);
    int IDX = indices.size(0);
    
    auto D = torch::zeros({M, N}, A.options().dtype(torch::kFloat32));
    
    dim3 threads(16, 16);
    dim3 blocks((M + 15) / 16, (IDX + 15) / 16);
    
    gather_gemm_scatter_kernel<<<blocks, threads>>>(

        reinterpret_cast<half*>(B.data_ptr<at::Half>()),
        indices.data_ptr<int>(),
        D.data_ptr<float>(),
        M, K, N, IDX, alpha, beta);
    
    return D;
}
