import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_fp16.h>
#include <cmath>

static inline __device__ int8_t float_to_int8_rn(float x) {
  uint32_t dst;
  asm volatile("cvt.rni.sat.s8.f32 %0, %1;" : "=r"(dst) : "f"(x));
  return reinterpret_cast<const int8_t&>(dst);
}

template <typename scalar_t>
__global__ void static_scaled_int8_quant_kernel_h100(
    const scalar_t* __restrict__ input, int8_t* __restrict__ output,
    const float scale, const int hidden_size) {
  const int tid = threadIdx.x;
  const int stride = blockDim.x;
  const int64_t token_idx = blockIdx.x;

  const scalar_t* row_in = input + token_idx * hidden_size;
  int8_t* row_out = output + token_idx * hidden_size;

  for (int i = tid; i < hidden_size; i += stride) {
    float v = static_cast<float>(row_in[i]);
    row_out[i] = float_to_int8_rn(v / scale);
  }
}

void static_scaled_int8_quant_cuda(torch::Tensor input, torch::Tensor out, float scale) {
  int hidden_size = input.size(-1);
  int num_tokens = input.numel() / hidden_size;
  dim3 grid(num_tokens);
  dim3 block(256);
  const cudaStream_t stream = at::cuda::getCurrentCUDAStream();
  static_scaled_int8_quant_kernel_h100<at::Half><<<grid, block, 0, stream>>>(
      input.data_ptr<at::Half>(), out.data_ptr<int8_t>(), scale, hidden_size);
}

torch::Tensor w8a8_gemm(torch::Tensor input, torch::Tensor weight, float scale) {
  auto options = torch::TensorOptions().dtype(torch::kFloat16).device(input.device());
  auto out = torch::empty({input.size(0), weight.size(0)}, options);
  auto qinput = torch::empty_like(input, torch::kInt8);
  static_scaled_int8_quant_cuda(input, qinput, scale);
  auto deq_in = qinput.to(torch::kFloat16) * scale;
  out = torch::matmul(deq_in, weight.t());
  return out;
}
