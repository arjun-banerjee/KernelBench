        uf.x = fmaxf(fminf(uf.x, limit), -limit);
        uf.y = fmaxf(fminf(uf.y, limit), -limit);
        g = float2_to_half2(gf);
        u = float2_to_half2(uf);
    }
    return packed_mul_device(packed_silu_device(g), u);
}

// H100-optimized kernel using vectorized loads with improved occupancy
template <typename scalar_t, bool use_vec, bool HAS_CLAMP>
__global__ void silu_and_mul_kernel(
    scalar_t* __restrict__ out,
    const scalar_t* __restrict__ input,
    const int d,
    const float limit) {
    
    const int64_t token_idx = blockIdx.x;
    const scalar_t* x_ptr = input + token_idx * 2 * d;  // Gate portion
    const scalar_t* y_ptr = x_ptr + d;                   // Up portion
    scalar_t* out_ptr = out + token_idx * d;

    if constexpr (use_vec) {
        // Vectorized path: use 128-bit loads (8 half elements)
        // H100 has excellent throughput for these operations
        using vec_t = uint4;  // 128-bit vector
        constexpr int VEC_ELTS = 16 / sizeof(scalar_t);  // 8 for half, 4 for float
        constexpr int PACKED_ELTS = 2;  // Number of packed elements (half2 = 2)
        
        const vec_t* x_vec = reinterpret_cast<const vec_t*>(x_ptr);
        const vec_t* y_vec = reinterpret_cast<const vec_t*>(y_ptr);
        vec_t* out_vec = reinterpret_cast<vec_t*>(out_ptr);
        
        const int num_vecs = d / VEC_ELTS;
        
        // Use __ldg for read-only cache optimization on H100
        for (int i = threadIdx.x; i < num_vecs; i += blockDim.x) {
            vec_t xv = __ldg(&x_vec[i]);
            vec_t yv = __ldg(&y_vec[i]);
            
            auto* x_halves = reinterpret_cast<half2*>(&xv);
            auto* y_halves = reinterpret_cast<half2*>(&yv);
            auto* out_halves = reinterpret_cast<half2*>(&xv);  // Reuse xv for output
            
            #pragma unroll
            for (int j = 0; j < VEC_ELTS / PACKED_ELTS; j++) {
                out_halves[j] = packed_compute_swiglu<HAS_CLAMP>(x_halves[j], y_halves[j], limit);
            }
            
            out_vec[i] = xv;
        }
        
        // Handle remaining elements
        const int remaining_start = num_vecs * VEC_ELTS;
        for (int idx = remaining_start + threadIdx.x; idx < d; idx += blockDim.x) {
            const scalar_t gate = __ldg(&x_ptr[idx]);
            const scalar_t up = __ldg(&y_ptr[idx]);
            out_ptr[idx] = compute_swiglu<scalar_t, HAS_CLAMP>(gate, up, limit);
        }
    } else {
        // Scalar path for unaligned data
        for (int64_t idx = threadIdx.x; idx < d; idx += blockDim.x) {
            const scalar_t gate = __ldg(&x_ptr[idx]);
            const scalar_t up = __ldg(&y_ptr[idx]);
            out_ptr[idx] = compute_swiglu<scalar_t, HAS_CLAMP>(gate, up, limit);
        }
    }
}

// Alternative kernel with interleaved processing for better cache utilization
template <typename scalar_t>
__global__ void silu_and_mul_kernel_interleaved(
    scalar_t* __restrict__ out,
    const scalar_t* __restrict__ input,
    const int d) {
    
    const int64_t token_idx = blockIdx.x;
    const scalar_t* in_ptr = input + token_idx * 2 * d;
    scalar_t* out_ptr = out + token_idx * d;

    // Process 4 elements at a time using half2 operations
    const int stride = blockDim.x;
    const int vec_size = 4;  // Process 4 half values per iteration
    
    for (int i = threadIdx.x * vec_size; i < d; i += stride * vec_size) {
        // Load gate and up values
        half2 gate0, gate1, up0, up1;
        
        if (i + vec_size <= d) {
            gate0 = *reinterpret_cast<const half2*>(in_ptr + i);
            gate1 = *reinterpret_cast<const half2*>(in_ptr + i + 2);
            up0 = *reinterpret_cast<const half2*>(in_ptr + d + i);
            up1 = *reinterpret_cast<const half2*>(in_ptr + d + i + 2);
            
            gate0 = packed_mul_device(packed_silu_device(gate0), up0);
            gate1 = packed_mul_device(packed_silu_device(gate1), up1);
            
            *reinterpret_cast<half2*>(out_ptr + i) = gate0;
            *reinterpret_cast<half2*>(out_ptr + i + 2) = gate1;
        } else {
            // Handle remaining elements
            for (int j = i; j < d; j++) {
                if (j < d) {
                    const scalar_t gate = __ldg(&in_ptr[j]);
                    const scalar_t up = __ldg(&in_ptr[d + j]);
                    out_ptr[j] = silu_device(gate) * up;
                }
            }
        }
    }
}

// Main entry point function
void silu_and_mul_impl(torch::Tensor& out, torch::Tensor& input) {
    int d = input.size(-1) / 2;
    int64_t num_tokens = input.numel() / input.size(-1);
    
    if (num_tokens == 0) return;
    
    dim3 grid(num_tokens);
    
    // H100-optimized block sizing
    // Aim for good occupancy while minimizing register pressure
    int block_size;
    const bool is_aligned = (reinterpret_cast<uintptr_t>(input.data_ptr()) & 15) == 0 &&
                           (reinterpret_cast<uintptr_t>(out.data_ptr()) & 15) == 0;
    const bool use_vec = is_aligned && (d % 8 == 0);  // 128-bit = 8 half elements
    
    if (use_vec) {
        // For vectorized path, use blocks of 256-512 threads
        // H100 can handle larger blocks efficiently
        block_size = std::min((d / 8 + 31) / 32 * 32, 512);
        block_size = std::max(block_size, 128);  // Minimum 128 threads
    } else {
        block_size = std::min(d, 1024);
        block_size = std::max(block_size, 128);
    }
    
    dim3 block(block_size);
    
    cudaStream_t stream = c10::cuda::getCurrentCUDAStream();
    
    AT_DISPATCH_FLOATING_TYPES_AND2(
        at::ScalarType::Half, at::ScalarType::BFloat16,
        input.scalar_type(), "silu_and_mul_kernel", [&] {
            if (use_vec) {
                silu_and_mul_kernel<scalar_t, true, false>
                    <<<grid, block, 0, stream>>>(
                        out.data_ptr<scalar_t>(),
                        input.data_ptr<scalar_t>(),
                        d, 0.0f);
            } else {
                silu_and_mul_kernel<scalar_t, false, false>
                    <<<grid, block, 0, stream>>>(
                        out.data_ptr<scalar_t>(),
                        input.data_ptr<scalar_t>(),

            }
        });
}
