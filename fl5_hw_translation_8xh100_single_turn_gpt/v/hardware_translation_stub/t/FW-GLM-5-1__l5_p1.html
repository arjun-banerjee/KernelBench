    __shared__ half k_chunk[CHUNK_SIZE][HEAD_SIZE];
    __shared__ half v_chunk[HEAD_SIZE][CHUNK_SIZE];
    __shared__ float logits_chunk[CHUNK_SIZE];
    __shared__ float out_acc[HEAD_SIZE];

    const half* query_ptr = query + seq_idx * q_stride + q_head_idx * HEAD_SIZE;
    for (int i = threadIdx.x; i < HEAD_SIZE; i += BLOCK_THREADS) {
        q_shared[i] = query_ptr[i];
    }

    for (int i = threadIdx.x; i < HEAD_SIZE; i += BLOCK_THREADS) {
        out_acc[i] = 0.0f;
    }

    float global_max = -INFINITY;
    float global_sum = 0.0f;
    
    const int num_chunks = (seq_len + CHUNK_SIZE - 1) / CHUNK_SIZE;
    
    for (int chunk_idx = 0; chunk_idx < num_chunks; chunk_idx++) {
        const int chunk_start = chunk_idx * CHUNK_SIZE;
        const int chunk_len = min(CHUNK_SIZE, seq_len - chunk_start);

        const int block_start = chunk_start / BLOCK_SIZE;
        const int block_end = (chunk_start + chunk_len + BLOCK_SIZE - 1) / BLOCK_SIZE;

            const int block_idx = global_token / BLOCK_SIZE;
            const int block_offset = global_token % BLOCK_SIZE;

            const half* k_block_ptr = key_cache + physical_block_id * kv_block_stride + 
                                       kv_head_idx * kv_head_stride;
            const int x_idx = head_dim % 8;
            const int head_x_idx = head_dim / 8;
            k_chunk[token_in_chunk][head_dim] = k_block_ptr[head_x_idx * BLOCK_SIZE * 8 + block_offset * 8 + x_idx];
            
            // Load V
            const half* v_block_ptr = value_cache + physical_block_id * kv_block_stride + 
                                       kv_head_idx * kv_head_stride;
            v_chunk[head_dim][token_in_chunk] = v_block_ptr[head_dim * BLOCK_SIZE + block_offset];
        }

        float chunk_max = -INFINITY;

            float score = 0.0f;
            // Use tensor cores for QK computation
            #pragma unroll 8
            for (int j = lane_id; j < HEAD_SIZE; j += WARP_SIZE) {
                score += __half2float(q_shared[j]) * __half2float(k_chunk[i][j]);
            }
            
            // Warp reduction
            #pragma unroll
            for (int offset = WARP_SIZE / 2; offset > 0; offset /= 2) {
                score += __shfl_down_sync(0xffffffff, score, offset);
            }

        const float scale_old = expf(global_max - new_max);
        const float scale_chunk = expf(chunk_max - new_max);
        
        // Compute exp and sum
        float chunk_sum = 0.0f;
        for (int i = warp_id; i < chunk_len; i += NUM_WARPS) {
            const float exp_score = expf(logits_chunk[i] - new_max);
            logits_chunk[i] = exp_score;
            chunk_sum += exp_score;
        }
        
        // Warp reduce chunk sum
        #pragma unroll
        for (int offset = WARP_SIZE / 2; offset > 0; offset /= 2) {
            chunk_sum += __shfl_down_sync(0xffffffff, chunk_sum, offset);
        }
        
        __shared__ float chunk_sum_shared;
        if (lane_id == 0) chunk_sum_shared = chunk_sum;
        __syncthreads();
        chunk_sum = chunk_sum_shared;
        
        // Update global sum
        const float new_sum = global_sum * scale_old + chunk_sum;

        for (int head_dim = warp_id; head_dim < HEAD_SIZE; head_dim += NUM_WARPS) {
            float val = 0.0f;
            for (int i = 0; i < chunk_len; i++) {
                val += logits_chunk[i] * __half2float(v_chunk[head_dim][i]);
            }
            
            // Update accumulator with online softmax correction
            out_acc[head_dim] = out_acc[head_dim] * scale_old + val;
        }
        
        global_max = new_max;
        global_sum = new_sum;
        __syncthreads();
    }
    
    // Final output: normalize by sum and convert to half
    for (int i = threadIdx.x; i < HEAD_SIZE; i += BLOCK_THREADS) {
        out[seq_idx * gridDim.x * HEAD_SIZE + q_head_idx * HEAD_SIZE + i] = 
            __float2half(out_acc[i] / global_sum);
    }
}
