    float max_score = -INFINITY;
    float sum_exp = 0.0f;

    for (int kv_start = 0; kv_start < seq_len; kv_start += BLOCK_SIZE) {

        for (int kv_local = tid; kv_local < kv_end * HEAD_SIZE; kv_local += blockDim.x) {

            for (int e = 0; e < elems_per_thread; e++) {

            }

            score = warp_reduce_sum(score);
            
            if (tid % WARP_SIZE == 0) {
                s_scores[kv_local] = score * scale;
            }
        }
        __syncthreads();
        
        // Find max score in this tile
        for (int kv_local = tid; kv_local < kv_end; kv_local += blockDim.x) {
            float tile_max = s_scores[kv_local];
            tile_max = block_reduce_max(tile_max, s_reduce);
            if (tid == 0) {
                max_score = fmaxf(max_score, tile_max);
            }
        }
        __syncthreads();

        for (int kv_local = tid; kv_local < kv_end * HEAD_SIZE; kv_local += blockDim.x) {
            const int kv_pos = kv_local / HEAD_SIZE;
            const int h = kv_local % HEAD_SIZE;
            s_v[kv_pos][h] = v_base[(kv_start + kv_pos) * HEAD_SIZE + h];
        }
        __syncthreads();

        // Note: This is a simplified version; proper implementation needs global max

            float score = s_scores[kv_local];
            float weight = expf(score - max_score);
            
            #pragma unroll
            for (int e = 0; e < elems_per_thread; e++) {
                o_acc[e] += weight * __half2float(s_v[kv_local][head_offset + e]);
            }
        }
        __syncthreads();
    }
    
    // Write output
    #pragma unroll
    for (int e = 0; e < elems_per_thread; e++) {
        o_ptr[head_offset + e] = __float2half(o_acc[e]);
    }
}

// Flash Attention style kernel optimized for H100
template<int HEAD_SIZE, int BLOCK_M, int BLOCK_N, int NUM_WARPS>
__global__ void flash_attention_h100(
    const half* __restrict__ Q,
    const half* __restrict__ K,
    const half* __restrict__ V,
    half* __restrict__ O,
    const int batch_size,
    const int num_heads,
    const int num_kv_heads,
    const int seq_len) {
    
    // Shared memory tiles - H100 has 228KB per SM
    __shared__ half s_Q[BLOCK_M][HEAD_SIZE];
    __shared__ half s_K[BLOCK_N][HEAD_SIZE];
    __shared__ half s_V[BLOCK_N][HEAD_SIZE];
    __shared__ float s_S[BLOCK_M][BLOCK_N];
    __shared__ float s_O[BLOCK_M][HEAD_SIZE];
    
    const int batch_idx = blockIdx.x;
    const int head_idx = blockIdx.y;
    const int m_block = blockIdx.z;  // Which block of Q sequence
    
    // GQA mapping
    const int kv_head_idx = head_idx / (num_heads / num_kv_heads);
    
    // Strides
    const int q_stride_b = num_heads * seq_len * HEAD_SIZE;
    const int kv_stride_b = num_kv_heads * seq_len * HEAD_SIZE;
    
    const half* Q_ptr = Q + batch_idx * q_stride_b + head_idx * seq_len * HEAD_SIZE;
    const half* K_ptr = K + batch_idx * kv_stride_b + kv_head_idx * seq_len * HEAD_SIZE;
    const half* V_ptr = V + batch_idx * kv_stride_b + kv_head_idx * seq_len * HEAD_SIZE;
    half* O_ptr = O + batch_idx * q_stride_b + head_idx * seq_len * HEAD_SIZE;
    
    const int tid = threadIdx.x;
    const int warp_id = tid / WARP_SIZE;
    const int lane_id = tid % WARP_SIZE;
    const int num_warps = NUM_WARPS;
    
    const float scale = 1.0f / sqrtf((float)HEAD_SIZE);
    
    // Range of Q this block processes
    const int m_start = m_block * BLOCK_M;
    const int m_end = min(m_start + BLOCK_M, seq_len);
    const int m_size = m_end - m_start;
    
    // Each thread handles part of the head dimension
    const int threads_per_row = HEAD_SIZE / 16;  // Each thread handles 16 elements
    const int row_per_warp = WARP_SIZE / threads_per_row;
    
    // Initialize output accumulator for this Q block
    float o_acc[BLOCK_M][16] = {0.0f};  // Each thread accumulates for its assigned Q rows
    float m_acc[BLOCK_M] = {-INFINITY};
    float l_acc[BLOCK_M] = {0.0f};
    
    // Load Q block into shared memory
    for (int m = 0; m < m_size; m++) {
        const int q_pos = m_start + m;
        for (int h = tid; h < HEAD_SIZE; h += blockDim.x) {
            s_Q[m][h] = Q_ptr[q_pos * HEAD_SIZE + h];
        }
    }
    __syncthreads();
    
    // Process KV in blocks
    for (int n_start = 0; n_start < seq_len; n_start += BLOCK_N) {
        const int n_end = min(n_start + BLOCK_N, seq_len);
        const int n_size = n_end - n_start;
        
        // Load K, V blocks
        for (int n = 0; n < n_size; n++) {
            const int kv_pos = n_start + n;
            for (int h = tid; h < HEAD_SIZE; h += blockDim.x) {
                s_K[n][h] = K_ptr[kv_pos * HEAD_SIZE + h];
                s_V[n][h] = V_ptr[kv_pos * HEAD_SIZE + h];
            }
        }
        __syncthreads();
        
        // Compute attention scores Q * K^T
        // Assign each warp to compute scores for subset of Q positions
        const int m_per_warp = (m_size + num_warps - 1) / num_warps;
        const int m_lo = warp_id * m_per_warp;
        const int m_hi = min(m_lo + m_per_warp, m_size);
        
        for (int m = m_lo; m < m_hi; m++) {
            float max_s = -INFINITY;
            
            // Compute scores for all N positions
            for (int n = lane_id; n < n_size; n += WARP_SIZE) {
                float score = 0.0f;
                
                // Dot product - vectorized using ldg
                const half* q_row = s_Q[m];
                const half* k_row = s_K[n];
                
                for (int h = 0; h < HEAD_SIZE; h += 4) {
                    float4 q_vec = {__half2float(q_row[h]), __half2float(q_row[h+1]),
                                    __half2float(q_row[h+2]), __half2float(q_row[h+3])};
                    float4 k_vec = {__half2float(k_row[h]), __half2float(k_row[h+1]),
                                    __half2float(k_row[h+2]), __half2float(k_row[h+3])};
                    score += q_vec.x * k_vec.x + q_vec.y * k_vec.y + 
                             q_vec.z * k_vec.z + q_vec.w * k_vec.w;
                }
                
                score *= scale;
                s_S[m][n] = score;
                max_s = fmaxf(max_s, score);
            }
            
            // Warp reduction for max
            max_s = warp_reduce_max(max_s);
            
            // Update global max with online softmax
            if (lane_id == 0) {
                float new_max = fmaxf(m_acc[m], max_s);
                // Rescale accumulator
                float rescale = expf(m_acc[m] - new_max);
                // This is simplified; proper flash attention handles this correctly
                m_acc[m] = new_max;
            }
        }
        __syncthreads();
        
        // Softmax and V accumulation (simplified)
        for (int m = 0; m < m_size; m++) {
            float row_max = -INFINITY;
            for (int n = 0; n < n_size; n++) {
                row_max = fmaxf(row_max, s_S[m][n]);
            }
            
            float row_sum = 0.0f;
            for (int n = 0; n < n_size; n++) {
                row_sum += expf(s_S[m][n] - row_max);
            }
            
            // Accumulate weighted V
            if (warp_id < m_size && warp_id < BLOCK_M) {
                const int m_idx = warp_id;
                for (int e = lane_id; e < 16; e += WARP_SIZE) {
                    const int h_base = e;
                    float acc = 0.0f;
                    for (int n = 0; n < n_size; n++) {
                        float weight = expf(s_S[m_idx][n] - row_max) / row_sum;
                        acc += weight * __half2float(s_V[n][h_base]);
                    }
                    o_acc[m_idx][e] = acc;
                }
            }
        }
        __syncthreads();
    }
    
    // Write output
    for (int m = 0; m < m_size; m++) {
        const int q_pos = m_start + m;
        for (int h = tid; h < HEAD_SIZE; h += blockDim.x) {
            float val = 0.0f;
            // Accumulate from thread-local storage
            O_ptr[q_pos * HEAD_SIZE + h] = __float2half(val);
        }
    }
}

// High-performance SDPA kernel for H100
__global__ void sdpa_h100_kernel(
    const half* __restrict__ query,
    const half* __restrict__ key, 
    const half* __restrict__ value,
    half* __restrict__ output,
    const int batch_size,
    const int num_heads,
    const int num_kv_heads,
    const int seq_len,
    const int head_size) {

    extern __shared__ char smem[];
    half* s_kv = reinterpret_cast<half*>(smem);
    
    const int bid = blockIdx.x;
    const int batch_idx = bid / num_heads;
    const int head_idx = bid % num_heads;
    const int q_pos = threadIdx.y;  // Each thread block handles multiple Q positions
    const int tid_x = threadIdx.x;

    const half* v_base = value + batch_idx * kv_stride + kv_head_idx * seq_len * head_size;
    half* o_base = output + batch_idx * q_stride + head_idx * seq_len * head_size;
    
    const float scale = 1.0f / sqrtf((float)head_size);

    constexpr int VEC_SIZE = 4;
    const int h_start = tid_x * VEC_SIZE;

            }

                }
            }
        }
        __syncthreads();

        }
    }
