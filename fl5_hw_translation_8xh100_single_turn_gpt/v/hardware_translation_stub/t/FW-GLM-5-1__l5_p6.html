// Hopper-optimized reduction kernel using async copies and warpgroup operations
template<typename T, int TILE_SIZE, int NUM_THREADS>
__launch_bounds__(NUM_THREADS, 2)  // 2 thread blocks per SM for better occupancy
__global__ void allreduce_hopper_kernel(
    T* __restrict__ local_buffer,
    T** __restrict__ peer_buffers,  // Array of pointers to peer buffers
    Signal** __restrict__ signals,  // Signals for coordination
    T* __restrict__ output,
    size_t num_elements,
    int rank,
    int world_size,
    int step) {

    __shared__ __align__(16) T shared_data[TILE_SIZE];
    __shared__ int block_phase;
    
    auto grid = cg::this_grid();
    auto block = cg::this_thread_block();
    auto warp = cg::tiled_partition<32>(block);
    
    const int tid = threadIdx.x;
    const int warp_id = threadIdx.x / WARP_SIZE;
    const int lane_id = threadIdx.x % WARP_SIZE;
    const int num_warps = blockDim.x / WARP_SIZE;
    
    // Calculate work range for this block
    const size_t block_start = blockIdx.x * TILE_SIZE;
    const size_t block_end = min(block_start + TILE_SIZE, num_elements);
    const size_t block_size = block_end - block_start;
    
    if (block_start >= num_elements) return;
    
    // Phase 1: Write local data to all peer buffers (scatter)
    // In Hopper, use cp.async for non-blocking stores
    for (int peer = 0; peer < world_size; peer++) {
        if (peer == rank) continue;
        
        T* peer_buf = peer_buffers[peer];
        // Offset into peer's buffer based on rank
        T* write_ptr = peer_buf + rank * num_elements + block_start;
        
        // Coalesced async copy to peer buffer
        #pragma unroll 4
        for (size_t i = tid; i < block_size; i += blockDim.x) {
            // Use non-coherent store for write-and-forget pattern
            asm volatile("st.global.L1::no_allocate [%0], %1;" 
                        : : "l"(write_ptr + i), "r"(local_buffer[block_start + i]));
        }
    }
    
    // Signal completion of write phase
    if (tid == 0) {
        atomicAdd_system((int*)&signals[rank]->flag[step % 4], 1);
    }
    
    block.sync();
    
    // Phase 2: Wait for all peers to signal, then reduce
    // Spin-wait with exponential backoff for peer signals
    for (int peer = 0; peer < world_size; peer++) {
        if (peer == rank) continue;
        
        // Wait for peer to write its tile to our buffer
        volatile int64_t* peer_flag = &signals[peer]->flag[step % 4];
        int spins = 0;
        while (*peer_flag < step + 1) {
            // Exponential backoff
            if (spins < 100) {
                spins++;
            } else {
                __nanosleep(100);  // 100ns sleep
            }
        }
    }
    
    // Load and reduce data from all peers
    // Initialize accumulator with local data
    #pragma unroll 8
    for (size_t i = tid; i < block_size; i += blockDim.x) {
        T sum = local_buffer[block_start + i];
        
        // Add contributions from all peers
        for (int peer = 0; peer < world_size; peer++) {
            if (peer == rank) continue;
            T* peer_src = peer_buffers[rank] + peer * num_elements + block_start + i;
            
            // Hopper: use L1 cache for repeated reduction reads
            T val;
            asm volatile("ld.global.L1::evict_first %0, [%1];" 
                        : "=r"(val) : "l"(peer_src));
            sum = __hadd(sum, val);
        }
        
        // Store result
        if (block_start + i < num_elements) {
            output[block_start + i] = sum;
        }
    }
}

                                 size_t num_elements) {
