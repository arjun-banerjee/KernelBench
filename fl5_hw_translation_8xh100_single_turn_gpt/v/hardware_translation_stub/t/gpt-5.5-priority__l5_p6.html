// Vectorized multiply-by-small-int for contiguous tensors.  The benchmark input is
// fp16 [512,4096], contiguous.  One CTA processes 256 half2 values (1024 bytes)
// per iteration; grid-stride keeps launch overhead low and saturates H100 DRAM.
__global__ __launch_bounds__(256, 4)
void scale_half_kernel(const half* __restrict__ x, half* __restrict__ y,
                       int64_t n, float scale) {
    int64_t n2 = n >> 1;
    const half2* __restrict__ x2 = reinterpret_cast<const half2*>(x);
    half2* __restrict__ y2 = reinterpret_cast<half2*>(y);
    half2 s2 = __float2half2_rn(scale);

    int64_t idx = (int64_t)blockIdx.x * blockDim.x + threadIdx.x;
    int64_t stride = (int64_t)blockDim.x * gridDim.x;

    // Unroll 4x to reduce loop overhead and improve memory-level parallelism.
    for (int64_t i = idx * 4; i < n2; i += stride * 4) {
        if (i + 0 < n2) y2[i + 0] = __hmul2(x2[i + 0], s2);
        if (i + 1 < n2) y2[i + 1] = __hmul2(x2[i + 1], s2);
        if (i + 2 < n2) y2[i + 2] = __hmul2(x2[i + 2], s2);
        if (i + 3 < n2) y2[i + 3] = __hmul2(x2[i + 3], s2);
    }
    if ((n & 1) && idx == 0) {
        y[n - 1] = __float2half(__half2float(x[n - 1]) * scale);
    }
}
