from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>

template <bool IS_NEOX>
__global__ void rotary_embedding_kernel(
    const half* __restrict__ query,
    const half* __restrict__ key,
    const half* __restrict__ cos,
    const half* __restrict__ sin,
    half* __restrict__ query_out,
    half* __restrict__ key_out,
    const int num_tokens,
    const int num_heads,
    const int num_kv_heads,
    const int head_size,
    const int rot_dim) {

    int token_idx = blockIdx.x;
    int head_idx = blockIdx.y;
    int dim_idx = threadIdx.x;

    if (token_idx >= num_tokens || head_idx >= num_heads || dim_idx >= rot_dim / 2) return;

    int embed_dim = rot_dim / 2;
    int x_idx, y_idx;

    if (IS_NEOX) {
        x_idx = dim_idx;
        y_idx = embed_dim + dim_idx;
    } else {
        x_idx = 2 * dim_idx;
        y_idx = 2 * dim_idx + 1;
    }

    float cos_f = __half2float(cos[token_idx * rot_dim + dim_idx]);
    float sin_f = __half2float(sin[token_idx * rot_dim + dim_idx]);

    // Query
    if (head_idx < num_heads) {
        int q_base = token_idx * num_heads * head_size + head_idx * head_size + (IS_NEOX ? 0 : 0);
        float x_f = __half2float(query[token_idx * num_heads * head_size + head_idx * head_size + x_idx]);
        float y_f = __half2float(query[token_idx * num_heads * head_size + head_idx * head_size + y_idx]);
        query_out[token_idx * num_heads * head_size + head_idx * head_size + x_idx] = __float2half(x_f * cos_f - y_f * sin_f);
        query_out[token_idx * num_heads * head_size + head_idx * head_size + y_idx] = __float2half(y_f * cos_f + x_f * sin_f);
    }

    // Key
    if (head_idx < num_kv_heads) {
        float x_f = __half2float(key[token_idx * num_kv_heads * head_size + head_idx * head_size + x_idx]);
        float y_f = __half2float(key[token_idx * num_kv_heads * head_size + head_idx * head_size + y_idx]);
        key_out[token_idx * num_kv_heads * head_size + head_idx * head_size + x_idx] = __float2half(x_f * cos_f - y_f * sin_f);
        key_out[token_idx * num_kv_heads * head_size + head_idx * head_size + y_idx] = __float2half(y_f * cos_f + x_f * sin_f);
    }
}

std::vector<torch::Tensor> rotary_embedding_cuda(
    torch::Tensor query,
    torch::Tensor key,
    torch::Tensor cos,
    torch::Tensor sin,
    int rot_dim) {

    int num_tokens = query.size(0);
    int num_heads = query.size(1);
    int head_size = query.size(2);
    int num_kv_heads = key.size(1);

    auto query_out = torch::empty_like(query);
    auto key_out = torch::empty_like(key);

    dim3 grid(num_tokens, num_heads);
    dim3 block(rot_dim / 2);

    rotary_embedding_kernel<false><<<grid, block>>>(
        (const half*)query.data_ptr<at::Half>(),
        (const half*)key.data_ptr<at::Half>(),
        (const half*)cos.data_ptr<at::Half>(),
        (const half*)sin.data_ptr<at::Half>(),
        (half*)query_out.data_ptr<at::Half>(),
        (half*)key_out.data_ptr<at::Half>(),
        num_tokens, num_heads, num_kv_heads, head_size, rot_dim);

    return {query_out, key_out};
}
