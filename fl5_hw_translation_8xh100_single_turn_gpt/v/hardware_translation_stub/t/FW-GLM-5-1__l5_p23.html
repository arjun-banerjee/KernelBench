        // Load A tile (row major assumed for A, but contract says column major)
        // A is M x K, stored row major: A[m, k] = A[m*K + k]
        if (global_m < M) {
            for (int k = tid; k < 32; k += 128) {
                if (k_start + k < K) {
                    sA[local_n * 32 + k] = A[global_m * K + k_start + k];
                }
            }

        // B[k, n] where n = indices[block_n + local_n]
        if (valid_n >= 0 && valid_n < N) {
            for (int k = tid; k < 32; k += 128) {
                if (k_start + k < K) {
                    sB[local_m * 32 + k] = B[valid_n * K + k_start + k];
                }
            }

        __syncthreads();
        
        // Compute
        if (global_m < M && valid_n >= 0) {
            for (int k = 0; k < k_end - k_start; k++) {
                accum += __half2float(sA[local_m * 32 + k]) * 
                         __half2float(sB[local_n * 32 + k]);
            }
        }
        
        __syncthreads();
    }
    
    // Write output with scatter
    if (global_m < M && valid_n >= 0 && valid_n < N) {
        float c_val = (beta != 0.0f) ? C[valid_n * M + global_m] : 0.0f;
        D[valid_n * M + global_m] = alpha * accum + beta * c_val;
    }
}

// Vectorized load/store kernel for better memory throughput
__global__ void __launch_bounds__(256) vectorized_gemm_gs_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    const float* __restrict__ C,
    float* __restrict__ D,
    const int* __restrict__ indices,
    int M, int N, int K, int index_size,
    float alpha, float beta
) {
    // Each block handles one output column (from index)
    // All threads in block compute different rows
    
    const int idx_pos = blockIdx.x;
    if (idx_pos >= index_size) return;
    
    const int col_idx = indices[idx_pos];
    const int tid = threadIdx.x;
    const int lane_id = tid & 31;
    const int warp_id = tid >> 5;
    
    // Each thread handles multiple rows
    const int rows_per_thread = (M + 255) / 256;
    const int start_row = tid * rows_per_thread;
    const int end_row = min(start_row + rows_per_thread, M);
    
    // Process K in chunks for better cache utilization
    const int K_CHUNK = 64;
    
    for (int row = start_row; row < end_row; row++) {
        float sum = 0.0f;
        
        // Vectorized loads
        for (int k = lane_id; k < K; k += 32) {
            // A is row major M x K: A[row, k]
            const half a_val = A[row * K + k];
            
            // B is column major K x N but we're gathering column col_idx
            // B[k, col_idx] = B[col_idx * K + k] for row-major B

            D[col_idx * M + row] = alpha * sum + beta * c_val;
        }
    }
}

        local_N = N // self.world_size

        self.register_buffer('indices', indices.to(torch.int32))

        start_col = sum(N // self.world_size + (1 if r < remainder else 0) for r in range(self.rank))
        end_col = start_col + local_N
        self.local_col_range = (start_col, end_col)

        K = A.size(1)
