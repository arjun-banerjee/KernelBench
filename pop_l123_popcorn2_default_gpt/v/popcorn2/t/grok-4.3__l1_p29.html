import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cmath>

__global__ void logsumexp_kernel(const float* __restrict__ input, float* __restrict__ output, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx < total) {
        int b = idx / D;
        int d = idx % D;
        float max_val = -INFINITY;
        for(int r = 0; r < R; r++) {
            float val = input[r * B * D + b * D + d];
            max_val = fmaxf(max_val, val);
        }
        float sum = 0.0f;
        for(int r = 0; r < R; r++) {
            float val = input[r * B * D + b * D + d];
            sum += expf(val - max_val);
        }
        output[b * D + d] = max_val + logf(sum);
    }
}

torch::Tensor logsumexp_cuda(torch::Tensor x) {
    TORCH_CHECK(x.dim() == 3, "Input must be 3D");
    int R = x.size(0);
    int B = x.size(1);
    int D = x.size(2);
    auto output = torch::empty({B, D}, x.options());
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    logsumexp_kernel<<<blocks, threads>>>(x.data_ptr<float>(), output.data_ptr<float>(), R, B, D);
    return output;
}
