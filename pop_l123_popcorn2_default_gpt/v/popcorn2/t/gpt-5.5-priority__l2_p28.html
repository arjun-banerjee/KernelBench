import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Fused GroupNorm + SiLU (Swish), a common post-conv / residual-branch pattern in
    ResNet-family backbones (SiLU instead of ReLU) and in spatial blocks of diffusion
    transformers (DiT) where features are (N, C, H, W).

    Reference pipeline for a single fused kernel: normalize channels with
    `GroupNorm`, then apply `SiLU` elementwise (`x * sigmoid(x)`).
    """

    def __init__(self, num_channels: int, num_groups: int, eps: float=1e-05, affine: bool=True):
        super().__init__()
        assert num_channels % num_groups == 0
        self.norm = nn.GroupNorm(num_groups=num_groups, num_channels=num_channels, eps=eps, affine=affine)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C, H, W) feature map
        Returns:
            Same shape as x
        """
        return F.silu(self.norm(x))

    def forward(self, x: torch.Tensor) -> torch.Tensor:

    sum = warp_reduce_sum(sum);
    sumsq = warp_reduce_sum(sumsq);

        sh_sum[wid] = sum;
        sh_sumsq[wid] = sumsq;
    }
    __syncthreads();
    int nwarps = blockDim.x >> 5;
    sum = (threadIdx.x < nwarps) ? sh_sum[lane] : 0.0f;
    sumsq = (threadIdx.x < nwarps) ? sh_sumsq[lane] : 0.0f;
    if (wid == 0) {
        sum = warp_reduce_sum(sum);
        sumsq = warp_reduce_sum(sumsq);
    }
    if (threadIdx.x == 0) {
        float invM = 1.0f / (float)M;
        float mean = sum * invM;
        float var = sumsq * invM - mean * mean;
        var = var < 0.0f ? 0.0f : var;
        s_mean = mean;
        s_rstd = rsqrtf(var + eps);
    }
    __syncthreads();
    float mean = s_mean;
    float rstd = s_rstd;

    for (int lc = 0; lc < cg; ++lc) {
        float wv = 1.0f;
        float bv = 0.0f;
        if constexpr (AFFINE) {
            int c = g * cg + lc;
            wv = weight[c];
            bv = bias[c];

        int off = base + lc * HW;

            float y = (x[off + hw] - mean) * rstd;
            if constexpr (AFFINE) y = fmaf(y, wv, bv);
            float sig = 1.0f / (1.0f + expf(-y));
            out[off + hw] = y * sig;
        }
    }
}
