    """
    Fused GroupNorm + SiLU (Swish), a common post-conv / residual-branch pattern in
    ResNet-family backbones (SiLU instead of ReLU) and in spatial blocks of diffusion
    transformers (DiT) where features are (N, C, H, W).

    Reference pipeline for a single fused kernel: normalize channels with
    `GroupNorm`, then apply `SiLU` elementwise (`x * sigmoid(x)`).
    """

    def __init__(self, num_channels: int, num_groups: int, eps: float=1e-05, affine: bool=True):
        super().__init__()
        assert num_channels % num_groups == 0
        self.norm = nn.GroupNorm(num_groups=num_groups, num_channels=num_channels, eps=eps, affine=affine)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C, H, W) feature map
        Returns:
            Same shape as x
        """
        return F.silu(self.norm(x))

// Efficient fused GroupNorm + SiLU kernel
// Grid: (N, num_groups) - each block handles one group in one sample
template <typename scalar_t>
__global__ void fused_groupnorm_silu_kernel(
    const scalar_t* __restrict__ input,
    scalar_t* __restrict__ output,
    const scalar_t* __restrict__ weight,
    const scalar_t* __restrict__ bias,
    const int N, const int C, const int H, const int W,
    const int num_groups, const int channels_per_group,
    const float eps) {
    
    const int n = blockIdx.x;
    const int g = blockIdx.y;
    
    const int group_start_c = g * channels_per_group;
    const int spatial_size = H * W;
    const int group_size = channels_per_group * spatial_size;
    
    const int tid = threadIdx.x;
    const int num_threads = blockDim.x;
    
    // Strided loop over all elements in the group
    float local_sum = 0.0f;
    float local_sum_sq = 0.0f;
    
    for (int idx = tid; idx < group_size; idx += num_threads) {
        const int c_in_group = idx / spatial_size;
        const int spatial_idx = idx % spatial_size;
        const int c = group_start_c + c_in_group;
        const int h = spatial_idx / W;
        const int w = spatial_idx % W;
        
        const int input_idx = ((n * C + c) * H + h) * W + w;
        const float val = static_cast<float>(input[input_idx]);
        local_sum += val;
        local_sum_sq += val * val;
    }
    
    // Block-level reduction using shared memory
    __shared__ float s_sum[256];  // Max threads
    __shared__ float s_sum_sq[256];
    
    s_sum[tid] = local_sum;
    s_sum_sq[tid] = local_sum_sq;
    __syncthreads();
    
    // Reduce in shared memory
    for (int stride = num_threads >> 1; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_sum[tid] += s_sum[tid + stride];
            s_sum_sq[tid] += s_sum_sq[tid + stride];
        }
        __syncthreads();
    }
    
    // Compute mean and inv_std
    __shared__ float mean, inv_std;
    if (tid == 0) {
        mean = s_sum[0] / group_size;
        float variance = s_sum_sq[0] / group_size - mean * mean;
        inv_std = 1.0f / sqrtf(variance + eps);
    }
    __syncthreads();
    
    // Apply normalization, affine, and SiLU
    for (int idx = tid; idx < group_size; idx += num_threads) {
        const int c_in_group = idx / spatial_size;
        const int spatial_idx = idx % spatial_size;
        const int c = group_start_c + c_in_group;
        const int h = spatial_idx / W;
        const int w = spatial_idx % W;
        
        const int input_idx = ((n * C + c) * H + h) * W + w;
        
        float val = static_cast<float>(input[input_idx]);
        
        // Normalize
        val = (val - mean) * inv_std;
        
        // Affine
        if (weight != nullptr) {
            val = val * static_cast<float>(weight[c]) + static_cast<float>(bias[c]);
        }
        
        // SiLU: x * sigmoid(x) = x / (1 + exp(-x))
        // Fast approximate sigmoid using __expf
        val = val / (1.0f + __expf(-val));
        
        output[input_idx] = static_cast<scalar_t>(val);
    }
}
