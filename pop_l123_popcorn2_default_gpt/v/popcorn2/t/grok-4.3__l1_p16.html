import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Weighted expert-output combine back into token-major order.

    Expert outputs are scattered back to token positions and scaled by the
    routing weight associated with each expert contribution.
    """

    def __init__(self):
        super().__init__()

    def forward(self, expert_hidden: torch.Tensor, token_idx: torch.Tensor, gates: torch.Tensor, num_tokens: int) -> torch.Tensor:
        out = torch.zeros(num_tokens, expert_hidden.shape[1], dtype=expert_hidden.dtype, device=expert_hidden.device)
        for row in range(expert_hidden.shape[0]):
            token = int(token_idx[row].item())
            out[token] += gates[row] * expert_hidden[row]
        return out

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void moe_combine_kernel(float* __restrict__ out, const float* __restrict__ expert_hidden, const int64_t* __restrict__ token_idx, const float* __restrict__ gates, int num_rows, int hidden_dim) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = num_rows * hidden_dim;
    if (idx >= total) return;
    int row = idx / hidden_dim;
    int col = idx % hidden_dim;
    float val = gates[row] * expert_hidden[row * hidden_dim + col];
    int64_t token = token_idx[row];
    atomicAdd(&out[token * hidden_dim + col], val);
}

torch::Tensor moe_combine_cuda(torch::Tensor expert_hidden, torch::Tensor token_idx, torch::Tensor gates, int num_tokens) {
    auto num_rows = expert_hidden.size(0);
    auto hidden_dim = expert_hidden.size(1);
    auto options = torch::TensorOptions().dtype(expert_hidden.dtype()).device(expert_hidden.device());
    auto out = torch::zeros({num_tokens, hidden_dim}, options);
    int total = num_rows * hidden_dim;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    moe_combine_kernel<<<blocks, threads>>>(
        out.data_ptr<float>(),
        expert_hidden.data_ptr<float>(),
        token_idx.data_ptr<int64_t>(),
        gates.data_ptr<float>(),
        num_rows,
        hidden_dim
    );
    return out;
}
