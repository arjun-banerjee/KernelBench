# Load the extension
fused_ext = load_inline(
    name='fused_ln_gelu_conv_v4',
    cpp_sources=cpp_source,
    cuda_sources=cuda_source,
    functions=['fused_forward'],
    extra_cuda_cflags=['-O3'],
    verbose=False
)

class ModelNew(nn.Module):
    def __init__(self, channels, seq_len, kernel_size, dilation):
        super(ModelNew, self).__init__()
        self.channels = channels
        self.seq_len = seq_len
        self.kernel_size = kernel_size
        self.dilation = dilation
        
        self.norm = nn.LayerNorm([channels, seq_len])
        self.conv = nn.Conv1d(channels, channels, kernel_size, padding=dilation, dilation=dilation)
        self.eps = 1e-5
    
    def forward(self, x):
        x = x.contiguous()
        weight = self.conv.weight.contiguous()
        bias = self.conv.bias.contiguous() if self.conv.bias is not None else torch.zeros(self.channels, device=x.device, dtype=x.dtype)
        gamma = self.norm.weight.contiguous()
        beta = self.norm.bias.contiguous() if self.norm.bias is not None else torch.zeros(self.channels, self.seq_len, device=x.device, dtype=x.dtype)
        
        output = fused_ext.fused_forward(x, weight, bias, gamma, beta, self.kernel_size, self.dilation, self.eps)
        return output
