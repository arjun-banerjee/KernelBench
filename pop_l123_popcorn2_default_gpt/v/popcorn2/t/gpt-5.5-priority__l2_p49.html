#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__device__ __forceinline__ float gelu_exact(float x) {
    return 0.5f * x * (1.0f + erff(x * 0.70710678118654752440f));
}

__global__ void gelu_sum4_kernel(const float* __restrict__ x, float* __restrict__ out, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < N) {
        const float* x0 = x;
        const float* x1 = x + N;
        const float* x2 = x + 2 * N;
        const float* x3 = x + 3 * N;
        float s = gelu_exact(x0[idx]);
        s += gelu_exact(x1[idx]);
        s += gelu_exact(x2[idx]);
        s += gelu_exact(x3[idx]);
        out[idx] = s;
    }
}

__global__ void gelu_sum_generic_kernel(const float* __restrict__ x, float* __restrict__ out,
                                        int R, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < N) {
        float s = 0.0f;
        for (int r = 0; r < R; ++r) {
            s += gelu_exact(x[(long long)r * N + idx]);
        }
        out[idx] = s;
    }
}

torch::Tensor gelu_sum_forward(torch::Tensor x) {
    const int R = (int)x.size(0);
    const int B = (int)x.size(1);
    const int D = (int)x.size(2);
    auto out = torch::empty({B, D}, x.options());

    }

}
