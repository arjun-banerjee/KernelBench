import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Self-normalized importance sampling estimate.
    Given samples from a proposal distribution q, computes importance
    weights w_i = p(x_i)/q(x_i), normalizes them, and returns both
    the weighted estimate of E_p[f(x)] and the effective sample size.

    Target p is a mixture of Gaussians; proposal q is a single Gaussian.
    """

    def __init__(self, dim, num_components=3):
        super().__init__()
        self.dim = dim
        self.num_components = num_components
        self.register_buffer('mix_weights', torch.ones(num_components) / num_components)
        mus = torch.randn(num_components, dim) * 2.0
        self.register_buffer('mix_mus', mus)
        self.register_buffer('mix_stds', torch.ones(num_components, dim))
        self.register_buffer('q_mu', torch.zeros(dim))
        self.register_buffer('q_std', torch.ones(dim) * 3.0)

    def _log_p(self, x):
        """Log density of target mixture."""
        diff = x.unsqueeze(1) - self.mix_mus.unsqueeze(0)
        log_comp = -0.5 * (diff / self.mix_stds.unsqueeze(0)) ** 2 - self.mix_stds.log().unsqueeze(0)
        log_comp = log_comp.sum(-1)
        return torch.logsumexp(log_comp + self.mix_weights.log().unsqueeze(0), dim=-1)

    def _log_q(self, x):
        """Log density of proposal."""
        return (-0.5 * ((x - self.q_mu) / self.q_std) ** 2 - self.q_std.log()).sum(-1)

    def forward(self, samples: torch.Tensor) -> tuple:
        """
        Args:
            samples: (N, dim) – samples drawn from q
        Returns:
            (weighted_mean, ess):
                weighted_mean: (dim,) – importance-weighted mean estimate of E_p[x]
                ess: scalar – effective sample size
        """
        log_w = self._log_p(samples) - self._log_q(samples)
        log_w = log_w - torch.logsumexp(log_w, dim=0)
        w = torch.exp(log_w)
        weighted_mean = (w.unsqueeze(-1) * samples).sum(dim=0)
        ess = 1.0 / (w ** 2).sum()
        return (weighted_mean, ess)

__device__ float logsumexp_device(float* vals, int n) {
    float maxv = -1e30f;
    for (int i=0; i<n; i++) maxv = fmaxf(maxv, vals[i]);
    float sum = 0.0f;
    for (int i=0; i<n; i++) sum += expf(vals[i] - maxv);
    return maxv + logf(sum);
}

__global__ void compute_logw_kernel(
    const float* __restrict__ samples,
    const float* __restrict__ mix_mus,
    const float* __restrict__ mix_stds,
    const float* __restrict__ mix_weights_log,
    const float* __restrict__ q_mu,
    const float* __restrict__ q_std,
    float* __restrict__ log_w,
    int N, int D, int K) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= N) return;
    const float* x = samples + i * D;
    // log_p
    float log_comps[8]; // assume K<=8
    for (int c=0; c<K; c++) {
        float lc = 0.0f;
        for (int d=0; d<D; d++) {
            float diff = x[d] - mix_mus[c*D + d];
            float s = mix_stds[c*D + d];
            lc += -0.5f * diff*diff / (s*s) - logf(s);
        }
        log_comps[c] = lc + mix_weights_log[c];
    }
    float log_p = logsumexp_device(log_comps, K);
    // log_q
    float lq = 0.0f;
    for (int d=0; d<D; d++) {
        float diff = x[d] - q_mu[d];
        float s = q_std[d];
        lq += -0.5f * diff*diff / (s*s) - logf(s);
    }
    log_w[i] = log_p - lq;
}

__global__ void normalize_and_compute_kernel(
    const float* __restrict__ samples,
    const float* __restrict__ log_w_in,
    float* __restrict__ weighted_mean,
    float* __restrict__ ess_out,
    int N, int D) {
    extern __shared__ float smem[];
    // first find logsumexp of log_w
    float local_max = -1e30f;
    for (int i=threadIdx.x; i<N; i+=blockDim.x) {
        local_max = fmaxf(local_max, log_w_in[i]);
    }
    // reduce max in block
    smem[threadIdx.x] = local_max;
    __syncthreads();
    for (int s=blockDim.x/2; s>0; s>>=1) {
        if (threadIdx.x < s) smem[threadIdx.x] = fmaxf(smem[threadIdx.x], smem[threadIdx.x+s]);
        __syncthreads();
    }
    float max_log = smem[0];
    __syncthreads();
    float sum_exp = 0.0f;
    for (int i=threadIdx.x; i<N; i+=blockDim.x) {
        sum_exp += expf(log_w_in[i] - max_log);
    }
    smem[threadIdx.x] = sum_exp;
    __syncthreads();
    for (int s=blockDim.x/2; s>0; s>>=1) {
        if (threadIdx.x < s) smem[threadIdx.x] += smem[threadIdx.x+s];
        __syncthreads();
    }
    float log_sum = max_log + logf(smem[0]);
    // now normalize w and compute weighted
    for (int d=0; d<D; d++) {
        float wsum = 0.0f;
        for (int i=threadIdx.x; i<N; i+=blockDim.x) {
            float w = expf(log_w_in[i] - log_sum);
            wsum += w * samples[i*D + d];
        }
        // block reduce wsum
        smem[threadIdx.x] = wsum;
        __syncthreads();
        for (int s=blockDim.x/2; s>0; s>>=1) {
            if (threadIdx.x < s) smem[threadIdx.x] += smem[threadIdx.x+s];
            __syncthreads();
        }
        if (threadIdx.x == 0) weighted_mean[d] = smem[0];
        __syncthreads();
    }
    // ess
    float sum_w2 = 0.0f;
    for (int i=threadIdx.x; i<N; i+=blockDim.x) {
        float w = expf(log_w_in[i] - log_sum);
        sum_w2 += w * w;
    }

        if (threadIdx.x < s) smem[threadIdx.x] += smem[threadIdx.x+s];
        __syncthreads();
    }
    if (threadIdx.x == 0) *ess_out = 1.0f / smem[0];
}

std::vector<torch::Tensor> importance_sampling_cuda(
    torch::Tensor samples,
    torch::Tensor mix_mus,
    torch::Tensor mix_stds,
    torch::Tensor mix_weights_log,
    torch::Tensor q_mu,
    torch::Tensor q_std) {
    int N = samples.size(0);
    int D = samples.size(1);
    int K = mix_mus.size(0);
    auto log_w = torch::empty({N}, samples.options());
    auto weighted_mean = torch::empty({D}, samples.options());
    auto ess = torch::empty({1}, samples.options());
    
    dim3 block(256);
    dim3 grid((N + 255)/256);
    compute_logw_kernel<<<grid, block>>>(
        samples.data_ptr<float>(),
        mix_mus.data_ptr<float>(),
        mix_stds.data_ptr<float>(),
        mix_weights_log.data_ptr<float>(),
        q_mu.data_ptr<float>(),
        q_std.data_ptr<float>(),

        log_w.data_ptr<float>(),
        weighted_mean.data_ptr<float>(),
        ess.data_ptr<float>(),
        N, D);
    
    return {weighted_mean, ess.squeeze()};
}
