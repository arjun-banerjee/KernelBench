import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Voxel-grid pooling for 3-D molecular point clouds.  Discretizes
    continuous 3-D coordinates into a regular grid and pools per-atom
    features into each voxel.  Used in 3-D-CNN approaches for molecular
    property prediction and ligand docking (e.g., PointVoxel-CNN,
    3D-Infomax).
    """

    def __init__(self, feat_dim, grid_size=32, voxel_size=1.0):
        super().__init__()
        self.feat_dim = feat_dim
        self.grid_size = grid_size
        self.voxel_size = voxel_size
        self.linear = nn.Linear(feat_dim, feat_dim)

    def forward(self, coords: torch.Tensor, features: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords:   (B, N, 3) – atom coordinates
            features: (B, N, feat_dim) – per-atom features
        Returns:
            voxel_grid: (B, feat_dim, G, G, G) – pooled voxel features
        """
        (B, N, _) = coords.shape
        G = self.grid_size
        features = self.linear(features)
        center = coords.mean(dim=1, keepdim=True)
        shifted = (coords - center) / self.voxel_size + G / 2.0
        indices = shifted.long().clamp(0, G - 1)
        voxel_grid = coords.new_zeros(B, self.feat_dim, G, G, G)
        ix = indices[:, :, 0]
        iy = indices[:, :, 1]
        iz = indices[:, :, 2]
        batch_idx = torch.arange(B, device=coords.device).unsqueeze(1).expand(-1, N)
        for d in range(self.feat_dim):
            feat_d = features[:, :, d]
            voxel_grid[:, d].index_put_((batch_idx.reshape(-1), ix.reshape(-1), iy.reshape(-1), iz.reshape(-1)), feat_d.reshape(-1), accumulate=True)
        return voxel_grid
feat_dim = 32
grid_size = 32
voxel_size = 1.0
num_atoms = 256
batch_size = 8

__global__ __launch_bounds__(256) void scatter32_kernel(const float* __restrict__ coords,
                                      const float* __restrict__ features,
                                      const float* __restrict__ weight,
                                      const float* __restrict__ bias,
                                      const float* __restrict__ means,
                                      float* __restrict__ out,
                                      int B, int N, float inv_voxel) {
    int lane = threadIdx.x & 31;
    int warp_in_block = threadIdx.x >> 5;
    int warps_per_block = 8;
    int warp_global = blockIdx.x * warps_per_block + warp_in_block;
    int warp_stride = gridDim.x * warps_per_block;
    int total_atoms = B * N;
    for (int atom = warp_global; atom < total_atoms; atom += warp_stride) {
        int b = atom / N;
        int n = atom - b * N;
        const float* cptr = coords + ((long long)b * N + n) * 3;
        const float* mptr = means + b * 3;
        float x = (cptr[0] - mptr[0]) * inv_voxel + 16.0f;
        float y = (cptr[1] - mptr[1]) * inv_voxel + 16.0f;
        float z = (cptr[2] - mptr[2]) * inv_voxel + 16.0f;
        int ix = (int)x, iy = (int)y, iz = (int)z;
        ix = ix < 0 ? 0 : (ix > 31 ? 31 : ix);
        iy = iy < 0 ? 0 : (iy > 31 ? 31 : iy);
        iz = iz < 0 ? 0 : (iz > 31 ? 31 : iz);
        const float* f = features + (long long)atom * 32;
        const float* w = weight + lane * 32;
        float acc = bias[lane];
        #pragma unroll
        for (int k=0; k<32; ++k) acc = fmaf(f[k], w[k], acc);
        long long off = (((long long)b * 32 + lane) << 15) + (ix << 10) + (iy << 5) + iz;
        atomicAdd(out + off, acc);
    }
}

__global__ void scatter_generic_kernel(const float* __restrict__ coords, const float* __restrict__ features, const float* __restrict__ weight, const float* __restrict__ bias, const float* __restrict__ means, float* __restrict__ out, int B, int N, int D, int G, float inv_voxel) {
    long long total = (long long)B*N*D;
    for (long long tid=(long long)blockIdx.x*blockDim.x+threadIdx.x; tid<total; tid += (long long)blockDim.x*gridDim.x) {
        int d = tid % D; long long atom = tid / D; int n = atom % N; int b = atom / N;
        const float* c = coords + ((long long)b*N+n)*3; const float* m=means+b*3;
        float halfG=(float)G*0.5f;
        int ix=(int)((c[0]-m[0])*inv_voxel+halfG), iy=(int)((c[1]-m[1])*inv_voxel+halfG), iz=(int)((c[2]-m[2])*inv_voxel+halfG);
        ix=ix<0?0:(ix>=G?G-1:ix); iy=iy<0?0:(iy>=G?G-1:iy); iz=iz<0?0:(iz>=G?G-1:iz);
        const float* f=features+atom*D; const float* w=weight+(long long)d*D; float acc=bias[d];
        for(int k=0;k<D;k++) acc=fmaf(f[k],w[k],acc);
        long long off=(((((long long)b*D+d)*G+ix)*G+iy)*G+iz); atomicAdd(out+off,acc);
    }
}
