"""
Reference: shard the last dimension per rank, then all_gather to reconstruct (tensor-parallel style).

Multi-GPU: each rank keeps one column block; output is concat along last dim (matches full input).
Single-GPU: returns x unchanged.
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_rank, get_world_size, is_distributed_run, maybe_init_process_group
dim = 80

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunks = x.chunk(ws, dim=-1)
        local = chunks[rank].contiguous() * self.scale
        gathered = [torch.empty_like(local) for _ in range(ws)]
        dist.all_gather(gathered, local)
        return torch.cat(gathered, dim=-1)

    const int64_t idx = (blockIdx.x * blockDim.x + threadIdx.x) * 4;
    
    if (idx + 3 < size) {
        // Use vectorized load/store
        float4 in_vec = *reinterpret_cast<const float4*>(input + idx);
        float4 out_vec;
        out_vec.x = in_vec.x * scale;
        out_vec.y = in_vec.y * scale;
        out_vec.z = in_vec.z * scale;
        out_vec.w = in_vec.w * scale;
        *reinterpret_cast<float4*>(output + idx) = out_vec;
    } else {
        // Handle remaining elements
        for (int i = 0; i < 4 && idx + i < size; i++) {
            output[idx + i] = input[idx + i] * scale;
        }
    }
}

torch::Tensor scale_cuda(torch::Tensor input, float scale) {
    auto output = torch::empty_like(input);
    const int64_t size = input.numel();
    
    const int threads = 256;
    const int64_t blocks = (size + threads * 4 - 1) / (threads * 4);
    
    AT_DISPATCH_FLOATING_TYPES(input.scalar_type(), "scale_kernel", ([&] {
        scale_kernel_vec4<scalar_t><<<blocks, threads>>>(
            input.data_ptr<scalar_t>(),
            output.data_ptr<scalar_t>(),
            static_cast<scalar_t>(scale),
            size
        );
    }));
    
    return output;
}

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        
        if not is_distributed_run() or ws == 1:
            # Non-distributed: just scale the entire tensor
            return scale_kernel.scale_cuda(x, self.scale.item())
        
        # Distributed case: use PyTorch's all_gather
        import torch.distributed as dist
        chunks = x.chunk(ws, dim=-1)
        local = chunks[rank].contiguous() * self.scale
        gathered = [torch.empty_like(local) for _ in range(ws)]
        dist.all_gather(gathered, local)
        return torch.cat(gathered, dim=-1)
