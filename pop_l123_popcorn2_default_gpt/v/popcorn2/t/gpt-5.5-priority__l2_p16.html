import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Hamiltonian Monte Carlo (HMC) proposal using leapfrog integration.
    Simulates Hamiltonian dynamics on a batched set of particles to
    produce proposals with high acceptance probability.  The target
    distribution is a standard Gaussian for benchmarking.
    """

    def __init__(self, dim, num_leapfrog_steps=10, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.num_leapfrog_steps = num_leapfrog_steps
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))

    def _grad_log_prob(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -(x - self.mu)

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (x ** 2).sum(dim=-1)

    def forward(self, q: torch.Tensor) -> tuple:
        """
        Args:
            q: (B, dim) – current positions
        Returns:
            (q_new, accepted):
                q_new:    (B, dim)
                accepted: (B,)
        """
        eps = self.step_size
        p = torch.randn_like(q)
        current_p = p.clone()
        current_q = q.clone()
        p = p + 0.5 * eps * self._grad_log_prob(q)
        for _ in range(self.num_leapfrog_steps - 1):
            q = q + eps * p
            p = p + eps * self._grad_log_prob(q)
        q = q + eps * p
        p = p + 0.5 * eps * self._grad_log_prob(q)
        p = -p
        current_H = -self._log_prob(current_q) + 0.5 * (current_p ** 2).sum(-1)
        proposed_H = -self._log_prob(q) + 0.5 * (p ** 2).sum(-1)
        log_alpha = current_H - proposed_H
        log_u = torch.log(torch.rand(q.shape[0], device=q.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        q_new = torch.where(accepted.unsqueeze(-1), q, current_q)
        return (q_new, accepted)

    int tid = threadIdx.x;
    int idx = b * D + tid;
    float sum = 0.0f;
    float qn = 0.0f;
    if (tid < D) {
        float q = q0[idx];
        float p = p0[idx];
        qn = acoef * q + bcoef * p;
        float pn = ccoef * q + dcoef * p;
        sum = 0.5f * (q * q + p * p - qn * qn - pn * pn);
    }
    float log_alpha = block_reduce_sum(sum);
    __shared__ int accept_s;
    if (tid == 0) {
        float uu = u[b];
        uu = uu < 1.0e-10f ? 1.0e-10f : uu;
        bool ac = (logf(uu) < log_alpha);
        accept_s = ac ? 1 : 0;
        acc_out[b] = ac;
    }
    __syncthreads();
    if (tid < D) out[idx] = accept_s ? qn : q0[idx];
}

static inline void coeffs(int steps, float eps, float &a, float &b, float &c, float &d) {
    a = 1.0f; b = 0.0f; c = -0.5f * eps; d = 1.0f;
    for (int i = 0; i < steps - 1; ++i) {
        a = a + eps * c;
        b = b + eps * d;
        c = c - eps * a;
        d = d - eps * b;
    }
    a = a + eps * c;
    b = b + eps * d;
    c = c - 0.5f * eps * a;
    d = d - 0.5f * eps * b;
}

std::vector<torch::Tensor> hmc_forward(torch::Tensor q, torch::Tensor p, torch::Tensor u,
                                       int64_t steps, double epsd) {
    auto qc = q.contiguous();
    auto pc = p.contiguous();
    auto uc = u.contiguous();
    int B = (int)qc.size(0);
    int D = (int)qc.size(1);
    float a, b, c, d;
    coeffs((int)steps, (float)epsd, a, b, c, d);
    auto out = torch::empty_like(qc);
    auto acc = torch::empty({B}, torch::TensorOptions().device(q.device()).dtype(torch::kBool));
    hmc_kernel<<<B, 128, 0, at::cuda::getCurrentCUDAStream()>>>(
        qc.data_ptr<float>(), pc.data_ptr<float>(), uc.data_ptr<float>(),
        out.data_ptr<float>(), acc.data_ptr<bool>(), D, a, b, c, d);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return {out, acc};
}
