import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Pack tokens into expert-major order for MoE execution.

    Each token is assigned to one expert and one slot within that expert's
    buffer. The kernel writes token activations into the packed dispatch layout.
    """

    def __init__(self):
        super().__init__()

    def forward(self, token_hidden: torch.Tensor, expert_idx: torch.Tensor, slot_idx: torch.Tensor, expert_offsets: torch.Tensor) -> torch.Tensor:
        num_rows = int(expert_offsets[-1].item())
        out = torch.zeros(num_rows, token_hidden.shape[1], dtype=token_hidden.dtype, device=token_hidden.device)
        for token in range(token_hidden.shape[0]):
            expert = int(expert_idx[token].item())
            row = int(expert_offsets[expert].item() + slot_idx[token].item())
            out[row] = token_hidden[token]
        return out

__global__ void moe_dispatch_permute_kernel(
    const float* __restrict__ token_hidden,
    const int64_t* __restrict__ expert_idx,
    const int64_t* __restrict__ slot_idx,
    const int64_t* __restrict__ expert_offsets,
    float* __restrict__ out,
    int64_t num_tokens,
    int64_t hidden_dim) {
    
    int64_t token = blockIdx.x * blockDim.x + threadIdx.x;
    if (token >= num_tokens) return;
    
    // Get expert and compute output row
    int64_t expert = expert_idx[token];
    int64_t row = expert_offsets[expert] + slot_idx[token];
    
    // Copy the hidden vector using vectorized loads/stores
    const float4* src_vec = reinterpret_cast<const float4*>(token_hidden + token * hidden_dim);
    float4* dst_vec = reinterpret_cast<float4*>(out + row * hidden_dim);
    
    int64_t vec_len = hidden_dim / 4;  // number of float4 elements
    
    for (int64_t i = 0; i < vec_len; i++) {
        dst_vec[i] = src_vec[i];

    // Handle remaining elements
    for (int64_t i = vec_len * 4; i < hidden_dim; i++) {
        out[row * hidden_dim + i] = token_hidden[token * hidden_dim + i];

}
