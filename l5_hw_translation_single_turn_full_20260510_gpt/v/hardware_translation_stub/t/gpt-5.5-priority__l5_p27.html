// 32x32 output tile, 32-deep K tile.  16x16 threads, each thread computes a 2x2 micro-tile.
// Designed for the fixed small strided batched SGEMM shape used by the model on Hopper.
template<int BM, int BN, int BK>
__global__ __launch_bounds__(256, 2)
void bmm_32x32_kernel(const float* __restrict__ A,
                      const float* __restrict__ B,
                      float* __restrict__ C,
                      int batch_count, int M, int K, int N,
                      float alpha, float beta) {
    __shared__ float As[BM][BK + 1];
    __shared__ float Bs[BK][BN + 1];

    const int tx = threadIdx.x;          // 0..15
    const int ty = threadIdx.y;          // 0..15
    const int tid = ty * 16 + tx;

    const int b = blockIdx.z;
    const int row_base = blockIdx.y * BM;
    const int col_base = blockIdx.x * BN;

    const int r0 = row_base + ty * 2;
    const int r1 = r0 + 1;
    const int c0 = col_base + tx * 2;
    const int c1 = c0 + 1;

    float acc00 = 0.0f, acc01 = 0.0f, acc10 = 0.0f, acc11 = 0.0f;

    const float* Ab = A + ((long long)b * M * K);
    const float* Bb = B + ((long long)b * K * N);

    #pragma unroll 1
    for (int kk = 0; kk < K; kk += BK) {
        // Cooperative, coalesced load of A tile and B tile.  Padding removes shared-bank conflicts.
        #pragma unroll
        for (int idx = tid; idx < BM * BK; idx += 256) {
            int rr = idx / BK;
            int cc = idx - rr * BK;
            int gr = row_base + rr;
            int gk = kk + cc;
            As[rr][cc] = (gr < M && gk < K) ? Ab[(long long)gr * K + gk] : 0.0f;
        }
        #pragma unroll
        for (int idx = tid; idx < BK * BN; idx += 256) {
            int rr = idx / BN;
            int cc = idx - rr * BN;
            int gk = kk + rr;
            int gc = col_base + cc;
            Bs[rr][cc] = (gk < K && gc < N) ? Bb[(long long)gk * N + gc] : 0.0f;
        }
        __syncthreads();

            float b0 = Bs[q][tx * 2];

            acc01 = fmaf(a0, b1, acc01);

}

torch::Tensor forward_cuda(torch::Tensor A, torch::Tensor B, double alpha_d, double beta_d) {
    CHECK_CUDA(A); CHECK_CUDA(B);
    CHECK_CONTIGUOUS(A); CHECK_CONTIGUOUS(B);
    TORCH_CHECK(A.scalar_type() == torch::kFloat32, "A must be float32");
    TORCH_CHECK(B.scalar_type() == torch::kFloat32, "B must be float32");
    TORCH_CHECK(A.dim() == 3 && B.dim() == 3, "expected rank-3 tensors");

    int batch_count = (int)A.size(0);
    int M = (int)A.size(1);
    int K = (int)A.size(2);
    int N = (int)B.size(2);
    TORCH_CHECK(B.size(0) == batch_count && B.size(1) == K, "incompatible B shape");

        batch_count, M, K, N, (float)alpha_d, (float)beta_d);
    return C;
}
