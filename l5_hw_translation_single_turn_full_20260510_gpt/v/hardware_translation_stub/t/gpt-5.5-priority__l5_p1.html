#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <math_constants.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

static __device__ __forceinline__ float warp_sum(float v) {
    v += __shfl_down_sync(0xffffffff, v, 16);
    v += __shfl_down_sync(0xffffffff, v, 8);
    v += __shfl_down_sync(0xffffffff, v, 4);
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

static __device__ __forceinline__ float warp_max(float v) {
    v = fmaxf(v, __shfl_down_sync(0xffffffff, v, 16));
    v = fmaxf(v, __shfl_down_sync(0xffffffff, v, 8));
    v = fmaxf(v, __shfl_down_sync(0xffffffff, v, 4));
    v = fmaxf(v, __shfl_down_sync(0xffffffff, v, 2));

// Fixed-shape grouped-query attention for B=4, Hq=32, Hkv=8, L=512, D=128.
// One CTA computes one output row (batch, query head, query position).  The
// implementation keeps the query vector resident in registers, streams K and V
// coalesced, and performs the softmax in FP32.  It intentionally avoids forming
// the repeated-KV tensor or the score matrix used by the PyTorch reference.
__global__ __launch_bounds__(128, 4)
void gqa_512x128_kernel(const half* __restrict__ q,
                        const half* __restrict__ k,
                        const half* __restrict__ v,
                        half* __restrict__ out) {
    constexpr int B = 4;
    constexpr int HQ = 32;
    constexpr int HKV = 8;
    constexpr int L = 512;
    constexpr int D = 128;
    constexpr int GROUP = HQ / HKV;
    constexpr float SCALE = 0.08838834764831845f; // 1/sqrt(128)

    __shared__ float logits[L];
    __shared__ float red[4];

    const int row = blockIdx.x;
    const int t = threadIdx.x; // also the head dimension for output
    const int qi = row & (L - 1);
    const int tmp = row >> 9;
    const int hq = tmp & (HQ - 1);
    const int b = tmp >> 5;
    const int hkv = hq / GROUP;

    const half* qptr = q + (((b * HQ + hq) * L + qi) * D);
    const half* kbase = k + ((b * HKV + hkv) * L * D);
    const half* vbase = v + ((b * HKV + hkv) * L * D);
    half* optr = out + (((b * HQ + hq) * L + qi) * D);

    const float qv = __half2float(qptr[t]);

    // QK^T: one coalesced 128-wide dot product per key position.
    #pragma unroll 1
    for (int sj = 0; sj < L; ++sj) {
        float prod = qv * __half2float(kbase[sj * D + t]);
        float dot = block_sum_128(prod, red);
        if (t == 0) logits[sj] = dot * SCALE;
        __syncthreads();
    }

    // Numerically stable softmax over the 512 keys.
    float m = -CUDART_INF_F;
    #pragma unroll
    for (int sj = t; sj < L; sj += 128) m = fmaxf(m, logits[sj]);
    m = block_max_128(m, red);

    float denom = 0.0f;
    #pragma unroll
    for (int sj = t; sj < L; sj += 128) {
        float e = __expf(logits[sj] - m);
        logits[sj] = e;
        denom += e;
    }
    denom = block_sum_128(denom, red);
    const float inv_denom = 1.0f / denom;
    __syncthreads();

    // P*V, one output channel per thread; V loads are coalesced for every key.
    float acc = 0.0f;
    #pragma unroll 1
    for (int sj = 0; sj < L; ++sj) {
        acc += (logits[sj] * inv_denom) * __half2float(vbase[sj * D + t]);
    }
    optr[t] = __float2half_rn(acc);
}

torch::Tensor gqa_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
    auto out = torch::empty_like(q);
    const int blocks = 4 * 32 * 512;
    const dim3 grid(blocks);
    const dim3 block(128);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    gqa_512x128_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()));
    return out;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gqa_forward", &gqa_forward, "Hopper optimized fixed-size GQA forward");
}
