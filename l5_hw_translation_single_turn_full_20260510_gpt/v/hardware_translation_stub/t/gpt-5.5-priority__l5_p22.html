#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <mma.h>

using namespace nvcuda;

constexpr int BM = 64;
constexpr int BN = 64;
constexpr int BK = 32;
constexpr int WARPS = 8;
constexpr int THREADS = WARPS * 32;

__global__ __launch_bounds__(THREADS, 2)
void h100_tf32_gemm_64x64x32_kernel(const float* __restrict__ A,
                                    const float* __restrict__ B,
                                    float* __restrict__ C,
                                    int M, int N, int K,
                                    float alpha) {
  __shared__ float As[BM * BK];
  __shared__ float Bs[BK * BN];

  const int tid = threadIdx.x;
  const int warp_id = tid >> 5;
  const int block_m = blockIdx.y * BM;
  const int block_n = blockIdx.x * BN;

  // Eight warps cover a 64x64 CTA tile.  Each warp computes a 16x32 tile
  // as two 16x16 WMMA accumulators.
  const int warp_m = warp_id >> 1;       // 0..3
  const int warp_n_pair = warp_id & 1;   // 0..1
  const int wm = warp_m * 16;
  const int wn = warp_n_pair * 32;

  wmma::fragment<wmma::accumulator, 16, 16, 8, float> acc0;
  wmma::fragment<wmma::accumulator, 16, 16, 8, float> acc1;
  wmma::fill_fragment(acc0, 0.0f);
  wmma::fill_fragment(acc1, 0.0f);

  for (int kb = 0; kb < K; kb += BK) {
    // Cooperative, coalesced global -> shared load.  Values are explicitly
    // rounded to TF32 before tensor-core consumption, matching TF32 GEMM math.
    #pragma unroll
    for (int idx = tid; idx < BM * BK; idx += THREADS) {
      int r = idx / BK;
      int c = idx - r * BK;
      int gr = block_m + r;
      int gc = kb + c;
      float v = 0.0f;
      if (gr < M && gc < K) v = A[gr * K + gc];
      As[idx] = __float_to_tf32(v);
    }

    #pragma unroll
    for (int idx = tid; idx < BK * BN; idx += THREADS) {
      int r = idx / BN;
      int c = idx - r * BN;
      int gr = kb + r;
      int gc = block_n + c;
      float v = 0.0f;
      if (gr < K && gc < N) v = B[gr * N + gc];
      Bs[idx] = __float_to_tf32(v);
    }

    __syncthreads();

    #pragma unroll
    for (int kk = 0; kk < BK; kk += 8) {
      wmma::fragment<wmma::matrix_a, 16, 16, 8, wmma::precision::tf32, wmma::row_major> a_frag;
      wmma::fragment<wmma::matrix_b, 16, 16, 8, wmma::precision::tf32, wmma::row_major> b_frag0;
      wmma::fragment<wmma::matrix_b, 16, 16, 8, wmma::precision::tf32, wmma::row_major> b_frag1;

      wmma::load_matrix_sync(a_frag,  As + wm * BK + kk, BK);
      wmma::load_matrix_sync(b_frag0, Bs + kk * BN + wn, BN);
      wmma::load_matrix_sync(b_frag1, Bs + kk * BN + wn + 16, BN);
      wmma::mma_sync(acc0, a_frag, b_frag0, acc0);
      wmma::mma_sync(acc1, a_frag, b_frag1, acc1);
    }

    __syncthreads();
  }

  if (alpha != 1.0f) {
    #pragma unroll
    for (int i = 0; i < acc0.num_elements; ++i) {
      acc0.x[i] *= alpha;
      acc1.x[i] *= alpha;
    }
  }

  const int out_m = block_m + wm;
  const int out_n0 = block_n + wn;
  const int out_n1 = out_n0 + 16;

  // The benchmark dimensions are multiples of 64.  Keep a guarded path for
  // completeness without penalizing the common full-tile case.
  if (out_m + 15 < M && out_n1 + 15 < N) {
    wmma::store_matrix_sync(C + out_m * N + out_n0, acc0, N, wmma::mem_row_major);

  } else {
    __shared__ float Stmp[WARPS * 2 * 16 * 16];
    float* s0 = Stmp + warp_id * 2 * 16 * 16;
    float* s1 = s0 + 16 * 16;
    wmma::store_matrix_sync(s0, acc0, 16, wmma::mem_row_major);
    wmma::store_matrix_sync(s1, acc1, 16, wmma::mem_row_major);

      if (out_m + r < M && out_n1 + c < N) C[(out_m + r) * N + out_n1 + c] = s1[i];
    }
  }
}

torch::Tensor gemm_forward(torch::Tensor a, torch::Tensor b, double alpha) {
  const int M = (int)a.size(0);
  const int K = (int)a.size(1);
  const int N = (int)b.size(1);
