import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor h100_gqa_causal_attention(torch::Tensor q, torch::Tensor k, torch::Tensor v);

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <cmath>

namespace {

constexpr int D_CONST = 128;
constexpr int ROWS_PER_CTA = 8;
constexpr int TILE_K = 64;
constexpr int THREADS = 256;

__device__ __forceinline__ float warp_sum(float x) {
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        x += __shfl_down_sync(0xffffffffu, x, off);
    }
    return x;
}

__global__ __launch_bounds__(THREADS, 2)
void gqa_causal_attn_kernel(const half* __restrict__ q,
                            const half* __restrict__ k,
                            const half* __restrict__ v,
                            half* __restrict__ out,
                            int B, int H, int KVH, int S) {
    extern __shared__ half smem[];
    half* __restrict__ sk = smem;
    half* __restrict__ sv = smem + TILE_K * D_CONST;

    const int q_block = blockIdx.x;
    const int h = blockIdx.y;
    const int b = blockIdx.z;
    const int warp = threadIdx.x >> 5;
    const int lane = threadIdx.x & 31;

    const int q_start = q_block * ROWS_PER_CTA;
    const int q_pos = q_start + warp;
    const int kv_group = H / KVH;
    const int kvh = h / kv_group;

    float qv0 = 0.f, qv1 = 0.f, qv2 = 0.f, qv3 = 0.f;
    if (q_pos < S) {
        const half* qptr = q + (((b * H + h) * S + q_pos) * D_CONST);
        qv0 = __half2float(qptr[lane]);
        qv1 = __half2float(qptr[lane + 32]);
        qv2 = __half2float(qptr[lane + 64]);
        qv3 = __half2float(qptr[lane + 96]);
    }

    float acc0 = 0.f, acc1 = 0.f, acc2 = 0.f, acc3 = 0.f;
    float m = -3.4028234663852886e38f;
    float l = 0.f;
    const float scale = 0.08838834764831845f; // 1 / sqrt(128)
    const int max_q_in_block = min(S - 1, q_start + ROWS_PER_CTA - 1);

    for (int ks = 0; ks <= max_q_in_block; ks += TILE_K) {
        const int valid = min(TILE_K, S - ks);
        const int elems = valid * D_CONST;
        const half* kbase = k + (((b * KVH + kvh) * S + ks) * D_CONST);
        const half* vbase = v + (((b * KVH + kvh) * S + ks) * D_CONST);

        for (int idx = threadIdx.x; idx < elems; idx += THREADS) {
            sk[idx] = kbase[idx];
            sv[idx] = vbase[idx];
        }
        __syncthreads();

        if (q_pos < S) {
            int upto = q_pos - ks + 1;
            if (upto > valid) upto = valid;
            if (upto > 0) {
                for (int j = 0; j < upto; ++j) {
                    const half* kptr = sk + j * D_CONST;
                    float dot = qv0 * __half2float(kptr[lane])
                              + qv1 * __half2float(kptr[lane + 32])
                              + qv2 * __half2float(kptr[lane + 64])
                              + qv3 * __half2float(kptr[lane + 96]);
                    dot = warp_sum(dot);
                    float logit = __shfl_sync(0xffffffffu, dot, 0) * scale;

                    float new_m = fmaxf(m, logit);
                    float alpha = __expf(m - new_m);
                    float beta  = __expf(logit - new_m);
                    const half* vptr = sv + j * D_CONST;
                    acc0 = acc0 * alpha + beta * __half2float(vptr[lane]);
                    acc1 = acc1 * alpha + beta * __half2float(vptr[lane + 32]);
                    acc2 = acc2 * alpha + beta * __half2float(vptr[lane + 64]);
                    acc3 = acc3 * alpha + beta * __half2float(vptr[lane + 96]);
                    l = l * alpha + beta;
                    m = new_m;
                }
            }
        }
        __syncthreads();
    }

    if (q_pos < S) {
        const float inv_l = 1.0f / l;
        half* optr = out + (((b * H + h) * S + q_pos) * D_CONST);
        optr[lane]      = __float2half_rn(acc0 * inv_l);
        optr[lane + 32] = __float2half_rn(acc1 * inv_l);
        optr[lane + 64] = __float2half_rn(acc2 * inv_l);
        optr[lane + 96] = __float2half_rn(acc3 * inv_l);
    }
}

} // namespace

torch::Tensor h100_gqa_causal_attention(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
    const int B = static_cast<int>(q.size(0));
    const int H = static_cast<int>(q.size(1));
    const int S = static_cast<int>(q.size(2));
    const int D = static_cast<int>(q.size(3));
    const int KVH = static_cast<int>(k.size(1));

    auto out = torch::empty_like(q);
    if (D != D_CONST || q.scalar_type() != torch::kFloat16 || k.scalar_type() != torch::kFloat16 || v.scalar_type() != torch::kFloat16) {
        TORCH_CHECK(false, "h100_gqa_causal_attention supports only fp16 head_size=128");
    }

    dim3 grid((S + ROWS_PER_CTA - 1) / ROWS_PER_CTA, H, B);
    dim3 block(THREADS);
    size_t shmem = 2ull * TILE_K * D_CONST * sizeof(half);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    gqa_causal_attn_kernel<<<grid, block, shmem, stream>>>(
        reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()),
        B, H, KVH, S);
    return out;
}
