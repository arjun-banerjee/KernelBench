#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <vector>
#include <stdint.h>

static __forceinline__ __device__ int8_t f32_to_i8_rn_sat(float x) {
  uint32_t dst;
  asm volatile("cvt.rni.sat.s8.f32 %0, %1;" : "=r"(dst) : "f"(x));
  return reinterpret_cast<const int8_t&>(dst);
}

static __forceinline__ __device__ float warp_reduce_max(float v) {
  v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 16));
  v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 8));
  v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 4));
  v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 2));
  v = fmaxf(v, __shfl_down_sync(0xffffffffu, v, 1));
  return v;
}

// 256 threads is a good point for the fixed hidden size used here (4096):
// enough warps to cover memory latency without adding the reduction overhead of
// larger CTAs.  Shared memory usage is intentionally tiny to maximize residency
// on Hopper.
template <typename scalar_t>
__global__ __launch_bounds__(256, 3)
void dyn_quant_kernel(const scalar_t* __restrict__ x,
                      int8_t* __restrict__ y,
                      float* __restrict__ scales,
                      int hidden_size) {
  const int row = blockIdx.x;
  const int tid = threadIdx.x;
  const scalar_t* __restrict__ row_x = x + (int64_t)row * hidden_size;
  int8_t* __restrict__ row_y = y + (int64_t)row * hidden_size;

  float tmax = 0.0f;

#pragma unroll 4
  for (int i = tid; i < hidden_size; i += blockDim.x) {
    float v = fabsf((float)row_x[i]);
    tmax = fmaxf(tmax, v);
  }

  tmax = warp_reduce_max(tmax);

  __shared__ float warp_max[8];
  if ((tid & 31) == 0) warp_max[tid >> 5] = tmax;
  __syncthreads();

  float bmax = 0.0f;
  if (tid < 8) bmax = warp_max[tid];
  if (tid < 32) bmax = warp_reduce_max(bmax);

  __shared__ float inv_scale_sh;
  if (tid == 0) {
    const float absmax = bmax;
    scales[row] = absmax * (1.0f / 127.0f);
    inv_scale_sh = (absmax == 0.0f) ? 0.0f : (127.0f / absmax);
  }
  __syncthreads();

  const float inv_s = inv_scale_sh;
#pragma unroll 4
  for (int i = tid; i < hidden_size; i += blockDim.x) {
    row_y[i] = f32_to_i8_rn_sat((float)row_x[i] * inv_s);
  }
}

std::vector<torch::Tensor> h100_dynamic_scaled_int8_quant(torch::Tensor input) {
  TORCH_CHECK(input.is_cuda(), "input must be CUDA");
  TORCH_CHECK(input.is_contiguous(), "input must be contiguous");
  TORCH_CHECK(input.dim() >= 1, "input must have at least one dimension");

  const c10::cuda::OptionalCUDAGuard device_guard(device_of(input));
  const int hidden_size = (int)input.size(-1);
  const int64_t num_tokens64 = input.numel() / hidden_size;
  TORCH_CHECK(num_tokens64 <= INT_MAX, "too many rows");
  const int num_tokens = (int)num_tokens64;

  auto out = torch::empty(input.sizes(), input.options().dtype(torch::kInt8));
  auto scales = torch::empty({num_tokens}, input.options().dtype(torch::kFloat32));

  const dim3 grid(num_tokens);
  const dim3 block(256);
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();

  if (input.scalar_type() == torch::kFloat16) {
    dyn_quant_kernel<at::Half><<<grid, block, 0, stream>>>(
        input.data_ptr<at::Half>(), out.data_ptr<int8_t>(),
        scales.data_ptr<float>(), hidden_size);
  } else if (input.scalar_type() == torch::kFloat32) {
    dyn_quant_kernel<float><<<grid, block, 0, stream>>>(
        input.data_ptr<float>(), out.data_ptr<int8_t>(),
        scales.data_ptr<float>(), hidden_size);
  } else {
    TORCH_CHECK(false, "only float16 and float32 inputs are supported");
  }

  return {out, scales};
}
