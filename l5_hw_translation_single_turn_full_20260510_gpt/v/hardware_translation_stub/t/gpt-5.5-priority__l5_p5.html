import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ __launch_bounds__(256, 4)
void rotary_h100_kernel(
    half* __restrict__ q,
    half* __restrict__ k,
    const half* __restrict__ cos,
    const half* __restrict__ sin,
    int num_tokens,
    int num_q_heads,
    int num_k_heads,
    int head_size,
    int rot_dim) {

    const int token = blockIdx.x;
    const int tid = threadIdx.x;
    const int half_dim = rot_dim >> 1;

    __shared__ half scos[128];
    __shared__ half ssin[128];

    if (tid < rot_dim) {
        const int off = token * rot_dim + tid;
        scos[tid] = cos[off];
        ssin[tid] = sin[off];
    }
    __syncthreads();

    const int total_heads = num_q_heads + num_k_heads;
    const int total_pairs = total_heads * half_dim;

    for (int linear = tid; linear < total_pairs; linear += blockDim.x) {
        const int head_all = linear / half_dim;
        const int d0 = linear - head_all * half_dim;
        const int d1 = d0 + half_dim;

        half* __restrict__ base_ptr;
        int head;
        int heads_for_tensor;
        if (head_all < num_q_heads) {
            head = head_all;
            heads_for_tensor = num_q_heads;
            base_ptr = q + ((token * num_q_heads + head) * head_size);
        } else {
            head = head_all - num_q_heads;
            heads_for_tensor = num_k_heads;
            base_ptr = k + ((token * num_k_heads + head) * head_size);
        }

        (void)heads_for_tensor;

        const float x = __half2float(base_ptr[d0]);
        const float y = __half2float(base_ptr[d1]);
        const float c0 = __half2float(scos[d0]);
        const float s0 = __half2float(ssin[d0]);
        const float c1 = __half2float(scos[d1]);
        const float s1 = __half2float(ssin[d1]);

        // NeoX/LLaMA rotate_half convention for separate full-width cos/sin:
        // first half:  x*cos - y*sin; second half: y*cos + x*sin.
        base_ptr[d0] = __float2half_rn(x * c0 - y * s0);
        base_ptr[d1] = __float2half_rn(y * c1 + x * s1);
    }
}

void rotary_forward_cuda(torch::Tensor q, torch::Tensor k,
                         torch::Tensor cos, torch::Tensor sin,
                         int64_t rot_dim) {
    const int num_tokens = (int)q.size(0);
    const int num_q_heads = (int)q.size(1);
    const int head_size = (int)q.size(2);
    const int num_k_heads = (int)k.size(1);

    const dim3 grid(num_tokens);
    const dim3 block(256);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    rotary_h100_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(cos.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(sin.data_ptr<at::Half>()),
        num_tokens, num_q_heads, num_k_heads, head_size, (int)rot_dim);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("rotary_forward_cuda", &rotary_forward_cuda, "H100 optimized rotary embedding (in-place)");
}
'''

        super().__init__()
        self.rot_dim = int(rot_dim)
