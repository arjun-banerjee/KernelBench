#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>

#include <mma.h>

#define __CUDA_ARCH__ 900

// Fixed-shape forward for:
// q: [B, H, N, 128], k/v: [B, KVH, N, 128], bf16
// Non-causal scaled dot-product attention with grouped-query heads.
template<int BR, int BC, int D>
__global__ __launch_bounds__(128, 2)
void gqa_flash_bf16_hdim128_kernel(
    const __nv_bfloat16* __restrict__ q,
    const __nv_bfloat16* __restrict__ k,
    const __nv_bfloat16* __restrict__ v,
    __nv_bfloat16* __restrict__ out,
    int B, int H, int KVH, int N)
{
    constexpr float kScale = 0.08838834764831845f; // 1 / sqrt(128)

    const int tid = threadIdx.x;
    const int warp_id = tid >> 5;
    const int bh = blockIdx.x;
    const int b = bh / H;
    const int h = bh - b * H;
    const int group = H / KVH;
    const int kvh = h / group;
    const int row0 = blockIdx.y * BR;

    const __nv_bfloat16* q_base = q + (((b * H + h) * N + row0) * D);
    const __nv_bfloat16* k_base = k + ((b * KVH + kvh) * N * D);
    const __nv_bfloat16* v_base = v + ((b * KVH + kvh) * N * D);
    __nv_bfloat16* o_base = out + (((b * H + h) * N + row0) * D);

    __shared__ float scores[BR * BC];             // [16, 64]
    __shared__ __nv_bfloat16 probs[BR * BC];      // [16, 64]
    __shared__ float acc[BR * D];                 // online numerator [16, 128]
    __shared__ float tmp[BR * D];                 // current block P @ V [16, 128]
    __shared__ float m[BR];
    __shared__ float l[BR];
    __shared__ float alpha[BR];
    __shared__ float m_new_s[BR];

    for (int i = tid; i < BR * D; i += blockDim.x) acc[i] = 0.0f;
    if (tid < BR) {
        m[tid] = -CUDART_INF_F;
        l[tid] = 0.0f;
    }
    __syncthreads();

    for (int col0 = 0; col0 < N; col0 += BC) {
        // QK^T: four warps compute the four 16-column panels of a 16x64 score tile.
        if (warp_id < 4) {
            const int n_off = warp_id * 16;
            wmma::fragment<wmma::matrix_a, 16, 16, 16, __nv_bfloat16, wmma::row_major> a_frag;
            wmma::fragment<wmma::matrix_b, 16, 16, 16, __nv_bfloat16, wmma::col_major> b_frag;
            wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;
            wmma::fill_fragment(c_frag, 0.0f);

            for (int kk = 0; kk < D; kk += 16) {

                // Treat K storage [seq, dim] as a column-major [dim, seq] tile.
                wmma::load_matrix_sync(b_frag, k_base + (col0 + n_off) * D + kk, D);
                wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);

            wmma::store_matrix_sync(scores + n_off, c_frag, BC, wmma::mem_row_major);

            for (int c = lane8; c < BC; c += 8) {

            }

                mx = fmaxf(mx, __shfl_down_sync(0xffffffff, mx, off, 8));
            }
            if (lane8 == 0) {
                const float old_m = m[r8];
                const float new_m = fmaxf(old_m, mx);
                m_new_s[r8] = new_m;
                alpha[r8] = isinf(old_m) ? 0.0f : __expf(old_m - new_m);
            }

        __syncthreads();

        // P @ V: use tensor cores.  Four warps cover 64 output dimensions per pass.
        #pragma unroll
        for (int pass = 0; pass < 2; ++pass) {
            const int d_off = pass * 64 + warp_id * 16;
            wmma::fragment<wmma::matrix_a, 16, 16, 16, __nv_bfloat16, wmma::row_major> pa_frag;
            wmma::fragment<wmma::matrix_b, 16, 16, 16, __nv_bfloat16, wmma::row_major> vb_frag;
            wmma::fragment<wmma::accumulator, 16, 16, 16, float> pv_frag;
            wmma::fill_fragment(pv_frag, 0.0f);

                wmma::load_matrix_sync(vb_frag, v_base + (col0 + kk) * D + d_off, D);
                wmma::mma_sync(pv_frag, pa_frag, vb_frag, pv_frag);
            }
            wmma::store_matrix_sync(tmp + d_off, pv_frag, D, wmma::mem_row_major);
        }
        __syncthreads();

        for (int i = tid; i < BR * D; i += blockDim.x) {
            const int rr = i / D;
            acc[i] = acc[i] * alpha[rr] + tmp[i];
        }
        __syncthreads();
    }

    for (int i = tid; i < BR * D; i += blockDim.x) {
        const int rr = i / D;
        const float inv_l = 1.0f / l[rr];
        o_base[i] = __float2bfloat16(acc[i] * inv_l);
    }
}

    auto out = torch::empty_like(q);
    const int B = (int)q.size(0);
    const int H = (int)q.size(1);

        reinterpret_cast<__nv_bfloat16*>(out.data_ptr<at::BFloat16>()),

    return out;
}
