import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_rms_norm_cpp = r"""
torch::Tensor rms_norm_h100_cuda(torch::Tensor x, torch::Tensor weight, double eps);

_rms_norm_cuda = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

static __forceinline__ __device__ float warp_sum(float v) {
    v += __shfl_down_sync(0xffffffff, v, 16);
    v += __shfl_down_sync(0xffffffff, v, 8);
    v += __shfl_down_sync(0xffffffff, v, 4);
    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

// Specialized Hopper-friendly RMSNorm for x: [rows, 4096] fp16 and fp32 weight.
// One CTA per token.  The whole row is read once into registers (8 half2/thread),
// then reused after the block reduction, avoiding the second input pass.
__global__ __launch_bounds__(256, 4)
void rms_norm_4096_f16_kernel(const half* __restrict__ x,
                              const float* __restrict__ weight,
                              half* __restrict__ out,
                              int rows,
                              float eps) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    if (row >= rows) return;

    const half2* __restrict__ x2 = reinterpret_cast<const half2*>(x + ((size_t)row << 12));
    half2* __restrict__ out2 = reinterpret_cast<half2*>(out + ((size_t)row << 12));
    const float2* __restrict__ w2 = reinterpret_cast<const float2*>(weight);

    half2 vals[8];
    float ss = 0.0f;

#pragma unroll
    for (int k = 0; k < 8; ++k) {
        const int idx = tid + (k << 8);  // tid + k*256, covers 2048 half2s
        half2 h = x2[idx];
        vals[k] = h;
        float2 f = __half22float2(h);
        ss = fmaf(f.x, f.x, ss);
        ss = fmaf(f.y, f.y, ss);
    }

    ss = warp_sum(ss);

    __shared__ float warp_partials[8];
    __shared__ float scale_s;
    const int lane = tid & 31;
    const int warp = tid >> 5;
    if (lane == 0) warp_partials[warp] = ss;
    __syncthreads();

    float block_sum = (tid < 8) ? warp_partials[lane] : 0.0f;
    if (warp == 0) {
        block_sum = warp_sum(block_sum);
        if (tid == 0) {
            scale_s = rsqrtf(block_sum * 0.000244140625f + eps); // /4096
        }
    }
    __syncthreads();

    const float scale = scale_s;
#pragma unroll
    for (int k = 0; k < 8; ++k) {
        const int idx = tid + (k << 8);
        float2 f = __half22float2(vals[k]);
        float2 w = w2[idx];
        out2[idx] = __floats2half2_rn(f.x * scale * w.x, f.y * scale * w.y);
    }
}

torch::Tensor rms_norm_h100_cuda(torch::Tensor x, torch::Tensor weight, double eps) {
    TORCH_CHECK(x.is_cuda(), "x must be CUDA");
    TORCH_CHECK(weight.is_cuda(), "weight must be CUDA");
    TORCH_CHECK(x.scalar_type() == torch::kFloat16, "optimized kernel expects fp16 input");
    TORCH_CHECK(weight.scalar_type() == torch::kFloat32, "optimized kernel expects fp32 weight");
    TORCH_CHECK(x.is_contiguous(), "x must be contiguous");
    TORCH_CHECK(weight.is_contiguous(), "weight must be contiguous");
    TORCH_CHECK(x.dim() == 2 && x.size(1) == 4096, "optimized kernel expects shape [N, 4096]");

    auto out = torch::empty_like(x);
    const int rows = (int)x.size(0);
    const dim3 block(256);
    const dim3 grid(rows);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    const half* x_ptr = reinterpret_cast<const half*>(x.data_ptr<at::Half>());
    half* out_ptr = reinterpret_cast<half*>(out.data_ptr<at::Half>());
    const float* w_ptr = weight.data_ptr<float>();

    rms_norm_4096_f16_kernel<<<grid, block, 0, stream>>>(x_ptr, w_ptr, out_ptr, rows, (float)eps);
    return out;
}
