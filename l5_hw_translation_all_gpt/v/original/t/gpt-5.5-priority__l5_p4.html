#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

// H100-optimized SwiGLU for FP16: out = silu(x[:,:d]) * x[:,d:]
// One CTA per token/row.  Threads process half2 values for coalesced 32-bit
// loads/stores and use native half2 exp/mul/add to keep register pressure low.
__global__ __launch_bounds__(256, 4)
void swiglu_half2_kernel(const half* __restrict__ x,
                         half* __restrict__ out,
                         int rows, int d) {
    const int row = blockIdx.x;
    if (row >= rows) return;

    const half* __restrict__ gate = x + ((size_t)row) * (2 * (size_t)d);
    const half* __restrict__ up   = gate + d;
    half* __restrict__ dst = out + ((size_t)row) * (size_t)d;

    const int tid = threadIdx.x;
    const int d2 = d >> 1;

    const half2* __restrict__ g2p = reinterpret_cast<const half2*>(gate);
    const half2* __restrict__ u2p = reinterpret_cast<const half2*>(up);
    half2* __restrict__ o2p = reinterpret_cast<half2*>(dst);

    const half2 one = __float2half2_rn(1.0f);

    // For D=2048 and blockDim=256, each thread handles exactly four half2s.
    // Stride loop keeps the kernel correct for other D as well.
    #pragma unroll 4
    for (int i = tid; i < d2; i += blockDim.x) {
        half2 g = __ldg(g2p + i);
        half2 u = __ldg(u2p + i);
        half2 e = h2exp(__hneg2(g));              // exp(-gate)
        half2 sig = __h2div(one, __hadd2(one, e));
        half2 y = __hmul2(__hmul2(g, sig), u);
        o2p[i] = y;
    }

    // Odd-d fallback (not used by the target model, but makes the op robust).
    if ((d & 1) && tid == 0) {
        int idx = d - 1;
        float gf = __half2float(gate[idx]);
        float uf = __half2float(up[idx]);
        dst[idx] = __float2half_rn((gf / (1.0f + expf(-gf))) * uf);
    }
}

    TORCH_CHECK(x.size(-1) % 2 == 0, "last dimension must be even");

    if (!x.is_contiguous()) x = x.contiguous();
