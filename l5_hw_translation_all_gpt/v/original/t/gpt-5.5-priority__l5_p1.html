    unsigned mask = 0xffffffffu;
    v += __shfl_down_sync(mask, v, 16);

// warps cooperate: one warp per query. K/V are staged once per CTA in 64-token

// avoids re-reading the same KV vectors for the 8 causal queries in the block.
__global__ __launch_bounds__(256, 2)
void gqa_causal_attn_hdim128_kernel(

    const half* __restrict__ k,
    const half* __restrict__ v,
    half* __restrict__ out,

    const int b = blockIdx.z;
