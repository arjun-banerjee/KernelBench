#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <stdint.h>

// Fixed problem family used by the benchmark: M=16, K=N=4096, group_size=128.
// The kernel is nevertheless dimensioned from tensor sizes (N multiple of 32,
// K multiple of 128).  One warp computes a 4 x 32 output tile.  This gives
// 4*(N/32) blocks for M=16, enough parallelism on H100 while keeping the
// frequently reused activation tile in shared memory.

namespace {
constexpr int GROUP = 128;
constexpr int TILE_M = 4;
constexpr int TILE_N = 32;

__global__ __launch_bounds__(32, 16)
void int4_woq_gemm_m4n32_kernel(
    const half* __restrict__ x,             // [M, K]
    const unsigned char* __restrict__ qw,   // [K, N/2]
    const half* __restrict__ scales,        // [K/128, N]
    const half* __restrict__ zeros,         // [K/128, N]
    half* __restrict__ out,                 // [M, N]
    int M, int K, int N) {

  const int lane = threadIdx.x;             // 0..31, exactly one warp/block
  const int n = blockIdx.x * TILE_N + lane;
  const int m_base = blockIdx.y * TILE_M;
  const int groups = K / GROUP;
  const int n_half = N >> 1;

  extern __shared__ half sx[];              // TILE_M * GROUP halfs

  float acc0 = 0.0f;
  float acc1 = 0.0f;
  float acc2 = 0.0f;
  float acc3 = 0.0f;

  #pragma unroll 1
  for (int g = 0; g < groups; ++g) {
    // Cooperatively stage a 4x128 activation slice.  For the benchmark M is
    // exactly 16, but bounds make edge tiles safe.
    #pragma unroll
    for (int off = lane; off < TILE_M * GROUP; off += 32) {
      const int mi = off / GROUP;
      const int kk = off - mi * GROUP;
      const int m = m_base + mi;
      sx[off] = (m < M) ? x[m * K + g * GROUP + kk] : __float2half(0.0f);
    }
    __syncwarp();

    float sc = 0.0f;
    float zp = 0.0f;
    if (n < N) {
      sc = __half2float(scales[g * N + n]);
      zp = __half2float(zeros[g * N + n]);
    }

    if (n < N) {
      const int byte_col = n >> 1;
      const bool high = (n & 1) != 0;

      #pragma unroll
      for (int kk = 0; kk < GROUP; ++kk) {
        const unsigned char packed = qw[(g * GROUP + kk) * n_half + byte_col];
        const int qv = high ? (int)(packed >> 4) : (int)(packed & 0x0f);

        // Match PyTorch dequantization semantics closely: dequantized weight is
        // materialized as FP16 before the matmul.
        const half wh = __float2half_rn(((float)qv - zp) * sc);
        const float w = __half2float(wh);

        acc0 = fmaf(__half2float(sx[0 * GROUP + kk]), w, acc0);
        acc1 = fmaf(__half2float(sx[1 * GROUP + kk]), w, acc1);
        acc2 = fmaf(__half2float(sx[2 * GROUP + kk]), w, acc2);
        acc3 = fmaf(__half2float(sx[3 * GROUP + kk]), w, acc3);
      }
    }
    __syncwarp();
  }

  if (n < N) {
    if (m_base + 0 < M) out[(m_base + 0) * N + n] = __float2half_rn(acc0);
    if (m_base + 1 < M) out[(m_base + 1) * N + n] = __float2half_rn(acc1);
    if (m_base + 2 < M) out[(m_base + 2) * N + n] = __float2half_rn(acc2);
    if (m_base + 3 < M) out[(m_base + 3) * N + n] = __float2half_rn(acc3);
  }
}
} // namespace

                                 torch::Tensor qweight,
                                 torch::Tensor scales,
                                 torch::Tensor zeros) {
  TORCH_CHECK(x.is_cuda() && qweight.is_cuda() && scales.is_cuda() && zeros.is_cuda(),

  TORCH_CHECK(x.scalar_type() == at::kHalf, "x must be float16");

  TORCH_CHECK(qweight.scalar_type() == at::kByte, "qweight must be uint8");
  TORCH_CHECK(x.is_contiguous() && qweight.is_contiguous() && scales.is_contiguous() && zeros.is_contiguous(),

  const int M = (int)x.size(0);
  const int K = (int)x.size(1);
  const int N = (int)scales.size(1);
  TORCH_CHECK(qweight.size(0) == K && qweight.size(1) == N / 2, "bad qweight shape");
  TORCH_CHECK((K % GROUP) == 0, "K must be divisible by 128");
  TORCH_CHECK((N % TILE_N) == 0, "N must be divisible by 32");

  auto out = torch::empty({M, N}, x.options());

  const dim3 block(32);

    def __init__(self, k: int, n: int, group_size: int = 128):
        super().__init__()
        self.k = k
        self.n = n
        self.group_size = group_size
        num_groups = k // group_size
        self.register_buffer(
            "qweight",
            torch.randint(0, 256, (k, n // 2), dtype=torch.uint8, device="cuda"),
        )

            torch.randn(num_groups, n, dtype=torch.float16, device="cuda") * 0.01,
