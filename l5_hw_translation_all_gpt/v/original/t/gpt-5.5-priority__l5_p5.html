import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <vector>

#define NUM_HEADS 32
#define NUM_KV_HEADS 8
#define HEAD_SIZE 128
#define ROT_DIM 64
#define HALF_ROT 32

__global__ __launch_bounds__(256, 4)
void rope_h100_kernel(const __half* __restrict__ q,
                      const __half* __restrict__ k,
                      const __half* __restrict__ cos,
                      const __half* __restrict__ sin,
                      __half* __restrict__ qo,
                      __half* __restrict__ ko) {
    const int token = blockIdx.x;
    const int tid = threadIdx.x;

    __shared__ __half cs[ROT_DIM * 2];
    if (tid < ROT_DIM) {
        cs[tid] = __ldg(cos + token * ROT_DIM + tid);
        cs[ROT_DIM + tid] = __ldg(sin + token * ROT_DIM + tid);
    }
    __syncthreads();

    const int64_t q_token_base = (int64_t)token * NUM_HEADS * HEAD_SIZE;
    const int64_t k_token_base = (int64_t)token * NUM_KV_HEADS * HEAD_SIZE;

    // Rotate dimensions [0, 63].  One work item produces the paired outputs
    // d and d+32 for a single head, avoiding reloading the source pair twice.
    constexpr int q_pairs = NUM_HEADS * HALF_ROT;
    constexpr int k_pairs = NUM_KV_HEADS * HALF_ROT;
    constexpr int total_pairs = q_pairs + k_pairs;

    for (int t = tid; t < total_pairs; t += blockDim.x) {
        const bool is_q = (t < q_pairs);
        const int u = is_q ? t : (t - q_pairs);
        const int head = u / HALF_ROT;
        const int d = u - head * HALF_ROT;

        const __half* __restrict__ in;
        __half* __restrict__ out;
        int64_t base;
        if (is_q) {
            in = q;
            out = qo;
            base = q_token_base + (int64_t)head * HEAD_SIZE;
        } else {
            in = k;
            out = ko;
            base = k_token_base + (int64_t)head * HEAD_SIZE;
        }

        const float x1 = __half2float(__ldg(in + base + d));
        const float x2 = __half2float(__ldg(in + base + d + HALF_ROT));
        const float c1 = __half2float(cs[d]);
        const float s1 = __half2float(cs[ROT_DIM + d]);
        const float c2 = __half2float(cs[d + HALF_ROT]);
        const float s2 = __half2float(cs[ROT_DIM + d + HALF_ROT]);

        // Reference: x * cos + rotate_half(x) * sin, where
        // rotate_half([x1, x2]) = [-x2, x1].
        out[base + d] = __float2half_rn(fmaf(x1, c1, -x2 * s1));
        out[base + d + HALF_ROT] = __float2half_rn(fmaf(x2, c2, x1 * s2));
    }

    // Copy the non-rotary tail [64, 127].  It is 128-byte aligned per head;
    // use 16-byte vector transactions for high HBM efficiency on Hopper.
    constexpr int q_vecs = NUM_HEADS * 8;      // 64 halfs / 8 halfs per uint4
    constexpr int k_vecs = NUM_KV_HEADS * 8;
    constexpr int total_vecs = q_vecs + k_vecs;
    for (int v = tid; v < total_vecs; v += blockDim.x) {
        const bool is_q = (v < q_vecs);
        const int u = is_q ? v : (v - q_vecs);
        const int head = u >> 3;
        const int vec = u & 7;
        if (is_q) {
            const int64_t off = q_token_base + (int64_t)head * HEAD_SIZE + ROT_DIM;
            const uint4* __restrict__ src = reinterpret_cast<const uint4*>(q + off);
            uint4* __restrict__ dst = reinterpret_cast<uint4*>(qo + off);
            dst[vec] = src[vec];
        } else {
            const int64_t off = k_token_base + (int64_t)head * HEAD_SIZE + ROT_DIM;
            const uint4* __restrict__ src = reinterpret_cast<const uint4*>(k + off);
            uint4* __restrict__ dst = reinterpret_cast<uint4*>(ko + off);
            dst[vec] = src[vec];
        }
    }
}

std::vector<torch::Tensor> rope_forward(torch::Tensor query,
                                        torch::Tensor key,
                                        torch::Tensor cos,
                                        torch::Tensor sin) {
    auto q_out = torch::empty_like(query);
    auto k_out = torch::empty_like(key);

    const int num_tokens = query.size(0);
    const dim3 grid(num_tokens);
    const dim3 block(256);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    rope_h100_kernel<<<grid, block, 0, stream>>>(
        reinterpret_cast<const __half*>(query.data_ptr<at::Half>()),
        reinterpret_cast<const __half*>(key.data_ptr<at::Half>()),
        reinterpret_cast<const __half*>(cos.data_ptr<at::Half>()),
        reinterpret_cast<const __half*>(sin.data_ptr<at::Half>()),
        reinterpret_cast<__half*>(q_out.data_ptr<at::Half>()),
        reinterpret_cast<__half*>(k_out.data_ptr<at::Half>()));

    return {q_out, k_out};
}
