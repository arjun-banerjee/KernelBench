import torch
import torch.nn as nn
import torch.nn.functional as F


class ModelNew(nn.Module):
    def __init__(self, num_heads: int, num_kv_heads: int, head_dim: int):
        super().__init__()
        self.num_heads = num_heads
        self.num_kv_heads = num_kv_heads
        self.head_dim = head_dim
        self.groups = num_heads // num_kv_heads
        self.scale = head_dim ** -0.5

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor) -> torch.Tensor:
        # Use PyTorch's fused FlashAttention backend directly with grouped-query
        # attention semantics.  This avoids the explicit repeat_interleave of K/V
        # in the reference model while producing the same causal SDPA result.
        return F.scaled_dot_product_attention(
            q, k, v, scale=self.scale, is_causal=True, enable_gqa=True
        )
