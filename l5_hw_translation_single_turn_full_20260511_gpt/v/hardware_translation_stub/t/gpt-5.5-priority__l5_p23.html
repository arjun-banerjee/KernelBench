__global__ void zero_kernel(float* __restrict__ out, int total) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = blockDim.x * gridDim.x;
    for (int i = idx; i < total; i += stride) out[i] = 0.0f;
}

// One warp computes one 16x16 tile of C = A[:, :] * B[:, indices].
// A is row-major MxK half, B is row-major KxN half. Output is row-major MxN float,
// scattered to the original selected columns.
__global__ void gather_gemm_scatter_wmma_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    const long long* __restrict__ indices,
    float* __restrict__ out,
    int M, int N, int K, int index_size, float alpha) {

#if defined(__CUDA_ARCH__) && (__CUDA_ARCH__ >= 700)
    const int tile_n = blockIdx.x;  // over selected columns
    const int tile_m = blockIdx.y;  // over rows
    const int lane = threadIdx.x;   // 0..31; exactly one warp per block

    __shared__ half As[16 * 16];
    __shared__ half Bs[16 * 16];
    __shared__ float Cs[16 * 16];

    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> c_frag;
    wmma::fill_fragment(c_frag, 0.0f);

    const int base_m = tile_m * 16;
    const int base_j = tile_n * 16;

    for (int kk0 = 0; kk0 < K; kk0 += 16) {
        // Cooperative load. 512 half values total; each lane loads 16 values split over A and B.
        #pragma unroll
        for (int t = lane; t < 256; t += 32) {
            int r = t >> 4;
            int c = t & 15;

            int gk = kk0 + c;
            half av = __float2half(0.0f);
            if (gm < M && gk < K) av = A[gm * K + gk];

            int r = t >> 4;
            int c = t & 15;

    __syncthreads();

#endif
}

    const int M = (int)A.size(0);
    const int K = (int)A.size(1);
    const int N = (int)B.size(1);
