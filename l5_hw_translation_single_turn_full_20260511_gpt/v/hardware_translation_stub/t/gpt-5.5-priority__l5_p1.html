// Fixed-shape Hopper-oriented grouped-query FP16 attention kernel.
// q: [B, 32, 512, 128], k/v: [B, 8, 512, 128], out: [B, 32, 512, 128]
template<int HEAD_SIZE, int SEQ_LEN, int NUM_HEADS, int NUM_KV_HEADS, int BLOCK_THREADS>
__global__ __launch_bounds__(BLOCK_THREADS, 2)
void gqa_attention_kernel(const half* __restrict__ q,
                          const half* __restrict__ k,
                          const half* __restrict__ v,
                          half* __restrict__ out,
                          int B) {
    constexpr int GROUP = NUM_HEADS / NUM_KV_HEADS;
    constexpr float SCALE = 0.08838834764831845f; // 1 / sqrt(128)

    const int s_q = blockIdx.x;
    const int h = blockIdx.y;
    const int b = blockIdx.z;
    const int kvh = h / GROUP;
    const int tid = threadIdx.x;

    __shared__ __align__(16) half q_sh[HEAD_SIZE];
    __shared__ float logits[SEQ_LEN];

    const half* q_ptr = q + (((b * NUM_HEADS + h) * SEQ_LEN + s_q) * HEAD_SIZE);
    const half* k_base = k + (((b * NUM_KV_HEADS + kvh) * SEQ_LEN) * HEAD_SIZE);
    const half* v_base = v + (((b * NUM_KV_HEADS + kvh) * SEQ_LEN) * HEAD_SIZE);
    half* out_ptr = out + (((b * NUM_HEADS + h) * SEQ_LEN + s_q) * HEAD_SIZE);

    if (tid < HEAD_SIZE) q_sh[tid] = q_ptr[tid];
    __syncthreads();

    const half2* q2 = reinterpret_cast<const half2*>(q_sh);
    float local_max = -CUDART_INF_F;

    for (int m = tid; m < SEQ_LEN; m += BLOCK_THREADS) {
        const half2* k2 = reinterpret_cast<const half2*>(k_base + m * HEAD_SIZE);

        logits[m] = l;
        local_max = fmaxf(local_max, l);

        float e = __expf(logits[m] - max_val);
        logits[m] = e;
        local_sum += e;

        }
        out_ptr[tid] = __float2half_rn(acc);
    }
}
