import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

M = 5120
N = 4096
K = 4096
ALPHA = 1
BETA = 0

cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void gemm_kernel(const int8_t* __restrict__ A, const int8_t* __restrict__ B, int32_t* __restrict__ C, int32_t* __restrict__ D, int32_t alpha, int32_t beta, int K, int N) {
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;

    if (row >= M || col >= N) return;

    int32_t sum = 0;
    for (int k = 0; k < K; ++k) {
        sum += int32_t(A[row * K + k]) * int32_t(B[k * N + col]);
    }

    int32_t c_val = C[row * N + col];
    D[row * N + col] = alpha * sum + beta * c_val;
}

void gemm_cuda(const int8_t* A, const int8_t* B, int32_t* C, int32_t* D, int32_t alpha, int32_t beta, int K, int N) {
    dim3 blockSize(16, 16);
    dim3 gridSize((N + blockSize.x - 1) / blockSize.x, (M + blockSize.y - 1) / blockSize.y);

    gemm_kernel<<<gridSize, blockSize>>> (A, B, C, D, alpha, beta, K, N);
    cudaDeviceSynchronize();
}
'''
