import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Hopper-targeted fixed-size GEMM module for C = A[M,K] @ W[K,N]
# Input A is fp32. Internally compute with TF32 tensor cores via WMMA for accuracy within fp32 tol.

_cuda_src = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <mma.h>

using namespace nvcuda;

#ifndef __CUDA_ARCH__
#define __CUDA_ARCH__ 900
#endif

// One warp computes one 16x16 tile. Grid covers M/16 x N/16 tiles.
// A row-major [M,K], B row-major [K,N], C row-major [M,N].
// For the benchmark sizes (5120,4096,4096) all dimensions are multiples of 16.
__global__ void gemm_tf32_warp_kernel(const float* __restrict__ A,
                                      const float* __restrict__ B,
                                      float* __restrict__ C,
                                      int M, int N, int K,
                                      float alpha) {
#if __CUDA_ARCH__ >= 800
    int tile_m = blockIdx.y;
    int tile_n_base = blockIdx.x * 4;   // 4 warps per block along N
    int warp_id = threadIdx.x >> 5;
    int lane = threadIdx.x & 31;
    int tile_n = tile_n_base + warp_id;
    if (tile_n * 16 >= N || tile_m * 16 >= M) return;

    const int row = tile_m * 16;
    const int col = tile_n * 16;

    wmma::fragment<wmma::matrix_a, 16, 16, 8, wmma::precision::tf32, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 8, wmma::precision::tf32, wmma::row_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 8, float> c_frag;
    wmma::fill_fragment(c_frag, 0.0f);

    #pragma unroll 4
    for (int k0 = 0; k0 < K; k0 += 8) {
        wmma::load_matrix_sync(a_frag, A + row * K + k0, K);
        wmma::load_matrix_sync(b_frag, B + k0 * N + col, N);
        // Explicit conversion to TF32 for tensor cores, matching PyTorch default matmul on NVIDIA.
        #pragma unroll
        for (int i = 0; i < a_frag.num_elements; ++i) a_frag.x[i] = wmma::__float_to_tf32(a_frag.x[i]);
        #pragma unroll
        for (int i = 0; i < b_frag.num_elements; ++i) b_frag.x[i] = wmma::__float_to_tf32(b_frag.x[i]);
        wmma::mma_sync(c_frag, a_frag, b_frag, c_frag);
    }

    if (alpha != 1.0f) {
        #pragma unroll
        for (int i = 0; i < c_frag.num_elements; ++i) c_frag.x[i] *= alpha;
    }
    wmma::store_matrix_sync(C + row * N + col, c_frag, N, wmma::mem_row_major);
#endif
}

// Fallback/general small boundary kernel, also used if shape is not multiple of tile.
__global__ void gemm_simt_kernel(const float* __restrict__ A,
                                 const float* __restrict__ B,
                                 float* __restrict__ C,
                                 int M, int N, int K,
                                 float alpha) {
    int col = blockIdx.x * blockDim.x + threadIdx.x;
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    if (row >= M || col >= N) return;
    float sum = 0.0f;
    for (int k = 0; k < K; ++k) sum += A[row*K + k] * B[k*N + col];
    C[row*N + col] = alpha * sum;
}

torch::Tensor gemm_forward(torch::Tensor A, torch::Tensor B, double alpha_d) {
    TORCH_CHECK(A.is_cuda() && B.is_cuda(), "A and B must be CUDA tensors");
    TORCH_CHECK(A.dtype() == torch::kFloat32 && B.dtype() == torch::kFloat32, "expected fp32 tensors");
    TORCH_CHECK(A.dim() == 2 && B.dim() == 2, "expected matrices");
    TORCH_CHECK(A.is_contiguous() && B.is_contiguous(), "expected contiguous tensors");
    int M = (int)A.size(0);
    int K = (int)A.size(1);
    TORCH_CHECK((int)B.size(0) == K, "K mismatch");
    int N = (int)B.size(1);
    auto C = torch::empty({M, N}, A.options());
    float alpha = (float)alpha_d;

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    if ((M % 16 == 0) && (N % 16 == 0) && (K % 8 == 0)) {
        dim3 block(128, 1, 1);          // 4 warps
        dim3 grid((N + 63) / 64, (M + 15) / 16, 1);
        gemm_tf32_warp_kernel<<<grid, block, 0, stream>>>(A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(), M, N, K, alpha);
    } else {
        dim3 block(16, 16, 1);
        dim3 grid((N + 15) / 16, (M + 15) / 16, 1);
        gemm_simt_kernel<<<grid, block, 0, stream>>>(A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(), M, N, K, alpha);
    }
    return C;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gemm_forward", &gemm_forward, "Hopper TF32 GEMM forward");
}
'''
