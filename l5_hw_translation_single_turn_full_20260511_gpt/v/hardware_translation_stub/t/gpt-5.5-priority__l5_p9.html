import torch.nn as nn

import triton
import triton.language as tl

def _gqa_flash_fwd_kernel(Q, K, V, O,
                          SCALE: tl.constexpr,
                          N_CTX: tl.constexpr,
                          H: tl.constexpr,
                          HK: tl.constexpr,

    pid_bh = tl.program_id(1)

    h = pid_bh - b * H
