    int idx = scalar_start + tid;
    for (; idx < end; idx += BLOCK_SIZE) {
        uint32_t dst;
        asm volatile("cvt.rni.sat.s8.f32 %0, %1;" : "=r"(dst) : "f"(static_cast<float>(row_in[idx]) * inv_s));
        row_out[idx] = reinterpret_cast<const int8_t&>(dst);
    }
}

torch::Tensor dynamic_scaled_int8_quant(torch::Tensor const& input) {
    TORCH_CHECK(input.is_contiguous());
    TORCH_CHECK(input.scalar_type() == at::ScalarType::Half);
    int hidden_size = input.size(-1);
    int num_tokens = input.numel() / hidden_size;

    auto out = torch::empty_like(input, input.options().dtype(torch::kInt8));
    auto scales = torch::empty({num_tokens}, input.options().dtype(torch::kFloat32));

    const at::cuda::OptionalCUDAGuard device_guard(device_of(input));
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    int target_total_blocks = 256;
    int blocks_per_token = std::max(1, target_total_blocks / num_tokens);
    int max_blocks_per_token = std::max(1, hidden_size / 64);
    if (blocks_per_token > max_blocks_per_token) blocks_per_token = max_blocks_per_token;
    if (blocks_per_token > 64) blocks_per_token = 64;

    int elems_per_block = (hidden_size + blocks_per_token - 1) / blocks_per_token;
    elems_per_block = ((elems_per_block + 7) / 8) * 8;

    int grid = num_tokens * blocks_per_token;
    auto partial_max = torch::empty({num_tokens * blocks_per_token}, scales.options());

    int block_size;
    if (elems_per_block >= 2048) block_size = 256;
    else if (elems_per_block >= 1024) block_size = 128;
    else if (elems_per_block >= 512) block_size = 64;
    else block_size = 32;

    switch (block_size) {
        case 32:
            dynamic_find_scale_kernel<32><<<grid, 32, 0, stream>>>(
                input.data_ptr<at::Half>(), partial_max.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
        case 64:
            dynamic_find_scale_kernel<64><<<grid, 64, 0, stream>>>(
                input.data_ptr<at::Half>(), partial_max.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
        case 128:
            dynamic_find_scale_kernel<128><<<grid, 128, 0, stream>>>(
                input.data_ptr<at::Half>(), partial_max.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
        case 256:
            dynamic_find_scale_kernel<256><<<grid, 256, 0, stream>>>(
                input.data_ptr<at::Half>(), partial_max.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
    }

    dynamic_reduce_scale_kernel<<<num_tokens, 32, 0, stream>>>(
        partial_max.data_ptr<float>(), scales.data_ptr<float>(), blocks_per_token);

    switch (block_size) {
        case 32:
            dynamic_quant_apply_kernel<32><<<grid, 32, 0, stream>>>(
                input.data_ptr<at::Half>(), out.data_ptr<int8_t>(), scales.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
        case 64:
            dynamic_quant_apply_kernel<64><<<grid, 64, 0, stream>>>(
                input.data_ptr<at::Half>(), out.data_ptr<int8_t>(), scales.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
        case 128:
            dynamic_quant_apply_kernel<128><<<grid, 128, 0, stream>>>(
                input.data_ptr<at::Half>(), out.data_ptr<int8_t>(), scales.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
        case 256:
            dynamic_quant_apply_kernel<256><<<grid, 256, 0, stream>>>(
                input.data_ptr<at::Half>(), out.data_ptr<int8_t>(), scales.data_ptr<float>(), hidden_size, blocks_per_token);
            break;
    }

    return out;
}
