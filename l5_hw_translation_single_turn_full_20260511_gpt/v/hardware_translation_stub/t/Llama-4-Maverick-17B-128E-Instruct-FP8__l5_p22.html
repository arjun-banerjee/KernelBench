__global__ void gemm_kernel(float *A, float *B, float *C, float *D, float alpha, float beta, int K, int N) {

    int thread_idx_n = threadIdx.y;

    int global_thread_idx_m = block_idx_m * tile_size_m + thread_idx_m;

    // Initialize the accumulator

    for (int k = 0; k < K; k += tile_size_k) {
