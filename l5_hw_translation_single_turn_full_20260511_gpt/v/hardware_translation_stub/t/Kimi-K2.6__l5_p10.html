    float m_reg = -1e20f;
    float l_reg = 0.0f;

    const int num_col_tiles = N / Bc_fwd;
    for (int tile_col = 0; tile_col < num_col_tiles; ++tile_col) {
        int col_start = tile_col * Bc_fwd;
        // Load K and V tiles
        for (int idx = tid; idx < Bc_fwd * D; idx += blockDim.x) {
            int c = idx / D;
            int d = idx % D;
            int kv_idx = ((b * H_kv + h_k) * N + col_start + c) * D + d;
            k_smem[c][d] = k[kv_idx];
            v_smem[c][d] = v[kv_idx];
        }
        __syncthreads();

        // Process each column j of the tile with online softmax
        for (int j = 0; j < Bc_fwd; ++j) {
            int i = tid;                     // each thread owns one row
            float s = 0.0f;
            #pragma unroll
            for (int d = 0; d < D; ++d) {
                s += static_cast<float>(q_smem[i][d]) * static_cast<float>(k_smem[j][d]);
            }

            float m_old = m_reg;
            float m_new = fmaxf(m_old, s);
            float exp_scale = expf(m_old - m_new);
            float exp_s     = expf(s - m_new);
            l_reg = l_reg * exp_scale + exp_s;

            #pragma unroll
            for (int d = 0; d < D; ++d) {
                o_acc[i][d] = o_acc[i][d] * exp_scale + exp_s * static_cast<float>(v_smem[j][d]);
            }
            m_reg = m_new;
        }
        __syncthreads(); // before overwriting K/V smem
    }

    // Write O and LSE
    {
        int i = tid;
        float lse_val = m_reg + logf(l_reg);
        lse[((b * H_q + h_q) * N + row_start + i)] = lse_val;
        #pragma unroll
        for (int d = 0; d < D; ++d) {
            float val = o_acc[i][d] / l_reg;
            o[((b * H_q + h_q) * N + row_start + i) * D + d] = to_bf16(val);
        }
    }
}

//------------------------------------------------------------------------------
// D-vector kernel: D_i = sum_d dout_i,d * o_i,d
//------------------------------------------------------------------------------
__global__ void compute_d_kernel(
    const scalar_t* __restrict__ o,
    const scalar_t* __restrict__ dout,
    float* __restrict__ D_vec,
    int B, int H_q, int N, int Ddim)
{
    int b = blockIdx.x;
    int h_q = blockIdx.y;
    int tile_row = blockIdx.z;
    int row_start = tile_row * Br_fwd;
    int tid = threadIdx.x;

    __shared__ scalar_t o_smem [Br_fwd][D + D_PAD_BF16];
    __shared__ scalar_t do_smem[Br_fwd][D + D_PAD_BF16];

    for (int idx = tid; idx < Br_fwd * Ddim; idx += blockDim.x) {
        int r = idx / Ddim;
        int d = idx % Ddim;
        int g = ((b * H_q + h_q) * N + row_start + r) * Ddim + d;
        o_smem [r][d] = o[g];
        do_smem[r][d] = dout[g];
    }
    __syncthreads();

    int i = tid;
    float acc = 0.0f;
    #pragma unroll
    for (int d = 0; d < Ddim; ++d) {
        acc += static_cast<float>(do_smem[i][d]) * static_cast<float>(o_smem[i][d]);
    }
    D_vec[((b * H_q + h_q) * N + row_start + i)] = acc;
}

//------------------------------------------------------------------------------
// dQ kernel
//------------------------------------------------------------------------------
__global__ void bwd_dq_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ dout,
    const float* __restrict__ lse,
    const float* __restrict__ D_vec,
    scalar_t* __restrict__ dq,
    int B, int H_q, int H_kv, float scale)
{
    int b = blockIdx.x;
    int h_q = blockIdx.y;
    int tile_row = blockIdx.z;
    int h_k = h_q / (H_q / H_kv);
    int row_start = tile_row * Br_dq;
    int tid = threadIdx.x;

    __shared__ scalar_t q_smem [Br_dq][D + D_PAD_BF16];
    __shared__ scalar_t do_smem[Br_dq][D + D_PAD_BF16];
    __shared__ scalar_t k_smem [Bc_dq][D + D_PAD_BF16];
    __shared__ scalar_t v_smem [Bc_dq][D + D_PAD_BF16];
    __shared__ float    dq_acc [Br_dq][D + D_PAD_FLOAT];

    // Load Q and dO tiles
    for (int idx = tid; idx < Br_dq * D; idx += blockDim.x) {
        int r = idx / D;
        int d = idx % D;
        int g = ((b * H_q + h_q) * N + row_start + r) * D + d;
        q_smem [r][d] = q[g];
        do_smem[r][d] = dout[g];
    }
    for (int idx = tid; idx < Br_dq * (D + D_PAD_FLOAT); idx += blockDim.x) {
        int r = idx / (D + D_PAD_FLOAT);
        int c = idx % (D + D_PAD_FLOAT);
        if (r < Br_dq && c < D + D_PAD_FLOAT) dq_acc[r][c] = 0.0f;
    }
    __syncthreads();

    const int num_col_tiles = N / Bc_dq;
    for (int tile_col = 0; tile_col < num_col_tiles; ++tile_col) {
        int col_start = tile_col * Bc_dq;
        for (int idx = tid; idx < Bc_dq * D; idx += blockDim.x) {
            int c = idx / D;
            int d = idx % D;
            int kv = ((b * H_kv + h_k) * N + col_start + c) * D + d;
            k_smem[c][d] = k[kv];
            v_smem[c][d] = v[kv];
        }
        __syncthreads();

        int i = tid;
        float lse_i = lse[((b * H_q + h_q) * N + row_start + i)];
        float D_i   = D_vec[((b * H_q + h_q) * N + row_start + i)];

        for (int j = 0; j < Bc_dq; ++j) {
            float s  = 0.0f;
            float dp = 0.0f;
            #pragma unroll
            for (int d = 0; d < D; ++d) {
                s  += static_cast<float>(q_smem[i][d]) * static_cast<float>(k_smem[j][d]);
                dp += static_cast<float>(do_smem[i][d]) * static_cast<float>(v_smem[j][d]);
            }
            float p  = expf(s * scale - lse_i);
            float ds = (dp - D_i) * p * scale;
            #pragma unroll
            for (int d = 0; d < D; ++d) {
                dq_acc[i][d] += ds * static_cast<float>(k_smem[j][d]);
            }
        }
        __syncthreads();
    }

    // Write dQ
    {
        int i = tid;
        #pragma unroll
        for (int d = 0; d < D; ++d) {
            float val = dq_acc[i][d];
            dq[((b * H_q + h_q) * N + row_start + i) * D + d] = to_bf16(val);
        }
    }
}

//------------------------------------------------------------------------------
// dK / dV kernel (outputs to expanded per-q-head buffers)
//------------------------------------------------------------------------------
__global__ void bwd_dkdv_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ dout,
    const float* __restrict__ lse,
    const float* __restrict__ D_vec,
    float* __restrict__ dk_exp,
    float* __restrict__ dv_exp,
    int B, int H_q, int H_kv, float scale)
{
    int b = blockIdx.x;
    int h_q = blockIdx.y;
    int tile_col = blockIdx.z;
    int h_k = h_q / (H_q / H_kv);
    int col_start = tile_col * Bc_dkdv;
    int tid = threadIdx.x;

    __shared__ scalar_t k_smem [Bc_dkdv][D + D_PAD_BF16];
    __shared__ scalar_t v_smem [Bc_dkdv][D + D_PAD_BF16];
    __shared__ scalar_t q_smem [Br_dkdv][D + D_PAD_BF16];
    __shared__ scalar_t do_smem[Br_dkdv][D + D_PAD_BF16];
    __shared__ float    dk_acc [Bc_dkdv][D + D_PAD_FLOAT];
    __shared__ float    dv_acc [Bc_dkdv][D + D_PAD_FLOAT];

    // Load K/V tile
    for (int idx = tid; idx < Bc_dkdv * D; idx += blockDim.x) {
        int c = idx / D;
        int d = idx % D;
        int kv = ((b * H_kv + h_k) * N + col_start + c) * D + d;
        k_smem[c][d] = k[kv];
        v_smem[c][d] = v[kv];
    }
    for (int idx = tid; idx < Bc_dkdv * (D + D_PAD_FLOAT); idx += blockDim.x) {
        int c = idx / (D + D_PAD_FLOAT);
        int d = idx % (D + D_PAD_FLOAT);
        if (c < Bc_dkdv) {
            dk_acc[c][d] = 0.0f;
            dv_acc[c][d] = 0.0f;
        }
    }
    __syncthreads();

    const int num_row_tiles = N / Br_dkdv;
    for (int tile_row = 0; tile_row < num_row_tiles; ++tile_row) {
        int row_start = tile_row * Br_dkdv;
        // Load Q and dO tiles
        for (int idx = tid; idx < Br_dkdv * D; idx += blockDim.x) {
            int r = idx / D;
            int d = idx % D;
            int g = ((b * H_q + h_q) * N + row_start + r) * D + d;
            q_smem [r][d] = q[g];
            do_smem[r][d] = dout[g];
        }
        __syncthreads();

        for (int i = 0; i < Br_dkdv; ++i) {
            float lse_i = lse[((b * H_q + h_q) * N + row_start + i)];
            float D_i   = D_vec[((b * H_q + h_q) * N + row_start + i)];

            __shared__ float ds_col[Bc_dkdv + 4];
            __shared__ float p_col [Bc_dkdv + 4];

            if (tid < Bc_dkdv) {
                int j = tid;
                float s  = 0.0f;
                float dp = 0.0f;
                #pragma unroll
                for (int d = 0; d < D; ++d) {
                    s  += static_cast<float>(q_smem[i][d]) * static_cast<float>(k_smem[j][d]);
                    dp += static_cast<float>(do_smem[i][d]) * static_cast<float>(v_smem[j][d]);
                }
                float p  = expf(s * scale - lse_i);
                float ds = (dp - D_i) * p * scale;
                ds_col[j] = ds;
                p_col [j] = p;
            }
            __syncthreads();

            for (int idx = tid; idx < Bc_dkdv * D; idx += blockDim.x) {
                int j = idx / D;
                int d = idx % D;
                float qv  = static_cast<float>(q_smem[i][d]);
                float dov = static_cast<float>(do_smem[i][d]);
                dk_acc[j][d] += ds_col[j] * qv;
                dv_acc[j][d] += p_col[j] * dov;
            }
            __syncthreads();
        }
    }

    // Write expanded dK / dV
    for (int idx = tid; idx < Bc_dkdv * D; idx += blockDim.x) {
        int c = idx / D;
        int d = idx % D;
        int out = ((b * H_q + h_q) * N + col_start + c) * D + d;
        dk_exp[out] = dk_acc[c][d];
        dv_exp[out] = dv_acc[c][d];
    }
}

//------------------------------------------------------------------------------
// GQA reduction kernel: sum over query-head group
//------------------------------------------------------------------------------
__global__ void reduce_gqa_kernel(
    const float* __restrict__ dk_exp,
    const float* __restrict__ dv_exp,
    scalar_t* __restrict__ dk_out,
    scalar_t* __restrict__ dv_out,
    int B, int H_kv, int N, int Ddim, int group)
{
    int b = blockIdx.x;
    int h_kv = blockIdx.y;
    int n = blockIdx.z;
    int d = threadIdx.x;
    if (d >= Ddim) return;

    float dk_sum = 0.0f;
    float dv_sum = 0.0f;
    for (int g = 0; g < group; ++g) {
        int h_q = h_kv * group + g;
        int idx = ((b * H_kv * group + h_q) * N + n) * Ddim + d;
        dk_sum += dk_exp[idx];
        dv_sum += dv_exp[idx];
    }
    int out_idx = ((b * H_kv + h_kv) * N + n) * Ddim + d;
    dk_out[out_idx] = to_bf16(dk_sum);
    dv_out[out_idx] = to_bf16(dv_sum);
}
