from torch.utils.cpp_extension import load_inline

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <c10/cuda/CUDAGuard.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

__inline__ __device__ float warp_sum(float v) {
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) v += __shfl_down_sync(0xffffffff, v, off);
    return v;
}

__inline__ __device__ float warp_max(float v) {
    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) v = fmaxf(v, __shfl_down_sync(0xffffffff, v, off));
    return v;
}

// Fixed-shape grouped-query attention kernel for H100.
// q: [B, 32, S, 128], k/v: [B, 8, S, 128], out: [B, 32, S, 128]
// One warp computes one query row. Four warps per CTA, which gives high occupancy
// and avoids CTA-wide reductions in the score loop.
__global__ __launch_bounds__(128, 2)
void gqa_attn_fwd_kernel(const half* __restrict__ q,
                         const half* __restrict__ k,
                         const half* __restrict__ v,
                         half* __restrict__ out,
                         int B, int S, float scale) {
    constexpr int H = 32;
    constexpr int HKV = 8;
    constexpr int D = 128;
    constexpr int WARPS_PER_BLOCK = 4;

    const int tid = threadIdx.x;
    const int lane = tid & 31;
    const int warp = tid >> 5;
    const int bh = blockIdx.x;
    const int q_base = (blockIdx.y * WARPS_PER_BLOCK) + warp;
    if (q_base >= S) return;

    const int b = bh / H;
    const int h = bh - b * H;
    if (b >= B) return;
    const int kvh = h >> 2;  // 32 query heads / 8 KV heads = group size 4.

    extern __shared__ float smem[]; // WARPS_PER_BLOCK * S floats
    float* logits = smem + warp * S;

    const half* qptr = q + (((b * H + h) * S + q_base) * D);
    const half* kbase = k + ((b * HKV + kvh) * S * D);
    const half* vbase = v + ((b * HKV + kvh) * S * D);
    half* optr = out + (((b * H + h) * S + q_base) * D);

    // Keep the query vector resident in registers: four elements per lane.
    float q0 = __half2float(qptr[lane]);
    float q1 = __half2float(qptr[lane + 32]);
    float q2 = __half2float(qptr[lane + 64]);
    float q3 = __half2float(qptr[lane + 96]);

    // QK^T.  Each warp writes a full logit row to shared memory.
    for (int j = 0; j < S; ++j) {

        float acc = q0 * __half2float(kp[lane]);
        acc = fmaf(q1, __half2float(kp[lane + 32]), acc);
        acc = fmaf(q2, __half2float(kp[lane + 64]), acc);
        acc = fmaf(q3, __half2float(kp[lane + 96]), acc);
        acc = warp_sum(acc);
        if (lane == 0) logits[j] = acc * scale;
    }
    __syncwarp();

    // Online softmax over the row, parallelized across lanes.
    float m = -3.402823466e38f;
    for (int j = lane; j < S; j += 32) m = fmaxf(m, logits[j]);
    m = warp_max(m);
    m = __shfl_sync(0xffffffff, m, 0);

    const float inv_denom = 1.0f / denom;

        o1 = fmaf(w, __half2float(vp[lane + 32]), o1);
        o2 = fmaf(w, __half2float(vp[lane + 64]), o2);
        o3 = fmaf(w, __half2float(vp[lane + 96]), o3);
    }

    optr[lane]      = __float2half_rn(o0);
    optr[lane + 32] = __float2half_rn(o1);
    optr[lane + 64] = __float2half_rn(o2);
    optr[lane + 96] = __float2half_rn(o3);
}

    const at::cuda::OptionalCUDAGuard device_guard(device_of(q));

    const int S = (int)q.size(2);
