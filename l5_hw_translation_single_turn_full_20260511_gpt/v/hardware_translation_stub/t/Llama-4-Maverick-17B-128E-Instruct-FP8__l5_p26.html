import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

BATCH = 96
M = 384
N = 192
K = 384
D1 = 12
ALPHA = 1.0
BETA = 0.0

# CUDA kernel
cuda_source = """
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void gemm_permute_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    const half* __restrict__ C,
    half* __restrict__ D,
    int batch_count,
    int M,
    int N,
    int K,
    float alpha,
    float beta,
    int D1
) {
    int batch_idx = blockIdx.z;
    int m = blockIdx.y * blockDim.y + threadIdx.y;
    int n = blockIdx.x * blockDim.x + threadIdx.x;

    if (m >= M || n >= N) return;

    // Permute A and B
    int permuted_m = m;
    int permuted_n = n;
    int permuted_k = blockIdx.z % D1;
    int original_k = blockIdx.z / D1;
    const half* A_ptr = A + (batch_idx / D1) * M * K + permuted_m * K + original_k;
    const half* B_ptr = B + (batch_idx / D1) * K * N + original_k * N + permuted_n;

    float sum = 0.0f;
    for (int k = 0; k < K; ++k) {
        sum += __hmul(A_ptr[k], B_ptr[k * N]);
    }

    // Apply alpha and beta
    float result = alpha * sum + beta * __float2half(C[batch_idx * M * N + m * N + n]);

    // Store result in D with permutation
    D[(batch_idx / D1) * M * N * D1 + permuted_m * N * D1 + permuted_k * N + permuted_n] = __float2half(result);
}

"""

# Load CUDA kernel
cuda_module = load_inline(
    name='gemm_permute_cuda',
    cuda_sources=[cuda_source],
    extra_cuda_cflags=['-O3', '--use_fast_math'],
    functions=['gemm_permute_kernel']
)

# PyTorch module
class ModelNew(nn.Module):
    def __init__(self, batch_count, K, N, D1, alpha, beta):
        super(ModelNew, self).__init__()
        self.batch_count = batch_count
        self.K = K
        self.N = N
        self.M = M
        self.D1 = D1
        self.alpha = alpha
        self.beta = beta

    def forward(self, A):
        B = torch.randn(self.batch_count, self.K, self.N, dtype=torch.float16).cuda()
        C = torch.randn(self.batch_count, M, N, dtype=torch.float16).cuda()
        D = torch.empty(self.batch_count * M * N, dtype=torch.float16).cuda()

        threads_per_block = (16, 16, 1)
        blocks_per_grid = ((N + 15) // 16, (M + 15) // 16, self.batch_count)

        cuda_module.gemm_permute_kernel[blocks_per_grid, threads_per_block](A, B, C, D, self.batch_count, M, N, K, self.alpha, self.beta, self.D1)
        return D.view(self.batch_count // self.D1, M, self.D1, N)
