__global__ void grouped_gemm_f16_kernel(const float* __restrict__ A,
                                        const half* __restrict__ B,

    const int tile_m = blockIdx.y * 32;

    const int tid = threadIdx.x;

    const int warp_n = warp & 1;    // 0,1,0,1

    extern __shared__ unsigned char smem_raw[];
    half* As = reinterpret_cast<half*>(smem_raw);              // 32 x 16
    half* Bs = As + 32 * 16;                                   // 16 x 32
    float* Cs = reinterpret_cast<float*>(Bs + 16 * 32);        // 4 x 16 x 16

    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::row_major> b_frag;

class ModelNew(nn.Module):
    def __init__(self, groups, k, n, alpha, beta):
        super().__init__()
        self.groups = int(groups)
        self.k = int(k)
        self.n = int(n)
        self.alpha = float(alpha)
        self.beta = float(beta)
        # The original CUTLASS workload is a grouped fp16 GEMM.  Keep B in fp16
        # so the custom kernel can use Hopper tensor cores directly.
        self.B = nn.Parameter(torch.randn(self.groups, self.k, self.n, device="cuda", dtype=torch.float16))

    def load_state_dict(self, state_dict, strict=True, assign=False):
        # Be permissive about the reference module's parameter name.  Copy the
        # first tensor with the grouped-GEMM RHS shape into our fp16 RHS matrix.
        with torch.no_grad():
            copied = False
            for _, v in state_dict.items():
                if torch.is_tensor(v) and tuple(v.shape) == (self.groups, self.k, self.n):
                    self.B.copy_(v.to(device=self.B.device, dtype=self.B.dtype))
                    copied = True
                    break
        return torch.nn.modules.module._IncompatibleKeys([], [] if copied else list(state_dict.keys()))

    def forward(self, x):
        if not x.is_contiguous():
            x = x.contiguous()
        return _ext.grouped_gemm(x, self.B, self.alpha)
