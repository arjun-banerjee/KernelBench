#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>
#include <stdint.h>

static __device__ __forceinline__ half2 h2_mul_scalar(half2 x, float s) {
    return __hmul2(x, __float2half2_rn(s));
}

__global__ __launch_bounds__(256, 4)
void scale_half2_kernel(const half* __restrict__ x, half* __restrict__ y, int n_half2, int n, float scale) {
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = blockDim.x * gridDim.x;
    const half2* __restrict__ x2 = reinterpret_cast<const half2*>(x);
    half2* __restrict__ y2 = reinterpret_cast<half2*>(y);

    if (scale == 1.0f) {
        // Vectorized pure copy path.  Use two half2 values (64 bits) per loop when possible.
        const uint64_t* __restrict__ x64 = reinterpret_cast<const uint64_t*>(x);
        uint64_t* __restrict__ y64 = reinterpret_cast<uint64_t*>(y);
        int n64 = n >> 2;
        for (int i = tid; i < n64; i += stride) {
            y64[i] = x64[i];
        }
        int rem_start = n64 << 1;
        for (int i = rem_start + tid; i < n_half2; i += stride) {
            y2[i] = x2[i];
        }
        if ((n & 1) && tid == 0) y[n - 1] = x[n - 1];
    } else {
        for (int i = tid; i < n_half2; i += stride) {
            y2[i] = h2_mul_scalar(x2[i], scale);
        }
        if ((n & 1) && tid == 0) y[n - 1] = __float2half_rn(__half2float(x[n - 1]) * scale);
    }
}

torch::Tensor ar_forward(torch::Tensor x, double scale_d) {
    TORCH_CHECK(x.is_cuda(), "input must be CUDA");
    TORCH_CHECK(x.scalar_type() == at::ScalarType::Half, "input must be fp16");
    auto y = torch::empty_like(x);
    int64_t n64 = x.numel();
    if (n64 == 0) return y;
    int n = (int)n64;
    int n_half2 = n >> 1;
    const int threads = 256;
    int blocks = (n_half2 + threads - 1) / threads;
    if (blocks < 1) blocks = 1;
    if (blocks > 2048) blocks = 2048;
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    scale_half2_kernel<<<blocks, threads, 0, stream>>>(
        reinterpret_cast<const half*>(x.data_ptr<at::Half>()),
        reinterpret_cast<half*>(y.data_ptr<at::Half>()),
        n_half2, n, (float)scale_d);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return y;
}
