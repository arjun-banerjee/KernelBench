import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <vector>

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif
#ifndef CHECK_CONTIGUOUS
#define CHECK_CONTIGUOUS(x) TORCH_CHECK(x.is_contiguous(), #x " must be contiguous")
#endif

// Hopper-targeted rotary embedding for the fixed evaluation layout:
// q:   [tokens, 32, 128] fp16 contiguous
// k:   [tokens,  8, 128] fp16 contiguous
// cos: [tokens, rot_dim] fp16 contiguous
// sin: [tokens, rot_dim] fp16 contiguous
// Applies GPT-NeoX/LLaMA rotate-half style to the first rot_dim channels.
template<int NUM_Q_HEADS, int NUM_K_HEADS, int HEAD_SIZE, int BLOCK_THREADS>
__global__ __launch_bounds__(BLOCK_THREADS, 4)
void rotary_neox_full_kernel(
    half* __restrict__ q,
    half* __restrict__ k,
    const half* __restrict__ cos,
    const half* __restrict__ sin,
    const int rot_dim) {

    const int token = blockIdx.x;
    const int half_dim = rot_dim >> 1;

    extern __shared__ half smem[];
    half* sh_cos = smem;
    half* sh_sin = smem + rot_dim;

    // Cache one token's cos/sin in shared memory.  All heads for the token
    // reuse these values, so this removes many redundant global loads.
    for (int d = threadIdx.x; d < rot_dim; d += BLOCK_THREADS) {
        sh_cos[d] = cos[token * rot_dim + d];
        sh_sin[d] = sin[token * rot_dim + d];
    }
    __syncthreads();

    const int q_pairs = NUM_Q_HEADS * half_dim;
    const int total_pairs = (NUM_Q_HEADS + NUM_K_HEADS) * half_dim;

    for (int idx = threadIdx.x; idx < total_pairs; idx += BLOCK_THREADS) {
        if (idx < q_pairs) {
            const int h = idx / half_dim;
            const int off = idx - h * half_dim;
            half* __restrict__ base = q + ((token * NUM_Q_HEADS + h) * HEAD_SIZE);

            const float x = __half2float(base[off]);
            const float y = __half2float(base[off + half_dim]);
            const float c0 = __half2float(sh_cos[off]);
            const float s0 = __half2float(sh_sin[off]);
            const float c1 = __half2float(sh_cos[off + half_dim]);
            const float s1 = __half2float(sh_sin[off + half_dim]);

            base[off]            = __float2half_rn(fmaf(-y, s0, x * c0));
            base[off + half_dim] = __float2half_rn(fmaf( x, s1, y * c1));
        } else {
            const int j = idx - q_pairs;
            const int h = j / half_dim;
            const int off = j - h * half_dim;
            half* __restrict__ base = k + ((token * NUM_K_HEADS + h) * HEAD_SIZE);

            const float x = __half2float(base[off]);
            const float y = __half2float(base[off + half_dim]);
            const float c0 = __half2float(sh_cos[off]);
            const float s0 = __half2float(sh_sin[off]);
            const float c1 = __half2float(sh_cos[off + half_dim]);
            const float s1 = __half2float(sh_sin[off + half_dim]);

            base[off]            = __float2half_rn(fmaf(-y, s0, x * c0));
            base[off + half_dim] = __float2half_rn(fmaf( x, s1, y * c1));
        }
    }
}

std::vector<torch::Tensor> rotary_forward(torch::Tensor q,
                                          torch::Tensor k,
                                          torch::Tensor cos,
                                          torch::Tensor sin,
                                          int64_t rot_dim_i64) {
    CHECK_CUDA(q); CHECK_CUDA(k); CHECK_CUDA(cos); CHECK_CUDA(sin);
    CHECK_CONTIGUOUS(q); CHECK_CONTIGUOUS(k); CHECK_CONTIGUOUS(cos); CHECK_CONTIGUOUS(sin);
    TORCH_CHECK(q.scalar_type() == at::kHalf, "q must be fp16");
    TORCH_CHECK(k.scalar_type() == at::kHalf, "k must be fp16");
    TORCH_CHECK(cos.scalar_type() == at::kHalf, "cos must be fp16");
    TORCH_CHECK(sin.scalar_type() == at::kHalf, "sin must be fp16");

    const int tokens = (int)q.size(0);
    const int rot_dim = (int)rot_dim_i64;

    constexpr int NUM_Q_HEADS = 32;
    constexpr int NUM_K_HEADS = 8;
    constexpr int HEAD_SIZE = 128;
    constexpr int THREADS = 512;

    const size_t shmem = (size_t)(2 * rot_dim) * sizeof(half);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    rotary_neox_full_kernel<NUM_Q_HEADS, NUM_K_HEADS, HEAD_SIZE, THREADS>
        <<<tokens, THREADS, shmem, stream>>>(
            reinterpret_cast<half*>(q.data_ptr<at::Half>()),
            reinterpret_cast<half*>(k.data_ptr<at::Half>()),
            reinterpret_cast<const half*>(cos.data_ptr<at::Half>()),
            reinterpret_cast<const half*>(sin.data_ptr<at::Half>()),
            rot_dim);

    return {q, k};
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("rotary_forward", &rotary_forward, "H100 optimized fp16 rotary embedding");
}
'''
