__global__ void __launch_bounds__(256) gather_gemm_scatter_kernel(
    const __half* __restrict__ A,
    const __half* __restrict__ B,
    float* __restrict__ D,
    const int* __restrict__ indices,
    int M, int K, int N_idx, int N_full, float alpha, float beta
) {
    const int TILE = 32;

    int by = blockIdx.y;
    int tx = threadIdx.x;
    int ty = threadIdx.y;

    int row = by * TILE + ty;
    int col_idx = bx * TILE + tx;
    int b_col = (col_idx < N_idx) ? indices[col_idx] : -1;
