template <typename scalar_t>
__global__ void __launch_bounds__(256, 4)
dynamic_scaled_int8_quant_kernel(
    const scalar_t* __restrict__ input,
    int8_t* __restrict__ output,
    float* __restrict__ scale_out,
    const int hidden_size) {
    
    const int tid = threadIdx.x;
    const int stride = blockDim.x;
    const int64_t token_idx = blockIdx.x;
    const int lane_id = tid & 31;
    const int warp_id = tid >> 5;
    const int num_warps = stride >> 5;
    
    const scalar_t* __restrict__ row_in = input + token_idx * hidden_size;
    int8_t* __restrict__ row_out = output + token_idx * hidden_size;
    
    float thread_max = 0.0f;
    const int vec_size = 8;
    const int aligned_end = (hidden_size / vec_size) * vec_size;
    
    for (int i = tid * vec_size; i < aligned_end; i += stride * vec_size) {
        const uint4* vec_ptr = reinterpret_cast<const uint4*>(row_in + i);
        uint4 vec_data = __ldg(vec_ptr);
        const half* h_ptr = reinterpret_cast<const half*>(&vec_data);
        
        #pragma unroll
        for (int j = 0; j < 8; j++) {
            float v = __half2float(h_ptr[j]);
            thread_max = fmaxf(thread_max, fabsf(v));
        }
    }
    
    for (int i = aligned_end + tid; i < hidden_size; i += stride) {
        float v = static_cast<float>(row_in[i]);
        thread_max = fmaxf(thread_max, fabsf(v));
    }
    
    float warp_max = warp_reduce_max(thread_max);
    __shared__ float warp_maxes[8];
    
    if (lane_id == 0) {
        warp_maxes[warp_id] = warp_max;
    }
    __syncthreads();
    
    float block_max = 0.0f;
    if (warp_id == 0) {
        block_max = (lane_id < num_warps) ? warp_maxes[lane_id] : 0.0f;
        block_max = warp_reduce_max(block_max);
        
        if (lane_id == 0) {
            scale_out[blockIdx.x] = block_max / 127.0f;
            warp_maxes[0] = block_max;
        }
    }
    __syncthreads();
    
    float absmax = warp_maxes[0];
    float inv_s = (absmax == 0.0f) ? 0.0f : 127.0f / absmax;
    
    for (int i = tid * vec_size; i < aligned_end; i += stride * vec_size) {
        const uint4* vec_ptr = reinterpret_cast<const uint4*>(row_in + i);
        uint4 vec_data = __ldg(vec_ptr);
        const half* h_ptr = reinterpret_cast<const half*>(&vec_data);
        
        #pragma unroll
        for (int j = 0; j < 8; j++) {
            float v = __half2float(h_ptr[j]);
            row_out[i + j] = float_to_int8_rn(__fmul_rn(v, inv_s));
        }
    }
    
    for (int i = aligned_end + tid; i < hidden_size; i += stride) {
        float v = static_cast<float>(row_in[i]);
        row_out[i] = float_to_int8_rn(__fmul_rn(v, inv_s));
    }
}
