        self.register_buffer('qweight', torch.zeros(K, N // self.pack_factor, dtype=torch.int32))
        self.register_buffer('scales', torch.zeros(self.num_groups, N, dtype=torch.float16))
        self.register_buffer('zeros', torch.zeros(self.num_groups, N // self.pack_factor, dtype=torch.int32))
        self.register_buffer('bias', torch.zeros(N, dtype=torch.float16))
        self.register_buffer('workspace', torch.zeros(256, dtype=torch.int32))
        
        self._quantize_weights()
    
    def _quantize_weights(self):
        """Quantize FP16 weights to packed INT4 with group-wise scales."""
        K, N = self.weight_fp16.shape
        
        # Group-wise quantization
        weight_groups = self.weight_fp16.view(self.num_groups, self.GROUP_SIZE, N)
        
        # Compute scales (max abs value per group)
        scales = weight_groups.abs().amax(dim=1).clamp(min=1e-8)
        
        # Quantize to INT4 [-8, 7]
        weight_scaled = weight_groups / scales.unsqueeze(1)
        weight_int = torch.round(weight_scaled * 8).clamp(-8, 7).to(torch.int8)
        
        # Pack 8 INT4 into int32
        weight_packed = torch.zeros(K, N // 8, dtype=torch.int32)
        for i in range(8):
            vals = (weight_int[:, i::8] & 0xF).to(torch.int32)
            weight_packed |= vals << (i * 4)
        
        self.qweight.copy_(weight_packed)
        self.scales.copy_((scales / 8).to(torch.float16))
    
    def forward(self, x):
        M, K = x.shape
        N = self.N
        
        if marlin_h100_module is not None:
            c = marlin_h100_module.marlin_gemm_h100(
                x.cuda(), None,
                self.qweight.cuda(),
                None,
                self.scales.cuda(),
                None, None, None,
                self.workspace.cuda(),
                0, M, N, K,
                True, False, False, False
            )
            return c
        else:
            return torch.matmul(x.cuda(), self.weight_fp16.cuda())
