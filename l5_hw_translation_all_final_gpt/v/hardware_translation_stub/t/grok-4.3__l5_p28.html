__global__ void gemm_fp16(const at::Half* __restrict__ A, const at::Half* __restrict__ B, at::Half* __restrict__ C, int M, int N, int K) {
    const int TILE = 32;
    __shared__ at::Half As[TILE][TILE];
    __shared__ at::Half Bs[TILE][TILE];
    int row = blockIdx.y * TILE + threadIdx.y;
    int col = blockIdx.x * TILE + threadIdx.x;
    float sum = 0.0f;
    for (int t = 0; t < (K + TILE-1)/TILE; t++) {
        int kA = t*TILE + threadIdx.x;
        As[threadIdx.y][threadIdx.x] = (row < M && kA < K) ? A[row*K + kA] : __float2half(0.0f);
        int kB = t*TILE + threadIdx.y;
        Bs[threadIdx.y][threadIdx.x] = (kB < K && col < N) ? B[kB*N + col] : __float2half(0.0f);
        __syncthreads();
        for (int k=0; k<TILE; k++) {
            sum += __half2float(As[threadIdx.y][k]) * __half2float(Bs[k][threadIdx.x]);
        }
        __syncthreads();
    }
    if (row < M && col < N) C[row*N + col] = __float2half(sum);
}

torch::Tensor custom_gemm(torch::Tensor A, torch::Tensor B) {
    int M = A.size(0);
    int K = A.size(1);
    int N = B.size(1);
    auto Af = A.to(torch::kFloat16);
    auto Bf = B.to(torch::kFloat16);

}
