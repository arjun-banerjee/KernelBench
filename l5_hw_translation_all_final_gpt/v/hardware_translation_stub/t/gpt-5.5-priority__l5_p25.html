    for (int k0 = 0; k0 < 256; k0 += 16) {
        As[ty][tx] = Ag[row * 256 + k0 + tx];
        Bs[ty][tx] = Bg[(k0 + ty) * 256 + col];
        __syncthreads();
        #pragma unroll
        for (int k = 0; k < 16; ++k) {
            acc = fmaf(As[ty][k], Bs[k][tx], acc);
        }
        __syncthreads();
    }
    C[(g * 256 + row) * 256 + col] = acc * alpha;
}

__global__ void bmm256_halfout_kernel(const half* __restrict__ A,
                                      const half* __restrict__ B,
                                      half* __restrict__ C,
                                      float alpha) {
    __shared__ half As[16][16];
    __shared__ half Bs[16][16];
    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int row = blockIdx.y * 16 + ty;
    int col = blockIdx.x * 16 + tx;
    int g   = blockIdx.z;

    const half* Ag = A + g * 65536;
    const half* Bg = B + g * 65536;
    float acc = 0.0f;

    #pragma unroll
    for (int k0 = 0; k0 < 256; k0 += 16) {
        As[ty][tx] = Ag[row * 256 + k0 + tx];
        Bs[ty][tx] = Bg[(k0 + ty) * 256 + col];
        __syncthreads();
        #pragma unroll
        for (int k = 0; k < 16; ++k) {
            acc = fmaf(__half2float(As[ty][k]), __half2float(Bs[k][tx]), acc);
        }
        __syncthreads();
    }
    C[(g * 256 + row) * 256 + col] = __float2half_rn(acc * alpha);
}

torch::Tensor bmm_forward(torch::Tensor A, torch::Tensor B, double alpha) {
    CHECK_CUDA(A);
    CHECK_CUDA(B);
    const int G = (int)A.size(0);
    auto C = torch::empty({A.size(0), A.size(1), B.size(2)}, A.options());
    dim3 block(16, 16);
    dim3 grid(16, 16, G);
    if (A.scalar_type() == torch::kFloat32 && B.scalar_type() == torch::kFloat32) {
        bmm256_f32_kernel<<<grid, block>>>(A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(), (float)alpha);
    } else {
        bmm256_halfout_kernel<<<grid, block>>>((const half*)A.data_ptr<at::Half>(),
                                               (const half*)B.data_ptr<at::Half>(),
                                               (half*)C.data_ptr<at::Half>(),
                                               (float)alpha);
    }
    return C;
}
