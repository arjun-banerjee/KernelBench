__global__ void h100_linear_kernel(const half* __restrict__ A,
                                   const half* __restrict__ W,
                                   half* __restrict__ C) {
    // Computes C[16,4096] = A[16,4096] @ W[4096,4096].
    // One warp produces one 16x16 output tile.  W is row-major KxN.

    constexpr int K = 4096;
    constexpr int N = 4096;
    const int n0 = blockIdx.x * 16;

    __shared__ float tmp[256];
    wmma::fragment<wmma::matrix_a, 16,16,16, half, wmma::row_major> af;
    wmma::fragment<wmma::matrix_b, 16,16,16, half, wmma::row_major> bf;
    wmma::fragment<wmma::accumulator, 16,16,16, float> acc;
    wmma::fill_fragment(acc, 0.0f);

    #pragma unroll 4
    for (int k0 = 0; k0 < K; k0 += 16) {
        wmma::load_matrix_sync(af, A + k0, K);
        wmma::load_matrix_sync(bf, W + k0 * N + n0, N);
        wmma::mma_sync(acc, af, bf, acc);
