torch::Tensor flash_attention_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    int num_heads,
    int num_kv_heads,
    int head_size) {

    const int batch_size = q.size(0);
    const int seq_len = q.size(2);
    const float scale = 1.0f / sqrtf(static_cast<float>(head_size));

    auto out = torch::zeros_like(q);

    constexpr int BLOCK_SIZE = 64;
    dim3 grid(num_heads, batch_size);
    dim3 block(256);

    switch (head_size) {

            flash_attention_fp16_kernel<128, BLOCK_SIZE><<<grid, block>>>(
                reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
                reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
                reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
                reinterpret_cast<half*>(out.data_ptr<at::Half>()),
                batch_size, num_heads, num_kv_heads, seq_len, scale);
            break;
        default:
            AT_ASSERT(false);
    }
    cudaDeviceSynchronize();
    return out;
}
