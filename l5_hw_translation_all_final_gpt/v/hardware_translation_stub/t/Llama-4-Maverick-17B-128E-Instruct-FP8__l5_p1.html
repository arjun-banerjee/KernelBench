import torch
from torch.utils.cpp_extension import load_inline

# Define the CUDA kernel code with modifications for H100 (Hopper architecture)
cuda_source = '''
#include <cuda_runtime.h>
#include <cuda_fp16.h>

#define WARP_SIZE 32
#define NUM_WARPS 4
#define BLOCK_SIZE 16
#define NUM_THREADS (WARP_SIZE * NUM_WARPS)
#define HEAD_SIZE 128

__device__ __forceinline__ float warp_reduce_sum(float sum) {
    #pragma unroll
    for (int offset = WARP_SIZE / 2; offset > 0; offset /= 2) {
        sum += __shfl_xor_sync(0xffffffff, sum, offset);
    }
    return sum;
}

__global__ void paged_attention_v1_kernel(half* out_ptr, half* query_ptr, half* key_cache_ptr, half* value_cache_ptr,
                                          int num_kv_heads, float scale, int* block_tables_ptr, int* seq_lens_ptr,
                                          int max_num_blocks_per_seq, float* alibi_slopes_ptr, int q_stride,
                                          int kv_block_stride, int kv_head_stride, float* k_scale_ptr, float* v_scale_ptr,
                                          int tp_rank, int blocksparse_local_blocks, int blocksparse_vert_stride,
                                          int blocksparse_block_size, int blocksparse_head_sliding_step) {
    int head_idx = blockIdx.x;
    int seq_idx = blockIdx.y;
    int warp_idx = threadIdx.x / WARP_SIZE;
    int lane_idx = threadIdx.x % WARP_SIZE;

    // Load query
    half* query = query_ptr + seq_idx * q_stride + head_idx * HEAD_SIZE;
    half query_reg[HEAD_SIZE / WARP_SIZE];
    for (int i = 0; i < HEAD_SIZE / WARP_SIZE; i++) {
        query_reg[i] = query[lane_idx + i * WARP_SIZE];
    }

    // Shared memory for logits and outputs
    __shared__ float logits[BLOCK_SIZE];
    __shared__ float outputs[NUM_WARPS / 2 * HEAD_SIZE];

    // Initialize logits and outputs
    if (threadIdx.x < BLOCK_SIZE) {
        logits[threadIdx.x] = -INFINITY;
    }
    if (threadIdx.x < NUM_WARPS / 2 * HEAD_SIZE) {
        outputs[threadIdx.x] = 0.0f;
    }
    __syncthreads();

    // Compute attention
    for (int block_idx = 0; block_idx < max_num_blocks_per_seq; block_idx++) {
        int block_table = block_tables_ptr[seq_idx * max_num_blocks_per_seq + block_idx];
        if (block_table == -1) break;

        half* key_cache = key_cache_ptr + block_table * kv_block_stride + (head_idx % num_kv_heads) * kv_head_stride;
        half* value_cache = value_cache_ptr + block_table * kv_block_stride + (head_idx % num_kv_heads) * kv_head_stride;

        // Load key and value
        half key_reg[HEAD_SIZE / WARP_SIZE];
        half value_reg[HEAD_SIZE / WARP_SIZE];
        for (int i = 0; i < HEAD_SIZE / WARP_SIZE; i++) {
            key_reg[i] = key_cache[lane_idx + i * WARP_SIZE];
            value_reg[i] = value_cache[lane_idx + i * WARP_SIZE];
        }

        // Compute QK^T
        float qk = 0.0f;
        for (int i = 0; i < HEAD_SIZE / WARP_SIZE; i++) {
            qk += __hmul(query_reg[i], key_reg[i]);
        }
        qk = warp_reduce_sum(qk) * scale;

        // Store logits
        if (lane_idx == 0) {
            logits[block_idx] = qk;
        }
        __syncthreads();

        // Compute attention output
        for (int i = warp_idx; i < BLOCK_SIZE; i += NUM_WARPS / 2) {
            float logit = logits[i];
            float output = 0.0f;
            for (int j = 0; j < HEAD_SIZE / WARP_SIZE; j++) {
                output += __hmul(value_reg[j], logit);
            }
            outputs[threadIdx.x] += output;
        }
        __syncthreads();
    }

    // Store output
    half* out = out_ptr + seq_idx * q_stride + head_idx * HEAD_SIZE;
    for (int i = 0; i < HEAD_SIZE / WARP_SIZE; i++) {
        out[lane_idx + i * WARP_SIZE] = __float2half(outputs[lane_idx + i * WARP_SIZE]);
