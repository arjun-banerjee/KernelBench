import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda_fp16.h>

__global__ void batched_gemm_fp16_kernel(
    const __half* __restrict__ A,
    const __half* __restrict__ B,
    float* __restrict__ C,
    int batch, int M, int K, int N,
    float alpha, float beta) {

    int b = blockIdx.z;
    int row = blockIdx.y * blockDim.y + threadIdx.y;
    int col = blockIdx.x * blockDim.x + threadIdx.x;

    if (row < M && col < N && b < batch) {
        float sum = 0.0f;
        for (int k = 0; k < K; ++k) {
            float a = __half2float(A[b * M * K + row * K + k]);
            float bb = __half2float(B[b * K * N + k * N + col]);
            sum += a * bb;
        }
        C[b * M * N + row * N + col] = alpha * sum + beta * C[b * M * N + row * N + col];
    }
}

torch::Tensor batched_gemm_fp16(torch::Tensor A, torch::Tensor B, float alpha, float beta) {
