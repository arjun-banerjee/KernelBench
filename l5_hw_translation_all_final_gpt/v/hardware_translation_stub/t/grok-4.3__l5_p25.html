import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_fp16.h>
#include <cuda_runtime.h>

__global__ void batched_gemm_fp16_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    half* __restrict__ C,
    int G, int M, int K, int N,
    float alpha, float beta) {
    
    const int g = blockIdx.z;
    const int m = blockIdx.y * blockDim.y + threadIdx.y;
    const int n = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (g >= G || m >= M || n >= N) return;
    
    const half* A_g = A + g * M * K;
    const half* B_g = B + g * K * N;
    half* C_g = C + g * M * N;
    
    float sum = 0.0f;
    for (int k = 0; k < K; ++k) {
        sum += __half2float(A_g[m * K + k]) * __half2float(B_g[k * N + n]);
    }
    
    float c_val = __half2float(C_g[m * N + n]);
    C_g[m * N + n] = __float2half(alpha * sum + beta * c_val);
}

torch::Tensor batched_gemm_cuda(torch::Tensor A, torch::Tensor B, float alpha, float beta) {
    auto G = A.size(0);
    auto M = A.size(1);
    auto K = A.size(2);
    auto N = B.size(2);
    
    auto A16 = A.to(torch::kHalf).contiguous();
    auto B16 = B.to(torch::kHalf).contiguous();
    
    auto C = torch::zeros({G, M, N}, A.options().dtype(torch::kHalf));
    
    dim3 block(16, 16);
    dim3 grid((N + 15) / 16, (M + 15) / 16, G);
    
    batched_gemm_fp16_kernel<<<grid, block>>>(
        reinterpret_cast<half*>(A16.data_ptr<at::Half>()),
        reinterpret_cast<half*>(B16.data_ptr<at::Half>()),
        reinterpret_cast<half*>(C.data_ptr<at::Half>()),
        G, M, K, N, alpha, beta
    );
    
    cudaDeviceSynchronize();
    return C.to(torch::kFloat32);
}
