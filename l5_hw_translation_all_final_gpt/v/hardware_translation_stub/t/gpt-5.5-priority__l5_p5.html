import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp = r'''
#include <torch/extension.h>

_cuda = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_fp16.h>

// In-place rotary embedding for the fixed benchmark shape:
// q [512,32,128], k [512,8,128], cos/sin [512,64], fp16.
// Uses 4 CTAs per token (10 heads/CTA) to expose more parallelism on H100 while
// keeping each warp's memory accesses contiguous across the 32 rotary lanes.
__global__ __launch_bounds__(128, 4)
void rope_h100_kernel(half* __restrict__ q,
                      half* __restrict__ k,
                      const half* __restrict__ cos,
                      const half* __restrict__ sin) {
    const int token = blockIdx.x;
    const int chunk = blockIdx.y;
    const int tid = threadIdx.x;
    const int hbase = chunk * 10;

    const half* __restrict__ ct = cos + (long)token * 64;
    const half* __restrict__ st = sin + (long)token * 64;

    #pragma unroll
    for (int i = tid; i < 320; i += 128) {
        const int p = i & 31;          // rotary lane within half dimension
        const int h = hbase + (i >> 5); // combined q/k head id [0,40)

        const bool is_q = h < 32;
        const int hh = is_q ? h : (h - 32);
        half* __restrict__ arr = is_q ? q : k;
        const long base = is_q ? (((long)token * 32 + hh) * 128)
                               : (((long)token *  8 + hh) * 128);

        const float x  = __half2float(arr[base + p]);
        const float y  = __half2float(arr[base + p + 32]);
        const float c0 = __half2float(ct[p]);
        const float s0 = __half2float(st[p]);
        const float c1 = __half2float(ct[p + 32]);
        const float s1 = __half2float(st[p + 32]);

        arr[base + p]      = __float2half_rn(x * c0 - y * s0);
        arr[base + p + 32] = __float2half_rn(y * c1 + x * s1);
    }
}

std::vector<torch::Tensor> rope_(torch::Tensor q, torch::Tensor k,
                                 torch::Tensor cos, torch::Tensor sin,
                                 int64_t rot_dim) {
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    dim3 grid(512, 4);
    rope_h100_kernel<<<grid, 128, 0, stream>>>(
        reinterpret_cast<half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(cos.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(sin.data_ptr<at::Half>()));
    return {q, k};
}
'''
