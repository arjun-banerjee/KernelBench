import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Channel-wise gated delta attention recurrence.

    The recurrent state decays independently for each output channel, which
    makes the update closer to channel-gated delta attention variants used in
    fast long-context baselines.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-2)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void gated_delta_attention_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    float* __restrict__ output,
    int batch_size,
    int seq_len,
    int num_heads,
    int head_dim
) {
    int bh = blockIdx.x;
    int b = bh / num_heads;
    int h = bh % num_heads;
    int d = head_dim;

    extern __shared__ float state[];
    
    // Initialize state to 0
    for (int i = threadIdx.x; i < d * d; i += blockDim.x) {
        state[i] = 0.0f;
    }
    __syncthreads();

    for (int t = 0; t < seq_len; t++) {
        // Update state: state[i][j] = gate[b,h,t,j] * state[i][j] + k[b,h,t,i] * v[b,h,t,j]
        for (int idx = threadIdx.x; idx < d * d; idx += blockDim.x) {
            int i = idx / d;
            int j = idx % d;
            float g = gate[((b * seq_len + t) * num_heads + h) * d + j];
            float kv = k[((b * seq_len + t) * num_heads + h) * d + i] * 
                       v[((b * seq_len + t) * num_heads + h) * d + j];
            state[idx] = g * state[idx] + kv;
        }
        __syncthreads();

        // Compute y: y[j] = sum_i q[b,h,t,i] * state[i][j]
        for (int j = threadIdx.x; j < d; j += blockDim.x) {
            float sum = 0.0f;
            for (int i = 0; i < d; i++) {
                sum += q[((b * seq_len + t) * num_heads + h) * d + i] * state[i * d + j];
            }
            output[((b * seq_len + t) * num_heads + h) * d + j] = sum;
        }
        __syncthreads();
    }
}

torch::Tensor gated_delta_attention_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    torch::Tensor gate
) {
    auto batch_size = q.size(0);
    auto seq_len = q.size(1);
    auto num_heads = q.size(2);
    auto head_dim = q.size(3);

    auto output = torch::zeros({batch_size, seq_len, num_heads, head_dim}, q.options());

    int bh = batch_size * num_heads;
    int threads = 256;  // enough for d=24, 576 elements
    size_t shmem = head_dim * head_dim * sizeof(float);

    gated_delta_attention_kernel<<<bh, threads, shmem>>>(
        q.data_ptr<float>(),
        k.data_ptr<float>(),
        v.data_ptr<float>(),
        gate.data_ptr<float>(),
        output.data_ptr<float>(),
        batch_size,
        seq_len,
        num_heads,
        head_dim
    );

}
