    def __init__(self, channels, seq_len, kernel_size, dilation):
        super(ModelNew, self).__init__()
        self.channels = channels
        self.seq_len = seq_len
        self.kernel_size = kernel_size
        self.dilation = dilation
        
        # Initialize parameters (same as reference model)
        self.norm = nn.LayerNorm([channels, seq_len])
        self.conv = nn.Conv1d(channels, channels, kernel_size, 
                              padding=dilation, dilation=dilation)
        
        # Store compiled extension
        self.ext = extension
    
    def forward(self, x):
        batch_size = x.shape[0]
        C = self.channels
        L = self.seq_len
        K = self.kernel_size
        dilation = self.dilation
        num_elements = C * L  # elements per batch item
        
        # Ensure contiguous
        x_contig = x.contiguous()
        x_flat = x_contig.view(-1)
        
        # Allocate output tensors
        gelu_out = torch.empty_like(x_flat)
        out = torch.empty_like(x_flat)
        
        # Get weight/bias as contiguous
        gamma = self.norm.weight.contiguous().view(-1)
        beta = self.norm.bias.contiguous().view(-1)
        weight = self.conv.weight.contiguous()
        bias = self.conv.bias.contiguous()
        
        # LayerNorm + GELU: two-pass fused approach
        threads = 512
        blocks = batch_size
        stats = torch.empty(batch_size, 2, dtype=x.dtype, device=x.device)
        shared_mem = threads * 2 * 4  # 2 floats per thread
        
        # Pass 1: compute statistics
        self.ext.ln_gelu_pass1_kernel(
            x_flat, stats, num_elements,
            grid=(blocks,), block=(threads,), shared_mem=shared_mem
        )
        
        # Pass 2: normalize, affine transform, and GELU
        self.ext.ln_gelu_pass2_kernel(
            x_flat, gelu_out, stats, gamma, beta, num_elements,
            grid=(blocks,), block=(threads,)
        )
        
        # Fused Conv1d + Residual
        total_elements = batch_size * num_elements
        threads_conv = 256
        blocks_conv = (total_elements + threads_conv - 1) // threads_conv
        
        self.ext.conv_residual_kernel(
            gelu_out, out, x_flat, weight, bias,
            C, L, K, dilation,
            grid=(blocks_conv,), block=(threads_conv,)
        )
        
        return out.view(batch_size, C, L)
