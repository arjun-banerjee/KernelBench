import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Fused 1D convolution + GroupNorm for temporal sequences (N, C, T).

    Typical in time-series heads, audio / motor trajectories, and sequence models
    that use strided or padded Conv1d followed by per-channel normalization without
    collapsing the time dimension (GroupNorm over `out_channels` on (N, C_out, T')).

    Reference fusion target: `Conv1d` → `GroupNorm` (no activation), matching a
    common stem or residual branch in temporal backbones.
    """

    def __init__(self, in_channels: int, out_channels: int, num_groups: int, kernel_size: int=3, stride: int=1, padding: int=1, dilation: int=1, eps: float=1e-05, affine: bool=True):
        super().__init__()
        assert out_channels % num_groups == 0
        self.conv = nn.Conv1d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, dilation=dilation, bias=False)
        self.norm = nn.GroupNorm(num_groups=num_groups, num_channels=out_channels, eps=eps, affine=affine)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C_in, T) temporal feature map
        Returns:
            (N, C_out, T') with T' determined by conv stride / padding / kernel
        """
        return self.norm(self.conv(x))

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAException.h>

__global__ void groupnorm_kernel(const float* __restrict__ y,
                                 const float* __restrict__ gamma,
                                 const float* __restrict__ beta,
                                 float* __restrict__ out,
                                 int N, int C, int T, int G,
                                 float eps, bool affine) {
    extern __shared__ float smem[];
    float* ssum = smem;
    float* ssq  = smem + blockDim.x;

    int ng = blockIdx.x;
    int n = ng / G;
    int g = ng - n * G;
    int cpg = C / G;
    int M = cpg * T;
    int tid = threadIdx.x;

    float sum = 0.0f;
    float sq = 0.0f;

    int base_c = g * cpg;
    for (int idx = tid; idx < M; idx += blockDim.x) {
        int cr = idx / T;
        int t = idx - cr * T;
        int c = base_c + cr;
        float v = y[((n * C + c) * T) + t];
        sum += v;
        sq += v * v;
    }

    ssum[tid] = sum;
    ssq[tid] = sq;
    __syncthreads();

    for (int off = blockDim.x >> 1; off > 0; off >>= 1) {
        if (tid < off) {
            ssum[tid] += ssum[tid + off];
            ssq[tid] += ssq[tid + off];
        }
        __syncthreads();
    }

    float mean = ssum[0] / (float)M;
    float var = ssq[0] / (float)M - mean * mean;
    var = var < 0.0f ? 0.0f : var;
    float inv_std = rsqrtf(var + eps);

    for (int idx = tid; idx < M; idx += blockDim.x) {
        int cr = idx / T;
        int t = idx - cr * T;
        int c = base_c + cr;
        int off = ((n * C + c) * T) + t;
        float v = (y[off] - mean) * inv_std;
        if (affine) {
            v = v * gamma[c] + beta[c];
        }
        out[off] = v;
    }
}

torch::Tensor gn_forward_cuda(torch::Tensor y, torch::Tensor gamma, torch::Tensor beta,
                              int64_t num_groups, double eps, bool affine) {
    auto out = torch::empty_like(y);
    const int N = (int)y.size(0);
    const int C = (int)y.size(1);
    const int T = (int)y.size(2);
    const int G = (int)num_groups;

    const int threads = 256;
    const int blocks = N * G;
    const size_t shmem = 2 * threads * sizeof(float);

    groupnorm_kernel<<<blocks, threads, shmem, at::cuda::getCurrentCUDAStream()>>>(
        y.data_ptr<float>(),
        affine ? gamma.data_ptr<float>() : nullptr,
        affine ? beta.data_ptr<float>() : nullptr,
        out.data_ptr<float>(),
        N, C, T, G, (float)eps, affine
    );
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return out;
}
'''
