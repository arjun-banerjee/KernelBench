class Model(nn.Module):

        super().__init__()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        (r, b, h) = x.shape

        parts = [x[i] for i in range(R)]
        gathered = torch.cat(parts, dim=-1)
        return self.proj(gathered)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

R = 4
H = 32
K = 112

_cpp_src = r'''
#include <torch/extension.h>

torch::Tensor vgcl_forward(torch::Tensor x, torch::Tensor weight, torch::Tensor bias);
'''

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAStream.h>

#define R_CONST 4
#define H_CONST 32
#define K_CONST 112
#define N_CONST 128

template<int KVAL>
__global__ void vgcl_kernel(const float* __restrict__ x,
                            const float* __restrict__ w,
                            const float* __restrict__ bias,
                            float* __restrict__ out,
                            int B) {
    int b = blockIdx.x;
    int k = threadIdx.x;
    if (b >= B || k >= KVAL) return;

    const float* wk = w + k * N_CONST;
    float acc = bias[k];

    // x layout is contiguous (R, B, H); virtual concatenation order is r-major.
    const float* x0 = x + b * H_CONST;
    const float* x1 = x + (B + b) * H_CONST;
    const float* x2 = x + (2 * B + b) * H_CONST;
    const float* x3 = x + (3 * B + b) * H_CONST;

    #pragma unroll
    for (int h = 0; h < H_CONST; ++h) acc += x0[h] * wk[h];
    #pragma unroll
    for (int h = 0; h < H_CONST; ++h) acc += x1[h] * wk[H_CONST + h];
    #pragma unroll
    for (int h = 0; h < H_CONST; ++h) acc += x2[h] * wk[2 * H_CONST + h];
    #pragma unroll
    for (int h = 0; h < H_CONST; ++h) acc += x3[h] * wk[3 * H_CONST + h];

    out[b * KVAL + k] = acc;
}

torch::Tensor vgcl_forward(torch::Tensor x, torch::Tensor weight, torch::Tensor bias) {
    const int B = (int)x.size(1);
    auto out = torch::empty({B, K_CONST}, x.options());

    const int threads = 128;
    dim3 grid(B);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    vgcl_kernel<K_CONST><<<grid, threads, 0, stream>>>(
        x.data_ptr<float>(), weight.data_ptr<float>(), bias.data_ptr<float>(), out.data_ptr<float>(), B);
    return out;
}
'''
