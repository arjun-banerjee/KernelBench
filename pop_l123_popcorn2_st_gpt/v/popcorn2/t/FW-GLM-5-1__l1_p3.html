import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Sparse message passing with CSR adjacency.

    For each destination node, aggregate weighted source-node features from the
    incoming edges in its CSR row. This is a core primitive in sparse graph
    neural network layers and graph convolution operators.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, col_idx: torch.Tensor, edge_weight: torch.Tensor, node_feat: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        feat_dim = node_feat.shape[1]
        out = torch.zeros(num_nodes, feat_dim, dtype=node_feat.dtype, device=node_feat.device)
        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end > start:
                src = col_idx[start:end].long()
                weights = edge_weight[start:end].unsqueeze(-1)
                out[dst] = (node_feat[src] * weights).sum(dim=0)
        return out

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template <typename scalar_t>
__global__ void csr_spmm_kernel(
    const int64_t* __restrict__ row_ptr,
    const int64_t* __restrict__ col_idx,
    const scalar_t* __restrict__ edge_weight,
    const scalar_t* __restrict__ node_feat,
    scalar_t* __restrict__ output,
    int num_nodes,
    int feat_dim) {
    
    // Each thread computes one output element: output[dst][fid]
    int dst = blockIdx.x * blockDim.x + threadIdx.x;
    int fid = blockIdx.y * blockDim.y + threadIdx.y;
    
    if (dst >= num_nodes || fid >= feat_dim) return;
    
    int start = row_ptr[dst];
    int end = row_ptr[dst + 1];
    
    scalar_t sum = scalar_t(0);
    
    for (int e = start; e < end; e++) {
        int src = col_idx[e];
        sum += edge_weight[e] * node_feat[src * feat_dim + fid];
    }
    
    output[dst * feat_dim + fid] = sum;
}

torch::Tensor csr_spmm_cuda(
    torch::Tensor row_ptr,
    torch::Tensor col_idx,
    torch::Tensor edge_weight,
    torch::Tensor node_feat) {
    
    int num_nodes = row_ptr.numel() - 1;
    int feat_dim = node_feat.size(1);
    
    auto output = torch::zeros({num_nodes, feat_dim}, 
        torch::dtype(node_feat.scalar_type()).device(node_feat.device()));
    
    dim3 threadsPerBlock(16, 16);
    dim3 numBlocks((num_nodes + threadsPerBlock.x - 1) / threadsPerBlock.x,
                   (feat_dim + threadsPerBlock.y - 1) / threadsPerBlock.y);
    
    AT_DISPATCH_FLOATING_TYPES(node_feat.scalar_type(), "csr_spmm_kernel", ([&] {
        csr_spmm_kernel<scalar_t><<<numBlocks, threadsPerBlock>>>(
            row_ptr.data_ptr<int64_t>(),
            col_idx.data_ptr<int64_t>(),
            edge_weight.data_ptr<scalar_t>(),
            node_feat.data_ptr<scalar_t>(),
            output.data_ptr<scalar_t>(),
            num_nodes,
            feat_dim);
    }));
    
    return output;
}
