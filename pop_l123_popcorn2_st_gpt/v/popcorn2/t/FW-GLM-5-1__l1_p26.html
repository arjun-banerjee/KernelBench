import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Self-normalized importance sampling estimate.
    Given samples from a proposal distribution q, computes importance
    weights w_i = p(x_i)/q(x_i), normalizes them, and returns both
    the weighted estimate of E_p[f(x)] and the effective sample size.

    Target p is a mixture of Gaussians; proposal q is a single Gaussian.
    """

    def __init__(self, dim, num_components=3):
        super().__init__()
        self.dim = dim
        self.num_components = num_components
        self.register_buffer('mix_weights', torch.ones(num_components) / num_components)
        mus = torch.randn(num_components, dim) * 2.0
        self.register_buffer('mix_mus', mus)
        self.register_buffer('mix_stds', torch.ones(num_components, dim))
        self.register_buffer('q_mu', torch.zeros(dim))
        self.register_buffer('q_std', torch.ones(dim) * 3.0)

    def _log_p(self, x):
        """Log density of target mixture."""
        diff = x.unsqueeze(1) - self.mix_mus.unsqueeze(0)
        log_comp = -0.5 * (diff / self.mix_stds.unsqueeze(0)) ** 2 - self.mix_stds.log().unsqueeze(0)
        log_comp = log_comp.sum(-1)
        return torch.logsumexp(log_comp + self.mix_weights.log().unsqueeze(0), dim=-1)

    def _log_q(self, x):
        """Log density of proposal."""
        return (-0.5 * ((x - self.q_mu) / self.q_std) ** 2 - self.q_std.log()).sum(-1)

    def forward(self, samples: torch.Tensor) -> tuple:
        """
        Args:
            samples: (N, dim) – samples drawn from q
        Returns:
            (weighted_mean, ess):
                weighted_mean: (dim,) – importance-weighted mean estimate of E_p[x]
                ess: scalar – effective sample size
        """
        log_w = self._log_p(samples) - self._log_q(samples)
        log_w = log_w - torch.logsumexp(log_w, dim=0)
        w = torch.exp(log_w)
        weighted_mean = (w.unsqueeze(-1) * samples).sum(dim=0)
        ess = 1.0 / (w ** 2).sum()
        return (weighted_mean, ess)

    auto global_sum_exp = torch::zeros({1}, options);
    
    const int threads = 256;
    const int blocks = std::min((num_samples + threads - 1) / threads, 1024);

            num_samples, dim, num_components);
    }));

    AT_DISPATCH_FLOATING_TYPES(samples.scalar_type(), "find_max", ([&] {

            global_sum_exp.data_ptr<scalar_t>(),
            num_samples);
    }));
    
    // Compute log_sum_w
    float log_sum_w = global_max.item<float>() + logf(global_sum_exp.item<float>());
    auto log_sum_w_tensor = torch::tensor(log_sum_w, options);
    
    // Kernel 3: Normalize and compute weighted mean and ESS
    AT_DISPATCH_FLOATING_TYPES(samples.scalar_type(), "normalize_and_reduce", ([&] {

            log_sum_w,

            weighted_mean.data_ptr<scalar_t>(),
            sum_w_sq.data_ptr<scalar_t>(),
            num_samples, dim);
    }));

    return {weighted_mean, torch::tensor(ess, options)};

class ModelNew(nn.Module):
    """
    Self-normalized importance sampling estimate with custom CUDA kernels.
    """
    
    def __init__(self, dim, num_components=3):
        super().__init__()
        self.dim = dim
        self.num_components = num_components
        self.register_buffer('mix_weights', torch.ones(num_components) / num_components)
        mus = torch.randn(num_components, dim) * 2.0
        self.register_buffer('mix_mus', mus)
        self.register_buffer('mix_stds', torch.ones(num_components, dim))
        self.register_buffer('q_mu', torch.zeros(dim))
        self.register_buffer('q_std', torch.ones(dim) * 3.0)
    
    def forward(self, samples: torch.Tensor) -> tuple:
        """
        Args:
            samples: (N, dim) - samples drawn from q
        Returns:
            (weighted_mean, ess):
                weighted_mean: (dim,) - importance-weighted mean estimate
                ess: scalar - effective sample size
        """
        # Precompute log weights and log q_std for kernel efficiency
        mix_log_weights = torch.log(self.mix_weights)
        q_log_std = torch.log(self.q_std)
        
        # Call custom CUDA kernel
        result = importance_sampling_module.importance_sampling_cuda(
            samples,
            self.mix_mus,
            self.mix_stds,
            mix_log_weights,
            self.q_mu,
            q_log_std
        )
        
        return (result[0], result[1])
