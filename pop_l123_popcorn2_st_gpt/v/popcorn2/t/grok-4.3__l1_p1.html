import torch

class Model(nn.Module):
    """
    Performs a depthwise 1D convolution where each input channel is convolved with its own filter.
    Fundamental primitive in Hyena, Mamba, and genomic sequence models (Enformer, HyenaDNA).

    Args:
        channels (int): Number of input/output channels.
        kernel_size (int): Size of the convolution kernel.
        stride (int, optional): Stride of the convolution. Defaults to 1.
        padding (int, optional): Padding applied to the input. Defaults to 0.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """

    def __init__(self, channels: int, kernel_size: int, stride: int=1, padding: int=0, bias: bool=False):
        super(Model, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.conv1d(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void depthwise_conv1d_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias_ptr,
    float* __restrict__ output,
    const int batch_size,
    const int channels,
    const int in_len,
    const int out_len,
    const int kernel_size,
    const int stride,
    const int padding,
    const bool has_bias
) {
    const int total_outputs = batch_size * channels * out_len;
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= total_outputs) return;

    const int b = idx / (channels * out_len);
    const int rem = idx % (channels * out_len);
    const int c = rem / out_len;
    const int o = rem % out_len;

    float sum = 0.0f;
    const float* w_base = weight + c * kernel_size;
    const float* in_base = input + (b * channels + c) * in_len;

    #pragma unroll
    for (int k = 0; k < kernel_size; ++k) {
        int i_pos = o * stride - padding + k;
        if (i_pos >= 0 && i_pos < in_len) {
            sum += in_base[i_pos] * w_base[k];
        }
    }

    if (has_bias) {
        sum += bias_ptr[c];
    }
    output[idx] = sum;
}

torch::Tensor depthwise_conv1d_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias,
    int stride,
    int padding
) {
    const int batch_size = input.size(0);
    const int channels = input.size(1);
    const int in_len = input.size(2);
    const int kernel_size = weight.size(2);
    const int out_len = (in_len + 2 * padding - kernel_size) / stride + 1;
    const bool has_bias = bias.defined() && bias.numel() > 0;

    auto output = torch::empty({batch_size, channels, out_len}, input.options());

    const int total = batch_size * channels * out_len;
    const int threads = 256;
    const int blocks = (total + threads - 1) / threads;

    depthwise_conv1d_kernel<<<blocks, threads>>>(
        input.data_ptr<float>(),
        weight.data_ptr<float>(),
        has_bias ? bias.data_ptr<float>() : nullptr,
        output.data_ptr<float>(),
        batch_size,
        channels,
        in_len,
        out_len,
        kernel_size,
        stride,
        padding,
        has_bias
    );

    return output;
}
