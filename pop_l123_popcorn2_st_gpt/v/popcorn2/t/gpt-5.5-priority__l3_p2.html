import torch
import torch.nn as nn
import torch.nn.functional as F

def _linear_beta_schedule(num_timesteps: int, beta_start: float, beta_end: float) -> torch.Tensor:
    return torch.linspace(beta_start, beta_end, num_timesteps)

class Model(nn.Module):
    """
    Fused **multi-step** diffusion denoising (VP-DDPM-style) in a single ``forward``.

    Instead of one scheduler step per Python call, this reference unrolls
    ``num_fused_steps`` consecutive **deterministic** reverse updates (posterior mean,
    no sampler noise)—a typical fusion target for robotics / world-model diffusion
    heads where latency matters.

    Each inner step: predict ``ε_θ(x, t)`` with a small MLP, then apply the standard
    DDPM mean update from discrete time ``t`` to ``t-1``. Batch elements may start
    from different ``t_start``; inputs must satisfy ``t_start >= num_fused_steps`` so
    indices stay valid.
    """

    def __init__(self, dim: int, num_fused_steps: int, total_timesteps: int=1000):
        super().__init__()
        assert num_fused_steps >= 1
        assert total_timesteps >= num_fused_steps + 1
        self.dim = dim
        self.num_fused_steps = num_fused_steps
        self.total_timesteps = total_timesteps
        self.eps_net = nn.Sequential(nn.Linear(dim + 1, dim * 4), nn.SiLU(), nn.Linear(dim * 4, dim * 4), nn.SiLU(), nn.Linear(dim * 4, dim))
        betas = _linear_beta_schedule(total_timesteps, 0.0001, 0.02)
        alphas = 1.0 - betas
        alphas_cumprod = torch.cumprod(alphas, dim=0)
        self.register_buffer('betas', betas)
        self.register_buffer('alphas', alphas)
        self.register_buffer('alphas_cumprod', alphas_cumprod)
        self.register_buffer('sqrt_recip_alphas', torch.sqrt(1.0 / alphas))
        self.register_buffer('coef_eps', betas / torch.sqrt(1.0 - alphas_cumprod))

    def forward(self, x: torch.Tensor, t_start: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, dim) noisy latent at discrete time ``t_start`` (same semantics as DDPM).
            t_start: (B,) int64 timesteps in ``[num_fused_steps, total_timesteps - 1]``.
        Returns:
            (B, dim) latent after ``num_fused_steps`` fused denoise steps toward 0.
        """
        (B, d) = x.shape
        assert d == self.dim
        assert t_start.shape == (B,)
        t_start = t_start.long()
        x_cur = x
        for k in range(self.num_fused_steps):
            t_idx = t_start - k
            t_norm = t_idx.float() / float(self.total_timesteps)
            eps = self.eps_net(torch.cat([x_cur, t_norm.unsqueeze(-1)], dim=-1))
            c1 = self.sqrt_recip_alphas[t_idx].view(B, 1)
            c2 = self.coef_eps[t_idx].view(B, 1)
            x_cur = c1 * (x_cur - c2 * eps)
        return x_cur

class ModelNew(nn.Module):

    def forward(self, x: torch.Tensor, t_start: torch.Tensor) -> torch.Tensor:

__global__ __launch_bounds__(320, 2)
void fused_diffusion_dim80_kernel(const float* __restrict__ x,
                                  const int64_t* __restrict__ t_start,
                                  const float* __restrict__ w1,
                                  const float* __restrict__ b1,
                                  const float* __restrict__ w2,
                                  const float* __restrict__ b2,
                                  const float* __restrict__ w3,
                                  const float* __restrict__ b3,
                                  const float* __restrict__ sqrt_recip,
                                  const float* __restrict__ coef_eps,
                                  float* __restrict__ y,
                                  int B, int num_steps, int total_timesteps) {
    const int bid = blockIdx.x;
    const int tid = threadIdx.x;

    __shared__ float xs[80];
    __shared__ float h1[320];
    __shared__ float h2[320];

    if (tid < 80) xs[tid] = x[bid * 80 + tid];
    __syncthreads();

    const int t0 = (int)t_start[bid];

    #pragma unroll 1
    for (int k = 0; k < num_steps; ++k) {
        const int t = t0 - k;
        const float tn = ((float)t) / ((float)total_timesteps);

        // Linear(81 -> 320) + SiLU.  One thread computes one output neuron.
        float s1 = b1[tid];
        const float* __restrict__ w1r = w1 + tid * 81;
        #pragma unroll
        for (int i = 0; i < 80; ++i) {
            s1 = fmaf(w1r[i], xs[i], s1);
        }
        s1 = fmaf(w1r[80], tn, s1);
        h1[tid] = silu_f(s1);
        __syncthreads();

        // Linear(320 -> 320) + SiLU.
        float s2 = b2[tid];
        const float* __restrict__ w2r = w2 + tid * 320;
        #pragma unroll 4
        for (int i = 0; i < 320; ++i) {
            s2 = fmaf(w2r[i], h1[i], s2);
        }
        h2[tid] = silu_f(s2);
        __syncthreads();

        // Linear(320 -> 80), then DDPM deterministic mean update.
        if (tid < 80) {
            float s3 = b3[tid];
            const float* __restrict__ w3r = w3 + tid * 320;
            #pragma unroll 4
            for (int i = 0; i < 320; ++i) {
                s3 = fmaf(w3r[i], h2[i], s3);
            }
            const float c1 = sqrt_recip[t];
            const float c2 = coef_eps[t];
            xs[tid] = c1 * (xs[tid] - c2 * s3);
        }
        __syncthreads();
    }

    if (tid < 80) y[bid * 80 + tid] = xs[tid];
}

torch::Tensor fused_forward(torch::Tensor x, torch::Tensor t_start,
                           torch::Tensor w1, torch::Tensor b1,
                           torch::Tensor w2, torch::Tensor b2,
                           torch::Tensor w3, torch::Tensor b3,
                           torch::Tensor sqrt_recip, torch::Tensor coef_eps,
                           int64_t num_steps, int64_t total_timesteps) {
    auto y = torch::empty_like(x);
    const int B = (int)x.size(0);
    if (B == 0) return y;

    dim3 block(320);
    dim3 grid(B);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    fused_diffusion_dim80_kernel<<<grid, block, 0, stream>>>(
        x.data_ptr<float>(),
        t_start.data_ptr<int64_t>(),
        w1.data_ptr<float>(), b1.data_ptr<float>(),
        w2.data_ptr<float>(), b2.data_ptr<float>(),
        w3.data_ptr<float>(), b3.data_ptr<float>(),
        sqrt_recip.data_ptr<float>(), coef_eps.data_ptr<float>(),
        y.data_ptr<float>(), B, (int)num_steps, (int)total_timesteps);
    return y;
}

    return torch.linspace(beta_start, beta_end, num_timesteps)

        assert num_fused_steps >= 1
        assert total_timesteps >= num_fused_steps + 1

        self.num_fused_steps = num_fused_steps

            nn.Linear(dim * 4, dim * 4), nn.SiLU(),
