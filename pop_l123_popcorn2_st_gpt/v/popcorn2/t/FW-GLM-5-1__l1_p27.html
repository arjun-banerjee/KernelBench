import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Generate the fixed permutation (same as reference)
N = 72
_g = torch.Generator()
_g.manual_seed(10)
PERM = torch.randperm(N, generator=_g)

# CUDA kernel with shared memory optimization
cuda_source = """
__global__ void permutation_kernel(const float* __restrict__ input,
                                    float* __restrict__ output,
                                    const int64_t* __restrict__ perm,
                                    int B, int N) {
    // Shared memory for permutation indices
    extern __shared__ int64_t s_perm[];
    
    // Cooperative load of permutation into shared memory
    for (int i = threadIdx.x; i < N; i += blockDim.x) {
        s_perm[i] = perm[i];
    }
    __syncthreads();
    
    // Process elements
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * N;
    
    if (idx < total) {
        int b = idx / N;
        int j = idx % N;
        output[idx] = input[b * N + s_perm[j]];
    }
}

torch::Tensor permutation_forward(torch::Tensor x, torch::Tensor perm) {
    int B = x.size(0);
    int N = x.size(1);
    
    auto x_cont = x.contiguous();
    auto perm_cont = perm.contiguous();
    auto output = torch::empty_like(x_cont);
    
    int threads = 256;
    int blocks = (B * N + threads - 1) / threads;
    int shared_mem = N * sizeof(int64_t);

        x_cont.data_ptr<float>(),
        output.data_ptr<float>(),
        perm_cont.data_ptr<int64_t>(),

    );
    
    return output;
}
