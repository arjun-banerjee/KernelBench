import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        assert kernel_size % 2 == 1, 'odd kernel_size keeps T unchanged with symmetric padding'
        assert dilation >= 1
        self.padding = (kernel_size - 1) * dilation // 2
        self.dw = nn.Conv1d(in_channels, in_channels, kernel_size, padding=self.padding, dilation=dilation, groups=in_channels, bias=False)
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

class ModelNew(nn.Module):
    """KernelBench candidate entry point; default delegates to ``Model``."""

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        self._impl = Model(in_channels, out_channels, kernel_size, dilation=dilation)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self._impl(x)

    functions=['depthwise_silu_cuda', 'pointwise_silu_cuda'],
    verbose=False,
    extra_cuda_cflags=['-O3', '-use_fast_math']
)


class ModelNew(nn.Module):
    """
    Optimized fused depthwise-pointwise 1D temporal convolution with SiLU activation.
    
    Uses custom CUDA kernels for:
    1. Fused depthwise conv + SiLU
    2. Fused pointwise conv + SiLU
    """
    
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int = 1):
        super().__init__()
        assert kernel_size % 2 == 1, 'odd kernel_size keeps T unchanged with symmetric padding'
        assert dilation >= 1
        
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.dilation = dilation
        self.padding = (kernel_size - 1) * dilation // 2
        
        # Depthwise conv weights: (in_channels, 1, kernel_size)
        self.dw_weight = nn.Parameter(torch.empty(in_channels, 1, kernel_size))
        # Pointwise conv weights: (out_channels, in_channels, 1)
        self.pw_weight = nn.Parameter(torch.empty(out_channels, in_channels, 1))
        
        # Initialize weights same as nn.Conv1d
        nn.init.kaiming_uniform_(self.dw_weight, a=5**0.5)
        nn.init.kaiming_uniform_(self.pw_weight, a=5**0.5)
        
        # For correctness verification, we'll use the reference model
        self._reference = None
    
    def _init_reference(self):
        """Initialize reference model with same weights."""
        if self._reference is None:
            ref = nn.Module()
            ref.dw = nn.Conv1d(
                self.in_channels, self.in_channels, self.kernel_size,
                padding=self.padding, dilation=self.dilation,
                groups=self.in_channels, bias=False
            )
            ref.pw = nn.Conv1d(
                self.in_channels, self.out_channels, kernel_size=1, bias=False
            )
            # Copy weights
            ref.dw.weight.data = self.dw_weight.data.clone()
            ref.pw.weight.data = self.pw_weight.data.clone()
            self._reference = ref
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features
        Returns:
            (B, C_out, T) output features
        """
        # Ensure contiguous input
        x = x.contiguous()
        
        # Step 1: Fused depthwise conv + SiLU
        # dw_weight shape: (in_channels, 1, kernel_size) -> squeeze to (in_channels, kernel_size)
        dw_w = self.dw_weight.squeeze(1).contiguous()
        intermediate = fused_conv.depthwise_silu_cuda(x, dw_w, self.dilation)
        
        # Step 2: Fused pointwise conv + SiLU  
        # pw_weight shape: (out_channels, in_channels, 1) -> squeeze to (out_channels, in_channels)
        pw_w = self.pw_weight.squeeze(2).contiguous()
        output = fused_conv.pointwise_silu_cuda(intermediate, pw_w)
        
        return output
