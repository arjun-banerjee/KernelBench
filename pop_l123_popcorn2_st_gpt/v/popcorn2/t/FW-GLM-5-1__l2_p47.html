    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int R, int B, int H, int K) {
    
    const int total_B = B;
    const int total_K = K;
    const int inner_dim = R * H;
    
    // Grid-stride loop over batch dimension
    // Each block handles multiple (b, k) outputs
    const int batch_stride = gridDim.x;
    const int k_stride = gridDim.y;
    
    const int batch_start = blockIdx.x;
    const int k_start = blockIdx.y * blockDim.x * VEC_SIZE + threadIdx.x * VEC_SIZE;
    
    // Shared memory for weight tile - helps with reuse across batch
    extern __shared__ float weight_tile[];
    
    // Process multiple batches per block
    for (int b = batch_start; b < total_B; b += batch_stride) {
        // Load bias values into registers
        float bias_vals[VEC_SIZE];
        #pragma unroll
        for (int v = 0; v < VEC_SIZE; v++) {
            int k_idx = k_start + v;
            bias_vals[v] = (k_idx < total_K) ? bias[k_idx] : 0.0f;
        }
        
        // Accumulator for each output element
        float sum[VEC_SIZE];
        #pragma unroll
        for (int v = 0; v < VEC_SIZE; v++) {
            sum[v] = bias_vals[v];
        }
        
        // Compute dot product for all inner_dim elements
        // Process in tiles for better cache utilization
        const int TILE_SIZE = 32;
        
        for (int tile_start = 0; tile_start < inner_dim; tile_start += TILE_SIZE) {
            int tile_end = min(tile_start + TILE_SIZE, inner_dim);
            
            // Cooperatively load weight tile into shared memory
            for (int j_offset = threadIdx.x; j_offset < TILE_SIZE && (tile_start + j_offset) < inner_dim; j_offset += blockDim.x) {
                int j = tile_start + j_offset;
                // Load weight for all k values that this thread handles
                #pragma unroll
                for (int v = 0; v < VEC_SIZE; v++) {
                    int k_idx = k_start + v;
                    if (k_idx < total_K && (tile_start + j_offset) < inner_dim) {
                        weight_tile[v * TILE_SIZE + j_offset] = weight[k_idx * inner_dim + j];
                    }
                }
            }
            
            __syncthreads();
            
            // Compute partial dot product using shared weight tile
            for (int j_offset = 0; j_offset < tile_end - tile_start; j_offset++) {
                int j = tile_start + j_offset;
                int h = j % H;
                int r = j / H;
                
                // Input access: input is (R, B, H), need input[r, b, h]
                float in_val = input[r * B * H + b * H + h];
                
                #pragma unroll
                for (int v = 0; v < VEC_SIZE; v++) {
                    sum[v] += in_val * weight_tile[v * TILE_SIZE + j_offset];
                }
            }
            
            __syncthreads();
        }
        
        // Write output
        #pragma unroll
        for (int v = 0; v < VEC_SIZE; v++) {
            int k_idx = k_start + v;
            if (k_idx < total_K) {
                output[b * K + k_idx] = sum[v];
            }
        }
    }
}

void launch_fused_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int R, int B, int H, int K) {
    
    const int vec_size = VEC_SIZE;
    const int threads_per_block = 128;
    
    // Grid configuration: batch blocks + K blocks
    dim3 grid(B, (K + threads_per_block * vec_size - 1) / (threads_per_block * vec_size));
    dim3 block(threads_per_block);
    
    // Shared memory size: VEC_SIZE * TILE_SIZE floats
    size_t shared_mem = VEC_SIZE * 32 * sizeof(float);
    
    fused_allgather_linear_kernel<<<grid, block, shared_mem>>>(
        input, weight, bias, output, R, B, H, K);
}
