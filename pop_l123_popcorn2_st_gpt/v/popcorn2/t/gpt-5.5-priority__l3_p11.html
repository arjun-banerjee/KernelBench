Dimensions are fixed; weights are trainable parameters with default init.

from kernelbench.distributed_collectives import default_device

class Model(nn.Module):

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define H0 48
#define H1 64
#define H2 40
#define H3 24

__global__ void mlp3_kernel(const float* __restrict__ h0,
                            const float* __restrict__ w1,
                            const float* __restrict__ w2,
                            const float* __restrict__ w3,
                            float* __restrict__ out,
                            int N) {
    int row = blockIdx.x;
    int tid = threadIdx.x;
    if (row >= N) return;

    __shared__ float s1[H1];
    __shared__ float s2[H2];

    const float* x = h0 + row * H0;

    // h1 = relu(x @ W1.T).  One thread computes one output element.
    if (tid < H1) {
        const float* ww = w1 + tid * H0;
        float acc = 0.0f;
        #pragma unroll
        for (int k = 0; k < H0; ++k) {
            acc = fmaf(x[k], ww[k], acc);
        }
        s1[tid] = acc > 0.0f ? acc : 0.0f;
