import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Evidence Lower Bound (ELBO) computation for a Variational Autoencoder.
    Combines a reconstruction term (binary cross-entropy) with the
    KL divergence between the approximate posterior q(z|x) = N(mu, sigma^2)
    and the prior p(z) = N(0, I).  The core loss function in VAE training.
    """

    def __init__(self, input_dim, hidden_dim, latent_dim):
        super().__init__()
        self.encoder = nn.Sequential(nn.Linear(input_dim, hidden_dim), nn.ReLU())
        self.fc_mu = nn.Linear(hidden_dim, latent_dim)
        self.fc_logvar = nn.Linear(hidden_dim, latent_dim)
        self.decoder = nn.Sequential(nn.Linear(latent_dim, hidden_dim), nn.ReLU(), nn.Linear(hidden_dim, input_dim), nn.Sigmoid())

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (B, input_dim) – input data in [0, 1]
        Returns:
            (elbo, recon_loss, kl_div):
                elbo:       (B,)
                recon_loss: (B,)
                kl_div:     (B,)
        """
        h = self.encoder(x)
        mu = self.fc_mu(h)
        logvar = self.fc_logvar(h)
        std = torch.exp(0.5 * logvar)
        z = mu + std * torch.randn_like(std)
        x_recon = self.decoder(z)
        recon = F.binary_cross_entropy(x_recon, x, reduction='none').sum(-1)
        kl = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(-1)
        elbo = -(recon + kl)
        return (elbo, recon, kl)

    const scalar_t* __restrict__ bias,
    const scalar_t* __restrict__ x_target,
    scalar_t* __restrict__ bce_partial,
    int batch_size, int hidden_dim, int input_dim) {
    
    int b = blockIdx.y;
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (b < batch_size && i < input_dim) {
        scalar_t sum = bias[i];
        for (int h = 0; h < hidden_dim; h++) {
            sum += h2[b * hidden_dim + h] * weight[i * hidden_dim + h];
        }
        scalar_t x_recon = sigmoid_op(sum);
        bce_partial[b * input_dim + i] = bce_op(x_recon, x_target[b * input_dim + i]);
    }
}

template <typename scalar_t>
__global__ void sum_reduce_kernel(
    const scalar_t* __restrict__ input,
    scalar_t* __restrict__ output,
    int batch_size, int dim) {
    
    int b = blockIdx.x;
    if (b < batch_size) {
        scalar_t sum = scalar_t(0);
        for (int i = 0; i < dim; i++) {
            sum += input[b * dim + i];
        }
        output[b] = sum;
    }
}

template <typename scalar_t>
__global__ void elbo_kernel(
    const scalar_t* __restrict__ recon,
    const scalar_t* __restrict__ kl,
    scalar_t* __restrict__ elbo,
    int batch_size) {
    
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b < batch_size) {
        elbo[b] = -(recon[b] + kl[b]);
    }
}

            (self.fc_mu_weight, self.fc_mu_bias),
            (self.fc_logvar_weight, self.fc_logvar_bias),
            (self.decoder1_weight, self.decoder1_bias),
            (self.decoder2_weight, self.decoder2_bias)
        ]:
            nn.init.kaiming_uniform_(weight, a=math.sqrt(5))
            fan_in = weight.size(1)
