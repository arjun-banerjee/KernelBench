import torch
import torch.nn as nn
import math

class Model(nn.Module):
    """
    Outer-product mean: converts MSA representation to a pair
    representation update.  Used in AlphaFold2 to communicate information
    from the MSA stack to the pair stack.  For each pair (i,j), the
    outer product of projected MSA features at positions i and j is
    averaged over the MSA depth dimension.
    """

    def __init__(self, msa_dim, pair_dim, hidden_dim=32):
        super().__init__()
        self.layer_norm = nn.LayerNorm(msa_dim)
        self.proj_a = nn.Linear(msa_dim, hidden_dim)
        self.proj_b = nn.Linear(msa_dim, hidden_dim)
        self.out_proj = nn.Linear(hidden_dim * hidden_dim, pair_dim)

    def forward(self, msa_repr: torch.Tensor, msa_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            msa_repr: (B, S, N, msa_dim) – MSA representation
            msa_mask: (B, S, N)          – 1 for valid rows/positions
        Returns:
            pair update (B, N, N, pair_dim)
        """
        x = self.layer_norm(msa_repr)
        a = self.proj_a(x)
        b = self.proj_b(x)
        mask = msa_mask.unsqueeze(-1)
        a = a * mask
        b = b * mask
        outer = torch.einsum('bsih,bsjk->bijhk', a, b)
        denom = msa_mask.sum(dim=1).unsqueeze(-1).unsqueeze(-1).unsqueeze(-1).clamp(min=1)
        outer = outer / denom
        (B, N1, N2, H1, H2) = outer.shape
        outer = outer.reshape(B, N1, N2, H1 * H2)
        return self.out_proj(outer)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <c10/cuda/CUDAStream.h>

#define C_DIM 64
#define H_DIM 48
#define O_DIM 64

__global__ void ln_proj_kernel(const float* __restrict__ x,
                               const float* __restrict__ mask,
                               const float* __restrict__ ln_w,
                               const float* __restrict__ ln_b,
                               const float* __restrict__ pa_w,
                               const float* __restrict__ pa_b,
                               const float* __restrict__ pb_w,
                               const float* __restrict__ pb_b,
                               float* __restrict__ A,
                               float* __restrict__ Bv,
                               int total_rows, int N) {
    int row = blockIdx.x;
    if (row >= total_rows) return;
    int tid = threadIdx.x;
    __shared__ float red[128];

    const float* xr = x + ((long long)row) * C_DIM;
    float v = 0.0f;
    if (tid < C_DIM) v = xr[tid];
    red[tid] = v;
    __syncthreads();
    for (int off = 64; off > 0; off >>= 1) {
        if (tid < off) red[tid] += red[tid + off];
        __syncthreads();
    }
    float mean = red[0] * (1.0f / (float)C_DIM);

    float d = 0.0f;
    if (tid < C_DIM) { float t = xr[tid] - mean; d = t * t; }
    red[tid] = d;
    __syncthreads();

        if (tid < off) red[tid] += red[tid + off];
        __syncthreads();

// T = A @ W_flat, where columns are (output_channel, hidden_k).
__global__ void aw_gemm_kernel(const float* __restrict__ A,
                               const float* __restrict__ out_w,
                               float* __restrict__ T,
                               int M) {
    const int TILE = 16;
    __shared__ float As[TILE][TILE];
    __shared__ float Ws[TILE][TILE];
    int row = blockIdx.y * TILE + threadIdx.y;
    int col = blockIdx.x * TILE + threadIdx.x; // col = o*H + k
    float acc = 0.0f;

    #pragma unroll
    for (int t0 = 0; t0 < H_DIM; t0 += TILE) {
        if (row < M) As[threadIdx.y][threadIdx.x] = A[(long long)row * H_DIM + (t0 + threadIdx.x)];
        else As[threadIdx.y][threadIdx.x] = 0.0f;

        int h = t0 + threadIdx.y;
        if (col < O_DIM * H_DIM) {
            int o = col / H_DIM;
            int k = col - o * H_DIM;
            // out_w is [O, H*H] with flattened index h*H+k
            Ws[threadIdx.y][threadIdx.x] = out_w[(long long)o * H_DIM * H_DIM + h * H_DIM + k];
        } else {
            Ws[threadIdx.y][threadIdx.x] = 0.0f;
        }
        __syncthreads();
        #pragma unroll
        for (int q = 0; q < TILE; ++q) acc = fmaf(As[threadIdx.y][q], Ws[q][threadIdx.x], acc);
        __syncthreads();
    }
    if (row < M && col < O_DIM * H_DIM) T[(long long)row * (O_DIM * H_DIM) + col] = acc;
}

__global__ void output_kernel(const float* __restrict__ T,
                              const float* __restrict__ Bv,
                              const float* __restrict__ out_b,
                              float* __restrict__ Y,
                              int Bsz, int S, int N) {
    const int TILE = 16;
    __shared__ float shT[TILE][H_DIM];
    __shared__ float shB[TILE][H_DIM];

    int tj = threadIdx.x;
    int ti = threadIdx.y;
    int j = blockIdx.x * TILE + tj;
    int i = blockIdx.y * TILE + ti;
    int zo = blockIdx.z;
    int b = zo / O_DIM;
    int o = zo - b * O_DIM;

    float acc = 0.0f;
    int linear = ti * TILE + tj;
    float invS = 1.0f / (float)S;

    for (int s = 0; s < S; ++s) {
        long long base_i = ((long long)(b * S + s) * N + (blockIdx.y * TILE)) * (O_DIM * H_DIM) + o * H_DIM;
        long long base_j = ((long long)(b * S + s) * N + (blockIdx.x * TILE)) * H_DIM;
        // cooperative load 16*48 values for each operand
        for (int idx = linear; idx < TILE * H_DIM; idx += TILE * TILE) {
            int r = idx / H_DIM;
            int k = idx - r * H_DIM;
            int gi = blockIdx.y * TILE + r;
            int gj = blockIdx.x * TILE + r;
            shT[r][k] = (gi < N) ? T[base_i + (long long)r * (O_DIM * H_DIM) + k] : 0.0f;
            shB[r][k] = (gj < N) ? Bv[base_j + (long long)r * H_DIM + k] : 0.0f;
        }
        __syncthreads();
        if (i < N && j < N) {
            #pragma unroll
            for (int k = 0; k < H_DIM; ++k) acc = fmaf(shT[ti][k], shB[tj][k], acc);
        }
        __syncthreads();
    }
    if (i < N && j < N) {
        Y[(((long long)b * N + i) * N + j) * O_DIM + o] = out_b[o] + acc * invS;
    }
}
