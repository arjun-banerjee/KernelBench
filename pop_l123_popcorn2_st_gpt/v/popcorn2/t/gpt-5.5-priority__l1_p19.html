import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues
        Returns:
            dist: (B, N, N) – pairwise Euclidean distance matrix
        """
        diff = coords.unsqueeze(2) - coords.unsqueeze(1)
        dist = torch.sqrt((diff ** 2).sum(dim=-1) + 1e-08)
        return dist

// 16x16 tiled kernel.  Each block computes one output tile for one batch item.
// Coordinates are contiguous as (B, N, 3), output is contiguous as (B, N, N).
template<int TILE>
__global__ void pairwise_dist_tiled_kernel(const float* __restrict__ coords,
                                           float* __restrict__ out,
                                           int B, int N) {
    __shared__ float xi[TILE][3];
    __shared__ float xj[TILE][3];

    const int tx = threadIdx.x;   // column within tile (j)
    const int ty = threadIdx.y;   // row within tile (i)
    const int j = blockIdx.x * TILE + tx;
    const int i = blockIdx.y * TILE + ty;
    const int b = blockIdx.z;

    const int row_i = blockIdx.y * TILE + tx;
    const int col_j = blockIdx.x * TILE + tx;
    const float* base = coords + ((long long)b * N * 3);

    // Use first row of threads to cooperatively load the two coordinate tiles.
    if (ty == 0) {
        if (row_i < N) {
            const float* p = base + row_i * 3;
            xi[tx][0] = p[0]; xi[tx][1] = p[1]; xi[tx][2] = p[2];
        }
        if (col_j < N) {
            const float* p = base + col_j * 3;
            xj[tx][0] = p[0]; xj[tx][1] = p[1]; xj[tx][2] = p[2];
        }
    }
    __syncthreads();
