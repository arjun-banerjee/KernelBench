import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Edge-only sampled dense-dense matmul.

    For each graph edge (src, dst), compute a dot product between source and
    destination node embeddings. This pattern appears in link prediction,
    sampled attention, and graph contrastive learning.
    """

    def __init__(self):
        super().__init__()

    def forward(self, src_idx: torch.Tensor, dst_idx: torch.Tensor, src_feat: torch.Tensor, dst_feat: torch.Tensor) -> torch.Tensor:
        src = src_feat[src_idx.long()]
        dst = dst_feat[dst_idx.long()]
        return (src * dst).sum(dim=-1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void sampled_dense_dense_matmul_kernel(
    const int32_t* __restrict__ src_idx,
    const int32_t* __restrict__ dst_idx,
    const float* __restrict__ src_feat,
    const float* __restrict__ dst_feat,
    float* __restrict__ output,
    const int num_edges,
    const int feat_dim,
    const int src_stride,
    const int dst_stride
) {
    int edge_idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (edge_idx >= num_edges) return;

    int src_node = src_idx[edge_idx];
    int dst_node = dst_idx[edge_idx];

    const float* src_ptr = src_feat + src_node * src_stride;
    const float* dst_ptr = dst_feat + dst_node * dst_stride;

    float sum = 0.0f;
    for (int i = 0; i < feat_dim; ++i) {
        sum += src_ptr[i] * dst_ptr[i];
    }

    output[edge_idx] = sum;
}

torch::Tensor sampled_dense_dense_matmul_cuda(
    torch::Tensor src_idx,
    torch::Tensor dst_idx,
    torch::Tensor src_feat,
    torch::Tensor dst_feat
) {
    int num_edges = src_idx.size(0);
    int feat_dim = src_feat.size(1);
    int num_src = src_feat.size(0);
    int num_dst = dst_feat.size(0);

    auto output = torch::empty({num_edges}, src_feat.options());

    const int threads = 256;
    const int blocks = (num_edges + threads - 1) / threads;

    sampled_dense_dense_matmul_kernel<<<blocks, threads>>>(
        src_idx.data_ptr<int32_t>(),
        dst_idx.data_ptr<int32_t>(),
        src_feat.data_ptr<float>(),
        dst_feat.data_ptr<float>(),
        output.data_ptr<float>(),
        num_edges,
        feat_dim,
        src_feat.stride(0),
        dst_feat.stride(0)
    );

    return output;
}
'''
