import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_hidden = 256

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor linear256_forward(torch::Tensor x, torch::Tensor weight);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

#define TILE 16

__global__ void linear256_kernel(const float* __restrict__ x,
                                 const float* __restrict__ w,
                                 float* __restrict__ y,
                                 int rows) {
    __shared__ float As[TILE][TILE];
    __shared__ float Bs[TILE][TILE + 1];

    const int tx = threadIdx.x;
    const int ty = threadIdx.y;
    const int row = blockIdx.y * TILE + ty;
    const int col = blockIdx.x * TILE + tx;

    float acc = 0.0f;

    #pragma unroll
    for (int t = 0; t < 16; ++t) {
        const int kx = t * TILE + tx;
        As[ty][tx] = (row < rows) ? x[row * 256 + kx] : 0.0f;

        // Cooperative load of a 16x16 tile of W^T.  Weight is stored as
        // w[out, k], so using tx as k gives contiguous loads for each output.
        const int wout = blockIdx.x * TILE + ty;
        const int wk = t * TILE + tx;
        Bs[tx][ty] = w[wout * 256 + wk];

        __syncthreads();

        #pragma unroll
        for (int k = 0; k < TILE; ++k) {
            acc = fmaf(As[ty][k], Bs[k][tx], acc);
        }
        __syncthreads();
    }

    if (row < rows) {
        y[row * 256 + col] = acc;
    }
}

torch::Tensor linear256_forward(torch::Tensor x, torch::Tensor weight) {
    TORCH_CHECK(x.is_cuda(), "x must be a CUDA tensor");
    TORCH_CHECK(weight.is_cuda(), "weight must be a CUDA tensor");
    TORCH_CHECK(x.scalar_type() == torch::kFloat32, "only float32 is supported");
    TORCH_CHECK(weight.scalar_type() == torch::kFloat32, "only float32 weights are supported");
    TORCH_CHECK(x.numel() % 256 == 0, "last dimension must be 256");

    auto y = torch::empty_like(x);
    const int rows = (int)(x.numel() / 256);
    if (rows == 0) return y;

    dim3 block(TILE, TILE);
    dim3 grid(16, (rows + TILE - 1) / TILE);
    linear256_kernel<<<grid, block, 0, at::cuda::getCurrentCUDAStream()>>>(
        x.data_ptr<float>(), weight.data_ptr<float>(), y.data_ptr<float>(), rows);
    return y;
}
"""
