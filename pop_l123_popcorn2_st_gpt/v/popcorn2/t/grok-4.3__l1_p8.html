import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Per-row top-k selection over CSR edge segments.

    Returns the top-k values and their original edge indices for each node's
    neighborhood. Useful for graph sparsification, hard attention, and
    candidate-pruning workloads.
    """

    def __init__(self, k):
        super().__init__()
        self.k = k

    def forward(self, row_ptr: torch.Tensor, edge_scores: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        num_nodes = row_ptr.numel() - 1
        topk_vals = torch.full((num_nodes, self.k), float('-inf'), dtype=edge_scores.dtype, device=edge_scores.device)
        topk_idx = torch.full((num_nodes, self.k), -1, dtype=torch.int64, device=edge_scores.device)
        for node in range(num_nodes):
            start = int(row_ptr[node].item())
            end = int(row_ptr[node + 1].item())
            if end <= start:
                continue
            segment = edge_scores[start:end]
            curr_k = min(self.k, end - start)
            (vals, idx) = torch.topk(segment, k=curr_k, dim=0)
            topk_vals[node, :curr_k] = vals
            topk_idx[node, :curr_k] = idx.long() + start
        return (topk_vals, topk_idx)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cfloat>

__global__ void segment_topk_kernel(
    const float* __restrict__ edge_scores,
    const int64_t* __restrict__ row_ptr,
    float* __restrict__ topk_vals,
    int64_t* __restrict__ topk_idx,
    int num_nodes,
    int k
) {
    int node = blockIdx.x;
    if (node >= num_nodes) return;

    int start = row_ptr[node];
    int end = row_ptr[node + 1];
    int seg_len = end - start;

    if (seg_len <= 0) {
        for (int i = threadIdx.x; i < k; i += blockDim.x) {
            topk_vals[node * k + i] = -FLT_MAX;
            topk_idx[node * k + i] = -1;
        }
        return;
    }

    int curr_k = min(k, seg_len);

    extern __shared__ float sdata[];
    float* svals = sdata;
    int* sidx = (int*)(svals + seg_len);

    for (int i = threadIdx.x; i < seg_len; i += blockDim.x) {
        svals[i] = edge_scores[start + i];
        sidx[i] = i;
    }
    __syncthreads();

    // Simple selection for top-k (small k, small seg_len)
    for (int pos = 0; pos < curr_k; ++pos) {
        // Find max in [pos, seg_len)
        float local_max = -FLT_MAX;
        int local_idx = -1;
        for (int i = threadIdx.x + pos; i < seg_len; i += blockDim.x) {
            if (svals[i] > local_max) {
                local_max = svals[i];
                local_idx = i;
            }
        }

        // Reduce across threads
        __shared__ float max_vals[32];
        __shared__ int max_idxs[32];
        int lane = threadIdx.x % 32;
        int warp = threadIdx.x / 32;

        max_vals[lane] = local_max;
        max_idxs[lane] = local_idx;
        __syncthreads();

        if (threadIdx.x == 0) {
            float gmax = -FLT_MAX;
            int gidx = -1;
            for (int w = 0; w < blockDim.x; ++w) {
                if (max_vals[w] > gmax) {
                    gmax = max_vals[w];
                    gidx = max_idxs[w];

            svals[pos] = gmax;
            sidx[pos] = gidx;
            svals[gidx] = tmpv;
            sidx[gidx] = tmpi;

            topk_idx[node * k + pos] = gidx + start;
        }
        __syncthreads();
    }
}

}
