"""
Reference: shard the last dimension per rank, then all_gather to reconstruct (tensor-parallel style).

Multi-GPU: each rank keeps one column block; output is concat along last dim (matches full input).
Single-GPU: returns x unchanged.
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_rank, get_world_size, is_distributed_run, maybe_init_process_group
dim = 80

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunks = x.chunk(ws, dim=-1)
        local = chunks[rank].contiguous() * self.scale
        gathered = [torch.empty_like(local) for _ in range(ws)]
        dist.all_gather(gathered, local)
        return torch.cat(gathered, dim=-1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void scale_kernel(const float* __restrict__ x, float* __restrict__ out, float scale, int n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        out[idx] = x[idx] * scale;
    }
}

__global__ void scale_kernel_vec4(const float* __restrict__ x, float* __restrict__ out, float scale, int n) {
    int idx = (blockIdx.x * blockDim.x + threadIdx.x) * 4;
    if (idx + 3 < n) {
        float4 val = *reinterpret_cast<const float4*>(x + idx);
        val.x *= scale;
        val.y *= scale;
        val.z *= scale;
        val.w *= scale;
        *reinterpret_cast<float4*>(out + idx) = val;
    } else {
        for (int i = idx; i < n && i < idx + 4; ++i) {
            out[i] = x[i] * scale;
        }
    }
}

torch::Tensor scale_cuda(torch::Tensor x, float scale) {
    auto out = torch::empty_like(x);
    int n = x.numel();
    const int threads = 256;
    if (n % 4 == 0) {
        int blocks = (n / 4 + threads - 1) / threads;
        scale_kernel_vec4<<<blocks, threads>>>(x.data_ptr<float>(), out.data_ptr<float>(), scale, n);
