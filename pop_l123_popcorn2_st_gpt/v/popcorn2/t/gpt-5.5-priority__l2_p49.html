    acc += 0.5f * v3 * (1.0f + erff(v3 * INV_SQRT2_F));
    out[idx] = acc;
}

torch::Tensor gelu_sum_forward(torch::Tensor x) {
    const int R = (int)x.size(0);
    const int B = (int)x.size(1);
    const int D = (int)x.size(2);
    auto out = torch::empty({B, D}, x.options());
    const int N = B * D;
    const int threads = 128;
    const int blocks = (N + threads - 1) / threads;
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const float* xp = x.data_ptr<float>();
    float* op = out.data_ptr<float>();
    if (R == 4) {
        gelu_sum_kernel_r4<<<blocks, threads, 0, stream>>>(xp, op, B, D, N);
    } else {
        gelu_sum_kernel<<<blocks, threads, 0, stream>>>(xp, op, R, B, D, N);
    }
    return out;
}
