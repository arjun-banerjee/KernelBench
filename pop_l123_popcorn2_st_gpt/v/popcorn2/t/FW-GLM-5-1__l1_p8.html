import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Per-row top-k selection over CSR edge segments.

    Returns the top-k values and their original edge indices for each node's
    neighborhood. Useful for graph sparsification, hard attention, and
    candidate-pruning workloads.
    """

    def __init__(self, k):
        super().__init__()
        self.k = k

    def forward(self, row_ptr: torch.Tensor, edge_scores: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        num_nodes = row_ptr.numel() - 1
        topk_vals = torch.full((num_nodes, self.k), float('-inf'), dtype=edge_scores.dtype, device=edge_scores.device)
        topk_idx = torch.full((num_nodes, self.k), -1, dtype=torch.int64, device=edge_scores.device)
        for node in range(num_nodes):
            start = int(row_ptr[node].item())
            end = int(row_ptr[node + 1].item())
            if end <= start:
                continue
            segment = edge_scores[start:end]
            curr_k = min(self.k, end - start)
            (vals, idx) = torch.topk(segment, k=curr_k, dim=0)
            topk_vals[node, :curr_k] = vals
            topk_idx[node, :curr_k] = idx.long() + start
        return (topk_vals, topk_idx)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <limits>

template <typename scalar_t>
__global__ void segment_topk_kernel(
    const int64_t* __restrict__ row_ptr,
    const scalar_t* __restrict__ edge_scores,
    scalar_t* __restrict__ topk_vals,
    int64_t* __restrict__ topk_idx,
    const int num_nodes,
    const int k
) {
    const int node = blockIdx.x * blockDim.x + threadIdx.x;
    if (node >= num_nodes) return;
    
    const int start = row_ptr[node];
    const int end = row_ptr[node + 1];
    
    // Local arrays for top-k (support up to k=64)
    scalar_t local_vals[64];
    int64_t local_idx[64];
    
    // Initialize with default values
    for (int i = 0; i < k; i++) {
        local_vals[i] = -INFINITY;
        local_idx[i] = -1;
    }
    
    // Process segment elements
    for (int i = start; i < end; i++) {
        const scalar_t val = edge_scores[i];
        const int64_t idx = i;
        
        // Insertion sort: find position and insert
        for (int j = 0; j < k; j++) {
            if (val > local_vals[j]) {
                // Shift elements down
                for (int l = k - 1; l > j; l--) {
                    local_vals[l] = local_vals[l - 1];
                    local_idx[l] = local_idx[l - 1];
                }
                local_vals[j] = val;
                local_idx[j] = idx;
                break;
            }
        }
    }
    
    // Write results to global memory
    for (int i = 0; i < k; i++) {
        topk_vals[node * k + i] = local_vals[i];
        topk_idx[node * k + i] = local_idx[i];
    }
}

std::tuple<torch::Tensor, torch::Tensor> segment_topk_cuda(
    torch::Tensor row_ptr,
    torch::Tensor edge_scores,
    int64_t k
) {
    const int num_nodes = row_ptr.numel() - 1;

    auto topk_vals = torch::full({num_nodes, static_cast<long>(k)}, 
                                  -std::numeric_limits<float>::infinity(), 
                                  edge_scores.options());
    auto topk_idx = torch::full({num_nodes, static_cast<long>(k)}, 
                                  -1,

            topk_idx.data_ptr<int64_t>(),

            static_cast<int>(k)

    }));

}
