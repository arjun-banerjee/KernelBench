"""
Reference: shard the last dimension per rank, then all_gather to reconstruct (tensor-parallel style).

Multi-GPU: each rank keeps one column block; output is concat along last dim (matches full input).
Single-GPU: returns x unchanged.
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_rank, get_world_size, is_distributed_run, maybe_init_process_group
dim = 80

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(0.5))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        rank = get_rank()
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunks = x.chunk(ws, dim=-1)
        local = chunks[rank].contiguous() * self.scale
        gathered = [torch.empty_like(local) for _ in range(ws)]
        dist.all_gather(gathered, local)
        return torch.cat(gathered, dim=-1)

__global__ void scale_scalar_kernel(const float* __restrict__ x,
                                    const float* __restrict__ scale,
                                    float* __restrict__ y,
                                    long long n) {
    const float s = scale[0];
    long long i = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    const long long stride = (long long)blockDim.x * gridDim.x;
    for (; i < n; i += stride) {
        y[i] = x[i] * s;
    }
}

__global__ void scale_vec4_kernel(const float4* __restrict__ x,
                                  const float* __restrict__ scale,
                                  float4* __restrict__ y,
                                  long long n4) {
    const float s = scale[0];
    long long i = (long long)blockIdx.x * blockDim.x + threadIdx.x;
    const long long stride = (long long)blockDim.x * gridDim.x;
    for (; i < n4; i += stride) {
        float4 v = x[i];
        v.x *= s;
        v.y *= s;
        v.z *= s;
        v.w *= s;
        y[i] = v;
    }
}

torch::Tensor scale_forward(torch::Tensor x, torch::Tensor scale) {
    auto y = torch::empty_like(x);
    const long long n = (long long)x.numel();
    if (n == 0) return y;

    const float* xp = x.data_ptr<float>();
    const float* sp = scale.data_ptr<float>();
    float* yp = y.data_ptr<float>();

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const int threads = 256;

    uintptr_t xa = reinterpret_cast<uintptr_t>(xp);
    uintptr_t ya = reinterpret_cast<uintptr_t>(yp);
    if (((n & 3LL) == 0) && ((xa & 15) == 0) && ((ya & 15) == 0)) {
        const long long n4 = n >> 2;
        int blocks = (int)((n4 + threads - 1) / threads);
        if (blocks > 4096) blocks = 4096;
        scale_vec4_kernel<<<blocks, threads, 0, stream>>>(
            reinterpret_cast<const float4*>(xp), sp, reinterpret_cast<float4*>(yp), n4);
    } else {
        int blocks = (int)((n + threads - 1) / threads);
        if (blocks > 4096) blocks = 4096;
        scale_scalar_kernel<<<blocks, threads, 0, stream>>>(xp, sp, yp, n);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return y;
}
