import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Voxel-grid pooling for 3-D molecular point clouds.  Discretizes
    continuous 3-D coordinates into a regular grid and pools per-atom
    features into each voxel.  Used in 3-D-CNN approaches for molecular
    property prediction and ligand docking (e.g., PointVoxel-CNN,
    3D-Infomax).
    """

    def __init__(self, feat_dim, grid_size=32, voxel_size=1.0):
        super().__init__()
        self.feat_dim = feat_dim
        self.grid_size = grid_size
        self.voxel_size = voxel_size
        self.linear = nn.Linear(feat_dim, feat_dim)

    def forward(self, coords: torch.Tensor, features: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords:   (B, N, 3) – atom coordinates
            features: (B, N, feat_dim) – per-atom features
        Returns:
            voxel_grid: (B, feat_dim, G, G, G) – pooled voxel features
        """
        (B, N, _) = coords.shape
        G = self.grid_size
        features = self.linear(features)
        center = coords.mean(dim=1, keepdim=True)
        shifted = (coords - center) / self.voxel_size + G / 2.0
        indices = shifted.long().clamp(0, G - 1)
        voxel_grid = coords.new_zeros(B, self.feat_dim, G, G, G)
        ix = indices[:, :, 0]
        iy = indices[:, :, 1]
        iz = indices[:, :, 2]
        batch_idx = torch.arange(B, device=coords.device).unsqueeze(1).expand(-1, N)
        for d in range(self.feat_dim):
            feat_d = features[:, :, d]
            voxel_grid[:, d].index_put_((batch_idx.reshape(-1), ix.reshape(-1), iy.reshape(-1), iz.reshape(-1)), feat_d.reshape(-1), accumulate=True)
        return voxel_grid
feat_dim = 32
grid_size = 32
voxel_size = 1.0
num_atoms = 256
batch_size = 8

def get_inputs():

    features = torch.randn(bs, na, feat_dim)
    return [coords, features]

def get_init_inputs():
    return [feat_dim, grid_size, voxel_size]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void voxel_pool_kernel(
    const float* __restrict__ features,
    const long* __restrict__ indices,
    float* __restrict__ voxel_grid,
    const int B, const int N, const int F, const int G
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= B * N) return;
    int b = idx / N;
    int n = idx % N;
    long ix = indices[idx * 3];
    long iy = indices[idx * 3 + 1];
    long iz = indices[idx * 3 + 2];
    const float* feat_ptr = features + (b * N + n) * F;
    for (int f = 0; f < F; ++f) {
        float val = feat_ptr[f];
        atomicAdd(voxel_grid + ((b * F + f) * G + ix) * G * G + iy * G + iz, val);
    }
}

torch::Tensor voxel_pool_cuda(torch::Tensor features, torch::Tensor indices, int G) {
    int B = features.size(0);
    int N = features.size(1);
    int F = features.size(2);
    auto options = torch::TensorOptions().dtype(torch::kFloat32).device(features.device());
    auto voxel_grid = torch::zeros({B, F, G, G, G}, options);
    int threads = 256;
    int blocks = (B * N + threads - 1) / threads;
    voxel_pool_kernel<<<blocks, threads>>>(
        features.data_ptr<float>(),
        indices.data_ptr<long>(),
        voxel_grid.data_ptr<float>(),
        B, N, F, G
    );
    return voxel_grid;
}
'''

cpp_source = r'''
torch::Tensor voxel_pool_cuda(torch::Tensor features, torch::Tensor indices, int G);
'''

_voxel_pool = load_inline(
    name='voxel_pool_ext',
    cpp_sources=cpp_source,
    cuda_sources=cuda_source,
    functions=['voxel_pool_cuda'],
    verbose=False
)

class ModelNew(nn.Module):
    def __init__(self, feat_dim, grid_size=32, voxel_size=1.0):
        super().__init__()
        self.feat_dim = feat_dim
        self.grid_size = grid_size
        self.voxel_size = voxel_size
        self.linear = nn.Linear(feat_dim, feat_dim)

    def forward(self, coords: torch.Tensor, features: torch.Tensor) -> torch.Tensor:
        B, N, _ = coords.shape
        G = self.grid_size
        features = self.linear(features)
        center = coords.mean(dim=1, keepdim=True)
        shifted = (coords - center) / self.voxel_size + G / 2.0
        indices = shifted.long().clamp(0, G - 1).contiguous()
        return _voxel_pool.voxel_pool_cuda(features.contiguous(), indices.view(B * N, 3), G)
