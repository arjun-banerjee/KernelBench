import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Differentiable Smith-Waterman-style local sequence alignment scoring.
    Uses a soft-max relaxation of the DP recurrence so gradients can flow
    through the alignment matrix.  
    """

    def __init__(self, gap_open=-1.0, temperature=1.0):
        super().__init__()
        self.gap_open = gap_open
        self.temperature = temperature

    def forward(self, score_matrix: torch.Tensor) -> torch.Tensor:
        """
        Args:
            score_matrix: (B, M, N) – per-position similarity scores
        Returns:
            alignment_score: (B,) – soft local-alignment score (log-sum-exp over DP cells)
        """
        (B, M, N) = score_matrix.shape
        neg_inf = torch.tensor(float('-inf'), device=score_matrix.device, dtype=score_matrix.dtype)
        H = score_matrix.new_full((B, M + 1, N + 1), 0.0)
        for i in range(1, M + 1):
            for j in range(1, N + 1):
                match = H[:, i - 1, j - 1] + score_matrix[:, i - 1, j - 1]
                delete = H[:, i - 1, j] + self.gap_open
                insert = H[:, i, j - 1] + self.gap_open
                zero = torch.zeros(B, device=score_matrix.device)
                stack = torch.stack([zero, match, delete, insert], dim=-1)
                H[:, i, j] = self.temperature * torch.logsumexp(stack / self.temperature, dim=-1)
        return H.view(B, -1).max(dim=-1).values

#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void smith_waterman_kernel(
    const float* __restrict__ score_matrix,
    float* __restrict__ H,
    float* __restrict__ output,
    int B, int M, int N,
    float gap_open, float temperature
) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    if (b >= B) return;

    const int MN = (M + 1) * (N + 1);
    float* Hb = H + b * MN;

    // Initialize first row and column to 0
    for (int i = 0; i <= M; ++i) {
        Hb[i * (N + 1) + 0] = 0.0f;
    }
    for (int j = 0; j <= N; ++j) {
        Hb[0 * (N + 1) + j] = 0.0f;
    }

    float inv_temp = 1.0f / temperature;

    for (int i = 1; i <= M; ++i) {
        for (int j = 1; j <= N; ++j) {
            float match = Hb[(i-1) * (N+1) + (j-1)] + score_matrix[b * M * N + (i-1) * N + (j-1)];
            float del = Hb[(i-1) * (N+1) + j] + gap_open;
            float ins = Hb[i * (N+1) + (j-1)] + gap_open;
            float max_val = fmaxf(0.0f, fmaxf(match, fmaxf(del, ins)));
            // soft version
            float z0 = 0.0f;
            float z1 = match;
            float z2 = del;
            float z3 = ins;
            float m = fmaxf(fmaxf(z0, z1), fmaxf(z2, z3));
            float sum = expf((z0 - m) * inv_temp) + expf((z1 - m) * inv_temp) +
                        expf((z2 - m) * inv_temp) + expf((z3 - m) * inv_temp);
            float lse = m + logf(sum) * temperature;
            Hb[i * (N+1) + j] = lse;
        }
    }

    // Find max over all cells
    float max_score = 0.0f;
    for (int idx = 0; idx < MN; ++idx) {
        max_score = fmaxf(max_score, Hb[idx]);
    }
    output[b] = max_score;
}

torch::Tensor smith_waterman_cuda(torch::Tensor score_matrix, float gap_open, float temperature) {
    int B = score_matrix.size(0);
    int M = score_matrix.size(1);
    int N = score_matrix.size(2);

    auto options = score_matrix.options();
    auto H = torch::empty({B, M+1, N+1}, options);
    auto output = torch::empty({B}, options);

    dim3 block(32);
    dim3 grid((B + 31) / 32);

    smith_waterman_kernel<<<grid, block>>>(
        score_matrix.data_ptr<float>(),
        H.data_ptr<float>(),
        output.data_ptr<float>(),
        B, M, N, gap_open, temperature
    );

    return output;
}
