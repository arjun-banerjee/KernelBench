"""
R int32 bitmasks per batch element: reduce with bitwise OR across virtual ranks, then popcount (32-bit).

Output shape (B,) float32: number of set bits in the unsigned lower 32 bits of the OR result.
"""
import torch
import torch.nn as nn
from kernelbench.distributed_collectives import default_device
R = 4
B = 48

def _popcount_u32(v: torch.Tensor) -> torch.Tensor:
    """v: int32 or int64 tensor (B,), return float32 popcount in 0..32."""
    b = v.shape[0]
    u = v.to(torch.int64) & 4294967295
    c = torch.zeros(b, dtype=torch.int64, device=v.device)
    for k in range(32):
        c = c + (u >> k & 1)
    return c.to(torch.float32)

class Model(nn.Module):

    def __init__(self):
        super().__init__()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        acc = x[0].to(torch.int64)
        for r in range(1, R):
            acc = acc | x[r].to(torch.int64)
        acc = acc & 4294967295
        return _popcount_u32(acc.to(torch.int32))

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r'''
#include <torch/extension.h>
#include <cstdint>

// CUDA launcher declaration
void bitmask_popcount_launcher(const torch::Tensor& x, torch::Tensor& out);

torch::Tensor bitmask_popcount_forward(torch::Tensor x) {
    auto out = torch::empty({x.size(1)}, torch::TensorOptions().device(x.device()).dtype(torch::kFloat32));
    bitmask_popcount_launcher(x, out);
    return out;
}
'''

_cuda_src = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cstdint>
#include <c10/cuda/CUDAGuard.h>

__global__ void bitmask_popcount_kernel(const int32_t* __restrict__ x,
                                        float* __restrict__ out,
                                        int B) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < B) {
        // Reference uses exactly R=4 virtual ranks and counts the unsigned
        // lower 32 bits after bitwise OR.
        uint32_t v = (uint32_t)x[i];
        v |= (uint32_t)x[B + i];
        v |= (uint32_t)x[2 * B + i];
        v |= (uint32_t)x[3 * B + i];
        out[i] = (float)__popc(v);
    }
}

void bitmask_popcount_launcher(const torch::Tensor& x, torch::Tensor& out) {
    const at::cuda::OptionalCUDAGuard device_guard(device_of(x));
    int B = (int)x.size(1);
    const int threads = 128;
    const int blocks = (B + threads - 1) / threads;
    bitmask_popcount_kernel<<<blocks, threads, 0, at::cuda::getCurrentCUDAStream()>>>(
        reinterpret_cast<const int32_t*>(x.data_ptr<int32_t>()),
        out.data_ptr<float>(),
        B);
}
'''
