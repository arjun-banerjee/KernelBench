import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Scalar-gated delta-rule linear attention recurrence.

    A recurrent state matrix is decayed by a scalar gate per head, updated by a
    rank-1 key/value outer product, then read out by a query vector.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor, gate: torch.Tensor, beta: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []
        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-1).unsqueeze(-1)
            beta_t = beta[:, t].unsqueeze(-1).unsqueeze(-1)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + beta_t * kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)
        return torch.stack(outputs, dim=1)

__global__ void gated_delta_attention_kernel_v2(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    const float* __restrict__ beta,
    float* __restrict__ output,
    int batch_size,
    int seq_len,
    int num_heads,
    int head_dim
) {
    extern __shared__ float smem[];
    float* state = smem;

    int bh = blockIdx.x;
    int b = bh / num_heads;
    int h = bh % num_heads;
    int tid = threadIdx.x;
    int D = head_dim;

    // Init state
    for (int i = tid; i < D * D; i += blockDim.x) {
        state[i] = 0.0f;
    }
    __syncthreads();

    for (int t = 0; t < seq_len; t++) {
        float g = gate[(b * seq_len + t) * num_heads + h];
        float be = beta[(b * seq_len + t) * num_heads + h];

        // Update state with rank-1
        for (int i = tid; i < D * D; i += blockDim.x) {
            int d1 = i / D;
            int d2 = i % D;
            float kv = k[((b * seq_len + t) * num_heads + h) * D + d1] * v[((b * seq_len + t) * num_heads + h) * D + d2];
            state[i] = g * state[i] + be * kv;
        }
        __syncthreads();

        // Compute y: y[d2] = sum_d1 q[d1] * state[d1, d2]
        for (int d2 = tid; d2 < D; d2 += blockDim.x) {
            float sum = 0.0f;
            for (int d1 = 0; d1 < D; d1++) {
                sum += q[((b * seq_len + t) * num_heads + h) * D + d1] * state[d1 * D + d2];
            }
            output[((b * seq_len + t) * num_heads + h) * D + d2] = sum;
        }
        __syncthreads();
    }
}

torch::Tensor gated_delta_attention_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    torch::Tensor gate,
    torch::Tensor beta
) {
    auto batch_size = q.size(0);
    auto seq_len = q.size(1);
    auto num_heads = q.size(2);
    auto head_dim = q.size(3);

    auto output = torch::zeros({batch_size, seq_len, num_heads, head_dim}, q.options());

    int threads = 256;
    dim3 blocks(batch_size * num_heads);

    size_t smem_size = head_dim * head_dim * sizeof(float);

    gated_delta_attention_kernel_v2<<<blocks, threads, smem_size>>>(
        q.data_ptr<float>(),
        k.data_ptr<float>(),
        v.data_ptr<float>(),
        gate.data_ptr<float>(),
        beta.data_ptr<float>(),
        output.data_ptr<float>(),
        batch_size,
        seq_len,
        num_heads,
        head_dim
    );

    return output;
}
