import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a depthwise-separable 1D convolution: a depthwise Conv1d (each channel independently)
    followed by a pointwise Conv1d (1x1, mixes channels). Used in Enformer and genomic sequence
    models for efficient spatial+channel factorized processing.

    Args:
        in_channels (int): Number of channels in the input.
        out_channels (int): Number of channels produced by the pointwise convolution.
        kernel_size (int): Kernel size of the depthwise convolution.
        stride (int, optional): Stride of the depthwise convolution. Defaults to 1.
        padding (int, optional): Padding for the depthwise convolution. Defaults to 0.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int=1, padding: int=0, bias: bool=False):
        super(Model, self).__init__()
        self.depthwise = nn.Conv1d(in_channels, in_channels, kernel_size, stride=stride, padding=padding, groups=in_channels, bias=bias)
        self.pointwise = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.depthwise(x)
        x = self.pointwise(x)
        return x

// Fused depthwise separable 1D convolution kernel
// Each thread handles one output element: (batch, out_channel, position)
// Computes depthwise conv for all in_channels at that position, then pointwise reduction

__global__ void fused_depthwise_pointwise_kernel(
    const float* __restrict__ input,      // [batch, in_channels, seq_len]
    const float* __restrict__ depthwise_weight,  // [in_channels, 1, kernel_size]
    const float* __restrict__ depthwise_bias,    // [in_channels] or nullptr
    const float* __restrict__ pointwise_weight,  // [out_channels, in_channels, 1]
    const float* __restrict__ pointwise_bias,    // [out_channels] or nullptr
    float* __restrict__ output,           // [batch, out_channels, seq_len_out]
    int batch_size, int in_channels, int out_channels, 
    int seq_len, int seq_len_out, int kernel_size, int stride, int padding
) {
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    
    // Total outputs: batch * out_channels * seq_len_out
    const int total_outputs = batch_size * out_channels * seq_len_out;
    
    if (tid >= total_outputs) return;
    
    // Decode output indices
    int out_pos = tid % seq_len_out;
    int tmp = tid / seq_len_out;
    int oc = tmp % out_channels;
    int b = tmp / out_channels;
    
    // Compute input position range for depthwise convolution
    // output_pos = (input_pos - padding + kernel_size/2) / stride for depthwise
    // For stride=1: input_pos = output_pos - kernel_size/2 + padding + k
    // Actually: out_pos = (in_pos + 2*padding - kernel_size) / stride + 1
    // Simplified for stride=1: in_pos_start = out_pos - padding
    // But need to handle boundaries properly
    
    int in_pos_start = out_pos - padding;
    
    // Accumulate pointwise convolution
    float sum = 0.0f;
    
    // Process each input channel
    for (int ic = 0; ic < in_channels; ic++) {
        // Compute depthwise convolution for this channel at this position
        float depthwise_sum = 0.0f;
        
        // Depthwise weight index: [ic, 0, k]
        const float* dw_weight_ptr = depthwise_weight + ic * kernel_size;
        
        // Input base for this batch and channel
        const float* input_ptr = input + b * in_channels * seq_len + ic * seq_len;

        for (int k = 0; k < kernel_size; k++) {

        sum += pointwise_bias[oc];
    }
    
    // Write output
    output[b * out_channels * seq_len_out + oc * seq_len_out + out_pos] = sum;
}

__global__ void fused_kernel_optimized(
    const float* __restrict__ input,
    const float* __restrict__ depthwise_weight,
    const float* __restrict__ depthwise_bias,
    const float* __restrict__ pointwise_weight,
    const float* __restrict__ pointwise_bias,
    float* __restrict__ output,

    const int pos_start = seq_block * SEQ_PER_BLOCK;

    const int warp_idx = threadIdx.y;
    const int block_threads = blockDim.x * blockDim.y;

    const int tile_in_start = pos_start - padding;
    const int tile_in_size = SEQ_PER_BLOCK + kernel_size - 1;

    // Process in_channels one at a time or in tiles
    for (int ic = 0; ic < in_channels; ic++) {
        // Load input tile for this channel
        for (int i = lane_idx + warp_idx * blockDim.x; i < tile_in_size; i += block_threads) {

                    input[batch_idx * in_channels * seq_len + ic * seq_len + global_pos];
            } else {
                shared_input[ic * tile_in_size + i] = 0.0f;
            }
        }
    }

    const int local_pos = lane_idx;
    const int local_oc = warp_idx;

                }

                sum += dw_sum * pointwise_weight[oc * in_channels + ic];
            }

                sum += pointwise_bias[oc];
            }
            
            output[batch_idx * out_channels * seq_len_out + oc * seq_len_out + out_pos] = sum;
        }
    }
}
