import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Multi-head row-wise softmax over CSR edge scores.

    This extends scalar graph edge softmax to multiple attention heads while
    preserving sparse CSR layout. Each row is normalized independently for each
    head.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, edge_scores: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        out = torch.empty_like(edge_scores)
        for node in range(num_nodes):
            start = int(row_ptr[node].item())
            end = int(row_ptr[node + 1].item())
            if end > start:
                out[start:end] = torch.softmax(edge_scores[start:end], dim=0)

        sum_per_head[h] = 0.0f;
    }
    __syncthreads();
    
    for (int i = tid; i < total_elems; i += num_threads) {
        const int head = i % num_heads;
        const float val = expf(data[i] - max_per_head[head]);
        data[i] = val;
        atomicAdd(&sum_per_head[head], val);
    }
    __syncthreads();
    
    // Normalize and write output
    for (int i = tid; i < total_elems; i += num_threads) {
        const int edge = i / num_heads;
        const int head = i % num_heads;
        const int idx = (start + edge) * num_heads + head;
        out[idx] = data[i] / sum_per_head[head];
    }
}

    Multi-head row-wise softmax over CSR edge scores.

    """
    
    def __init__(self):
        super().__init__()
    
    def forward(self, row_ptr: torch.Tensor, edge_scores: torch.Tensor) -> torch.Tensor:
        # Handle device and dtype
        assert edge_scores.is_cuda, "edge_scores must be on CUDA"
        assert row_ptr.is_cuda, "row_ptr must be on CUDA"
        
        # Infer num_heads from tensor shape
        # edge_scores shape: (num_edges_total, num_heads) for contiguous layout
        if edge_scores.dim() == 1:
            num_heads = 1
        else:
            num_heads = edge_scores.size(1)
        
        # Ensure contiguous and correct dtype
        edge_scores_f32 = edge_scores.contiguous().float()
        row_ptr_int = row_ptr.contiguous().int()
        
        # Call CUDA kernel
        out = csr_softmax_cuda.csr_softmax_cuda(row_ptr_int, edge_scores_f32, num_heads)

            out = out.to(edge_scores.dtype)
