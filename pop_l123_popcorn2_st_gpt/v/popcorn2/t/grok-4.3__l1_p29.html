import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cfloat>

__global__ void logsumexp_kernel(const float* __restrict__ input, float* __restrict__ output,
                                 int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx >= total) return;

    int b = idx / D;
    int d = idx % D;

    float max_val = -FLT_MAX;
    for (int r = 0; r < R; ++r) {
        float val = input[r * B * D + b * D + d];
        max_val = fmaxf(max_val, val);
    }

    float sum_exp = 0.0f;
    for (int r = 0; r < R; ++r) {
        float val = input[r * B * D + b * D + d];
        sum_exp += expf(val - max_val);
    }

    output[b * D + d] = max_val + logf(sum_exp);
}

torch::Tensor logsumexp_cuda(torch::Tensor x) {
    TORCH_CHECK(x.dim() == 3, "Input must be 3D");
    TORCH_CHECK(x.is_contiguous(), "Input must be contiguous");
    TORCH_CHECK(x.dtype() == torch::kFloat32, "Only float32 supported");

    int R = x.size(0);
    int B = x.size(1);
