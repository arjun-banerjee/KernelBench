import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Scalar-gated delta-rule linear attention recurrence.

    A recurrent state matrix is decayed by a scalar gate per head, updated by a
    rank-1 key/value outer product, then read out by a query vector.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor, gate: torch.Tensor, beta: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []
        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-1).unsqueeze(-1)
            beta_t = beta[:, t].unsqueeze(-1).unsqueeze(-1)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + beta_t * kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)
        return torch.stack(outputs, dim=1)

            #pragma unroll
            for (int off = 16; off > 0; off >>= 1) {
                x += __shfl_down_sync(0xffffffffu, x, off);
            }
        }
    }
    return x;
}

// Fast path: one CUDA block computes one output column for one (batch, head).
// For the common head_dim <= 256, each state element for this column lives in a register.
__global__ void gated_delta_reg_kernel(const float* __restrict__ q,
                                       const float* __restrict__ k,
                                       const float* __restrict__ v,
                                       const float* __restrict__ gate,
                                       const float* __restrict__ beta,
                                       float* __restrict__ out,
                                       int B, int S, int H, int D) {
    int col = blockIdx.x % D;
    int bh = blockIdx.x / D;
    int h = bh % H;
    int b = bh / H;
    int tid = threadIdx.x;

    float state = 0.0f;
    const long long bh_base = ((long long)b * S * H + h) * D;
    const long long gh_base = (long long)b * S * H + h;

    for (int t = 0; t < S; ++t) {
        long long vec_base = bh_base + (long long)t * H * D;
        long long gate_idx = gh_base + (long long)t * H;
        float g = gate[gate_idx];
        float be = beta[gate_idx];
        float vj = v[vec_base + col];

        float contrib = 0.0f;
        if (tid < D) {
            float st = g * state + be * k[vec_base + tid] * vj;
            state = st;
            contrib = q[vec_base + tid] * st;
        }

        float sum = block_sum(contrib);
        if (tid == 0) out[vec_base + col] = sum;
        if (blockDim.x > 32) __syncthreads();
    }
}

// Very large head_dim fallback; keeps the column state in shared memory and lets
// threads own multiple rows.  This is not normally used for the benchmark center,
// but avoids dimension assumptions.
__global__ void gated_delta_shared_kernel(const float* __restrict__ q,
                                          const float* __restrict__ k,
                                          const float* __restrict__ v,
                                          const float* __restrict__ gate,
                                          const float* __restrict__ beta,
                                          float* __restrict__ out,
                                          int B, int S, int H, int D) {
    extern __shared__ float state[];
    int col = blockIdx.x % D;
    int bh = blockIdx.x / D;
    int h = bh % H;
    int b = bh / H;
    int tid = threadIdx.x;

    for (int i = tid; i < D; i += blockDim.x) state[i] = 0.0f;
    __syncthreads();

    const long long bh_base = ((long long)b * S * H + h) * D;
    const long long gh_base = (long long)b * S * H + h;

    for (int t = 0; t < S; ++t) {
        long long vec_base = bh_base + (long long)t * H * D;
        long long gate_idx = gh_base + (long long)t * H;
        float g = gate[gate_idx];
        float be = beta[gate_idx];
        float vj = v[vec_base + col];

        float contrib = 0.0f;
        for (int i = tid; i < D; i += blockDim.x) {
            float st = g * state[i] + be * k[vec_base + i] * vj;
            state[i] = st;
            contrib += q[vec_base + i] * st;
        }

        float sum = block_sum(contrib);
        if (tid == 0) out[vec_base + col] = sum;
        __syncthreads();
    }
}

torch::Tensor gated_delta_forward_cuda(torch::Tensor q, torch::Tensor k, torch::Tensor v,
                                      torch::Tensor gate, torch::Tensor beta) {
    const int B = (int)q.size(0);
    const int S = (int)q.size(1);
    const int H = (int)q.size(2);
    const int D = (int)q.size(3);
    auto out = torch::empty_like(q);

    const int blocks = B * H * D;
    int threads;
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    if (D <= 256) {
        if (D <= 32) threads = 32;
        else if (D <= 64) threads = 64;
        else if (D <= 128) threads = 128;
        else threads = 256;
        gated_delta_reg_kernel<<<blocks, threads, 0, stream>>>(
            q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(),
            gate.data_ptr<float>(), beta.data_ptr<float>(), out.data_ptr<float>(),
            B, S, H, D);
    } else {
        threads = 256;
        gated_delta_shared_kernel<<<blocks, threads, (size_t)D * sizeof(float), stream>>>(
            q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(),
            gate.data_ptr<float>(), beta.data_ptr<float>(), out.data_ptr<float>(),
            B, S, H, D);
    }
    return out;
}
