"""
Reference: all_to_all_single with equal splits (NCCL all_to_all family).

With identical inputs on every rank, a symmetric equal-split exchange is an identity on the
concatenated layout; we multiply by a learnable scalar so the module is non-trivial.
"""
import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_world_size, is_distributed_run, maybe_init_process_group
flat_dim = 64

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(1.25))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        (b, d) = x.shape
        assert d == flat_dim, f'expected last dim {flat_dim}, got {d}'
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunk = d // ws
        assert chunk * ws == d, 'flat_dim must be divisible by world_size'
        flat = x.reshape(b * d)
        out_flat = torch.empty_like(flat)
        sizes = [chunk * b] * ws
        dist.all_to_all_single(out_flat, flat, output_split_sizes=sizes, input_split_sizes=sizes)
        return out_flat.reshape(b, d) * self.scale

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void scale_kernel(const float* input, float* output, float scale, int n) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx < n) {
        output[idx] = input[idx] * scale;
    }
}

torch::Tensor scale_forward(torch::Tensor x, float scale) {
    auto output = torch::empty_like(x);
    int n = x.numel();
    const int block_size = 256;
    const int num_blocks = (n + block_size - 1) / block_size;
    scale_kernel<<<num_blocks, block_size>>>(x.data_ptr<float>(), output.data_ptr<float>(), scale, n);
    return output;
}
'''

cpp_source = '''
torch::Tensor scale_forward(torch::Tensor x, float scale);
'''

module = load_inline(
    name='scale_module',
    cpp_sources=[cpp_source],
    cuda_sources=[cuda_source],
    functions=['scale_forward'],
    verbose=False
)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(1.25))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return module.scale_forward(x, self.scale.item())
