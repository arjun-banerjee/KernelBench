    This isolates the MLA-style compressed-cache pattern: a hidden activation is
    projected into a low-rank latent space and then expanded back into separate
    key and value tensors.
    """

        super().__init__()

    def forward(self, hidden: torch.Tensor, w_down: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, model_dim) = hidden.shape
        rank = w_down.shape[1]
        kv_heads = 4
        head_dim = w_up_k.shape[1] // kv_heads

        v = v.view(batch_size, seq_len, kv_heads, head_dim)

        #pragma unroll 4
        for (int r = 0; r < R; r++) {
            float latent_val = latent_shared[r];
            sum_k += latent_val * w_up_k[r * out_dim + h];
            sum_v += latent_val * w_up_v[r * out_dim + h];
        }
        
        k_out[bi * out_dim + h] = sum_k;
        v_out[bi * out_dim + h] = sum_v;
    }
}

// Batch processing kernel - multiple batch elements per block for better efficiency
__global__ void fused_mla_kernel_batch(
    const float* __restrict__ hidden,
    const float* __restrict__ w_down,
    const float* __restrict__ w_up_k,
    const float* __restrict__ w_up_v,
    float* __restrict__ k_out,
    float* __restrict__ v_out,
    int BS, int D, int R, int out_dim, int batch_per_block) {
    
    extern __shared__ float smem[];
    float* latent_batch = smem;  // batch_per_block * R floats
    
    int block_batch_start = blockIdx.x * batch_per_block;
    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int batch_in_block = ty;  // Which batch element within the block
    int bi = block_batch_start + batch_in_block;
    
    if (bi >= BS) return;
    
    // Phase 1: Each thread computes one latent value for its batch element
    if (tx < R) {
        float sum = 0.0f;
        for (int d = 0; d < D; d++) {
            sum += hidden[bi * D + d] * w_down[d * R + tx];
        }
        latent_batch[batch_in_block * R + tx] = sum;
    }
    __syncthreads();
    
    // Phase 2: Each thread computes one output column for k and v
    int h = tx;
    if (h < out_dim) {
        float sum_k = 0.0f, sum_v = 0.0f;
        for (int r = 0; r < R; r++) {
            float latent_val = latent_batch[batch_in_block * R + r];
            sum_k += latent_val * w_up_k[r * out_dim + h];
            sum_v += latent_val * w_up_v[r * out_dim + h];
        }
        k_out[bi * out_dim + h] = sum_k;
        v_out[bi * out_dim + h] = sum_v;
    }
}

torch::Tensor fused_mla_forward(
    torch::Tensor hidden,
    torch::Tensor w_down,
    torch::Tensor w_up_k,
    torch::Tensor w_up_v) {
    
    int batch_size = hidden.size(0);
    int seq_len = hidden.size(1);
    int model_dim = hidden.size(2);
    int BS = batch_size * seq_len;
    int D = model_dim;
    int R = w_down.size(1);
    int out_dim = w_up_k.size(1);
    
    // Flatten hidden to (BS, D)
    auto hidden_flat = hidden.view({BS, D});
    
    // Allocate output tensors
    auto options = hidden.options();
    auto k_out = torch::empty({BS, out_dim}, options);
    auto v_out = torch::empty({BS, out_dim}, options);
    
    // Launch kernel - one block per batch element
    int threads = 128;  // >= out_dim (128)
    int blocks = BS;
    int shared_mem = R * sizeof(float);  // latent_shared
    
    fused_mla_kernel<<<blocks, threads, shared_mem>>>(
        hidden_flat.data_ptr<float>(),
        w_down.data_ptr<float>(),
        w_up_k.data_ptr<float>(),
        w_up_v.data_ptr<float>(),
        k_out.data_ptr<float>(),
        v_out.data_ptr<float>(),
        BS, D, R, out_dim
    );
    
    // Reshape outputs and stack
    int kv_heads = 4;
    int head_dim = out_dim / kv_heads;
    
    auto k = k_out.view({batch_size, seq_len, kv_heads, head_dim});
    auto v = v_out.view({batch_size, seq_len, kv_heads, head_dim});
    
    return torch::stack({k, v}, /*dim=*/2);
}

class ModelNew(nn.Module):
    """
    Low-rank expansion of compressed KV activations into multi-head K/V tensors.
    
    Uses a fused CUDA kernel that:
    1. Computes latent = hidden @ w_down
    2. Computes k = latent @ w_up_k and v = latent @ w_up_v
    
    The latent values are kept in shared memory, avoiding global memory round-trip.
    """
    
    def __init__(self):
        super().__init__()
    
    def forward(self, hidden: torch.Tensor, w_down: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        return fused_mla_ops.fused_mla_forward_opt(hidden, w_down, w_up_k, w_up_v)
