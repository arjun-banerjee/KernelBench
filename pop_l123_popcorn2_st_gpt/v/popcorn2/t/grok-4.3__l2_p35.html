import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Fused 1D convolution + GroupNorm for temporal sequences (N, C, T).

    Typical in time-series heads, audio / motor trajectories, and sequence models
    that use strided or padded Conv1d followed by per-channel normalization without
    collapsing the time dimension (GroupNorm over `out_channels` on (N, C_out, T')).

    Reference fusion target: `Conv1d` → `GroupNorm` (no activation), matching a
    common stem or residual branch in temporal backbones.
    """

    def __init__(self, in_channels: int, out_channels: int, num_groups: int, kernel_size: int=3, stride: int=1, padding: int=1, dilation: int=1, eps: float=1e-05, affine: bool=True):
        super().__init__()
        assert out_channels % num_groups == 0
        self.conv = nn.Conv1d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, dilation=dilation, bias=False)
        self.norm = nn.GroupNorm(num_groups=num_groups, num_channels=out_channels, eps=eps, affine=affine)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (N, C_in, T) temporal feature map
        Returns:
            (N, C_out, T') with T' determined by conv stride / padding / kernel
        """
        return self.norm(self.conv(x))

cuda_source = '''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void fused_conv1d_groupnorm_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ gamma,
    const float* __restrict__ beta,
    float* __restrict__ output,
    int N, int C_in, int T_in, int C_out, int T_out,
    int K, int stride, int padding, int num_groups,
    float eps, int channels_per_group
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = N * C_out * T_out;
    if (idx >= total) return;

    int t_out = idx % T_out;
    int c_out = (idx / T_out) % C_out;
    int n = idx / (T_out * C_out);

    float sum = 0.0f;
    for (int k = 0; k < K; ++k) {
        int t_in = t_out * stride + k - padding;
        if (t_in >= 0 && t_in < T_in) {
            for (int c_in = 0; c_in < C_in; ++c_in) {
                float w = weight[(c_out * C_in + c_in) * K + k];
                float xval = input[n * C_in * T_in + c_in * T_in + t_in];
                sum += w * xval;
            }
        }
    }
    output[idx] = sum;
}

torch::Tensor fused_conv_groupnorm(
    torch::Tensor x,
    torch::Tensor weight,
    torch::Tensor gamma,
    torch::Tensor beta,
    int num_groups,
    int kernel_size,
    int stride,
    int padding,
    float eps
) {
    int N = x.size(0);
    int C_in = x.size(1);
    int T_in = x.size(2);
    int C_out = weight.size(0);
    int K = kernel_size;
    int T_out = (T_in + 2 * padding - K) / stride + 1;

    auto options = torch::TensorOptions().dtype(torch::kFloat32).device(x.device());
    auto conv_out = torch::empty({N, C_out, T_out}, options);

    int channels_per_group = C_out / num_groups;
    int total = N * C_out * T_out;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;

    fused_conv1d_groupnorm_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(),
        weight.data_ptr<float>(),
        gamma.data_ptr<float>(),
        beta.data_ptr<float>(),
        conv_out.data_ptr<float>(),
        N, C_in, T_in, C_out, T_out,
        K, stride, padding, num_groups,
        eps, channels_per_group
    );

    // GroupNorm on conv_out
    auto out_reshaped = conv_out.view({N, num_groups, channels_per_group, T_out});
    auto mean = out_reshaped.mean(dim={2, 3}, keepdim=true);
    auto var = out_reshaped.var(dim={2, 3}, unbiased=false, keepdim=true);
    out_reshaped = (out_reshaped - mean) / torch::sqrt(var + eps);
    out_reshaped = out_reshaped * gamma.view({1, num_groups, channels_per_group, 1})
                   + beta.view({1, num_groups, channels_per_group, 1});
    return out_reshaped.view({N, C_out, T_out});
}
