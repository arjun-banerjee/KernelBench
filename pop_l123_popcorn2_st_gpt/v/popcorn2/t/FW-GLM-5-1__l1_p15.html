class Model(nn.Module):
    """
    Scatter-add edge features into destination-node accumulators.

    This is a canonical graph primitive for COO-formatted workloads where edge
    messages are already materialized and must be reduced into node states.
    """

        super().__init__()
        self.num_nodes = num_nodes

    def forward(self, dst_idx: torch.Tensor, edge_feat: torch.Tensor) -> torch.Tensor:

        out = torch.zeros(self.num_nodes, feat_dim, dtype=edge_feat.dtype, device=edge_feat.device)

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r"""
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void scatter_add_kernel(
    const int* __restrict__ dst_idx,
    const float* __restrict__ edge_feat,
    float* __restrict__ out,
    const int num_edges,
    const int feat_dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = num_edges * feat_dim;
    
    if (idx >= total) return;
    
    int edge_id = idx / feat_dim;
    int feat_id = idx - edge_id * feat_dim;
    
    int dst = __ldg(&dst_idx[edge_id]);
    float val = __ldg(&edge_feat[idx]);
    atomicAdd(&out[dst * feat_dim + feat_id], val);
}

torch::Tensor scatter_add_cuda(
    torch::Tensor dst_idx,
    torch::Tensor edge_feat,
    int num_nodes
) {
    int num_edges = edge_feat.size(0);
    int feat_dim = edge_feat.size(1);
    
    auto out = torch::zeros({num_nodes, feat_dim}, edge_feat.options());
    
    int total = num_edges * feat_dim;
    const int threads = 256;
    int blocks = (total + threads - 1) / threads;
    
    scatter_add_kernel<<<blocks, threads>>>(
        dst_idx.data_ptr<int>(),
        edge_feat.data_ptr<float>(),
        out.data_ptr<float>(),
        num_edges,
        feat_dim
    );

}
