import math
import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Scaled low-precision attention with explicit dequantization semantics.

    The inputs are quantized tensors plus per-head scales. This mirrors the data
    path of FP8-style scaled matmul attention even though the reference computes
    everything through explicit dequantization in PyTorch.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q_q: torch.Tensor, k_q: torch.Tensor, v_q: torch.Tensor, q_scale: torch.Tensor, k_scale: torch.Tensor, v_scale: torch.Tensor) -> torch.Tensor:
        q = q_q.float() * q_scale.view(1, 1, -1, 1)
        k = k_q.float() * k_scale.view(1, 1, -1, 1)
        v = v_q.float() * v_scale.view(1, 1, -1, 1)
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        out = torch.zeros_like(q)
        scale = 1.0 / math.sqrt(head_dim)
        for b in range(batch_size):
            for h in range(num_heads):
                for t in range(seq_len):
                    q_t = q[b, t, h]
                    k_hist = k[b, :t + 1, h]
                    v_hist = v[b, :t + 1, h]
                    scores = (k_hist * q_t.unsqueeze(0)).sum(dim=-1) * scale
                    attn = torch.softmax(scores, dim=0)
                    out[b, t, h] = (attn.unsqueeze(-1) * v_hist).sum(dim=0)
        return out

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

__global__ void scaled_attention_kernel(
    const int8_t* __restrict__ q_q,
    const int8_t* __restrict__ k_q,
    const int8_t* __restrict__ v_q,
    const float* __restrict__ q_scale,
    const float* __restrict__ k_scale,
    const float* __restrict__ v_scale,
    float* __restrict__ out,
    int batch_size,
    int seq_len,
    int num_heads,
    int head_dim
) {
    int b = blockIdx.x;
    int h = blockIdx.y;
    int t = threadIdx.x;

    if (t >= seq_len) return;

    extern __shared__ float sdata[];

    float scale = 1.0f / sqrtf((float)head_dim);

    float q_t[32]; // max head_dim=32
    for (int d = 0; d < head_dim; d++) {
        int idx = ((b * seq_len + t) * num_heads + h) * head_dim + d;
        q_t[d] = (float)q_q[idx] * q_scale[h];
    }

    float scores[256]; // max seq_len=256 assume
    for (int s = 0; s <= t; s++) {
        float dot = 0.0f;
        for (int d = 0; d < head_dim; d++) {
            int kidx = ((b * seq_len + s) * num_heads + h) * head_dim + d;
            float k_val = (float)k_q[kidx] * k_scale[h];
            dot += k_val * q_t[d];
        }
        scores[s] = dot * scale;
    }

    float max_val = -1e30f;
    for (int s = 0; s <= t; s++) {
        if (scores[s] > max_val) max_val = scores[s];
    }

        scores[s] = expf(scores[s] - max_val);
        sum_exp += scores[s];
    }
    for (int s = 0; s <= t; s++) {
        scores[s] /= sum_exp;
    }

        float val = 0.0f;
        for (int s = 0; s <= t; s++) {
            int vidx = ((b * seq_len + s) * num_heads + h) * head_dim + d;
            float v_val = (float)v_q[vidx] * v_scale[h];
            val += scores[s] * v_val;
        }
        int oidx = ((b * seq_len + t) * num_heads + h) * head_dim + d;
        out[oidx] = val;
    }
}
