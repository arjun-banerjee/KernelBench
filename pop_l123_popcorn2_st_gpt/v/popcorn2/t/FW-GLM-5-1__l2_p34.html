import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_world_size, is_distributed_run, maybe_init_process_group
flat_dim = 64

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(1.25))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        (b, d) = x.shape
        assert d == flat_dim, f'expected last dim {flat_dim}, got {d}'
        if not is_distributed_run() or ws == 1:
            return x * self.scale
        chunk = d // ws
        assert chunk * ws == d, 'flat_dim must be divisible by world_size'

        out_flat = torch.empty_like(flat)
        sizes = [chunk * b] * ws
        dist.all_to_all_single(out_flat, flat, output_split_sizes=sizes, input_split_sizes=sizes)
        return out_flat.reshape(b, d) * self.scale

        output[i] = input[i] * scale;
    }
}

torch::Tensor scale_cuda(torch::Tensor input, torch::Tensor scale) {
    // Ensure contiguous tensors
    auto input_contiguous = input.contiguous();
    
    // Create output tensor
    auto output = torch::empty_like(input_contiguous);
    
    const int64_t num_elements = input_contiguous.numel();
    if (num_elements == 0) {
        return output;
    }
    
    // Kernel launch configuration
    const int threads = 256;
    const int blocks = (num_elements + threads - 1) / threads;
    
    // Get scalar value from scale tensor
    float scale_val = scale.item<float>();
    
    AT_DISPATCH_FLOATING_TYPES(input_contiguous.scalar_type(), "scale_kernel", ([&] {
        scale_kernel<scalar_t><<<blocks, threads>>>(
            input_contiguous.data_ptr<scalar_t>(),
            output.data_ptr<scalar_t>(),
            static_cast<scalar_t>(scale_val),
            num_elements
        );
    }));
    
    return output;
}
