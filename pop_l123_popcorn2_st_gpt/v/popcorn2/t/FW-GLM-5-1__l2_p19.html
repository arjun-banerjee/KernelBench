import torch
import torch.nn as nn

class Model(nn.Module):
    """
    1D depthwise separable convolution block: pointwise Conv1d (channel mixing), depthwise Conv1d (spatial),
    GELU activation, and residual addition. Used in genomic foundation models (Enformer) for efficient
    sequence processing of DNA/RNA.
    """

    def __init__(self, channels, expanded_channels, kernel_size):
        super(Model, self).__init__()
        self.pointwise = nn.Conv1d(channels, expanded_channels, 1)
        self.depthwise = nn.Conv1d(expanded_channels, expanded_channels, kernel_size, padding=kernel_size // 2, groups=expanded_channels)
        self.activation = nn.GELU()
        self.project_back = nn.Conv1d(expanded_channels, channels, 1)

    def forward(self, x):
        residual = x
        x = self.pointwise(x)
        x = self.depthwise(x)
        x = self.activation(x)
        x = self.project_back(x)
        x = x + residual
        return x

    // Each thread computes a subset of expanded_channels
    for (int e = threadIdx.x; e < expanded_channels; e += blockDim.x) {
        float pw_sum = pw_bias[e];
        for (int ch = 0; ch < channels; ch++) {
            float in_val = input[batch * channels * seq_len + ch * seq_len + seq_pos];
            float w_val = pw_weight[e * channels + ch];
            pw_sum += in_val * w_val;
        }
        shared_pw[e] = pw_sum;
    }
    __syncthreads();
    
    // Compute depthwise and GELU for all expanded channels
    int half_k = kernel_size / 2;
    for (int e = threadIdx.x; e < expanded_channels; e += blockDim.x) {
        float dw_sum = dw_bias[e];
        
        // For depthwise, need to access neighboring sequence positions
        // These neighbors might have different pointwise results
        for (int k = 0; k < kernel_size; k++) {
            int neighbor_pos = seq_pos - half_k + k;
            if (neighbor_pos >= 0 && neighbor_pos < seq_len) {
                // Need pointwise result for (batch, e, neighbor_pos)
                // This requires computing it - shared memory won't work across seq positions
                // Let's compute directly
                float pw_val = pw_bias[e];
                for (int ch = 0; ch < channels; ch++) {
                    float in_val = input[batch * channels * seq_len + ch * seq_len + neighbor_pos];
                    float w_val = pw_weight[e * channels + ch];
                    pw_val += in_val * w_val;
                }
                dw_sum += pw_val * dw_weight[e * kernel_size + k];
            }
        }
        shared_gelu[e] = gelu_approx(dw_sum);
    }
    __syncthreads();
    
    // Project back
    for (int e = 0; e < expanded_channels; e++) {
        result += shared_gelu[e] * pb_weight[out_ch * expanded_channels + e];
    }
    
    // Add residual
    result += input[batch * channels * seq_len + out_ch * seq_len + seq_pos];
    
    output[batch * channels * seq_len + out_ch * seq_len + seq_pos] = result;
}

// Alternative: Use multiple simpler kernels for better parallelism
// Kernel 1: Pointwise conv
__global__ void pointwise_conv_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size, int in_channels, int out_channels, int seq_len
) {
    const int batch = blockIdx.x;
    const int out_ch = blockIdx.y;
    const int seq_pos = blockIdx.z * blockDim.x + threadIdx.x;
    
    if (seq_pos >= seq_len) return;
    
    float sum = bias[out_ch];
    for (int in_ch = 0; in_ch < in_channels; in_ch++) {
        sum += input[batch * in_channels * seq_len + in_ch * seq_len + seq_pos] 
             * weight[out_ch * in_channels + in_ch];
    }
    
    output[batch * out_channels * seq_len + out_ch * seq_len + seq_pos] = sum;
}

// Kernel 2: Depthwise conv
__global__ void depthwise_conv_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size, int channels, int seq_len, int kernel_size, int padding
) {
    const int batch = blockIdx.x;
    const int ch = blockIdx.y;
    const int seq_pos = blockIdx.z * blockDim.x + threadIdx.x;
    
    if (seq_pos >= seq_len) return;
    
    float sum = bias[ch];
    int half_k = kernel_size / 2;
    
    for (int k = 0; k < kernel_size; k++) {
        int input_pos = seq_pos - half_k + k;
        if (input_pos >= 0 && input_pos < seq_len) {
            sum += input[batch * channels * seq_len + ch * seq_len + input_pos]
                 * weight[ch * kernel_size + k];
        }
    }
    
    output[batch * channels * seq_len + ch * seq_len + seq_pos] = sum;
}

// Kernel 3: GELU + project back + residual add (fused)
__global__ void gelu_project_residual_kernel(
    const float* __restrict__ gelu_input,
    const float* __restrict__ pb_weight,
    const float* __restrict__ pb_bias,
    const float* __restrict__ residual,
    float* __restrict__ output,
    int batch_size, int channels, int expanded_channels, int seq_len
) {
    const int batch = blockIdx.x;
    const int out_ch = blockIdx.y;
    const int seq_pos = blockIdx.z * blockDim.x + threadIdx.x;
    
    if (seq_pos >= seq_len) return;
    
    // Project back with GELU applied inline
    float sum = pb_bias[out_ch];
    for (int e = 0; e < expanded_channels; e++) {
        float gelu_in = gelu_input[batch * expanded_channels * seq_len + e * seq_len + seq_pos];
        float gelu_out = gelu_approx(gelu_in);
        sum += gelu_out * pb_weight[out_ch * expanded_channels + e];
    }
    
    // Add residual
    sum += residual[batch * channels * seq_len + out_ch * seq_len + seq_pos];
    
    output[batch * channels * seq_len + out_ch * seq_len + seq_pos] = sum;
}

// Optimized fused kernel using shared memory and cooperative loading
// Each thread block handles one (batch, seq_pos), computes all output channels
template <int BLOCK_SIZE, int EXPANDED_CH_PER_ITER>
__global__ void fused_optimized_kernel(
    const float* __restrict__ input,
    const float* __restrict__ pw_weight,
    const float* __restrict__ pw_bias,
    const float* __restrict__ dw_weight,
    const float* __restrict__ dw_bias,
    const float* __restrict__ pb_weight,
    const float* __restrict__ pb_bias,
    float* __restrict__ output,
    int batch_size, int channels, int expanded_channels, int seq_len, int kernel_size
) {
    const int batch = blockIdx.x;
    const int seq_pos = blockIdx.y;
    const int out_ch_base = blockIdx.z * BLOCK_SIZE;
    const int out_ch = out_ch_base + threadIdx.x;
    
    // Shared memory layout:
    // - input_channels[BLOCK_SIZE][channels]: cached input for this seq_pos
    // - pw_output[expanded_channels]: pointwise output for all exp channels
    // - dw_output[expanded_channels]: depthwise output for all exp channels
    
    extern __shared__ float smem[];
    float* smem_input = smem;  // channels * BLOCK_SIZE elements (but we only need channels for one seq_pos)
    float* smem_pw = &smem[channels];  // expanded_channels elements
    float* smem_dw = &smem[channels + expanded_channels];  // expanded_channels elements
    
    // Cache input for this (batch, seq_pos)
    for (int ch = threadIdx.x; ch < channels; ch += BLOCK_SIZE) {
        smem_input[ch] = input[batch * channels * seq_len + ch * seq_len + seq_pos];
    }
    __syncthreads();
    
    // Compute pointwise for all expanded channels
    // Each thread handles multiple expanded channels
    for (int e = threadIdx.x; e < expanded_channels; e += BLOCK_SIZE) {
        float pw_sum = pw_bias[e];
        for (int ch = 0; ch < channels; ch++) {
            pw_sum += smem_input[ch] * pw_weight[e * channels + ch];
        }
        smem_pw[e] = pw_sum;
    }
    __syncthreads();
    
    // Compute depthwise for all expanded channels
    // This requires access to neighboring seq positions
    int half_k = kernel_size / 2;
    for (int e = threadIdx.x; e < expanded_channels; e += BLOCK_SIZE) {
        float dw_sum = dw_bias[e];
        for (int k = 0; k < kernel_size; k++) {
            int neighbor_pos = seq_pos - half_k + k;
            if (neighbor_pos >= 0 && neighbor_pos < seq_len) {
                // Need to recompute pointwise for neighbor position
                float pw_val = pw_bias[e];
                for (int ch = 0; ch < channels; ch++) {
                    pw_val += input[batch * channels * seq_len + ch * seq_len + neighbor_pos]
                            * pw_weight[e * channels + ch];
                }
                dw_sum += pw_val * dw_weight[e * kernel_size + k];
            }
        }
        smem_dw[e] = gelu_approx(dw_sum);
    }
    __syncthreads();
    
    // Project back and write output
    if (out_ch < channels) {
        float sum = pb_bias[out_ch];
        for (int e = 0; e < expanded_channels; e++) {
            sum += smem_dw[e] * pb_weight[out_ch * expanded_channels + e];
        }
        // Add residual
        sum += smem_input[out_ch];
        output[batch * channels * seq_len + out_ch * seq_len + seq_pos] = sum;
    }
}
