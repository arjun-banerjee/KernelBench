import torch
import torch.nn as nn
import math

class Model(nn.Module):
    """
    One iteration of Expectation-Maximization for a Gaussian Mixture Model.
    E-step: compute soft assignments (responsibilities).
    M-step: update means, covariances, and mixing weights.
    The dominant cost is the batched evaluation of K Gaussian densities
    over N data points.
    """

    def __init__(self, dim, num_components):
        super().__init__()
        self.dim = dim
        self.K = num_components
        self.register_buffer('mus', torch.randn(num_components, dim))
        self.register_buffer('covs', torch.eye(dim).unsqueeze(0).expand(num_components, -1, -1).clone())
        self.register_buffer('log_pi', torch.zeros(num_components))

    def _log_gaussian(self, x, mu, cov):
        """Log N(x | mu, cov) for batched x."""
        D = mu.shape[-1]
        diff = x - mu.unsqueeze(0)
        L = torch.linalg.cholesky(cov)
        solve = torch.linalg.solve_triangular(L, diff.unsqueeze(-1), upper=False).squeeze(-1)
        maha = (solve ** 2).sum(-1)
        log_det = 2.0 * L.diagonal().log().sum()
        return -0.5 * (D * math.log(2 * math.pi) + log_det + maha)

    def forward(self, x: torch.Tensor) -> tuple:
        """
        Args:
            x: (N, dim) – data points
        Returns:
            (new_mus, new_covs, new_log_pi, log_lik):
                new_mus:    (K, dim)
                new_covs:   (K, dim, dim)
                new_log_pi: (K,)
                log_lik:    scalar – total log-likelihood
        """
        (N, D) = x.shape
        K = self.K
        log_pi = self.log_pi - torch.logsumexp(self.log_pi, dim=0)
        log_resp = torch.stack([self._log_gaussian(x, self.mus[k], self.covs[k]) + log_pi[k] for k in range(K)], dim=-1)
        log_lik = torch.logsumexp(log_resp, dim=-1).sum()
        log_resp = log_resp - torch.logsumexp(log_resp, dim=-1, keepdim=True)
        resp = torch.exp(log_resp)
        N_k = resp.sum(dim=0).clamp(min=1e-08)
        new_mus = resp.t() @ x / N_k.unsqueeze(-1)
        new_covs = torch.zeros_like(self.covs)
        for k in range(K):
            diff = x - new_mus[k].unsqueeze(0)
            weighted = diff * resp[:, k:k + 1]
            new_covs[k] = weighted.t() @ diff / N_k[k] + 0.0001 * torch.eye(D, device=x.device)
        new_log_pi = N_k.log() - math.log(N)
        return (new_mus, new_covs, new_log_pi, log_lik)

std::vector<torch::Tensor> gmm_forward(torch::Tensor x,

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>
#include <math.h>

#define D_ 8
#define K_ 5
#define NSTATS_ (K_ + K_*D_ + K_*D_*D_)

__global__ void accumulate_kernel(const float* __restrict__ x,
                                  const float* __restrict__ mus,
                                  const float* __restrict__ log_pi,
                                  int N,
                                  float* __restrict__ stats,
                                  float* __restrict__ loglik) {
    __shared__ float s_stats[NSTATS_];
    __shared__ float s_ll[128];
    const int tid = threadIdx.x;
    for (int i = tid; i < NSTATS_; i += blockDim.x) s_stats[i] = 0.0f;
    s_ll[tid] = 0.0f;
    __syncthreads();

    int n = blockIdx.x * blockDim.x + tid;
    if (n < N) {
        float xv[D_];
        #pragma unroll
        for (int d = 0; d < D_; ++d) xv[d] = x[n * D_ + d];

        float maxlp = log_pi[0];
        #pragma unroll
        for (int k = 1; k < K_; ++k) maxlp = fmaxf(maxlp, log_pi[k]);
        float slp = 0.0f;
        #pragma unroll
        for (int k = 0; k < K_; ++k) slp += expf(log_pi[k] - maxlp);
        float logpi_norm_const = maxlp + logf(slp);

        const float c = -0.5f * (float)D_ * 1.8378770664093453f; // log(2*pi)
        float score[K_];
        float m = -3.402823466e+38f;
        #pragma unroll
        for (int k = 0; k < K_; ++k) {
            float dist = 0.0f;
            #pragma unroll
            for (int d = 0; d < D_; ++d) {
                float diff = xv[d] - mus[k * D_ + d];
                dist += diff * diff;
            }
            float s = c - 0.5f * dist + (log_pi[k] - logpi_norm_const);
            score[k] = s;
            m = fmaxf(m, s);
        }
        float esum = 0.0f;
        #pragma unroll
        for (int k = 0; k < K_; ++k) esum += expf(score[k] - m);
        float lden = m + logf(esum);
        s_ll[tid] = lden;

        #pragma unroll
        for (int k = 0; k < K_; ++k) {
            float r = expf(score[k] - lden);
            atomicAdd(&s_stats[k], r);
            int sx_base = K_ + k * D_;
            #pragma unroll
            for (int a = 0; a < D_; ++a) {
                float rxa = r * xv[a];
                atomicAdd(&s_stats[sx_base + a], rxa);
            }
            int sxx_base = K_ + K_ * D_ + k * D_ * D_;
            #pragma unroll
            for (int a = 0; a < D_; ++a) {
                float rxa = r * xv[a];
                #pragma unroll
                for (int b = 0; b < D_; ++b) {
                    atomicAdd(&s_stats[sxx_base + a * D_ + b], rxa * xv[b]);
                }
            }
        }
    }
    __syncthreads();

    // reduce log-likelihood inside the block
    for (int off = 64; off > 0; off >>= 1) {
        if (tid < off) s_ll[tid] += s_ll[tid + off];
        __syncthreads();
    }
    if (tid == 0) atomicAdd(loglik, s_ll[0]);

    for (int i = tid; i < NSTATS_; i += blockDim.x) {
        atomicAdd(&stats[i], s_stats[i]);
    }
}

__global__ void finalize_kernel(const float* __restrict__ stats,
                                int N,
                                float* __restrict__ new_mus,
                                float* __restrict__ new_covs,
                                float* __restrict__ new_log_pi) {
    int idx = threadIdx.x + blockIdx.x * blockDim.x;
    if (idx < K_ * D_) {
        int k = idx / D_;
        int d = idx - k * D_;
        float nk = fmaxf(stats[k], 1.0e-8f);
        new_mus[idx] = stats[K_ + k * D_ + d] / nk;
    }
    if (idx < K_) {
        float nk = fmaxf(stats[idx], 1.0e-8f);
        new_log_pi[idx] = logf(nk) - logf((float)N);
    }
    __syncthreads();
    if (idx < K_ * D_ * D_) {
        int k = idx / (D_ * D_);
        int rem = idx - k * D_ * D_;
        int a = rem / D_;
        int b = rem - a * D_;
        float nk = fmaxf(stats[k], 1.0e-8f);
        float ma = stats[K_ + k * D_ + a] / nk;
        float mb = stats[K_ + k * D_ + b] / nk;
        float exx = stats[K_ + K_ * D_ + k * D_ * D_ + a * D_ + b] / nk;
        new_covs[k * D_ * D_ + a * D_ + b] = exx - ma * mb + ((a == b) ? 1.0e-4f : 0.0f);
    }
}

std::vector<torch::Tensor> gmm_forward(torch::Tensor x,
                                       torch::Tensor mus,
                                       torch::Tensor covs,
                                       torch::Tensor log_pi) {
    const int N = (int)x.size(0);
    auto opts = x.options();
    auto stats = torch::zeros({NSTATS_}, opts);
    auto loglik = torch::zeros({}, opts);
    auto new_mus = torch::empty({K_, D_}, opts);
    auto new_covs = torch::empty({K_, D_, D_}, opts);
    auto new_log_pi = torch::empty({K_}, opts);

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    dim3 block(128);
    dim3 grid((N + 127) / 128);
    accumulate_kernel<<<grid, block, 0, stream>>>(x.data_ptr<float>(), mus.data_ptr<float>(), log_pi.data_ptr<float>(), N, stats.data_ptr<float>(), loglik.data_ptr<float>());

    return {new_mus, new_covs, new_log_pi, loglik};
}

    def __init__(self, dim, num_components):
        super().__init__()
        self.dim = dim
        self.K = num_components
        self.register_buffer('mus', torch.randn(num_components, dim))
        self.register_buffer('covs', torch.eye(dim).unsqueeze(0).expand(num_components, -1, -1).clone())
        self.register_buffer('log_pi', torch.zeros(num_components))

    def forward(self, x: torch.Tensor) -> tuple:
        if not x.is_contiguous():
            x = x.contiguous()
        mus = self.mus.contiguous()
        log_pi = self.log_pi.contiguous()
        out = _gmm_ext.gmm_forward(x, mus, self.covs, log_pi)
        return (out[0], out[1], out[2], out[3])
