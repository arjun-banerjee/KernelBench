import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Forward algorithm for Hidden Markov Models.  Computes the log-
    likelihood log p(observations) by iterating the forward recursion
    in log-space for numerical stability.  The dominant inner loop in
    HMM training (Baum-Welch) and Viterbi decoding.
    """

    def __init__(self, num_states, num_obs):
        super().__init__()
        self.num_states = num_states
        self.num_obs = num_obs
        self.log_init = nn.Parameter(torch.zeros(num_states))
        self.log_trans = nn.Parameter(torch.zeros(num_states, num_states))
        self.log_emit = nn.Parameter(torch.zeros(num_states, num_obs))

    def _normalize_log(self, x, dim=-1):
        return x - torch.logsumexp(x, dim=dim, keepdim=True)

    def forward(self, observations: torch.Tensor) -> torch.Tensor:
        """
        Args:
            observations: (B, T) – integer observation indices (as long tensor)
        Returns:
            log_lik: (B,) – log p(observations) under the HMM
        """
        (B, T) = observations.shape
        log_pi = self._normalize_log(self.log_init)
        log_A = self._normalize_log(self.log_trans, dim=-1)
        log_B = self._normalize_log(self.log_emit, dim=-1)
        emit_0 = log_B[:, observations[:, 0]].t()
        log_alpha = log_pi.unsqueeze(0) + emit_0
        for t in range(1, T):
            log_alpha = log_alpha.unsqueeze(-1) + log_A.unsqueeze(0)
            log_alpha = torch.logsumexp(log_alpha, dim=1)
            emit_t = log_B[:, observations[:, t]].t()
            log_alpha = log_alpha + emit_t
        log_lik = torch.logsumexp(log_alpha, dim=-1)
        return log_lik

    s_data[tid] = local_max;
    __syncthreads();
    for (int s = blockDim.x >> 1; s > 0; s >>= 1) {
        if (tid < s) {
            s_data[tid] = fmaxf(s_data[tid], s_data[tid + s]);
        }
        __syncthreads();
    }
    float max_val = s_data[0];
    __syncthreads();
    
    if (max_val > -INFINITY) {
        for (int i = tid; i < size; i += blockDim.x) {
            local_sum += expf(data[i] - max_val);
        }
        
        s_data[tid] = local_sum;
        __syncthreads();
        for (int s = blockDim.x >> 1; s > 0; s >>= 1) {
            if (tid < s) {
                s_data[tid] += s_data[tid + s];
            }
            __syncthreads();
        }
        float log_sum = logf(s_data[0]);
        
        for (int i = tid; i < size; i += blockDim.x) {
            data[i] -= log_sum;
        }
    }
}

__global__ void hmm_forward_kernel(
    const int* __restrict__ observations,
    const float* __restrict__ log_pi,
    const float* __restrict__ log_A,
    const float* __restrict__ log_B,
    float* __restrict__ log_lik,
    int B, int T, int num_states, int num_obs
) {
    // observations: (B, T)
    // log_pi: (num_states,) - already normalized
    // log_A: (num_states, num_states) - already normalized
    // log_B: (num_states, num_obs) - already normalized
    // log_lik: (B,) output
    
    extern __shared__ float s_data[];
    
    int b = blockIdx.x;  // batch index
    int j = threadIdx.x; // state index
    int tid = threadIdx.x;
    int block_dim = blockDim.x;
    
    if (b >= B) return;
    
    const int* obs_b = observations + b * T;
    
    // Use shared memory for alpha values (ping-pong buffers)
    float* alpha_cur = s_data;                          // first num_states elements
    float* alpha_next = s_data + num_states;            // next num_states elements
    float* reduction = s_data + 2 * num_states;         // for reductions
    
    // Initialize: alpha[j] = log_pi[j] + log_B[j, obs[0]]
    int obs_0 = obs_b[0];
    float local_alpha = -INFINITY;
    if (j < num_states) {
        local_alpha = log_pi[j] + log_B[j * num_obs + obs_0];
        alpha_cur[j] = local_alpha;
    }
    __syncthreads();
    
    // Forward recursion through time
    for (int t = 1; t < T; t++) {
        int obs_t = obs_b[t];
        
        // Compute new alpha for state j
        // alpha_new[j] = logsumexp_i(alpha[i] + log_A[i,j]) + log_B[j, obs_t]
        
        float max_val = -INFINITY;
        
        // Compute logsumexp
        if (j < num_states) {
            float sum = 0.0f;
            
            // Sequential scan through states (small num_states=16)
            for (int i = 0; i < num_states; i++) {
                float v = alpha_cur[i] + log_A[i * num_states + j];
                if (v > max_val) max_val = v;
            }
            
            if (max_val > -INFINITY) {
                for (int i = 0; i < num_states; i++) {
                    float v = alpha_cur[i] + log_A[i * num_states + j];
                    sum += expf(v - max_val);
                }
                local_alpha = max_val + logf(sum) + log_B[j * num_obs + obs_t];
            } else {
                local_alpha = -INFINITY;
            }
            
            alpha_next[j] = local_alpha;
        }
        __syncthreads();
        
        // Swap buffers
        float* tmp = alpha_cur;
        alpha_cur = alpha_next;
        alpha_next = tmp;
    }
    
    // Final logsumexp over states
    float local_val = (j < num_states) ? alpha_cur[j] : -INFINITY;
    
    // Reduce max
    reduction[tid] = local_val;
    __syncthreads();
    for (int s = block_dim >> 1; s > 0; s >>= 1) {
        if (tid < s) {
            reduction[tid] = fmaxf(reduction[tid], reduction[tid + s]);
        }
        __syncthreads();
    }
    float global_max = reduction[0];
    __syncthreads();
    
    // Reduce sum of exp
    float exp_val = (local_val > -INFINITY) ? expf(local_val - global_max) : 0.0f;
    reduction[tid] = exp_val;
    __syncthreads();
    for (int s = block_dim >> 1; s > 0; s >>= 1) {
        if (tid < s) {
            reduction[tid] += reduction[tid + s];
        }
        __syncthreads();
    }
    
    if (tid == 0) {
        log_lik[b] = global_max + logf(reduction[0]);
    }
}

class ModelNew(nn.Module):
    """
    HMM Forward algorithm with custom CUDA kernel.
    Computes log-likelihood log p(observations) using the forward recursion
    in log-space for numerical stability.
    """

    def __init__(self, num_states, num_obs):
        super().__init__()
        self.num_states = num_states
        self.num_obs = num_obs
        self.log_init = nn.Parameter(torch.zeros(num_states))
        self.log_trans = nn.Parameter(torch.zeros(num_states, num_states))
        self.log_emit = nn.Parameter(torch.zeros(num_states, num_obs))

    def forward(self, observations: torch.Tensor) -> torch.Tensor:
        """
        Args:
            observations: (B, T) - integer observation indices (long tensor)
        Returns:
            log_lik: (B,) - log p(observations) under the HMM
        """
        B, T = observations.shape
        
        # Ensure inputs are contiguous and on GPU
        obs = observations.contiguous()
        log_pi = self.log_init.contiguous()
        log_A = self.log_trans.contiguous()
        log_B = self.log_emit.contiguous()
        log_lik = torch.empty(B, dtype=torch.float32, device=obs.device)
        
        # Call custom CUDA kernel
        hmm_ext.hmm_forward_cuda(obs, log_pi, log_A, log_B, log_lik)
        
        return log_lik
