import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Stack of planar normalizing flow layers.  Each layer applies the
    invertible transform f(z) = z + u * tanh(w^T z + b), with a
    log-determinant-Jacobian correction.  Used to transform a simple
    base distribution (e.g. standard Gaussian) into a complex posterior.
    """

    def __init__(self, dim, num_flows):
        super().__init__()
        self.dim = dim
        self.num_flows = num_flows
        self.w = nn.ParameterList([nn.Parameter(torch.randn(dim)) for _ in range(num_flows)])
        self.u = nn.ParameterList([nn.Parameter(torch.randn(dim)) for _ in range(num_flows)])
        self.b = nn.ParameterList([nn.Parameter(torch.zeros(1)) for _ in range(num_flows)])

    def _constrain_u(self, w, u):
        """Ensure invertibility: w^T u >= -1."""
        wtu = (w * u).sum()
        m = -1 + F.softplus(wtu)
        u_hat = u + (m - wtu) * w / (w ** 2).sum()
        return u_hat

    def forward(self, z: torch.Tensor) -> tuple:
        """
        Args:
            z: (B, dim) – samples from base distribution
        Returns:
            (z_out, sum_log_det):
                z_out:       (B, dim) – transformed samples
                sum_log_det: (B,)     – total log |det df/dz|
        """
        sum_log_det = torch.zeros(z.shape[0], device=z.device)
        for i in range(self.num_flows):
            w = self.w[i]
            u = self._constrain_u(w, self.u[i])
            b = self.b[i]
            linear = z @ w + b
            h = torch.tanh(linear)
            h_prime = 1.0 - h ** 2
            z = z + u.unsqueeze(0) * h.unsqueeze(-1)
            psi = h_prime.unsqueeze(-1) * w.unsqueeze(0)
            log_det = torch.log(torch.abs(1.0 + psi @ u) + 1e-08)
            sum_log_det = sum_log_det + log_det
        return (z, sum_log_det)

        red1[tid] = sum1; red2[tid] = sum2;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) { red1[tid] += red1[tid + stride]; red2[tid] += red2[tid + stride]; }
            __syncthreads();
        }
        if (tid == 0) {
            float wtu = red1[0];
            float w2 = red2[0];
            float m = -1.0f + softplusf_stable(wtu);
            s_alpha = (m - wtu) / w2;
            s_wuhat = m;
        }
        __syncthreads();

        float dot = 0.0f;
        for (int d = tid; d < D; d += blockDim.x) dot += zsh[d] * w[d];
        red1[tid] = dot;
        __syncthreads();
        for (int stride = blockDim.x >> 1; stride > 0; stride >>= 1) {
            if (tid < stride) red1[tid] += red1[tid + stride];
            __syncthreads();
        }
        if (tid == 0) {
            float h = tanhf(red1[0] + bs[f][0]);
            s_h = h;
            float hp = 1.0f - h*h;
            s_logacc += logf(fabsf(1.0f + hp * s_wuhat) + 1.0e-8f);
        }
        __syncthreads();
        for (int d = tid; d < D; d += blockDim.x) zsh[d] += (u[d] + s_alpha * w[d]) * s_h;
        __syncthreads();
    }
    for (int d = tid; d < D; d += blockDim.x) z_out[row * D + d] = zsh[d];
    if (tid == 0) log_out[row] = s_logacc;
}

std::vector<torch::Tensor> planar_forward(torch::Tensor z,
                                          std::vector<torch::Tensor> w_list,
                                          std::vector<torch::Tensor> u_list,
                                          std::vector<torch::Tensor> b_list) {
    CHECK_CUDA(z);
    auto zc = z.contiguous();
    CHECK_CONTIGUOUS(zc);
    TORCH_CHECK(zc.scalar_type() == at::kFloat, "only float32 supported");
    int64_t B64 = zc.size(0);
    int64_t D64 = zc.size(1);
    int NF = (int)w_list.size();
    TORCH_CHECK((int)u_list.size() == NF && (int)b_list.size() == NF, "parameter list size mismatch");

    auto out = torch::empty_like(zc);
    auto log = torch::empty({B64}, zc.options());
    auto opts_long = torch::TensorOptions().device(zc.device()).dtype(torch::kInt64);
    auto wptr_t = torch::empty({NF}, opts_long);
    auto uptr_t = torch::empty({NF}, opts_long);
    auto bptr_t = torch::empty({NF}, opts_long);

    std::vector<uint64_t> h_w(NF), h_u(NF), h_b(NF);
    for (int i = 0; i < NF; ++i) {
        CHECK_CUDA(w_list[i]); CHECK_CUDA(u_list[i]); CHECK_CUDA(b_list[i]);
        TORCH_CHECK(w_list[i].scalar_type() == at::kFloat && u_list[i].scalar_type() == at::kFloat && b_list[i].scalar_type() == at::kFloat, "only float32 params supported");
        auto wc = w_list[i].contiguous();
        auto uc = u_list[i].contiguous();
        auto bc = b_list[i].contiguous();
        // Parameters are already contiguous in normal use; keep tensors alive through lists.
        if (!w_list[i].is_contiguous()) w_list[i] = wc;
        if (!u_list[i].is_contiguous()) u_list[i] = uc;
        if (!b_list[i].is_contiguous()) b_list[i] = bc;
        h_w[i] = (uint64_t)w_list[i].data_ptr<float>();
        h_u[i] = (uint64_t)u_list[i].data_ptr<float>();
        h_b[i] = (uint64_t)b_list[i].data_ptr<float>();
    }
    cudaMemcpyAsync(wptr_t.data_ptr<int64_t>(), h_w.data(), NF * sizeof(uint64_t), cudaMemcpyHostToDevice, at::cuda::getCurrentCUDAStream());
    cudaMemcpyAsync(uptr_t.data_ptr<int64_t>(), h_u.data(), NF * sizeof(uint64_t), cudaMemcpyHostToDevice, at::cuda::getCurrentCUDAStream());
    cudaMemcpyAsync(bptr_t.data_ptr<int64_t>(), h_b.data(), NF * sizeof(uint64_t), cudaMemcpyHostToDevice, at::cuda::getCurrentCUDAStream());

    int B = (int)B64;
    int D = (int)D64;
    if (D == 48 && NF == 8) {
        planar_kernel_fixed<48,8><<<B, 64, 0, at::cuda::getCurrentCUDAStream()>>>(
            zc.data_ptr<float>(), out.data_ptr<float>(), log.data_ptr<float>(),
            (const float* const*)wptr_t.data_ptr<int64_t>(),
            (const float* const*)uptr_t.data_ptr<int64_t>(),
            (const float* const*)bptr_t.data_ptr<int64_t>(), B);
    } else {
        int threads = 256;
        size_t shmem = (size_t)(D + 2 * threads) * sizeof(float);
        planar_kernel_generic<<<B, threads, shmem, at::cuda::getCurrentCUDAStream()>>>(
            zc.data_ptr<float>(), out.data_ptr<float>(), log.data_ptr<float>(),
            (const float* const*)wptr_t.data_ptr<int64_t>(),
            (const float* const*)uptr_t.data_ptr<int64_t>(),
            (const float* const*)bptr_t.data_ptr<int64_t>(), B, D, NF);
    }
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return {out, log};
}

class ModelNew(nn.Module):
    def __init__(self, dim, num_flows):
        super().__init__()
        self.dim = dim
        self.num_flows = num_flows
        self.w = nn.ParameterList([nn.Parameter(torch.randn(dim)) for _ in range(num_flows)])
        self.u = nn.ParameterList([nn.Parameter(torch.randn(dim)) for _ in range(num_flows)])
        self.b = nn.ParameterList([nn.Parameter(torch.zeros(1)) for _ in range(num_flows)])

    def _constrain_u(self, w, u):
        wtu = (w * u).sum()
        m = -1 + F.softplus(wtu)
        return u + (m - wtu) * w / (w ** 2).sum()

    def forward(self, z: torch.Tensor) -> tuple:
        if z.is_cuda and z.dtype is torch.float32:
            return tuple(_ext.planar_forward(z, list(self.w), list(self.u), list(self.b)))
        # Non-CUDA fallback for completeness (not used in benchmark timing).
        sum_log_det = torch.zeros(z.shape[0], device=z.device, dtype=z.dtype)
        for i in range(self.num_flows):
            w = self.w[i]
            u = self._constrain_u(w, self.u[i])
            b = self.b[i]
            linear = z @ w + b
            h = torch.tanh(linear)
            h_prime = 1.0 - h ** 2
            z = z + u.unsqueeze(0) * h.unsqueeze(-1)
            psi = h_prime.unsqueeze(-1) * w.unsqueeze(0)
            log_det = torch.log(torch.abs(1.0 + psi @ u) + 1e-08)
            sum_log_det = sum_log_det + log_det
        return (z, sum_log_det)
