import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Low-rank expansion of compressed KV activations into multi-head K/V tensors.

    This isolates the MLA-style compressed-cache pattern: a hidden activation is
    projected into a low-rank latent space and then expanded back into separate
    key and value tensors.
    """

    def __init__(self):
        super().__init__()

    def forward(self, hidden: torch.Tensor, w_down: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, model_dim) = hidden.shape
        rank = w_down.shape[1]
        kv_heads = 4
        head_dim = w_up_k.shape[1] // kv_heads
        latent = hidden.reshape(batch_size * seq_len, model_dim) @ w_down
        k = latent @ w_up_k
        v = latent @ w_up_v
        k = k.view(batch_size, seq_len, kv_heads, head_dim)
        v = v.view(batch_size, seq_len, kv_heads, head_dim)
        return torch.stack((k, v), dim=2)

                                          const float* __restrict__ w_down,
                                          const float* __restrict__ w_up_k,
                                          const float* __restrict__ w_up_v,
                                          float* __restrict__ out,
                                          int rows, int M, int R, int N) {
    extern __shared__ float smem[];
    float* hs = smem;
    float* lat = smem + M;

    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    if (row >= rows) return;

    const float* hrow = hidden + (long long)row * M;
    for (int i = tid; i < M; i += blockDim.x) hs[i] = hrow[i];
    __syncthreads();

    for (int r = tid; r < R; r += blockDim.x) {
        float acc = 0.0f;
        for (int m = 0; m < M; ++m) {
            acc = fmaf(hs[m], w_down[(long long)m * R + r], acc);
        }
        lat[r] = acc;
    }
    __syncthreads();

    const int total = 2 * N;
    float* orow = out + (long long)row * total;
    for (int o = tid; o < total; o += blockDim.x) {
        const bool is_v = (o >= N);
        const int col = is_v ? (o - N) : o;
        const float* __restrict__ wu = is_v ? w_up_v : w_up_k;
        float acc = 0.0f;
        for (int r = 0; r < R; ++r) {
            acc = fmaf(lat[r], wu[(long long)r * N + col], acc);
        }
        orow[o] = acc;
    }
}
