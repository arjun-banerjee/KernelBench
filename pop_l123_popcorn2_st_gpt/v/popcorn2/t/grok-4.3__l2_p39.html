import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        assert kernel_size % 2 == 1, 'odd kernel_size keeps T unchanged with symmetric padding'
        assert dilation >= 1
        self.padding = (kernel_size - 1) * dilation // 2
        self.dw = nn.Conv1d(in_channels, in_channels, kernel_size, padding=self.padding, dilation=dilation, groups=in_channels, bias=False)
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

class ModelNew(nn.Module):
    """KernelBench candidate entry point; default delegates to ``Model``."""

    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        self._impl = Model(in_channels, out_channels, kernel_size, dilation=dilation)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self._impl(x)

    const scalar_t* __restrict__ input,
    const scalar_t* __restrict__ dw_weight,
    const scalar_t* __restrict__ pw_weight,
    scalar_t* __restrict__ output,
    int B, int C, int T, int K, int D, int P, int Cout) {
    
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * Cout * T;
    if (idx >= total) return;
    
    int t = idx % T;
    int c_out = (idx / T) % Cout;
    int b = idx / (T * Cout);
    
    scalar_t val = 0;
    for (int c_in = 0; c_in < C; ++c_in) {
        scalar_t sum = 0;
        for (int k = 0; k < K; ++k) {
            int t_in = t + (k - K/2) * D;
            if (t_in >= 0 && t_in < T) {
                scalar_t in_val = input[b * C * T + c_in * T + t_in];
                sum += in_val * dw_weight[c_in * K + k];
            }
        }
        scalar_t silu = sum / (1.0f + expf(-sum));
        val += silu * pw_weight[c_out * C + c_in];
    }
    output[b * Cout * T + c_out * T + t] = val;
}

torch::Tensor fused_conv(torch::Tensor x, torch::Tensor dw_w, torch::Tensor pw_w, int padding, int dilation, int kernel_size) {
    int B = x.size(0);
    int C = x.size(1);
    int T = x.size(2);
    int Cout = pw_w.size(0);
    auto output = torch::empty({B, Cout, T}, x.options());
    
    int total = B * Cout * T;
    const int threads = 256;
    int blocks = (total + threads - 1) / threads;
    
    AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "fused", ([&] {
        fused_dw_pw_silu_kernel<scalar_t><<<blocks, threads>>>(
            x.data_ptr<scalar_t>(),
            dw_w.data_ptr<scalar_t>(),
            pw_w.data_ptr<scalar_t>(),
            output.data_ptr<scalar_t>(),
            B, C, T, kernel_size, dilation, padding, Cout);
    }));
    return output;
}

class ModelNew(nn.Module):
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, dilation: int=1):
        super().__init__()
        assert kernel_size % 2 == 1
        self.padding = (kernel_size - 1) * dilation // 2
        self.dw = nn.Conv1d(in_channels, in_channels, kernel_size, padding=self.padding, dilation=dilation, groups=in_channels, bias=False)
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)
        self.kernel_size = kernel_size
        self.dilation = dilation
        self.fused = module

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.fused.fused_conv(x, self.dw.weight, self.pw.weight, self.padding, self.dilation, self.kernel_size)
