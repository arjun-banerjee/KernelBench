import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Triangular multiplicative update – outgoing edges.
    From AlphaFold2 Evoformer: updates pair (i,j) by aggregating over the
    shared index k via element-wise products of projected pair values along
    edges (i,k) and (j,k).
    """

    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            pair_repr: (B, N, N, pair_dim)
            pair_mask: (B, N, N, 1)
        Returns:
            updated pair representation (B, N, N, pair_dim)
        """
        mask = pair_mask.unsqueeze(-1)
        x = self.layer_norm_in(pair_repr)
        a = self.proj_a(x) * torch.sigmoid(self.gate_a(x)) * mask
        b = self.proj_b(x) * torch.sigmoid(self.gate_b(x)) * mask
        out = torch.einsum('bikh,bjkh->bijh', a, b)
        out = self.layer_norm_out(out)
        out = self.out_proj(out)
        out = torch.sigmoid(self.out_gate(pair_repr)) * out
        return out

    AT_DISPATCH_FLOATING_TYPES(pair_repr.scalar_type(), "fused_output_gate", ([&] {
        fused_output_gate_kernel<scalar_t><<<blocks, threads>>>(
            pair_repr.data_ptr<scalar_t>(),
            gate_weight.data_ptr<scalar_t>(),
            gate_bias.data_ptr<scalar_t>(),
            out.data_ptr<scalar_t>(),
            output.data_ptr<scalar_t>(),
            B, N, dim);
    }));
    C10_CUDA_CHECK(cudaGetLastError());
}

void einsum_triangular_cuda(
    const at::Tensor& a,
    const at::Tensor& b,
    at::Tensor& out) {
    
    const int B = a.size(0);
    const int N = a.size(1);
    const int H = a.size(3);
    
    // Use tiled kernel for better memory access
    const int TILE_SIZE = 32;
    const dim3 blocks(N, N, B * H);
    const int threads = TILE_SIZE;
    
    AT_DISPATCH_FLOATING_TYPES(a.scalar_type(), "einsum_tiled", ([&] {
        einsum_tiled_kernel<scalar_t, TILE_SIZE><<<blocks, threads>>>(
            a.data_ptr<scalar_t>(),
            b.data_ptr<scalar_t>(),
            out.data_ptr<scalar_t>(),
            B, N, H);
    }));
    C10_CUDA_CHECK(cudaGetLastError());
}

# Compile the CUDA extension
try:
    triangular_ops = load_inline(
        name='triangular_ops',
        cpp_sources='',
        cuda_sources=cuda_source,
        functions=['fused_proj_gate_mask_cuda', 'layer_norm_cuda', 
                   'fused_output_gate_cuda', 'einsum_triangular_cuda'],
        verbose=False,
        extra_cuda_cflags=['-O3', '-lineinfo']
    )
except Exception as e:
    print(f"Failed to compile CUDA extension: {e}")
    triangular_ops = None


class ModelNew(nn.Module):
    """
    Triangular multiplicative update - outgoing edges with custom CUDA kernels.
    """

    def __init__(self, pair_dim, hidden_dim):
        super().__init__()
        self.pair_dim = pair_dim
        self.hidden_dim = hidden_dim
        
        # LayerNorm parameters (using nn.Parameter to match original)
        self.layer_norm_in = nn.LayerNorm(pair_dim)
        self.layer_norm_out = nn.LayerNorm(hidden_dim)
        
        # Linear layers for projections
        self.proj_a = nn.Linear(pair_dim, hidden_dim)
        self.gate_a = nn.Linear(pair_dim, hidden_dim)
        self.proj_b = nn.Linear(pair_dim, hidden_dim)
        self.gate_b = nn.Linear(pair_dim, hidden_dim)
        self.out_proj = nn.Linear(hidden_dim, pair_dim)
        self.out_gate = nn.Linear(pair_dim, pair_dim)

    def forward(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """
        Args:
            pair_repr: (B, N, N, pair_dim)
            pair_mask: (B, N, N)
        Returns:
            updated pair representation (B, N, N, pair_dim)
        """
        if triangular_ops is None or not pair_repr.is_cuda:
            # Fallback to reference implementation
            return self._forward_reference(pair_repr, pair_mask)
        
        B, N, _, _ = pair_repr.shape
        
        # Squeeze mask to B,N,N for custom kernel
        mask = pair_mask  # Already B,N,N
        
        # LayerNorm input
        x = self.layer_norm_in(pair_repr)
        
        # Fused proj_a * sigmoid(gate_a) * mask
        a = torch.empty(B, N, N, self.hidden_dim, device=pair_repr.device, dtype=pair_repr.dtype)
        triangular_ops.fused_proj_gate_mask_cuda(
            x, 
            self.proj_a.weight, self.proj_a.bias,
            self.gate_a.weight, self.gate_a.bias,
            mask,
            a
        )
        
        # Fused proj_b * sigmoid(gate_b) * mask
        b = torch.empty(B, N, N, self.hidden_dim, device=pair_repr.device, dtype=pair_repr.dtype)
        triangular_ops.fused_proj_gate_mask_cuda(
            x,
            self.proj_b.weight, self.proj_b.bias,
            self.gate_b.weight, self.gate_b.bias,
            mask,
            b
        )
        
        # Einsum: bikh,bjkh->bijh
        out = torch.empty(B, N, N, self.hidden_dim, device=pair_repr.device, dtype=pair_repr.dtype)
        triangular_ops.einsum_triangular_cuda(a, b, out)
        
        # LayerNorm output
        out = self.layer_norm_out(out)
        
        # Output projection
        out = self.out_proj(out)
        
        # Fused output gate: sigmoid(gate(pair_repr)) * out
        result = torch.empty(B, N, N, self.pair_dim, device=pair_repr.device, dtype=pair_repr.dtype)
        triangular_ops.fused_output_gate_cuda(
            pair_repr,
            self.out_gate.weight, self.out_gate.bias,
            out,
            result
        )
        
        return result
    
    def _forward_reference(self, pair_repr: torch.Tensor, pair_mask: torch.Tensor) -> torch.Tensor:
        """Reference implementation for fallback."""
        mask = pair_mask.unsqueeze(-1)
        x = self.layer_norm_in(pair_repr)
        a = self.proj_a(x) * torch.sigmoid(self.gate_a(x)) * mask
        b = self.proj_b(x) * torch.sigmoid(self.gate_b(x)) * mask
        out = torch.einsum('bikh,bjkh->bijh', a, b)
        out = self.layer_norm_out(out)
        out = self.out_proj(out)
        out = torch.sigmoid(self.out_gate(pair_repr)) * out
        return out
