import torch
import torch.nn as nn
import math

class Model(nn.Module):
    """
    Radial Basis Function (RBF) expansion of interatomic distances.
    Standard featurization in molecular graph neural networks (SchNet,
    DimeNet, GemNet) that converts scalar distances into a feature vector
    of Gaussian-basis evaluations.
    """

    def __init__(self, num_rbf=64, cutoff=5.0):
        super().__init__()
        self.num_rbf = num_rbf
        self.cutoff = cutoff
        offsets = torch.linspace(0.0, cutoff, num_rbf)
        self.register_buffer('offsets', offsets)
        coeff = -0.5 / (offsets[1] - offsets[0]).item() ** 2
        self.coeff = coeff

    def forward(self, distances: torch.Tensor) -> torch.Tensor:
        """
        Args:
            distances: (B, N, N) – pairwise distance matrix (or (B, E) edge list)
        Returns:
            rbf_features: (*distances.shape, num_rbf) – expanded features
        """
        d_scaled = distances.unsqueeze(-1)
        envelope = 0.5 * (torch.cos(math.pi * d_scaled / self.cutoff) + 1.0)
        envelope = envelope * (d_scaled < self.cutoff).float()
        rbf = torch.exp(self.coeff * (d_scaled - self.offsets) ** 2)
        return rbf * envelope
num_atoms = 256
num_rbf = 64
cutoff = 5.0

def get_inputs():

    distances = torch.sqrt((diff ** 2).sum(-1) + 1e-08)
    return [distances]

def get_init_inputs():
    return [num_rbf, cutoff]

#include <ATen/cuda/CUDAContext.h>

#include <vector>

static constexpr float PI_F = 3.14159265358979323846f;

__global__ __launch_bounds__(256, 4)
void rbf_warp_kernel_64(const float* __restrict__ distances,
                        const float* __restrict__ offsets,
                        float* __restrict__ out,
                        long long n,
                        float coeff,
                        float cutoff,
                        float inv_cutoff) {
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int lane = threadIdx.x & 31;
    const long long i = ((long long)tid) >> 5;  // one warp per input distance
    if (i >= n) return;

    float d = 0.0f;
    if (lane == 0) d = distances[i];
    d = __shfl_sync(0xffffffffu, d, 0);

    float env = 0.0f;
    if (d < cutoff) {
        env = 0.5f * (cosf(PI_F * d * inv_cutoff) + 1.0f);
    }

    const long long base = i * 64LL;
    float off0 = offsets[lane];
    float x0 = d - off0;
    out[base + lane] = env * expf(coeff * x0 * x0);

    float off1 = offsets[lane + 32];
    float x1 = d - off1;
    out[base + lane + 32] = env * expf(coeff * x1 * x1);
}

void rbf_warp_kernel_generic(const float* __restrict__ distances,
                             const float* __restrict__ offsets,
                             float* __restrict__ out,
                             long long n,
                             int nr,
                             float coeff,
                             float cutoff,
                             float inv_cutoff) {
    const int tid = blockIdx.x * blockDim.x + threadIdx.x;
    const int lane = threadIdx.x & 31;
    const long long i = ((long long)tid) >> 5;  // one warp per input distance
    if (i >= n) return;

    float d = 0.0f;
    if (lane == 0) d = distances[i];
    d = __shfl_sync(0xffffffffu, d, 0);

    float env = 0.0f;
    if (d < cutoff) {
        env = 0.5f * (cosf(PI_F * d * inv_cutoff) + 1.0f);

    const long long base = i * (long long)nr;
    for (int k = lane; k < nr; k += 32) {
        float x = d - offsets[k];
        out[base + k] = env * expf(coeff * x * x);
    }
}

    def __init__(self, num_rbf=64, cutoff=5.0):
        super().__init__()
        self.num_rbf = num_rbf
        self.cutoff = cutoff
        offsets = torch.linspace(0.0, cutoff, num_rbf)
