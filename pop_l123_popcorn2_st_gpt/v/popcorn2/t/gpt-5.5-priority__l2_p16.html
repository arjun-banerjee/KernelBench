import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Hamiltonian Monte Carlo (HMC) proposal using leapfrog integration.
    Simulates Hamiltonian dynamics on a batched set of particles to
    produce proposals with high acceptance probability.  The target
    distribution is a standard Gaussian for benchmarking.
    """

    def __init__(self, dim, num_leapfrog_steps=10, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.num_leapfrog_steps = num_leapfrog_steps
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))

    def _grad_log_prob(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -(x - self.mu)

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (x ** 2).sum(dim=-1)

    def forward(self, q: torch.Tensor) -> tuple:
        """
        Args:
            q: (B, dim) – current positions
        Returns:
            (q_new, accepted):
                q_new:    (B, dim)
                accepted: (B,)
        """
        eps = self.step_size
        p = torch.randn_like(q)
        current_p = p.clone()
        current_q = q.clone()
        p = p + 0.5 * eps * self._grad_log_prob(q)
        for _ in range(self.num_leapfrog_steps - 1):
            q = q + eps * p
            p = p + eps * self._grad_log_prob(q)
        q = q + eps * p
        p = p + 0.5 * eps * self._grad_log_prob(q)
        p = -p
        current_H = -self._log_prob(current_q) + 0.5 * (current_p ** 2).sum(-1)
        proposed_H = -self._log_prob(q) + 0.5 * (p ** 2).sum(-1)
        log_alpha = current_H - proposed_H
        log_u = torch.log(torch.rand(q.shape[0], device=q.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        q_new = torch.where(accepted.unsqueeze(-1), q, current_q)
        return (q_new, accepted)

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>
#include <math.h>

#ifndef BLOCK_SIZE
#define BLOCK_SIZE 128
#endif

__global__ void hmc_kernel_f32(const float* __restrict__ q0,
                               const float* __restrict__ p0,
                               const float* __restrict__ uni,
                               float* __restrict__ q_out,
                               bool* __restrict__ accepted,
                               int B, int D, int L, float eps) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    if (b >= B) return;

    __shared__ float s_cur[BLOCK_SIZE];
    __shared__ float s_prop[BLOCK_SIZE];
    __shared__ int s_accept;

    float cur_sum = 0.0f;
    float prop_sum = 0.0f;
    int base = b * D;

    // Compute the leapfrog trajectory and the two quadratic Hamiltonians.
    for (int c = tid; c < D; c += BLOCK_SIZE) {
        int idx = base + c;
        float q = q0[idx];
        float p = p0[idx];

        cur_sum += q * q + p * p;

        // p = p - 0.5*eps*q
        p = p + (-0.5f * eps) * q;

        // for _ in range(num_steps - 1): q += eps*p; p -= eps*q
        for (int it = 0; it < L - 1; ++it) {
            q = q + eps * p;
            p = p + (-eps) * q;
        }

        // final full q step and half p step.  The subsequent p=-p does not
        // change p^2, so it is omitted for the Hamiltonian.
        q = q + eps * p;
        p = p + (-0.5f * eps) * q;

        prop_sum += q * q + p * p;
    }

    s_cur[tid] = cur_sum;
    s_prop[tid] = prop_sum;
    __syncthreads();

    // fixed-size block reduction; D is small (80 in the benchmark), but this
    // also works for larger D through the per-thread strided loop above.
    #pragma unroll
    for (int stride = BLOCK_SIZE / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_cur[tid] += s_cur[tid + stride];
            s_prop[tid] += s_prop[tid + stride];
        }
        __syncthreads();
    }

    if (tid == 0) {
        float log_alpha = 0.5f * (s_cur[0] - s_prop[0]);
        float uu = uni[b];
        if (uu < 1.0e-10f) uu = 1.0e-10f;
        int acc = (logf(uu) < log_alpha);
        s_accept = acc;
        accepted[b] = (bool)acc;
    }
    __syncthreads();

    // Write selected proposal.  Recomputing q is cheaper than a second global
    // temporary and keeps the whole HMC update/acceptance in one CUDA launch.
    int acc = s_accept;
    for (int c = tid; c < D; c += BLOCK_SIZE) {
        int idx = base + c;
        float q_initial = q0[idx];
        float q = q_initial;
        float p = p0[idx];

        p = p + (-0.5f * eps) * q;
        for (int it = 0; it < L - 1; ++it) {
            q = q + eps * p;
            p = p + (-eps) * q;
        }
        q = q + eps * p;

        q_out[idx] = acc ? q : q_initial;
    }
}

std::vector<torch::Tensor> hmc_forward_cuda(torch::Tensor q, torch::Tensor p, torch::Tensor u,
                                            int64_t num_steps, double step_size) {
    const int B = (int)q.size(0);
    const int D = (int)q.size(1);
    auto q_out = torch::empty_like(q);
    auto acc = torch::empty({B}, q.options().dtype(torch::kBool));

                                        B, D, (int)num_steps, (float)step_size);
    }

}
