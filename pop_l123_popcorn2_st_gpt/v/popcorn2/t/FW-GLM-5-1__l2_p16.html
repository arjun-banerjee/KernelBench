import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Hamiltonian Monte Carlo (HMC) proposal using leapfrog integration.
    Simulates Hamiltonian dynamics on a batched set of particles to
    produce proposals with high acceptance probability.  The target
    distribution is a standard Gaussian for benchmarking.
    """

    def __init__(self, dim, num_leapfrog_steps=10, step_size=0.1):
        super().__init__()
        self.dim = dim
        self.num_leapfrog_steps = num_leapfrog_steps
        self.step_size = step_size
        self.register_buffer('mu', torch.zeros(dim))

    def _grad_log_prob(self, x: torch.Tensor) -> torch.Tensor:
        """Gradient of log p(x) for N(0, I)."""
        return -(x - self.mu)

    def _log_prob(self, x: torch.Tensor) -> torch.Tensor:
        return -0.5 * (x ** 2).sum(dim=-1)

    def forward(self, q: torch.Tensor) -> tuple:
        """
        Args:
            q: (B, dim) – current positions
        Returns:
            (q_new, accepted):
                q_new:    (B, dim)
                accepted: (B,)
        """
        eps = self.step_size
        p = torch.randn_like(q)
        current_p = p.clone()
        current_q = q.clone()
        p = p + 0.5 * eps * self._grad_log_prob(q)
        for _ in range(self.num_leapfrog_steps - 1):
            q = q + eps * p
            p = p + eps * self._grad_log_prob(q)
        q = q + eps * p
        p = p + 0.5 * eps * self._grad_log_prob(q)
        p = -p
        current_H = -self._log_prob(current_q) + 0.5 * (current_p ** 2).sum(-1)
        proposed_H = -self._log_prob(q) + 0.5 * (p ** 2).sum(-1)
        log_alpha = current_H - proposed_H
        log_u = torch.log(torch.rand(q.shape[0], device=q.device).clamp(min=1e-10))
        accepted = log_u < log_alpha
        q_new = torch.where(accepted.unsqueeze(-1), q, current_q)
        return (q_new, accepted)

    const float* __restrict__ q_in,
    float* __restrict__ q_out,
    bool* __restrict__ accepted,
    unsigned long long seed,
    int batch_size,
    int dim,
    int num_leapfrog_steps,
    float step_size
) {
    int b = blockIdx.x;
    int tid = threadIdx.x;
    
    if (b >= batch_size) return;
    
    // Shared memory: q[dim], p[dim], current_q[dim], current_p[dim]
    extern __shared__ float shared_data[];
    float* q = shared_data;
    float* p = q + dim;
    float* current_q = p + dim;
    float* current_p = current_q + dim;
    
    // Initialize RNG state - unique per thread
    curandState state;
    curand_init(seed, b * blockDim.x + tid, 0, &state);
    
    // Parallel load current position and generate momentum
    for (int d = tid; d < dim; d += blockDim.x) {
        int idx = b * dim + d;
        float q_val = q_in[idx];
        q[d] = q_val;
        current_q[d] = q_val;
        float p_val = curand_normal(&state);
        p[d] = p_val;
        current_p[d] = p_val;
    }
    __syncthreads();
    
    float eps = step_size;
    
    // Half step for momentum at start
    for (int d = tid; d < dim; d += blockDim.x) {
        p[d] = p[d] + 0.5f * eps * (-q[d]);
    }
    __syncthreads();
    
    // Leapfrog integration steps
    for (int step = 0; step < num_leapfrog_steps - 1; step++) {
        // Full position update
        for (int d = tid; d < dim; d += blockDim.x) {
            q[d] = q[d] + eps * p[d];
        }
        __syncthreads();
        
        // Full momentum update
        for (int d = tid; d < dim; d += blockDim.x) {
            p[d] = p[d] + eps * (-q[d]);
        }
        __syncthreads();
    }
    
    // Final full position update
    for (int d = tid; d < dim; d += blockDim.x) {
        q[d] = q[d] + eps * p[d];
    }
    __syncthreads();
    
    // Half step for momentum at end
    for (int d = tid; d < dim; d += blockDim.x) {
        p[d] = p[d] + 0.5f * eps * (-q[d]);
    }
    __syncthreads();
    
    // Negate momentum
    for (int d = tid; d < dim; d += blockDim.x) {
        p[d] = -p[d];
    }
    __syncthreads();
    
    // Compute local Hamiltonian contributions
    float local_H_current = 0.0f;
    float local_H_proposed = 0.0f;
    for (int d = tid; d < dim; d += blockDim.x) {
        local_H_current += 0.5f * current_q[d] * current_q[d];
        local_H_current += 0.5f * current_p[d] * current_p[d];
        local_H_proposed += 0.5f * q[d] * q[d];
        local_H_proposed += 0.5f * p[d] * p[d];
    }
    
    // Block reduction using warp shuffles
    for (int offset = 16; offset > 0; offset >>= 1) {
        local_H_current += __shfl_down_sync(0xffffffff, local_H_current, offset);
        local_H_proposed += __shfl_down_sync(0xffffffff, local_H_proposed, offset);
    }
    
    __shared__ float H_current_shared[32];
    __shared__ float H_proposed_shared[32];
    
    int lane = tid & 31;
    int warp_id = tid >> 5;
    
    if (lane == 0) {
        H_current_shared[warp_id] = local_H_current;
        H_proposed_shared[warp_id] = local_H_proposed;
    }
    __syncthreads();
    
    // Final reduction by first warp
    int num_warps = (blockDim.x + 31) / 32;
    float final_H_current = 0.0f;
    float final_H_proposed = 0.0f;
    
    if (tid < num_warps) {
        final_H_current = H_current_shared[tid];
        final_H_proposed = H_proposed_shared[tid];
        for (int offset = num_warps / 2; offset > 0; offset >>= 1) {
            final_H_current += __shfl_down_sync(0xffffffff, final_H_current, offset);
            final_H_proposed += __shfl_down_sync(0xffffffff, final_H_proposed, offset);
        }
    }
    
    if (tid == 0) {
        float log_alpha = final_H_current - final_H_proposed;
        float u = curand_uniform(&state);
        float log_u = logf(fmaxf(1e-10f, u));
        bool accept = log_u < log_alpha;
        accepted[b] = accept;
    }
    __syncthreads();
    
    // Store result
    bool accept = accepted[b];
    for (int d = tid; d < dim; d += blockDim.x) {
        int idx = b * dim + d;
        q_out[idx] = accept ? q[d] : current_q[d];
    }
}

std::tuple<torch::Tensor, torch::Tensor> hmc_forward(
    torch::Tensor q_in,
    int num_leapfrog_steps,
    float step_size
) {
    const int batch_size = q_in.size(0);
    const int dim = q_in.size(1);
    
    auto options = q_in.options();
    auto q_out = torch::empty({batch_size, dim}, options);
    auto accepted = torch::empty({batch_size}, options.dtype(torch::kBool));
    
    unsigned long long seed = clock64();
    
    const int threads = 64;  // Good for dim=80
    const int blocks = batch_size;
    const int shared_mem = 4 * dim * sizeof(float);
    
    hmc_kernel<<<blocks, threads, shared_mem>>>(
        q_in.data_ptr<float>(),
        q_out.data_ptr<float>(),
        accepted.data_ptr<bool>(),
        seed,
        batch_size,
        dim,
        num_leapfrog_steps,
        step_size
    );
    
    return std::make_tuple(q_out, accepted);
}
