With identical inputs on every rank, a symmetric equal-split exchange is an identity on the
concatenated layout; we multiply by a learnable scalar so the module is non-trivial.

import torch
import torch.distributed as dist
import torch.nn as nn
from kernelbench.distributed_collectives import default_device, get_world_size, is_distributed_run, maybe_init_process_group
flat_dim = 64

class Model(nn.Module):

    def __init__(self):
        super().__init__()
        self.scale = nn.Parameter(torch.tensor(1.25))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        maybe_init_process_group()
        ws = get_world_size()
        (b, d) = x.shape
        assert d == flat_dim, f'expected last dim {flat_dim}, got {d}'
        if not is_distributed_run() or ws == 1:
            return x * self.scale

        assert chunk * ws == d, 'flat_dim must be divisible by world_size'

        sizes = [chunk * b] * ws
        dist.all_to_all_single(out_flat, flat, output_split_sizes=sizes, input_split_sizes=sizes)
        return out_flat.reshape(b, d) * self.scale

#include <torch/extension.h>

#include <cuda.h>

__global__ void scale64_kernel(const float* __restrict__ x, const float* __restrict__ s, float* __restrict__ y, int n4) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n4) {
        float sc = __ldg(s);
        const float4* __restrict__ x4 = reinterpret_cast<const float4*>(x);
        float4* __restrict__ y4 = reinterpret_cast<float4*>(y);
        float4 v = x4[i];
        v.x *= sc; v.y *= sc; v.z *= sc; v.w *= sc;
        y4[i] = v;
    }
}

__global__ void scale_kernel(const float* __restrict__ x, const float* __restrict__ s, float* __restrict__ y, int n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) y[i] = x[i] * __ldg(s);
}
