import torch
import torch.nn as nn

class Model(nn.Module):
    """
    **Fused probability-flow ODE integration** (flow-matching style) in one ``forward``.

    Models the ODE ``dx/dt = v_θ(x, t)`` on ``t ∈ [0, 1]`` with a learned velocity
    field. A **single fused pass** unrolls **fixed-step Euler** integration for
    ``num_steps`` substeps—typical for **real-time trajectory / action generation**
    where latency is dominated by repeated ``v_θ`` evaluations.

    This is the continuous-time analogue of a rectified-flow / conditional-flow
    sampler: one ``forward`` returns an approximate **x(1)** from **x(0)** without a
    Python loop across separate module calls (fusion target for the integrator +
    field).
    """

    def __init__(self, dim: int, num_steps: int):
        super().__init__()
        assert dim >= 1 and num_steps >= 1
        self.dim = dim
        self.num_steps = num_steps
        self.register_buffer('dt', torch.tensor(1.0 / float(num_steps)))
        self.velocity = nn.Sequential(nn.Linear(dim + 1, dim * 4), nn.SiLU(), nn.Linear(dim * 4, dim * 4), nn.SiLU(), nn.Linear(dim * 4, dim))

    def forward(self, x0: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x0: (B, dim) initial state at ``t = 0`` (e.g. noise or anchor trajectory).
        Returns:
            (B, dim) approximate state at ``t = 1`` after fused Euler integration.
        """
        (B, d) = x0.shape
        assert d == self.dim
        x = x0
        dt = self.dt.to(dtype=x0.dtype, device=x0.device)
        for k in range(self.num_steps):
            t = (float(k) + 0.5) / float(self.num_steps)
            t_b = torch.full((B, 1), t, device=x0.device, dtype=x0.dtype)
            inp = torch.cat([x, t_b], dim=-1)
            v = self.velocity(inp)
            x = x + dt * v
        return x

        super().__init__()
        self._impl = Model(dim, num_steps)

    def forward(self, x0: torch.Tensor) -> torch.Tensor:
        return self._impl(x0)

        __syncthreads();
        
        // Euler step
        for (int i = thread_id; i < dim; i += num_threads) {
            x[i] += dt * v[i];
        }
        __syncthreads();
    }
    
    // Output
    for (int i = thread_id; i < dim; i += num_threads) {
        output[batch_idx * dim + i] = x[i];
    }
}

    size_t shared_mem_opt = (shared_mem_size + w1_size + hidden_size) * sizeof(float);
    
    if (shared_mem_opt < 48000) {  // Less than 48KB
        threads_per_block = 128;
        flow_matching_ode_kernel_opt<<<blocks, threads_per_block, shared_mem_opt>>>(
            x0.data_ptr<float>(),

class ModelNew(nn.Module):
    """
    Fused probability-flow ODE integration with custom CUDA kernel.
    
    Replaces the Python loop with a single kernel that performs all
    num_steps Euler iterations while keeping state in shared memory.
    """

    def __init__(self, dim: int, num_steps: int):
        super().__init__()
        assert dim >= 1 and num_steps >= 1
        self.dim = dim
        self.num_steps = num_steps
        self.hidden_size = dim * 4
        
        # Store dt as a buffer for consistency with original model
        self.register_buffer('dt', torch.tensor(1.0 / float(num_steps)))
        
        # Define the velocity network layers
        # Structure: Linear(dim+1, dim*4) -> SiLU -> Linear(dim*4, dim*4) -> SiLU -> Linear(dim*4, dim)
        self.velocity = nn.Sequential(
            nn.Linear(dim + 1, self.hidden_size),
            nn.SiLU(),
            nn.Linear(self.hidden_size, self.hidden_size),
            nn.SiLU(),
            nn.Linear(self.hidden_size, dim)
        )

    def forward(self, x0: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x0: (B, dim) initial state at t = 0
        Returns:
            (B, dim) approximate state at t = 1 after fused Euler integration
        """
        B, d = x0.shape
        assert d == self.dim
        
        # For small batch sizes or when not on CUDA, fall back to PyTorch
        if not x0.is_cuda or B < 4:
            return self._forward_torch(x0)
        
        # Ensure float32 for CUDA kernel
        input_dtype = x0.dtype
        if input_dtype != torch.float32:
            x0 = x0.float()
        
        # Extract weights from velocity network
        # velocity[0] = Linear(dim+1, hidden_size)
        # velocity[2] = Linear(hidden_size, hidden_size)
        # velocity[4] = Linear(hidden_size, dim)
        w1 = self.velocity[0].weight.contiguous()  # (hidden_size, dim+1)
        b1 = self.velocity[0].bias.contiguous()     # (hidden_size,)
        w2 = self.velocity[2].weight.contiguous()   # (hidden_size, hidden_size)
        b2 = self.velocity[2].bias.contiguous()     # (hidden_size,)
        w3 = self.velocity[4].weight.contiguous()    # (dim, hidden_size)
        b3 = self.velocity[4].bias.contiguous()     # (dim,)
        
        # Run fused CUDA kernel
        output = flow_matching_ext.flow_matching_ode_forward(
            x0, w1, b1, w2, b2, w3, b3,
            self.dim, self.hidden_size, self.num_steps, self.dt.item()
        )
        
        # Restore original dtype if needed
        if input_dtype != torch.float32:
            output = output.to(input_dtype)
        
        return output
    
    def _forward_torch(self, x0: torch.Tensor) -> torch.Tensor:
        """Fallback PyTorch implementation for non-CUDA tensors."""
        B, d = x0.shape
        x = x0
        dt = self.dt.to(dtype=x0.dtype, device=x0.device)
        
        for k in range(self.num_steps):
            t = (float(k) + 0.5) / float(self.num_steps)
            t_b = torch.full((B, 1), t, device=x0.device, dtype=x0.dtype)
            inp = torch.cat([x, t_b], dim=-1)
            v = self.velocity(inp)
            x = x + dt * v
        
        return x
