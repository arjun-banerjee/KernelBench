import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void scatter_add_kernel(const int32_t* __restrict__ dst_idx,
                                   const float* __restrict__ edge_feat,
                                   float* __restrict__ out,
                                   int num_edges,
                                   int feat_dim) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = num_edges * feat_dim;
    if (idx < total) {
        int edge = idx / feat_dim;
        int f = idx % feat_dim;
        int32_t dst = dst_idx[edge];
        atomicAdd(&out[dst * feat_dim + f], edge_feat[edge * feat_dim + f]);
    }
}

torch::Tensor coo_scatter_add(torch::Tensor dst_idx, torch::Tensor edge_feat, int64_t num_nodes) {
    int num_edges = dst_idx.size(0);
    int feat_dim = edge_feat.size(1);
    auto out = torch::zeros({num_nodes, feat_dim}, edge_feat.options());
    int threads = 256;
    int blocks = (num_edges * feat_dim + threads - 1) / threads;
    scatter_add_kernel<<<blocks, threads>>>(dst_idx.data_ptr<int32_t>(),
                                            edge_feat.data_ptr<float>(),
                                            out.data_ptr<float>(),
                                            num_edges,
                                            feat_dim);
    return out;
}
