import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Reparameterization trick for differentiable sampling from a diagonal
    Gaussian.  Separates stochasticity from parameters so that gradients
    flow through mu and log_var.  Also demonstrates a "mixture of
    Gaussians" reparameterization with learnable component weights.
    """

    def __init__(self, latent_dim, num_components=1):
        super().__init__()
        self.latent_dim = latent_dim
        self.num_components = num_components
        if num_components > 1:
            self.logits = nn.Parameter(torch.zeros(num_components))

    def forward(self, mu: torch.Tensor, log_var: torch.Tensor) -> tuple:
        """
        Args:
            mu:      (B, K, D) or (B, D) – means
            log_var: (B, K, D) or (B, D) – log-variances
        Returns:
            (z, log_q_z):
                z:       (B, D) – sampled latents
                log_q_z: (B,)   – log-density of sample under q
        """
        std = torch.exp(0.5 * log_var)
        if self.num_components > 1 and mu.dim() == 3:
            weights = torch.softmax(self.logits, dim=0)
            comp = torch.multinomial(weights.expand(mu.shape[0], -1), 1).squeeze(-1)
            batch_idx = torch.arange(mu.shape[0], device=mu.device)
            mu_k = mu[batch_idx, comp]
            std_k = std[batch_idx, comp]
            z = mu_k + std_k * torch.randn_like(mu_k)
            log_probs = -0.5 * ((z.unsqueeze(1) - mu) / std) ** 2 - log_var * 0.5 - 0.5 * torch.log(torch.tensor(2 * 3.14159265))
            log_probs = log_probs.sum(-1) + weights.log().unsqueeze(0)
            log_q_z = torch.logsumexp(log_probs, dim=1)
        else:
            if mu.dim() == 3:
                mu = mu[:, 0]
                std = std[:, 0]
                log_var = log_var[:, 0]
            z = mu + std * torch.randn_like(mu)
            log_q_z = (-0.5 * ((z - mu) / std) ** 2 - 0.5 * log_var - 0.5 * torch.log(torch.tensor(2 * 3.14159265))).sum(-1)
        return (z, log_q_z)

#include <cmath>

constexpr float LOG_2PI = 1.8378770064093453f;

// Single component: compute z and intermediate log probabilities
__global__ void single_comp_forward_kernel(
    const float* __restrict__ mu,
    const float* __restrict__ log_var,
    const float* __restrict__ noise,
    float* __restrict__ z,
    float* __restrict__ log_probs_per_dim,
    int total_elements,
    int latent_dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (idx < total_elements) {
        float m = mu[idx];
        float lv = log_var[idx];
        float s = expf(0.5f * lv);
        float n = noise[idx];
        
        float z_val = m + s * n;
        z[idx] = z_val;
        
        // (z - mu) / std = n
        float log_prob = -0.5f * n * n - 0.5f * lv - 0.5f * LOG_2PI;
        log_probs_per_dim[idx] = log_prob;
    }
}

__global__ void reduce_log_probs_kernel(
    const float* __restrict__ log_probs_per_dim,
    float* __restrict__ log_q_z,
    int batch_size,
    int latent_dim
) {
    extern __shared__ float s_data[];
    
    int batch_idx = blockIdx.x;
    int tid = threadIdx.x;
    int blockSize = blockDim.x;
    
    float sum = 0.0f;
    for (int i = tid; i < latent_dim; i += blockSize) {
        sum += log_probs_per_dim[batch_idx * latent_dim + i];
    }
    s_data[tid] = sum;
    __syncthreads();
    
    for (int stride = blockSize / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            s_data[tid] += s_data[tid + stride];
        }
        __syncthreads();
    }
    
    if (tid == 0) {
        log_q_z[batch_idx] = s_data[0];
    }
}

// Multi-component: compute z by sampling from selected component
__global__ void compute_z_multi_kernel(
    const float* __restrict__ mu,
    const float* __restrict__ log_var,
    const int* __restrict__ comp_indices,
    const float* __restrict__ noise,
    float* __restrict__ z,
    int batch_size,
    int num_components,
    int latent_dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * latent_dim;
    
    if (idx < total) {
        int b = idx / latent_dim;
        int d = idx % latent_dim;
        int c = comp_indices[b];
        
        int src_idx = b * num_components * latent_dim + c * latent_dim + d;
        
        float m = mu[src_idx];
        float lv = log_var[src_idx];
        float s = expf(0.5f * lv);
        float n = noise[idx];
        
        z[idx] = m + s * n;
    }
}

// Multi-component: compute log_probs for all components
__global__ void multi_comp_log_probs_kernel(
    const float* __restrict__ mu,
    const float* __restrict__ std,
    const float* __restrict__ log_var,
    const float* __restrict__ z,
    float* __restrict__ log_probs,
    int batch_size,
    int num_components,
    int latent_dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * num_components * latent_dim;
    
    if (idx < total) {
        int b = idx / (num_components * latent_dim);
        int remaining = idx % (num_components * latent_dim);
        int c = remaining / latent_dim;
        int d = remaining % latent_dim;
        
        int z_idx = b * latent_dim + d;
        
        float m = mu[idx];
        float s = std[idx];
        float lv = log_var[idx];
        float z_val = z[z_idx];
        
        float diff = (z_val - m) / s;
        float log_prob = -0.5f * diff * diff - 0.5f * lv - 0.5f * LOG_2PI;
        log_probs[idx] = log_prob;
    }
}

__global__ void sum_and_add_weights_kernel(
    const float* __restrict__ log_probs,
    const float* __restrict__ log_weights,
    float* __restrict__ log_probs_summed,
    int batch_size,
    int num_components,
    int latent_dim
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = batch_size * num_components;
    
    if (idx < total) {
        int b = idx / num_components;
        int c = idx % num_components;
        
        float sum = 0.0f;
        for (int d = 0; d < latent_dim; d++) {
            int prob_idx = idx * latent_dim + d;
            sum += log_probs[prob_idx];
        }
        
        log_probs_summed[idx] = sum + log_weights[c];
    }
}

__global__ void logsumexp_components_kernel(
    const float* __restrict__ log_probs_summed,
    float* __restrict__ log_q_z,
    int batch_size,
    int num_components
) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (b < batch_size) {
        float max_val = log_probs_summed[b * num_components];
        for (int c = 1; c < num_components; c++) {
            float val = log_probs_summed[b * num_components + c];
            if (val > max_val) max_val = val;
        }
        
        float sum_exp = 0.0f;
        for (int c = 0; c < num_components; c++) {
            sum_exp += expf(log_probs_summed[b * num_components + c] - max_val);
        }
        
        log_q_z[b] = max_val + logf(sum_exp);
    }
}

std::tuple<torch::Tensor, torch::Tensor> reparam_single_cuda(
    torch::Tensor mu,
    torch::Tensor log_var) {
    
    mu = mu.contiguous();
    log_var = log_var.contiguous();
    
    int batch_size = mu.size(0);
    int latent_dim = mu.size(1);
    int total = batch_size * latent_dim;
    
    auto noise = torch::randn_like(mu);
    auto z = torch::empty_like(mu);
    auto log_probs_per_dim = torch::empty_like(mu);
    auto log_q_z = torch::empty({batch_size}, mu.options());
    
    const int threads = 256;
    const int blocks = (total + threads - 1) / threads;
    
    single_comp_forward_kernel<<<blocks, threads>>>(
        mu.data_ptr<float>(),
        log_var.data_ptr<float>(),
        noise.data_ptr<float>(),
        z.data_ptr<float>(),
        log_probs_per_dim.data_ptr<float>(),
        total,
        latent_dim
    );
    
    int blockSize = 256;
    size_t shared_mem = blockSize * sizeof(float);
    reduce_log_probs_kernel<<<batch_size, blockSize, shared_mem>>>(
        log_probs_per_dim.data_ptr<float>(),
        log_q_z.data_ptr<float>(),
        batch_size,
        latent_dim
    );
    
    return std::make_tuple(z, log_q_z);
}

    const int threads = 256;

    sum_and_add_weights_kernel<<<blocks_sum, threads>>>(

    int blocks_lse = (batch_size + threads - 1) / threads;

        log_probs_summed.data_ptr<float>(),
        log_q_z.data_ptr<float>(),
        batch_size,
        num_components

    return std::make_tuple(z, log_q_z);
