import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void fused_concat_linear_kernel(
    const float* __restrict__ x,      // (R, B, H)
    const float* __restrict__ weight, // (K, R*H)
    const float* __restrict__ bias,   // (K,)
    float* __restrict__ output,       // (B, K)
    int R, int B, int H, int K, int RH
) {
    int b = blockIdx.x * blockDim.x + threadIdx.x;
    int k = blockIdx.y * blockDim.y + threadIdx.y;

    if (b >= B || k >= K) return;

    float sum = bias[k];
    const float* w_row = weight + k * RH;

    for (int r = 0; r < R; ++r) {
        const float* x_chunk = x + (r * B + b) * H;
        const float* w_chunk = w_row + r * H;
        for (int h = 0; h < H; ++h) {
            sum += x_chunk[h] * w_chunk[h];
        }
    }

    output[b * K + k] = sum;
}

torch::Tensor fused_concat_linear_cuda(
    torch::Tensor x,
    torch::Tensor weight,
    torch::Tensor bias
) {
    int R = x.size(0);
    int B = x.size(1);
    int H = x.size(2);
    int K = weight.size(0);
    int RH = R * H;

    auto output = torch::empty({B, K}, x.options());

    dim3 threads(16, 16);
    dim3 blocks((B + 15) / 16, (K + 15) / 16);

    fused_concat_linear_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(),
        weight.data_ptr<float>(),
        bias.data_ptr<float>(),
        output.data_ptr<float>(),
        R, B, H, K, RH
    );

    return output;
}
