import torch

class Model(nn.Module):
    """
    Channel-wise gated delta attention recurrence.

    The recurrent state decays independently for each output channel, which
    makes the update closer to channel-gated delta attention variants used in
    fast long-context baselines.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []
        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-2)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)
        return torch.stack(outputs, dim=1)

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

template <typename scalar_t>
__global__ void delta_attention_kernel(
    const scalar_t* __restrict__ q,
    const scalar_t* __restrict__ k,
    const scalar_t* __restrict__ v,
    const scalar_t* __restrict__ gate,
    scalar_t* __restrict__ output,
    const int batch_size,
    const int seq_len,
    const int num_heads,
    const int head_dim
) {
    const int bh = blockIdx.x;
    const int b = bh / num_heads;
    const int h = bh % num_heads;
    
    if (b >= batch_size) return;
    
    const int D = head_dim;
    extern __shared__ scalar_t state[];
    
    // Initialize state to 0
    for (int i = threadIdx.x; i < D * D; i += blockDim.x) {
        state[i] = 0.0f;
    }
    __syncthreads();
    
    const int base_offset = ((b * seq_len + 0) * num_heads + h) * D;
    const int stride_bh = seq_len * num_heads * D;
    const int stride_t = num_heads * D;
    
    for (int t = 0; t < seq_len; t++) {
        // Load gate_t [D], k_t [D], v_t [D], q_t [D]
        scalar_t gate_t[32];
        scalar_t k_t[32];
        scalar_t v_t[32];
        scalar_t q_t[32];
        
        #pragma unroll
        for (int d = threadIdx.x; d < D; d += blockDim.x) {
            int idx = base_offset + t * stride_t + d;
            gate_t[d] = gate[idx];
            k_t[d] = k[idx];
            v_t[d] = v[idx];
            q_t[d] = q[idx];
        }
        __syncthreads(); // not strictly needed but safe
        
        // Update state: state = gate_t.unsqueeze(-2) * state + outer(k, v)
        // gate_t.unsqueeze(-2): multiply each row d' by gate[d']
        for (int i = threadIdx.x; i < D * D; i += blockDim.x) {
            int row = i / D;  // d' (v dim)
            int col = i % D;  // d (k dim)
            scalar_t g = gate_t[col];
            scalar_t kv = k_t[col] * v_t[row];
            state[i] = g * state[i] + kv;
        }
        __syncthreads();
        
        // Compute y_t = sum_d' q[d] * state[d, d']   wait indices: q.unsqueeze(-1)*state -> sum(-2)
        // state[d_row (k), d_col (v)] , q.unsqueeze(-1) means q outer with state last dim? 
        // q.unsqueeze(-1) [D,1] * state[D,D] -> sum(-2) sums over first D -> [D]
        // actually: y[d_out] = sum_d q[d] * state[d, d_out]
        scalar_t y[32];
        for (int d_out = threadIdx.x; d_out < D; d_out += blockDim.x) {
            scalar_t sum = 0.0f;
            for (int d = 0; d < D; d++) {
                sum += q_t[d] * state[d * D + d_out];
            }
            y[d_out] = sum;
        }
        
        // Write output
        for (int d = threadIdx.x; d < D; d += blockDim.x) {
            int out_idx = ((b * seq_len + t) * num_heads + h) * D + d;
            output[out_idx] = y[d];
        }
        __syncthreads();
    }
}

torch::Tensor delta_attention_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    torch::Tensor gate
) {
    const int batch_size = q.size(0);
    const int seq_len = q.size(1);
    const int num_heads = q.size(2);
    const int head_dim = q.size(3);
    
    auto output = torch::empty({batch_size, seq_len, num_heads, head_dim}, q.options());

        output.data_ptr<float>(),
        batch_size, seq_len, num_heads, head_dim

}
