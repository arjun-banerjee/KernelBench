import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Multi-head sparse aggregation with per-edge attention weights.

    Each edge carries one weight per head, and source node features are arranged
    as `(num_nodes, num_heads, head_dim)`. The output aggregates weighted
    source features into `(num_nodes, num_heads, head_dim)`.
    """

    def __init__(self):
        super().__init__()

    def forward(self, row_ptr: torch.Tensor, col_idx: torch.Tensor, edge_weight: torch.Tensor, node_feat: torch.Tensor) -> torch.Tensor:
        num_nodes = row_ptr.numel() - 1
        num_heads = node_feat.shape[1]
        head_dim = node_feat.shape[2]
        out = torch.zeros(num_nodes, num_heads, head_dim, dtype=node_feat.dtype, device=node_feat.device)
        for dst in range(num_nodes):
            start = int(row_ptr[dst].item())
            end = int(row_ptr[dst + 1].item())
            if end > start:
                src = col_idx[start:end].long()
                weights = edge_weight[start:end].unsqueeze(-1)
                out[dst] = (node_feat[src] * weights).sum(dim=0)
        return out

#include <torch/extension.h>

torch::Tensor csr_spmm_forward(torch::Tensor row_ptr,
                               torch::Tensor col_idx,
                               torch::Tensor edge_weight,

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cstdint>

// Optimized for the benchmark shape: node_feat is (N, 8, 32).  One CUDA
// block handles one destination node; its 8 warps handle the 8 heads, with
// lane 0..31 computing the 32 feature dimensions.  Each thread accumulates
// one output element in the original edge order.
template <typename row_t, typename col_t>
__global__ void csr_spmm_8x32_kernel(const row_t* __restrict__ row_ptr,
                                     const col_t* __restrict__ col_idx,
                                     const float* __restrict__ edge_weight,
                                     const float* __restrict__ node_feat,
                                     float* __restrict__ out,
                                     int num_nodes) {
    int dst = blockIdx.x;
    int tid = threadIdx.x;  // 0..255
    if (dst >= num_nodes) return;

    int h = tid >> 5;       // / 32
    int d = tid & 31;       // % 32

    row_t start = row_ptr[dst];
    row_t end   = row_ptr[dst + 1];

    float acc = 0.0f;
    #pragma unroll 1
    for (row_t e = start; e < end; ++e) {
        int src = static_cast<int>(col_idx[e]);
        float w = edge_weight[static_cast<long long>(e) * 8 + h];
        acc += node_feat[static_cast<long long>(src) * 256 + tid] * w;
    }
    out[static_cast<long long>(dst) * 256 + tid] = acc;
}

// Generic fallback for unexpected head/dim sizes, still entirely custom CUDA.
template <typename row_t, typename col_t>
__global__ void csr_spmm_generic_kernel(const row_t* __restrict__ row_ptr,
                                        const col_t* __restrict__ col_idx,
                                        const float* __restrict__ edge_weight,
                                        const float* __restrict__ node_feat,
                                        float* __restrict__ out,
                                        int num_nodes, int num_heads, int head_dim,
                                        int total) {
    int dst = blockIdx.x;
    int tid = threadIdx.x;
    if (dst >= num_nodes) return;
    for (int idx = tid; idx < total; idx += blockDim.x) {
        int h = idx / head_dim;
        row_t start = row_ptr[dst];
        row_t end   = row_ptr[dst + 1];
        float acc = 0.0f;
        for (row_t e = start; e < end; ++e) {
            int src = static_cast<int>(col_idx[e]);
            float w = edge_weight[static_cast<long long>(e) * num_heads + h];
            acc += node_feat[static_cast<long long>(src) * total + idx] * w;
        }
        out[static_cast<long long>(dst) * total + idx] = acc;
    }
}

template <typename row_t, typename col_t>
static inline void launch_typed(torch::Tensor row_ptr,
                                torch::Tensor col_idx,
                                torch::Tensor edge_weight,
                                torch::Tensor node_feat,
                                torch::Tensor out,
                                int num_nodes, int num_heads, int head_dim) {
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

        csr_spmm_8x32_kernel<row_t, col_t><<<num_nodes, 256, 0, stream>>>(
            row_ptr.data_ptr<row_t>(),
            col_idx.data_ptr<col_t>(),
            edge_weight.data_ptr<float>(),
            node_feat.data_ptr<float>(),
            out.data_ptr<float>(),
