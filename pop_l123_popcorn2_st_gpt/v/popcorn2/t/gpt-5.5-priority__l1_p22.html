import torch
import torch.nn as nn

class Model(nn.Module):
    """
    SE(3)-invariant linear layer.  Maps node features that are augmented
    with 3-D coordinate information through a linear transform that
    respects rotational and translational invariance.  The layer operates
    on (scalar_features, vector_features) pairs where scalar features
    transform trivially and vector features transform as 3-D vectors.
    Used in equivariant GNNs (EGNN, PaiNN, TFN).
    """

    def __init__(self, scalar_in, scalar_out, vector_in, vector_out):
        super().__init__()
        self.scalar_linear = nn.Linear(scalar_in + vector_in, scalar_out)
        self.vector_weight = nn.Parameter(torch.randn(vector_out, vector_in))
        nn.init.xavier_uniform_(self.vector_weight)
        self.vector_gate = nn.Linear(scalar_in + vector_in, vector_out)

    def forward(self, scalar_feat: torch.Tensor, vector_feat: torch.Tensor) -> tuple:
        """
        Args:
            scalar_feat: (B, N, scalar_in) – invariant node features
            vector_feat: (B, N, vector_in, 3) – equivariant vector features
        Returns:
            (scalar_out, vector_out):
                scalar_out: (B, N, scalar_out)
                vector_out: (B, N, vector_out, 3)
        """
        vec_norm = vector_feat.norm(dim=-1)
        combined = torch.cat([scalar_feat, vec_norm], dim=-1)
        s_out = self.scalar_linear(combined)
        v_out = torch.einsum('oi,bnid->bnod', self.vector_weight, vector_feat)
        gate = torch.sigmoid(self.vector_gate(combined)).unsqueeze(-1)
        v_out = gate * v_out
        return (s_out, v_out)

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <vector>

__global__ void se3_kernel(const float* __restrict__ scalar,
                           const float* __restrict__ vec,
                           const float* __restrict__ sw,
                           const float* __restrict__ sb,
                           const float* __restrict__ vw,
                           const float* __restrict__ gw,
                           const float* __restrict__ gb,
                           float* __restrict__ sout,
                           float* __restrict__ vout,
                           int M, int S_in, int S_out, int V_in, int V_out) {
    int row = blockIdx.x;
    int tid = threadIdx.x;
    int C = S_in + V_in;
    extern __shared__ float comb[];

    // Build the concatenated feature [scalar, ||vector||] once per (B,N) row.
    for (int k = tid; k < C; k += blockDim.x) {
        float val;
        if (k < S_in) {
            val = scalar[row * S_in + k];
        } else {
            int i = k - S_in;
            int base = (row * V_in + i) * 3;
            float x = vec[base + 0];
            float y = vec[base + 1];
            float z = vec[base + 2];
            val = sqrtf(x * x + y * y + z * z);
        }
        comb[k] = val;
    }
    __syncthreads();

    // Scalar linear: (S_out x C) * comb + bias.
    for (int o = tid; o < S_out; o += blockDim.x) {
        float acc = sb[o];
        const float* wrow = sw + o * C;
        #pragma unroll 4
        for (int k = 0; k < C; ++k) {
            acc = fmaf(comb[k], wrow[k], acc);
        }
        sout[row * S_out + o] = acc;
    }

    // Gate linear + sigmoid, fused with vector projection for this output channel.
    for (int o = tid; o < V_out; o += blockDim.x) {
        float gacc = gb[o];
        const float* grow = gw + o * C;
        #pragma unroll 4
        for (int k = 0; k < C; ++k) {
            gacc = fmaf(comb[k], grow[k], gacc);
        }
        float gate = 1.0f / (1.0f + expf(-gacc));
        const float* vwrow = vw + o * V_in;
        for (int d = 0; d < 3; ++d) {
            float vacc = 0.0f;
            #pragma unroll
            for (int i = 0; i < 16; ++i) {
                if (i < V_in) {
                    vacc = fmaf(vwrow[i], vec[(row * V_in + i) * 3 + d], vacc);
                }
            }
            vout[(row * V_out + o) * 3 + d] = gate * vacc;
        }
    }
}

                                            torch::Tensor vector_feat,
                                            torch::Tensor scalar_weight,
                                            torch::Tensor scalar_bias,
                                            torch::Tensor vector_weight,
                                            torch::Tensor gate_weight,
                                            torch::Tensor gate_bias) {
    const int B = (int)scalar_feat.size(0);
    const int N = (int)scalar_feat.size(1);
    const int S_in = (int)scalar_feat.size(2);
    const int V_in = (int)vector_feat.size(2);
    const int S_out = (int)scalar_weight.size(0);
    const int V_out = (int)vector_weight.size(0);

}
