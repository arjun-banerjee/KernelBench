import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Channel-wise gated delta attention recurrence.

    The recurrent state decays independently for each output channel, which
    makes the update closer to channel-gated delta attention variants used in
    fast long-context baselines.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q: torch.Tensor, k: torch.Tensor, v: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []
        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-2)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)
        return torch.stack(outputs, dim=1)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
#include <torch/extension.h>

torch::Tensor delta_attn_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, torch::Tensor gate);
"""

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <c10/cuda/CUDAStream.h>

#define WARP_SIZE 32

template<int WARPS>
__global__ void delta_attn_kernel(const float* __restrict__ q,
                                  const float* __restrict__ k,
                                  const float* __restrict__ v,
                                  const float* __restrict__ gate,
                                  float* __restrict__ out,
                                  int B, int T, int H, int D) {
    int b = blockIdx.x;
    int h = blockIdx.y;
    int tid = threadIdx.x;
    int lane = tid & 31;
    int warp = tid >> 5;

    extern __shared__ float smem[];
    float* state = smem;                  // laid out as state[j][i], contiguous across i
    float* qs = state + D * D;
    float* ks = qs + D;

    int DD = D * D;
    for (int idx = tid; idx < DD; idx += blockDim.x) {
        state[idx] = 0.0f;
    }
    __syncthreads();

    for (int t = 0; t < T; ++t) {
        int base = (((b * T + t) * H + h) * D);

        for (int i = tid; i < D; i += blockDim.x) {
            qs[i] = q[base + i];
            ks[i] = k[base + i];
        }
        __syncthreads();

        // Each warp owns one or more output channels j.  Lanes cover the summed dimension i.
        for (int j = warp; j < D; j += WARPS) {
            float gj = gate[base + j];
            float vj = v[base + j];
            float acc = 0.0f;

            int row = j * D;
            for (int i = lane; i < D; i += WARP_SIZE) {
                float s = gj * state[row + i] + ks[i] * vj;
                state[row + i] = s;
                acc += qs[i] * s;
            }

            #pragma unroll
            for (int off = 16; off > 0; off >>= 1) {
                acc += __shfl_down_sync(0xffffffff, acc, off);
            }
            if (lane == 0) {
                out[base + j] = acc;
            }
        }
        __syncthreads();
    }
}

torch::Tensor delta_attn_forward(torch::Tensor q, torch::Tensor k, torch::Tensor v, torch::Tensor gate) {
    const int B = (int)q.size(0);
    const int T = (int)q.size(1);
    const int H = (int)q.size(2);
    const int D = (int)q.size(3);

    auto out = torch::empty_like(q);
    const int WARPS = 8;
    dim3 block(WARPS * 32);
    dim3 grid(B, H);
    size_t shmem = (size_t)(D * D + 2 * D) * sizeof(float);

    cudaStream_t stream = c10::cuda::getCurrentCUDAStream();
    delta_attn_kernel<WARPS><<<grid, block, shmem, stream>>>(
        q.data_ptr<float>(), k.data_ptr<float>(), v.data_ptr<float>(), gate.data_ptr<float>(),
        out.data_ptr<float>(), B, T, H, D);
    return out;
}
"""
