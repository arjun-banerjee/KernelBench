import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Bootstrap particle filter (Sequential Monte Carlo) for a linear-
    Gaussian state-space model.  Implements the predict–weight–resample
    cycle for one time step.  The state transition is x_{t} = A x_{t-1} + w
    and the observation model is y_t = C x_t + v.
    """

    def __init__(self, state_dim, obs_dim, num_particles):
        super().__init__()
        self.state_dim = state_dim
        self.obs_dim = obs_dim
        self.num_particles = num_particles
        self.register_buffer('A', 0.9 * torch.eye(state_dim))
        self.register_buffer('C', torch.randn(obs_dim, state_dim) * 0.5)
        self.process_noise = 0.1
        self.obs_noise = 0.5

    def forward(self, particles: torch.Tensor, log_weights: torch.Tensor, observation: torch.Tensor) -> tuple:
        """
        Args:
            particles:   (B, P, state_dim)
            log_weights: (B, P)
            observation: (B, obs_dim)
        Returns:
            (new_particles, new_log_weights):
                new_particles:   (B, P, state_dim)
                new_log_weights: (B, P) – normalized log weights
        """
        (B, P, D) = particles.shape
        pred = particles @ self.A.t() + self.process_noise * torch.randn_like(particles)
        obs_pred = pred @ self.C.t()
        diff = observation.unsqueeze(1) - obs_pred
        log_lik = -0.5 * (diff ** 2).sum(-1) / self.obs_noise ** 2
        new_log_w = log_weights + log_lik
        new_log_w = new_log_w - torch.logsumexp(new_log_w, dim=1, keepdim=True)
        weights = torch.exp(new_log_w)
        cumsum = torch.cumsum(weights, dim=1)
        u = (torch.arange(P, device=particles.device, dtype=particles.dtype) + torch.rand(B, 1, device=particles.device)) / P
        indices = torch.searchsorted(cumsum, u)
        indices = indices.clamp(0, P - 1)
        batch_idx = torch.arange(B, device=particles.device).unsqueeze(1).expand(-1, P)
        resampled = pred[batch_idx, indices]
        uniform_log_w = torch.full_like(new_log_w, -torch.log(torch.tensor(float(P))))
        return (resampled, uniform_log_w)

#include <cuda.h>
#include <cuda_runtime.h>

__global__ void particle_filter_kernel(
    const float* __restrict__ particles,
    const float* __restrict__ log_weights,
    const float* __restrict__ observation,
    const float* __restrict__ A,
    const float* __restrict__ C,
    float* __restrict__ resampled,
    float* __restrict__ uniform_log_w,
    float process_noise,
    float obs_noise,
    int B,
    int P,
    int D,
    int O,
    unsigned long long seed
) {
    int b = blockIdx.x;
    int p = threadIdx.x;
    if (b >= B || p >= P) return;

    extern __shared__ float smem[];
    float* pred_s = smem;
    float* obs_pred_s = smem + P * D;
    float* weights_s = smem + P * D + P * O;
    float* cumsum_s = smem + P * D + P * O + P;

    curandState state;
    curand_init(seed, b * P + p, 0, &state);

    // Predict: pred = particles @ A.T + noise
    for (int d = 0; d < D; d++) {
        float val = 0.0f;
        for (int k = 0; k < D; k++) {
            val += particles[(b * P + p) * D + k] * A[k * D + d];
        }
        float noise = curand_normal(&state) * process_noise;
        pred_s[p * D + d] = val + noise;
    }

    __syncthreads();

    // obs_pred = pred @ C.T
    for (int o = 0; o < O; o++) {
        float val = 0.0f;
        for (int d = 0; d < D; d++) {
            val += pred_s[p * D + d] * C[d * O + o];
        }
        obs_pred_s[p * O + o] = val;

    __syncthreads();

    // log_lik
    float log_lik = 0.0f;

        log_lik -= 0.5f * diff * diff / (obs_noise * obs_noise);
    }

    float new_log_w = log_weights[b * P + p] + log_lik;
    weights_s[p] = new_log_w;
    __syncthreads();

        sum_exp = sum;
    }
    __syncthreads();

    weights_s[p] = expf(nlogw);  // now normalized weights
    __syncthreads();

        for (int i = 1; i < P; i++) {

        }
    }
    __syncthreads();

    float u_base = curand_uniform(&state);

    // binary search
    int low = 0, high = P - 1;

    torch::Tensor particles,
