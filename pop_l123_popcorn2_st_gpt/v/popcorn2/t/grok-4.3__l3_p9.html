import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Bayesian linear regression with a conjugate Gaussian prior.
    Given features X and targets y, computes the posterior distribution
    over weights: p(w | X, y) = N(mu_post, Sigma_post) in closed form
    via the matrix inversion lemma / Cholesky decomposition.
    """

    def __init__(self, input_dim, prior_precision=1.0, noise_precision=1.0):
        super().__init__()
        self.input_dim = input_dim
        self.prior_precision = prior_precision
        self.noise_precision = noise_precision

    def forward(self, X: torch.Tensor, y: torch.Tensor, X_test: torch.Tensor) -> tuple:
        """
        Args:
            X:      (B, N, D) – training features
            y:      (B, N, 1) – training targets
            X_test: (B, M, D) – test features
        Returns:
            (pred_mean, pred_var):
                pred_mean: (B, M, 1)
                pred_var:  (B, M)
        """
        D = X.shape[-1]
        eye = torch.eye(D, device=X.device, dtype=X.dtype).unsqueeze(0)
        Lambda_post = self.prior_precision * eye + self.noise_precision * (X.transpose(-1, -2) @ X)
        L = torch.linalg.cholesky(Lambda_post)
        rhs = self.noise_precision * (X.transpose(-1, -2) @ y)
        mu_post = torch.cholesky_solve(rhs, L)
        pred_mean = X_test @ mu_post
        Sigma_post = torch.cholesky_inverse(L)
        pred_var = (X_test @ Sigma_post * X_test).sum(-1) + 1.0 / self.noise_precision
        return (pred_mean, pred_var)

#include <cuda.h>
#include <cuda_runtime.h>
#include <ATen/ATen.h>
#include <ATen/cuda/CUDAContext.h>

#define D 16

#define N 256

__global__ void bayesian_lr_kernel(
    const float* __restrict__ X,
    const float* __restrict__ y,
    const float* __restrict__ X_test,
    float* __restrict__ pred_mean,
    float* __restrict__ pred_var,
    float prior_precision,
    float noise_precision,
    int batch_size
) {
    int b = blockIdx.x;
    if (b >= batch_size) return;

    extern __shared__ float smem[];

    float* Lambda = smem;                    // D*D
    float* L = &smem[D*D];                   // D*D
    float* rhs = &smem[2*D*D];               // D*1
    float* mu = &smem[2*D*D + D];            // D*1
    float* Sigma = &smem[2*D*D + 2*D];       // D*D
    float* temp = &smem[3*D*D + 2*D];        // D*D + D*1 for workspace

    const float* Xb = X + b * N * D;
    const float* yb = y + b * N;
    const float* Xt = X_test + b * M * D;

    // Compute X^T @ X
    for (int i = 0; i < D; i++) {
        for (int j = 0; j < D; j++) {
            float sum = 0.0f;
            for (int k = 0; k < N; k++) {
                sum += Xb[k*D + i] * Xb[k*D + j];
            }
            Lambda[i*D + j] = prior_precision * (i==j ? 1.0f : 0.0f) + noise_precision * sum;
        }
    }

    // Cholesky decomposition L L^T = Lambda, lower triangular
    for (int i = 0; i < D; i++) {
        for (int j = 0; j <= i; j++) {
            float sum = 0.0f;
            for (int k = 0; k < j; k++) {
                sum += L[i*D + k] * L[j*D + k];

            if (i == j) {

            } else {

            }
        }
    }

    // rhs = noise_precision * X^T @ y

        float sum = 0.0f;
        for (int k = 0; k < N; k++) {
            sum += Xb[k*D + i] * yb[k];

        for (int j = 0; j < i; j++) {
            sum -= L[i*D + j] * mu[j];

        mu[i] = sum / L[i*D + i];

            sum += Xt[m*D + d] * mu[d];
        }

    }

        for (int i = 0; i < D; i++) {
