import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Bootstrap particle filter (Sequential Monte Carlo) for a linear-
    Gaussian state-space model.  Implements the predict–weight–resample
    cycle for one time step.  The state transition is x_{t} = A x_{t-1} + w
    and the observation model is y_t = C x_t + v.
    """

    def __init__(self, state_dim, obs_dim, num_particles):
        super().__init__()
        self.state_dim = state_dim
        self.obs_dim = obs_dim
        self.num_particles = num_particles
        self.register_buffer('A', 0.9 * torch.eye(state_dim))
        self.register_buffer('C', torch.randn(obs_dim, state_dim) * 0.5)
        self.process_noise = 0.1
        self.obs_noise = 0.5

    def forward(self, particles: torch.Tensor, log_weights: torch.Tensor, observation: torch.Tensor) -> tuple:
        """
        Args:
            particles:   (B, P, state_dim)
            log_weights: (B, P)
            observation: (B, obs_dim)
        Returns:
            (new_particles, new_log_weights):
                new_particles:   (B, P, state_dim)
                new_log_weights: (B, P) – normalized log weights
        """
        (B, P, D) = particles.shape
        pred = particles @ self.A.t() + self.process_noise * torch.randn_like(particles)
        obs_pred = pred @ self.C.t()
        diff = observation.unsqueeze(1) - obs_pred
        log_lik = -0.5 * (diff ** 2).sum(-1) / self.obs_noise ** 2
        new_log_w = log_weights + log_lik
        new_log_w = new_log_w - torch.logsumexp(new_log_w, dim=1, keepdim=True)
        weights = torch.exp(new_log_w)
        cumsum = torch.cumsum(weights, dim=1)
        u = (torch.arange(P, device=particles.device, dtype=particles.dtype) + torch.rand(B, 1, device=particles.device)) / P
        indices = torch.searchsorted(cumsum, u)
        indices = indices.clamp(0, P - 1)
        batch_idx = torch.arange(B, device=particles.device).unsqueeze(1).expand(-1, P)
        resampled = pred[batch_idx, indices]
        uniform_log_w = torch.full_like(new_log_w, -torch.log(torch.tensor(float(P))))
        return (resampled, uniform_log_w)

        float log_lik = -0.5f * diff_sq_sum / (obs_noise * obs_noise);
        
        new_log_w[p] = log_weights_in[batch_idx * P + p] + log_lik;
    }
    __syncthreads();
    
    // Step 2: Logsumexp normalization
    logsumexp_block(new_log_w, P, &lse_val, tid);
    __syncthreads();
    
    for (int p = tid; p < P; p += BLOCK_SIZE) {
        new_log_w[p] -= lse_val;
    }
    __syncthreads();
    
    // Step 3: Compute weights
    for (int p = tid; p < P; p += BLOCK_SIZE) {
        cumsum[p] = expf(new_log_w[p]);
    }
    __syncthreads();
    
    // Step 4: Cumulative sum (inclusive scan)
    if (tid == 0) {
        float sum = 0.0f;
        for (int p = 0; p < P; p++) {
            sum += cumsum[p];
            cumsum[p] = sum;
        }
    }
    __syncthreads();
    
    // Step 5: Get random value
    if (tid == 0) {
        rand_val = rand_u[batch_idx];
    }
    __syncthreads();
    
    // Step 6: Binary search for resampling indices
    for (int p = tid; p < P; p += BLOCK_SIZE) {
        float u = (static_cast<float>(p) + rand_val) / static_cast<float>(P);
        
        int lo = 0, hi = P - 1;
        while (lo < hi) {
            int mid = (lo + hi) / 2;
            if (cumsum[mid] < u) {
                lo = mid + 1;
            } else {
                hi = mid;
            }
        }
        indices[p] = min(lo, P - 1);
    }
    __syncthreads();
    
    // Step 7: Resample
    for (int p = tid; p < P; p += BLOCK_SIZE) {
        int idx = indices[p];
        for (int d = 0; d < D; d++) {
            resampled_out[batch_idx * P * D + p * D + d] = pred[idx * D + d];
        }
    }
    
    // Step 8: Set uniform log weights
    float uniform_log_w = -logf(static_cast<float>(P));
    for (int p = tid; p < P; p += BLOCK_SIZE) {
        log_weights_out[batch_idx * P + p] = uniform_log_w;
    }
}
"""

class ModelNew(nn.Module):
    def __init__(self, state_dim, obs_dim, num_particles):
        super().__init__()
        self.state_dim = state_dim
        self.obs_dim = obs_dim
        self.num_particles = num_particles
        self.register_buffer('A', 0.9 * torch.eye(state_dim))
        self.register_buffer('C', torch.randn(obs_dim, state_dim) * 0.5)
        self.process_noise = 0.1
        self.obs_noise = 0.5
    
    def forward(self, particles, log_weights, observation):
        B, P, D = particles.shape
        
        # Generate random values in PyTorch for deterministic correctness
        noise = torch.randn_like(particles) * self.process_noise
        rand_u = torch.rand(B, 1, device=particles.device, dtype=particles.dtype)
        
        resampled, new_log_weights = particle_filter_ext.particle_filter_forward_cuda(
            particles.contiguous(),
            noise.contiguous(),
            log_weights.contiguous(),
            observation.contiguous(),
            self.A.contiguous(),
            self.C.contiguous(),
            rand_u.contiguous(),
            self.obs_noise
        )
        
        return resampled, new_log_weights
