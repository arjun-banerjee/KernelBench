import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void mean_reduce_kernel(const float* __restrict__ x, float* __restrict__ out, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx < total) {
        int b = idx / D;
        int d = idx % D;
        float sum = 0.0f;
        #pragma unroll
        for(int r = 0; r < 8; r++) {  // unroll for small R<=8
            if (r < R) {
                sum += x[(r * B + b) * D + d];
            }
        }
        out[b * D + d] = sum / (float)R;
    }
}

torch::Tensor mean_reduce_cuda(torch::Tensor x) {
    TORCH_CHECK(x.is_cuda(), "x must be CUDA tensor");
    TORCH_CHECK(x.dim() == 3, "x must be 3D");
    int R = x.size(0);
    int B = x.size(1);
    int D = x.size(2);
    auto out = torch::empty({B, D}, x.options());
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    mean_reduce_kernel<<<blocks, threads>>>(x.data_ptr<float>(), out.data_ptr<float>(), R, B, D);
    return out;
}
'''

cpp_source = '''
torch::Tensor mean_reduce_cuda(torch::Tensor x);

_mean_module = load_inline(
