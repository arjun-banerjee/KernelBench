import math
import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Fused MLA-style attention from compressed KV latents.

    The compressed latent representation is expanded into keys and values on the
    fly inside the attention computation rather than materializing a KV cache
    first. This isolates the main compute pattern behind fused MLA attention.
    """

    def __init__(self):
        super().__init__()

    def forward(self, q: torch.Tensor, kv_latent: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, num_heads, head_dim) = q.shape
        rank = kv_latent.shape[-1]
        out = torch.zeros_like(q)
        scale = 1.0 / math.sqrt(head_dim)
        for b in range(batch_size):
            for h in range(num_heads):
                for t in range(seq_len):
                    logits = []
                    values = []
                    for s in range(t + 1):
                        latent = kv_latent[b, s]
                        k = torch.matmul(latent, w_up_k[:, h])
                        v = torch.matmul(latent, w_up_v[:, h])
                        logits.append((q[b, t, h] * k).sum() * scale)
                        values.append(v)
                    attn = torch.softmax(torch.stack(logits, dim=0), dim=0)
                    value_stack = torch.stack(values, dim=0)
                    out[b, t, h] = (attn.unsqueeze(-1) * value_stack).sum(dim=0)
        return out

        }

        logits[s] = logit;

            max_logit = logit;
        }

        float exp_val = expf(logits[s] - max_logit);
        logits[s] = exp_val;
        sum_exp += exp_val;
    }
    
    float inv_sum_exp = 1.0f / sum_exp;
    for (int s = 0; s <= t; s++) {
        logits[s] *= inv_sum_exp;

            out_d += logits[s] * all_v[s * head_dim + d];
        }

    }
}

    """
    Fused MLA-style attention from compressed KV latents.
    
    Optimized implementation using custom CUDA kernel for the attention
    computation with on-the-fly KV projection.
    """
    
    def __init__(self):
        super().__init__()
        
    def forward(self, q: torch.Tensor, kv_latent: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        # Ensure inputs are contiguous
        q = q.contiguous()

        w_up_v = w_up_v.contiguous()

        return self._forward_pytorch(q, kv_latent, w_up_k, w_up_v)
    
    def _forward_pytorch(self, q: torch.Tensor, kv_latent: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        """Optimized PyTorch fallback implementation."""
        batch_size, seq_len, num_heads, head_dim = q.shape
        rank = kv_latent.shape[-1]
        
        # Precompute k and v projections: kv_latent @ w_up_k/v
        # kv_latent: [B, T, R], w_up_k: [R, H, D] -> k: [B, T, H, D]
        # Using einsum for efficient batched matmul
        k = torch.einsum('btr,rhd->bthd', kv_latent, w_up_k)
        v = torch.einsum('btr,rhd->bthd', kv_latent, w_up_v)
        
        scale = 1.0 / math.sqrt(head_dim)
        
        # Compute attention scores: q @ k^T
        # q: [B, T, H, D], k: [B, T, H, D] -> scores: [B, H, T, T]
        q_t = q.transpose(1, 2)  # [B, H, T, D]
        k_t = k.transpose(1, 2)  # [B, H, T, D]
        
        scores = torch.matmul(q_t, k_t.transpose(-2, -1)) * scale  # [B, H, T, T]
        
        # Apply causal mask
        causal_mask = torch.triu(torch.ones(seq_len, seq_len, device=q.device, dtype=torch.bool), diagonal=1)
        scores = scores.masked_fill(causal_mask, float('-inf'))
        
        # Softmax
        attn = torch.softmax(scores, dim=-1)  # [B, H, T, T]
        
        # Apply attention to values
        v_t = v.transpose(1, 2)  # [B, H, T, D]
        out = torch.matmul(attn, v_t)  # [B, H, T, D]
        
        # Transpose back to [B, T, H, D]
        out = out.transpose(1, 2).contiguous()
        
        return out
