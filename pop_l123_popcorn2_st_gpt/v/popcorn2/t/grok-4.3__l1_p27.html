import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

# Fixed permutation (same as reference)
_g = torch.Generator()
_g.manual_seed(10)
N = 72
PERM = torch.randperm(N, generator=_g)

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void permute_kernel(const float* __restrict__ x, float* __restrict__ y, const int* __restrict__ perm, int B, int N) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * N;
    if (idx < total) {
        int b = idx / N;
        int j = idx % N;
        int src = perm[j];
        y[idx] = x[b * N + src];
    }
}

torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm) {
    auto B = x.size(0);
    auto N = x.size(1);
    auto y = torch::empty({B, N}, x.options());
    
    const int threads = 256;
    const int blocks = (B * N + threads - 1) / threads;
    
    permute_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(),
        y.data_ptr<float>(),
        perm.data_ptr<int>(),
        B,
        N
    );
    
    return y;
}
'''

cpp_source = '''
torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm);
'''

# Compile the extension
permute_ext = load_inline(
    name='permute_ext',
    cpp_sources=[cpp_source],
    cuda_sources=[cuda_source],
    functions=['permute_cuda'],
    verbose=False,
    with_cuda=True
)

class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        self.register_buffer('perm', PERM)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return permute_ext.permute_cuda(x, self.perm.to(device=x.device, dtype=torch.int32))
