import math
import torch
import torch.nn as nn
import torch.nn.functional as F

class Model(nn.Module):
    """
    Single-layer multi-head cross-attention with RoPE on Q and K, attention
    dropout, and output projection—intended as a fusion target for robotics
    stacks (e.g. proprio / action queries attending to vision or memory).

    Pipeline (conceptually one fused kernel): linear Q from query stream,
    linear K/V from key–value stream → reshape heads → apply RoPE separately
    along query positions (T_q) and key positions (T_k) → scaled dot-product,
    softmax, dropout on weights → aggregate values → output projection.
    """

    def __init__(self, dim: int, num_heads: int, dropout: float=0.0, rope_base: float=10000.0):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        assert self.head_dim % 2 == 0, 'RoPE requires an even head dimension'
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.out_proj = nn.Linear(dim, dim, bias=False)
        self.dropout = nn.Dropout(dropout)
        self.scale = 1.0 / math.sqrt(self.head_dim)
        inv_freq = 1.0 / rope_base ** (torch.arange(0, self.head_dim, 2).float() / self.head_dim)
        self.register_buffer('inv_freq', inv_freq, persistent=False)

    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        (x1, x2) = x.chunk(2, dim=-1)
        return torch.cat([-x2, x1], dim=-1)

    def _rope(self, x: torch.Tensor, seq_len: int) -> torch.Tensor:
        """Apply RoPE along the sequence dimension (last-but-one after head). x: (B, H, L, d)."""
        t = torch.arange(seq_len, device=x.device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        emb = torch.cat([freqs, freqs], dim=-1)
        cos = emb.cos().view(1, 1, seq_len, self.head_dim)
        sin = emb.sin().view(1, 1, seq_len, self.head_dim)
        return x * cos + self._rotate_half(x) * sin

    def forward(self, x_query: torch.Tensor, x_kv: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x_query: (B, T_q, dim) — e.g. robot or decoder tokens
            x_kv:    (B, T_k, dim) — e.g. encoder / memory tokens
        Returns:
            (B, T_q, dim)
        """
        (B, T_q, _) = x_query.shape
        (_, T_k, _) = x_kv.shape
        (h, d) = (self.num_heads, self.head_dim)
        q = self.q_proj(x_query).view(B, T_q, h, d).transpose(1, 2)
        k = self.k_proj(x_kv).view(B, T_k, h, d).transpose(1, 2)
        v = self.v_proj(x_kv).view(B, T_k, h, d).transpose(1, 2)
        q = self._rope(q, T_q)
        k = self._rope(k, T_k)
        attn = torch.matmul(q, k.transpose(-2, -1)) * self.scale
        attn = F.softmax(attn, dim=-1)
        attn = self.dropout(attn)
        out = torch.matmul(attn, v)
        out = out.transpose(1, 2).contiguous().view(B, T_q, self.dim)
        return self.out_proj(out)

    int v_base = batch * num_heads * seq_len_kv * head_dim + head * seq_len_kv * head_dim;
    int o_base = batch * num_heads * seq_len_q * head_dim + head * seq_len_q * head_dim;
    
    float scale = 1.0f / sqrtf((float)head_dim);
    
    // Load Q vector into registers
    extern __shared__ float shared_mem[];
    float* q_local = shared_mem;  // head_dim elements
    float* k_block = &shared_mem[head_dim];  // BLOCK_SIZE * head_dim elements
    float* v_block = &shared_mem[head_dim + BLOCK_SIZE * head_dim];  // BLOCK_SIZE * head_dim elements
    
    // Load Q
    for (int i = tid; i < head_dim; i += stride) {
        q_local[i] = q[q_base + q_idx * head_dim + i];
    }
    __syncthreads();
    
    // Initialize output accumulator and online softmax state
    float max_val = -FLT_MAX;
    float sum_val = 0.0f;
    float out_local[64];  // Assuming head_dim <= 64
    
    for (int i = 0; i < head_dim; i++) {
        out_local[i] = 0.0f;
    }
    
    // Process K,V in blocks
    for (int k_start = 0; k_start < seq_len_kv; k_start += BLOCK_SIZE) {
        int k_end = min(k_start + BLOCK_SIZE, seq_len_kv);
        int block_len = k_end - k_start;
        
        // Load K and V block into shared memory
        for (int i = tid; i < BLOCK_SIZE * head_dim; i += stride) {
            int k_idx = k_start + i / head_dim;
            int d_idx = i % head_dim;
            if (k_idx < seq_len_kv) {
                k_block[i] = k[k_base + k_idx * head_dim + d_idx];
                v_block[i] = v[v_base + k_idx * head_dim + d_idx];
            }
        }
        __syncthreads();
        
        // Compute Q*K^T for each k in the block
        for (int k = 0; k < block_len; k++) {
            // Each thread computes part of the dot product
            float qk_local = 0.0f;
            for (int i = tid; i < head_dim; i += stride) {
                qk_local += q_local[i] * k_block[k * head_dim + i];
            }
            
            // Reduce within block
            __shared__ float qk_shared[BLOCK_SIZE];
            qk_shared[tid] = qk_local;
            __syncthreads();
            
            for (int s = stride / 2; s > 0; s >>= 1) {
                if (tid < s) {
                    qk_shared[tid] += qk_shared[tid + s];
                }
                __syncthreads();
            }
            
            float qk = qk_shared[0] * scale;
            
            // Online softmax update (only one thread needs to do this)
            if (tid == 0) {
                float new_max = fmaxf(max_val, qk);
                float exp_val = expf(qk - new_max);
                float scale_factor = expf(max_val - new_max);
                
                for (int i = 0; i < head_dim; i++) {
                    out_local[i] = out_local[i] * scale_factor + exp_val * v_block[k * head_dim + i];
                }
                
                sum_val = sum_val * scale_factor + exp_val;
                max_val = new_max;
            }
            __syncthreads();
        }
    }
    
    // Normalize and write output
    if (tid == 0) {
        float inv_sum = 1.0f / sum_val;
        for (int i = 0; i < head_dim; i++) {
            out[o_base + q_idx * head_dim + i] = out_local[i] * inv_sum;
        }
    }
}

class ModelNew(nn.Module):
    """
    Fused multi-head cross-attention with RoPE.
    Uses custom Flash Attention kernel for the attention computation.
    """
    
    def __init__(self, dim: int, num_heads: int, dropout: float=0.0, rope_base: float=10000.0):
        super().__init__()
        assert dim % num_heads == 0
        self.dim = dim
        self.num_heads = num_heads
        self.head_dim = dim // num_heads
        assert self.head_dim % 2 == 0, 'RoPE requires an even head dimension'
        
        # Linear projections (weights only, no bias)
        self.q_proj = nn.Linear(dim, dim, bias=False)
        self.k_proj = nn.Linear(dim, dim, bias=False)
        self.v_proj = nn.Linear(dim, dim, bias=False)
        self.out_proj = nn.Linear(dim, dim, bias=False)
        
        self.dropout = nn.Dropout(dropout)
        self.scale = 1.0 / math.sqrt(self.head_dim)
        
        # RoPE inverse frequencies
        inv_freq = 1.0 / rope_base ** (torch.arange(0, self.head_dim, 2).float() / self.head_dim)
        self.register_buffer('inv_freq', inv_freq, persistent=False)
    
    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        x1, x2 = x.chunk(2, dim=-1)
        return torch.cat([-x2, x1], dim=-1)
    
    def _rope(self, x: torch.Tensor, seq_len: int) -> torch.Tensor:
        """Apply RoPE along the sequence dimension. x: (B, H, L, d)."""
        t = torch.arange(seq_len, device=x.device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        emb = torch.cat([freqs, freqs], dim=-1)
        cos = emb.cos().view(1, 1, seq_len, self.head_dim)
        sin = emb.sin().view(1, 1, seq_len, self.head_dim)
        return x * cos + self._rotate_half(x) * sin
    
    def forward(self, x_query: torch.Tensor, x_kv: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x_query: (B, T_q, dim)
            x_kv: (B, T_k, dim)
        Returns:
            (B, T_q, dim)
        """
        B, T_q, _ = x_query.shape
        _, T_k, _ = x_kv.shape
        h, d = self.num_heads, self.head_dim
        
        # Linear projections - optimized with contiguous tensor operations
        q = self.q_proj(x_query)
        k = self.k_proj(x_kv)
        v = self.v_proj(x_kv)
        
        # Reshape and transpose in one operation for better memory access
        q = q.view(B, T_q, h, d).transpose(1, 2).contiguous()
        k = k.view(B, T_k, h, d).transpose(1, 2).contiguous()
        v = v.view(B, T_k, h, d).transpose(1, 2).contiguous()
        
        # Apply RoPE
        q = self._rope(q, T_q)
        k = self._rope(k, T_k)
        
        # Use custom Flash Attention kernel
        if q.is_cuda and q.dtype == torch.float32:
            out = flash_attention_cuda.flash_attention_fwd(q, k, v)
        else:
            # Fallback to PyTorch implementation
            attn = torch.matmul(q, k.transpose(-2, -1)) * self.scale
            attn = F.softmax(attn, dim=-1)
            attn = self.dropout(attn)
            out = torch.matmul(attn, v)
        
        # Reshape output
        out = out.transpose(1, 2).contiguous().view(B, T_q, self.dim)
        
        # Output projection
        return self.out_proj(out)
