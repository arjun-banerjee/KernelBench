import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Grounded top-2 MoE router.

    Router logits are biased by token-expert similarity against learned
    grounding embeddings, then normalized over the selected top-2 experts.
    """

    def __init__(self):
        super().__init__()

    def forward(self, token_hidden: torch.Tensor, router_logits: torch.Tensor, expert_ground: torch.Tensor, alpha: float) -> torch.Tensor:
        grounded = router_logits + alpha * (token_hidden @ expert_ground.t())
        (top_vals, top_idx) = torch.topk(grounded, k=2, dim=-1)
        top_weights = torch.softmax(top_vals, dim=-1)
        return torch.stack((top_idx.to(torch.float32), top_weights), dim=-1)
num_tokens = 8192

num_experts = 16

#include <cuda.h>
#include <cuda_runtime.h>
#include <c10/cuda/CUDAStream.h>

template <typename scalar_t>
__global__ void grounded_top2_kernel(
    const scalar_t* __restrict__ token_hidden,
    const scalar_t* __restrict__ router_logits,
    const scalar_t* __restrict__ expert_ground,
    scalar_t* __restrict__ output,
    const float alpha,
    const int num_tokens,
    const int hidden_dim,
    const int num_experts) {
    
    extern __shared__ float smem[];
    float* shared_expert_ground = smem;  // num_experts * hidden_dim floats
    
    // Load expert_ground into shared memory
    for (int i = threadIdx.x; i < num_experts * hidden_dim; i += blockDim.x) {
        shared_expert_ground[i] = static_cast<float>(expert_ground[i]);
    }
    __syncthreads();
    
    const int token_idx = blockIdx.x;
    if (token_idx >= num_tokens) return;
    
    const int tid = threadIdx.x;
    
    // Each block processes one token
    // Thread block size should be >= num_experts
    
    // Compute similarity scores for this token
    // token_hidden[token_idx] @ expert_ground.T
    // Result: num_experts values
    
    float grounded[64];  // Assuming num_experts <= 64
    float local_router[64];
    
    // Initialize grounded with router_logits
    for (int e = 0; e < num_experts; e++) {
        local_router[e] = static_cast<float>(router_logits[token_idx * num_experts + e]);
        grounded[e] = local_router[e];
    }
    
    // Add alpha * (token_hidden @ expert_ground.T)
    // For each expert e, compute dot(token_hidden, expert_ground[e])
    for (int e = 0; e < num_experts; e++) {
        float dot = 0.0f;
        // Parallel reduction within the block for dot product
        // Each thread handles a portion of hidden_dim
        for (int h = tid; h < hidden_dim; h += blockDim.x) {
            float th = static_cast<float>(token_hidden[token_idx * hidden_dim + h]);
            float eg = shared_expert_ground[e * hidden_dim + h];
            dot += th * eg;
        }
        
        // Reduce within warp using shuffle
        for (int offset = warpSize / 2; offset > 0; offset /= 2) {
            dot += __shfl_down_sync(0xffffffff, dot, offset);
        }
        
        // First thread in each warp writes to shared memory
        __shared__ float warp_results[32];  // At most 32 warps
        int lane_id = tid & 31;
        int warp_id = tid >> 5;
        
        if (lane_id == 0) {
            warp_results[warp_id] = dot;
        }
        __syncthreads();
        
        // First warp reduces warp_results
        float final_dot = 0.0f;
        if (warp_id == 0) {
            for (int i = lane_id; i < blockDim.x / 32; i += 32) {
                final_dot += warp_results[i];
            }
            for (int offset = warpSize / 2; offset > 0; offset /= 2) {
                final_dot += __shfl_down_sync(0xffffffff, final_dot, offset);
            }
        }
        
        // Broadcast final_dot to all threads
        if (tid == 0) {
            warp_results[0] = final_dot;
        }
        __syncthreads();
        final_dot = warp_results[0];
        
        grounded[e] += alpha * final_dot;
    }
    
    __syncthreads();
    
    // Find top-2
    float top1_val = -INFINITY;
    float top2_val = -INFINITY;
    int top1_idx = 0;
    int top2_idx = 0;
    
    for (int e = 0; e < num_experts; e++) {
        float v = grounded[e];
        if (v > top1_val) {
            top2_val = top1_val;
            top2_idx = top1_idx;
            top1_val = v;
            top1_idx = e;
        } else if (v > top2_val) {
            top2_val = v;
            top2_idx = e;
        }
    }
    
    // Softmax on top-2 values
    float max_val = fmaxf(top1_val, top2_val);
    float exp1 = expf(top1_val - max_val);
    float exp2 = expf(top2_val - max_val);
    float sum_exp = exp1 + exp2;
    float w1 = exp1 / sum_exp;
    float w2 = exp2 / sum_exp;
    
    // Output: (num_tokens, 2, 2)
    // output[token_idx, expert_num, (idx/weight)]
    // [idx0, w0, idx1, w1] for this token
    if (tid == 0) {

        output[token_idx * 4 + 1] = static_cast<scalar_t>(w1);
        output[token_idx * 4 + 2] = static_cast<scalar_t>(static_cast<float>(top2_idx));
        output[token_idx * 4 + 3] = static_cast<scalar_t>(w2);
    }
}

    torch::Tensor token_hidden,
    torch::Tensor router_logits,
    torch::Tensor expert_ground,
    double alpha) {

    auto output = torch::empty({num_tokens, 2, 2}, token_hidden.options());

    const int threads_per_block = 256;

            expert_ground.data_ptr<scalar_t>(),
            output.data_ptr<scalar_t>(),
            static_cast<float>(alpha),
            num_tokens,
            hidden_dim,
            num_experts);

}
