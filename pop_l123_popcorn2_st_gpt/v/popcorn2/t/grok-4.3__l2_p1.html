import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Low-rank expansion of compressed KV activations into multi-head K/V tensors.

    This isolates the MLA-style compressed-cache pattern: a hidden activation is
    projected into a low-rank latent space and then expanded back into separate
    key and value tensors.
    """

    def __init__(self):
        super().__init__()

    def forward(self, hidden: torch.Tensor, w_down: torch.Tensor, w_up_k: torch.Tensor, w_up_v: torch.Tensor) -> torch.Tensor:
        (batch_size, seq_len, model_dim) = hidden.shape
        rank = w_down.shape[1]
        kv_heads = 4
        head_dim = w_up_k.shape[1] // kv_heads
        latent = hidden.reshape(batch_size * seq_len, model_dim) @ w_down
        k = latent @ w_up_k
        v = latent @ w_up_v
        k = k.view(batch_size, seq_len, kv_heads, head_dim)
        v = v.view(batch_size, seq_len, kv_heads, head_dim)
        return torch.stack((k, v), dim=2)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void mla_expansion_kernel(
    const float* __restrict__ hidden,
    const float* __restrict__ w_down,
    const float* __restrict__ w_up_k,
    const float* __restrict__ w_up_v,
    float* __restrict__ output,
    int batch_size,
    int seq_len,
    int model_dim,
    int rank,
    int kv_heads,
    int head_dim
) {
    int total_tokens = batch_size * seq_len;
    int head_dim_total = kv_heads * head_dim;
    int out_head_dim = head_dim;

    int token_idx = blockIdx.x;
    int head_idx = blockIdx.y;
    int dim_idx = threadIdx.x;

    if (token_idx >= total_tokens || head_idx >= kv_heads || dim_idx >= head_dim) return;

    extern __shared__ float s_latent[];

    // Compute latent for this token if first thread in block or use shared
    // Simpler: each block handles one token, compute latent in shared
    if (dim_idx < rank) {
        float sum = 0.0f;
        for (int d = 0; d < model_dim; ++d) {
            sum += hidden[token_idx * model_dim + d] * w_down[d * rank + dim_idx];
        }
        s_latent[dim_idx] = sum;
    }
    __syncthreads();

    // Now compute k and v for this head
    float k_val = 0.0f;
    float v_val = 0.0f;
    for (int r = 0; r < rank; ++r) {
        float l = s_latent[r];
        k_val += l * w_up_k[r * head_dim_total + head_idx * head_dim + dim_idx];
        v_val += l * w_up_v[r * head_dim_total + head_idx * head_dim + dim_idx];
    }

    // Output layout: (batch, seq, 2, kv_heads, head_dim)
    int b = token_idx / seq_len;
    int s = token_idx % seq_len;
    int out_offset_k = ((b * seq_len + s) * 2 * kv_heads + 0 * kv_heads + head_idx) * head_dim + dim_idx;
    int out_offset_v = ((b * seq_len + s) * 2 * kv_heads + 1 * kv_heads + head_idx) * head_dim + dim_idx;
    output[out_offset_k] = k_val;
    output[out_offset_v] = v_val;
}
