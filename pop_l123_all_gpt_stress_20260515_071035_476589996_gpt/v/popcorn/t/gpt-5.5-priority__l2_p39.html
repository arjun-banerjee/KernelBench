class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        kernel_size: int,
        dilation: int = 1,
    ):
        super().__init__()
        assert kernel_size % 2 == 1, "odd kernel_size keeps T unchanged with symmetric padding"
        assert dilation >= 1
        self.padding = ((kernel_size - 1) * dilation) // 2
        self.dw = nn.Conv1d(
            in_channels,
            in_channels,
            kernel_size,
            padding=self.padding,
            dilation=dilation,
            groups=in_channels,
            bias=False,
        )
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

    int row1 = row0 + 16;
    int col0 = blockIdx.x * BN + tx;
    int col1 = col0 + 16;

    float acc00 = 0.0f, acc01 = 0.0f, acc10 = 0.0f, acc11 = 0.0f;

    int b0 = row0 / T, t0 = row0 - b0 * T;
    int b1 = row1 / T, t1 = row1 - b1 * T;

    for (int k0 = 0; k0 < C; k0 += BK) {
        int kA0 = k0 + tx;
        int kA1 = kA0 + 16;
        As[ty][tx]       = (row0 < M && kA0 < C) ? tf32(a[(b0 * C + kA0) * T + t0]) : 0.0f;
        As[ty][tx + 16]  = (row0 < M && kA1 < C) ? tf32(a[(b0 * C + kA1) * T + t0]) : 0.0f;
        As[ty + 16][tx]  = (row1 < M && kA0 < C) ? tf32(a[(b1 * C + kA0) * T + t1]) : 0.0f;
        As[ty + 16][tx + 16] = (row1 < M && kA1 < C) ? tf32(a[(b1 * C + kA1) * T + t1]) : 0.0f;

        Bs[ty][tx]       = (kB0 < C && col0 < O) ? tf32(w[col0 * C + kB0]) : 0.0f;
        Bs[ty][tx + 16]  = (kB0 < C && col1 < O) ? tf32(w[col1 * C + kB0]) : 0.0f;
        Bs[ty + 16][tx]  = (kB1 < C && col0 < O) ? tf32(w[col0 * C + kB1]) : 0.0f;
        Bs[ty + 16][tx + 16] = (kB1 < C && col1 < O) ? tf32(w[col1 * C + kB1]) : 0.0f;
        __syncthreads();

        #pragma unroll
        for (int kk = 0; kk < BK; ++kk) {
            float a0 = As[ty][kk];
            float a1 = As[ty + 16][kk];
            float b0v = Bs[kk][tx];
            float b1v = Bs[kk][tx + 16];
            acc00 += a0 * b0v;
            acc01 += a0 * b1v;
            acc10 += a1 * b0v;
            acc11 += a1 * b1v;
        }
        __syncthreads();
