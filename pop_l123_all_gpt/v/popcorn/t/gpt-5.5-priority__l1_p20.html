import torch
import torch.nn as nn
import math


class Model(nn.Module):
    """
    Rotary Position Embedding (RoPE) applied to protein residue sequences.
    Used in ESM-2 and similar protein language models.  Encodes absolute
    position via rotation of query/key pairs in 2-D subspaces, preserving
    relative-position information in dot-product attention.
    """

    def __init__(self, dim, max_seq_len=2048, base=10000.0):
        super().__init__()
        inv_freq = 1.0 / (base ** (torch.arange(0, dim, 2).float() / dim))
        self.register_buffer("inv_freq", inv_freq)
        self.max_seq_len = max_seq_len
        self.dim = dim

    def _build_cos_sin(self, seq_len: int, device: torch.device):
        t = torch.arange(seq_len, device=device, dtype=self.inv_freq.dtype)
        freqs = torch.outer(t, self.inv_freq)
        emb = torch.cat([freqs, freqs], dim=-1)
        return emb.cos(), emb.sin()

    @staticmethod
    def _rotate_half(x: torch.Tensor) -> torch.Tensor:
        x1, x2 = x.chunk(2, dim=-1)
        return torch.cat([-x2, x1], dim=-1)

    def forward(self, q: torch.Tensor, k: torch.Tensor) -> tuple:
        """
        Args:
            q: (B, H, N, D)  – queries
            k: (B, H, N, D)  – keys
        Returns:
            (q_rot, k_rot) each (B, H, N, D) with RoPE applied
        """
        N = q.shape[2]
        cos, sin = self._build_cos_sin(N, q.device)  # (N, D)
        cos = cos.unsqueeze(0).unsqueeze(0)
        sin = sin.unsqueeze(0).unsqueeze(0)
        q_rot = q * cos + self._rotate_half(q) * sin
        k_rot = k * cos + self._rotate_half(k) * sin
        return q_rot, k_rot

__device__ __forceinline__ void rotate4(const float4 a0, const float4 a1,
                                        const float4 c, const float4 s,
                                        float4 &o0, float4 &o1) {
    o0.x = a0.x*c.x - a1.x*s.x; o1.x = a1.x*c.x + a0.x*s.x;
    o0.y = a0.y*c.y - a1.y*s.y; o1.y = a1.y*c.y + a0.y*s.y;
    o0.z = a0.z*c.z - a1.z*s.z; o1.z = a1.z*c.z + a0.z*s.z;
    o0.w = a0.w*c.w - a1.w*s.w; o1.w = a1.w*c.w + a0.w*s.w;
}

__global__ __launch_bounds__(256) void rope64_vec4_kernel(
    const float4* __restrict__ q4,
    const float4* __restrict__ k4,
    const float4* __restrict__ cos4,
    const float4* __restrict__ sin4,
    float4* __restrict__ qo4,
    float4* __restrict__ ko4,
    int tasks, int N) {

    int idx = blockIdx.x * 256 + threadIdx.x;
    if (idx >= tasks) return;

    // dim is fixed by benchmark to 64. One task processes four dimensions
    // from the first half and the matching four from the second half.
    int row = idx >> 3;      // B*H*N row
    int chunk = idx & 7;     // 0..7 float4 chunks in the half dimension
    int n = row % N;
    int row4 = row << 4;     // 64 floats == 16 float4 per row

    float4 c = cos4[(n << 3) + chunk];
    float4 s = sin4[(n << 3) + chunk];

    float4 q0 = q4[row4 + chunk];
    float4 q1 = q4[row4 + 8 + chunk];
    float4 k0 = k4[row4 + chunk];
    float4 k1 = k4[row4 + 8 + chunk];

    float4 qo0, qo1, ko0, ko1;
    rotate4(q0, q1, c, s, qo0, qo1);
    rotate4(k0, k1, c, s, ko0, ko1);

    qo4[row4 + chunk] = qo0;
    qo4[row4 + 8 + chunk] = qo1;
    ko4[row4 + chunk] = ko0;
    ko4[row4 + 8 + chunk] = ko1;
}
