            int t_input = t - padding + k * dilation;
            if (t_input >= 0 && t_input < seq_len) {
                float val = input[(b * in_channels + ic) * seq_len + t_input];
                float w = dw_weight[ic * kernel_size + k];
                dw_sum += val * w;
            }
        }
        
        // SiLU: x * sigmoid(x)
        float sig = 1.0f / (1.0f + expf(-dw_sum));
        float activated = dw_sum * sig;
        float pw_w = pw_weight[oc * in_channels + ic];
        sum += activated * pw_w;
    }
    
    output[(b * out_channels + oc) * seq_len + t] = sum;
}
