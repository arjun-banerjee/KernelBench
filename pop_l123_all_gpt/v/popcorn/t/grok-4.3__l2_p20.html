import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Scalar-gated delta-rule linear attention recurrence.

    A recurrent state matrix is decayed by a scalar gate per head, updated by a
    rank-1 key/value outer product, then read out by a query vector.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        gate: torch.Tensor,
        beta: torch.Tensor,
    ) -> torch.Tensor:
        batch_size, seq_len, num_heads, head_dim = q.shape
        state = torch.zeros(batch_size, num_heads, head_dim, head_dim, dtype=q.dtype, device=q.device)
        outputs = []

        for t in range(seq_len):
            gate_t = gate[:, t].unsqueeze(-1).unsqueeze(-1)
            beta_t = beta[:, t].unsqueeze(-1).unsqueeze(-1)
            kv_outer = k[:, t].unsqueeze(-1) * v[:, t].unsqueeze(-2)
            state = gate_t * state + beta_t * kv_outer
            y_t = (q[:, t].unsqueeze(-1) * state).sum(dim=-2)
            outputs.append(y_t)

        return torch.stack(outputs, dim=1)

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void delta_rule_kernel(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    const float* __restrict__ beta,
    float* __restrict__ output,
    int batch_size,
    int seq_len,
    int num_heads,
    int head_dim
) {
    int bh = blockIdx.x;
    int b = bh / num_heads;
    int h = bh % num_heads;
    if (b >= batch_size) return;

    extern __shared__ float state[];  // head_dim * head_dim

    int D = head_dim;
    for (int i = threadIdx.x; i < D * D; i += blockDim.x) {
        state[i] = 0.0f;
    }
    __syncthreads();

    for (int t = 0; t < seq_len; t++) {
        float g = gate[(b * seq_len + t) * num_heads + h];
        float be = beta[(b * seq_len + t) * num_heads + h];

        // update state: state = g * state + be * outer(k, v)
        for (int i = threadIdx.x; i < D * D; i += blockDim.x) {
            int row = i / D;
            int col = i % D;
            float kv = k[((b * seq_len + t) * num_heads + h) * D + row] *
                       v[((b * seq_len + t) * num_heads + h) * D + col];
            float old = state[i];
            state[i] = g * old + be * kv;
        }
        __syncthreads();

        // compute y = q @ state  (sum over dim -2, i.e. q row vector * state)
        for (int j = threadIdx.x; j < D; j += blockDim.x) {
            float sum = 0.0f;
            for (int i = 0; i < D; i++) {
                float qval = q[((b * seq_len + t) * num_heads + h) * D + i];
                sum += qval * state[i * D + j];
            }
            output[((b * seq_len + t) * num_heads + h) * D + j] = sum;
        }
        __syncthreads();
    }
}

torch::Tensor delta_rule_cuda(
    torch::Tensor q,
    torch::Tensor k,
    torch::Tensor v,
    torch::Tensor gate,
    torch::Tensor beta
) {
    int batch_size = q.size(0);
    int seq_len = q.size(1);
    int num_heads = q.size(2);
    int head_dim = q.size(3);

    auto output = torch::zeros({batch_size, seq_len, num_heads, head_dim}, q.options());

    int threads = 256;
    int blocks = batch_size * num_heads;
    int smem = head_dim * head_dim * sizeof(float);

    delta_rule_kernel<<<blocks, threads, smem>>>(
        q.data_ptr<float>(),
        k.data_ptr<float>(),
        v.data_ptr<float>(),
        gate.data_ptr<float>(),
        beta.data_ptr<float>(),
        output.data_ptr<float>(),
        batch_size, seq_len, num_heads, head_dim
    );

    return output;
}
