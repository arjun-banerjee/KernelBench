class Model(nn.Module):
    """
    **Fused probability-flow ODE integration** (flow-matching style) in one ``forward``.

    Models the ODE ``dx/dt = v_θ(x, t)`` on ``t ∈ [0, 1]`` with a learned velocity
    field. A **single fused pass** unrolls **fixed-step Euler** integration for
    ``num_steps`` substeps—typical for **real-time trajectory / action generation**
    where latency is dominated by repeated ``v_θ`` evaluations.

    This is the continuous-time analogue of a rectified-flow / conditional-flow
    sampler: one ``forward`` returns an approximate **x(1)** from **x(0)** without a
    Python loop across separate module calls (fusion target for the integrator +
    field).
    """

    def __init__(self, dim: int, num_steps: int):
        super().__init__()
        assert dim >= 1 and num_steps >= 1
        self.dim = dim
        self.num_steps = num_steps
        self.register_buffer("dt", torch.tensor(1.0 / float(num_steps)))

        self.velocity = nn.Sequential(
            nn.Linear(dim + 1, dim * 4),
            nn.SiLU(),
            nn.Linear(dim * 4, dim * 4),
            nn.SiLU(),
            nn.Linear(dim * 4, dim),
        )

    def forward(self, x0: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x0: (B, dim) initial state at ``t = 0`` (e.g. noise or anchor trajectory).
        Returns:
            (B, dim) approximate state at ``t = 1`` after fused Euler integration.
        """
        B, d = x0.shape
        assert d == self.dim
        x = x0
        dt = self.dt.to(dtype=x0.dtype, device=x0.device)
        for k in range(self.num_steps):
            t = (float(k) + 0.5) / float(self.num_steps)
            t_b = torch.full((B, 1), t, device=x0.device, dtype=x0.dtype)
            inp = torch.cat([x, t_b], dim=-1)
            v = self.velocity(inp)
            x = x + dt * v
        return x

    v += __shfl_down_sync(0xffffffff, v, 2);
    v += __shfl_down_sync(0xffffffff, v, 1);
    return v;
}

__device__ void global_barrier(int total_blocks) {
    __syncthreads();
    if (threadIdx.x == 0) {
        __threadfence();
        unsigned int s = *((volatile unsigned int*)&g_bar_sense);
        unsigned int ticket = atomicAdd(&g_bar_count, 1u);
        if (ticket == (unsigned int)(total_blocks - 1)) {
            g_bar_count = 0;
            __threadfence();
            atomicAdd(&g_bar_sense, 1u);
        } else {
            while (*((volatile unsigned int*)&g_bar_sense) == s) { }
        }
    }
    __syncthreads();
}

__global__ void ode64_grid_kernel(const float* __restrict__ x0,
                                  const float* __restrict__ w0, const float* __restrict__ b0,
                                  const float* __restrict__ w1, const float* __restrict__ b1,
                                  const float* __restrict__ w2, const float* __restrict__ b2,
                                  float* __restrict__ xbuf, float* __restrict__ h1, float* __restrict__ h2,
                                  int num_steps, int total_blocks) {
    int b = blockIdx.x;
    int tile = blockIdx.y;
    int tid = threadIdx.x;
    int lane = tid & 31;
    int warp = tid >> 5;
    int idx = tile * 8 + warp;  // 128 independent warps per batch item for grid.y=16

    if (tile == 0 && tid < 64) xbuf[b * 64 + tid] = x0[b * 64 + tid];
    global_barrier(total_blocks);

    float dt = 1.0f / (float)num_steps;
    for (int k = 0; k < num_steps; ++k) {
        float t = ((float)k + 0.5f) * dt;

        // Layer 0: 65 -> 256. Each warp computes one output; with 128 warps two outputs/warp over loop.
        for (int o = idx; o < 256; o += 128) {
            const float* ww = w0 + o * 65;
            const float* xb = xbuf + b * 64;
            float sum = 0.0f;
            sum = xb[lane] * ww[lane];
            sum = fmaf(xb[lane + 32], ww[lane + 32], sum);
            if (lane == 0) sum = fmaf(t, ww[64], sum);
            sum = warp_sum(sum);
            if (lane == 0) h1[b * 256 + o] = silu_f(sum + b0[o]);
        }
        global_barrier(total_blocks);

        // Layer 1: 256 -> 256.
        for (int o = idx; o < 256; o += 128) {
            const float* ww = w1 + o * 256;
            const float* hp = h1 + b * 256;
            float sum = 0.0f;
            #pragma unroll
            for (int j = lane; j < 256; j += 32) sum = fmaf(hp[j], ww[j], sum);
            sum = warp_sum(sum);
            if (lane == 0) h2[b * 256 + o] = silu_f(sum + b1[o]);
        }
        global_barrier(total_blocks);

        // Layer 2: 256 -> 64 and Euler update x in-place.
        if (idx < 64) {
            int o = idx;
            const float* ww = w2 + o * 256;
            const float* hp = h2 + b * 256;
            float sum = 0.0f;
            #pragma unroll
            for (int j = lane; j < 256; j += 32) sum = fmaf(hp[j], ww[j], sum);
            sum = warp_sum(sum);
            if (lane == 0) xbuf[b * 64 + o] = fmaf(dt, sum + b2[o], xbuf[b * 64 + o]);
        }
        global_barrier(total_blocks);
    }
}

                               torch::Tensor w1, torch::Tensor b1, torch::Tensor w2,
                               torch::Tensor b2, int64_t num_steps) {
    int B = x0.size(0);
    auto opts = x0.options();
    auto xbuf = torch::empty({B, 64}, opts);
    auto h1 = torch::empty({B, 256}, opts);
    auto h2 = torch::empty({B, 256}, opts);
    dim3 grid(B, 16);

                                     w0.data_ptr<float>(), b0.data_ptr<float>(),

                                     xbuf.data_ptr<float>(), h1.data_ptr<float>(), h2.data_ptr<float>(),

    return xbuf;
}
