from torch.utils.cpp_extension import load_inline

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void depthwise_conv1d_kernel(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size,
    int channels,
    int seq_len,
    int kernel_size,
    int padding,
    int stride,
    bool has_bias
) {
    // Each thread handles one output element
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    
    int total_outputs = batch_size * channels * seq_len;
    if (tid >= total_outputs) return;
    
    int out_pos = tid % seq_len;
    int c = (tid / seq_len) % channels;
    int b = tid / (channels * seq_len);
    
    // Weight index: kernel for channel c is at index c
    // Weight layout: [channels, 1, kernel_size] -> indexed as [c * kernel_size + k]
    
    float sum = 0.0f;
    
    // For each kernel position
    for (int k = 0; k < kernel_size; k++) {
        int in_pos = out_pos * stride + k - padding;
        if (in_pos >= 0 && in_pos < seq_len) {
            int in_idx = b * channels * seq_len + c * seq_len + in_pos;
            int w_idx = c * kernel_size + k;
            sum += input[in_idx] * weight[w_idx];
        }
    }
    
    if (has_bias) {
        sum += bias[c];
    }
    
    int out_idx = b * channels * seq_len + c * seq_len + out_pos;
    output[out_idx] = sum;
}

torch::Tensor depthwise_conv1d_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    c10::optional<torch::Tensor> bias,
    int64_t stride,
    int64_t padding,
    int64_t groups
) {
    int batch_size = input.size(0);
    int channels = input.size(1);
    int seq_len = input.size(2);
    int kernel_size = weight.size(2);

    def __init__(self, channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(ModelNew, self).__init__()
        self.channels = channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # Create weights same as Conv1d for depthwise
        # Depthwise: weight shape is [channels, 1, kernel_size]
        self.weight = nn.Parameter(torch.empty(channels, 1, kernel_size))
        nn.init.kaiming_uniform_(self.weight, a=5 ** 0.5)
