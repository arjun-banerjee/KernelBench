import torch
import torch.nn as nn

class Model(nn.Module):
    """
    Performs a depthwise 1D convolution where each input channel is convolved with its own filter.
    Fundamental primitive in Hyena, Mamba, and genomic sequence models (Enformer, HyenaDNA).

    Args:
        channels (int): Number of input/output channels.
        kernel_size (int): Size of the convolution kernel.
        stride (int, optional): Stride of the convolution. Defaults to 1.
        padding (int, optional): Padding applied to the input. Defaults to 0.
        bias (bool, optional): If True, adds a learnable bias. Defaults to False.
    """
    def __init__(self, channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(Model, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.conv1d(x)

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = '''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void depthwise_conv1d_kernel_v2(
    const float* __restrict__ input,
    const float* __restrict__ weight,
    const float* __restrict__ bias,
    float* __restrict__ output,
    int batch_size,
    int channels,
    int seq_len,
    int kernel_size,
    int stride,
    int padding,
    bool has_bias
) {
    int b = blockIdx.x;
    int c = blockIdx.y;
    int s = threadIdx.x + blockIdx.z * blockDim.x;

    if (s >= seq_len) return;

    float sum = 0.0f;
    #pragma unroll
    for (int k = 0; k < kernel_size; ++k) {
        int in_pos = s * stride + k - padding;
        if (in_pos >= 0 && in_pos < seq_len) {
            sum += input[(b * channels + c) * seq_len + in_pos] * weight[c * kernel_size + k];
        }
    }
    if (has_bias) sum += bias[c];
    output[(b * channels + c) * seq_len + s] = sum;
}

torch::Tensor depthwise_conv1d_cuda(
    torch::Tensor input,
    torch::Tensor weight,
    torch::Tensor bias,
    int stride,
    int padding
) {
    int batch_size = input.size(0);
    int channels = input.size(1);
    int seq_len = input.size(2);
    int kernel_size = weight.size(2);
    bool has_bias = bias.defined() && bias.numel() > 0;

    auto output = torch::empty({batch_size, channels, seq_len}, input.options());

    dim3 threads(256);
    dim3 blocks(batch_size, channels, (seq_len + 255) / 256);

    depthwise_conv1d_kernel_v2<<<blocks, threads>>>(
        input.data_ptr<float>(),
        weight.data_ptr<float>(),
        has_bias ? bias.data_ptr<float>() : nullptr,
        output.data_ptr<float>(),
        batch_size, channels, seq_len, kernel_size, stride, padding, has_bias
    );

    return output;
}

class ModelNew(nn.Module):
    def __init__(self, channels, kernel_size, stride=1, padding=0, bias=False):
        super(ModelNew, self).__init__()
        self.conv1d = nn.Conv1d(channels, channels, kernel_size, stride=stride, padding=padding, groups=channels, bias=bias)
        self.stride = stride
        self.padding = padding

    def forward(self, x):
        weight = self.conv1d.weight
        bias = self.conv1d.bias if self.conv1d.bias is not None else torch.tensor([], device=x.device)
        return module.depthwise_conv1d_cuda(x, weight, bias, self.stride, self.padding)
