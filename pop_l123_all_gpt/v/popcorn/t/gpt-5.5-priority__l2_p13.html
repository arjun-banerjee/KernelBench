def _fake_quant_symmetric_int8(x: torch.Tensor, scale: torch.Tensor) -> torch.Tensor:
    """Symmetric int8 fake-quant: round(x/scale) clamped to [-127, 127], times scale."""
    q = torch.clamp(torch.round(x / scale), -127.0, 127.0)
    return q * scale


class Model(nn.Module):
    """
    **QuantVLA selective layout fusion** (reference target for a fused CUDA/Triton kernel).

    Vision–language–action stacks often use **token-wise mixed precision**: some
    timesteps or modalities tolerate aggressive quantization while others stay in
    full precision. This module fuses in **one forward**:

      1. **Layout routing** — ``sigmoid(layout_logits)`` yields a per-token mix
         coefficient in ``(0, 1)`` (quant vs FP residual).
      2. **Quantitative path** — global symmetric **int8-style fake quantization**
         (scale from the tensor max, ``/ 127``).
      3. **Selective blend** — ``α · x_quant + (1 - α) · x``.
      4. **Head** — linear projection on the mixed activations.

    This matches a “quantitative selective layout” block: the layout logits (from a
    router, modality tags, or calibration) choose how strongly each position uses the
    quantized layout vs the identity layout.
    """

    def __init__(self, dim: int):
        super().__init__()
        self.dim = dim
        self.out_proj = nn.Linear(dim, dim, bias=True)

    def forward(self, x: torch.Tensor, layout_logits: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, T, D) VLA token activations (fused modalities).
            layout_logits: (B, T) unbounded logits; larger → more mass on quantized path.
        Returns:
            (B, T, D) updated activations.
        """
        B, T, D = x.shape
        assert D == self.dim
        assert layout_logits.shape == (B, T)

        alpha = torch.sigmoid(layout_logits).unsqueeze(-1)
        with torch.no_grad():
            amax = x.detach().abs().max().clamp(min=1e-8)
        scale = amax / 127.0
        x_q = _fake_quant_symmetric_int8(x, scale)
        mixed = alpha * x_q + (1.0 - alpha) * x
        return self.out_proj(mixed)

#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <math.h>

#define DVAL 128

__global__ void reduce_amax_kernel(const float* __restrict__ x, float* __restrict__ scale, int n) {
    __shared__ float smem[256];
    int tid = threadIdx.x;
    float m = 0.0f;
    for (int i = tid; i < n; i += 256) m = fmaxf(m, fabsf(x[i]));
    smem[tid] = m;
    __syncthreads();
    for (int s = 128; s > 0; s >>= 1) {
        if (tid < s) smem[tid] = fmaxf(smem[tid], smem[tid + s]);
        __syncthreads();
    }
    if (tid == 0) scale[0] = fmaxf(smem[0], 1.0e-8f) * (1.0f / 127.0f);
}

__device__ __forceinline__ float mix_val(float xv, float inv_scale, float scale, float alpha) {
    float q = nearbyintf(xv * inv_scale);
    q = fminf(127.0f, fmaxf(-127.0f, q));
    return fmaf(alpha, q * scale - xv, xv);
}

__global__ void warp4_kernel(const float* __restrict__ x,
                             const float* __restrict__ logits,
                             const float* __restrict__ weight,
                             const float* __restrict__ bias,
                             const float* __restrict__ scale_ptr,
                             float* __restrict__ out,
                             int N) {
    int lane = threadIdx.x;
    int wy = threadIdx.y;
    int row = blockIdx.y * 8 + wy;
    int c0 = blockIdx.x * 4;
    if (row >= N) return;

    float scale = scale_ptr[0];
    float inv_scale = 1.0f / scale;
    float alpha = 1.0f / (1.0f + __expf(-logits[row]));

    const float* xp = x + row * DVAL;
    const float* w0 = weight + (c0 + 0) * DVAL;
    const float* w1 = w0 + DVAL;
    const float* w2 = w1 + DVAL;
    const float* w3 = w2 + DVAL;

    float a0 = 0.0f, a1 = 0.0f, a2 = 0.0f, a3 = 0.0f;

    int k = lane;
    float m = mix_val(xp[k], inv_scale, scale, alpha);
    a0 = fmaf(m, w0[k], a0); a1 = fmaf(m, w1[k], a1); a2 = fmaf(m, w2[k], a2); a3 = fmaf(m, w3[k], a3);
    k = lane + 32;
    m = mix_val(xp[k], inv_scale, scale, alpha);
    a0 = fmaf(m, w0[k], a0); a1 = fmaf(m, w1[k], a1); a2 = fmaf(m, w2[k], a2); a3 = fmaf(m, w3[k], a3);
    k = lane + 64;
    m = mix_val(xp[k], inv_scale, scale, alpha);
    a0 = fmaf(m, w0[k], a0); a1 = fmaf(m, w1[k], a1); a2 = fmaf(m, w2[k], a2); a3 = fmaf(m, w3[k], a3);
    k = lane + 96;
    m = mix_val(xp[k], inv_scale, scale, alpha);
    a0 = fmaf(m, w0[k], a0); a1 = fmaf(m, w1[k], a1); a2 = fmaf(m, w2[k], a2); a3 = fmaf(m, w3[k], a3);

    #pragma unroll
    for (int off = 16; off > 0; off >>= 1) {
        a0 += __shfl_down_sync(0xffffffff, a0, off);
        a1 += __shfl_down_sync(0xffffffff, a1, off);
        a2 += __shfl_down_sync(0xffffffff, a2, off);
        a3 += __shfl_down_sync(0xffffffff, a3, off);
    }

    if (lane == 0) {
        float* op = out + row * DVAL + c0;
        op[0] = a0 + bias[c0 + 0];
        op[1] = a1 + bias[c0 + 1];
        op[2] = a2 + bias[c0 + 2];
        op[3] = a3 + bias[c0 + 3];
    }
}

torch::Tensor quant_vla_forward(torch::Tensor x, torch::Tensor layout_logits, torch::Tensor weight, torch::Tensor bias) {
    auto xc = x.contiguous();
    auto lc = layout_logits.contiguous();
    auto wc = weight.contiguous();
    auto bc = bias.contiguous();
    int N = (int)(xc.size(0) * xc.size(1));

    auto out = torch::empty_like(xc);
    auto scale = torch::empty({1}, xc.options());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    reduce_amax_kernel<<<1, 256, 0, stream>>>(xc.data_ptr<float>(), scale.data_ptr<float>(), N * DVAL);
    dim3 block(32, 8);
    dim3 grid(32, (N + 7) / 8);
    warp4_kernel<<<grid, block, 0, stream>>>(xc.data_ptr<float>(), lc.data_ptr<float>(), wc.data_ptr<float>(), bc.data_ptr<float>(), scale.data_ptr<float>(), out.data_ptr<float>(), N);
    return out;
}
'''

_ext = load_inline(
    name='quant_vla_ext_final',
    cpp_sources=cpp_src,
    cuda_sources=cuda_src,
    functions=['quant_vla_forward'],
    extra_cuda_cflags=['-O3', '--use_fast_math'],
    extra_cflags=['-O3'],
    verbose=False,
)

class ModelNew(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        self.dim = dim
        self.out_proj = nn.Linear(dim, dim, bias=True)

    def forward(self, x: torch.Tensor, layout_logits: torch.Tensor) -> torch.Tensor:
        return _ext.quant_vla_forward(x, layout_logits, self.out_proj.weight, self.out_proj.bias)
