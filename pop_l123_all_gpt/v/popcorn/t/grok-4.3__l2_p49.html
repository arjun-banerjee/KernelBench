import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__device__ __forceinline__ float gelu(float x) {
    return x * 0.5f * (1.0f + erff(x * 0.7071067811865475f));
}

__global__ void gelu_sum_kernel(const float* __restrict__ x, float* __restrict__ out, int R, int B, int D) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * D;
    if (idx >= total) return;
    float sum = 0.0f;
    for (int r = 0; r < R; r++) {
        float val = x[r * total + idx];
        sum += gelu(val);
    }
    out[idx] = sum;
}

torch::Tensor gelu_sum_cuda(torch::Tensor x) {
    int R = x.size(0);
    int B = x.size(1);
    int D = x.size(2);
    auto out = torch::empty({B, D}, x.options());
    int total = B * D;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    gelu_sum_kernel<<<blocks, threads>>>(x.data_ptr<float>(), out.data_ptr<float>(), R, B, D);
    return out;
}
"""

torch::Tensor gelu_sum_cuda(torch::Tensor x);
