#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

// Column-based approach: all threads load same weight column at once
template <typename scalar_t>
__global__ void __launch_bounds__(128) gather_linear_kernel(
    const scalar_t* __restrict__ x,  // shape (R, B, H)
    const scalar_t* __restrict__ weight,  // shape (K, R*H) row-major
    const scalar_t* __restrict__ bias,  // shape (K)
    scalar_t* __restrict__ out,  // shape (B, K)
    int R, int B, int H, int K
) {
    const int b = blockIdx.x;  // batch index
    const int tid = threadIdx.x;
    const int RH = R * H;
    
    // Shared memory for x vector
    __shared__ scalar_t xs[128];  // R*H = 128
    // Shared memory for a column of weight (K values)
    __shared__ scalar_t w_col[128];  // K values, padded to 128
    
    // Load x into shared memory cooperatively
    for (int i = tid; i < RH; i += blockDim.x) {
        int r = i / H;
        int h = i % H;
        xs[i] = x[r * B * H + b * H + h];
    }
    __syncthreads();
    
    // Initialize output with bias
    scalar_t sum = (tid < K) ? bias[tid] : scalar_t(0);
    
    // Process weight columns one at a time
    // All threads load the same column rh at once, coalesced!
    for (int rh = 0; rh < RH; rh++) {
        // Load weight column rh: weight[k, rh] = weight[k * RH + rh]
        // Threads 0..K-1 load weight[0..K-1, rh] -> offsets 0*RH+rh, 1*RH+rh, etc.
        // These are strided by RH=128, NOT coalesced!
        // But we can use __ldg for read-only cache
        
        scalar_t w_val = (tid < K) ? __ldg(&weight[tid * RH + rh]) : scalar_t(0);
        w_col[tid] = w_val;
        __syncthreads();
        
        // Multiply by x[rh] and accumulate
        sum += xs[rh] * w_col[tid];
    }
    
    // Write result
    if (tid < K) {
        out[b * K + tid] = sum;
    }
}

torch::Tensor gather_linear_cuda(
    torch::Tensor x,  // shape (R, B, H)
    torch::Tensor weight,  // shape (K, R*H)
    torch::Tensor bias  // shape (K)
) {
    const int R = x.size(0);
    const int B = x.size(1);
    const int H = x.size(2);
    const int K = weight.size(0);
    
    auto out = torch::empty({B, K}, x.options());
    
    // One block per batch, 128 threads
    const int threads = 128;
    const dim3 grid(B);
    
    AT_DISPATCH_FLOATING_TYPES(x.scalar_type(), "gather_linear_kernel", ([&] {
        gather_linear_kernel<scalar_t><<<grid, threads>>>(
            x.data_ptr<scalar_t>(),
            weight.data_ptr<scalar_t>(),
            bias.data_ptr<scalar_t>(),
            out.data_ptr<scalar_t>(),
            R, B, H, K

    }));
    
    return out;
}
