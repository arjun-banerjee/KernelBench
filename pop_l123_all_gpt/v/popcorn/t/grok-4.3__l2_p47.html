"""
Virtual all-gather on the last dim: stack R tensors of shape (B, H) into (B, R*H), then Linear.
"""

import torch
import torch.nn as nn

from kernelbench.distributed_collectives import default_device

R = 4
B = 12
H = 32
K = 56


class Model(nn.Module):
    def __init__(self):
        super().__init__()
        self.proj = nn.Linear(R * H, K, bias=True)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        r, b, h = x.shape
        assert r == R and h == H
        parts = [x[i] for i in range(R)]
        gathered = torch.cat(parts, dim=-1)
        return self.proj(gathered)

    return [torch.randn(R, B, H, device=dev, generator=g)]

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void fused_allgather_linear_kernel(
    const float* __restrict__ x,  // (R, B, H)
    const float* __restrict__ weight,  // (K, R*H)
    const float* __restrict__ bias,    // (K,)
    float* __restrict__ out,           // (B, K)
    int R, int B, int H, int K
) {
    int b = blockIdx.x;
    int k = threadIdx.x;
    if (b >= B || k >= K) return;

    float sum = bias[k];
    const int RH = R * H;
    for (int r = 0; r < R; ++r) {
        for (int h = 0; h < H; ++h) {
            float val = x[r * B * H + b * H + h];
            float w = weight[k * RH + r * H + h];
            sum += val * w;
        }
    }
    out[b * K + k] = sum;
}

torch::Tensor fused_allgather_linear(
    torch::Tensor x,
    torch::Tensor weight,
    torch::Tensor bias
) {
    int R = x.size(0);
    int B = x.size(1);
    int H = x.size(2);
    int K = weight.size(0);
    auto out = torch::empty({B, K}, x.options());
    dim3 grid(B);
    dim3 block(K);
    fused_allgather_linear_kernel<<<grid, block>>>(
        x.data_ptr<float>(),
        weight.data_ptr<float>(),
        bias.data_ptr<float>(),
        out.data_ptr<float>(),
        R, B, H, K
    );
    return out;
}
'''
