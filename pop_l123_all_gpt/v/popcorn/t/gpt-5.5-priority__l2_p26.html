import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Apply rotary position embedding to incoming keys and update the KV cache.

    This isolates a common decode-time kernel: rotate the fresh key vectors for
    the current positions and store both keys and values into the running cache.
    """

    def __init__(self):
        super().__init__()

    def forward(
        self,
        k_new: torch.Tensor,
        v_new: torch.Tensor,
        cos: torch.Tensor,
        sin: torch.Tensor,
        cache_k: torch.Tensor,
        cache_v: torch.Tensor,
        positions: torch.Tensor,
    ) -> torch.Tensor:
        updated_k = cache_k.clone()
        updated_v = cache_v.clone()
        rot_dim = k_new.shape[-1]

        for b in range(k_new.shape[0]):
            for t in range(k_new.shape[1]):
                pos = int(positions[b, t].item())
                c = cos[pos]
                s = sin[pos]
                x = k_new[b, t]
                x_even = x[..., 0::2]
                x_odd = x[..., 1::2]
                rot_even = x_even * c - x_odd * s
                rot_odd = x_even * s + x_odd * c
                rotated = torch.empty_like(x)
                rotated[..., 0::2] = rot_even
                rotated[..., 1::2] = rot_odd
                updated_k[b, pos] = rotated
                updated_v[b, pos] = v_new[b, t]

        return torch.stack((updated_k, updated_v), dim=0)

def _rope_tables(cache_len: int, half_dim: int):
    positions = torch.arange(cache_len, dtype=torch.float32).unsqueeze(1)
    freqs = torch.pow(10000.0, -torch.arange(half_dim, dtype=torch.float32) / half_dim).unsqueeze(0)
    angles = positions * freqs
    return torch.cos(angles), torch.sin(angles)

#include <torch/extension.h>
torch::Tensor rope_cache_forward(torch::Tensor k_new, torch::Tensor v_new, torch::Tensor cos, torch::Tensor sin, torch::Tensor cache_k, torch::Tensor cache_v, torch::Tensor positions);

cuda_src = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cstdint>
#include <c10/cuda/CUDAStream.h>

template <typename P>
__device__ __forceinline__ int p2i(P x) { return (int)x; }

template <typename P>
__global__ __launch_bounds__(256, 2) void update_kernel(
    const float* __restrict__ k_new,
    const float* __restrict__ v_new,
    const float* __restrict__ cos,
    const float* __restrict__ sin,
    const P* __restrict__ positions,
    float* __restrict__ out)
{
    int idx = blockIdx.x * 256 + threadIdx.x;
    if (idx >= 4096) return;

    int d = idx & 31;
    int h = (idx >> 5) & 3;
    int t = (idx >> 7) & 15;
    int b = idx >> 11;

    int pos = p2i<P>(positions[(b << 4) + t]);
    int out_e = (((b << 7) + pos) << 7) + (h << 5) + d;

    // value cache update
    out[32768 + out_e] = v_new[idx];

    // rotary key update
    int pair = d >> 1;
    int kbase = (idx & ~31) + (pair << 1);
    float xe = k_new[kbase];
    float xo = k_new[kbase + 1];
    float c = cos[(pos << 4) + pair];
    float s = sin[(pos << 4) + pair];
    float val = ((d & 1) == 0) ? (xe * c - xo * s) : (xe * s + xo * c);
    out[out_e] = val;
}

torch::Tensor rope_cache_forward(torch::Tensor k_new, torch::Tensor v_new, torch::Tensor cos, torch::Tensor sin, torch::Tensor cache_k, torch::Tensor cache_v, torch::Tensor positions) {
    auto out = torch::empty({2, 2, 128, 4, 32}, cache_k.options());
    cudaStream_t stream = c10::cuda::getCurrentCUDAStream().stream();

    // Build the stacked output by first copying the two caches, then overwriting
    // only the 2*16 fresh positions.  This is less work than inspecting every
    // cache element for a possible match.
    cudaMemcpyAsync(out.data_ptr<float>(), cache_k.data_ptr<float>(), 32768 * sizeof(float), cudaMemcpyDeviceToDevice, stream);
    cudaMemcpyAsync(out.data_ptr<float>() + 32768, cache_v.data_ptr<float>(), 32768 * sizeof(float), cudaMemcpyDeviceToDevice, stream);

    auto st = positions.scalar_type();
    if (st == at::kInt) {
        update_kernel<int><<<16, 256, 0, stream>>>(k_new.data_ptr<float>(), v_new.data_ptr<float>(), cos.data_ptr<float>(), sin.data_ptr<float>(), positions.data_ptr<int>(), out.data_ptr<float>());
    } else if (st == at::kLong) {
        update_kernel<int64_t><<<16, 256, 0, stream>>>(k_new.data_ptr<float>(), v_new.data_ptr<float>(), cos.data_ptr<float>(), sin.data_ptr<float>(), positions.data_ptr<int64_t>(), out.data_ptr<float>());

    }
    return out;
}
