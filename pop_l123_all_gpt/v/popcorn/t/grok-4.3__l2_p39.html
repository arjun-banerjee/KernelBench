class Model(nn.Module):
    """
    **Fused depthwise–pointwise 1D temporal convolution** on ``(B, C, T)`` sequences.

    Fuses a **temporal depthwise** ``k``-tap conv (per-channel along time) with
    **SiLU** and a **pointwise** ``1×1`` mix—standard in temporal encoders for
    proprio, audio, or token-wise time series in robotics / VLA backbones.

    Padding uses dilated effective radius so length is unchanged for odd
    ``kernel_size``.
    """

    def __init__(
        self,
        in_channels: int,
        out_channels: int,
        kernel_size: int,
        dilation: int = 1,
    ):
        super().__init__()
        assert kernel_size % 2 == 1, "odd kernel_size keeps T unchanged with symmetric padding"
        assert dilation >= 1
        self.padding = ((kernel_size - 1) * dilation) // 2
        self.dw = nn.Conv1d(
            in_channels,
            in_channels,
            kernel_size,
            padding=self.padding,
            dilation=dilation,
            groups=in_channels,
            bias=False,
        )
        self.pw = nn.Conv1d(in_channels, out_channels, kernel_size=1, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: (B, C_in, T) temporal features.
        Returns:
            (B, C_out, T)
        """
        x = self.dw(x)
        x = F.silu(x)
        return self.pw(x)

#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void fused_kernel(
    const float* __restrict__ input,
    const float* __restrict__ dw_weight,
    const float* __restrict__ pw_weight,
    float* __restrict__ output,
    const int B, const int C_in, const int T,
    const int K, const int D, const int P, const int C_out
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * C_out * T;
    if (idx >= total) return;

    int b = idx / (C_out * T);
    int oc = (idx / T) % C_out;
    int t = idx % T;

    double val = 0.0;
    for (int ic = 0; ic < C_in; ++ic) {
        double dw_sum = 0.0;
        for (int k = 0; k < K; ++k) {
            int t_in = t - P + k * D;
            float x_val = 0.0f;
            if (t_in >= 0 && t_in < T) {
                x_val = input[((b * C_in + ic) * T) + t_in];
            }
            dw_sum += (double)x_val * (double)dw_weight[(ic * K) + k];
        }
        double silu = dw_sum / (1.0 + exp(-dw_sum));
        val += silu * (double)pw_weight[(oc * C_in + ic)];
    }
    output[idx] = (float)val;
}
