"""
Virtual all-gather on the last dim: stack R tensors of shape (B, H) into (B, R*H), then Linear.
"""

import torch
import torch.nn as nn

from kernelbench.distributed_collectives import default_device

R = 4
B = 12
H = 32
K = 56


class Model(nn.Module):
    def __init__(self):
        super().__init__()
        self.proj = nn.Linear(R * H, K, bias=True)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        r, b, h = x.shape
        assert r == R and h == H
        parts = [x[i] for i in range(R)]
        gathered = torch.cat(parts, dim=-1)
        return self.proj(gathered)


def get_inputs():
    dev = default_device()
    g = torch.Generator(device=dev)
    g.manual_seed(9)
    return [torch.randn(R, B, H, device=dev, generator=g)]


def get_init_inputs():
    return []

import torch.nn as nn
from torch.utils.cpp_extension import load_inline

torch::Tensor fwd_cuda(torch::Tensor x, torch::Tensor w, torch::Tensor b);
torch::Tensor forward(torch::Tensor x, torch::Tensor w, torch::Tensor b) {
    return fwd_cuda(x, w, b);
}
'''

_cuda_src = r'''
#include <torch/extension.h>
#include <cuda_runtime.h>

__global__ void kernel(const float* __restrict__ x,
                       const float* __restrict__ w,
                       const float* __restrict__ bias,
                       float* __restrict__ y) {
    int b = blockIdx.x;
    int k = blockIdx.y * 32 + (threadIdx.x >> 5);
    int lane = threadIdx.x & 31;
    float s = 0.0f;
    if (k < 56) {
        #pragma unroll
        for (int r = 0; r < 4; ++r) {
            s = fmaf(x[(r * 12 + b) * 32 + lane], w[k * 128 + r * 32 + lane], s);
        }
        #pragma unroll
        for (int off = 16; off > 0; off >>= 1) {
            s += __shfl_down_sync(0xffffffff, s, off);
        }
        if (lane == 0) y[b * 56 + k] = s + bias[k];
    }
}

torch::Tensor fwd_cuda(torch::Tensor x, torch::Tensor w, torch::Tensor bias) {
    auto y = torch::empty({12, 56}, x.options());
    kernel<<<dim3(12, 2), 1024>>>(x.data_ptr<float>(), w.data_ptr<float>(), bias.data_ptr<float>(), y.data_ptr<float>());
    return y;
}
