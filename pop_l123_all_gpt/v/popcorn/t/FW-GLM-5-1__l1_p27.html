"""
Fixed permutation along the feature axis: y[b, j] = x[b, perm[j]].
"""

import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

from kernelbench.distributed_collectives import default_device

B = 10
N = 36

_g = torch.Generator()
_g.manual_seed(10)
PERM = torch.randperm(N, generator=_g)

# Define the CUDA kernel
cuda_source = """
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void permute_kernel(const float* __restrict__ x, float* __restrict__ y, 
                               const int* __restrict__ perm, int B, int N) {
    // Use 1D indexing - each thread handles one element
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    int total = B * N;
    
    if (idx < total) {
        int b = idx / N;
        int j = idx % N;
        int src_j = perm[j];
        y[idx] = x[b * N + src_j];
    }
}

torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm) {
    int B = x.size(0);
    int N = x.size(1);
    
    auto y = torch::empty_like(x);
    
    int total = B * N;
    int threads = 256;
    int blocks = (total + threads - 1) / threads;
    
    permute_kernel<<<blocks, threads>>>(
        x.data_ptr<float>(),
        y.data_ptr<float>(),
        perm.data_ptr<int>(),
        B, N
    );
    
    return y;
}
"""

cpp_source = """
torch::Tensor permute_cuda(torch::Tensor x, torch::Tensor perm);
"""

# Compile the extension
permute_ext = load_inline(
    name="permute_ext",
    cpp_sources=cpp_source,
    cuda_sources=cuda_source,
    functions=["permute_cuda"],
    verbose=True,
)


class ModelNew(nn.Module):
    def __init__(self):
        super().__init__()
        # Register the permutation as int32 buffer
        self.register_buffer('perm', PERM.to(torch.int32))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # perm is already int32, just move to correct device
        return permute_ext.permute_cuda(x, self.perm.to(device=x.device))


def get_inputs():
    dev = default_device()
    g = torch.Generator(device=dev)
    g.manual_seed(10)
    return [torch.randn(B, N, device=dev, generator=g)]

    return []
