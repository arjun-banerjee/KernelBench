import torch
import torch.nn as nn


class Model(nn.Module):
    """
    Compute all-pairs Euclidean distance matrix from a 3-D point cloud.
    A core primitive in molecular dynamics simulations, docking, and
    distance-geometry-based structure prediction.
    """

    def __init__(self):
        super().__init__()

    def forward(self, coords: torch.Tensor) -> torch.Tensor:
        """
        Args:
            coords: (B, N, 3) – 3-D coordinates of N atoms / residues
        Returns:
            dist: (B, N, N) – pairwise Euclidean distance matrix
        """
        diff = coords.unsqueeze(2) - coords.unsqueeze(1)  # (B, N, N, 3)
        dist = torch.sqrt((diff ** 2).sum(dim=-1) + 1e-8)
        return dist

#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>

__global__ void pairwise_sym_kernel(const float* __restrict__ coords, float* __restrict__ out, int B, int N) {
    int tx = threadIdx.x;
    int ty = threadIdx.y;
    int tile_j = blockIdx.x;
    int tile_i = blockIdx.y;
    if (tile_j < tile_i) return;

    int b = blockIdx.z;
    int i = tile_i * 16 + ty;
    int j = tile_j * 16 + tx;

    if (i < N && j < N) {
        const float* base = coords + ((long long)b * N) * 3;
        float xi = base[i * 3 + 0];
        float yi = base[i * 3 + 1];
        float zi = base[i * 3 + 2];
        float xj = base[j * 3 + 0];
        float yj = base[j * 3 + 1];
        float zj = base[j * 3 + 2];
        float dx = xi - xj;
        float dy = yi - yj;
        float dz = zi - zj;
        float d = sqrtf(dx * dx + dy * dy + dz * dz + 1.0e-8f);
        out[((long long)b * N + i) * N + j] = d;
        if (tile_j != tile_i) {
            out[((long long)b * N + j) * N + i] = d;
        }
    }
}

torch::Tensor pairwise_forward(torch::Tensor coords) {
    auto c = coords.contiguous();
    int B = (int)c.size(0);
    int N = (int)c.size(1);
    auto out = torch::empty({B, N, N}, c.options());
    dim3 block(16, 16);
    dim3 grid((N + 15) / 16, (N + 15) / 16, B);
    pairwise_sym_kernel<<<grid, block>>>(c.data_ptr<float>(), out.data_ptr<float>(), B, N);
    return out;
}
