#include <cuda_runtime.h>

__global__ void lse_kernel(const float* __restrict__ x, float* __restrict__ out) {
    int idx = threadIdx.x;
    if (idx < 320) {
        float s = __expf(x[idx]) + __expf(x[idx + 320]) + __expf(x[idx + 640]) + __expf(x[idx + 960]);
        out[idx] = __logf(s);
    }
}

}
