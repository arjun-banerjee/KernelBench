#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

#define TILE_M 128
#define TILE_N 128
#define TILE_K 16
#define THREADS_PER_BLOCK 256

__global__ void gemm_fp16_kernel(const half* __restrict__ A, const half* __restrict__ B, float* __restrict__ C,
                                 int M, int N, int K, float alpha, float beta) {
    __shared__ half As[TILE_M][TILE_K + 1];
    __shared__ half Bs[TILE_K][TILE_N + 1];

    int bx = blockIdx.x;
    int by = blockIdx.y;
    int tx = threadIdx.x % 16;
    int ty = threadIdx.x / 16;

    int row = by * TILE_M + ty * 8;
    int col = bx * TILE_N + tx * 8;

    float acc[8][8] = {0.0f};

    for (int t = 0; t < (K + TILE_K - 1) / TILE_K; ++t) {
        for (int i = 0; i < 8; ++i) {
            for (int j = 0; j < TILE_K; j += 16) {
                int global_row = row + i;
                int global_col = t * TILE_K + (threadIdx.x % 16) + j;
                if (global_row < M && global_col < K) {
                    As[ty * 8 + i][threadIdx.x % 16 + j] = __float2half(A[global_row * K + global_col]);
                } else {
                    As[ty * 8 + i][threadIdx.x % 16 + j] = __float2half(0.0f);
                }
            }
        }

        for (int i = 0; i < TILE_K; ++i) {
            for (int j = 0; j < 8; ++j) {
                int global_row = t * TILE_K + i;
                int global_col = col + j * 16 + (threadIdx.x / 16);
                if (global_row < K && global_col < N) {
                    Bs[i][tx * 8 + j] = __float2half(B[global_row * N + global_col]);
                } else {
                    Bs[i][tx * 8 + j] = __float2half(0.0f);
                }
            }
        }

        __syncthreads();

        for (int k = 0; k < TILE_K; ++k) {
            half a_vals[8];
            half b_vals[8];
            for (int i = 0; i < 8; ++i) a_vals[i] = As[ty * 8 + i][k];
            for (int j = 0; j < 8; ++j) b_vals[j] = Bs[k][tx * 8 + j];

                }
                C[global_row * N + global_col] = val;
            }
        }
    }
}

torch::Tensor gemm_cuda(torch::Tensor A, int K, int N, float alpha, float beta) {
    int M = A.size(0);
    auto options = torch::TensorOptions().dtype(torch::kFloat32).device(A.device());
    auto C = torch::zeros({M, N}, options);
    
    auto A_half = A.to(torch::kFloat16).contiguous();
    auto B = torch::randn({K, N}, torch::TensorOptions().dtype(torch::kFloat16).device(A.device())).contiguous();
    
    dim3 blocks((N + TILE_N - 1) / TILE_N, (M + TILE_M - 1) / TILE_M);
    dim3 threads(THREADS_PER_BLOCK);

        reinterpret_cast<half*>(A_half.data_ptr<at::Half>()),
        reinterpret_cast<half*>(B.data_ptr<at::Half>()),
        C.data_ptr<float>(),
        M, N, K, alpha, beta
    );
    
    cudaDeviceSynchronize();
    return C;
}
