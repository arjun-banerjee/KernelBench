// C = alpha * A @ B + beta, with A [batch,M,K], B [batch,K,N], C [batch,M,N]
// Row-major tensors.  Tuned for the fixed problem family in the prompt (K=129,N=219,B=17)

template<int BM, int BN, int BK>
__global__ __launch_bounds__(256, 2)
void bmm_add_kernel_f32(const float* __restrict__ A,
                        const float* __restrict__ B,
                        float* __restrict__ C,
                        int batch, int M, int K, int N,
                        float alpha, float beta) {
    __shared__ float As[BM][BK + 1];
    __shared__ float Bs[BK][BN + 1];

    const int tx = threadIdx.x;  // 0..15
    const int ty = threadIdx.y;  // 0..15
    const int b  = blockIdx.z;
    const int row = blockIdx.y * BM + ty;
    const int col0 = blockIdx.x * BN + tx;
    const int col1 = col0 + 16;

    float acc1 = 0.0f;

    const float* __restrict__ Bb = B + ((long long)b * K * N);
