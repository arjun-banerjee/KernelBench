// 128-bit aligned vector for coalesced global memory access.
template <typename T, int VEC_SIZE>
struct alignas(sizeof(T) * VEC_SIZE) Vec { T data[VEC_SIZE]; };

// Vectorized RMSNorm kernel – one block per row, warp-shuffle reduction.
template <typename T, int VEC_SIZE, int BLOCK_SIZE>
__global__ __launch_bounds__(BLOCK_SIZE, 4)
void rms_norm_kernel(
    T* __restrict__ out,
    const T* __restrict__ input,
    int64_t input_stride,
    const T* __restrict__ weight,
    float epsilon,
    int num_tokens,
    int hidden_size) {

    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int nvec = hidden_size / VEC_SIZE;

    const T* in_row = input + static_cast<int64_t>(row) * input_stride;
    T* out_row = out + static_cast<int64_t>(row) * hidden_size;

    float variance = 0.0f;
    using VecT = Vec<T, VEC_SIZE>;

    // Load and accumulate variance (vectorized).
    for (int i = tid; i < nvec; i += BLOCK_SIZE) {
        VecT v = reinterpret_cast<const VecT*>(in_row)[i];
        #pragma unroll
        for (int j = 0; j < VEC_SIZE; ++j) {
            float x = float(v.data[j]);
            variance += x * x;
        }
    }

    // Intra-warp reduction.
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        variance += __shfl_down_sync(0xffffffff, variance, offset);
    }

    __shared__ float warp_sum[32];   // 32 warps max per block
    const int lane = tid & 31;
    const int warp_id = tid >> 5;
    if (lane == 0) {
        warp_sum[warp_id] = variance;
    }
    __syncthreads();

    // Inter-warp reduction using warp 0.
    if (warp_id == 0) {
        float val = (lane < (BLOCK_SIZE >> 5)) ? warp_sum[lane] : 0.0f;
        #pragma unroll
        for (int offset = 16; offset > 0; offset >>= 1) {
            val += __shfl_down_sync(0xffffffff, val, offset);
        }
        if (lane == 0) {
            warp_sum[0] = rsqrtf(val / static_cast<float>(hidden_size) + epsilon);
        }

    float rms = warp_sum[0];

        }
        reinterpret_cast<VecT*>(out_row)[i] = v_out;
    }
}
