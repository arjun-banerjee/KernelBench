import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <mma.h>

using namespace nvcuda;

__global__ __launch_bounds__(256, 2)
void h100_tf32_gemm_kernel(const float* __restrict__ A,
                           const float* __restrict__ B,
                           float* __restrict__ C,
                           int M, int N, int K,
                           float alpha) {
  // 8 warps per CTA.  CTA tile = 64 x 32, each warp computes one 16 x 16 tile.
  int warp_id = threadIdx.x >> 5;
  int warp_m = warp_id & 3;
  int warp_n = warp_id >> 2;

  int row = blockIdx.y * 64 + warp_m * 16;
  int col = blockIdx.x * 32 + warp_n * 16;

  if (row >= M || col >= N) return;

  wmma::fragment<wmma::accumulator, 16, 16, 8, float> acc;
  wmma::fill_fragment(acc, 0.0f);

  #pragma unroll 1
  for (int k0 = 0; k0 < K; k0 += 8) {
    wmma::fragment<wmma::matrix_a, 16, 16, 8, wmma::precision::tf32, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 8, wmma::precision::tf32, wmma::row_major> b_frag;

    wmma::load_matrix_sync(a_frag, A + row * K + k0, K);
    wmma::load_matrix_sync(b_frag, B + k0 * N + col, N);
    wmma::mma_sync(acc, a_frag, b_frag, acc);
  }

  if (alpha != 1.0f) {
    #pragma unroll
    for (int i = 0; i < acc.num_elements; ++i) acc.x[i] *= alpha;
  }

  wmma::store_matrix_sync(C + row * N + col, acc, N, wmma::mem_row_major);
}

// A small scalar FP32 cleanup kernel for non-multiple edge cases.  It is not used for the
// benchmark dimensions (5120 x 4096 x 4096) but keeps the operator fully general.
__global__ void fp32_edge_kernel(const float* __restrict__ A,
                                 const float* __restrict__ B,
                                 float* __restrict__ C,
                                 int M, int N, int K, float alpha) {
  int col = blockIdx.x * blockDim.x + threadIdx.x;
  int row = blockIdx.y * blockDim.y + threadIdx.y;
  if (row >= M || col >= N) return;
  if ((row < (M & ~15)) && (col < (N & ~15)) && ((K & 7) == 0)) return;
  float sum = 0.0f;
  for (int k = 0; k < K; ++k) sum += A[row * K + k] * B[k * N + col];
  C[row * N + col] = sum * alpha;
}

torch::Tensor gemm_forward(torch::Tensor A, torch::Tensor B, double alpha_d) {
  TORCH_CHECK(A.is_cuda() && B.is_cuda(), "A and B must be CUDA tensors");
  TORCH_CHECK(A.scalar_type() == torch::kFloat32 && B.scalar_type() == torch::kFloat32, "A and B must be float32");
  TORCH_CHECK(A.dim() == 2 && B.dim() == 2, "A and B must be matrices");
  TORCH_CHECK(A.size(1) == B.size(0), "incompatible GEMM dimensions");

  if (!A.is_contiguous()) A = A.contiguous();
  if (!B.is_contiguous()) B = B.contiguous();

  int M = (int)A.size(0);
  int K = (int)A.size(1);
  int N = (int)B.size(1);
  float alpha = (float)alpha_d;

  auto C = torch::empty({M, N}, A.options());
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();

  if ((K & 7) == 0) {
    dim3 block(256, 1, 1);
    dim3 grid((N + 31) / 32, (M + 63) / 64, 1);
    h100_tf32_gemm_kernel<<<grid, block, 0, stream>>>(A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(), M, N, K, alpha);
  }
  if ((M & 15) || (N & 15) || (K & 7)) {
    dim3 b2(16, 16, 1);
    dim3 g2((N + 15) / 16, (M + 15) / 16, 1);
    fp32_edge_kernel<<<g2, b2, 0, stream>>>(A.data_ptr<float>(), B.data_ptr<float>(), C.data_ptr<float>(), M, N, K, alpha);
  }
  return C;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("gemm_forward", &gemm_forward, "Hopper TF32 GEMM forward");
}
