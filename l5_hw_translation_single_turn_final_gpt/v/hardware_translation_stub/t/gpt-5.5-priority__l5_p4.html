import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_CPP_SRC = r"""
#include <torch/extension.h>

torch::Tensor silu_mul_forward(torch::Tensor input);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("silu_mul_forward", &silu_mul_forward, "Hopper optimized SiLU-and-mul forward (CUDA)");
}
"""

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

// One CTA handles one token.  For the benchmark shape (D=2048) each CTA has
// 256 threads and each thread computes eight output half values using four
// half2 operations.  Loads/stores are 128-bit and fully coalesced.
__global__ __launch_bounds__(256, 4)
void silu_mul_vec8_kernel(half* __restrict__ out,
                          const half* __restrict__ input,
                          const int d) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int vecs_per_row = d >> 3;  // 8 half values per int4

    const half* __restrict__ xrow = input + (static_cast<long long>(row) * 2LL * d);
    const half* __restrict__ yrow = xrow + d;
    half* __restrict__ orow = out + (static_cast<long long>(row) * d);

    const int4* __restrict__ x4 = reinterpret_cast<const int4*>(xrow);
    const int4* __restrict__ y4 = reinterpret_cast<const int4*>(yrow);
    int4* __restrict__ o4 = reinterpret_cast<int4*>(orow);

    const half2 one = __float2half2_rn(1.0f);

    for (int v = tid; v < vecs_per_row; v += blockDim.x) {
        int4 xv4 = x4[v];
        int4 yv4 = y4[v];
        int4 ov4;

        half2* __restrict__ xv = reinterpret_cast<half2*>(&xv4);
        half2* __restrict__ yv = reinterpret_cast<half2*>(&yv4);
        half2* __restrict__ ov = reinterpret_cast<half2*>(&ov4);

        #pragma unroll
        for (int j = 0; j < 4; ++j) {
            // silu(x) * y = x / (1 + exp(-x)) * y.  Hopper has fast native
            // half2 transcendental support; fp16 tolerance is 1e-2, so this
            // avoids scalar float unpacking while remaining numerically stable.
            const half2 x = xv[j];
            const half2 y = yv[j];
            const half2 sig_den = __hadd2(one, h2exp(__hneg2(x)));
            const half2 silu = __h2div(x, sig_den);
            ov[j] = __hmul2(silu, y);
        }
        o4[v] = ov4;
    }
}

// Scalar/half2 cleanup for non-multiple-of-8 D.  Not used by the supplied
// contract (D=2048), but keeps the operator well-defined for compatible inputs.
__global__ __launch_bounds__(256, 4)
void silu_mul_h2_kernel(half* __restrict__ out,
                        const half* __restrict__ input,
                        const int d) {
    const int row = blockIdx.x;
    const int tid = threadIdx.x;
    const int h2n = d >> 1;

    const half* __restrict__ xrow_h = input + (static_cast<long long>(row) * 2LL * d);
    const half* __restrict__ yrow_h = xrow_h + d;
    half* __restrict__ orow_h = out + (static_cast<long long>(row) * d);

    const half2* __restrict__ xrow = reinterpret_cast<const half2*>(xrow_h);
    const half2* __restrict__ yrow = reinterpret_cast<const half2*>(yrow_h);
    half2* __restrict__ orow = reinterpret_cast<half2*>(orow_h);
    const half2 one = __float2half2_rn(1.0f);

    for (int i = tid; i < h2n; i += blockDim.x) {
        const half2 x = xrow[i];
        const half2 y = yrow[i];
        const half2 sig_den = __hadd2(one, h2exp(__hneg2(x)));
        orow[i] = __hmul2(__h2div(x, sig_den), y);
    }

    if ((d & 1) && tid == 0) {
        const int i = d - 1;
        const float x = __half2float(xrow_h[i]);
        const float y = __half2float(yrow_h[i]);
        orow_h[i] = __float2half_rn((x / (1.0f + __expf(-x))) * y);
    }
}

torch::Tensor silu_mul_forward(torch::Tensor input) {
    TORCH_CHECK(input.is_cuda(), "input must be a CUDA tensor");
    TORCH_CHECK(input.scalar_type() == at::kHalf, "input must be float16");
    TORCH_CHECK(input.dim() >= 1, "input must have at least one dimension");
    TORCH_CHECK(input.size(-1) % 2 == 0, "last dimension must be even");

    const int64_t last = input.size(-1);
    const int d = static_cast<int>(last / 2);
    const int64_t num_tokens = input.numel() / last;

    auto out_sizes = input.sizes().vec();
    out_sizes.back() = d;
    torch::Tensor out = torch::empty(out_sizes, input.options());
    if (num_tokens == 0) return out;

    const half* in_ptr = reinterpret_cast<const half*>(input.data_ptr<at::Half>());
    half* out_ptr = reinterpret_cast<half*>(out.data_ptr<at::Half>());
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();

    dim3 grid(static_cast<unsigned int>(num_tokens));
    dim3 block(256);

    if ((d & 7) == 0) {
        silu_mul_vec8_kernel<<<grid, block, 0, stream>>>(out_ptr, in_ptr, d);
    } else {
        silu_mul_h2_kernel<<<grid, block, 0, stream>>>(out_ptr, in_ptr, d);
    }
    return out;
}
"""
