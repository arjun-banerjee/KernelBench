// Simple batched GEMM kernel optimized for small matrices
__global__ void simple_batched_gemm_kernel(
    const int num_groups,
    const int M, const int N, const int K,
    const float alpha, const float beta,
    const float* __restrict__ A,
    const float* __restrict__ B,
    float* __restrict__ C) {
    
    // Each block handles one group
    const int batch_idx = blockIdx.x;
    if (batch_idx >= num_groups) return;
    
    const int row = blockIdx.y * blockDim.y + threadIdx.y;
    const int col = blockIdx.z * blockDim.z + threadIdx.z;
    
    if (row >= M || col >= N) return;
    
    const float* A_batch = A + batch_idx * M * K;
    const float* B_batch = B + batch_idx * K * N;
    float* C_batch = C + batch_idx * M * N;
    
    float sum = 0.0f;
    
    for (int k = 0; k < K; k++) {
        sum += A_batch[row * K + k] * B_batch[k * N + col];
    }
    
    C_batch[row * N + col] = alpha * sum + beta * C_batch[row * N + col];
}

// Optimized FP16 batched GEMM using Tensor Cores

            }
            
            float alpha_f = __half2float(alpha);
            float beta_f = __half2float(beta);
            float c_val = __half2float(C_batch[out_row * N + out_col]);
            
            C_batch[out_row * N + out_col] = __float2half(alpha_f * sum + beta_f * c_val);
        }
    }
}

// High-performance kernel using shared memory tiling
template<int BLOCK_SIZE, int TILE_K>
__global__ void tiled_gemm_kernel(
    const int num_groups,
    const int M, const int N, const int K,
    const float alpha, const float beta,
    const float* __restrict__ A,
    const float* __restrict__ B,
    float* __restrict__ C) {
    
    const int batch_idx = blockIdx.x;
    if (batch_idx >= num_groups) return;
    
    const int block_row = blockIdx.y;
    const int block_col = blockIdx.z;
    
    const int row = block_row * BLOCK_SIZE + threadIdx.y;
    const int col = block_col * BLOCK_SIZE + threadIdx.x;
    
    const int ty = threadIdx.y;
    const int tx = threadIdx.x;
    
    __shared__ float As[BLOCK_SIZE][TILE_K];
    __shared__ float Bs[TILE_K][BLOCK_SIZE];
    
    const float* A_batch = A + batch_idx * M * K;
    const float* B_batch = B + batch_idx * K * N;
    float* C_batch = C + batch_idx * M * N;
    
    float sum = 0.0f;
    
    for (int k = 0; k < K; k += TILE_K) {
        // Load tiles into shared memory
        if (row < M && (k + tx) < K)
            As[ty][tx] = A_batch[row * K + k + tx];
        else
            As[ty][tx] = 0.0f;
            
        if ((k + ty) < K && col < N)
            Bs[ty][tx] = B_batch[(k + ty) * N + col];
        else
            Bs[ty][tx] = 0.0f;
        
        __syncthreads();
        
        // Compute partial dot product
        for (int kk = 0; kk < TILE_K; kk++) {
            sum += As[ty][kk] * Bs[kk][tx];
        }
        
        __syncthreads();
    }
    
    if (row < M && col < N) {
        C_batch[row * N + col] = alpha * sum + beta * C_batch[row * N + col];
    }
}
