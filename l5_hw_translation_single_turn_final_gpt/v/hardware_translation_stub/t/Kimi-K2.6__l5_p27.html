    __shared__ float C_smem[BLOCK_M * BLOCK_N];

    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::col_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> acc_frag;
    wmma::fill_fragment(acc_frag, 0.0f);

    int batch_stride_B = N * K;
    int batch_stride_C = M * N;

    const float* A_b = A + b * batch_stride_A;
    const float* B_b = B + b * batch_stride_B;

            int k = idx % BLOCK_K;

            int gk = k0 + k;
            half val;
            if (gm < M && gk < K) {
                val = __float2half(A_b[gm * K + gk]);
            } else {
                val = __float2half(0.0f);
            }
            A_smem[m * BLOCK_K + k] = val;
        }

        for (int idx = tid; idx < BLOCK_N * BLOCK_K; idx += blockDim.x) {
            int n = idx / BLOCK_K;
            int k = idx % BLOCK_K;
            int gn = n0 + n;
            int gk = k0 + k;
            half val;
            if (gn < N && gk < K) {
                val = __float2half(B_b[gn * K + gk]);
            } else {
                val = __float2half(0.0f);
            }
            B_smem[n * BLOCK_K + k] = val;
        }

        __syncthreads();

        #pragma unroll
        for (int k_frag = 0; k_frag < BLOCK_K; k_frag += 16) {
            wmma::load_matrix_sync(a_frag, A_smem + m_offset * BLOCK_K + k_frag, BLOCK_K);
            wmma::load_matrix_sync(b_frag, B_smem + n_offset * BLOCK_K + k_frag, BLOCK_K);
            wmma::mma_sync(acc_frag, a_frag, b_frag, acc_frag);
        }

        __syncthreads();
    }

    wmma::store_matrix_sync(C_smem + m_offset * BLOCK_N + n_offset, acc_frag, BLOCK_N, wmma::mem_row_major);

    __syncthreads();

    for (int idx = tid; idx < BLOCK_M * BLOCK_N; idx += blockDim.x) {
        int m = idx / BLOCK_N;
        int n = idx % BLOCK_N;
        int gm = m0 + m;
        int gn = n0 + n;
        if (gm < M && gn < N) {
            C_b[gm * N + gn] = alpha * C_smem[m * BLOCK_N + n] + beta;
        }
    }
}
