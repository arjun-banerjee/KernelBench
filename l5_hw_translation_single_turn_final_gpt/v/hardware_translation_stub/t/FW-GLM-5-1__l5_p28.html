// Optimized GEMM kernel for Hopper
template<int NUM_WARPS_M, int NUM_WARPS_N, int NUM_STAGES>
__global__ void int8_gemm_hopper_kernel(
    const int8_t* __restrict__ A,
    const int8_t* __restrict__ B,
    int32_t* __restrict__ C,
    const int M, const int N, const int K,
    const int32_t alpha, const int32_t beta) {
    
    // Thread block and warp configuration
    constexpr int WARPS_PER_BLOCK = NUM_WARPS_M * NUM_WARPS_N;
    constexpr int THREADS_PER_WARP = 32;
    constexpr int THREADS_PER_BLOCK = WARPS_PER_BLOCK * THREADS_PER_WARP;
    
    // Shared memory tile sizes
    constexpr int TILE_K = THREADBLOCK_K / NUM_STAGES;
    constexpr int SHMEM_A_SIZE = THREADBLOCK_M * TILE_K;
    constexpr int SHMEM_B_SIZE = TILE_K * THREADBLOCK_N;
    
    // Shared memory with padding for bank conflict avoidance
    extern __shared__ uint8_t shared_memory[];
    int8_t* shmem_A = reinterpret_cast<int8_t*>(shared_memory);
    int8_t* shmem_B = reinterpret_cast<int8_t*>(shared_memory + SHMEM_A_SIZE * sizeof(int8_t) * 2);
    
    // Thread and warp indices
    const int thread_idx = threadIdx.x;
    const int warp_idx = thread_idx / THREADS_PER_WARP;
    const int lane_idx = thread_idx % THREADS_PER_WARP;
    
    const int warp_m = warp_idx / NUM_WARPS_N;
    const int warp_n = warp_idx % NUM_WARPS_N;
    
    // Output tile coordinates
    const int block_m = blockIdx.y;
    const int block_n = blockIdx.x;
    
    const int global_m = block_m * THREADBLOCK_M + warp_m * WARP_M;
    const int global_n = block_n * THREADBLOCK_N + warp_n * WARP_N;
    
    // Accumulator register file - 16x16 int32 values
    int32_t acc[MMA_M * MMA_N] = {0};
    
    // Pipeline stage tracking
    int stage = 0;
    
    // Main loop over K dimension
    const int num_k_tiles = (K + TILE_K - 1) / TILE_K;
    
    // Prefetch first tiles
    for (int s = 0; s < min(NUM_STAGES, num_k_tiles); s++) {
        const int k_offset = s * TILE_K;
        
        // Cooperative loading of A tile
        for (int i = thread_idx; i < THREADBLOCK_M * TILE_K; i += THREADS_PER_BLOCK) {
            const int row = i / TILE_K;
            const int col = i % TILE_K;
            const int a_row = block_m * THREADBLOCK_M + row;
            const int a_col = k_offset + col;
            if (a_row < M && a_col < K) {
                shmem_A[s * THREADBLOCK_M * TILE_K + row * (TILE_K + 8) + col] = A[a_row * K + a_col];
            }
        }
        
        // Cooperative loading of B tile
        for (int i = thread_idx; i < TILE_K * THREADBLOCK_N; i += THREADS_PER_BLOCK) {
            const int row = i / THREADBLOCK_N;
            const int col = i % THREADBLOCK_N;
            const int b_row = k_offset + row;
            const int b_col = block_n * THREADBLOCK_N + col;
            if (b_row < K && b_col < N) {
                shmem_B[s * TILE_K * (THREADBLOCK_N + 8) + row * (THREADBLOCK_N + 8) + col] = B[b_row * N + b_col];
            }
        }
    }
    __syncthreads();
    
    // Main computation loop with pipelining
    for (int k_tile = 0; k_tile < num_k_tiles; k_tile++) {
        
        // Load next stage while computing current
        const int next_stage = (stage + NUM_STAGES - 1) % NUM_STAGES;
        const int next_k_tile = k_tile + NUM_STAGES;
        
        if (next_k_tile < num_k_tiles) {
            const int k_offset = next_k_tile * TILE_K;
            
            // Prefetch next A tile
            for (int i = thread_idx; i < THREADBLOCK_M * TILE_K; i += THREADS_PER_BLOCK) {
                const int row = i / TILE_K;
                const int col = i % TILE_K;
                const int a_row = block_m * THREADBLOCK_M + row;
                const int a_col = k_offset + col;
                if (a_row < M && a_col < K) {
                    shmem_A[next_stage * THREADBLOCK_M * TILE_K + row * (TILE_K + 8) + col] = 
                        A[a_row * K + a_col];
                }
            }
            
            // Prefetch next B tile
            for (int i = thread_idx; i < TILE_K * THREADBLOCK_N; i += THREADS_PER_BLOCK) {
                const int row = i / THREADBLOCK_N;
                const int col = i % THREADBLOCK_N;
                const int b_row = k_offset + row;
                const int b_col = block_n * THREADBLOCK_N + col;
                if (b_row < K && b_col < N) {
                    shmem_B[next_stage * TILE_K * (THREADBLOCK_N + 8) + row * (THREADBLOCK_N + 8) + col] = 
                        B[b_row * N + b_col];
                }
            }
        }
        
        // Compute on current tile
        const int8_t* tile_A = &shmem_A[stage * THREADBLOCK_M * TILE_K + 
                                        warp_m * WARP_M * (TILE_K + 8)];
        const int8_t* tile_B = &shmem_B[stage * TILE_K * (THREADBLOCK_N + 8) + 
                                        warp_n * WARP_N];
        
        // Perform MMA operations
        #pragma unroll
        for (int k_step = 0; k_step < TILE_K; k_step += MMA_K) {
            #pragma unroll
            for (int m_idx = 0; m_idx < WARP_M / MMA_M; m_idx++) {
                #pragma unroll
                for (int n_idx = 0; n_idx < WARP_N / MMA_N; n_idx++) {
                    // Compute MMA tile coordinates
                    const int8_t* a_frag = tile_A + m_idx * MMA_M * (TILE_K + 8) + k_step;
                    const int8_t* b_frag = tile_B + k_step * (THREADBLOCK_N + 8) + n_idx * MMA_N;
                    
                    // Accumulator offset
                    const int acc_offset = m_idx * MMA_M * (WARP_N / MMA_N) * MMA_N + n_idx * MMA_N * MMA_M;
                    
                    // Fallback computation (works on all architectures)
                    for (int m = lane_idx / 4; m < MMA_M; m += 8) {
                        for (int n = lane_idx % 4; n < MMA_N; n += 4) {
                            int32_t sum = 0;
                            for (int k = 0; k < MMA_K; k++) {
                                sum += static_cast<int32_t>(a_frag[m * (TILE_K + 8) + k]) *
                                       static_cast<int32_t>(b_frag[k * (THREADBLOCK_N + 8) + n]);
                            }
                            acc[m * MMA_N + n] += sum;
                        }
                    }
                }
            }
        }
        
        stage = (stage + 1) % NUM_STAGES;
        __syncthreads();
    }
    
    // Store results to global memory
    #pragma unroll
    for (int m = lane_idx / 4; m < WARP_M; m += 8) {
        #pragma unroll
        for (int n = lane_idx % 4; n < WARP_N; n += 4) {
            const int out_m = global_m + m;
            const int out_n = global_n + n;
            if (out_m < M && out_n < N) {
                // Apply alpha and beta
                int32_t result = acc[(m % WARP_M) / (WARP_M / 4) * 64 + 
                                     (m % (WARP_M / 4)) * MMA_N + n % MMA_N];
                result = alpha * result + beta * C[out_m * N + out_n];
                C[out_m * N + out_n] = result;
            }
        }
    }
}
