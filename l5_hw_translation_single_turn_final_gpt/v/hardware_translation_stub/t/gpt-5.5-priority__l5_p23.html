import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_src = r'''
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <mma.h>

using namespace nvcuda;

__global__ void zero_kernel(float* __restrict__ out, int total) {
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int stride = blockDim.x * gridDim.x;
    for (int i = tid; i < total; i += stride) out[i] = 0.0f;
}

__global__ void gather_b_kernel(const half* __restrict__ B,
                                const int* __restrict__ indices,
                                half* __restrict__ Bsel,
                                int K, int N, int J) {
    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int total = K * J;
    for (int p = tid; p < total; p += blockDim.x * gridDim.x) {
        int j = p / K;
        int k = p - j * K;
        int col = indices[j];
        Bsel[p] = B[k * N + col];
    }
}

// One warp computes one 16x16 tile of A(MxK) * Bsel^T(JxK), then scatters columns.
__global__ void wmma_scatter_kernel(const half* __restrict__ A,
                                    const half* __restrict__ Bsel,
                                    const int* __restrict__ indices,
                                    float* __restrict__ out,
                                    int M, int N, int K, int J,
                                    float alpha) {
    int tile_m = blockIdx.y;
    int tile_j = blockIdx.x;
    int row0 = tile_m * 16;
    int j0 = tile_j * 16;

    wmma::fragment<wmma::matrix_a, 16, 16, 16, half, wmma::row_major> a_frag;
    wmma::fragment<wmma::matrix_b, 16, 16, 16, half, wmma::col_major> b_frag;
    wmma::fragment<wmma::accumulator, 16, 16, 16, float> acc_frag;
    wmma::fill_fragment(acc_frag, 0.0f);

    #pragma unroll
    for (int k0 = 0; k0 < 1024; k0 += 16) {
        const half* Aptr = A + row0 * K + k0;
        const half* Bptr = Bsel + j0 * K + k0;
        wmma::load_matrix_sync(a_frag, Aptr, K);
        wmma::load_matrix_sync(b_frag, Bptr, K);
        wmma::mma_sync(acc_frag, a_frag, b_frag, acc_frag);
    }

    __shared__ float smem[16 * 16];
    wmma::store_matrix_sync(smem, acc_frag, 16, wmma::mem_row_major);
    __syncthreads();

    int lane = threadIdx.x;
    for (int t = lane; t < 256; t += 32) {
        int r = t >> 4;
        int c = t & 15;
        int row = row0 + r;
        int jj = j0 + c;
        if (row < M && jj < J) {
            int col = indices[jj];
            out[row * N + col] = smem[t] * alpha;
        }
    }
}

torch::Tensor fused_forward(torch::Tensor A, torch::Tensor B, torch::Tensor indices, int K, int N, int J, double alpha) {
    const int M = (int)A.size(0);
    auto out = torch::empty({M, N}, torch::TensorOptions().device(A.device()).dtype(torch::kFloat32));
    auto Bsel = torch::empty({J, K}, torch::TensorOptions().device(A.device()).dtype(torch::kFloat16));

    int total_out = M * N;
    zero_kernel<<<(total_out + 255) / 256, 256>>>(out.data_ptr<float>(), total_out);

    int total_b = K * J;
    gather_b_kernel<<<(total_b + 255) / 256, 256>>>(
        reinterpret_cast<const half*>(B.data_ptr<at::Half>()),
        indices.data_ptr<int>(),

        reinterpret_cast<const half*>(Bsel.data_ptr<at::Half>()),
        indices.data_ptr<int>(),
        out.data_ptr<float>(),
        M, N, K, J, (float)alpha);
    return out;
}
