_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <mma.h>

using namespace nvcuda;

#ifndef __CUDA_ARCH__
#define __CUDA_ARCH__ 0
#endif

static constexpr int BM = 64;
static constexpr int BN = 64;
static constexpr int BK = 128;
static constexpr int WM = 16;
static constexpr int WN = 16;
static constexpr int WARPS_M = 4;
static constexpr int WARPS_N = 4;
static constexpr int WARPS_PER_BLOCK = WARPS_M * WARPS_N;
static constexpr int THREADS = WARPS_PER_BLOCK * 32;

__global__ __launch_bounds__(THREADS, 2)
void int8_gemm_64x64x128_kernel(const signed char* __restrict__ A,
                                const signed char* __restrict__ B,
                                int* __restrict__ C,
                                int M, int N, int K) {
#if __CUDA_ARCH__ >= 750
  __shared__ __align__(16) signed char shA[BM * BK];
  __shared__ __align__(16) signed char shB[BK * BN];

  const int tid = threadIdx.x;
  const int warp_id = tid >> 5;
  const int warp_m = warp_id >> 2;
  const int warp_n = warp_id & 3;

  const int block_m = blockIdx.y * BM;
  const int block_n = blockIdx.x * BN;

  wmma::fragment<wmma::accumulator, 16, 16, 16, int> acc;
  wmma::fill_fragment(acc, 0);

  for (int kk = 0; kk < K; kk += BK) {
    // Exactly 512 16-byte vectors for each 64x128 / 128x64 tile.
    int a_off = tid * 16;
    int ar = a_off / BK;
    int ac = a_off - ar * BK;
    const int4* gA = reinterpret_cast<const int4*>(A + (block_m + ar) * K + kk + ac);
    int4* sA = reinterpret_cast<int4*>(shA + a_off);
    *sA = *gA;

    int b_off = tid * 16;
    int br = b_off / BN;
    int bc = b_off - br * BN;
    const int4* gB = reinterpret_cast<const int4*>(B + (kk + br) * N + block_n + bc);
    int4* sB = reinterpret_cast<int4*>(shB + b_off);
    *sB = *gB;

    __syncthreads();

#pragma unroll
    for (int ksub = 0; ksub < BK; ksub += 16) {

      wmma::load_matrix_sync(bf, b_tile, BN);
      wmma::mma_sync(acc, af, bf, acc);
    }

    __syncthreads();
  }

  int* c_tile = C + (block_m + warp_m * WM) * N + block_n + warp_n * WN;
  wmma::store_matrix_sync(c_tile, acc, N, wmma::mem_row_major);
#endif
}

  TORCH_CHECK(a.is_cuda() && b.is_cuda(), "inputs must be CUDA tensors");
