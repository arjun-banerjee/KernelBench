#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <cuda_bf16.h>

template <typename T>
__global__ void flash_attn_kernel(
    const T* __restrict__ q,
    const T* __restrict__ k,
    const T* __restrict__ v,
    T* __restrict__ out,
    const int B, const int Hq, const int Hkv, const int S, const int D) {
    // Simplified efficient kernel for H100 - uses vectorized loads, shared mem tiling
    int batch = blockIdx.x;
    int head = blockIdx.y;
    int kv_head = head / (Hq / Hkv);
    int tid = threadIdx.x;
    int warp_id = tid / 32;
    int lane = tid % 32;

    // For simplicity, each block handles one (batch, head) pair, process seq in tiles
    extern __shared__ char smem[];
    float* s_q = (float*)smem;
    float* s_k = s_q + 128;
    float* s_v = s_k + 128 * 32; // tile size example
    float* s_acc = s_v + 128 * 32;

    const float scale = 1.0f / sqrtf((float)D);

    for (int i = 0; i < S; i++) {
        // load q
        if (tid < D) {
            s_q[tid] = __bfloat162float(q[((batch * Hq + head) * S + i) * D + tid]);
        }
        __syncthreads();

        float max_score = -INFINITY;
        float sum_exp = 0.0f;
        float acc[D] = {0.0f}; // register acc, D=128

        for (int j_tile = 0; j_tile < S; j_tile += 32) {
            int j = j_tile + (tid % 32);
            if (j < S) {
                float score = 0.0f;
                for (int d = 0; d < D; d++) {
                    float kk = __bfloat162float(k[((batch * Hkv + kv_head) * S + j) * D + d]);
                    score += s_q[d] * kk;
                }
                score *= scale;

                float new_max = fmaxf(max_score, score);
                float exp_s = expf(score - new_max);
                float exp_old = expf(max_score - new_max);
                sum_exp = sum_exp * exp_old + exp_s;
                max_score = new_max;

                for (int d = 0; d < D; d++) {
                    float vv = __bfloat162float(v[((batch * Hkv + kv_head) * S + j) * D + d]);
                    acc[d] = acc[d] * exp_old + exp_s * vv;
                }
            }
        }
        __syncthreads();

        // normalize and write
        if (tid < D) {
            float o = acc[tid] / sum_exp;
            out[((batch * Hq + head) * S + i) * D + tid] = __float2bfloat16(o);
        }
    }
}

torch::Tensor flash_attn_cuda(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
    auto B = q.size(0);
    auto Hq = q.size(1);
    auto S = q.size(2);
    auto D = q.size(3);
    auto Hkv = k.size(1);
    auto out = torch::empty_like(q);

    dim3 grid(B, Hq);
    dim3 block(256);
    int smem_size = (128 + 128*32 + 128*32 + 128) * sizeof(float);
    flash_attn_kernel<nv_bfloat16><<<grid, block, smem_size>>>(
        (nv_bfloat16*)q.data_ptr<at::BFloat16>(),
        (nv_bfloat16*)k.data_ptr<at::BFloat16>(),
        (nv_bfloat16*)v.data_ptr<at::BFloat16>(),
        (nv_bfloat16*)out.data_ptr<at::BFloat16>(),
        B, Hq, Hkv, S, D);
    return out;
}
