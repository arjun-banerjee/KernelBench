import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_CPP_SRC = r"""
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> rope_forward(torch::Tensor q,
                                        torch::Tensor k,
                                        torch::Tensor cos,
                                        torch::Tensor sin,
                                        int64_t rot_dim);

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def("rope_forward", &rope_forward, "Hopper optimized FP16 rotary embedding (in-place)");
}
"""

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <vector>

__global__ __launch_bounds__(512, 2)
void rotary_split_half_kernel(
    __half* __restrict__ q,
    __half* __restrict__ k,
    const __half* __restrict__ cos,
    const __half* __restrict__ sin,
    int num_tokens,
    int num_heads,
    int num_kv_heads,
    int head_size,
    int rot_dim) {

  const int token = blockIdx.x;
  const int tid = threadIdx.x;
  const int half_dim = rot_dim >> 1;

  extern __shared__ __half smem[];
  __half* sh_cos = smem;
  __half* sh_sin = smem + rot_dim;

  // Cos/sin are reused by all Q and K heads of this token.  Keep the small
  // per-token table in shared memory to avoid rereading it for every head.
  for (int d = tid; d < rot_dim; d += blockDim.x) {
    const int off = token * rot_dim + d;
    sh_cos[d] = cos[off];
    sh_sin[d] = sin[off];
  }
  __syncthreads();

  const int q_pairs = num_heads * half_dim;
  const int total_pairs = (num_heads + num_kv_heads) * half_dim;

  for (int linear = tid; linear < total_pairs; linear += blockDim.x) {
    __half* base;
    int local = linear;
    if (linear < q_pairs) {
      const int h = linear / half_dim;
      local = linear - h * half_dim;
      base = q + ((token * num_heads + h) * head_size);
    } else {
      const int kk = linear - q_pairs;
      const int h = kk / half_dim;
      local = kk - h * half_dim;
      base = k + ((token * num_kv_heads + h) * head_size);
    }

    const int d0 = local;
    const int d1 = local + half_dim;

    const float x0 = __half2float(base[d0]);
    const float x1 = __half2float(base[d1]);

    // rotate_half convention: [-x_second_half, x_first_half].
    const float c0 = __half2float(sh_cos[d0]);
    const float s0 = __half2float(sh_sin[d0]);
    const float c1 = __half2float(sh_cos[d1]);
    const float s1 = __half2float(sh_sin[d1]);

    base[d0] = __float2half_rn(fmaf(x0, c0, -x1 * s0));
    base[d1] = __float2half_rn(fmaf(x1, c1,  x0 * s1));
  }
}

std::vector<torch::Tensor> rope_forward(torch::Tensor q,
                                        torch::Tensor k,
                                        torch::Tensor cos,
                                        torch::Tensor sin,
                                        int64_t rot_dim64) {
  const int num_tokens = (int)q.size(0);
  const int num_heads = (int)q.size(1);
  const int head_size = (int)q.size(2);
  const int num_kv_heads = (int)k.size(1);
  const int rot_dim = (int)rot_dim64;

  dim3 grid(num_tokens);
  dim3 block(512);
  size_t shmem = (size_t)(2 * rot_dim) * sizeof(__half);

  rotary_split_half_kernel<<<grid, block, shmem, at::cuda::getCurrentCUDAStream()>>>(
      reinterpret_cast<__half*>(q.data_ptr<at::Half>()),
      reinterpret_cast<__half*>(k.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(cos.data_ptr<at::Half>()),
      reinterpret_cast<const __half*>(sin.data_ptr<at::Half>()),
      num_tokens, num_heads, num_kv_heads, head_size, rot_dim);

  return {q, k};
}
"""
