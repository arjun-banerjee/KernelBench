      auto* x_packed = reinterpret_cast<packed_t*>(&x);
      auto* y_packed = reinterpret_cast<packed_t*>(&y);
      auto* out_packed = reinterpret_cast<packed_t*>(&x);  // Reuse x for output
      
#pragma unroll
      for (int j = 0; j < PACKED_SIZE; j++) {
        out_packed[j] = packed_compute<packed_t, PACKED_ACT_FN, act_first, HAS_CLAMP>(
            x_packed[j], y_packed[j], limit);
      }
      
      if constexpr (use_256b) {
        st256(reinterpret_cast<uint4&>(x), &reinterpret_cast<uint4*>(out_vec)[i]);
      } else {
        st128(reinterpret_cast<uint2&>(x), &reinterpret_cast<uint2*>(out_vec)[i]);
      }
    }
    
    // Handle remaining elements
    const int remainder_start = num_vecs * VEC_SIZE;
    for (int idx = remainder_start + threadIdx.x; idx < d; idx += blockDim.x) {
      const scalar_t x = __ldg(&x_ptr[idx]);
      const scalar_t y = __ldg(&y_ptr[idx]);
      out_ptr[idx] = compute<scalar_t, ACT_FN, act_first, HAS_CLAMP>(x, y, limit);
    }
  } else {
    // Scalar fallback
    for (int64_t idx = threadIdx.x; idx < d; idx += blockDim.x) {
      const scalar_t x = __ldg(&x_ptr[idx]);
      const scalar_t y = __ldg(&y_ptr[idx]);
      out_ptr[idx] = compute<scalar_t, ACT_FN, act_first, HAS_CLAMP>(x, y, limit);
    }
  }
}

// Activation kernel with parameter (for fatrelu)
template <typename scalar_t, typename packed_t,
          scalar_t (*ACT_FN)(const scalar_t&, const float),
          packed_t (*PACKED_ACT_FN)(const packed_t&, const float), 
          bool use_vec, bool use_256b = false>
__global__ void act_and_mul_kernel_with_param_hopper(
    scalar_t* __restrict__ out, const scalar_t* __restrict__ input, const int d,
    const float param) {
  
  const scalar_t* x_ptr = input + blockIdx.x * 2 * d;
  const scalar_t* y_ptr = x_ptr + d;
  scalar_t* out_ptr = out + blockIdx.x * d;

  if constexpr (use_vec) {
    constexpr int ARCH_VEC_SIZE = use_256b ? 32 : 16;
    constexpr int VEC_SIZE = ARCH_VEC_SIZE / sizeof(scalar_t);
    constexpr int PACKED_SIZE = VEC_SIZE / 2;
    
    using vec_t = typename VecTraitsHopper<use_256b>::vec_t;
    
    const vec_t* x_vec = reinterpret_cast<const vec_t*>(x_ptr);
    const vec_t* y_vec = reinterpret_cast<const vec_t*>(y_ptr);
    vec_t* out_vec = reinterpret_cast<vec_t*>(out_ptr);
    const int num_vecs = d / VEC_SIZE;

    for (int i = threadIdx.x; i < num_vecs; i += blockDim.x) {
      vec_t x, y;
      if constexpr (use_256b) {
        ld256(reinterpret_cast<uint4&>(x), &reinterpret_cast<const uint4*>(x_vec)[i]);
        ld256(reinterpret_cast<uint4&>(y), &reinterpret_cast<const uint4*>(y_vec)[i]);
      } else {
        ld128(reinterpret_cast<uint2&>(x), &reinterpret_cast<const uint2*>(x_vec)[i]);
        ld128(reinterpret_cast<uint2&>(y), &reinterpret_cast<const uint2*>(y_vec)[i]);
      }
      
      auto* x_packed = reinterpret_cast<packed_t*>(&x);
      auto* y_packed = reinterpret_cast<packed_t*>(&y);
      auto* out_packed = reinterpret_cast<packed_t*>(&x);
      
#pragma unroll
      for (int j = 0; j < PACKED_SIZE; j++) {
        out_packed[j] = packed_mul(PACKED_ACT_FN(x_packed[j], param), y_packed[j]);
      }
      
      if constexpr (use_256b) {
        st256(reinterpret_cast<uint4&>(x), &reinterpret_cast<uint4*>(out_vec)[i]);
      } else {
        st128(reinterpret_cast<uint2&>(x), &reinterpret_cast<uint2*>(out_vec)[i]);
      }
    }
    
    const int remainder_start = num_vecs * VEC_SIZE;
    for (int idx = remainder_start + threadIdx.x; idx < d; idx += blockDim.x) {
      const scalar_t x = __ldg(&x_ptr[idx]);
      const scalar_t y = __ldg(&y_ptr[idx]);
      out_ptr[idx] = ACT_FN(x, param) * y;
    }
  } else {
    for (int64_t idx = threadIdx.x; idx < d; idx += blockDim.x) {
      const scalar_t x = __ldg(&x_ptr[idx]);
      const scalar_t y = __ldg(&y_ptr[idx]);
      out_ptr[idx] = ACT_FN(x, param) * y;
    }
  }
}

// Element-wise activation kernel
template <typename scalar_t, scalar_t (*ACT_FN)(const scalar_t&), bool use_vec,
          bool use_256b = false>
__global__ void activation_kernel_hopper(
    scalar_t* __restrict__ out,
    const scalar_t* __restrict__ input,
    const int d) {
  
  const scalar_t* in_ptr = input + blockIdx.x * d;
  scalar_t* out_ptr = out + blockIdx.x * d;

  if constexpr (use_vec) {
    constexpr int ARCH_VEC_SIZE = use_256b ? 32 : 16;
    constexpr int VEC_SIZE = ARCH_VEC_SIZE / sizeof(scalar_t);
    
    using vec_t = typename VecTraitsHopper<use_256b>::vec_t;
    
    const vec_t* in_vec = reinterpret_cast<const vec_t*>(in_ptr);
    vec_t* out_vec = reinterpret_cast<vec_t*>(out_ptr);
    const int num_vecs = d / VEC_SIZE;

    for (int i = threadIdx.x; i < num_vecs; i += blockDim.x) {
      vec_t v;
      if constexpr (use_256b) {
        ld256(reinterpret_cast<uint4&>(v), &reinterpret_cast<const uint4*>(in_vec)[i]);
      } else {
        v = __ldg(&in_vec[i]);
      }
      
      auto* vp = reinterpret_cast<scalar_t*>(&v);
#pragma unroll
      for (int j = 0; j < VEC_SIZE; j++) {
        vp[j] = ACT_FN(vp[j]);
      }
      
      if constexpr (use_256b) {
        st256(reinterpret_cast<uint4&>(v), &reinterpret_cast<uint4*>(out_vec)[i]);
      } else {
        out_vec[i] = v;
      }
    }
    
    const int remainder_start = num_vecs * VEC_SIZE;
    for (int idx = remainder_start + threadIdx.x; idx < d; idx += blockDim.x) {
      out_ptr[idx] = ACT_FN(__ldg(&in_ptr[idx]));
    }
  } else {
    for (int64_t idx = threadIdx.x; idx < d; idx += blockDim.x) {
      out_ptr[idx] = ACT_FN(__ldg(&in_ptr[idx]));
    }
  }
}
