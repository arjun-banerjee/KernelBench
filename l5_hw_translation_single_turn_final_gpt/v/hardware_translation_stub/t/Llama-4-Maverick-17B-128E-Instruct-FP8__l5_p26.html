import torch.nn as nn
from torch.utils.cpp_extension import load_inline

cuda_source = """
#include <cuda_runtime.h>
#include <cuda_fp16.h>

__global__ void gemm_permute_kernel(const half* A, const half* B, half* C, int batch, int M, int N, int K, int D1, float alpha, float beta) {
    int batch_idx = blockIdx.z;
    int m = blockIdx.y * blockDim.y + threadIdx.y;
    int n = blockIdx.x * blockDim.x + threadIdx.x;

    if (m >= M || n >= N) return;

    half2 acc = __float2half2_rn(0.0f);
    for (int k = 0; k < K; ++k) {
        half2 a = __halves2half2(A[batch_idx * M * K + m * K + k], A[batch_idx * M * K + m * K + k]);
        half2 b = __halves2half2(B[batch_idx * K * N + k * N + n], B[batch_idx * K * N + k * N + n]);
        acc = __hfma2(a, b, acc);
    }

    half result = __hadd(__low2half(acc), __high2half(acc));
    result = __float2half(alpha * __half2float(result) + beta * __half2float(C[batch_idx * M * N + m * N + n]));

    // Permute operation: Tensor4DPermuteBMM0213
    int new_batch_idx = batch_idx / D1;
    int d1_idx = batch_idx % D1;
    int permuted_idx = new_batch_idx * M * N * D1 + m * N * D1 + d1_idx * N + n;
    C[permuted_idx] = result;
}

torch::Tensor gemm_permute(torch::Tensor A, torch::Tensor B, torch::Tensor C, int D1, float alpha, float beta) {
    const int batch = A.size(0);
    const int M = A.size(1);
    const int K = A.size(2);
    const int N = B.size(2);

    auto output = torch::empty_like(C);

    dim3 block(16, 16);
    dim3 grid((N + block.x - 1) / block.x, (M + block.y - 1) / block.y, batch);

        A.data_ptr<at::Half>(),
        B.data_ptr<at::Half>(),

}
