#include <cuda_fp16.h>

__global__ void batched_gemm_kernel(float* A, float* B, float* C, int M, int N, int K, float alpha, float beta, int batch_count, int lda, int ldb, int ldc, long long batch_stride_A, long long batch_stride_B, long long batch_stride_C) {

    C[batch_idx * batch_stride_C + n_idx * ldc + m_idx] = accum;
