    int8_t* __restrict__ row_out = output + token_idx * hidden_size;

    float thread_max = 0.0f;
    
    // Vectorized scan for absmax
    const int vec_size = 8;
    const int stride = block_size * vec_size;
    const int base_idx = tid * vec_size;
    
    for (int i = base_idx; i < hidden_size; i += stride) {
        if (i + vec_size <= hidden_size) {
            const float4* vec_in = reinterpret_cast<const float4*>(row_in + i);
            float4 vals = *vec_in;
            half2* h2_ptr = reinterpret_cast<half2*>(&vals);
            
            #pragma unroll
            for (int j = 0; j < 4; ++j) {
                half2 h2_val = h2_ptr[j];
                float f0 = fabsf(__half2float(h2_val.x));
                float f1 = fabsf(__half2float(h2_val.y));
                thread_max = fmaxf(thread_max, f0);
                thread_max = fmaxf(thread_max, f1);
            }
        } else {
            for (int j = 0; j < vec_size && i + j < hidden_size; ++j) {
                float f = fabsf(static_cast<float>(row_in[i + j]));
                thread_max = fmaxf(thread_max, f);
            }
        }
    }
    
    // Block reduction using warp shuffles
    float block_max = block_reduce_max(thread_max, block_size);
    
    __shared__ float absmax_sh;
    if (tid == 0) {
        absmax_sh = block_max;
        scale_out[blockIdx.x] = absmax_sh / 127.0f;
    }
    __syncthreads();
    
    // Phase 2: Quantize with computed scale
    float absmax = absmax_sh;
    float inv_s = (absmax == 0.0f) ? 0.0f : 127.0f / absmax;
    
    for (int i = base_idx; i < hidden_size; i += stride) {
        if (i + vec_size <= hidden_size) {
            const float4* vec_in = reinterpret_cast<const float4*>(row_in + i);
            float4 vals = *vec_in;
            half2* h2_ptr = reinterpret_cast<half2*>(&vals);
            int8_t local_out[8];
            
            #pragma unroll
            for (int j = 0; j < 4; ++j) {
                half2 h2_val = h2_ptr[j];
                float f0 = __half2float(h2_val.x);
                float f1 = __half2float(h2_val.y);
                local_out[j*2] = float_to_int8_rn(f0 * inv_s);
                local_out[j*2 + 1] = float_to_int8_rn(f1 * inv_s);
            }
            
            *(reinterpret_cast<int2*>(row_out + i)) = *reinterpret_cast<int2*>(local_out);
        } else {
            for (int j = 0; j < vec_size && i + j < hidden_size; ++j) {
                row_out[i + j] = float_to_int8_rn(static_cast<float>(row_in[i + j]) * inv_s);
            }
        }
    }
}

// H100-optimized dynamic scaled int8 asymmetric quantization kernel
template <typename scalar_t, typename scale_t, typename azp_t>
__global__ void __launch_bounds__(512) dynamic_scaled_int8_azp_quant_kernel(
    const scalar_t* __restrict__ input,
    int8_t* __restrict__ output,
    scale_t* scale_out,
    azp_t* azp_out,
    const int hidden_size) {
    
    const int tid = threadIdx.x;
    const int block_size = blockDim.x;
    const int64_t token_idx = blockIdx.x;
    
    const scalar_t* __restrict__ row_in = input + token_idx * hidden_size;
    int8_t* __restrict__ row_out = output + token_idx * hidden_size;
    
    // Phase 1: Find min/max
    MinMax thread_mm;
    
    const int vec_size = 8;
    const int stride = block_size * vec_size;
    const int base_idx = tid * vec_size;
    
    for (int i = base_idx; i < hidden_size; i += stride) {
        if (i + vec_size <= hidden_size) {
            const float4* vec_in = reinterpret_cast<const float4*>(row_in + i);
            float4 vals = *vec_in;
            half2* h2_ptr = reinterpret_cast<half2*>(&vals);
            
            #pragma unroll
            for (int j = 0; j < 4; ++j) {
                half2 h2_val = h2_ptr[j];
                float f0 = __half2float(h2_val.x);
                float f1 = __half2float(h2_val.y);
                thread_mm += f0;
                thread_mm += f1;
            }
        } else {
            for (int j = 0; j < vec_size && i + j < hidden_size; ++j) {
                thread_mm += static_cast<float>(row_in[i + j]);
            }
        }
    }
    
    // Block reduction for MinMax
    MinMax block_mm = block_reduce_minmax(thread_mm, block_size);
    
    __shared__ float scale_sh;
    __shared__ azp_t azp_sh;
    
    if (tid == 0) {
        float s = (block_mm.max_val - block_mm.min_val) / 255.0f;
        float zp = nearbyintf(-128.0f - block_mm.min_val / s);
        scale_sh = s;
        azp_sh = static_cast<azp_t>(zp);
        scale_out[blockIdx.x] = s;
        azp_out[blockIdx.x] = azp_sh;
    }
    __syncthreads();
    
    // Phase 2: Quantize
    const float inv_s = 1.0f / scale_sh;
    const azp_t azp = azp_sh;
    
    for (int i = base_idx; i < hidden_size; i += stride) {
        if (i + vec_size <= hidden_size) {
            const float4* vec_in = reinterpret_cast<const float4*>(row_in + i);
            float4 vals = *vec_in;
            half2* h2_ptr = reinterpret_cast<half2*>(&vals);
            int8_t local_out[8];
            
            #pragma unroll
            for (int j = 0; j < 4; ++j) {
                half2 h2_val = h2_ptr[j];
                float f0 = __half2float(h2_val.x);
                float f1 = __half2float(h2_val.y);
                local_out[j*2] = int32_to_int8(float_to_int32_rn(f0 * inv_s) + azp);
                local_out[j*2 + 1] = int32_to_int8(float_to_int32_rn(f1 * inv_s) + azp);
            }
            
            *(reinterpret_cast<int2*>(row_out + i)) = *reinterpret_cast<int2*>(local_out);
        } else {
            for (int j = 0; j < vec_size && i + j < hidden_size; ++j) {
                const auto v = static_cast<float>(row_in[i + j]) * inv_s;
                row_out[i + j] = int32_to_int8(float_to_int32_rn(v) + azp);
            }
        }
    }
}

class ModelNew(nn.Module):
    """
    H100-optimized INT8 quantization module.
    
    Performs per-token symmetric dynamic quantization, computing scale per row
    and quantizing to int8. Optimized for Hopper architecture with:
    - Larger thread blocks (512 threads) for better occupancy
    - Vectorized 128-bit loads/stores (8 fp16 values per transaction)
    - Warp-level shuffle reductions instead of shared memory
    - Launch bounds for optimal register usage
    """

        self.K = K
        self.N = N
        self._kernel = _get_kernel()

        Quantize input tensor to int8 using per-row symmetric quantization.
        
        Args:

            Quantized tensor of shape [M, K] in int8
