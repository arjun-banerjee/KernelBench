    const int hq = tmp0 % Hq;
    const int b = tmp0 / Hq;
    const int hk = hq / (Hq / Hkv);
    const half* qptr = q + (((b * Hq + hq) * S + s_q) * D);
    const half* kbase = k + (((b * Hkv + hk) * S) * D);
    const half* vbase = v + (((b * Hkv + hk) * S) * D);
    half* optr = out + (((b * Hq + hq) * S + s_q) * D);
    extern __shared__ float smem[];
    float* scores = smem;
    int tid = threadIdx.x;
    const half2* q2 = reinterpret_cast<const half2*>(qptr);
    float row_max = -INFINITY;
    for (int sk = tid; sk <= s_q; sk += blockDim.x) {
        const half2* k2 = reinterpret_cast<const half2*>(kbase + sk * D);
        float dot = 0.0f;
#pragma unroll 4
        for (int i = 0; i < 64; ++i) {
            float2 qf = __half22float2(q2[i]);
            float2 kf = __half22float2(k2[i]);
            dot = fmaf(qf.x, kf.x, dot);
            dot = fmaf(qf.y, kf.y, dot);
        }
        dot *= scale;
        scores[sk] = dot;
        row_max = fmaxf(row_max, dot);
    }
#pragma unroll
    for (int off = 16; off > 0; off >>= 1) row_max = fmaxf(row_max, __shfl_down_sync(0xffffffff, row_max, off));
    __shared__ float wb[8];
    if ((tid & 31) == 0) wb[tid >> 5] = row_max;
    __syncthreads();
    row_max = (tid < 8) ? wb[tid] : -INFINITY;
    if (tid < 32) {
#pragma unroll
        for (int off = 16; off > 0; off >>= 1) row_max = fmaxf(row_max, __shfl_down_sync(0xffffffff, row_max, off));
        if (tid == 0) wb[0] = row_max;
    }
    __syncthreads();
    row_max = wb[0];
    float denom = 0.0f;
    for (int sk = tid; sk <= s_q; sk += blockDim.x) {
        float e = __expf(scores[sk] - row_max);
        scores[sk] = e;
        denom += e;
    }
#pragma unroll
    for (int off = 16; off > 0; off >>= 1) denom += __shfl_down_sync(0xffffffff, denom, off);
    if ((tid & 31) == 0) wb[tid >> 5] = denom;
    __syncthreads();
    denom = (tid < 8) ? wb[tid] : 0.0f;
    if (tid < 32) {
#pragma unroll
        for (int off = 16; off > 0; off >>= 1) denom += __shfl_down_sync(0xffffffff, denom, off);
        if (tid == 0) wb[0] = denom;
    }
    __syncthreads();
    float inv_denom = 1.0f / wb[0];
    for (int d = tid; d < D; d += blockDim.x) {
        float acc = 0.0f;
        for (int sk = 0; sk <= s_q; ++sk) {
            acc = fmaf(scores[sk] * inv_denom, __half2float(vbase[sk * D + d]), acc);
        }
        optr[d] = __float2half_rn(acc);
    }
}
