#define WARP_SIZE 32

__global__ void flash_attention_fp16_kernel(
    const __half* __restrict__ q,
    const __half* __restrict__ k,
    const __half* __restrict__ v,
    __half* __restrict__ out,
    const int batch_size,
    const int num_heads,
    const int num_kv_heads,
    const int seq_len,
    const int head_dim,
    const float scale) {

    const int b = blockIdx.x;
    const int h = blockIdx.y;
    const int kv_h = h / (num_heads / num_kv_heads);
    
    extern __shared__ __half smem[];
    __half* k_smem = smem;
    __half* v_smem = smem + seq_len * head_dim;

    const int tid = threadIdx.x;
    const int warp_id = tid / WARP_SIZE;
    const int lane = tid % WARP_SIZE;

    // Load K and V into shared memory (coalesced for H100)
    for (int i = tid; i < seq_len * head_dim; i += blockDim.x) {
        k_smem[i] = k[((b * num_kv_heads + kv_h) * seq_len * head_dim) + i];
        v_smem[i] = v[((b * num_kv_heads + kv_h) * seq_len * head_dim) + i];
    }
    __syncthreads();

    // Each thread computes one or more rows of Q @ K^T
    for (int row = tid; row < seq_len; row += blockDim.x) {
        float acc[8]; // assume head_dim=128, process in chunks
