    float o_local[HEAD_DIM / 4];
    float q_local[HEAD_DIM / 4];
    float m_max = -INFINITY;
    float d_sum = 0.0f;
    
    #pragma unroll
    for (int i = 0; i < HEAD_DIM / 4; i++) {
        o_local[i] = 0.0f;
        q_local[i] = 0.0f;
    }
    
    // Load Q into registers (each thread loads HEAD_DIM/4 elements)
    if (valid) {
        int base_idx = batch * num_heads * seq_len * HEAD_DIM + head * seq_len * HEAD_DIM + q_pos * HEAD_DIM;
        #pragma unroll
        for (int i = 0; i < HEAD_DIM / 4; i++) {
            int dim = q_col * (HEAD_DIM / 4) + i;
            q_local[i] = __half2float(q[base_idx + dim]);
        }
    }
    
    // Iterate through KV tiles
    for (int kv_start = 0; kv_start < seq_len; kv_start += KV_TILE) {
        int kv_end = min(kv_start + KV_TILE, seq_len);
        int kv_len = kv_end - kv_start;
        
        // Cooperatively load K and V tiles
        // Each thread loads multiple elements for memory coalescing
        #pragma unroll
        for (int i = 0; i < (KV_TILE * HEAD_DIM) / THREADS; i++) {
            int idx = tid + i * THREADS;
            int kv_row = idx / HEAD_DIM;
            int dim = idx % HEAD_DIM;
            int kv_pos = kv_start + kv_row;
            
            if (kv_pos < seq_len) {
                int k_idx = batch * num_kv_heads * seq_len * HEAD_DIM + kv_head * seq_len * HEAD_DIM + kv_pos * HEAD_DIM + dim;
                int v_idx = k_idx;
                k_smem[idx] = k[k_idx];
                v_smem[idx] = v[v_idx];
            } else {
                k_smem[idx] = __float2half(0.0f);
                v_smem[idx] = __float2half(0.0f);
            }
        }
        
        __syncthreads();
        
        // Compute attention for each KV row
        for (int kv_row = 0; kv_row < kv_len; kv_row++) {
            int kv_pos = kv_start + kv_row;
            
            // Causal mask
            if (valid && q_pos >= kv_pos) {
                // Compute partial dot product
                float dot = 0.0f;
                #pragma unroll
                for (int i = 0; i < HEAD_DIM / 4; i++) {
                    int dim = q_col * (HEAD_DIM / 4) + i;
                    dot += q_local[i] * __half2float(k_smem[kv_row * HEAD_DIM + dim]);
                }
                
                // Warp-level reduction
                #pragma unroll
                for (int offset = 2; offset > 0; offset >>= 1) {
                    dot += __shfl_down_sync(0xffffffff, dot, offset);
                }
                dot = __shfl_sync(0xffffffff, dot, 0) * scale;
                
                // Softmax update (only thread 0 in warp computes full values)
                float m_new = max(m_max, dot);
                float d_scale = exp(m_max - m_new);
                float dot_exp = exp(dot - m_new);
                
                // Update output
                #pragma unroll
                for (int i = 0; i < HEAD_DIM / 4; i++) {
                    int dim = q_col * (HEAD_DIM / 4) + i;
                    o_local[i] = o_local[i] * d_scale + dot_exp * __half2float(v_smem[kv_row * HEAD_DIM + dim]);
                }
                
                if (q_col == 0) {
                    d_sum = d_sum * d_scale + dot_exp;
                    m_max = m_new;
                }
            }
        }
        
        __syncthreads();
    }
    
    // Broadcast denominator from thread 0 in each warp
    d_sum = __shfl_sync(0xffffffff, d_sum, 0);
    
    // Write output
    if (valid) {
        int base_idx = batch * num_heads * seq_len * HEAD_DIM + head * seq_len * HEAD_DIM + q_pos * HEAD_DIM;
        #pragma unroll
        for (int i = 0; i < HEAD_DIM / 4; i++) {
            int dim = q_col * (HEAD_DIM / 4) + i;
            out[base_idx + dim] = __float2half(o_local[i] / max(d_sum, 1e-6f));
        }
    }
}
