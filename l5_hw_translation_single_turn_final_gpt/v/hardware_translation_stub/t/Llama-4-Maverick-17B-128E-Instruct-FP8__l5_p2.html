#define WARP_SIZE 32
#define NUM_WARPS 4
#define NUM_THREADS (WARP_SIZE * NUM_WARPS)
#define PARTITION_SIZE 512

template <typename T, int HEAD_SIZE>
__global__ void paged_attention_v2_kernel(T* out, float* exp_sums, float* max_logits, T* query, T* key_cache, T* value_cache, int num_kv_heads, float scale, int* block_tables, int* seq_lens, int max_num_blocks_per_seq) {
    // Kernel implementation for paged attention v2
    int num_seqs = gridDim.y;
    int num_heads = gridDim.x;

    int head_id = blockIdx.x;
    int partition_id = blockIdx.z;

    // Calculate the thread's position in the sequence and head
    int seq_len = seq_lens[seq_id];
    int num_partitions = (seq_len + PARTITION_SIZE - 1) / PARTITION_SIZE;
    if (partition_id >= num_partitions) return;

    // Load query, key, and value
    T query_vec[HEAD_SIZE / NUM_THREADS];
    T key_vec[HEAD_SIZE / NUM_THREADS];
    T value_vec[HEAD_SIZE / NUM_THREADS];

    // Shared memory for storing logits and outputs
    __shared__ float logits[PARTITION_SIZE];
    __shared__ float outputs[HEAD_SIZE];

    // Calculate logits and outputs
    float max_logit = -INFINITY;
    float sum_exp = 0.0f;
    for (int i = threadIdx.x; i < PARTITION_SIZE; i += NUM_THREADS) {
        // Load key and value from global memory
        int block_id = block_tables[seq_id * max_num_blocks_per_seq + partition_id * PARTITION_SIZE + i];
        key_vec[threadIdx.x % (HEAD_SIZE / NUM_THREADS)] = key_cache[block_id * num_kv_heads * HEAD_SIZE + head_id * HEAD_SIZE + threadIdx.x % (HEAD_SIZE / NUM_THREADS)];
