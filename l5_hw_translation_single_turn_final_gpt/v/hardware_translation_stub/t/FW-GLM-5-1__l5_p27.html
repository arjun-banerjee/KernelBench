// Main kernel using direct computation (most reliable for correctness)
__launch_bounds__(256, 1)
__global__ void batched_gemm_fp16_direct_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B, 
    half* __restrict__ C,
    half alpha,
    half beta,
    int M, int N, int K,
    long long batch_stride_A, 
    long long batch_stride_B,
    long long batch_stride_C,
    int lda, int ldb, int ldc,
    int batch_count) {
    
    const int batch_idx = blockIdx.z;
    const int m_idx = blockIdx.x * blockDim.x + threadIdx.x;
    const int n_idx = blockIdx.y * blockDim.y + threadIdx.y;
    
    if (batch_idx >= batch_count || m_idx >= M || n_idx >= N) return;
    
    const half* batch_A = A + batch_idx * batch_stride_A;
    const half* batch_B = B + batch_idx * batch_stride_B;
    half* batch_C = C + batch_idx * batch_stride_C;
    
    // Compute dot product for this output element
    float sum = 0.0f;
    
    for (int k = 0; k < K; k++) {
        // A[m][k] and B[k][n] in column-major
        half a_val = batch_A[m_idx + k * lda];
        half b_val = batch_B[k + n_idx * ldb];
        sum += __half2float(a_val) * __half2float(b_val);
    }
    
    // Apply alpha * result + beta * C_original
    half c_orig = batch_C[m_idx + n_idx * ldc];
    float result = __half2float(alpha) * sum + __half2float(beta) * __half2float(c_orig);
    
    batch_C[m_idx + n_idx * ldc] = __float2half(result);
}

// Optimized kernel using WMMA with shared memory tiling
__launch_bounds__(128, 4)
__global__ void batched_gemm_fp16_wmma_kernel(
    const half* __restrict__ A,
    const half* __restrict__ B,
    half* __restrict__ C,
    half alpha_val,
    half beta_val,
    int M, int N, int K,
    long long batch_stride_A,
    long long batch_stride_B, 
    long long batch_stride_C,
    int lda, int ldb, int ldc,
    int batch_count) {
    
    // Thread block handles 64x64 output tile
    // 4 warps (128 threads), each warp handles 32x32 via 2x2 WMMA
    
    const int batch_idx = blockIdx.z;
    const int tile_m = blockIdx.x * 64;
    const int tile_n = blockIdx.y * 64;
    
    if (batch_idx >= batch_count) return;
    
    const int warp_id = threadIdx.x >> 5;
    const int lane_id = threadIdx.x & 31;
    
    // Each warp processes 32x32 output
    const int warp_row = tile_m + ((warp_id & 1) << 5);  // 0 or 32
    const int warp_col = tile_n + ((warp_id >> 1) << 5); // 0 or 32
    
    const half* batch_A = A + batch_idx * batch_stride_A;
    const half* batch_B = B + batch_idx * batch_stride_B;
    half* batch_C = C + batch_idx * batch_stride_C;
    
    const float alpha = __half2float(alpha_val);
    const float beta = __half2float(beta_val);
    
    // Shared memory for A and B tiles
    __shared__ half smem_A[64][16];  // 64 rows x 16 cols
    __shared__ half smem_B[16][64];  // 16 rows x 64 cols
    
    // Accumulators for 32x32 output (2x2 WMMA tiles)
    fragment<accumulator, 16, 16, 16, float> acc[2][2];
    #pragma unroll
    for (int i = 0; i < 2; i++) {
        #pragma unroll
        for (int j = 0; j < 2; j++) {
            fill_fragment(acc[i][j], 0.0f);
        }
    }
    
    // Loop over K in tiles of 16
    const int K_tiles = (K + 15) >> 4;
    
    for (int k_tile = 0; k_tile < K_tiles; k_tile++) {
        const int k_base = k_tile << 4;
        const int k_size = min(16, K - k_base);
        
        // Cooperative load of A tile (64x16)
        for (int i = threadIdx.x; i < 64 * 16; i += 128) {
            int row = i & 63;
            int col = i >> 6;
            int global_row = tile_m + row;
            int global_col = k_base + col;
            if (global_row < M && global_col < K) {
                smem_A[row][col] = batch_A[global_row + global_col * lda];
            } else {
                smem_A[row][col] = __float2half(0.0f);
            }
        }
        
        // Cooperative load of B tile (16x64)
        for (int i = threadIdx.x; i < 16 * 64; i += 128) {
            int row = i & 15;
            int col = i >> 4;
            int global_row = k_base + row;
            int global_col = tile_n + col;
            if (global_row < K && global_col < N) {
                smem_B[row][col] = batch_B[global_row + global_col * ldb];
            } else {
                smem_B[row][col] = __float2half(0.0f);
            }
        }
        
        __syncthreads();
        
        // Each warp computes 32x32 output
        #pragma unroll
        for (int wm = 0; wm < 2; wm++) {
            #pragma unroll
            for (int wn = 0; wn < 2; wn++) {
                const int out_row = warp_row + (wm << 4);
                const int out_col = warp_col + (wn << 4);
                
                if (out_row >= M || out_col >= N) continue;
                
                fragment<matrix_a, 16, 16, 16, half, col_major> a_frag;
                fragment<matrix_b, 16, 16, 16, half, col_major> b_frag;
                
                // Load A from shared memory (row offset within tile)
                const int a_row_offset = (warp_id & 1) * 32 + (wm << 4);
                load_matrix_sync(a_frag, &smem_A[a_row_offset][0], 16);
                
                // Load B from shared memory (col offset within tile)
                const int b_col_offset = (warp_id >> 1) * 32 + (wn << 4);
                load_matrix_sync(b_frag, &smem_B[0][b_col_offset], 64);
                
                mma_sync(acc[wm][wn], a_frag, b_frag, acc[wm][wn]);
            }
        }
        
        __syncthreads();
    }
    
    // Store results with alpha/beta scaling
    #pragma unroll
    for (int wm = 0; wm < 2; wm++) {
        #pragma unroll
        for (int wn = 0; wn < 2; wn++) {
            const int out_row = warp_row + (wm << 4);
            const int out_col = warp_col + (wn << 4);
            
            if (out_row >= M || out_col >= N) continue;
            
            // Load original C
            fragment<matrix_a, 16, 16, 16, half, col_major> c_orig;
            load_matrix_sync(c_orig, &batch_C[out_row + out_col * ldc], ldc, mem_col_major);
            
            // Apply scaling
            #pragma unroll
            for (int i = 0; i < c_orig.num_elements; i++) {
                float c_val = __half2float(c_orig.x[i]);
                float a_val = acc[wm][wn].x[i];
                acc[wm][wn].x[i] = alpha * a_val + beta * c_val;
            }
            
            // Convert to half and store
            fragment<accumulator, 16, 16, 16, half> result;
            #pragma unroll
            for (int i = 0; i < result.num_elements; i++) {
                result.x[i] = __float2half(acc[wm][wn].x[i]);
            }
            
            store_matrix_sync(&batch_C[out_row + out_col * ldc], result, ldc, mem_col_major);
        }
    }
}
