template <bool IS_NEOX>
__global__ void rotary_embedding_kernel_fp16(
    const half* __restrict__ query,
    const half* __restrict__ key,
    const half* __restrict__ cos,
    const half* __restrict__ sin,
    half* __restrict__ out_query,
    half* __restrict__ out_key,
    const int num_tokens,
    const int num_heads,
    const int num_kv_heads,
    const int head_size,
    const int rot_dim) {

    const int token_idx = blockIdx.x;
    const int head_idx = blockIdx.y;
    const int tid = threadIdx.x;

    if (token_idx >= num_tokens || head_idx >= num_heads) return;

    const int embed_dim = rot_dim / 2;

    for (int i = tid; i < embed_dim; i += blockDim.x) {
        const int q_offset = token_idx * num_heads * head_size + head_idx * head_size;
        const int k_offset = token_idx * num_kv_heads * head_size + (head_idx % num_kv_heads) * head_size;

        int x_idx, y_idx;
        half c, s;

        if (IS_NEOX) {
            x_idx = i;
            y_idx = embed_dim + i;
            c = cos[token_idx * rot_dim + i];
            s = sin[token_idx * rot_dim + i];
        } else {
            x_idx = 2 * i;
            y_idx = 2 * i + 1;
            c = cos[token_idx * rot_dim + i];
            s = sin[token_idx * rot_dim + i];
        }

        float qx = __half2float(query[q_offset + x_idx]);
        float qy = __half2float(query[q_offset + y_idx]);
        float qx_new = qx * __half2float(c) - qy * __half2float(s);
        float qy_new = qy * __half2float(c) + qx * __half2float(s);
        out_query[q_offset + x_idx] = __float2half(qx_new);
        out_query[q_offset + y_idx] = __float2half(qy_new);

        if (head_idx < num_kv_heads) {
            float kx = __half2float(key[k_offset + x_idx]);
            float ky = __half2float(key[k_offset + y_idx]);
            float kx_new = kx * __half2float(c) - ky * __half2float(s);
            float ky_new = ky * __half2float(c) + kx * __half2float(s);
            out_key[k_offset + x_idx] = __float2half(kx_new);
            out_key[k_offset + y_idx] = __float2half(ky_new);
        }
    }
}

    torch::Tensor query,
    torch::Tensor key,
    torch::Tensor cos,
    torch::Tensor sin,
    torch::Tensor out_query,
    torch::Tensor out_key,
