import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_fp16.h>
#include <mma.h>

using namespace nvcuda;

#ifndef CHECK_CUDA
#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be a CUDA tensor")
#endif
#ifndef CHECK_CONTIGUOUS
#define CHECK_CONTIGUOUS(x) TORCH_CHECK(x.is_contiguous(), #x " must be contiguous")

// Computes C[M,N] = A[M,K] * W[N,K]^T for half inputs, half output.
// Specialised for the benchmark shape M=16, K=N=4096, but keeps M/N/K dynamic.
// One block computes one 16x16 output tile.  Eight warps split the K dimension,
// which gives enough independent CTAs to keep H100 occupied for the skinny-M GEMM.
__global__ __launch_bounds__(256, 2)
void h100_m16_gemm_kernel(const half* __restrict__ A,
                          const half* __restrict__ W,
                          half* __restrict__ C,
                          int M, int N, int K) {
    constexpr int WM = 16;
    constexpr int WN = 16;
    constexpr int WK = 16;
    constexpr int WARPS = 8;

    const int tid = threadIdx.x;
    const int warp = tid >> 5;
    const int lane = tid & 31;
    const int n0 = blockIdx.x * WN;
    const int m0 = blockIdx.y * WM;

    extern __shared__ float smem[]; // WARPS * 16 * 16 floats

    wmma::fragment<wmma::matrix_a, WM, WN, WK, half, wmma::row_major> a_frag;
    // Logical B is W^T (K x N).  Since W is row-major [N,K], a tile W[n,k]
    // is exactly a column-major tile of logical B with leading dimension K.
    wmma::fragment<wmma::matrix_b, WM, WN, WK, half, wmma::col_major> b_frag;
    wmma::fragment<wmma::accumulator, WM, WN, WK, float> acc;
    wmma::fill_fragment(acc, 0.0f);

    // Fast path for full tiles.  The benchmark always takes this path.
    if (m0 + 15 < M && n0 + 15 < N && (K & 15) == 0) {
        for (int k0 = warp * WK; k0 < K; k0 += WARPS * WK) {
            const half* Ap = A + m0 * K + k0;
            const half* Wp = W + n0 * K + k0;
            wmma::load_matrix_sync(a_frag, Ap, K);
            wmma::load_matrix_sync(b_frag, Wp, K);
            wmma::mma_sync(acc, a_frag, b_frag, acc);
        }

        float* part = smem + warp * WM * WN;
        wmma::store_matrix_sync(part, acc, WN, wmma::mem_row_major);
        __syncthreads();

        // 256 threads reduce the 8 partial sums for the 256 elements of the tile.
        if (tid < WM * WN) {
            float s = 0.0f;
            #pragma unroll
            for (int w = 0; w < WARPS; ++w) {
                s += smem[w * WM * WN + tid];
            }
            int mi = tid >> 4;
            int ni = tid & 15;
            C[(m0 + mi) * N + (n0 + ni)] = __float2half_rn(s);
        }
    } else {
        // General edge path.  Uses scalar accumulation; not performance critical.
        for (int idx = tid; idx < WM * WN; idx += blockDim.x) {
            int mi = idx >> 4;
            int ni = idx & 15;
            int m = m0 + mi;
            int n = n0 + ni;
            if (m < M && n < N) {
                float s = 0.0f;
                for (int k = 0; k < K; ++k) {
                    s += __half2float(A[m * K + k]) * __half2float(W[n * K + k]);
                }
                C[m * N + n] = __float2half_rn(s);
            }
        }
    }
}

torch::Tensor h100_m16_gemm(torch::Tensor A, torch::Tensor W) {
    CHECK_CUDA(A); CHECK_CUDA(W);
    CHECK_CONTIGUOUS(A); CHECK_CONTIGUOUS(W);
    TORCH_CHECK(A.scalar_type() == at::kHalf, "A must be float16");
    TORCH_CHECK(W.scalar_type() == at::kHalf, "W must be float16");
    TORCH_CHECK(A.dim() == 2 && W.dim() == 2, "expected rank-2 tensors");
    int64_t M64 = A.size(0), K64 = A.size(1), N64 = W.size(0);
    TORCH_CHECK(W.size(1) == K64, "weight shape must be [N,K]");
    TORCH_CHECK(M64 <= INT_MAX && N64 <= INT_MAX && K64 <= INT_MAX, "shape too large");
    auto C = torch::empty({M64, N64}, A.options());
    int M = (int)M64, N = (int)N64, K = (int)K64;
    dim3 block(256);
    dim3 grid((N + 15) / 16, (M + 15) / 16);
    size_t shmem = 8 * 16 * 16 * sizeof(float);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    h100_m16_gemm_kernel<<<grid, block, shmem, stream>>>(
        reinterpret_cast<const half*>(A.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(W.data_ptr<at::Half>()),
        reinterpret_cast<half*>(C.data_ptr<at::Half>()),
        M, N, K);
    return C;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {

}
