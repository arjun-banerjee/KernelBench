        fill_fragment(acc[i], 0);
    }

    int num_k_tiles = K / K_TILE;

    for (int k_tile = 0; k_tile < num_k_tiles; ++k_tile) {
        int k_offset = k_tile * K_TILE;
        int tid      = threadIdx.x;
        int nthreads = blockDim.x;

        /* Load A tile (M_TILE x K_TILE) into shared memory */
        int total_a = (M_TILE * K_TILE) / 16;
        #pragma unroll
        for (int idx = tid; idx < total_a; idx += nthreads) {
            int m = (idx * 16) / K_TILE;
            int k = (idx * 16) % K_TILE;
            const int4* src = reinterpret_cast<const int4*>(
                A + (block_m * M_TILE + m) * K + k_offset + k);
            int4* dst = reinterpret_cast<int4*>(&A_smem[m][k]);
            *dst = *src;
        }

        /* Load B tile (N_TILE x K_TILE) into shared memory.
           B is stored as N x K row-major, which is K x N column-major.
           B_smem[n][k] corresponds to element (n,k) in the N x K view. */
        int total_b = (N_TILE * K_TILE) / 16;
        #pragma unroll
        for (int idx = tid; idx < total_b; idx += nthreads) {
            int n = (idx * 16) / K_TILE;
            int k = (idx * 16) % K_TILE;
            const int4* src = reinterpret_cast<const int4*>(
                B + (block_n * N_TILE + n) * K + k_offset + k);
            int4* dst = reinterpret_cast<int4*>(&B_smem[n][k]);
            *dst = *src;
        }

        __syncthreads();

        /* Compute on the shared-memory tiles */
        #pragma unroll
        for (int k_step = 0; k_step < K_TILE; k_step += WMMA_K) {
            fragment<matrix_b, WMMA_M, WMMA_N, WMMA_K, signed char, col_major> b_frag;
            load_matrix_sync(b_frag,
                             (const signed char*)&B_smem[n_start][k_step],
                             K_TILE);

            #pragma unroll
            for (int m_frag = 0; m_frag < 4; ++m_frag) {
                int m = m_start + m_frag * WMMA_M;
                fragment<matrix_a, WMMA_M, WMMA_N, WMMA_K, signed char, row_major> a_frag;
                load_matrix_sync(a_frag,
                                 (const signed char*)&A_smem[m][k_step],
                                 K_TILE);
                mma_sync(acc[m_frag], a_frag, b_frag, acc[m_frag]);
            }
        }

        __syncthreads();
    }

    /* Store accumulator tiles to global memory */
    #pragma unroll
    for (int m_frag = 0; m_frag < 4; ++m_frag) {
        int m = block_m * M_TILE + m_start + m_frag * WMMA_M;
        int n = block_n * N_TILE + n_start;
        store_matrix_sync(D + m * N + n, acc[m_frag], N, mem_row_major);
    }
}

torch::Tensor forward_cuda(torch::Tensor A, torch::Tensor B, int M, int N, int K) {
    auto D = torch::empty({M, N},
                          torch::TensorOptions().dtype(torch::kInt32).device(A.device()));
    dim3 block(256);
    dim3 grid(N / N_TILE, M / M_TILE);
    gemm_int8<<<grid, block>>>(
        A.data_ptr<int8_t>(),
        B.data_ptr<int8_t>(),
        D.data_ptr<int32_t>(),
        M, N, K);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    return D;
}
