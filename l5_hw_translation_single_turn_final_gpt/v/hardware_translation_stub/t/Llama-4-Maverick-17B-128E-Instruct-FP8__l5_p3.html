template <typename scalar_t, int VEC_SIZE>
__global__ void rms_norm_kernel(
    scalar_t* __restrict__ out,           // [..., hidden_size]
    const scalar_t* __restrict__ input,   // [..., hidden_size]
    const scalar_t* __restrict__ weight,  // [hidden_size]
    const float epsilon, const int num_tokens, const int hidden_size) {
    __shared__ float s_variance;
    float variance = 0.0f;
    const int tid = threadIdx.x;
    const int block_size = blockDim.x;
    const int row_id = blockIdx.x;
    const scalar_t* input_row = input + row_id * hidden_size;
    scalar_t* out_row = out + row_id * hidden_size;

    auto vec_op = [&variance](const float4& vec) {
        float4 temp = vec;
        variance += temp.x * temp.x + temp.y * temp.y + temp.z * temp.z + temp.w * temp.w;
    };
    auto scalar_op = [&variance](const float& val) {
        variance += val * val;
    };

    // Vectorized load and reduction
    for (int i = tid; i < hidden_size / VEC_SIZE; i += block_size) {
        float4 vec = *reinterpret_cast<const float4*>(input_row + i * VEC_SIZE);
        vec_op(vec);
    }
    // Handle remaining elements
    for (int i = (hidden_size / VEC_SIZE) * VEC_SIZE + tid; i < hidden_size; i += block_size) {
        float val = static_cast<float>(input_row[i]);
        scalar_op(val);
    }

    using BlockReduce = cub::BlockReduce<float, 256>;
    __shared__ typename BlockReduce::TempStorage reduceStore;
    variance = BlockReduce(reduceStore).Reduce(variance, cub::Sum(), block_size);

    if (tid == 0) {
        s_variance = rsqrtf(variance / hidden_size + epsilon);
    }

    // Apply RMS Norm

template <typename scalar_t, int width>
__global__ void fused_add_rms_norm_kernel(
    scalar_t* __restrict__ input,  // [..., hidden_size]
    scalar_t* __restrict__ residual,      // [..., hidden_size]
    const scalar_t* __restrict__ weight,  // [hidden_size]

    const int block_size = blockDim.x;
    const int row_id = blockIdx.x;

    float variance = 0.0f;

    auto* input_v = reinterpret_cast<float4*>(input + row_id * hidden_size);
    auto* residual_v = reinterpret_cast<float4*>(residual + row_id * hidden_size);
    auto* weight_v = reinterpret_cast<const float4*>(weight);

    // Vectorized addition and reduction

        sum_vec.y = __hadd(input_vec.y, residual_vec.y);

    torch::Tensor& input,
    torch::Tensor& weight,
    float epsilon) {
    const auto num_tokens = input.size(0);
    const auto hidden_size = input.size(1);
    auto output = torch::empty_like(input);

    const int block_size = 256;
    const int VEC_SIZE = 4; // For float4 vectorization
    dim3 grid(num_tokens);
    dim3 block(block_size);

    AT_DISPATCH_FLOATING_TYPES_AND2(
        at::ScalarType::Half, at::ScalarType::BFloat16,
        input.scalar_type(), "rms_norm_kernel", [&] {
            vllm::rms_norm_kernel<scalar_t, VEC_SIZE><<<grid, block>>>(
                output.data_ptr<scalar_t>(),
                input.data_ptr<scalar_t>(),
