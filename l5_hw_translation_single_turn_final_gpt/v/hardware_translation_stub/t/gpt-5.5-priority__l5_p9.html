            scores[idx] = __expf(scores[idx] - newm_s[mi]);
        }
        // Rescale previous output accumulator by exp(old_m - new_m).
        for (int idx = tid; idx < BM * D; idx += blockDim.x) {
            int mi = idx / D;
            float alpha = __expf(m_s[mi] - newm_s[mi]);
            o_s[idx] *= alpha;
        }
        __syncthreads();

        // O += P V for this key tile.
        for (int idx = tid; idx < BM * D; idx += blockDim.x) {
            int mi = idx / D;
            int di = idx - mi * D;
            float acc = 0.0f;
            #pragma unroll
            for (int ni = 0; ni < BN; ++ni) {
                acc = fmaf(scores[mi * BN + ni], __bfloat162float(v_s[ni * D + di]), acc);
            }
            o_s[idx] += acc;
        }
        __syncthreads();

        if (tid < BM) m_s[tid] = newm_s[tid];
        __syncthreads();
    }

    // Normalize by softmax denominator and store BF16 output.
    for (int idx = tid; idx < BM * D; idx += blockDim.x) {
        int mi = idx / D;
        int di = idx - mi * D;
        int row = q_start + mi;
        if (row < N) {
            float val = o_s[idx] / l_s[mi];
            out[o_base + (long)row * D + di] = __float2bfloat16(val);
        }
    }
}
