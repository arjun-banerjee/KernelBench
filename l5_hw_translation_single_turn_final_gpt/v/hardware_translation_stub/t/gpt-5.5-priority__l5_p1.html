import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_cuda_src = r'''
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <math_constants.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

static __device__ __forceinline__ float warp_reduce_sum(float v) {
    #pragma unroll
    for (int mask = 16; mask > 0; mask >>= 1) v += __shfl_down_sync(0xffffffff, v, mask);
    return v;
}

static __device__ __forceinline__ float warp_reduce_max(float v) {
    #pragma unroll
    for (int mask = 16; mask > 0; mask >>= 1) v = fmaxf(v, __shfl_down_sync(0xffffffff, v, mask));
    return v;
}

static __device__ __forceinline__ float block_reduce_sum(float v) {
    __shared__ float smem[8];
    int lane = threadIdx.x & 31;
    int wid = threadIdx.x >> 5;
    v = warp_reduce_sum(v);
    if (lane == 0) smem[wid] = v;
    __syncthreads();
    v = (threadIdx.x < (blockDim.x >> 5)) ? smem[lane] : 0.0f;
    if (wid == 0) v = warp_reduce_sum(v);
    return v;
}

static __device__ __forceinline__ float block_reduce_max(float v) {
    __shared__ float smem[8];
    int lane = threadIdx.x & 31;
    int wid = threadIdx.x >> 5;
    v = warp_reduce_max(v);
    if (lane == 0) smem[wid] = v;
    __syncthreads();
    v = (threadIdx.x < (blockDim.x >> 5)) ? smem[lane] : -CUDART_INF_F;
    if (wid == 0) v = warp_reduce_max(v);
    return v;
}

// One CTA computes one (batch, query-head, query-position) row.
// Fixed benchmark dimensions: B=4, Hq=32, Hkv=8, S=512, D=128.
__global__ __launch_bounds__(128, 2)
void gqa_attention_fwd_kernel(const half* __restrict__ q,
                              const half* __restrict__ k,
                              const half* __restrict__ v,
                              half* __restrict__ out) {
    constexpr int B = 4;
    constexpr int H = 32;
    constexpr int HKV = 8;
    constexpr int S = 512;
    constexpr int D = 128;
    constexpr float SCALE = 0.08838834764831845f; // 1/sqrt(128)

    const int row = blockIdx.x;
    const int s_q = row & (S - 1);
    const int h = (row >> 9) & 31;
    const int b = row >> 14;
    const int hk = h >> 2;
    const int tid = threadIdx.x;

    __shared__ float logits[S];

    const half* q_ptr = q + (((b * H + h) * S + s_q) * D);
    const half* k_base = k + ((b * HKV + hk) * S * D);

    // Compute all q*k logits. With D==blockDim.x, each thread owns one channel.
    float qv = __half2float(q_ptr[tid]);
    #pragma unroll 1
    for (int sk = 0; sk < S; ++sk) {
        float prod = qv * __half2float(k_base[sk * D + tid]);
        float sum = block_reduce_sum(prod);
        if (tid == 0) logits[sk] = sum * SCALE;
        __syncthreads();
    }

    // Softmax over 512 keys.
    float local_m = -CUDART_INF_F;
    for (int i = tid; i < S; i += 128) local_m = fmaxf(local_m, logits[i]);
    float m = block_reduce_max(local_m);
    __syncthreads();

    float local_s = 0.0f;
    for (int i = tid; i < S; i += 128) {
        float e = __expf(logits[i] - m);
        logits[i] = e;
        local_s += e;
    }
    float denom = block_reduce_sum(local_s);
    float inv_denom = 1.0f / denom;
    __syncthreads();

    const half* v_base = v + ((b * HKV + hk) * S * D);
    float acc = 0.0f;
    #pragma unroll 1
    for (int sk = 0; sk < S; ++sk) {
        acc += (logits[sk] * inv_denom) * __half2float(v_base[sk * D + tid]);
    }
    out[((b * H + h) * S + s_q) * D + tid] = __float2half_rn(acc);
}

torch::Tensor gqa_attention(torch::Tensor q, torch::Tensor k, torch::Tensor v) {
    auto out = torch::empty_like(q);
    const int rows = q.size(0) * q.size(1) * q.size(2);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    gqa_attention_fwd_kernel<<<rows, 128, 0, stream>>>(
        reinterpret_cast<const half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(v.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>())
    );
    return out;
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("gqa_attention", &gqa_attention, "H100 optimized grouped-query attention forward (fp16)");
}
