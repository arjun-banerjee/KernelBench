__inline__ __device__ float warp_sum(float v) {
    #pragma unroll
    for (int offset = 16; offset > 0; offset >>= 1) {
        v += __shfl_down_sync(0xffffffff, v, offset);
    }

__inline__ __device__ float block_sum_256(float v) {
    __shared__ float smem[8];
    int lane = threadIdx.x & 31;
    int wid = threadIdx.x >> 5;
    v = warp_sum(v);
    if (lane == 0) smem[wid] = v;
    __syncthreads();
    v = (threadIdx.x < 8) ? smem[lane] : 0.0f;
    if (wid == 0) v = warp_sum(v);
    return v;
}

// Specialized Hopper-friendly RMSNorm for [512,4096] fp16, contiguous.
// One CTA per token. 256 threads: each thread processes exactly two 16-byte vectors.
__global__ __launch_bounds__(256, 2)

                             const half* __restrict__ w,
                             half* __restrict__ y,
                             float eps) {

    constexpr int NVEC = H / V;   // 512
    const int row = blockIdx.x;
    const int tid = threadIdx.x;

    const uint4* __restrict__ xv = reinterpret_cast<const uint4*>(x + row * H);
    const uint4* __restrict__ wv = reinterpret_cast<const uint4*>(w);
    uint4* __restrict__ yv = reinterpret_cast<uint4*>(y + row * H);

    float ss = 0.0f;

    // idx = tid and tid+256 cover all 512 vectors, coalesced 16B/thread loads.
    #pragma unroll
    for (int it = 0; it < 2; ++it) {
        int idx = tid + it * 256;
        uint4 a = xv[idx];
        half2 h0 = *reinterpret_cast<half2*>(&a.x);
        half2 h1 = *reinterpret_cast<half2*>(&a.y);
        half2 h2 = *reinterpret_cast<half2*>(&a.z);

// Generic fp16 fallback for same operation when the benchmark harness varies token count.
__global__ __launch_bounds__(256, 2)
void rmsnorm_f16_kernel(const half* __restrict__ x,
                        const half* __restrict__ w,
                        half* __restrict__ y,
                        int hidden,
                        float eps) {
    int row = blockIdx.x;
    int tid = threadIdx.x;
    const half* xr = x + (long long)row * hidden;
    half* yr = y + (long long)row * hidden;
    float ss = 0.0f;
    for (int i = tid; i < hidden; i += 256) {
        float v = __half2float(xr[i]);
        ss = fmaf(v, v, ss);
    }
    ss = block_sum_256(ss);
    __shared__ float s_scale;
    if (tid == 0) s_scale = rsqrtf(ss / (float)hidden + eps);
    __syncthreads();
    float scale = s_scale;
    for (int i = tid; i < hidden; i += 256) {
        float v = __half2float(xr[i]);
        float ww = __half2float(w[i]);
        yr[i] = __float2half_rn(v * scale * ww);
    }
}

    TORCH_CHECK(x.is_cuda(), "x must be CUDA");
    TORCH_CHECK(weight.is_cuda(), "weight must be CUDA");
    TORCH_CHECK(x.scalar_type() == at::kHalf, "x must be fp16");
    TORCH_CHECK(weight.scalar_type() == at::kHalf, "weight must be fp16");
    TORCH_CHECK(x.dim() == 2, "x must be 2D");

    const int hidden = (int)x.size(1);

    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    const half* xp = reinterpret_cast<const half*>(x.data_ptr<at::Half>());
