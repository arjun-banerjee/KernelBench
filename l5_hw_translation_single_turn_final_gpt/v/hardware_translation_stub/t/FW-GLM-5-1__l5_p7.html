    const scalar_t* __restrict__ scales,
    const scalar_t* __restrict__ zeros,
    int prob_m, int prob_n, int prob_k,
    int group_size, int num_groups
) {
    constexpr int TILE_M = 16;
    constexpr int TILE_N = 16 * THREAD_N_BLOCKS;
    constexpr int TILE_K = 16 * THREAD_K_BLOCKS;
    constexpr int NUM_THREADS = THREAD_N_BLOCKS >= 8 ? 256 : 128;
    
    constexpr int SH_A_SIZE = STAGES * TILE_M * TILE_K * sizeof(scalar_t);
    constexpr int SH_B_SIZE = STAGES * TILE_K * TILE_N / 8 * sizeof(uint32_t);
    constexpr int SH_S_SIZE = STAGES * (TILE_K / group_size) * TILE_N * sizeof(scalar_t);
    constexpr int SH_C_SIZE = TILE_M * TILE_N * sizeof(float);
    
    extern __shared__ uint8_t shmem[];
    
    scalar_t* sh_A = reinterpret_cast<scalar_t*>(shmem);
    uint32_t* sh_B = reinterpret_cast<uint32_t*>(shmem + SH_A_SIZE);
    scalar_t* sh_S = reinterpret_cast<scalar_t*>(shmem + SH_A_SIZE + SH_B_SIZE);
    float* sh_C = reinterpret_cast<float*>(shmem + SH_A_SIZE + SH_B_SIZE + SH_S_SIZE);
    
    const int block_m = blockIdx.y * TILE_M;
    const int block_n = blockIdx.x * TILE_N;
    
    if (block_m >= prob_m || block_n >= prob_n) return;
    
    // Initialize accumulator
    float acc[TILE_M][TILE_N / 16] = {{0}};
    
    const int warp_id = threadIdx.x / 32;
    const int lane_id = threadIdx.x % 32;
    
    // Main GEMM loop with pipelining
    for (int k_base = 0; k_base < prob_k; k_base += TILE_K) {
        // Load A tile
        for (int m = threadIdx.x; m < TILE_M && (block_m + m) < prob_m; m += NUM_THREADS) {
            for (int k = 0; k < TILE_K && (k_base + k) < prob_k; k += 1) {
                int a_idx = (block_m + m) * prob_k + k_base + k;
                sh_A[(k_base / TILE_K) * TILE_M * TILE_K + m * TILE_K + k] = A[a_idx];
            }
        }
        
        // Load B tile (quantized)
        for (int k = threadIdx.x; k < TILE_K / 2 && (k_base + k * 2) < prob_k; k += NUM_THREADS) {
            for (int n = 0; n < TILE_N / PACK_FACTOR_4BIT && (block_n + n * PACK_FACTOR_4BIT) < prob_n; n++) {
                int b_idx = (k_base + k * 2) / 2 * prob_n / PACK_FACTOR_4BIT + (block_n / PACK_FACTOR_4BIT) + n;
                sh_B[(k_base / TILE_K) * TILE_K * TILE_N / 8 + k * TILE_N / 8 + n] = B_qweight[b_idx];
            }
        }
        
        // Load scales
        for (int g = threadIdx.x; g < (TILE_K / group_size) && (k_base / group_size + g) < num_groups; g += NUM_THREADS) {
            for (int n = 0; n < TILE_N && (block_n + n) < prob_n; n++) {
                int s_idx = (k_base / group_size + g) * prob_n + block_n + n;
                sh_S[(k_base / TILE_K) * (TILE_K / group_size) * TILE_N + g * TILE_N + n] = scales[s_idx];
            }
        }
        
        __syncthreads();
        
        // Compute partial GEMM
        for (int k_tile = 0; k_tile < TILE_K && (k_base + k_tile) < prob_k; k_tile += 16) {
            // Load weights and dequantize
            half weight_frag[16][8];
            half a_frag[16][8];
            
            // Dequantize 4-bit weights to FP16
            for (int n_sub = 0; n_sub < 8; n_sub++) {
                for (int k_sub = 0; k_sub < 16; k_sub++) {
                    int packed_idx = (k_tile + k_sub) / 2 * (TILE_N / 8) + (n_sub * 2) / 8;
                    int sub_idx = (k_sub % 2) * 4 + ((n_sub * 2) % 8) / 2;
                    
                    uint32_t packed = sh_B[(k_base / TILE_K) * TILE_K * TILE_N / 8 + packed_idx];
                    uint8_t w4 = (packed >> (sub_idx * 4)) & 0xF;
                    
                    int group_idx = (k_base + k_tile + k_sub) / group_size;
                    half scale = sh_S[(k_base / TILE_K) * (TILE_K / group_size) * TILE_N + group_idx * TILE_N + n_sub * 2];
                    
                    weight_frag[k_sub][n_sub] = __hmul(__short2half_rn((short)w4), scale);
                }
            }
            
            // Load A fragment
            for (int m_sub = 0; m_sub < TILE_M; m_sub += 16) {
                for (int k_sub = 0; k_sub < 16; k_sub++) {
                    a_frag[k_sub][m_sub / 2] = sh_A[(k_base / TILE_K) * TILE_M * TILE_K + m_sub * TILE_K + k_tile + k_sub];
                }
            }
            
            // Compute using tensor core (WMMA)
            using namespace nvcuda::wmma;
            fragment<matrix_a, 16, 16, 16, half, row_major> a_mat;
            fragment<matrix_b, 16, 16, 16, half, col_major> b_mat;
            fragment<accumulator, 16, 16, 16, float> c_mat;
            
            // Load A and B
            load_matrix_sync(a_mat, &sh_A[(k_base / TILE_K) * TILE_M * TILE_K + k_tile], TILE_K);
            load_matrix_sync(b_mat, &weight_frag[0][0], 8);
            fill_fragment(c_mat, 0.0f);
            
            // MMA
            mma_sync(c_mat, a_mat, b_mat, c_mat);
            
            // Store result
            store_matrix_sync(&sh_C[0], c_mat, TILE_N, mem_row_major);
        }
    }
    
    // Write C
    for (int m = threadIdx.x; m < TILE_M && (block_m + m) < prob_m; m += NUM_THREADS) {
        for (int n = 0; n < TILE_N && (block_n + n) < prob_n; n++) {
            int c_idx = (block_m + m) * prob_n + block_n + n;
            C[c_idx] = __float2half_rn(sh_C[m * TILE_N + n]);
        }
    }
}

// Simpler optimized GEMM kernel for H100
template<int BLOCK_M, int BLOCK_N, int BLOCK_K, int STAGES>
__global__ void simple_marlin_gemm_h100(
    const half* __restrict__ A,
    const uint32_t* __restrict__ B_packed,
    half* __restrict__ C,
    const half* __restrict__ scales,
    const half* __restrict__ zeros,
    int M, int N, int K,
    int group_size
) {
    const int bm = blockIdx.y * BLOCK_M;
    const int bn = blockIdx.x * BLOCK_N;
    
    if (bm >= M || bn >= N) return;
    
    const int tid = threadIdx.x;
    const int warp_id = tid / 32;
    const int lane_id = tid % 32;
    
    // Thread tile assignment
    const int thread_m = warp_id / 2;
    const int thread_n = (warp_id % 2) * 8 + lane_id / 4;
    
    // Accumulator
    float sum = 0.0f;
    
    // Process K in tiles
    for (int bk = 0; bk < K; bk += BLOCK_K) {
        // Load A and B for this thread
        for (int k_local = 0; k_local < BLOCK_K; k_local += 32) {
            int k = bk + k_local + lane_id;
            if (k >= K) continue;
            
            // Load A element
            int a_m = bm + thread_m;
            half a_val = (a_m < M) ? A[a_m * K + k] : __float2half(0.0f);
            
            // Load and dequantize B element (4-bit packed)
            int b_n = bn + thread_n;
            if (b_n >= N) continue;
            
            int packed_idx = k * (N / 8) + b_n / 8;
            int sub_idx = b_n % 8;
            
            uint32_t packed = B_packed[packed_idx];
            uint8_t w4 = (packed >> (sub_idx * 4)) & 0xF;
            
            // Get scale
            int group_idx = k / group_size;
            half scale = scales[group_idx * N + b_n];
            
            // Dequantize
            half b_val = __hmul(__int2half_rn((int)w4), scale);
            
            // Accumulate
            sum += __half2float(__hmul(a_val, b_val));
        }
    }
    
    // Reduce within warp
    for (int offset = 16; offset > 0; offset /= 2) {
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    }
    
    // Write result
    if (lane_id == 0) {
        int c_m = bm + thread_m;
        int c_n = bn + thread_n;
        if (c_m < M && c_n < N) {
            C[c_m * N + c_n] = __float2half_rn(sum);
        }
    }
}

// Reference implementation for correctness
__global__ void marlin_gemm_reference(
    const half* __restrict__ A,
    const uint32_t* __restrict__ B_packed,
    half* __restrict__ C,
    const half* __restrict__ scales,
    const half* __restrict__ zeros,
    int M, int N, int K,
    int group_size
) {
    const int m = blockIdx.y * blockDim.y + threadIdx.y;
    const int n = blockIdx.x * blockDim.x + threadIdx.x;
    
    if (m >= M || n >= N) return;
    
    float sum = 0.0f;
    
    for (int k = 0; k < K; k++) {
        half a_val = A[m * K + k];
        
        // Dequantize B
        int packed_idx = k * (N / 8) + n / 8;
        int sub_idx = n % 8;
        uint32_t packed = B_packed[packed_idx];
        uint8_t w4 = (packed >> (sub_idx * 4)) & 0xF;
        
        int group_idx = k / group_size;
        half scale = scales[group_idx * N + n];
        half zero = zeros ? zeros[group_idx * N + n] : __float2half(0.0f);
        
        half b_val = __hsub(__hmul(__int2half_rn((int)w4), scale), zero);
        
        sum += __half2float(__hmul(a_val, b_val));
    }
    
    C[m * N + n] = __float2half_rn(sum);
}

// Optimized kernel with shared memory and cooperative loading
template<int BLOCK_M, int BLOCK_N, int BLOCK_K, int NUM_THREADS>
__launch_bounds__(NUM_THREADS)
__global__ void marlin_gemm_optimized_h100(
    const half* __restrict__ A,
    const uint32_t* __restrict__ B_packed,
    half* __restrict__ C,
    const half* __restrict__ scales,
    const half* __restrict__ zeros,
    int M, int N, int K,
    int group_size
) {
    __shared__ half sh_A[BLOCK_K][BLOCK_M];
    __shared__ uint32_t sh_B[BLOCK_K][BLOCK_N / 8];
    __shared__ half sh_S[BLOCK_K / 128 * BLOCK_N];
    
    const int bm = blockIdx.y * BLOCK_M;
    const int bn = blockIdx.x * BLOCK_N;
    
    if (bm >= M || bn >= N) return;
    
    const int tid = threadIdx.x;
    const int warp_id = tid / 32;
    const int lane_id = tid % 32;
    
    // Each thread handles multiple output elements
    constexpr int THREAD_M = BLOCK_M / (NUM_THREADS / 32);
    constexpr int THREAD_N = 8;
    
    float accum[THREAD_M][THREAD_N / 4] = {{0}};
    
    for (int bk = 0; bk < K; bk += BLOCK_K) {
        // Cooperative load of A
        for (int i = tid; i < BLOCK_M * BLOCK_K; i += NUM_THREADS) {
            int local_m = i / BLOCK_K;
            int local_k = i % BLOCK_K;
            int global_m = bm + local_m;
            int global_k = bk + local_k;
            sh_A[local_k][local_m] = (global_m < M && global_k < K) ? 
                A[global_m * K + global_k] : __float2half(0.0f);
        }
        
        // Cooperative load of B
        for (int i = tid; i < BLOCK_K * (BLOCK_N / 8); i += NUM_THREADS) {
            int local_k = i / (BLOCK_N / 8);
            int local_n = i % (BLOCK_N / 8);
            int global_k = bk + local_k;
            int global_n_packed = bn / 8 + local_n;
            if (global_k < K && global_n_packed < N / 8) {
                sh_B[local_k][local_n] = B_packed[global_k * (N / 8) + global_n_packed];
            }
        }
        
        // Load scales for this K block
        for (int i = tid; i < (BLOCK_K / group_size) * BLOCK_N; i += NUM_THREADS) {
            int local_g = i / BLOCK_N;
            int local_n = i % BLOCK_N;
            int global_g = (bk / group_size) + local_g;
            int global_n = bn + local_n;
            if (global_n < N) {
                sh_S[local_g * BLOCK_N + local_n] = scales[global_g * N + global_n];
            }
        }
        
        __syncthreads();
        
        // Compute
        for (int k_local = 0; k_local < BLOCK_K; k_local++) {
            half scale_local[THREAD_N / 4];
            half b_local[THREAD_N];
            
            int group_local = k_local / group_size;
            
            // Dequantize weights
            for (int tn = 0; tn < THREAD_N; tn++) {
                int n_local = lane_id * 4 + tn;
                if (n_local < BLOCK_N) {
                    uint32_t packed = sh_B[k_local][n_local / 8];
                    int sub_idx = n_local % 8;
                    uint8_t w4 = (packed >> (sub_idx * 4)) & 0xF;
                    half s = sh_S[group_local * BLOCK_N + n_local];
                    b_local[tn] = __hmul(__int2half_rn((int)w4), s);
                }
            }
            
            // Matrix multiply for this thread's tile
            for (int tm = 0; tm < THREAD_M; tm++) {
                int m_local = warp_id * THREAD_M + tm;
                half a_val = sh_A[k_local][m_local];
                
                for (int tn = 0; tn < THREAD_N / 4; tn++) {
                    accum[tm][tn] += __half2float(__hmul(a_val, b_local[tn * 4]));
                    accum[tm][tn] += __half2float(__hmul(a_val, b_local[tn * 4 + 1]));
                    accum[tm][tn] += __half2float(__hmul(a_val, b_local[tn * 4 + 2]));
                    accum[tm][tn] += __half2float(__hmul(a_val, b_local[tn * 4 + 3]));
                }
            }
        }
    }
    
    // Write C
    for (int tm = 0; tm < THREAD_M; tm++) {
        int m_global = bm + warp_id * THREAD_M + tm;
        if (m_global >= M) continue;
        
        for (int tn = 0; tn < THREAD_N / 4; tn++) {
            int n_global = bn + lane_id * 4 + tn * 4;
            if (n_global >= N) continue;
            
            C[m_global * N + n_global] = __float2half_rn(accum[tm][tn]);
        }
    }
}
