template <typename T>
__device__ inline T add_elem(T a, T b);

template <>
__device__ inline half add_elem<half>(half a, half b) {
    return __hadd(a, b);
}

template <>
__device__ inline float add_elem<float>(float a, float b) {
    return a + b;

__device__ inline nv_bfloat16 add_elem<nv_bfloat16>(nv_bfloat16 a, nv_bfloat16 b) {

template <typename T>
__global__ void __launch_bounds__(512, 2)
allreduce_kernel(
    Signal* peer_sigs[8],
    const T* const* __restrict__ peer_bufs,
    T* __restrict__ output,
    int numel,
    int rank,
    int world_size,
    int step) {

    // Signal readiness to all peers (including self for uniformity)
    if (threadIdx.x == 0) {
        #pragma unroll
        for (int i = 0; i < world_size; ++i) {
            atomicAdd((unsigned int*)&peer_sigs[i]->flags[rank], 1);
        }
    }
    __threadfence_system();

    // Wait until all peers have signalled readiness for this step
    if (threadIdx.x == 0) {
        #pragma unroll
        for (int i = 0; i < world_size; ++i) {
            while (atomicAdd((unsigned int*)&peer_sigs[rank]->flags[i], 0) < (unsigned int)step) {
                // spin-wait
            }
        }
    }
    __syncthreads();

    int tid = blockIdx.x * blockDim.x + threadIdx.x;
    int nthreads = blockDim.x * gridDim.x;

    if (sizeof(T) == 2) {
        // fp16 / bf16: vectorize 128 bits = 8 elements per thread
        const int VEC = 8;
        using Vec = uint4;
        const Vec* peers[8];
        #pragma unroll
        for (int i = 0; i < world_size; ++i) {
            peers[i] = reinterpret_cast<const Vec*>(peer_bufs[i]);
        }
        Vec* out_vec = reinterpret_cast<Vec*>(output);
        int nvec = numel / VEC;

        #pragma unroll 1
        for (int idx = tid; idx < nvec; idx += nthreads) {
            Vec v0 = peers[0][idx];
            half2 a0 = reinterpret_cast<const half2*>(&v0)[0];
            half2 a1 = reinterpret_cast<const half2*>(&v0)[1];
            half2 a2 = reinterpret_cast<const half2*>(&v0)[2];
            half2 a3 = reinterpret_cast<const half2*>(&v0)[3];

            #pragma unroll
            for (int r = 1; r < world_size; ++r) {
                Vec vr = peers[r][idx];
                half2 b0 = reinterpret_cast<const half2*>(&vr)[0];
                half2 b1 = reinterpret_cast<const half2*>(&vr)[1];
                half2 b2 = reinterpret_cast<const half2*>(&vr)[2];
                half2 b3 = reinterpret_cast<const half2*>(&vr)[3];
                a0 = __hadd2(a0, b0);
                a1 = __hadd2(a1, b1);
                a2 = __hadd2(a2, b2);
                a3 = __hadd2(a3, b3);
            }

            Vec res;
            reinterpret_cast<half2*>(&res)[0] = a0;
            reinterpret_cast<half2*>(&res)[1] = a1;
            reinterpret_cast<half2*>(&res)[2] = a2;
            reinterpret_cast<half2*>(&res)[3] = a3;
            out_vec[idx] = res;
        }

        // Tail scalar loop
        for (int idx = nvec * VEC + tid; idx < numel; idx += nthreads) {
            T sum = peer_bufs[0][idx];
            #pragma unroll
            for (int r = 1; r < world_size; ++r) {
                sum = add_elem(sum, peer_bufs[r][idx]);
            }
            output[idx] = sum;
        }
    } else {
        // fp32: vectorize 128 bits = 4 floats per thread
        const int VEC = 4;
        using Vec = float4;
        const Vec* peers[8];
        #pragma unroll
        for (int i = 0; i < world_size; ++i) {
            peers[i] = reinterpret_cast<const Vec*>(peer_bufs[i]);
        }
        Vec* out_vec = reinterpret_cast<Vec*>(output);
        int nvec = numel / VEC;

        #pragma unroll 1
        for (int idx = tid; idx < nvec; idx += nthreads) {
            Vec sum = peers[0][idx];
            #pragma unroll
            for (int r = 1; r < world_size; ++r) {
                Vec val = peers[r][idx];
                sum.x += val.x;
                sum.y += val.y;
                sum.z += val.z;
                sum.w += val.w;
            }
            out_vec[idx] = sum;
        }

        for (int idx = nvec * VEC + tid; idx < numel; idx += nthreads) {
            float sum = peer_bufs[0][idx];
            #pragma unroll
            for (int r = 1; r < world_size; ++r) {
                sum += peer_bufs[r][idx];
            }
            output[idx] = sum;
        }
    }
}

class CustomAllreduce {
public:
    int rank_;
    int world_size_;
    Signal* peer_sigs_[8];
    void** d_peer_bufs_;
    int step_counter_;

    CustomAllreduce(Signal* ipc_ptrs[8], void* /*rank_data*/, int64_t /*rank_data_size*/,
                    int rank, int world_size, bool /*fully_connected*/)
        : rank_(rank), world_size_(world_size), step_counter_(1) {
        for (int i = 0; i < world_size; ++i) {
            peer_sigs_[i] = ipc_ptrs[i];
        }
        cudaMalloc(&d_peer_bufs_, 8 * sizeof(void*));
    }

    ~CustomAllreduce() {
        cudaFree(d_peer_bufs_);
    }

    void register_buffer(void* fake_ipc_ptrs[8]) {
        cudaMemcpy(d_peer_bufs_, fake_ipc_ptrs, 8 * sizeof(void*), cudaMemcpyHostToDevice);
    }

    template <typename T>
    void allreduce(cudaStream_t stream, T* input, T* output, int numel) {
        void* local = reinterpret_cast<void*>(input);
        cudaMemcpyAsync(d_peer_bufs_ + rank_, &local, sizeof(void*), cudaMemcpyHostToDevice, stream);

        // Saturate H100 SMs: at least 132 blocks, at most 1024
        int block_size = 512;
        int grid_size = (numel + block_size * 8 - 1) / (block_size * 8);
        if (grid_size < 132) grid_size = 132;
        if (grid_size > 1024) grid_size = 1024;

        allreduce_kernel<T><<<grid_size, block_size, 0, stream>>>(
            peer_sigs_, reinterpret_cast<const T* const*>(d_peer_bufs_),
            output, numel, rank_, world_size_, step_counter_);
        step_counter_++;
    }

    std::pair<std::vector<uint8_t>, std::vector<int64_t>> get_graph_buffer_ipc_meta() {
        return {{}, {}};
    }

    void register_graph_buffers(const std::vector<std::string>& /*handles*/,
                                const std::vector<std::vector<int64_t>>& /*offsets*/) {}
};

bool _is_weak_contiguous(torch::Tensor& t) {
    return t.is_contiguous() ||
           (t.storage().nbytes() - t.storage_offset() * t.element_size() ==
            t.numel() * t.element_size());
}

void all_reduce(fptr_t _fa, torch::Tensor& inp, torch::Tensor& out,
                fptr_t _reg_buffer, int64_t reg_buffer_sz_bytes) {
    auto fa = reinterpret_cast<CustomAllreduce*>(_fa);
    const at::cuda::OptionalCUDAGuard device_guard(device_of(inp));
    auto stream = c10::cuda::getCurrentCUDAStream().stream();

    TORCH_CHECK_EQ(inp.scalar_type(), out.scalar_type());
    TORCH_CHECK_EQ(inp.numel(), out.numel());
    TORCH_CHECK(_is_weak_contiguous(out));
    TORCH_CHECK(_is_weak_contiguous(inp));
    auto input_size = inp.numel() * inp.element_size();
    auto reg_buffer = reinterpret_cast<void*>(_reg_buffer);
    if (reg_buffer) {
        TORCH_CHECK_LE(input_size, reg_buffer_sz_bytes);
        AT_CUDA_CHECK(cudaMemcpyAsync(reg_buffer, inp.data_ptr(), input_size,
                                      cudaMemcpyDeviceToDevice, stream));
    } else {
        reg_buffer = inp.data_ptr();
    }
    switch (out.scalar_type()) {
        case at::ScalarType::Float: {
            fa->allreduce<float>(stream, reinterpret_cast<float*>(reg_buffer),
                                 reinterpret_cast<float*>(out.data_ptr()), out.numel());
            break;
        }
        case at::ScalarType::Half: {
            fa->allreduce<half>(stream, reinterpret_cast<half*>(reg_buffer),
                                reinterpret_cast<half*>(out.data_ptr()), out.numel());
            break;
        }
        case at::ScalarType::BFloat16: {
            fa->allreduce<nv_bfloat16>(
                stream, reinterpret_cast<nv_bfloat16*>(reg_buffer),
                reinterpret_cast<nv_bfloat16*>(out.data_ptr()), out.numel());
            break;
        }
        default:
            throw std::runtime_error(
                "custom allreduce only supports float32, float16 and bfloat16");
    }
}
