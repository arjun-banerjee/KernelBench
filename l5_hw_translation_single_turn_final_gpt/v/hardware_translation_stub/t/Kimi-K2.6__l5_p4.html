/* ---------- swigluoai (interleaved) kernel ---------- */
template <typename scalar_t, bool use_vec, bool use_256b>
__global__ void swigluoai_and_mul_kernel(
    scalar_t* __restrict__ out,
    const scalar_t* __restrict__ input,
    int64_t num_tokens, int d, float alpha, float limit) {
    int token = blockIdx.x;
    if (token >= num_tokens) return;
    const scalar_t* in_ptr = input + token * 2 * d;
    scalar_t* out_ptr = out + token * d;

    if constexpr (use_vec) {
        constexpr int VEC_BYTES = use_256b ? 32 : 16;
        constexpr int PAIRS = VEC_BYTES / (2 * sizeof(scalar_t));
        using InVec = Vec<scalar_t, VEC_BYTES>;
        struct alignas(VEC_BYTES / 2) OutVec {
            static constexpr int NUM_ELTS = VEC_BYTES / (2 * sizeof(scalar_t));
            scalar_t v[NUM_ELTS];
        };
        const int num_vecs = d / PAIRS;
        const InVec* in_vec = reinterpret_cast<const InVec*>(in_ptr);
        OutVec* out_vec = reinterpret_cast<OutVec*>(out_ptr);
        for (int i = threadIdx.x; i < num_vecs; i += blockDim.x) {
            InVec iv = in_vec[i];
            OutVec ov;
            #pragma unroll
            for (int j = 0; j < PAIRS; ++j) {
                float gate = to_float_impl<scalar_t>(iv.v[j*2]);
                float up   = to_float_impl<scalar_t>(iv.v[j*2+1]);
                float g = fminf(gate, limit);
                float u = fmaxf(fminf(up, limit), -limit);
                float res = (u + 1.0f) * g / (1.0f + expf(-g * alpha));
                ov.v[j] = from_float_impl<scalar_t>(res);
            }
            out_vec[i] = ov;
        }
        #pragma unroll 1
        for (int i = num_vecs * PAIRS + threadIdx.x; i < d; i += blockDim.x) {
            float gate = to_float_impl<scalar_t>(in_ptr[2*i]);
            float up   = to_float_impl<scalar_t>(in_ptr[2*i+1]);
            float g = fminf(gate, limit);
            float u = fmaxf(fminf(up, limit), -limit);
            out_ptr[i] = from_float_impl<scalar_t>((u + 1.0f) * g / (1.0f + expf(-g * alpha)));
        }
    } else {
        #pragma unroll 1
        for (int i = threadIdx.x; i < d; i += blockDim.x) {
            float gate = to_float_impl<scalar_t>(in_ptr[2*i]);
            float up   = to_float_impl<scalar_t>(in_ptr[2*i+1]);
            float g = fminf(gate, limit);
            float u = fmaxf(fminf(up, limit), -limit);
            out_ptr[i] = from_float_impl<scalar_t>((u + 1.0f) * g / (1.0f + expf(-g * alpha)));
        }
    }
}

template <typename scalar_t>
static void launch_swigluoai_and_mul(
    scalar_t* out, const scalar_t* input,
    int64_t num_tokens, int d, float alpha, float limit, cudaStream_t stream) {
    int cc_major = at::cuda::getCurrentDeviceProperties()->major;
    bool use_256b = (cc_major >= 9);
    int vec_bytes = use_256b ? 32 : 16;
    int pairs = vec_bytes / (2 * sizeof(scalar_t));
    bool aligned = (((uintptr_t)input) % vec_bytes == 0) && (((uintptr_t)out) % (vec_bytes/2) == 0);
    bool use_vec = (d >= pairs) && (d % pairs == 0) && aligned;
    int block_size;
    if (use_vec) {
        int num_vecs = d / pairs;
        block_size = num_vecs;
        if (block_size > 256) block_size = 256;
        block_size = (block_size + 31) & ~31;
    } else {
        block_size = d;
        if (block_size > 512) block_size = 512;
        block_size = (block_size + 31) & ~31;
    }
    dim3 grid((int)num_tokens);
    if (use_vec && use_256b) {
        swigluoai_and_mul_kernel<scalar_t, true, true>
            <<<grid, block_size, 0, stream>>>(out, input, num_tokens, d, alpha, limit);
    } else if (use_vec) {
        swigluoai_and_mul_kernel<scalar_t, true, false>
            <<<grid, block_size, 0, stream>>>(out, input, num_tokens, d, alpha, limit);
    } else {
        swigluoai_and_mul_kernel<scalar_t, false, false>
            <<<grid, block_size, 0, stream>>>(out, input, num_tokens, d, alpha, limit);
    }
}
