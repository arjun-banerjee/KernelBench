import torch
import torch.nn as nn

class ModelNew(nn.Module):
    def __init__(self, K, N, alpha, beta):
        super().__init__()
        self.K = K
        self.N = N
        self.alpha = alpha
        self.beta = beta
        self.weight = nn.Parameter(torch.randn(K, N, dtype=torch.float32).cuda())
    def forward(self, x):
        return torch.matmul(x, self.weight)
