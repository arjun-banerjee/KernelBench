#include <torch/extension.h>

#include <cuda_fp16.h>
#include <cuda_runtime.h>

#define CHECK_CUDA(x) TORCH_CHECK(x.is_cuda(), #x " must be CUDA")

__device__ __forceinline__ float warp_sum(float v) {
    #pragma unroll
    for (int o = 16; o > 0; o >>= 1) v += __shfl_down_sync(0xffffffff, v, o);
    return v;
}

__device__ __forceinline__ float warp_max(float v) {
    #pragma unroll
    for (int o = 16; o > 0; o >>= 1) v = fmaxf(v, __shfl_down_sync(0xffffffff, v, o));
    return v;
}

// Fixed-shape Hopper kernel for causal grouped-query attention:
// q: [4, 32, 1024, 128], k/v: [4, 8, 1024, 128], fp16 contiguous.
// A CTA has 8 warps.  It computes two sequence rows for all four query heads
// sharing a KV head, improving L1/L2 reuse of the same K/V cache lines.
__global__ __launch_bounds__(256, 2)
void gqa_attn_gqa4x2_h2(const half* __restrict__ q,
                        const half* __restrict__ k,
                        const half* __restrict__ v,
                        half* __restrict__ out) {
    constexpr int H = 32;
    constexpr int KVH = 8;
    constexpr int S = 1024;
    constexpr int D = 128;
    constexpr int R = 2;

    extern __shared__ float sm[];  // 8 * S logits/probabilities.

    const int lane = threadIdx.x & 31;
    const int wid  = threadIdx.x >> 5;
    const int qin  = wid & 1;      // two neighboring query positions
    const int hg   = wid >> 1;     // four query heads per KV head

    const int tile = blockIdx.x;
    const int kvh  = blockIdx.y;
    const int b    = blockIdx.z;
    const int qpos = tile * R + qin;
    const int h    = kvh * 4 + hg;
    const int len  = qpos + 1;

    float* __restrict__ logits = sm + wid * S;

    const half* qptr = q + (((b * H + h) * S + qpos) * D);
    const half2* q2 = reinterpret_cast<const half2*>(qptr);
    const half2 qh0 = q2[lane];
    const half2 qh1 = q2[lane + 32];
    const float2 qf0 = __half22float2(qh0);
    const float2 qf1 = __half22float2(qh1);

    constexpr float scale = 0.08838834764831845f;  // 1 / sqrt(128)

    for (int kp = 0; kp < len; ++kp) {
        const half2* kk = reinterpret_cast<const half2*>(kbase + kp * D);
        const float2 kf0 = __half22float2(kk[lane]);
        const float2 kf1 = __half22float2(kk[lane + 32]);

        if (lane == 0) logits[kp] = s * scale;

    m = __shfl_sync(0xffffffff, m, 0);

        logits[i] = e;

        a0.y += w * vf0.y;
        a1.x += w * vf1.x;
        a1.y += w * vf1.y;

}
