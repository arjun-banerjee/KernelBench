cuda_src = r'''
#include <torch/extension.h>
#include <cuda_fp16.h>

// Problem-specialized FP16 RoPE for:
// q: [512, 32, 128], k: [512, 8, 128], cos/sin: [512, 64].
// In-place update and return q,k.  One CTA per token keeps index math simple on
// Hopper; half2 vectorization maps the 64 rotary dimensions to 16 lanes/head.
__global__ __launch_bounds__(256, 2)
void rope_token_kernel(half* __restrict__ q,
                       half* __restrict__ k,
                       const half* __restrict__ cos,
                       const half* __restrict__ sin) {
    const int tok = blockIdx.x;
    const half* __restrict__ cbase = cos + tok * 64;
    const half* __restrict__ sbase = sin + tok * 64;

    // 40 heads (32 q + 8 k) * 16 half2 pairs/head = 640 work items/token.
    #pragma unroll
    for (int i = threadIdx.x; i < 640; i += 256) {
        const int head = i >> 4;
        const int off = (i & 15) << 1;

        half* __restrict__ arr = (head < 32)
            ? (q + ((tok * 32 + head) << 7))
            : (k + ((tok * 8 + (head - 32)) << 7));

        const half2 x  = *reinterpret_cast<const half2*>(arr + off);
        const half2 y  = *reinterpret_cast<const half2*>(arr + 32 + off);
        const half2 c0 = *reinterpret_cast<const half2*>(cbase + off);
        const half2 s0 = *reinterpret_cast<const half2*>(sbase + off);
        const half2 c1 = *reinterpret_cast<const half2*>(cbase + 32 + off);
        const half2 s1 = *reinterpret_cast<const half2*>(sbase + 32 + off);

        *reinterpret_cast<half2*>(arr + off) = __hsub2(__hmul2(x, c0), __hmul2(y, s0));
        *reinterpret_cast<half2*>(arr + 32 + off) = __hadd2(__hmul2(y, c1), __hmul2(x, s1));
    }
}

std::vector<torch::Tensor> rope_h100(torch::Tensor q, torch::Tensor k,
                                     torch::Tensor cos, torch::Tensor sin,
                                     int64_t rot_dim) {
    (void)rot_dim;
    rope_token_kernel<<<512, 256>>>(
        reinterpret_cast<half*>(q.data_ptr<at::Half>()),
        reinterpret_cast<half*>(k.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(cos.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(sin.data_ptr<at::Half>()));
    return {q, k};
}
'''
