__inline__ __device__ float warp_sum(float x) {
    x += __shfl_down_sync(0xffffffff, x, 16);
    x += __shfl_down_sync(0xffffffff, x, 8);
    x += __shfl_down_sync(0xffffffff, x, 4);
    x += __shfl_down_sync(0xffffffff, x, 2);

__inline__ __device__ float block_sum(float v) {
    __shared__ float s[256];
    int tid = threadIdx.x;
    s[tid] = v;
    __syncthreads();
    #pragma unroll

        if (tid < o) s[tid] += s[tid + o];
        __syncthreads();
    }

}

    const half2* q2 = reinterpret_cast<const half2*>(qptr);

        half2 pa = __hmul2(qa, k2[lane]);
        half2 pb = __hmul2(qb, k2[lane + 32]);
