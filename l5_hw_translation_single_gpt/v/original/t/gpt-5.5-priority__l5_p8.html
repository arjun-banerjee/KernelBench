    def __init__(self, k: int, n: int):
        super().__init__()
        self.register_buffer(
            "weight", torch.randint(-128, 127, (n, k), dtype=torch.int8, device="cuda")
        )
        self.register_buffer(
            "weight_scale", torch.ones(n, dtype=torch.float32, device="cuda") / 127.0

        # forward; keeping a persistent=False packed buffer preserves state_dict

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return _w8a8_gemm.w8a8_gemm_cuda(x.contiguous(), self._wt_float_t)
