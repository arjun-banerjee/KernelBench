import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r"""
#include <torch/extension.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <stdint.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

// One warp computes 4 output columns x 16 rows.  Each lane owns one column
// (lane/8) and two rows (lane%8, lane%8+8).  The input activation x is shared
// across the four 8-lane groups with warp shuffles, while int4 weights/scales
// are broadcast inside each 8-lane group.
__global__ __launch_bounds__(128, 4)
void int4_gemm_m16_kernel(const half* __restrict__ x,
                          const uint8_t* __restrict__ qweight,
                          const half* __restrict__ scales,
                          const half* __restrict__ zeros,
                          half* __restrict__ out,
                          int M, int K, int N) {
    const int lane = threadIdx.x & 31;
    const int warp = threadIdx.x >> 5;       // 0..3
    const int sub  = lane >> 3;              // 0..3, output column within warp
    const int r    = lane & 7;               // row pair id

    const int n = blockIdx.x * 16 + warp * 4 + sub;
    const int m0 = r;
    const int m1 = r + 8;

    float acc0 = 0.0f;
    float acc1 = 0.0f;

    float sc = 0.0f;
    float zp = 0.0f;

    // full warp is active for all valid problem sizes used here
    const unsigned mask = 0xffffffffu;

    #pragma unroll 1
    for (int k = 0; k < K; ++k) {
        if ((k & 127) == 0) {
            const int g = k >> 7;
            float sc_lane = 0.0f;
            float zp_lane = 0.0f;
            if (r == 0 && n < N) {
                sc_lane = __half2float(scales[g * N + n]);
                zp_lane = __half2float(zeros[g * N + n]);
            }
            sc = __shfl_sync(mask, sc_lane, sub * 8, 8);
            zp = __shfl_sync(mask, zp_lane, sub * 8, 8);
        }

        // Load x only in the first 8-lane group and broadcast to the other
        // three column groups.  This cuts activation loads by 4x.
        float x0_lane = 0.0f;
        float x1_lane = 0.0f;
        if (sub == 0) {
            if (m0 < M) x0_lane = __half2float(x[m0 * K + k]);
            if (m1 < M) x1_lane = __half2float(x[m1 * K + k]);
        }
        const float xv0 = __shfl_sync(mask, x0_lane, r);
        const float xv1 = __shfl_sync(mask, x1_lane, r);

        float w_lane = 0.0f;
        if (r == 0 && n < N) {
            const uint8_t packed = qweight[k * (N >> 1) + (n >> 1)];
            const int q = (n & 1) ? (int)(packed >> 4) : (int)(packed & 0x0f);
            w_lane = ((float)q - zp) * sc;
        }
        const float w = __shfl_sync(mask, w_lane, sub * 8, 8);
        acc0 = fmaf(xv0, w, acc0);
        acc1 = fmaf(xv1, w, acc1);
    }

    if (n < N) {
        if (m0 < M) out[m0 * N + n] = __float2half_rn(acc0);
        if (m1 < M) out[m1 * N + n] = __float2half_rn(acc1);
    }
}

torch::Tensor int4_gemm_cuda(torch::Tensor x,
                             torch::Tensor qweight,
                             torch::Tensor scales,
                             torch::Tensor zeros) {
    const int M = (int)x.size(0);
    const int K = (int)x.size(1);
    const int N = (int)scales.size(1);
    auto out = torch::empty({M, N}, x.options());

    dim3 block(128);
    dim3 grid((N + 15) / 16);
    int4_gemm_m16_kernel<<<grid, block>>>(
        reinterpret_cast<const half*>(x.data_ptr<at::Half>()),
        reinterpret_cast<const uint8_t*>(qweight.data_ptr<uint8_t>()),
        reinterpret_cast<const half*>(scales.data_ptr<at::Half>()),
        reinterpret_cast<const half*>(zeros.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()),
        M, K, N);
    return out;
}
"""

class ModelNew(nn.Module):
    def __init__(self, k: int, n: int, group_size: int = 128):
        super().__init__()
        num_groups = k // group_size
        self.group_size = group_size
        self.register_buffer(
            "qweight",
            torch.randint(0, 256, (k, n // 2), dtype=torch.uint8).cuda(),
        )
        self.register_buffer(
            "scales",
            torch.randn(num_groups, n, dtype=torch.float16).cuda() * 0.01,
        )
        self.register_buffer(
            "zeros",
            torch.zeros(num_groups, n, dtype=torch.float16).cuda() + 8.0,
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return _int4_ext.int4_gemm_cuda(x, self.qweight, self.scales, self.zeros)
