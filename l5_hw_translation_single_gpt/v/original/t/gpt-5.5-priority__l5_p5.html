import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline


_rope_h100_cpp = r"""
#include <torch/extension.h>
#include <vector>

std::vector<torch::Tensor> rope_forward(torch::Tensor query,
                                        torch::Tensor key,
                                        torch::Tensor cos,
                                        torch::Tensor sin);
"""

_rope_h100_cuda = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <vector>

// Specialized for the benchmark layout: [T, H, 128], ROT_DIM=64, Neox-style.
// One CTA owns one token.  It stages cos/sin in shared memory once and applies
// the same values to all Q and K heads, reducing redundant HBM reads on Hopper.
__global__ __launch_bounds__(256, 4)
void rope_h100_kernel(const half* __restrict__ query,
                      const half* __restrict__ key,
                      const half* __restrict__ cos,
                      const half* __restrict__ sin,
                      half* __restrict__ query_out,
                      half* __restrict__ key_out,
                      const int num_heads,
                      const int num_kv_heads) {
    const int token = blockIdx.x;

    extern __shared__ float smem[];
    float* __restrict__ c = smem;       // 64 floats
    float* __restrict__ s = smem + 64;  // 64 floats

    if (threadIdx.x < 64) {
        const int off = token * 64 + threadIdx.x;
        c[threadIdx.x] = __half2float(cos[off]);
        s[threadIdx.x] = __half2float(sin[off]);
    }
    __syncthreads();

    // Per head: 32 pair-rotation tasks + 32 half2 copy tasks for dims [64,128).
    const int tasks_per_head = 64;
    const int total_heads = num_heads + num_kv_heads;
    const int total_tasks = total_heads * tasks_per_head;

    for (int task = threadIdx.x; task < total_tasks; task += blockDim.x) {
        const int logical_head = task >> 6;   // /64
        const int u = task & 63;              // %64
        const bool is_q = logical_head < num_heads;
        const int h = is_q ? logical_head : (logical_head - num_heads);

        const half* __restrict__ in;
        half* __restrict__ out;
        if (is_q) {
            const int base = (token * num_heads + h) * 128;
            in = query + base;
            out = query_out + base;
        } else {
            const int base = (token * num_kv_heads + h) * 128;
            in = key + base;
            out = key_out + base;
        }

        if (u < 32) {
            // Neox rotate_half over the first rot_dim=64 values:
            // out[i]    = x*cos[i]    - y*sin[i]
            // out[i+32] = y*cos[i+32] + x*sin[i+32]
            const int i = u;
            const float x = __half2float(in[i]);
            const float y = __half2float(in[i + 32]);
            const float o0 = fmaf(-y, s[i],      x * c[i]);
            const float o1 = fmaf( x, s[i + 32], y * c[i + 32]);
            out[i]      = __float2half_rn(o0);
            out[i + 32] = __float2half_rn(o1);
        } else {
            // Copy the non-rotary tail, two fp16 values per task.
            const int d = 64 + ((u - 32) << 1);
            const half2 v = *reinterpret_cast<const half2*>(in + d);
            *reinterpret_cast<half2*>(out + d) = v;
        }
    }
}

std::vector<torch::Tensor> rope_forward(torch::Tensor query,
                                        torch::Tensor key,
                                        torch::Tensor cos,
                                        torch::Tensor sin) {
    auto query_out = torch::empty_like(query);
    auto key_out = torch::empty_like(key);

    const int num_tokens = static_cast<int>(query.size(0));
    const int num_heads = static_cast<int>(query.size(1));
    const int num_kv_heads = static_cast<int>(key.size(1));

    const half* q_ptr = reinterpret_cast<const half*>(query.data_ptr<at::Half>());
    const half* k_ptr = reinterpret_cast<const half*>(key.data_ptr<at::Half>());
    const half* c_ptr = reinterpret_cast<const half*>(cos.data_ptr<at::Half>());
    const half* s_ptr = reinterpret_cast<const half*>(sin.data_ptr<at::Half>());
    half* qo_ptr = reinterpret_cast<half*>(query_out.data_ptr<at::Half>());
    half* ko_ptr = reinterpret_cast<half*>(key_out.data_ptr<at::Half>());

    constexpr int threads = 256;
    constexpr int shmem = 128 * sizeof(float);
    rope_h100_kernel<<<num_tokens, threads, shmem, at::cuda::getCurrentCUDAStream()>>>(
        q_ptr, k_ptr, c_ptr, s_ptr, qo_ptr, ko_ptr, num_heads, num_kv_heads);

    return {query_out, key_out};
}
"""

_rope_h100 = load_inline(
    name="rope_h100_token_shared_ext",
    cpp_sources=_rope_h100_cpp,
    cuda_sources=_rope_h100_cuda,
    functions=["rope_forward"],
    verbose=False,
    extra_cflags=["-O3"],
    extra_cuda_cflags=["-O3", "--use_fast_math", "-Xptxas", "-O3"],
)


class ModelNew(nn.Module):
    def __init__(self, rot_dim: int):
        super().__init__()
        self.rot_dim = rot_dim
        self._rope = _rope_h100

    def forward(self, query: torch.Tensor, key: torch.Tensor,
                cos: torch.Tensor, sin: torch.Tensor):
        return self._rope.rope_forward(query, key, cos, sin)
