import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cuda_src = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>

// One CTA per token, vectorized as half2.  For the benchmark shape
// (512 tokens, D=2048), 256 threads gives four half2 results per thread.
__global__ __launch_bounds__(256, 2)
void swiglu_half2_kernel(const half* __restrict__ x,
                         half* __restrict__ out,
                         const int d) {
    const int token = blockIdx.x;
    const half* __restrict__ gate = x + ((size_t)token * 2u * (size_t)d);
    const half* __restrict__ up   = gate + d;
    half* __restrict__ dst = out + ((size_t)token * (size_t)d);

    const half2* __restrict__ g2p = reinterpret_cast<const half2*>(gate);
    const half2* __restrict__ u2p = reinterpret_cast<const half2*>(up);
    half2* __restrict__ o2p = reinterpret_cast<half2*>(dst);

    const int n2 = d >> 1;
    #pragma unroll 4
    for (int i = threadIdx.x; i < n2; i += blockDim.x) {
        half2 gh = __ldg(g2p + i);
        half2 uh = __ldg(u2p + i);

        float2 gf = __half22float2(gh);
        float2 uf = __half22float2(uh);

        // SiLU(g) * up = g / (1 + exp(-g)) * up.  __expf is more than
        // accurate enough for fp16 output tolerance and maps to fast SFU ops.
        float sx = gf.x / (1.0f + __expf(-gf.x));
        float sy = gf.y / (1.0f + __expf(-gf.y));
        float2 rf;
        rf.x = sx * uf.x;
        rf.y = sy * uf.y;
        o2p[i] = __float22half2_rn(rf);
    }

    // d is 2048 in the target model, but keep an odd-d cleanup for robustness.
    if ((d & 1) && threadIdx.x == 0) {
        int idx = d - 1;
        float g = __half2float(gate[idx]);
        float u = __half2float(up[idx]);
        dst[idx] = __float2half_rn((g / (1.0f + __expf(-g))) * u);
    }
}

torch::Tensor swiglu_cuda(torch::Tensor x) {
    const int64_t last = x.size(-1);
    const int d = (int)(last / 2);
    const int64_t tokens = x.numel() / last;

    auto out_sizes = x.sizes().vec();
    out_sizes.back() = d;
    auto out = torch::empty(out_sizes, x.options());

    const int threads = 256;
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    swiglu_half2_kernel<<<(unsigned)tokens, threads, 0, stream>>>(
        reinterpret_cast<const half*>(x.data_ptr<at::Half>()),
        reinterpret_cast<half*>(out.data_ptr<at::Half>()),
        d);
    return out;
}
"""
