            const long k_base = (((long)b * HKV + hkv) * S + j) * HD;
            float p = expf(scores[j] - L);
            float ds = p * (dps[j] - Dval);
            acc = fmaf(ds, __bfloat162float(k[k_base + d]), acc);
        }
        dq[q_base + d] = __float2bfloat16(acc * scale);
    }
}

__global__ __launch_bounds__(128, 2)
void dkdv_kernel(const __nv_bfloat16* __restrict__ q,
                 const __nv_bfloat16* __restrict__ k,
                 const __nv_bfloat16* __restrict__ v,
                 const __nv_bfloat16* __restrict__ dout,
                 const float* __restrict__ lse,
                 const float* __restrict__ delta,
                 __nv_bfloat16* __restrict__ dk,
                 __nv_bfloat16* __restrict__ dv,
                 int B, int H, int HKV, int S, int groups,
                 float scale) {
    __shared__ float red[8];
    int idx = blockIdx.x;
    int j = idx % S;
    int hkv = (idx / S) % HKV;
    int b = idx / (S * HKV);
    int tid = threadIdx.x;

    for (int d = tid; d < HD; d += blockDim.x) {
        float dk_acc = 0.0f;
        float dv_acc = 0.0f;
        const long kv_base = (((long)b * HKV + hkv) * S + j) * HD;
        float kjd = __bfloat162float(k[kv_base + d]);
        float vjd = __bfloat162float(v[kv_base + d]);
        (void)kjd; (void)vjd;

        for (int g = 0; g < groups; ++g) {
            int h = hkv * groups + g;
            for (int i = j; i < S; ++i) {
                const long q_base  = (((long)b * H + h) * S + i) * HD;
                const long do_base = q_base;
                float part_qk = 0.0f;
                float part_dp = 0.0f;
                for (int dd = threadIdx.x; dd < HD; dd += blockDim.x) {
                    float qv = __bfloat162float(q[q_base + dd]);
                    float kv = __bfloat162float(k[kv_base + dd]);
                    float dov = __bfloat162float(dout[do_base + dd]);
                    float vv = __bfloat162float(v[kv_base + dd]);
                    part_qk = fmaf(qv, kv, part_qk);
                    part_dp = fmaf(dov, vv, part_dp);
                }
                float score = block_reduce_sum(part_qk, red) * scale;
                float dp = block_reduce_sum(part_dp, red);
                float L = lse[((long)b * H + h) * S + i];
                float Di = delta[((long)b * H + h) * S + i];
                float p = expf(score - L);
                float ds = p * (dp - Di);
                float qid = __bfloat162float(q[q_base + d]);
                float dod = __bfloat162float(dout[do_base + d]);
                dk_acc = fmaf(ds, qid, dk_acc);
                dv_acc = fmaf(p, dod, dv_acc);
            }
        }
        dk[kv_base + d] = __float2bfloat16(dk_acc * scale);
        dv[kv_base + d] = __float2bfloat16(dv_acc);
    }
}

std::vector<torch::Tensor> flash_bwd_cuda(torch::Tensor q, torch::Tensor k, torch::Tensor v, torch::Tensor dout) {
    CHECK_CUDA(q); CHECK_CUDA(k); CHECK_CUDA(v); CHECK_CUDA(dout);
    CHECK_CONTIGUOUS(q); CHECK_CONTIGUOUS(k); CHECK_CONTIGUOUS(v); CHECK_CONTIGUOUS(dout);
    TORCH_CHECK(q.scalar_type() == at::kBFloat16, "q must be bf16");
    TORCH_CHECK(k.scalar_type() == at::kBFloat16, "k must be bf16");
    int B = (int)q.size(0);
    int H = (int)q.size(1);
    int S = (int)q.size(2);
    int D = (int)q.size(3);
    int HKV = (int)k.size(1);
    TORCH_CHECK(D == HD, "only head_dim=128 supported");
    TORCH_CHECK(S <= 2048, "sequence length too large for this kernel");
    int groups = H / HKV;

    auto dq = torch::empty_like(q);
    auto dk = torch::empty_like(k);
    auto dv = torch::empty_like(v);
    auto opts_f = torch::TensorOptions().device(q.device()).dtype(torch::kFloat32);
    auto lse = torch::empty({B, H, S}, opts_f);
    auto delta = torch::empty({B, H, S}, opts_f);

    const float scale = rsqrtf((float)HD);
    cudaStream_t stream = at::cuda::getCurrentCUDAStream();
    dim3 block(128);
    dim3 grid_rows(B * H * S);
    size_t shmem = (size_t)(2 * S + 32) * sizeof(float);
    row_lse_d_dq_kernel<<<grid_rows, block, shmem, stream>>>(
        reinterpret_cast<const __nv_bfloat16*>(q.data_ptr<at::BFloat16>()),
        reinterpret_cast<const __nv_bfloat16*>(k.data_ptr<at::BFloat16>()),
        reinterpret_cast<const __nv_bfloat16*>(v.data_ptr<at::BFloat16>()),
        reinterpret_cast<const __nv_bfloat16*>(dout.data_ptr<at::BFloat16>()),
        reinterpret_cast<__nv_bfloat16*>(dq.data_ptr<at::BFloat16>()),
        lse.data_ptr<float>(), delta.data_ptr<float>(), B, H, HKV, S, groups, scale);

    dim3 grid_kv(B * HKV * S);
    dkdv_kernel<<<grid_kv, block, 0, stream>>>(
        reinterpret_cast<const __nv_bfloat16*>(q.data_ptr<at::BFloat16>()),
        reinterpret_cast<const __nv_bfloat16*>(k.data_ptr<at::BFloat16>()),
        reinterpret_cast<const __nv_bfloat16*>(v.data_ptr<at::BFloat16>()),
        reinterpret_cast<const __nv_bfloat16*>(dout.data_ptr<at::BFloat16>()),
        lse.data_ptr<float>(), delta.data_ptr<float>(),
        reinterpret_cast<__nv_bfloat16*>(dk.data_ptr<at::BFloat16>()),
        reinterpret_cast<__nv_bfloat16*>(dv.data_ptr<at::BFloat16>()),
        B, H, HKV, S, groups, scale);

    return {dq, dk, dv};
}
