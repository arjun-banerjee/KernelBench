_cuda_src = r"""
#include <torch/extension.h>
#include <cuda.h>
#include <cuda_runtime.h>
#include <cuda_fp16.h>
#include <math_constants.h>

#ifndef WARP_SIZE
#define WARP_SIZE 32
#endif

// Fixed-shape Hopper-oriented FP16 causal GQA attention for:
// q: [4, 32, 512, 128], k/v: [4, 8, 512, 128]
// One CTA computes one output row.  Logits are kept in shared memory so QK is
// evaluated only once; V accumulation is coalesced across the head dimension.
__global__ __launch_bounds__(256, 2)
void gqa_causal_attn_512x128_kernel(const half* __restrict__ q,
                                    const half* __restrict__ k,
                                    const half* __restrict__ v,
                                    half* __restrict__ out) {
    constexpr int BATCH = 4;
    constexpr int NUM_HEADS = 32;
    constexpr int NUM_KV_HEADS = 8;
    constexpr int GROUPS = NUM_HEADS / NUM_KV_HEADS;
    constexpr int SEQ_LEN = 512;
    constexpr int HEAD_SIZE = 128;
    constexpr int THREADS = 256;
    constexpr float SCALE = 0.08838834764831845f; // 1/sqrt(128)

    const int t = blockIdx.x;    // query position
    const int h = blockIdx.y;    // query head
    const int b = blockIdx.z;    // batch
    const int tid = threadIdx.x;
    const int kvh = h / GROUPS;

    const half* __restrict__ qrow = q + (((b * NUM_HEADS + h) * SEQ_LEN + t) * HEAD_SIZE);
    const half* __restrict__ kbase = k + (((b * NUM_KV_HEADS + kvh) * SEQ_LEN) * HEAD_SIZE);
    const half* __restrict__ vbase = v + (((b * NUM_KV_HEADS + kvh) * SEQ_LEN) * HEAD_SIZE);
    half* __restrict__ orow = out + (((b * NUM_HEADS + h) * SEQ_LEN + t) * HEAD_SIZE);

    __shared__ half2 q_sh[HEAD_SIZE / 2];
    __shared__ float logits[SEQ_LEN];
    __shared__ float red[THREADS];

    if (tid < HEAD_SIZE / 2) {
        q_sh[tid] = reinterpret_cast<const half2*>(qrow)[tid];
    }
    __syncthreads();

    float local_max = -CUDART_INF_F;

    // Compute scaled QK logits for valid causal positions j <= t.
    for (int j = tid; j <= t; j += THREADS) {
        const half2* __restrict__ krow2 = reinterpret_cast<const half2*>(kbase + j * HEAD_SIZE);
        float dot = 0.0f;
#pragma unroll
        for (int i = 0; i < HEAD_SIZE / 2; ++i) {
            const float2 qf = __half22float2(q_sh[i]);
            const float2 kf = __half22float2(krow2[i]);
            dot = fmaf(qf.x, kf.x, dot);
            dot = fmaf(qf.y, kf.y, dot);
        }
        const float val = dot * SCALE;
        logits[j] = val;
        local_max = fmaxf(local_max, val);
    }

    red[tid] = local_max;
    __syncthreads();
#pragma unroll
    for (int stride = THREADS / 2; stride > 0; stride >>= 1) {
        if (tid < stride) red[tid] = fmaxf(red[tid], red[tid + stride]);
        __syncthreads();
    }
    const float row_max = red[0];

    float local_sum = 0.0f;
    for (int j = tid; j <= t; j += THREADS) {
        const float w = __expf(logits[j] - row_max);
        logits[j] = w;
        local_sum += w;
    }

    red[tid] = local_sum;
    __syncthreads();
#pragma unroll
    for (int stride = THREADS / 2; stride > 0; stride >>= 1) {
        if (tid < stride) red[tid] += red[tid + stride];
        __syncthreads();
    }
    const float inv_sum = 1.0f / red[0];
    __syncthreads();

    // Accumulate O = softmax(QK) V.  Threads 0..127 map to the contiguous
    // feature dimension, yielding coalesced V reads and output writes.
    if (tid < HEAD_SIZE) {

#pragma unroll 1

            acc = fmaf(logits[j], __half2float(vbase[j * HEAD_SIZE + tid]), acc);
        }
        orow[tid] = __float2half_rn(acc * inv_sum);
    }
}
